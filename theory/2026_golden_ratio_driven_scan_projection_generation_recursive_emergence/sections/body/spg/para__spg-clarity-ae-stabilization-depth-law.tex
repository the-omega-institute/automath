\paragraph{几乎处处稳定、风险阈值与算术读出}
边界柱厚度与 Gödel 双对数标度给出的并不只是清晰度指数的存在性。把命题 \ref{prop:spg-clarity-bayes-optimality} 的 Bayes 最优前缀判决与上述指数律并置之后，还可把“误差剖面可和”“几乎处处最终定解”“目标风险所需扫描深度”以及“边界柱 Gödel 码的双对数读出”压缩为同一条定量链。

\begin{theorem}[可和扫描误差下的 Bayes 前缀判决稳定性]\label{thm:spg-bayes-finite-mistakes-summable}
沿用命题 \ref{prop:spg-clarity-bayes-optimality} 的记号。若
\[
\sum_{m=1}^{\infty}\varepsilon_m(\mathsf P;\mu)<\infty,
\]
则对 \(\mu\)-几乎处处的 \(\omega\in X\) 都有
\[
\sum_{m=1}^{\infty}\ind_{\mathsf P\triangle \mathsf P_m^\star}(\omega)<\infty.
\]
因而 \(\ind_{\mathsf P_m^\star}(\omega)\to \ind_{\mathsf P}(\omega)\) 对 \(\mu\)-几乎处处的 \(\omega\) 成立；等价地，存在随机但有限的 \(m_0(\omega)\)，使得对所有 \(m\ge m_0(\omega)\) 都有
\[
\ind_{\mathsf P_m^\star}(\omega)=\ind_{\mathsf P}(\omega).
\]
\end{theorem}
\begin{proof}
令
\[
E_m:=\mathsf P\triangle \mathsf P_m^\star.
\]
由命题 \ref{prop:spg-clarity-bayes-optimality} 可知
\[
\mu(E_m)=\mu(\mathsf P\triangle \mathsf P_m^\star)=\varepsilon_m(\mathsf P;\mu).
\]
故 \(\sum_{m\ge 1}\mu(E_m)<\infty\)。由 Borel--Cantelli 引理，
\[
\mu\bigl(E_m\ \text{无穷多次发生}\bigr)=0.
\]
于是对 \(\mu\)-几乎处处的 \(\omega\)，仅有有限多个 \(m\) 满足 \(\omega\in E_m\)，亦即
\[
\sum_{m=1}^{\infty}\ind_{\mathsf P\triangle \mathsf P_m^\star}(\omega)<\infty.
\]
这正说明 \(\ind_{\mathsf P_m^\star}(\omega)\) 最终稳定到 \(\ind_{\mathsf P}(\omega)\)。证毕。
\end{proof}

\begin{corollary}[边界维数严格小于一时的几乎处处最终定解]\label{cor:spg-boundary-dimension-ae-stabilization}
设定理 \ref{thm:spg-clarity-volume-boundary-scaling} 的厚度假设成立，并且对某个 \(d\in[0,1)\) 有
\[
N_m(\partial\mathsf P)=\lambda_{\mathrm{cyl}}^{dm+o(m)}.
\]
则
\[
\varepsilon_m(\mathsf P;\mu)=\lambda_{\mathrm{cyl}}^{-(1-d)m+o(m)}.
\]
特别地，
\[
\sum_{m=1}^{\infty}\varepsilon_m(\mathsf P;\mu)<\infty,
\]
从而 \(\ind_{\mathsf P_m^\star}\to \ind_{\mathsf P}\) 在 \(\mu\)-几乎处处意义下最终稳定。
\end{corollary}
\begin{proof}
由定理 \ref{thm:spg-clarity-volume-boundary-scaling} 的双边估计，
\[
\theta c_-\,\lambda_{\mathrm{cyl}}^{-m}N_m(\partial\mathsf P)
\le
\varepsilon_m(\mathsf P;\mu)
\le
\frac{c_+}{2}\lambda_{\mathrm{cyl}}^{-m}N_m(\partial\mathsf P).
\]
代入 \(N_m(\partial\mathsf P)=\lambda_{\mathrm{cyl}}^{dm+o(m)}\) 即得
\[
\varepsilon_m(\mathsf P;\mu)=\lambda_{\mathrm{cyl}}^{-(1-d)m+o(m)}.
\]
由于 \(d<1\)，右端为指数衰减，故级数收敛。再由定理 \ref{thm:spg-bayes-finite-mistakes-summable} 即得几乎处处最终定解。证毕。
\end{proof}

\begin{theorem}[风险阈值的扫描深度渐近律]\label{thm:spg-clarity-risk-threshold-depth-law}
在推论 \ref{cor:spg-boundary-dimension-ae-stabilization} 的假设下，对 \(\delta\in(0,1)\) 定义达到风险阈值所需的最小扫描深度
\[
m_\delta(\mathsf P):=\min\{m\ge 1:\ \varepsilon_m(\mathsf P;\mu)\le \delta\}.
\]
则当 \(\delta\downarrow 0\) 时，
\[
m_\delta(\mathsf P)
=
\frac{\log(1/\delta)}{(1-d)\log\lambda_{\mathrm{cyl}}}
+o\!\bigl(\log(1/\delta)\bigr).
\]
若进一步有更强计数式
\[
N_m(\partial\mathsf P)=\lambda_{\mathrm{cyl}}^{dm+O(1)},
\]
则可提升为
\[
m_\delta(\mathsf P)
=
\frac{\log(1/\delta)}{(1-d)\log\lambda_{\mathrm{cyl}}}
+O(1).
\]
\leanverified{paper\_spg\_clarity\_risk\_threshold\_depth\_law}
\end{theorem}
\begin{proof}
由推论 \ref{cor:spg-boundary-dimension-ae-stabilization}，存在 \(u_m=o(m)\) 使
\[
\varepsilon_m(\mathsf P;\mu)
=
\lambda_{\mathrm{cyl}}^{-(1-d)m+u_m}.
\]
于是
\[
\log\frac{1}{\varepsilon_m(\mathsf P;\mu)}
=
(1-d)m\log\lambda_{\mathrm{cyl}}-u_m\log\lambda_{\mathrm{cyl}}.
\]
当 \(m=m_\delta(\mathsf P)\) 时，由 \(\varepsilon_{m_\delta}(\mathsf P;\mu)\le \delta<\varepsilon_{m_\delta-1}(\mathsf P;\mu)\) 与 \(u_m=o(m)\) 可解得
\[
m_\delta(\mathsf P)
=
\frac{\log(1/\delta)}{(1-d)\log\lambda_{\mathrm{cyl}}}
+o\!\bigl(\log(1/\delta)\bigr).
\]
若 \(N_m(\partial\mathsf P)=\lambda_{\mathrm{cyl}}^{dm+O(1)}\)，则同理有 \(\varepsilon_m(\mathsf P;\mu)=\lambda_{\mathrm{cyl}}^{-(1-d)m+O(1)}\)，反解后即得 \(O(1)\) 精度。证毕。
\end{proof}

\begin{corollary}[边界柱 Gödel 双对数维与清晰度指数的精确对应]\label{cor:spg-boundary-godel-clarity-exponent-equivalence}
在定理 \ref{thm:spg-clarity-volume-boundary-scaling} 的厚度假设下，若极限
\[
d_\partial(\mathsf P)
:=
\lim_{m\to\infty}
\frac{\log\log G_m^\partial(\mathsf P)}{m\log\lambda_{\mathrm{cyl}}}
\]
存在，则清晰度指数也存在，且
\[
\lim_{m\to\infty}
\frac{-\log \varepsilon_m(\mathsf P;\mu)}{m\log\lambda_{\mathrm{cyl}}}
=
1-d_\partial(\mathsf P).
\]
特别地，
\[
d_\partial(\mathsf P)<1
\quad\Longleftrightarrow\quad
\exists\,\gamma>0:\ 
\varepsilon_m(\mathsf P;\mu)\le e^{-\gamma m+o(m)},
\]
并且最优指数可取
\[
\gamma=(1-d_\partial(\mathsf P))\log\lambda_{\mathrm{cyl}}.
\]
\end{corollary}
\begin{proof}
定理 \ref{thm:spg-boundary-cylinder-godelization-clarity} 的证明已经给出
\[
\bigl|\log\log G_m^\partial(\mathsf P)-\log N_m(\partial\mathsf P)\bigr|
\le
A\log m+B
\]
对充分大 \(m\) 成立，其中 \(A,B\) 与 \(\mathsf P\) 无关。故
\[
\lim_{m\to\infty}
\frac{\log N_m(\partial\mathsf P)}{m\log\lambda_{\mathrm{cyl}}}
=
d_\partial(\mathsf P).
\]
于是
\[
N_m(\partial\mathsf P)
=
\lambda_{\mathrm{cyl}}^{d_\partial(\mathsf P)m+o(m)}.
\]
再由定理 \ref{thm:spg-clarity-volume-boundary-scaling}，
\[
\varepsilon_m(\mathsf P;\mu)
=
\lambda_{\mathrm{cyl}}^{-(1-d_\partial(\mathsf P))m+o(m)},
\]
从而
\[
\lim_{m\to\infty}
\frac{-\log \varepsilon_m(\mathsf P;\mu)}{m\log\lambda_{\mathrm{cyl}}}
=
1-d_\partial(\mathsf P).
\]
最后一组等价只是对该极限的重写。证毕。
\end{proof}

下述陈述把 Interleaving stability 算子用于 Bayes 前缀判决：深度偏移给出合法交错，缺陷集给出有限分辨率噪声预算。
\begin{theorem}[Bayes 前缀判决的可和交错缺陷稳定性]\label{distill:persistent_homology_persistence_modules_and_constructible_sheaf_persistence-stability_and_interleaving-spg-bayes-prefix-summable-interleaving-defect-transfer}
沿用命题 \ref{prop:spg-clarity-bayes-optimality} 的记号。设 \((\widehat{\mathsf P}_m)_{m\ge 1}\) 为另一族可测前缀判决。固定整数 \(h\ge 0\)。假设存在可测缺陷集 \(A_m^+,A_m^-\subseteq X\)，使得对所有 \(m\ge 1\) 都有
\[
\widehat{\mathsf P}_m\triangle \mathsf P_{m+h}^\star\subseteq A_m^+,
\qquad
\mathsf P_m^\star\triangle \widehat{\mathsf P}_{m+h}\subseteq A_m^-.
\]
记
\[
\widehat\varepsilon_m:=\mu(\widehat{\mathsf P}_m\triangle \mathsf P),
\qquad
\alpha_m^\pm:=\mu(A_m^\pm).
\]
则对每个 \(m\ge 1\)，
\[
\widehat\varepsilon_m\le \varepsilon_{m+h}(\mathsf P;\mu)+\alpha_m^+,
\qquad
\varepsilon_m(\mathsf P;\mu)\le \widehat\varepsilon_{m+h}+\alpha_m^-.
\]
并且有坏例子载体包含式
\[
\limsup_{m\to\infty}(\widehat{\mathsf P}_m\triangle\mathsf P)
\subseteq
\limsup_{m\to\infty} A_m^+
\cup
\limsup_{m\to\infty}(\mathsf P_{m+h}^\star\triangle\mathsf P).
\]
因此，若
\[
\sum_{m=1}^\infty \varepsilon_m(\mathsf P;\mu)<\infty,
\qquad
\sum_{m=1}^\infty \alpha_m^+<\infty,
\]
则
\[
\sum_{m=1}^\infty \ind_{\widehat{\mathsf P}_m\triangle\mathsf P}(\omega)<\infty
\]
对 \(\mu\)-几乎处处的 \(\omega\in X\) 成立。更一般地，若某可测集 \(G\subseteq X\) 满足
\[
\mu\Bigl(G\cap\limsup_{m\to\infty}(\mathsf P_{m+h}^\star\triangle\mathsf P)\Bigr)=0
\]
且
\[
\mu\Bigl(G\cap\limsup_{m\to\infty}(\widehat{\mathsf P}_m\triangle\mathsf P)\Bigr)>0,
\]
则
\[
\mu\Bigl(G\cap\limsup_{m\to\infty}A_m^+\Bigr)>0.
\]
特别地，在 Bayes 前缀本身仅犯有限次错误的区域上，任何正测度的无限错误坏例子都必须落入交错缺陷的 sticky limsup 载体。

若进一步满足推论 \ref{cor:spg-boundary-dimension-ae-stabilization} 的假设，令 \(\gamma:=1-d>0\)，并假设
\[
\alpha_m^+=\lambda_{\mathrm{cyl}}^{-\gamma m+o(m)},
\qquad
\alpha_m^-=o\bigl(\varepsilon_m(\mathsf P;\mu)\bigr),
\]
则
\[
\widehat\varepsilon_m=\lambda_{\mathrm{cyl}}^{-\gamma m+o(m)}.
\]
于是若
\[
\widehat m_\delta:=\min\{m\ge 1:\widehat\varepsilon_m\le\delta\},
\]
则当 \(\delta\downarrow0\) 时，
\[
\widehat m_\delta
=
\frac{\log(1/\delta)}{(1-d)\log\lambda_{\mathrm{cyl}}}
+o\!\bigl(\log(1/\delta)\bigr).
\]
\end{theorem}
\begin{proof}
置
\[
E_m:=\mathsf P_m^\star\triangle\mathsf P,
\qquad
\widehat E_m:=\widehat{\mathsf P}_m\triangle\mathsf P.
\]
若 \(\omega\in\widehat E_m\) 且 \(\omega\notin A_m^+\)，则由
\(\widehat{\mathsf P}_m\triangle\mathsf P_{m+h}^\star\subseteq A_m^+\) 可知 \(\omega\) 在 \(\widehat{\mathsf P}_m\) 与 \(\mathsf P_{m+h}^\star\) 中具有相同判决；既然它在 \(\widehat{\mathsf P}_m\) 与 \(\mathsf P\) 中判决不同，必有 \(\omega\in E_{m+h}\)。故
\[
\widehat E_m\subseteq A_m^+\cup E_{m+h}.
\]
取测度得到
\[
\widehat\varepsilon_m\le\alpha_m^++\varepsilon_{m+h}(\mathsf P;\mu).
\]
同理，若 \(\omega\in E_m\) 且 \(\omega\notin A_m^-\)，则 \(\omega\in\widehat E_{m+h}\)，故
\[
E_m\subseteq A_m^-\cup\widehat E_{m+h},
\]
从而
\[
\varepsilon_m(\mathsf P;\mu)\le\alpha_m^-+\widehat\varepsilon_{m+h}.
\]

由 \(\widehat E_m\subseteq A_m^+\cup E_{m+h}\) 对 \(m\) 取 limsup，得到
\[
\limsup_m\widehat E_m\subseteq \limsup_m A_m^+\cup\limsup_m E_{m+h}.
\]
若 \(\sum_m\varepsilon_m(\mathsf P;\mu)<\infty\) 且 \(\sum_m\alpha_m^+<\infty\)，则 \(\sum_m\mu(E_{m+h})<\infty\) 且 \(\sum_m\mu(A_m^+)<\infty\)。由 Borel--Cantelli 引理，\(\mu(\limsup_m E_{m+h})=0\) 且 \(\mu(\limsup_m A_m^+)=0\)。于是 \(\mu(\limsup_m\widehat E_m)=0\)，等价于 \(\widehat{\mathsf P}_m\) 对 \(\mu\)-几乎处处点仅犯有限次错误，即
\[
\sum_{m=1}^\infty \ind_{\widehat{\mathsf P}_m\triangle\mathsf P}(\omega)<\infty
\]
几乎处处成立。

对一般可测集 \(G\)，将上面的 limsup 包含式与 \(G\) 相交。若
\[
\mu\bigl(G\cap\limsup_m E_{m+h}\bigr)=0
\]
而
\[
\mu\bigl(G\cap\limsup_m\widehat E_m\bigr)>0,
\]
则只能有
\[
\mu\bigl(G\cap\limsup_m A_m^+\bigr)>0.
\]
这给出无限错误坏例子的载体分类。

最后假设推论 \ref{cor:spg-boundary-dimension-ae-stabilization} 的条件成立。该推论给出
\[
\varepsilon_m(\mathsf P;\mu)=\lambda_{\mathrm{cyl}}^{-\gamma m+o(m)}.
\]
由第一条风险不等式和 \(h\) 固定，
\[
\widehat\varepsilon_m
\le
\lambda_{\mathrm{cyl}}^{-\gamma(m+h)+o(m)}+\lambda_{\mathrm{cyl}}^{-\gamma m+o(m)}
=
\lambda_{\mathrm{cyl}}^{-\gamma m+o(m)}.
\]
由第二条风险不等式，
\[
\widehat\varepsilon_{m+h}\ge \varepsilon_m(\mathsf P;\mu)-\alpha_m^-
=
(1-o(1))\varepsilon_m(\mathsf P;\mu)
=
\lambda_{\mathrm{cyl}}^{-\gamma m+o(m)}.
\]
将指标重命名即得
\[
\widehat\varepsilon_m=\lambda_{\mathrm{cyl}}^{-\gamma m+o(m)}.
\]
因此对任意 \(\eta\in(0,\gamma)\)，充分大的 \(m\) 满足
\[
\lambda_{\mathrm{cyl}}^{-(\gamma+\eta)m}\le \widehat\varepsilon_m\le
\lambda_{\mathrm{cyl}}^{-(\gamma-\eta)m}.
\]
由 \(\widehat m_\delta\) 的定义，这给出
\[
\frac{\log(1/\delta)}{(\gamma+\eta)\log\lambda_{\mathrm{cyl}}}+O(1)
\le
\widehat m_\delta
\le
\frac{\log(1/\delta)}{(\gamma-\eta)\log\lambda_{\mathrm{cyl}}}+O(1).
\]
令 \(\eta\downarrow0\)，并代回 \(\gamma=1-d\)，即得所述扫描深度渐近律。证毕。
\end{proof}
\begin{proposition}[边界柱清晰度的谱求和判据]\label{prop:distill-connes-boundary-clarity-summability}
设定理 \ref{thm:spg-clarity-volume-boundary-scaling} 的厚度假设成立。对每个 \(m\ge1\)，取维数为 \(N_m(\partial\mathsf P)\) 的有限维 Hilbert 空间 \(H_m\)，并定义
\[
H_{\partial}:=\bigoplus_{m\ge1}H_m,\qquad
D_{\partial}\big|_{H_m}:=\lambda_{\mathrm{cyl}}^m\mathrm{Id}_{H_m}.
\]
记
\[
\zeta_{\partial}(s):=\operatorname{Tr}(D_{\partial}^{-s})
=\sum_{m\ge1}N_m(\partial\mathsf P)\lambda_{\mathrm{cyl}}^{-sm},\qquad
\delta_{\partial}:=\inf\{s>0:\zeta_{\partial}(s)<\infty\}.
\]
则
\[
\sum_{m=1}^{\infty}\varepsilon_m(\mathsf P;\mu)<\infty
\quad\Longleftrightarrow\quad
\zeta_{\partial}(1)<\infty.
\]
若进一步存在
\[
d=\lim_{m\to\infty}\frac{\log N_m(\partial\mathsf P)}{m\log\lambda_{\mathrm{cyl}}},
\]
则 \(\delta_{\partial}=d\)。特别地，在 \(d<1\) 时，\(D_{\partial}\) 的 \(1\)-可和性等价于 Bayes 前缀判决误差的可和性，并由定理 \ref{thm:spg-bayes-finite-mistakes-summable} 推出几乎处处最终定解。
\end{proposition}
\begin{proof}
由定理 \ref{thm:spg-clarity-volume-boundary-scaling}，存在正常数 \(C_-,C_+\) 使充分大 \(m\) 满足
\[
C_-\lambda_{\mathrm{cyl}}^{-m}N_m(\partial\mathsf P)\le
\varepsilon_m(\mathsf P;\mu)\le
C_+\lambda_{\mathrm{cyl}}^{-m}N_m(\partial\mathsf P).
\]
去掉有限多个初始项不影响级数收敛性，因此 \(\sum_m\varepsilon_m(\mathsf P;\mu)\) 收敛当且仅当
\[
\sum_m N_m(\partial\mathsf P)\lambda_{\mathrm{cyl}}^{-m}=\zeta_{\partial}(1)
\]
收敛。

设极限 \(d\) 存在。若 \(s>d\)，取 \(\rho>0\) 使 \(d+\rho<s\)。充分大 \(m\) 有
\[
N_m(\partial\mathsf P)\le \lambda_{\mathrm{cyl}}^{(d+\rho)m},
\]
故 \(\zeta_{\partial}(s)\) 被收敛几何级数
\[
\sum_m \lambda_{\mathrm{cyl}}^{(d+\rho-s)m}
\]
控制。若 \(0<s<d\)，取 \(\rho>0\) 使 \(s<d-\rho\)。充分大 \(m\) 有
\[
N_m(\partial\mathsf P)\ge \lambda_{\mathrm{cyl}}^{(d-\rho)m},
\]
故 \(\zeta_{\partial}(s)\) 的一般项不趋于零。于是 \(\delta_{\partial}=d\)；当 \(d=0\) 时同一上界对每个 \(s>0\) 成立，故 \(\delta_{\partial}=0\)。最后一项由第一部分与定理 \ref{thm:spg-bayes-finite-mistakes-summable} 立即得到。证毕。
\end{proof}
