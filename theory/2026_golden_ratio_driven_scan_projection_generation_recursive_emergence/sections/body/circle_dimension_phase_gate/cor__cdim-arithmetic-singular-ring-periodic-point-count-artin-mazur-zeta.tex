\begin{corollary}[乘法动力学的周期点计数与 Artin--Mazur \(\zeta\)-函数]\label{cor:cdim-arithmetic-singular-ring-periodic-point-count-artin-mazur-zeta}
令 \(S\subset\mathbb{P}\) 为有限素数集，记 \(s:=|S|\ge 1\)，并令 \(\mathfrak{R}_S\) 为算术奇异环。
固定整数 \(n\ge 2\)，考虑自同态 \([n]:\mathfrak{R}_S\to\mathfrak{R}_S\)。
则对任意 \(k\ge 1\) 有
\[
\operatorname{Fix}([n]^k)=\ker\bigl([n^k-1]:\mathfrak{R}_S\to \mathfrak{R}_S\bigr),
\]
并且若记
\(
(n^k-1)_{S^c}:=\prod_{p\notin S}p^{v_p(n^k-1)}
\)
为 \(n^k-1\) 的 \(S^c\)-部分，则
\[
\boxed{\;
|\operatorname{Fix}([n]^k)|=(n^k-1)^{s-1}\cdot (n^k-1)_{S^c}\; }.
\]
因此 \([n]\) 的 Artin--Mazur \(\zeta\)-函数可写为
\[
\zeta_{[n]}(z)
:=\exp\!\left(\sum_{k\ge 1}\frac{|\operatorname{Fix}([n]^k)|}{k}\,z^k\right)
=\exp\!\left(\sum_{k\ge 1}\frac{(n^k-1)^{s-1}(n^k-1)_{S^c}}{k}\,z^k\right).
\]
\leanverified{paper\_cdim\_periodic\_point\_count\_artin\_mazur\_zeta}
\end{corollary}

\begin{proof}
由于 \([n]^k=[n^k]\)，定点条件 \([n]^k(x)=x\) 等价于 \([n^k-1](x)=0\)，从而
\(
\operatorname{Fix}([n]^k)=\ker([n^k-1])
\)。
核的基数由定理 \ref{thm:cdim-solenoid-arithmetic-singular-ring-torsion-kernel-formula} 取 \(m=n^k-1\) 直接给出。证毕。
\end{proof}
\begin{definition}[乘法自同态的 \(S\)-尺度分母账本]\label{def:distill-cdim-scaling-denominator-ledger}
沿推论 \ref{cor:cdim-arithmetic-singular-ring-periodic-point-count-artin-mazur-zeta} 的记号，令
\[
A_{S,n}(k):=|\operatorname{Fix}([n]^k)|,
\qquad
D^{[n]}_{S}(k):=\prod_{p\in S}p^{v_p(n^k-1)},
\]
其中 \(v_p\) 为 \(p\)-进赋值。称 \(D^{[n]}_{S}\) 为 \([n]\) 的有限 \(S\)-尺度分母账本。若存在整数 \(B\ge 0\) 与函数
\[
E(k)=\prod_{p\in S}p^{e_p(k)},
\qquad e_p(k)\in\mathbb Z_{\ge0},
\]
满足对所有 \(p\in S\) 与 \(k\ge1\) 有 \(e_p(k)\le B\)，则称 \(E\) 为 \(B\)-有界的 \(S\)-分母账本。
\end{definition}

\begin{lemma}[有限尺度账本的次指数性与无界轴]\label{lem:distill-cdim-scaling-denominator-subexponential}
令 \(S\subset\mathbb P\) 为有限集且 \(n\ge2\)。则
\[
\lim_{k\to\infty}\bigl(D^{[n]}_{S}(k)\bigr)^{1/k}=1.
\]
进一步，若存在 \(p\in S\) 使 \(p\nmid n\)，则数列 \(v_p(n^k-1)\) 关于 \(k\) 无界；因而 \(D^{[n]}_{S}\) 不是任何 \(B\)-有界 \(S\)-分母账本。
\end{lemma}
\begin{proof}
若 \(p\mid n\)，则 \(n^k-1\equiv -1\pmod p\)，故 \(v_p(n^k-1)=0\)。设 \(p\nmid n\)。当 \(p\) 为奇素数时，记 \(\rho_p\) 为 \(n\) 在 \((\mathbb Z/p\mathbb Z)^\times\) 中的阶。若 \(\rho_p\nmid k\)，则 \(v_p(n^k-1)=0\)；若 \(\rho_p\mid k\)，由提升指数引理得
\[
v_p(n^k-1)=v_p\bigl(n^{\rho_p}-1\bigr)+v_p(k/\rho_p)
=v_p\bigl(n^{\rho_p}-1\bigr)+v_p(k),
\]
其中最后一步用到 \(p\nmid\rho_p\)。

当 \(p=2\) 且 \(2\nmid n\) 时，\(n\) 为奇数。若 \(k\) 为奇数，则
\[
v_2(n^k-1)=v_2(n-1),
\]
而若 \(k\) 为偶数，则提升指数引理给出
\[
v_2(n^k-1)=v_2(n^2-1)+v_2(k)-1.
\]
因此对每个 \(p\in S\) 均存在常数 \(C_p\) 使
\[
v_p(n^k-1)\le C_p+v_p(k)
\qquad(k\ge1).
\]
于是
\[
D^{[n]}_{S}(k)\le
\left(\prod_{p\in S}p^{C_p}\right)
\prod_{p\in S}p^{v_p(k)}
\le C k
\]
对某常数 \(C>0\) 成立，故 \(\bigl(D^{[n]}_{S}(k)\bigr)^{1/k}\to1\)。

若 \(p\in S\) 且 \(p\nmid n\)，则对奇素数 \(p\) 取 \(k=\rho_p p^a\)，上式给出
\[
v_p(n^k-1)=v_p(n^{\rho_p}-1)+a\to\infty.
\]
对 \(p=2\) 取 \(k=2^a\)，上述偶数情形同样给出 \(v_2(n^k-1)\to\infty\)。故相应 \(p\)-进尺度轴无界，结论成立。
\end{proof}

\begin{theorem}[周期点计数的尺度账本半径与有界迹阻碍]\label{thm:distill-cdim-scaling-denominator-shadow-radius}
沿推论 \ref{cor:cdim-arithmetic-singular-ring-periodic-point-count-artin-mazur-zeta} 的记号，设 \(s=|S|\ge1\)。则对所有 \(k\ge1\) 有精确分解
\[
A_{S,n}(k)=\frac{(n^k-1)^s}{D^{[n]}_{S}(k)}.
\]
因此
\[
\lim_{k\to\infty}A_{S,n}(k)^{1/k}=n^s.
\]
特别地，对数 Artin--Mazur 级数
\[
L_{S,n}(z):=\sum_{k\ge1}\frac{A_{S,n}(k)}{k}z^k
\]
的收敛半径恰为 \(n^{-s}\)。

若存在 \(p\in S\) 使 \(p\nmid n\)，则不存在整数 \(B\ge0\) 与 \(B\)-有界 \(S\)-分母账本 \(E\) 使得对所有 \(k\ge1\) 同时成立
\[
A_{S,n}(k)=\frac{(n^k-1)^s}{E(k)}.
\]
换言之，在含有真正外部尺度素数 \(p\nmid n\) 的情形下，周期点计数不能由有界 multiplicity ledger 承载；任何有限迹解释若要逐周期精确，必须保留无界的 \(p\)-进尺度账本。
\end{theorem}
\begin{proof}
由推论 \ref{cor:cdim-arithmetic-singular-ring-periodic-point-count-artin-mazur-zeta}，
\[
A_{S,n}(k)=(n^k-1)^{s-1}(n^k-1)_{S^c}.
\]
另一方面
\[
n^k-1=D^{[n]}_{S}(k)\,(n^k-1)_{S^c}.
\]
代入即得
\[
A_{S,n}(k)
=\bigl(D^{[n]}_{S}(k)(n^k-1)_{S^c}\bigr)^{s-1}(n^k-1)_{S^c}
=\frac{(n^k-1)^s}{D^{[n]}_{S}(k)}.
\]
由引理 \ref{lem:distill-cdim-scaling-denominator-subexponential}，\(D^{[n]}_{S}(k)^{1/k}\to1\)，而 \((n^k-1)^{1/k}\to n\)，故
\[
A_{S,n}(k)^{1/k}
=\frac{(n^k-1)^{s/k}}{\bigl(D^{[n]}_{S}(k)\bigr)^{1/k}}
\longrightarrow n^s.
\]
Cauchy--Hadamard 公式给出
\[
R^{-1}
=\limsup_{k\to\infty}\left(\frac{A_{S,n}(k)}{k}\right)^{1/k}
=n^s,
\]
因此 \(L_{S,n}(z)\) 的收敛半径为 \(R=n^{-s}\)。

最后，若存在 \(B\)-有界 \(E\) 满足
\[
A_{S,n}(k)=\frac{(n^k-1)^s}{E(k)}
\]
对所有 \(k\) 成立，则与已经证明的精确分解比较，因 \(n^k-1\ne0\) 且 \(A_{S,n}(k)>0\)，必有
\[
E(k)=D^{[n]}_{S}(k)
\qquad(k\ge1).
\]
但当某个 \(p\in S\) 满足 \(p\nmid n\) 时，引理 \ref{lem:distill-cdim-scaling-denominator-subexponential} 表明 \(v_p(D^{[n]}_{S}(k))=v_p(n^k-1)\) 沿一列 \(k\) 无界，这与 \(E\) 的 \(B\)-有界性矛盾。证毕。
\end{proof}
