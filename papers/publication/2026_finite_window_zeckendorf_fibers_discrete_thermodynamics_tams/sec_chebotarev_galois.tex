\begin{lemma}[Soundness of the finite-field certificates]
\label{lem:finite-field-certificate-soundness}
For each entry of \Cref{tab:galois-certificates-q9-17}, exact Euclidean
division in \(\FF_p[\lambda]\) gives a squarefree factorization of
\(P_q\bmod p\) with the displayed degree pattern.  Consequently the displayed
patterns occur as cycle types in \(\Gal(P_q/\QQ)\).
\end{lemma}

\begin{proof}
In the exact certificate format used for a table entry, the proof object
consists of monic factors in \(\FF_p[\lambda]\), their product identity with
\(P_q\bmod p\), and the Euclidean gcd identity
\[
  \gcd(P_q\bmod p, P_q'\bmod p)=1.
\]
The monic factors and the Bezout polynomials for the gcd identity are recorded
in the artifact of \Cref{prop:pinned-normalizer-certificate-table}.  All three
checks are exact polynomial identities in the finite field, obtained by reducing
the printed integer polynomial \(P_q\) modulo the displayed prime; no root
approximation or scalar moment data enter this certificate.  The gcd identity
says that the listed prime is unramified for \(P_q\).  Dedekind's
factorization theorem, in the squarefree case, identifies the degrees of the
irreducible factors of \(P_q\bmod p\) with the cycle lengths of a Frobenius
element in the Galois group.  Hence each displayed degree pattern is realized
as a cycle type in \(\Gal(P_q/\QQ)\).
\end{proof}

\begin{proposition}[Irreducibility in the resonance window]
\label{prop:irreducibility-window}
The exact certificate records of
Theorem~\ref{thm:arithmetic-window-certificates} imply that, for
\(q=9,\dots,17\), the certified
recurrence polynomial \(P_q\) is
irreducible over \(\QQ\).  Consequently it is the minimal recurrence polynomial
of the nonzero sequence \((S_q(m))_{m\ge m_0(q)}\).  Its degree sequence is
\[
  (d_9,d_{10},d_{11},d_{12},d_{13},d_{14},d_{15},d_{16},d_{17})
  =(7,9,9,13,11,13,11,13,13).
\]
\end{proposition}

\begin{proof}
The degree data are the orders of the kernel-certified recurrences listed in
Table~\ref{tab:appendix-recurrences-q9-17}.  By
\Cref{lem:finite-field-certificate-soundness}, for each \(q\) there exists a prime
\(p_{\mathrm{irr}}\) such that \(P_q\bmod p_{\mathrm{irr}}\) is irreducible in
\(\FF_{p_{\mathrm{irr}}}[\lambda]\).  Since \(P_q\) is monic with integer
coefficients, Gauss's lemma implies that \(P_q\) is irreducible over \(\QQ\).
By \Cref{prop:section7-recurrence-input}, \(P_q\) annihilates the tail
\((S_q(m))_{m\ge m_0(q)}\).  The monic minimal recurrence polynomial of this
tail divides every monic annihilating polynomial in \(\QQ[\lambda]\).  Since
the tail is not identically zero and \(P_q\) is irreducible, that divisor cannot
be \(1\); hence it is \(P_q\).
\end{proof}

\begin{lemma}[A transitive group with an \texorpdfstring{$(n-1)$}{(n-1)}-cycle is primitive]
\label{lem:nminus1-cycle-primitive}
Let \(G\le S_n\) be transitive with \(n\ge 3\).  If \(G\) contains an
\((n-1)\)-cycle, then \(G\) is primitive.
\end{lemma}

\begin{proof}
Let \(\tau\in G\) be an \((n-1)\)-cycle, and let \(\alpha\) be its unique fixed
point.  Consider a block \(B\subseteq\{1,\dots,n\}\) for the action of \(G\).

If \(\alpha\in B\), then \(B\setminus\{\alpha\}\) is \(\tau\)-invariant inside the
single \(\tau\)-orbit \(\{1,\dots,n\}\setminus\{\alpha\}\), so either
\(B=\{\alpha\}\) or \(B=\{1,\dots,n\}\).

If \(\alpha\notin B\), choose \(\beta\in B\) and \(g\in G\) with \(g(\beta)=\alpha\),
using transitivity.  Then \(g(B)\) is a block containing \(\alpha\), hence
trivial or all by the previous paragraph.  Therefore \(B\) itself is trivial or
all.
\end{proof}

\begin{theorem}[Full symmetric Galois groups in the resonance window]
\label{thm:galois-sd-window}
The exact certificate records of
Theorem~\ref{thm:arithmetic-window-certificates} imply that, for every
\(q=9,\dots,17\),
\[
  \Gal(P_q/\QQ)\cong S_{d_q}.
\]
\end{theorem}

\begin{proof}
Fix \(q\in\{9,\dots,17\}\), and let \(G_q:=\Gal(P_q/\QQ)\le S_{d_q}\) in its
natural action on the roots of \(P_q\).

By \Cref{prop:irreducibility-window}, the polynomial \(P_q\) is irreducible, so
\(G_q\) is transitive.  The factorization type \((d_q-1,1)\), interpreted by
\Cref{lem:finite-field-certificate-soundness}, exhibits an element of \(G_q\)
that is an \((d_q-1)\)-cycle, so \Cref{lem:nminus1-cycle-primitive} shows that
\(G_q\) is primitive.

The same certificate table also provides an element of prime cycle length
\[
  \ell_q\in\{3,5,5,7,7,7,7,7,7\},
\]
and in each case \(\ell_q\le d_q-3\).  Jordan's theorem therefore implies that
\(A_{d_q}\subseteq G_q\); see \cite[Thm.~3.3A]{DixonMortimer1996}.

Finally, an \((d_q-1)\)-cycle is odd because \(d_q-1\) is even for all
\(d_q\in\{7,9,11,13\}\).  Hence \(G_q\) is not contained in \(A_{d_q}\).  Since
\(A_{d_q}\subseteq G_q\le S_{d_q}\), we conclude that \(G_q=S_{d_q}\).
\end{proof}

\begin{corollary}[Chebotarev densities]
\label{cor:individual-chebotarev}
For each \(q=9,\dots,17\), the set of unramified primes \(p\) for which
\(P_q\bmod p\) is irreducible in \(\FF_p[\lambda]\) has natural density
\(1/d_q\).
\end{corollary}

\begin{proof}
By \Cref{thm:galois-sd-window} and the Chebotarev density theorem
\cite{Neukirch1999}, Frobenius classes are equidistributed in \(S_{d_q}\).
Irreducibility modulo \(p\) is equivalent to Frobenius being a
\(d_q\)-cycle, and the proportion of \(d_q\)-cycles in \(S_{d_q}\) is
\((d_q-1)!/d_q!=1/d_q\).
\end{proof}

\begin{lemma}[The partition constant is a root of the certified polynomial]
\label{lem:lambda-root-certified-polynomial}
For each \(q=9,\dots,17\), the partition constant \(\lambda_q\) satisfies
\[
  P_q(\lambda_q)=0.
\]
\end{lemma}

\begin{proof}
Fix \(q\in\{9,\dots,17\}\), write \(d=d_q\), and set
\[
  Q_q(z):=1-c_{q,1}z-\cdots-c_{q,d}z^d.
\]
Let
\[
  F_q(z):=\sum_{m\ge0} S_q(m)z^m.
\]
By \Cref{prop:section7-recurrence-input},
\[
  S_q(m)=\sum_{i=1}^d c_{q,i}S_q(m-i)
  \qquad(m\ge d+2).
\]
Therefore the coefficient of \(z^m\) in \(Q_q(z)F_q(z)\) vanishes for every
\(m\ge d+2\).  The finitely many initial coefficients \(m<d+2\) only form the
polynomial numerator \(N_q(z):=Q_q(z)F_q(z)\); they do not affect the radius of
convergence of \(F_q\).  Thus
\[
  F_q(z)=\frac{N_q(z)}{Q_q(z)}
\]
as a formal power series, and hence as a rational function in its disk of
convergence.

Now \(S_q(m)\ge0\) for every \(m\), and \Cref{thm:all-q-transfer} gives
\[
  S_q(m)\asymp_q \lambda_q^m.
\]
The Cauchy--Hadamard formula therefore gives radius of convergence
\(R_q=1/\lambda_q\) for \(F_q\).  Since the coefficients of \(F_q\) are
nonnegative, Pringsheim's theorem
\cite[Thm.~IV.6]{FlajoletSedgewick2009} shows that \(z=R_q\) is a singularity
of \(F_q\).  A cancellation between \(N_q\) and \(Q_q\) cannot remove this
singularity: after cancellation the same rational function would be analytic at
\(R_q\), contradicting Pringsheim.  Hence every denominator representation of
the rational function has a zero at \(R_q\), in particular the displayed
denominator does, so
\[
  Q_q(1/\lambda_q)=0.
\]
Multiplying by \(\lambda_q^d\) gives
\[
  0=\lambda_q^dQ_q(1/\lambda_q)=P_q(\lambda_q).
\]
\end{proof}

\begin{corollary}[Rational R\'enyi denominators are large]
\label{cor:renyi-rational-denominator}
For \(q\in\{9,\dots,17\}\), set
\[
  D_q:=\frac{q\log 2-\log \lambda_q}{(q-1)\log\varphi}.
\]
If \(D_q=a/b\in\QQ\) in lowest terms, then
\[
  b\ge \left\lceil\frac{d_q}{2}\right\rceil.
\]
In particular, the range \(q=9,\dots,17\) yields the lower bounds
\[
  b\ge 4,5,5,7,6,7,6,7,7
  \qquad (q=9,\dots,17).
\]
\end{corollary}

\begin{proof}
From the definition of \(D_q\),
\[
  \lambda_q = 2^q\varphi^{-(q-1)a/b}\in \QQ(\varphi^{1/b}).
\]
Since \(\QQ(\varphi)=\QQ(\sqrt5)\) is quadratic, the extension
\(\QQ(\varphi^{1/b})/\QQ\) has degree at most \(2b\).  By
\Cref{lem:lambda-root-certified-polynomial}, the number \(\lambda_q\) is a root
of \(P_q\).  Since \(P_q\) is irreducible of degree \(d_q\) by
\Cref{prop:irreducibility-window}, it is the minimal polynomial of
\(\lambda_q\) over \(\QQ\).  Therefore \([\QQ(\lambda_q):\QQ]=d_q\).
Therefore \(d_q\le 2b\), which is the claimed lower bound.
\end{proof}
