\section{Finite-state realizations, convex pressure, and canonical selection}
\label{sec:moment-kernel}

This section studies the collision moments \(S_q(m)\) through their finite-state
realizations in the length variable \(m\).  The language of symbolic dynamics
and Perron--Frobenius theory used below is standard; see
\cite{LindMarcus1995,Seneta2006}.  The pressure and large-deviation
perspective follows the thermodynamic formalism of
\cite{DemboZeitouni2010}.  Combined with \Cref{thm:all-q-transfer}, the
finite-state realizations turn Sanna's constants \(\lambda_q\) into
Perron roots, yielding algebraicity, convex pressure geometry, canonical
selection bands, and zero-temperature concentration.  The transducer
mechanism behind the kernel construction is recorded in
Appendix~\ref{app:transducer}.

\subsection{Finite-state realizations of the collision moments}

\begin{definition}[Collision moments]
\label{def:collision-moments}
For integers \(m\ge 0\) and \(q\ge 0\), define
\[
  S_q(m):=\sum_{x\in X_m} d_m(x)^q.
\]
Thus \(S_0(m)=\card{X_m}=F_{m+2}\), \(S_1(m)=2^m\), and for \(q\ge 2\),
\[
  S_q(m)
  =
  \card{\bigl\{(\omega^{(1)},\dots,\omega^{(q)})\in\Omega_m^q:
  \Fold_m(\omega^{(1)})=\cdots=\Fold_m(\omega^{(q)})\bigr\}}.
\]
In particular, \(S_q(0)=1\) for every \(q\ge0\).
\end{definition}

\begin{remark}
The convention includes the empty window \(m=0\), which is needed below for the
exact finite-state initial condition.
\end{remark}

\begin{theorem}[Finite-state collision kernels]
\label{thm:collision-kernel}
\leanverified{paper_projection_collision_kernel}
For every integer \(q\ge 2\), there exist a finite nonnegative integer matrix
\(A_q\) and nonnegative integer vectors \(u_q,v_q\) such that
\[
  S_q(m)=u_q^{\mathsf T}A_q^m v_q
  \qquad (m\ge 0).
\]
Moreover, \(A_q\) may be chosen trim: every state of the underlying automaton
lies on a directed path from the initial state to an accepting terminal state.
\end{theorem}

\begin{remark}
The realization used here is the trimmed \(m=0\)-inclusive automaton
reconstructed from the hash-pinned overflow-aware normalizer in
Appendix~\ref{app:transducer}.
\end{remark}

\begin{proof}
Appendix~\ref{app:transducer} makes the construction explicit.  Choose the
hash-pinned padded deterministic subsequential normalizer on reversed Fibonacci
words, and identify the finite-window fold with the fixed-width normalized
output after deleting its unique possible overflow digit; see
\Cref{thm:pinned-padded-normalizer-soundness,thm:appendix-fixed-width-overflow}.  The
finite table, the augmented one-copy states, and the histogram kernels are
reconstructed from the archived transition table by
\Cref{prop:auditable-kernel-reconstruction}.  By
\Cref{lem:appendix-prefix-stability}, once a prefix has been read, only a
bounded suffix of the visible output can still be altered by any further
continuation.  The synchronized collision automaton of
\Cref{prop:appendix-collision-automaton} stores exactly this unresolved suffix,
checks equality of the newly committed visible blocks at each step, and accepts
precisely those \(q\)-tuples with common finite-window fold.  Its adjacency
matrix is a finite nonnegative integer matrix \(A_q\), and the initial and
terminal conditions are encoded by nonnegative integer vectors \(u_q,v_q\).
Deleting any state that is not reachable from the initial synchronized state or
cannot reach an accepting terminal state does not change the accepted path
count.  Thus the automaton may be taken trim.  Let \(A_q\) be the adjacency
matrix of this trim automaton, let \(u_q\) select the initial state, and let
\(v_q\) mark the accepting terminal states.  Then
\[
  S_q(m)=u_q^{\mathsf T}A_q^m v_q
  \qquad (m\ge 0).
\]
\end{proof}

\begin{theorem}[Symmetric compression]
\label{thm:symmetric-compression}
\leanverified{paper_projection_symmetric_compression}
Let
\[
  k:=|\widehat Q|=2\,|Q|\,(2^{L+2}-1)
\]
be the number of states of the overflow-aware one-copy augmented fold automaton
of \Cref{prop:appendix-collision-automaton}.  The kernel \(A_q\) of
\Cref{thm:collision-kernel} admits an overflow-aware symmetric quotient
\(\widetilde A_q\) acting on histogram states of cardinality
\[
  \binom{k+q-1}{k-1},
\]
such that
\[
  S_q(m)=\widetilde u_q^{\mathsf T}\widetilde A_q^m\widetilde v_q
  \qquad\text{and}\qquad
  \rho(\widetilde A_q)=\rho(A_q)=r_q.
\]
In particular, for fixed \(k\) the state-space dimension grows polynomially in
\(q\).
\end{theorem}

\begin{remark}
The state count \(k=2|Q|(2^{L+2}-1)\) comes from the overflow-aware residual
bound \(L+1\) and the deletion flag in Appendix~\ref{app:transducer}.
\end{remark}

\begin{proof}
Let \(\widehat Q\) be the one-copy augmented state set from
\Cref{prop:appendix-collision-automaton}, so \(k=\card{\widehat Q}\).  The
symmetric group \(\mathfrak S_q\) acts on the synchronized state space
\(\widehat Q^q\) by permuting the \(q\) coordinates.  The transition relation
and the collision condition are invariant under this action, so the kernel
descends to the orbit space.  If \(s=(s_1,\dots,s_q)\in \widehat Q^q\), its
orbit is determined by the histogram
\[
  \eta_s(t):=\card{\{1\le i\le q:\ s_i=t\}}
  \qquad (t\in \widehat Q),
\]
namely a weak composition of \(q\) into \(k\) parts.  Hence the orbit space has
cardinality \(\binom{k+q-1}{k-1}\).

Because the synchronized update rule of
\Cref{prop:appendix-collision-automaton} is coordinatewise and depends only on
whether the newly committed visible blocks agree across the \(q\) coordinates,
orbit sums produce a nonnegative integer quotient matrix \(\widetilde A_q\)
with the stated realization formula.  The symmetric subspace of
\(\RR^{\widehat Q^q}\) is \(A_q\)-invariant and is naturally identified with
the histogram state space, so \(\widetilde A_q\) is precisely the restriction
of \(A_q\) to that invariant subspace.  Averaging a Perron eigenvector of
\(A_q\) over \(\mathfrak S_q\) produces a nonzero symmetric eigenvector with
the same eigenvalue, and therefore \(\rho(\widetilde A_q)=\rho(A_q)\).
\end{proof}

\begin{definition}[Pressure data and canonical laws]
\label{def:pressure-slopes}
Let
\[
  P_0:=\log\varphi,\qquad P_1:=\log2,\qquad
  P_q:=\log\lambda_q\quad(q\ge2),
\]
where \(\lambda_q\) is the partition constant transferred in
\Cref{thm:all-q-transfer}.  For \(q\ge1\), set
\[
  \Delta_q:=P_q-P_{q-1}.
\]
For \(m\ge0\) and \(q\ge1\), the \(q\)-canonical law on \(X_m\) is
\[
  \mu_{m,q}(x):=\frac{d_m(x)^q}{S_q(m)}.
\]
The visible law is
\[
  p_m(x):=\frac{d_m(x)}{2^m},
\]
and
\[
  D_m:=\max_{x\in X_m}d_m(x).
\]
\end{definition}

\begin{corollary}[Algebraic realization of the partition constants]
\label{cor:lambda-algebraic}
\leanverified{paper_projection_lambda_algebraic}
For every integer \(q\ge 2\),
\[
  \lambda_q=\rho(A_q)=\rho(\widetilde A_q).
\]
In particular, \(\lambda_q\) is an algebraic integer.  More precisely,
\(\lambda_q\) is a root of the monic integer characteristic polynomial of the
nonnegative integer histogram matrix \(\widetilde A_q\), whose dimension is
\(\binom{k+q-1}{k-1}\).
\end{corollary}

\begin{proof}
The equality \(\lambda_q=\rho(A_q)\) is the final assertion of
\Cref{thm:all-q-transfer}.  The symmetric compression theorem gives
\(\rho(A_q)=\rho(\widetilde A_q)\), with \(\widetilde A_q\) a nonnegative
integer matrix of dimension \(\binom{k+q-1}{k-1}\).  By the
Perron--Frobenius theorem, the spectral radius of a nonnegative matrix is an
eigenvalue.  Hence \(\lambda_q\) is an eigenvalue of the integer matrix
\(\widetilde A_q\), and so it is a zero of the monic polynomial
\(\det(XI-\widetilde A_q)\in\ZZ[X]\).  This is exactly the definition of an
algebraic integer.
\end{proof}

\begin{proposition}[Hankel positivity and cross-moment convexity]
\label{prop:hankel-psd}
For every \(m\ge0\), the infinite Hankel matrix
\[
  \bigl(S_{i+j}(m)\bigr)_{i,j\ge0}
\]
is positive semidefinite.  Consequently, for every \(q\ge1\),
\[
  S_q(m)^2\le S_{q-1}(m)S_{q+1}(m),
\]
and, more generally, the sequence \(q\mapsto \log S_q(m)\) is convex on the
nonnegative integers.
\end{proposition}

\begin{proof}
Let \(c_0,\dots,c_\ell\in\RR\).  Using the definition of the collision
moments,
\[
  \sum_{i,j=0}^{\ell}c_i c_j S_{i+j}(m)
  =
  \sum_{x\in X_m}\sum_{i,j=0}^{\ell}c_i c_j d_m(x)^{i+j}
  =
  \sum_{x\in X_m}\left(\sum_{i=0}^{\ell}c_i d_m(x)^i\right)^2
  \ge0.
\]
Thus every finite principal block of the Hankel matrix is positive
semidefinite.

Taking the \(2\times2\) principal minor with rows and columns
\(\{q-1,q\}\), or equivalently applying Cauchy's inequality to the functions
\(d_m(x)^{(q-1)/2}\) and \(d_m(x)^{(q+1)/2}\), gives
\[
  S_q(m)^2\le S_{q-1}(m)S_{q+1}(m).
\]
Since all \(S_q(m)\) are positive, this is the midpoint-convexity inequality
\[
  2\log S_q(m)\le \log S_{q-1}(m)+\log S_{q+1}(m).
\]
On the integer line, these adjacent second-difference inequalities are
equivalent to convexity of \(q\mapsto\log S_q(m)\).
\end{proof}

\begin{corollary}[Adjacent pressure convexity]
\label{cor:cross-q-logconvex}
\leanverified{paper_projection_adjacent_pressure_convexity}
For every \(q\ge1\),
\[
  P_q\le \frac{P_{q-1}+P_{q+1}}2.
\]
Equivalently,
\[
  \Delta_q\le \Delta_{q+1}.
\]
\end{corollary}

\begin{proof}
By \Cref{prop:hankel-psd},
\[
  S_q(m)^2\le S_{q-1}(m)S_{q+1}(m).
\]
Taking logarithms, dividing by \(m\), and letting \(m\to\infty\) gives the
claim.  For \(q=1\), the limits are
\[
  \frac1m\log S_0(m)\to\log\varphi=P_0,\qquad
  \frac1m\log S_1(m)=\log2=P_1,
\]
and \(\frac1m\log S_2(m)\to P_2\) by \Cref{thm:all-q-transfer}.  For
\(q\ge2\), all three limits follow from \Cref{thm:all-q-transfer}.  The slope
form is the same inequality rewritten as
\(P_q-P_{q-1}\le P_{q+1}-P_q\).
\end{proof}

\subsection{Pressure geometry and the zero-temperature regime}

\begin{theorem}[Global pressure convexity]
\label{thm:global-pressure-convexity}
\leanverified{paper_projection_global_pressure_convexity}
The pressure sequence \((P_q)_{q\ge0}\) is convex on the nonnegative integer
line.  Equivalently, its piecewise-linear interpolation on \([0,\infty)\) is a
convex function.  In particular, for integers \(0\le a<b<c\),
\[
  P_b\le \frac{c-b}{c-a}P_a+\frac{b-a}{c-a}P_c .
\]
\end{theorem}

\begin{proof}
\Cref{cor:cross-q-logconvex} says that the adjacent slopes
\(\Delta_q=P_q-P_{q-1}\) are increasing.  If \(0\le a<b<c\), then
\[
  \frac{P_b-P_a}{b-a}
  =
  \frac1{b-a}\sum_{j=a+1}^{b}\Delta_j
  \le
  \frac1{c-b}\sum_{j=b+1}^{c}\Delta_j
  =
  \frac{P_c-P_b}{c-b}.
\]
Multiplying by \((b-a)(c-b)\) and rearranging gives
\[
  P_b\le \frac{c-b}{c-a}P_a+\frac{b-a}{c-a}P_c.
\]
This three-point inequality is the convexity of the sequence, and it is
equivalent to convexity of the piecewise-linear interpolation.
\end{proof}

\begin{lemma}[Extremal finite-window fiber growth]
\label{lem:extremal-fiber-growth}
The largest finite-window fiber satisfies
\[
  \lim_{m\to\infty}\frac1m\log D_m=\frac12\log\varphi .
\]
\end{lemma}

\begin{proof}
By \Cref{cor:coefficient-spectrum}, \(D_m\) is the largest coefficient of
\[
  \prod_{j=1}^m(1+z^{F_j}).
\]
For \(0\le n<F_{m+2}\), this coefficient is at most
\(R^\dagger(n)=R(n)+R(n-1)\), where \(R\) is the ordinary Fibonacci
representation function.  Koc\'abov\'a, Mas\'akov\'a, and Pelantov\'a
\cite{KocabovaMasakovaPelantova2005} determine the maximum of \(R(n)\) on each
interval between consecutive Fibonacci numbers.  In the indexing of the
present paper, their formula gives absolute constants \(0<c<C<\infty\) such
that, for every \(k\ge1\),
\[
  c\varphi^{k/2}
  \le
  \max_{F_k\le n<F_{k+1}}R(n)
  \le
  C\varphi^{k/2}.
\]
The upper bound over the intervals below \(F_{m+2}\) gives
\[
  \max_{0\le n<F_{m+2}} R(n)\le C\varphi^{m/2}
\]
after increasing \(C\).  Hence \(D_m\le 2C\varphi^{m/2}\), and
\[
  \limsup_{m\to\infty}\frac1m\log D_m\le \frac12\log\varphi .
\]

For the lower bound, apply the same estimate on the interval
\([F_m,F_{m+1})\).  It gives an \(n_m<F_{m+1}\) such that
\[
  R(n_m)\ge c\varphi^{m/2}
\]
after decreasing \(c\) if necessary.
Since \(n_m<F_{m+1}\), the finite coefficient equals the indexed partition
coefficient:
\[
  [z^{n_m}]\prod_{j=1}^m(1+z^{F_j})=R^\dagger(n_m)\ge R(n_m).
\]
Thus \(D_m\ge c\varphi^{m/2}\), giving the matching lower exponential rate.
\end{proof}

\begin{corollary}[Endpoint law for the partition constants]
\label{cor:renyi-endpoint}
\leanverified{paper_projection_renyi_endpoint}
One has
\[
  \lim_{q\to\infty}\lambda_q^{1/q}=\varphi^{1/2}.
\]
\end{corollary}

\begin{proof}
\Cref{lem:extremal-fiber-growth} gives
\[
  \frac1m\log D_m\longrightarrow \frac12\log\varphi .
\]
For every fixed \(q\ge1\),
\[
  D_m^q\le S_q(m)\le F_{m+2}D_m^q.
\]
Taking logarithms, dividing by \(m\), and letting \(m\to\infty\), using
\Cref{thm:all-q-transfer}, gives
\[
  q\cdot\frac12\log\varphi
  \le
  P_q
  \le
  \log\varphi+q\cdot\frac12\log\varphi .
\]
After division by \(q\), the squeeze tends to
\(\frac12\log\varphi\), which is the displayed assertion.
\end{proof}

\begin{lemma}[Adjacent moment ratios]
\label{lem:adjacent-moment-ratios}
For every fixed integer \(q\ge1\),
\[
  \frac{S_{q+1}(m)}{S_q(m)}
  =
  \exp\bigl((\Delta_{q+1}+o(1))m\bigr).
\]
Moreover, for \(q\ge2\),
\[
  \frac{S_{q-1}(m)}{S_q(m)}
  =
  \exp\bigl((-\Delta_q+o(1))m\bigr),
\]
and for \(q=1\),
\[
  \frac{S_0(m)}{S_1(m)}
  =
  \exp\bigl((-\Delta_1+o(1))m\bigr).
\]
\end{lemma}

\begin{proof}
For fixed \(j\ge1\), \Cref{thm:all-q-transfer} gives
\[
  S_j(m)=\exp\bigl((P_j+o(1))m\bigr).
\]
Taking quotients with \(j=q+1,q\) gives the first display, and taking
quotients with \(j=q-1,q\) gives the second when \(q\ge2\).  For \(q=1\),
\[
  S_0(m)=F_{m+2}=\exp\bigl((P_0+o(1))m\bigr),
  \qquad
  S_1(m)=2^m=\exp(P_1m),
\]
so \(S_0(m)/S_1(m)=\exp((P_0-P_1+o(1))m)\), which is the final display.
\end{proof}

\begin{corollary}[Pressure slopes]
\label{cor:pressure-slopes}
\leanverified{paper_projection_pressure_slopes}
\[
  \Delta_1\le\Delta_2\le\cdots
  \qquad\text{and}\qquad
  \lim_{q\to\infty}\Delta_q=\frac12\log\varphi .
\]
\end{corollary}

\begin{proof}
The monotonicity is \Cref{cor:cross-q-logconvex}.  Let
\(L:=\lim_{q\to\infty}\Delta_q\in(-\infty,\infty]\); the limit exists because
the sequence is increasing.  Since
\[
  \frac{P_q}{q}=\frac1q\sum_{j=1}^q\Delta_j,
\]
\Cref{cor:renyi-endpoint} gives
\[
  \frac1q\sum_{j=1}^q\Delta_j\longrightarrow\frac12\log\varphi .
\]
For an increasing sequence, the Cesaro means converge to the same limit as the
sequence, when the latter exists in the extended real line: if
\(L>\frac12\log\varphi+\eta\), then all sufficiently large \(\Delta_j\) exceed
that value and so do the tail Cesaro means; if
\(L<\frac12\log\varphi-\eta\), then all terms are eventually below that value.
Thus \(L=\frac12\log\varphi\).
\end{proof}

\begin{theorem}[Discrete Gibbs selection]
\label{thm:gibbs-selection}
\leanverified{paper_projection_gibbs_selection}
Fix an integer \(q\ge1\).  For every \(\varepsilon>0\),
\[
  \mu_{m,q}\left\{x:\frac1m\log d_m(x)\notin
  [\Delta_q-\varepsilon,\Delta_{q+1}+\varepsilon]\right\}
  \le \exp\bigl(-(\varepsilon+o(1))m\bigr).
\]
\end{theorem}

\begin{proof}
Write
\[
  L_m^-:=\left\{x:d_m(x)\le e^{(\Delta_q-\varepsilon)m}\right\},
  \qquad
  L_m^+:=\left\{x:d_m(x)\ge e^{(\Delta_{q+1}+\varepsilon)m}\right\}.
\]
For the lower tail, if \(q\ge2\), then
\[
  \mu_{m,q}(L_m^-)
  =
  \frac{\sum_{L_m^-}d_m(x)^q}{S_q(m)}
  \le
  e^{(\Delta_q-\varepsilon)m}\frac{S_{q-1}(m)}{S_q(m)}
  =
  \exp\bigl((-\varepsilon+o(1))m\bigr)
\]
by \Cref{lem:adjacent-moment-ratios}.  The same computation for \(q=1\) uses
\(d_m(x)\le e^{(\Delta_1-\varepsilon)m}\) and
\(\sum_{L_m^-}1\le S_0(m)\), together with the \(q=1\) part of
\Cref{lem:adjacent-moment-ratios}.

For the upper tail,
\[
  \mu_{m,q}(L_m^+)
  =
  \frac{\sum_{L_m^+}d_m(x)^q}{S_q(m)}
  \le
  e^{-(\Delta_{q+1}+\varepsilon)m}\frac{S_{q+1}(m)}{S_q(m)}
  =
  \exp\bigl((-\varepsilon+o(1))m\bigr).
\]
The union bound gives the displayed estimate; the factor \(2\) is absorbed in
the \(o(1)\) term.
\end{proof}

\begin{theorem}[Microcanonical band bounds]
\label{thm:microcanonical-bands}
\leanverified{paper_projection_microcanonical_band_bounds}
Fix an integer \(q\ge1\) and \(\varepsilon>0\), and set
\[
  B_{m,q}(\varepsilon)
  :=
  \left\{
    x\in X_m:
    \Delta_q-\varepsilon\le \frac1m\log d_m(x)
    \le \Delta_{q+1}+\varepsilon
  \right\}.
\]
Then
\[
  \mu_{m,q}\bigl(X_m\setminus B_{m,q}(\varepsilon)\bigr)
  \le \exp\bigl(-(\varepsilon+o(1))m\bigr).
\]
Moreover
\[
  P_q-q(\Delta_{q+1}+\varepsilon)
  \le
  \liminf_{m\to\infty}\frac1m\log |B_{m,q}(\varepsilon)|,
\]
\[
  \limsup_{m\to\infty}\frac1m\log |B_{m,q}(\varepsilon)|
  \le
  P_q-q(\Delta_q-\varepsilon),
\]
and, with \(p_m(A):=2^{-m}\sum_{x\in A}d_m(x)\),
\[
  P_q-\log 2-(q-1)(\Delta_{q+1}+\varepsilon)
  \le
  \liminf_{m\to\infty}\frac1m\log p_m(B_{m,q}(\varepsilon)),
\]
\[
  \limsup_{m\to\infty}\frac1m\log p_m(B_{m,q}(\varepsilon))
  \le
  P_q-\log 2-(q-1)(\Delta_q-\varepsilon).
\]
\end{theorem}

\begin{proof}
The canonical-mass estimate is exactly \Cref{thm:gibbs-selection}.

Let \(B_m=B_{m,q}(\varepsilon)\).  On \(B_m\),
\[
  e^{(\Delta_q-\varepsilon)m}\le d_m(x)\le
  e^{(\Delta_{q+1}+\varepsilon)m}.
\]
The upper cardinality bound follows from
\[
  |B_m|\,e^{q(\Delta_q-\varepsilon)m}
  \le
  \sum_{x\in B_m}d_m(x)^q
  \le S_q(m).
\]
For the lower cardinality bound, \Cref{thm:gibbs-selection} gives
\(\sum_{x\in B_m}d_m(x)^q=S_q(m)(1-o(1))\).  Hence
\[
  S_q(m)(1-o(1))
  \le
  |B_m|\,e^{q(\Delta_{q+1}+\varepsilon)m}.
\]
Using \(S_q(m)=\exp((P_q+o(1))m)\) gives the two cardinality estimates.

For visible mass, the upper bound is
\[
  p_m(B_m)
  =
  2^{-m}\sum_{x\in B_m}d_m(x)
  \le
  2^{-m}|B_m|e^{(\Delta_{q+1}+\varepsilon)m}.
\]
Combining this with the upper cardinality estimate would give the weaker
endpoint \(\Delta_{q+1}\).  Instead use the lower edge of the band:
\[
  d_m(x)^q\ge e^{(q-1)(\Delta_q-\varepsilon)m}d_m(x)
  \qquad (x\in B_m),
\]
so
\[
  p_m(B_m)
  \le
  2^{-m}e^{-(q-1)(\Delta_q-\varepsilon)m}S_q(m),
\]
which is the displayed upper visible-mass bound.

For the lower visible-mass bound, use the upper edge of the band and the
canonical-mass estimate:
\[
  d_m(x)^q\le e^{(q-1)(\Delta_{q+1}+\varepsilon)m}d_m(x)
  \qquad (x\in B_m).
\]
Therefore
\[
  S_q(m)(1-o(1))
  \le
  e^{(q-1)(\Delta_{q+1}+\varepsilon)m}
  \sum_{x\in B_m}d_m(x),
\]
and after multiplying by \(2^{-m}\) and taking logarithmic limits we obtain
the lower visible-mass bound.
\end{proof}

\begin{corollary}[Limiting microcanonical exponent window]
\label{cor:microcanonical-exponent-window}
With \(B_{m,q}(\varepsilon)\) as in \Cref{thm:microcanonical-bands}, fix an
integer \(q\ge 1\).  For every \(\eta>0\), there is
\(\varepsilon_0>0\) such that, whenever \(0<\varepsilon<\varepsilon_0\),
that band satisfies
\[
  P_q-q\Delta_{q+1}-\eta
  \le
  \liminf_{m\to\infty}\frac1m\log |B_{m,q}(\varepsilon)|,
\]
\[
  \limsup_{m\to\infty}\frac1m\log |B_{m,q}(\varepsilon)|
  \le
  P_q-q\Delta_q+\eta,
\]
\[
  P_q-\log 2-(q-1)\Delta_{q+1}-\eta
  \le
  \liminf_{m\to\infty}\frac1m\log p_m\bigl(B_{m,q}(\varepsilon)\bigr),
\]
and
\[
  \limsup_{m\to\infty}\frac1m\log p_m\bigl(B_{m,q}(\varepsilon)\bigr)
  \le
  P_q-\log 2-(q-1)\Delta_q+\eta.
\]
Thus the narrowing canonical bands have cardinality exponents confined
to \([P_q-q\Delta_{q+1},\,P_q-q\Delta_q]\) and visible-mass exponents
confined to
\([P_q-\log 2-(q-1)\Delta_{q+1},\,
  P_q-\log 2-(q-1)\Delta_q]\), up to an arbitrary error.
\end{corollary}

\begin{proof}
\Cref{thm:microcanonical-bands} gives the four finite-\(\varepsilon\) bounds
with error terms \(q\varepsilon\), \(q\varepsilon\),
\((q-1)\varepsilon\), and \((q-1)\varepsilon\), respectively.  Choose
\(\varepsilon_0>0\) so that \(q\varepsilon_0<\eta\) and
\((q-1)\varepsilon_0<\eta\).  Substitution gives the stated estimates.
\end{proof}

\begin{theorem}[Zero-temperature law for the largest fiber]
\label{thm:max-fiber}
\leanverified{paper_projection_max_fiber}
One has
\[
  \lim_{m\to\infty} D_m^{1/m}=\varphi^{1/2}.
\]
\end{theorem}

\begin{proof}
This is \Cref{lem:extremal-fiber-growth} rewritten exponentially.
\end{proof}

\begin{theorem}[Diagonal high-moment law]
\label{thm:diagonal-high-moments}
\leanverified{paper_projection_diagonal_high_moments}
For every integer sequence \(q(m)\ge1\) with \(q(m)\to\infty\),
\[
  \log S_{q(m)}(m)
  =
  \left(\frac12\log\varphi+o(1)\right)m\,q(m).
\]
\end{theorem}

\begin{proof}
For every \(m\) and \(q\ge1\),
\[
  D_m^q\le S_q(m)\le |X_m|D_m^q=F_{m+2}D_m^q.
\]
With \(q=q(m)\), divide logarithms by \(m q(m)\):
\[
  \frac1m\log D_m
  \le
  \frac{1}{m q(m)}\log S_{q(m)}(m)
  \le
  \frac1m\log D_m+\frac{1}{m q(m)}\log F_{m+2}.
\]
The first term tends to \(\frac12\log\varphi\) by \Cref{thm:max-fiber}; the
last term tends to \(0\) because \(q(m)\to\infty\) and
\(\log F_{m+2}=O(m)\).
\end{proof}

\begin{theorem}[Zero-temperature concentration of diverging tilts]
\label{thm:zero-temp-concentration}
\leanverified{paper_projection_zero_temp_concentration}
Let \(q(m)\ge1\) be any integer sequence with \(q(m)\to\infty\).  Then for
every \(\varepsilon>0\),
\[
  \mu_{m,q(m)}
  \left\{x:\left|\frac1m\log d_m(x)-\frac12\log\varphi\right|>\varepsilon\right\}
  \longrightarrow 0 .
\]
More precisely, for all sufficiently large \(m\) the upper tail is empty, and
the lower tail has mass at most
\[
  \exp\left(m\log\varphi-\frac{\varepsilon}{2}m\,q(m)+O(1)\right).
\]
\end{theorem}

\begin{proof}
Set \(\alpha_\ast=\frac12\log\varphi\).  By \Cref{thm:max-fiber}, for all
sufficiently large \(m\),
\[
  e^{(\alpha_\ast-\varepsilon/2)m}\le D_m
  \le e^{(\alpha_\ast+\varepsilon)m}.
\]
The second inequality implies that no \(x\in X_m\) can satisfy
\(\frac1m\log d_m(x)>\alpha_\ast+\varepsilon\).

For the lower tail
\[
  A_m^-:=\{x:d_m(x)<e^{(\alpha_\ast-\varepsilon)m}\},
\]
we have
\[
  \mu_{m,q(m)}(A_m^-)
  =
  \frac{\sum_{A_m^-}d_m(x)^{q(m)}}{S_{q(m)}(m)}
  \le
  \frac{F_{m+2}e^{q(m)(\alpha_\ast-\varepsilon)m}}
       {D_m^{q(m)}}.
\]
Using \(D_m\ge e^{(\alpha_\ast-\varepsilon/2)m}\) gives
\[
  \mu_{m,q(m)}(A_m^-)
  \le
  F_{m+2}e^{-\varepsilon m q(m)/2}
  =
  \exp\left(m\log\varphi-\frac{\varepsilon}{2}m q(m)+O(1)\right),
\]
which tends to \(0\).
\end{proof}

\begin{corollary}[Exchange of thermodynamic and zero-temperature limits]
\label{cor:double-limit}
\leanverified{paper_projection_double_limit}
The fixed-tilt thermodynamic limit and the zero-temperature limit exchange:
\[
  \lim_{q\to\infty}\lim_{m\to\infty}\frac{1}{mq}\log S_q(m)
  =
  \lim_{\substack{m\to\infty\\ q(m)\to\infty}}
  \frac{1}{m q(m)}\log S_{q(m)}(m)
  =
  \frac12\log\varphi,
\]
where the second limit means along every integer sequence \(q(m)\to\infty\).
Moreover, for every such sequence \(q(m)\), the \(q(m)\)-canonical thickness
converges in probability to \(\frac12\log\varphi\).
\end{corollary}

\begin{proof}
For fixed \(q\), \Cref{thm:all-q-transfer} gives
\[
  \lim_{m\to\infty}\frac1{mq}\log S_q(m)=\frac{P_q}{q}.
\]
Letting \(q\to\infty\) and applying \Cref{cor:renyi-endpoint} gives the first
displayed limit.  The second displayed limit is
\Cref{thm:diagonal-high-moments}.  The final assertion is
\Cref{thm:zero-temp-concentration}.
\end{proof}

\subsection{One-sided thick-fiber envelopes}

\begin{definition}[Exponential thick fibers]
\label{def:thick-fibers}
For \(\alpha\ge0\), set
\[
  \mathcal N_m(\alpha):=
  \#\left\{x\in X_m:\frac1m\log d_m(x)\ge\alpha\right\}.
\]
\end{definition}

\begin{corollary}[Discrete pressure upper envelope for thick fibers]
\label{cor:thick-fiber-envelope}
\leanverified{paper_projection_thick_fiber_envelope}
For every \(\alpha\ge 0\),
\[
  \limsup_{m\to\infty}\frac{1}{m}\log \mathcal N_m(\alpha)
  \le
  \min\left\{
    \log\varphi,\,
    \inf_{q\ge 1}\bigl(\log \lambda_q-q\alpha\bigr)
  \right\},
\]
where the infimum runs over positive integers \(q\).  Setting
\[
  \alpha_\ast:=\frac12\log\varphi,
\]
this barrier is sharp in the sense that
\[
  \alpha>\alpha_\ast
  \Longrightarrow
  \mathcal N_m(\alpha)=0 \text{ for all sufficiently large }m,
\]
whereas
\[
  0\le \alpha<\alpha_\ast
  \Longrightarrow
  \mathcal N_m(\alpha)\ge 1 \text{ for all sufficiently large }m.
\]
\end{corollary}

\begin{proof}
Fix \(\alpha\ge 0\), and let \(q\ge 1\) be an integer.  Every
\(x\in X_m\) counted by \(\mathcal N_m(\alpha)\) contributes at least
\(e^{q\alpha m}\) to \(S_q(m)\), so
\[
  e^{q\alpha m}\mathcal N_m(\alpha)
  \le
  \sum_{\substack{x\in X_m\\ d_m(x)\ge e^{\alpha m}}} d_m(x)^q
  \le
  S_q(m).
\]
Hence
\[
  \limsup_{m\to\infty}\frac{1}{m}\log \mathcal N_m(\alpha)
  \le
  \log\lambda_q-q\alpha
\]
by \Cref{thm:all-q-transfer}.  Taking the infimum over \(q\ge 1\) gives the
pressure bound.  Since \(\mathcal N_m(\alpha)\le \card{X_m}=F_{m+2}\), we also
have
\[
  \limsup_{m\to\infty}\frac{1}{m}\log \mathcal N_m(\alpha)\le \log\varphi.
\]

For the threshold statement, \Cref{thm:max-fiber} gives
\(
\frac{1}{m}\log D_m\to \alpha_\ast.
\)
If \(\alpha>\alpha_\ast\), then eventually \(D_m<e^{\alpha m}\), so
\(\mathcal N_m(\alpha)=0\).  If \(0\le \alpha<\alpha_\ast\), then eventually
\(D_m\ge e^{\alpha m}\), so at least one fiber is counted.
\end{proof}
