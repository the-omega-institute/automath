\subsection{Pressure geometry and the zero-temperature regime}

\begin{theorem}[Global pressure convexity]
\label{thm:global-pressure-convexity}
\leanverified{paper_projection_global_pressure_convexity}
The pressure sequence \((P_q)_{q\ge0}\) is convex on the nonnegative integer
line.  Equivalently, its piecewise-linear interpolation on \([0,\infty)\) is a
convex function.  In particular, for integers \(0\le a<b<c\),
\[
  P_b\le \frac{c-b}{c-a}P_a+\frac{b-a}{c-a}P_c .
\]
\end{theorem}

\begin{proof}
\Cref{cor:cross-q-logconvex} says that the adjacent slopes
\(\Delta_q=P_q-P_{q-1}\) are increasing.  If \(0\le a<b<c\), then
\[
  \frac{P_b-P_a}{b-a}
  =
  \frac1{b-a}\sum_{j=a+1}^{b}\Delta_j
  \le
  \frac1{c-b}\sum_{j=b+1}^{c}\Delta_j
  =
  \frac{P_c-P_b}{c-b}.
\]
Multiplying by \((b-a)(c-b)\) and rearranging gives
\[
  P_b\le \frac{c-b}{c-a}P_a+\frac{b-a}{c-a}P_c.
\]
This three-point inequality is the convexity of the sequence, and it is
equivalent to convexity of the piecewise-linear interpolation.
\end{proof}

\begin{lemma}[Extremal finite-window fiber growth]
\label{lem:extremal-fiber-growth}
The largest finite-window fiber satisfies
\[
  \lim_{m\to\infty}\frac1m\log D_m=\frac12\log\varphi .
\]
\end{lemma}

\begin{proof}
By \Cref{cor:coefficient-spectrum}, \(D_m\) is the largest coefficient of
\[
  \prod_{j=1}^m(1+z^{F_j}).
\]
For \(0\le n<F_{m+2}\), this coefficient is at most
\(R^\dagger(n)=R(n)+R(n-1)\), where \(R\) is the ordinary Fibonacci
representation function.  Koc\'abov\'a, Mas\'akov\'a, and Pelantov\'a
\cite{KocabovaMasakovaPelantova2005} determine the maximum of \(R(n)\) on each
interval between consecutive Fibonacci numbers.  In the indexing of the
present paper, their formula gives absolute constants \(0<c<C<\infty\) such
that, for every \(k\ge1\),
\[
  c\varphi^{k/2}
  \le
  \max_{F_k\le n<F_{k+1}}R(n)
  \le
  C\varphi^{k/2}.
\]
The upper bound over the intervals below \(F_{m+2}\) gives
\[
  \max_{0\le n<F_{m+2}} R(n)\le C\varphi^{m/2}
\]
after increasing \(C\).  Hence \(D_m\le 2C\varphi^{m/2}\), and
\[
  \limsup_{m\to\infty}\frac1m\log D_m\le \frac12\log\varphi .
\]

For the lower bound, apply the same estimate on the interval
\([F_m,F_{m+1})\).  It gives an \(n_m<F_{m+1}\) such that
\[
  R(n_m)\ge c\varphi^{m/2}
\]
after decreasing \(c\) if necessary.
Since \(n_m<F_{m+1}\), the finite coefficient equals the indexed partition
coefficient:
\[
  [z^{n_m}]\prod_{j=1}^m(1+z^{F_j})=R^\dagger(n_m)\ge R(n_m).
\]
Thus \(D_m\ge c\varphi^{m/2}\), giving the matching lower exponential rate.
\end{proof}

\begin{corollary}[Endpoint law for the partition constants]
\label{cor:renyi-endpoint}
\leanverified{paper_projection_renyi_endpoint}
One has
\[
  \lim_{q\to\infty}\lambda_q^{1/q}=\varphi^{1/2}.
\]
\end{corollary}

\begin{proof}
\Cref{lem:extremal-fiber-growth} gives
\[
  \frac1m\log D_m\longrightarrow \frac12\log\varphi .
\]
For every fixed \(q\ge1\),
\[
  D_m^q\le S_q(m)\le F_{m+2}D_m^q.
\]
Taking logarithms, dividing by \(m\), and letting \(m\to\infty\), using
\Cref{thm:all-q-transfer}, gives
\[
  q\cdot\frac12\log\varphi
  \le
  P_q
  \le
  \log\varphi+q\cdot\frac12\log\varphi .
\]
After division by \(q\), the squeeze tends to
\(\frac12\log\varphi\), which is the displayed assertion.
\end{proof}

\begin{lemma}[Adjacent moment ratios]
\label{lem:adjacent-moment-ratios}
For every fixed integer \(q\ge1\),
\[
  \frac{S_{q+1}(m)}{S_q(m)}
  =
  \exp\bigl((\Delta_{q+1}+o(1))m\bigr).
\]
Moreover, for \(q\ge2\),
\[
  \frac{S_{q-1}(m)}{S_q(m)}
  =
  \exp\bigl((-\Delta_q+o(1))m\bigr),
\]
and for \(q=1\),
\[
  \frac{S_0(m)}{S_1(m)}
  =
  \exp\bigl((-\Delta_1+o(1))m\bigr).
\]
\end{lemma}

\begin{proof}
For fixed \(j\ge1\), \Cref{thm:all-q-transfer} gives
\[
  S_j(m)=\exp\bigl((P_j+o(1))m\bigr).
\]
Taking quotients with \(j=q+1,q\) gives the first display, and taking
quotients with \(j=q-1,q\) gives the second when \(q\ge2\).  For \(q=1\),
\[
  S_0(m)=F_{m+2}=\exp\bigl((P_0+o(1))m\bigr),
  \qquad
  S_1(m)=2^m=\exp(P_1m),
\]
so \(S_0(m)/S_1(m)=\exp((P_0-P_1+o(1))m)\), which is the final display.
\end{proof}

\begin{corollary}[Pressure slopes]
\label{cor:pressure-slopes}
\leanverified{paper_projection_pressure_slopes}
\[
  \Delta_1\le\Delta_2\le\cdots
  \qquad\text{and}\qquad
  \lim_{q\to\infty}\Delta_q=\frac12\log\varphi .
\]
\end{corollary}

\begin{proof}
The monotonicity is \Cref{cor:cross-q-logconvex}.  Let
\(L:=\lim_{q\to\infty}\Delta_q\in(-\infty,\infty]\); the limit exists because
the sequence is increasing.  Since
\[
  \frac{P_q}{q}=\frac1q\sum_{j=1}^q\Delta_j,
\]
\Cref{cor:renyi-endpoint} gives
\[
  \frac1q\sum_{j=1}^q\Delta_j\longrightarrow\frac12\log\varphi .
\]
For an increasing sequence, the Cesaro means converge to the same limit as the
sequence, when the latter exists in the extended real line: if
\(L>\frac12\log\varphi+\eta\), then all sufficiently large \(\Delta_j\) exceed
that value and so do the tail Cesaro means; if
\(L<\frac12\log\varphi-\eta\), then all terms are eventually below that value.
Thus \(L=\frac12\log\varphi\).
\end{proof}

\begin{theorem}[Discrete Gibbs selection]
\label{thm:gibbs-selection}
\leanverified{paper_projection_gibbs_selection}
Fix an integer \(q\ge1\).  For every \(\varepsilon>0\),
\[
  \mu_{m,q}\left\{x:\frac1m\log d_m(x)\notin
  [\Delta_q-\varepsilon,\Delta_{q+1}+\varepsilon]\right\}
  \le \exp\bigl(-(\varepsilon+o(1))m\bigr).
\]
\end{theorem}

\begin{proof}
Write
\[
  L_m^-:=\left\{x:d_m(x)\le e^{(\Delta_q-\varepsilon)m}\right\},
  \qquad
  L_m^+:=\left\{x:d_m(x)\ge e^{(\Delta_{q+1}+\varepsilon)m}\right\}.
\]
For the lower tail, if \(q\ge2\), then
\[
  \mu_{m,q}(L_m^-)
  =
  \frac{\sum_{L_m^-}d_m(x)^q}{S_q(m)}
  \le
  e^{(\Delta_q-\varepsilon)m}\frac{S_{q-1}(m)}{S_q(m)}
  =
  \exp\bigl((-\varepsilon+o(1))m\bigr)
\]
by \Cref{lem:adjacent-moment-ratios}.  The same computation for \(q=1\) uses
\(d_m(x)\le e^{(\Delta_1-\varepsilon)m}\) and
\(\sum_{L_m^-}1\le S_0(m)\), together with the \(q=1\) part of
\Cref{lem:adjacent-moment-ratios}.

For the upper tail,
\[
  \mu_{m,q}(L_m^+)
  =
  \frac{\sum_{L_m^+}d_m(x)^q}{S_q(m)}
  \le
  e^{-(\Delta_{q+1}+\varepsilon)m}\frac{S_{q+1}(m)}{S_q(m)}
  =
  \exp\bigl((-\varepsilon+o(1))m\bigr).
\]
The union bound gives the displayed estimate; the factor \(2\) is absorbed in
the \(o(1)\) term.
\end{proof}

\begin{theorem}[Microcanonical band bounds]
\label{thm:microcanonical-bands}
\leanverified{paper_projection_microcanonical_band_bounds}
Fix an integer \(q\ge1\) and \(\varepsilon>0\), and set
\[
  B_{m,q}(\varepsilon)
  :=
  \left\{
    x\in X_m:
    \Delta_q-\varepsilon\le \frac1m\log d_m(x)
    \le \Delta_{q+1}+\varepsilon
  \right\}.
\]
Then
\[
  \mu_{m,q}\bigl(X_m\setminus B_{m,q}(\varepsilon)\bigr)
  \le \exp\bigl(-(\varepsilon+o(1))m\bigr).
\]
Moreover
\[
  P_q-q(\Delta_{q+1}+\varepsilon)
  \le
  \liminf_{m\to\infty}\frac1m\log |B_{m,q}(\varepsilon)|,
\]
\[
  \limsup_{m\to\infty}\frac1m\log |B_{m,q}(\varepsilon)|
  \le
  P_q-q(\Delta_q-\varepsilon),
\]
and, with \(p_m(A):=2^{-m}\sum_{x\in A}d_m(x)\),
\[
  P_q-\log 2-(q-1)(\Delta_{q+1}+\varepsilon)
  \le
  \liminf_{m\to\infty}\frac1m\log p_m(B_{m,q}(\varepsilon)),
\]
\[
  \limsup_{m\to\infty}\frac1m\log p_m(B_{m,q}(\varepsilon))
  \le
  P_q-\log 2-(q-1)(\Delta_q-\varepsilon).
\]
\end{theorem}

\begin{proof}
The canonical-mass estimate is exactly \Cref{thm:gibbs-selection}.

Let \(B_m=B_{m,q}(\varepsilon)\).  On \(B_m\),
\[
  e^{(\Delta_q-\varepsilon)m}\le d_m(x)\le
  e^{(\Delta_{q+1}+\varepsilon)m}.
\]
The upper cardinality bound follows from
\[
  |B_m|\,e^{q(\Delta_q-\varepsilon)m}
  \le
  \sum_{x\in B_m}d_m(x)^q
  \le S_q(m).
\]
For the lower cardinality bound, \Cref{thm:gibbs-selection} gives
\(\sum_{x\in B_m}d_m(x)^q=S_q(m)(1-o(1))\).  Hence
\[
  S_q(m)(1-o(1))
  \le
  |B_m|\,e^{q(\Delta_{q+1}+\varepsilon)m}.
\]
Using \(S_q(m)=\exp((P_q+o(1))m)\) gives the two cardinality estimates.

For visible mass, the upper bound is
\[
  p_m(B_m)
  =
  2^{-m}\sum_{x\in B_m}d_m(x)
  \le
  2^{-m}|B_m|e^{(\Delta_{q+1}+\varepsilon)m}.
\]
Combining this with the upper cardinality estimate would give the weaker
endpoint \(\Delta_{q+1}\).  Instead use the lower edge of the band:
\[
  d_m(x)^q\ge e^{(q-1)(\Delta_q-\varepsilon)m}d_m(x)
  \qquad (x\in B_m),
\]
so
\[
  p_m(B_m)
  \le
  2^{-m}e^{-(q-1)(\Delta_q-\varepsilon)m}S_q(m),
\]
which is the displayed upper visible-mass bound.

For the lower visible-mass bound, use the upper edge of the band and the
canonical-mass estimate:
\[
  d_m(x)^q\le e^{(q-1)(\Delta_{q+1}+\varepsilon)m}d_m(x)
  \qquad (x\in B_m).
\]
Therefore
\[
  S_q(m)(1-o(1))
  \le
  e^{(q-1)(\Delta_{q+1}+\varepsilon)m}
  \sum_{x\in B_m}d_m(x),
\]
and after multiplying by \(2^{-m}\) and taking logarithmic limits we obtain
the lower visible-mass bound.
\end{proof}

\begin{corollary}[Limiting microcanonical exponent window]
\label{cor:microcanonical-exponent-window}
With \(B_{m,q}(\varepsilon)\) as in \Cref{thm:microcanonical-bands}, fix an
integer \(q\ge 1\).  For every \(\eta>0\), there is
\(\varepsilon_0>0\) such that, whenever \(0<\varepsilon<\varepsilon_0\),
that band satisfies
\[
  P_q-q\Delta_{q+1}-\eta
  \le
  \liminf_{m\to\infty}\frac1m\log |B_{m,q}(\varepsilon)|,
\]
\[
  \limsup_{m\to\infty}\frac1m\log |B_{m,q}(\varepsilon)|
  \le
  P_q-q\Delta_q+\eta,
\]
\[
  P_q-\log 2-(q-1)\Delta_{q+1}-\eta
  \le
  \liminf_{m\to\infty}\frac1m\log p_m\bigl(B_{m,q}(\varepsilon)\bigr),
\]
and
\[
  \limsup_{m\to\infty}\frac1m\log p_m\bigl(B_{m,q}(\varepsilon)\bigr)
  \le
  P_q-\log 2-(q-1)\Delta_q+\eta.
\]
Thus the narrowing canonical bands have cardinality exponents confined
to \([P_q-q\Delta_{q+1},\,P_q-q\Delta_q]\) and visible-mass exponents
confined to
\([P_q-\log 2-(q-1)\Delta_{q+1},\,
  P_q-\log 2-(q-1)\Delta_q]\), up to an arbitrary error.
\end{corollary}

\begin{proof}
\Cref{thm:microcanonical-bands} gives the four finite-\(\varepsilon\) bounds
with error terms \(q\varepsilon\), \(q\varepsilon\),
\((q-1)\varepsilon\), and \((q-1)\varepsilon\), respectively.  Choose
\(\varepsilon_0>0\) so that \(q\varepsilon_0<\eta\) and
\((q-1)\varepsilon_0<\eta\).  Substitution gives the stated estimates.
\end{proof}

\begin{theorem}[Zero-temperature law for the largest fiber]
\label{thm:max-fiber}
\leanverified{paper_projection_max_fiber}
One has
\[
  \lim_{m\to\infty} D_m^{1/m}=\varphi^{1/2}.
\]
\end{theorem}

\begin{proof}
This is \Cref{lem:extremal-fiber-growth} rewritten exponentially.
\end{proof}

\begin{theorem}[Diagonal high-moment law]
\label{thm:diagonal-high-moments}
\leanverified{paper_projection_diagonal_high_moments}
For every integer sequence \(q(m)\ge1\) with \(q(m)\to\infty\),
\[
  \log S_{q(m)}(m)
  =
  \left(\frac12\log\varphi+o(1)\right)m\,q(m).
\]
\end{theorem}

\begin{proof}
For every \(m\) and \(q\ge1\),
\[
  D_m^q\le S_q(m)\le |X_m|D_m^q=F_{m+2}D_m^q.
\]
With \(q=q(m)\), divide logarithms by \(m q(m)\):
\[
  \frac1m\log D_m
  \le
  \frac{1}{m q(m)}\log S_{q(m)}(m)
  \le
  \frac1m\log D_m+\frac{1}{m q(m)}\log F_{m+2}.
\]
The first term tends to \(\frac12\log\varphi\) by \Cref{thm:max-fiber}; the
last term tends to \(0\) because \(q(m)\to\infty\) and
\(\log F_{m+2}=O(m)\).
\end{proof}

\begin{theorem}[Zero-temperature concentration of diverging tilts]
\label{thm:zero-temp-concentration}
\leanverified{paper_projection_zero_temp_concentration}
Let \(q(m)\ge1\) be any integer sequence with \(q(m)\to\infty\).  Then for
every \(\varepsilon>0\),
\[
  \mu_{m,q(m)}
  \left\{x:\left|\frac1m\log d_m(x)-\frac12\log\varphi\right|>\varepsilon\right\}
  \longrightarrow 0 .
\]
More precisely, for all sufficiently large \(m\) the upper tail is empty, and
the lower tail has mass at most
\[
  \exp\left(m\log\varphi-\frac{\varepsilon}{2}m\,q(m)+O(1)\right).
\]
\end{theorem}

\begin{proof}
Set \(\alpha_\ast=\frac12\log\varphi\).  By \Cref{thm:max-fiber}, for all
sufficiently large \(m\),
\[
  e^{(\alpha_\ast-\varepsilon/2)m}\le D_m
  \le e^{(\alpha_\ast+\varepsilon)m}.
\]
The second inequality implies that no \(x\in X_m\) can satisfy
\(\frac1m\log d_m(x)>\alpha_\ast+\varepsilon\).

For the lower tail
\[
  A_m^-:=\{x:d_m(x)<e^{(\alpha_\ast-\varepsilon)m}\},
\]
we have
\[
  \mu_{m,q(m)}(A_m^-)
  =
  \frac{\sum_{A_m^-}d_m(x)^{q(m)}}{S_{q(m)}(m)}
  \le
  \frac{F_{m+2}e^{q(m)(\alpha_\ast-\varepsilon)m}}
       {D_m^{q(m)}}.
\]
Using \(D_m\ge e^{(\alpha_\ast-\varepsilon/2)m}\) gives
\[
  \mu_{m,q(m)}(A_m^-)
  \le
  F_{m+2}e^{-\varepsilon m q(m)/2}
  =
  \exp\left(m\log\varphi-\frac{\varepsilon}{2}m q(m)+O(1)\right),
\]
which tends to \(0\).
\end{proof}

\begin{corollary}[Exchange of thermodynamic and zero-temperature limits]
\label{cor:double-limit}
\leanverified{paper_projection_double_limit}
The fixed-tilt thermodynamic limit and the zero-temperature limit exchange:
\[
  \lim_{q\to\infty}\lim_{m\to\infty}\frac{1}{mq}\log S_q(m)
  =
  \lim_{\substack{m\to\infty\\ q(m)\to\infty}}
  \frac{1}{m q(m)}\log S_{q(m)}(m)
  =
  \frac12\log\varphi,
\]
where the second limit means along every integer sequence \(q(m)\to\infty\).
Moreover, for every such sequence \(q(m)\), the \(q(m)\)-canonical thickness
converges in probability to \(\frac12\log\varphi\).
\end{corollary}

\begin{proof}
For fixed \(q\), \Cref{thm:all-q-transfer} gives
\[
  \lim_{m\to\infty}\frac1{mq}\log S_q(m)=\frac{P_q}{q}.
\]
Letting \(q\to\infty\) and applying \Cref{cor:renyi-endpoint} gives the first
displayed limit.  The second displayed limit is
\Cref{thm:diagonal-high-moments}.  The final assertion is
\Cref{thm:zero-temp-concentration}.
\end{proof}
