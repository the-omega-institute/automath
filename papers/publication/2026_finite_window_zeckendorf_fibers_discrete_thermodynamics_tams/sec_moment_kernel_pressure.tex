\section{Integer pressure and a diagonal high-tilt limit}
\label{sec:integer-pressure}

For notational completeness, set
\[
 S_0(m):=|X_m|=F_{m+2},\qquad
 P_0:=\log\varphi,
\]
and, for each positive integer \(q\), set
\[
 P_q:=\lim_{m\to\infty}\frac1m\log S_q(m)=\log\lambda_q.
\]
The existence of the latter limit follows from
\Cref{thm:all-q-transfer}.  Thus \(q\) is an integer moment order in this
section; real tilts continue to be denoted by \(t\).

\begin{proposition}[Convexity of the integer pressure]
\label{prop:integer-pressure-convexity}
For every integer \(q\ge1\),
\[
  2P_q\le P_{q-1}+P_{q+1}.
\]
Equivalently, the adjacent differences \(P_q-P_{q-1}\) are nondecreasing.
\end{proposition}

\begin{proof}
For each fixed \(m\), Cauchy--Schwarz gives
\[
 \left(\sum_{x\in X_m}d_m(x)^q\right)^2
 \le
 \left(\sum_{x\in X_m}d_m(x)^{q-1}\right)
 \left(\sum_{x\in X_m}d_m(x)^{q+1}\right).
\]
This is \(S_q(m)^2\le S_{q-1}(m)S_{q+1}(m)\).  Taking logarithms,
dividing by \(m\), and passing to the established limits proves the claim.
\end{proof}

\begin{corollary}[Sanna's endpoint law: an alternative finite-window proof]
\label{cor:renyi-endpoint}
One has
\[
  \lim_{q\to\infty}\lambda_q^{1/q}=\varphi^{1/2}.
\]
This result is due to Sanna \cite[Thm.~1.2]{Sanna2025}; the proof below is an
alternative deduction from the exact finite-window maximum.
\end{corollary}

\begin{proof}
By \Cref{thm:unconditional-extremal-fibers},
\[
 \lim_{m\to\infty}\frac1m\log D_m=\frac12\log\varphi.
\]
For every fixed positive integer \(q\),
\[
 D_m^q\le S_q(m)\le F_{m+2}D_m^q.
\]
After taking logarithms, dividing by \(m\), and letting \(m\to\infty\),
\Cref{thm:all-q-transfer} gives
\[
 \frac q2\log\varphi
 \le \log\lambda_q
 \le \log\varphi+\frac q2\log\varphi.
\]
Division by \(q\), followed by \(q\to\infty\), proves the displayed limit.
\end{proof}

\begin{theorem}[Diagonal high-tilt limit]
\label{thm:diagonal-high-moments}
For every sequence of positive integers \(q_m\to\infty\),
\[
 \frac{1}{m q_m}\log S_{q_m}(m)
 \longrightarrow\frac12\log\varphi.
\]
More generally, for every real sequence \(t_m\ge1\) with
\(t_m\to\infty\),
\[
 \frac{1}{m t_m}\log S_{t_m}(m)
 \longrightarrow\frac12\log\varphi.
\]
\end{theorem}

\begin{proof}
For every real \(t\ge1\),
\[
 D_m^t\le S_t(m)\le |X_m|D_m^t=F_{m+2}D_m^t.
\]
Putting \(t=t_m\), taking logarithms, and dividing by \(mt_m\) gives
\[
 \frac1m\log D_m
 \le \frac{1}{mt_m}\log S_{t_m}(m)
 \le \frac1m\log D_m+\frac{\log F_{m+2}}{mt_m}.
\]
The first term tends to \(\frac12\log\varphi\), and the last term tends to
zero because \(\log F_{m+2}=O(m)\) and \(t_m\to\infty\).  The integer
statement is the specialization \(t_m=q_m\).
\end{proof}
