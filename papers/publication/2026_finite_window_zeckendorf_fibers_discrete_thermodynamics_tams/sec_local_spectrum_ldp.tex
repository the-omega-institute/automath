\begin{theorem}[Full large deviations across the freezing corner]
\label{thm:full-fiber-ldp}
Let \(\mathbf X_m\) be uniformly distributed on \(X_m\), and set
\[
 \begin{gathered}
 y_m(x):=\frac1m\log d_m(x),\qquad
 Y_m:=y_m(\mathbf X_m),\\
 t_c:=-\sigma_0,\qquad v_c:=P'_+(t_c),
 \qquad \alpha_+:=\frac12\log\varphi.
 \end{gathered}
\]
Then \((Y_m)\) satisfies a full large deviation principle with speed \(m\)
and good rate function
\begin{equation}
 I(\alpha)=\sup_{t\in\mathbb R}
 \{t\alpha-P(t)+\log\varphi\}.
 \label{eq:full-fiber-ldp-rate}
\end{equation}
Its effective domain is \([0,\alpha_+]\).  On the coexistence interval,
\begin{equation}
 I(\alpha)=\log\varphi-\sigma_0\alpha
 \qquad(0\le\alpha\le v_c).
 \label{eq:frozen-affine-rate}
\end{equation}
More precisely, for every \(0<\alpha<v_c\),
\begin{equation}
 \begin{aligned}
 &\lim_{\varepsilon\downarrow0}\liminf_{m\to\infty}
 \frac1m\log\Pr\{|Y_m-\alpha|<\varepsilon\}\\
 &\quad=
 \lim_{\varepsilon\downarrow0}\limsup_{m\to\infty}
 \frac1m\log\Pr\{|Y_m-\alpha|<\varepsilon\}\\
 &\quad=-\bigl(\log\varphi-\sigma_0\alpha\bigr).
 \end{aligned}
 \label{eq:frozen-local-large-deviation}
\end{equation}
\end{theorem}

\begin{proof}
For \(\alpha\in\mathbb R\) and \(\varepsilon>0\), write
\[
 A_m(\alpha,\varepsilon)
 :=\card{\left\{x\in X_m:
 \left|m^{-1}\log d_m(x)-\alpha\right|<\varepsilon\right\}}.
\]
We first establish the lower bound at the nonexposed slopes.  Fix
\(0<\alpha<v_c\), then choose \(t>t_c\) and put \(v=P'(t)\).  Strict
convexity gives \(v>v_c>\alpha\), so
\(\rho:=\alpha/v\) belongs to \((0,1)\).  Define the tilted law on \(X_n\) by
\[
 \nu_{n,t}(x):=\frac{d_n(x)^t}{S_t(n)}.
\]
For sufficiently small \(s>0\), the exponential Markov inequality and
\Cref{thm:all-real-pressure} give
\[
 \limsup_{n\to\infty}\frac1n
 \log\nu_{n,t}\{y_n\ge v+\delta\}
 \le P(t+s)-P(t)-s(v+\delta)<0.
\]
The corresponding estimate with \(s<0\) gives the same conclusion for
\(y_n\le v-\delta\).  Consequently,
\[
 \sum_{\substack{x\in X_n\\|y_n(x)-v|<\delta}}d_n(x)^t
 =\exp\{n(P(t)+o(1))\}.
\]
Every summand on the left is at most
\(\exp\{n(tv+|t|\delta)\}\), and hence
\begin{equation}
 \liminf_{n\to\infty}\frac1n\log A_n(v,\delta)
 \ge P(t)-tv-|t|\delta.
 \label{eq:exposed-source-count}
\end{equation}

We now transfer these source states to a larger layer without changing their
multiplicities.  The proof of \Cref{thm:fiber-partition-interval} gives the
explicit bijection
\[
 \chi_n(x)=
 \begin{cases}
  F_{n+2}+V_n(x),&0\le V_n(x)\le F_{n+1}-2,\\
  V_n(x),&F_{n+1}-1\le V_n(x)<F_{n+2},
 \end{cases}
\]
from \(X_n\) onto the integer points of
\(\mathcal I_n\sqcup\mathcal I_{n+1}\), with
\(d_n(x)=R(\chi_n(x))\).  Take
\(\delta<v/2\), so their multiplicities exceed two for all sufficiently
large \(n\).  The uniform layer-count lemma shows that each Weinstein
generating orbit supplies at most four source states to these two layers.
The specialized formulas in the proof of
\Cref{prop:weighted-generator-renewal} give
\[
 L([a]g)=L(g)+2a,\qquad
 L([a]\tau g)=L(g)+1+2a,
\]
and the two sigma branches increase these lengths by one.  Hence every orbit
meeting \(\mathcal I_n\sqcup\mathcal I_{n+1}\) has generator length at most
\(n+1\).
Keeping the generating number of every orbit which occurs therefore gives a
set \(\mathcal G_n\) such that
\begin{equation}
 \card{\mathcal G_n}\ge\frac14 A_n(v,\delta),
 \qquad L(g)\le n+1\quad(g\in\mathcal G_n),
 \label{eq:source-generating-orbits}
\end{equation}
and \(n^{-1}\log R(g)\in(v-\delta,v+\delta)\), since the orbit action
preserves \(R\).  Distinct elements of
\(\mathcal G_n\) generate disjoint orbits.

Let \(n_m=\lfloor\rho m\rfloor\).  Since
\(\rho m-1<n_m\le\rho m\) and \(L(g)\le n_m+1\), the inequality
\(L(g)<m\) holds uniformly for \(g\in\mathcal G_{n_m}\) once
\((1-\rho)m>2\).  Thus \(m-L(g)\) is a positive integer.  Its parity gives
exactly one of the two cases
\[
 z_m(g):=
 \begin{cases}
 [a]g,&m-L(g)=2a,\\
 [a]\tau g,&m-L(g)=2a+1.
 \end{cases}
\]
where \(a=(m-L(g))/2\) or \(a=(m-L(g)-1)/2\), respectively, is a
nonnegative integer.  They give \(L(z_m(g))=m\) and
\(R(z_m(g))=R(g)>1\).  Thus
\(z_m(g)\in\mathcal I_m\), and \(\chi_m^{-1}\) turns these into distinct
states of \(X_m\).  Moreover,
\[
 \left|m^{-1}\log R(z_m(g))-\alpha\right|
 \le \frac{n_m}{m}\delta
   +v\left|\frac{n_m}{m}-\rho\right|
 \le \rho\delta+\frac{\delta+v}{m}.
\]
Combining this injection with
\eqref{eq:exposed-source-count}--\eqref{eq:source-generating-orbits} yields,
for every \(\varepsilon>\rho\delta\),
\begin{equation}
 \liminf_{m\to\infty}\frac1m\log A_m(\alpha,\varepsilon)
 \ge \rho\bigl(P(t)-tv-|t|\delta\bigr).
 \label{eq:orbit-filling-lower-bound}
\end{equation}
First let \(\delta\downarrow0\), and then let \(t\downarrow t_c\).  Since
\(P(t)\to0\), \(P'(t)=v\to v_c\), and
\[
 \rho(P(t)-tv)=\alpha\left(\frac{P(t)}v-t\right)
 \longrightarrow-t_c\alpha=\sigma_0\alpha,
\]
we obtain
\begin{equation}
 \lim_{\varepsilon\downarrow0}\liminf_{m\to\infty}
 \frac1m\log A_m(\alpha,\varepsilon)
 \ge\sigma_0\alpha.
 \label{eq:frozen-count-lower}
\end{equation}

The matching upper bound uses the critical pressure directly.  On the set
counted by \(A_m(\alpha,\varepsilon)\),
\(d_m(x)^{-\sigma_0}\ge
\exp\{-\sigma_0m(\alpha+\varepsilon)\}\).  Therefore
\[
 A_m(\alpha,\varepsilon)
 e^{-\sigma_0m(\alpha+\varepsilon)}
 \le S_{-\sigma_0}(m)=e^{o(m)},
\]
which, together with \eqref{eq:frozen-count-lower}, proves the corresponding
counting limit.  Since \(\card{X_m}=F_{m+2}\) and
\(m^{-1}\log F_{m+2}\to\log\varphi\), this is
\eqref{eq:frozen-local-large-deviation}.

We next identify the convex dual.  If \(0\le\alpha\le v_c\), then
\(P(t)=0\) for \(t\le t_c\), while convexity and the right derivative at
\(t_c\) give \(P(t)\ge v_c(t-t_c)\) for \(t\ge t_c\).  It follows that
\(t\alpha-P(t)\le t_c\alpha\) for every \(t\), with equality at
\(t=t_c\), proving \eqref{eq:frozen-affine-rate}.

The exact maximum theorem gives
\[
 \max_{x\in X_m}m^{-1}\log d_m(x)\longrightarrow\alpha_+.
\]
For \(t>0\), the bounds
\(D_m^t\le S_t(m)\le F_{m+2}D_m^t\) imply
\[
 \alpha_+\le\frac{P(t)}t
 \le\alpha_++\frac{\log\varphi}{t}.
\]
By convexity, (P') is increasing and therefore has a limit in
\((0,+\infty]\).  The identity
\[
 \frac{P(t)}t=\frac{P(0)}t+\frac1t\int_0^t P'(u)\,du
\]
and Ces\`aro convergence show that this limit is \(\alpha_+\).  Thus
\(P'(t)\uparrow\alpha_+\) as \(t\to\infty\).  Strict convexity and
analyticity on \((t_c,\infty)\) now show that every
\(\alpha\in(v_c,\alpha_+)\) is exposed by a unique \(t>t_c\).  The tilted
estimate above with \(n=m\) supplies
\begin{equation}
 \lim_{\varepsilon\downarrow0}\liminf_{m\to\infty}
 \frac1m\log\Pr\{|Y_m-\alpha|<\varepsilon\}
 \ge P(t)-t\alpha-\log\varphi=-I(\alpha).
 \label{eq:exposed-local-large-deviation}
\end{equation}

Finally, the logarithmic moment generating functions under the uniform law
converge on all of \(\mathbb R\) to
\[
 \Lambda(t)=P(t)-\log\varphi.
\]
The variables are eventually supported in a fixed compact interval, so they
are exponentially tight.  The G\"artner--Ellis upper bound
\cite[Theorem~2.3.6(a)]{DemboZeitouni2010} therefore gives the
large-deviation upper bound for all closed sets.

It remains to justify the effective domain and endpoint lower bounds.  If
\(\alpha<0\), letting \(t\to-\infty\) in
\eqref{eq:full-fiber-ldp-rate} and using \(P(t)=0\) gives
\(I(\alpha)=+\infty\).  If \(\alpha>\alpha_+\), the limit
\(P(t)/t\to\alpha_+\) gives the same conclusion as \(t\to\infty\).
Conversely, if \(0\le\alpha\le\alpha_+\), then
\(t\alpha-P(t)\) is bounded above on \(( -\infty,t_c]\), on the compact
interval \([t_c,0]\), and on \([0,\infty)\), where
\(P(t)\ge t\alpha_+\).  Thus the effective domain is exactly
\([0,\alpha_+]\).  The rate is a lower-semicontinuous convex function, and
its finiteness on this closed interval implies continuity relative to the
interval, including at the endpoints.

We make the boundary passage in the relative topology explicit.  Let
\(G\subset\mathbb R\) be open and meet \([0,\alpha_+]\), and fix
\(a\in G\cap[0,\alpha_+]\).  If \(a\in(0,\alpha_+)\setminus\{v_c\}\), the
local lower bounds above apply directly.  If \(a=v_c\), choose
\(a_j\to v_c\) from either adjacent open interval; if \(a=\alpha_+\), choose
exposed points \(a_j\uparrow\alpha_+\).  Relative continuity gives
\(I(a_j)\to I(a)\), and openness puts \(a_j\) together with a sufficiently
small neighborhood inside \(G\) for all large \(j\).  Applying the local
bound first and then sending \(j\to\infty\) gives the lower bound at \(a\).
At \(a=0\), the two multiplicity-one states give the same conclusion
directly.  Taking the supremum over such \(a\), equivalently the negative
infimum of \(I\) over \(G\), proves the open-set lower bound.  The effective
domain is compact and \(I\) is lower semicontinuous, so \(I\) is good and the
full large deviation principle follows.
\end{proof}

\subsection{Microcanonical phase separation}

The renewal description also retains the generating layer, rather than only
the multiplicity.  Let
\[
 \mathcal A:=\{p/q:0<p<q,\ (p,q)=1\},
 \qquad \pi(p/q):=q^{-\sigma _0}.
\]
The critical identity \eqref{eq:weighted-renewal-at-one} gives
\(\sum_{\mathcal A}\pi=1\).  Let \((C_i,H_i)_{i\ge1}\) be independent with
\[
 \Pr\{(C_i,H_i)=(c(p/q),\log q)\}=\pi(p/q),
\]
put \(S_n=\sum_{i=1}^nC_i\), \(T_n=\sum_{i=1}^nH_i\), and define the marked
renewal measure
\[
 \mathcal U:=\sum_{n\ge1}\mathcal L(S_n,T_n).
\]
The moment estimates in the proof of \Cref{thm:all-real-pressure} show that
\[
 \mu_C:=\mathbb E C_1<\infty,
 \qquad \mu_H:=\mathbb E H_1<\infty,
 \qquad \frac{\mu_H}{\mu_C}=v_c.
\]

We shall use the following marked form of the already proved weighted renewal
identity.  It is recorded explicitly to retain the generating cost in the
conditional argument.  For $x\in X_m$ with \(d_m(x)>2\), let \(g_m(x)\) be
the unique generating number of the Weinstein orbit containing \(\chi_m(x)\).
Define
\[
 K_m(c):=
 \begin{cases}
  4,&c\le m-1,\\
  3,&c=m,\\
  1,&c=m+1,\\
  0,&c\ge m+2.
 \end{cases}
\]
For every Borel set \(B\subset\mathbb R\times(\log2,\infty)\),
\begin{equation}
 \card{\{x\in X_m:(L(g_m(x)),\log d_m(x))\in B\}}
 =\int_B K_m(s)e^{\sigma_0t}\,\mathcal U(ds,dt).
 \label{eq:marked-microcanonical-count}
\end{equation}
Indeed, for a generating number \(g\) of cost \(c=L(g)\), the four branches
and their layer indices are
\[
 \begin{array}{c|c}
 {}[a]g&c+2a\\
 {}[a]\tau g&c+1+2a\\
 \sigma([a]g)&c+1+2a\\
 \sigma([a]\tau g)&c+2+2a.
 \end{array}
\]
The free action makes the middle two points distinct.  Thus this orbit
contributes zero, one, or two points to \(\mathcal I_j\), according as
\(j<c\), \(j=c\), or \(j>c\).  Summing over the two disjoint layers
\(\mathcal I_m\sqcup\mathcal I_{m+1}\) gives exactly \(K_m(c)\).  The two
unit-multiplicity endpoints correspond to the empty word and are excluded by
the condition \(t>\log2\).

Under Weinstein's free-monoid bijection, a nonempty word
\(w=(p_1/q_1)\times\cdots\times(p_n/q_n)\) has
\[
 L(g)=S_n,\qquad \log R(g)=T_n,
 \qquad \pi(w)=e^{-\sigma_0T_n}.
\]
Multiplication of its renewal mass by \(e^{\sigma_0T_n}\), followed by the
orbit count \(K_m(S_n)\), therefore counts every corresponding finite-window
state once.  This proves \eqref{eq:marked-microcanonical-count}.

For the endpoint tilt, define the finite-window Gibbs measure
\[
 \widehat Q_m(x):=\frac{d_m(x)^{-\sigma_0}}{S_{-\sigma_0}(m)}
 \qquad(x\in X_m),
\]
and put \(\ell_m(x)=L(g_m(x))\) when \(d_m(x)>2\), while
\(\ell_m(x)=0\) when \(d_m(x)\le2\).

\begin{theorem}[Critical finite-size and coexistence law]
\label{thm:critical-finite-window-gibbs}
Let
\[
 \mu_C=
 \sum_{q\ge2}\sum_{\substack{1\le p<q\\(p,q)=1}}
 c(p/q)q^{-\sigma_0}.
\]
Then
\begin{equation}
 S_{-\sigma_0}(m)\sim\frac{4m}{\mu_C}.
 \label{eq:critical-linear-finite-size}
\end{equation}
If \(\widehat{\mathbf X}_m\) has law \(\widehat Q_m\) and
\(U\) is uniform on \([0,1]\), then
\begin{equation}
 \left(
  \frac1m\log d_m(\widehat{\mathbf X}_m),
  \frac1m\ell_m(\widehat{\mathbf X}_m)
 \right)
 \xrightarrow[m\to\infty]{\mathrm d}(v_cU,U).
 \label{eq:critical-uniform-coexistence}
\end{equation}
\end{theorem}

\begin{proof}
At \(s=\sigma_0\), the coefficients \(b_j(\sigma_0)\) form a probability
distribution on the positive integers by
\eqref{eq:weighted-renewal-at-one}.  Its span is one because its support
contains both \(3\) and \(5\), and its mean is \(\mu_C<\infty\).  The
arithmetic renewal theorem therefore gives
\[
 u_j(\sigma_0)\longrightarrow\frac1{\mu_C}.
\]
Substitution in the exact identity
\eqref{eq:weighted-window-renewal}, followed by Ces\`aro summation, proves
\eqref{eq:critical-linear-finite-size}.

We use the critical letter law \(\pi(p/q)=q^{-\sigma_0}\) and the marked
renewal variables \((S_n,T_n)\) defined above
\eqref{eq:marked-microcanonical-count}.  For every bounded continuous
function \(f\) on \([0,1]\), the arithmetic renewal limit and a Riemann-sum
argument give
\[
 \frac1m\sum_{j=0}^{m-1}u_j(\sigma_0)f(j/m)
 \longrightarrow\frac1{\mu_C}\int_0^1 f(y)\,dy.
\]
The terms at \(j=m,m+1\) are bounded, while
\(K_m(j)=4\) for \(j\le m-1\).  The total critical weight of the states
with multiplicity at most two is also bounded uniformly in \(m\).
Multiplying \eqref{eq:marked-microcanonical-count} by
\(e^{-\sigma_0t}\), and then using
\eqref{eq:critical-linear-finite-size}, proves
\begin{equation}
 \frac{\ell_m(\widehat{\mathbf X}_m)}m
 \xrightarrow{\mathrm d}U.
 \label{eq:critical-uniform-layer}
\end{equation}

It remains to retain the reward mark.  The strong law gives
\(T_n/S_n\to\mu_H/\mu_C=v_c\) almost surely.  For \(\delta>0\), let
\[
 A_m=\mathcal U\{0<S\le m+1,\ |T/S-v_c|>\delta\}.
\]
Every renewal step has cost at least three.  Hence, on each sample path, the
number of renewal epochs with \(S_n\le m+1\), divided by \(m\), is bounded;
the corresponding number of exceptional epochs divided by \(m\) tends to
zero by the strong law.  Dominated convergence yields \(A_m=o(m)\).
Since \(K_m\le4\), the marked identity and
\eqref{eq:critical-linear-finite-size} now imply
\[
 \frac1m\log d_m(\widehat{\mathbf X}_m)
 -v_c\frac1m\ell_m(\widehat{\mathbf X}_m)
 \xrightarrow{\Pr}0.
\]
Together with \eqref{eq:critical-uniform-layer}, this proves
\eqref{eq:critical-uniform-coexistence}.
\end{proof}

\begin{proposition}[Finite-prime-support coefficient interface]
\label{prop:prime-support-interface}
Fix distinct primes \(\mathcal P=\{p_1,\ldots,p_d\}\).  For
\(\mathbf a=(a_1,\ldots,a_d)\in\mathbb N^d\setminus\{\mathbf0\}\), put
\[
 q_{\mathbf a}:=\prod_{i=1}^d p_i^{a_i},
 \qquad
 A_{\mathbf a}(u):=
 \sum_{\substack{1\le h<q_{\mathbf a}\\(h,q_{\mathbf a})=1}}
 u^{c(h/q_{\mathbf a})},
\]
and define the formal series
\[
 A_{\mathcal P}(\mathbf z,u)
 :=\sum_{\mathbf a\ne\mathbf0}
 A_{\mathbf a}(u)\mathbf z^{\mathbf a}.
\]
If
\[
 U_j(q_{\mathbf n})
 :=\card{\{g:\text{\(g\) is a Weinstein generator, }
 R(g)=q_{\mathbf n},\ L(g)=j\}},
\]
with the empty-word convention \(U_0(1)=1\), then
\begin{equation}
 \sum_{\mathbf n\in\mathbb N^d}\sum_{j\ge0}
 U_j(q_{\mathbf n})\mathbf z^{\mathbf n}u^j
 =\frac1{1-A_{\mathcal P}(\mathbf z,u)}.
 \label{eq:prime-support-joint-renewal}
\end{equation}
For \(q_{\mathbf n}>2\), the exact finite-window count is
\begin{equation}
 C_m(q_{\mathbf n})
 =\sum_{j\ge0}K_m(j)U_j(q_{\mathbf n}).
 \label{eq:prime-support-window-interface}
\end{equation}
After the cost mark is removed, the series is rational:
\begin{equation}
 \sum_{\mathbf n\in\mathbb N^d}
 \left(\sum_{j\ge0}U_j(q_{\mathbf n})\right)\mathbf z^{\mathbf n}
 =\frac1{\displaystyle
 2-\prod_{i=1}^d\frac{1-z_i}{1-p_i z_i}}.
 \label{eq:prime-support-unmarked-rational}
\end{equation}
\end{proposition}

\begin{proof}
Under Weinstein's free-monoid parameterization, a letter with denominator
\(q_{\mathbf a}\) contributes exponent vector \(\mathbf a\), cost
\(c(h/q_{\mathbf a})\), and multiplier \(q_{\mathbf a}\).  Unique ordered
factorization into letters gives \eqref{eq:prime-support-joint-renewal} by the
formal sequence construction.  The four orbit branches counted in
\eqref{eq:marked-microcanonical-count} depend only on the generator cost and
give \eqref{eq:prime-support-window-interface}.  Finally,
\[
 \sum_{a\ge0}\phi_{\rm E}(p_i^a)z_i^a
 =\frac{1-z_i}{1-p_i z_i}.
\]
Multiplying over the primes shows that
\(A_{\mathcal P}(\mathbf z,1)\) is the displayed product minus one, which
proves \eqref{eq:prime-support-unmarked-rational}.
\end{proof}

\begin{remark}[Located active-cutoff obstruction]
\label{rem:prime-support-local-open-interface}
Equations \eqref{eq:prime-support-joint-renewal} and
\eqref{eq:prime-support-window-interface} locate, but do not solve, the
active arithmetic interface.  The unmarked rational function is in the scope
of smooth-point multivariate coefficient methods such as
Pemantle--Wilson \cite{PemantleWilson2002}; the nearest ordered-factorization
comparators are Lau \cite{Lau2001} and Hwang--Janson
\cite{HwangJanson2011}.  Those results do not retain the continued-fraction
cost at an exponent-conditioned lattice point with an error uniform in the
direction, and they do not provide the uniform integrability needed for the
unbounded cost.  Accordingly, no directional quenched law is asserted here,
even below the active cutoff.

Already for \(\mathcal P=\{2\}\), the normalized letter law used in the
dyadic renewal assigns mass \(3^{-a}\) to each fraction with denominator
\(2^a\).  Since
\(c(1/2^a)=2^{a+1}-1\), its cost \(C\) satisfies
\[
 \mathbb E C^2
 \ge\sum_{a\ge1}3^{-a}(2^{a+1}-1)^2=+\infty,
 \qquad
 \mathbb E e^{\theta C}=+\infty\quad(\theta>0).
\]
Its first moment is finite: the Panov--Liehl mean estimate used above makes
the contribution at exponent \(a\) at most
\(O(a^2(2/3)^a)\).  Thus neither a Gaussian local theorem nor a two-sided
Cram\'er tilt is available at \(u=1\).  A sharp asymptotic for
\(C_m(q_{\mathbf n})\) with the cutoff \(j=m+O(1)\) requires a lattice
stable or semistable local-renewal theorem uniform under multivariate exponent
conditioning, followed by the nonsmooth weights \(K_m(m)=3\) and
\(K_m(m+1)=1\).  No such theorem is proved or invoked here; this is the exact
remaining open input.
\end{remark}
