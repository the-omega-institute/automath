\section{Contextual Analytic Comparators}
\label{sec:analytic}

This appendix is purely contextual. The finite-state results in the
paper use only transfer-matrix counting and explicit prime-counting
bounds. Two classical analytic contrasts are worth recording: prime
support in Euler products often forces natural boundaries, and scalar
finite-matrix determinant models become \(2\pi i\)-periodic after the
substitution \(z=e^{-s}\). Neither statement is used in
\Cref{thm:effective-fixed-dfa-prime-obstruction,thm:multicomponent-finite-budget}.

\begin{theorem}[Classical: prime-supported Euler products have a natural boundary]
  \label{thm:euler-product-natural-boundary}
  Let \((b_n)_{n\ge1}\subseteq\ZZ_{\ge0}\), assume that \(b_p>0\) for
  infinitely many primes \(p\), and suppose that
  \[
    Z_b(z):=\prod_{n\ge1}(1-z^n)^{-b_n}
  \]
  converges to a holomorphic function on \(\abs z<1\). Then
  \(\abs z=1\) is a natural boundary for \(Z_b\).
  \leanverified{paper_prime_languages_euler_product_natural_boundary}
\end{theorem}

\begin{proof}
  This is the prime-supported case of Estermann's natural-boundary
  theorem for Euler products with non-negative integral exponents
  \cite{Estermann1928}; see also the P\'olya--Carlson formulation
  \cite{Polya1923,Carlson1921}. Under the displayed hypotheses,
  infinitely many prime-indexed primitive factors occur with positive
  exponent, so Estermann's theorem applies directly.
\end{proof}

\begin{proposition}[Classical: periodic scalar determinant models after \(z=e^{-s}\)]
  \label{prop:finite-zeta-imaginary-periodicity}
  For every finite square complex matrix \(M\), the scalar determinant
  model
  \[
    F_M(s):=\det(I-(e^{-s})M)^{-1}
  \]
  is \(2\pi i\)-periodic wherever it is holomorphic. Therefore it
  cannot agree on a non-empty right-half-plane open set with a
  nonconstant ordinary Dirichlet series such as the Riemann
  \(\zeta\)-function.
\end{proposition}

\begin{proof}
  The periodicity is the identity
  \[
    F_M(s+2\pi i)
    =\det(I-(e^{-(s+2\pi i)})M)^{-1}
    =F_M(s).
  \]
  If \(F(s)=\sum_{n\ge1}a_nn^{-s}\) is an ordinary Dirichlet series with
  some non-zero coefficient at \(n\ge2\), then
  \[
    G(s):=F(s+2\pi i)-F(s)
      =\sum_{n\ge1}a_n(e^{-2\pi i\log n}-1)n^{-s}
  \]
  has a non-zero first coefficient, where \(\log\) is the real
  logarithm. The usual uniqueness lemma for ordinary Dirichlet series,
  obtained by multiplying by the first index to the power \(\sigma\)
  and letting \(\sigma\to+\infty\), gives \(G(\sigma)\ne0\) for all
  sufficiently large real \(\sigma\). Thus \(F\) is not
  \(2\pi i\)-periodic, and the identity theorem prevents agreement with
  a scalar determinant model on such an open set.
\end{proof}
