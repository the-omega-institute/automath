\section{Finite-size crossover and the off-critical regimes}
\label{sec:finite-size-crossover}

The proof uses the exact orbit weights before taking a renewal limit.  This
is important in the critical window: normalizing the letter masses first
produces both a probability law and a separate fugacity, and the latter is
what tilts the macroscopic generator location.

\begin{lemma}[Near-critical triangular letter array]
\label{lem:near-critical-letter-array}
Suppose that \(s_m\to\sigma_0\) and
\(m\{1-B_{s_m}(1)\}\to\theta\).  Put
\[
 \beta_m=B_{s_m}(1),\qquad
 p_{m,j}=\frac{b_j(s_m)}{\beta_m},\qquad
 \widehat\mu_m=\sum_{j\ge1}jp_{m,j},
\]
and let \(C_{m,1},C_{m,2},\ldots\) have mass function \(p_{m,j}\).
Then
\begin{equation}
 s_m-\sigma_0=O(m^{-1}),\quad
 \beta_m=1-\frac{\theta}{m}+o(m^{-1}),\quad
 \widehat\mu_m=\mu_C+O(m^{-1}).
 \label{eq:near-critical-basic-expansions}
\end{equation}
If \(r_m/m\to x>0\), then, with
\(S_{m,r}=C_{m,1}+\cdots+C_{m,r}\),
\begin{equation}
 \frac{S_{m,r_m}-r_m\mu_C}{a_m}
 \xrightarrow{\rm d}x^{1/\alpha}\mathcal S_\alpha.
 \label{eq:near-critical-triangular-stable-limit}
\end{equation}
The weak law \(S_{m,r}/r\to\mu_C\) is uniform when
\(\delta m\le r\le A m\), for every fixed \(0<\delta<A<\infty\).
\end{lemma}

\begin{proof}
For \(s>2\), equation~\eqref{eq:weighted-renewal-at-one} gives
\[
 B_s(1)=\frac{\zeta(s-1)}{\zeta(s)}-1.
\]
This function is continuously differentiable near \(\sigma_0\), and its
derivative there is \(-\kappa<0\).  The inverse-function theorem therefore
turns \(B_{s_m}(1)-1=O(m^{-1})\) into
\(s_m-\sigma_0=O(m^{-1})\).  The first two assertions in
\eqref{eq:near-critical-basic-expansions} follow.  The Panov--Liehl bound
used below \eqref{eq:critical-cost-mean}, with one additional factor
\(\log q\), gives, uniformly for \(s\) in a compact neighborhood of
\(\sigma_0\),
\begin{equation}
 \sum_{p/q\in\mathcal A}
 (1+c(p/q))\log q\,q^{-s}<\infty.
 \label{eq:near-critical-differentiation-domination}
\end{equation}
Termwise differentiation is consequently valid for both the mass and the
first cost moment.  Thus \(\widehat\mu_m-\mu_C=O(|s_m-\sigma_0|)\), proving
the last assertion in \eqref{eq:near-critical-basic-expansions}.

For the stable limit, write
\[
 D_s(y)=\frac1{B_s(1)}
 \sum_{p/q\in\mathcal A}q^{-s}
 \{e^{iyc(p/q)}-1-iyc(p/q)\}.
\]
The domination in \eqref{eq:near-critical-differentiation-domination} and
dominated convergence show that
\begin{equation}
 \sup_{|s-\sigma_0|\le\varepsilon}
 |\partial_sD_s(y)|\longrightarrow0
 \quad (y\to0)
 \label{eq:compensated-transform-uniformity}
\end{equation}
for sufficiently small fixed \(\varepsilon>0\).  The critical stable-domain
calculation in \Cref{lem:one-sided-stable-domain} is equivalently
\[
 mD_{\sigma_0}(t/a_m)\longrightarrow\Psi_\alpha(t),
\]
where \(\Psi_\alpha\) is the exponent in
\eqref{eq:centered-stable-characteristic-exponent}.  Since
\(m|s_m-\sigma_0|=O(1)\),
\eqref{eq:compensated-transform-uniformity} gives
\[
 m\{D_{s_m}(t/a_m)-D_{\sigma_0}(t/a_m)\}\longrightarrow0.
\]
The difference between centering by \(\widehat\mu_m\) and by \(\mu_C\) is
\(r_mO(m^{-1})/a_m=o(1)\).  Taking logarithms of the one-letter
characteristic functions now proves
\eqref{eq:near-critical-triangular-stable-limit}.

Finally, \eqref{eq:near-critical-differentiation-domination} implies uniform
integrability of the near-critical letter laws.  Truncating the letters at
a fixed level, applying Chebyshev's inequality to the bounded variables,
and then removing the truncation proves the asserted uniform weak law.
\end{proof}

\begin{proof}[Proof of \Cref{thm:finite-size-fibonacci-crossover}]
Use the probability law from \Cref{lem:near-critical-letter-array}.  For a
word of length \(r\) and cost \(j\), its unnormalized inverse-power weight is
\(\beta_m^r\Pr\{S_{m,r}=j\}\).  The same orbit count as in
\eqref{eq:critical-gibbs-renewal-average} therefore gives, for bounded
\(f:\RR^2\to\mathbb C\),
\begin{align}
 &Z_m^R(-s_m)\,\mathbb E_{\mathbb G_{m,s_m}}
 f\left(\frac{J_m}{m},\frac{H_m-J_m/\mu_C}{a_m}\right)
 \notag\\
 &\quad=f(0,0)
 +2\sum_{r\ge1}\beta_m^r\mathbb E_m\left[
 \mathbf1_{\{S_{m,r}<m\}}
 f\left(\frac{S_{m,r}}m,
         \frac{r-S_{m,r}/\mu_C}{a_m}\right)\right]
 \notag\\
 &\qquad
 +\sum_{r\ge1}\beta_m^r\mathbb E_m\left[
 \mathbf1_{\{S_{m,r}=m\}}
 f\left(1,\frac{r-m/\mu_C}{a_m}\right)\right].
 \label{eq:near-critical-gibbs-renewal-average}
\end{align}
Here \(\mathbb E_m\) refers to the triangular letter law.  Since every
letter costs at least three, only \(r\le m/3\) can occur in the last sum.
Moreover \(\sup_{r\le m/3}\beta_m^r=O(1)\), while the events
\(\{S_{m,r}=m\}\), as \(r\) varies, are disjoint renewal events on each
sample path.  The last sum is therefore \(O(1)\), uniformly in \(m\).

Set \(t_{m,r}=r\mu_C/m\).  The uniform weak law from
\Cref{lem:near-critical-letter-array} permits the same interior truncation
as in the argument from \eqref{eq:critical-gibbs-characteristic-sum} to
\eqref{eq:critical-gibbs-limit-characteristic-function}: indices with
\(t_{m,r}\le\varepsilon\) or
\(|t_{m,r}-1|\le\varepsilon\) contribute \(O(\varepsilon m)\), indices
with \(t_{m,r}>1+\varepsilon\) contribute \(o(m)\), and in the remaining
range the event \(S_{m,r}<m\) may be removed at cost \(o(m)\).  Uniformly on
that interior range,
\begin{equation}
 \beta_m^r=\exp\{-\theta t_{m,r}/\mu_C+o(1)\}.
 \label{eq:near-critical-fugacity-limit}
\end{equation}
Taking \(f=1\) in \eqref{eq:near-critical-gibbs-renewal-average} and using
the Riemann-sum mesh \(\mu_C/m\) gives
\begin{align}
 \frac{Z_m^R(-s_m)}m
 &\longrightarrow \frac2{\mu_C}
 \int_0^1e^{-\theta t/\mu_C}\,dt
 \notag\\
 &=\begin{cases}
 \displaystyle\frac{2(1-e^{-\theta/\mu_C})}{\theta},&\theta\ne0,\\[6pt]
 \displaystyle\frac2{\mu_C},&\theta=0.
 \end{cases}
 \label{eq:finite-size-crossover-normalizer-proof}
\end{align}

It remains to retain the second coordinate in the same sum.  Take
\(f(x,y)=e^{i\xi x+i\eta y}\).  In the interior,
\(S_{m,r}/m\) may be replaced by \(t_{m,r}\).  By
\eqref{eq:near-critical-triangular-stable-limit}, uniformly there by the
subsequence argument used in the critical proof,
\[
 \mathbb E_m\exp\left\{
 i\eta\frac{r-S_{m,r}/\mu_C}{a_m}\right\}
 =\exp\left\{\frac{t_{m,r}}{\mu_C}
 \Psi_\alpha(-\eta/\mu_C)\right\}+o(1).
\]
After division by \eqref{eq:finite-size-crossover-normalizer-proof}, the
limiting joint characteristic function is
\begin{equation}
 \frac{\displaystyle
 \int_0^1e^{-\theta t/\mu_C+i\xi t}
 \exp\{(t/\mu_C)\Psi_\alpha(-\eta/\mu_C)\}\,dt}
 {\displaystyle\int_0^1e^{-\theta t/\mu_C}\,dt}.
 \label{eq:finite-size-crossover-characteristic-function}
\end{equation}
The homogeneity of \(\Psi_\alpha\) identifies this as the characteristic
function of the vector stated in
\eqref{eq:finite-size-fibonacci-crossover}.  The continuity theorem finishes
the proof.
\end{proof}

At \(\theta=0\), the density in
\eqref{eq:finite-size-crossover-location-density} is uniform and the scalar
limit in \eqref{eq:finite-size-crossover-normalizer} is \(2/\mu_C\).
Thus \Cref{thm:critical-fibonacci-gibbs-geometry} is exactly the central
member of the crossover theorem.

\begin{lemma}[Gaussian inversion of an exponentially tilted renewal]
\label{lem:gaussian-renewal-inversion}
Let \(P(z)=\sum_{j\ge1}p_jz^j\) be an aperiodic probability generating
function analytic in a disk of radius greater than one.  If its mean and
variance are \(\mu\) and \(\tau^2\), and
\(T_r=X_1+\cdots+X_r\), then
\begin{equation}
 \mathcal L\left(
 \left.\frac{r-j/\mu}{\sqrt j}\ \right|\ T_r=j
 \text{ for a renewal epoch }r\right)
 \xrightarrow{\rm d}\mathcal N\left(0,\frac{\tau^2}{\mu^3}\right),
 \label{eq:gaussian-renewal-inversion}
\end{equation}
where the conditioning weights epoch \(r\) proportionally to
\(\Pr\{T_r=j\}\).
\end{lemma}

\begin{proof}
The marked renewal coefficient is
\[
 V_j(y)=\sum_{r\ge0}y^r\Pr\{T_r=j\}
 =[z^j]\frac1{1-yP(z)}.
\]
Fix a small \(\delta>0\) and put
\(K_\delta:=\{e^{it}:|t|\le\delta\}\), the compact parameter arc used below.
For each fixed \(\lambda\), \(e^{i\lambda/\sqrt j}\in K_\delta\) for all
sufficiently large \(j\).
At \(y=1\), aperiodicity gives the strict inequality
\(|P(z)|<P(|z|)\) for every \(z\) off the positive real axis, so \(z=1\) is the
only zero of \(1-P(z)\) on the relevant circle \(|z|=1\).  The simple root
\(z=z(y)\) near one and the strict gap persist uniformly on \(K_\delta\):
after shrinking \(\delta\), there are \(\epsilon,\eta>0\) such that every other
zero of \(1-yP(z)\) is at least \(\epsilon\) from \(z(y)\), and on the remainder
of a contour enclosing \(z(y)\) one has
\[
  |1-yP(z)|\ge\eta\qquad(y\in K_\delta).
\]
Choosing the outer part of that contour with radius
\(R>\sup_{y\in K_\delta}|z(y)|\), its Cauchy integral is \(O(R^{-j})\)
uniformly in \(y\in K_\delta\).  Thus the residue at \(z(y)\) is the sole
dominant contribution uniformly on the parameter range in use.  Residue
extraction gives
\begin{equation}
 V_j(y)=\frac{z(y)^{-j}}{y z(y)P'(z(y))}\{1+o(1)\}
 \label{eq:marked-renewal-pole}
\end{equation}
uniformly in that neighborhood.  Put \(y=e^h\) and \(z(y)=e^{w(h)}\).
Since
\[
 \log P(e^w)=\mu w+\frac{\tau^2}{2}w^2+O(w^3),
\]
the equation \(h+\log P(e^{w(h)})=0\) yields
\[
 w(h)=-\frac h\mu-\frac{\tau^2}{2\mu^3}h^2+O(h^3).
\]
Insert \(h=i\lambda/\sqrt j\) into
\eqref{eq:marked-renewal-pole}, divide by \(V_j(1)\), and multiply by
\(e^{-i\lambda\sqrt j/\mu}\).  The linear terms cancel and the limit is
\(\exp\{-\lambda^2\tau^2/(2\mu^3)\}\), proving
\eqref{eq:gaussian-renewal-inversion}.
\end{proof}

\begin{proof}[Proof of \Cref{cor:fibonacci-three-regime}]
Write \(\rho_s=1+B_s(1)\).  If \(s>\sigma_0\), then \(B_s(1)<1\), and
\[
 \sum_{j\ge0}u_j(s)=U_s(1)=\frac1{1-B_s(1)}=\frac1{2-\rho_s}.
\]
Equation~\eqref{eq:standard-layer-renewal} and \(u_m(s)\to0\) give
\[
 Z_m^R(-s)\longrightarrow2U_s(1)-1=\frac{\rho_s}{2-\rho_s}.
\]
The same summability makes the tails of the total weight carried by
\(J_m\ge K\) tend to zero uniformly in \(m\), so \(J_m=O_{\mathbb P}(1)\).

Now let \(2<s<\sigma_0\).  Monotonicity gives a unique
\(z_s\in(0,1)\) with \(B_s(z_s)=1\).  Under the probability law
\[
 p_j=b_j(s)z_s^j,
\]
let \(v_j\) be its renewal mass.  Then
\[
 u_j(s)=z_s^{-j}v_j,\qquad
 v_j\longrightarrow\frac1{\mu_s},qquad
 \mu_s=z_sB_s'(z_s),
\]
by the aperiodic finite-mean renewal theorem.  Substitution in
\eqref{eq:standard-layer-renewal} gives
\[
 Z_m^R(-s)\sim
 \frac{1+z_s}{z_s(1-z_s)B_s'(z_s)}z_s^{-m}.
\]
More precisely, if \(K_m=m-J_m\), then
\begin{equation}
 \Pr\{K_m=0\}\longrightarrow\frac{1-z_s}{1+z_s},\qquad
 \Pr\{K_m=k\}\longrightarrow
 \frac{2(1-z_s)}{1+z_s}z_s^k\quad(k\ge1).
 \label{eq:subcritical-boundary-defect-law}
\end{equation}
This proves \(m-J_m=O_{\mathbb P}(1)\).

The probability generating function \(B_s(z_sz)\) is analytic for
\(|z|<z_s^{-1}\), is aperiodic, and has mean \(\mu_s\) and variance
\(\tau_s^2\).  Conditional on \(J_m=j\), the letter-count weights are
exactly the renewal-epoch weights in
\Cref{lem:gaussian-renewal-inversion}.  Combining that lemma with
\eqref{eq:subcritical-boundary-defect-law} and \(J_m/m\to1\) gives the
Gaussian limit in \eqref{eq:subcritical-gaussian-length}.  This completes
the three-regime proof.
\end{proof}
