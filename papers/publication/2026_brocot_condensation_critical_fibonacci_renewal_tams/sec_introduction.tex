\section{Introduction}\label{sec:introduction}

The Stern--Brocot and Farey constructions organize rational numbers by
continued-fraction complexity.  A natural statistical question is what a
typical rational looks like when a fixed complexity layer is biased toward
small denominators.  Here the complexity of the canonical regular continued
fraction
\[
 x=[0;a_1,\ldots,a_r],\qquad a_i\ge1,\quad a_r\ge2,
\]
is the digit sum \(a_1+\cdots+a_r=n\), and the weight is
\(\operatorname{den}(x)^{-s}\) for a fixed \(s>2\).  Although the continuant
in the denominator is nonmultiplicative across the digits, a single partial
quotient absorbs all but a tight amount of the digit sum, while the two
finite contexts on its sides converge to independent arithmetic laws.

For \(n\ge2\), let \(\mathcal Q_n\) be the set of these canonical fractions
of digit sum \(n\), and define
\begin{equation}
 Z_n(s):=\sum_{x\in\mathcal Q_n}\operatorname{den}(x)^{-s},
 \qquad
 \mathbb P_{n,s}\{x\}:=\frac{\operatorname{den}(x)^{-s}}{Z_n(s)}.
 \label{eq:brocot-probability}
\end{equation}
For a finite positive word \(\mathbf w\), let \(\mathcal K(\mathbf w)\) be
its regular continuant, \(|\mathbf w|_1\) its digit sum, and
\(\ell(\mathbf w)\) its length.  Write \(\mathcal W_{\rm L}\) for all
finite positive words, including the empty word, and
\(\mathcal W_{\rm R}\) for the canonical regular words, again including the
empty word.  Put
\[
 \rho_s=\frac{\zeta(s-1)}{\zeta(s)}.
\]

\begin{theorem}[Total-variation condensation for Brocot fractions]
\label{thm:brocot-condensation}
Fix \(s>2\).  Sample
\(x_n=[0;a_1,\ldots,a_{r_n}]\) according to \(\mathbb P_{n,s}\), and put
\[
 M_n:=\max_{1\le j\le r_n}a_j.
\]
On \(\{M_n>n/2\}\), let \(J_n\) be the unique index attaining the maximum
and define
\[
 U_n=(a_1,\ldots,a_{J_n-1}),
 \qquad
 V_n=(a_{J_n+1},\ldots,a_{r_n}).
\]
On the complementary event set \(U_n=V_n=\varnothing\), and extend \(J_n\)
arbitrarily, say by \(J_n=1\).

Let \(U\) and \(V\) be independent random words with laws
\begin{equation}
 \begin{aligned}
  \mathbb P\{U=\mathbf u\}
  &=\frac{\mathcal K(\mathbf u)^{-s}}{2\rho_s},
  &\mathbf u&\in\mathcal W_{\rm L},\\
  \mathbb P\{V=\mathbf v\}
  &=\frac{\mathcal K(\mathbf v)^{-s}}{\rho_s},
  &\mathbf v&\in\mathcal W_{\rm R}.
 \end{aligned}
 \label{eq:limiting-context-laws}
\end{equation}
Then, as \(n\to\infty\),
\begin{align}
 \mathbb P_{n,s}\{M_n>n/2\}&\longrightarrow1,
 \label{eq:brocot-unique-condensate}\\
 d_{\rm TV}\bigl(\mathcal L_{\mathbb P_{n,s}}(U_n,V_n),
                   \mathcal L(U,V)\bigr)&\longrightarrow0.
 \label{eq:brocot-context-tv}
\end{align}
Consequently, if \(D:=|U|_1+|V|_1\), then
\begin{equation}
 n-M_n\xrightarrow{\rm TV}D,
 \qquad
 \frac{M_n}{n}\xrightarrow{\mathbb P}1,
 \label{eq:brocot-defect-limit}
\end{equation}
where
\begin{equation}
 \mathbb P\{D=k\}
 =\frac1{2\rho_s^2}
 \sum_{\substack{\mathbf u\in\mathcal W_{\rm L},\
                   \mathbf v\in\mathcal W_{\rm R}\\
                   |\mathbf u|_1+|\mathbf v|_1=k}}
 \{\mathcal K(\mathbf u)\mathcal K(\mathbf v)\}^{-s},
 \qquad k\ge0.
 \label{eq:brocot-defect-law}
\end{equation}
The location data satisfy
\begin{equation}
 (J_n-1,r_n-J_n)\xrightarrow{\rm TV}
 (\ell(U),\ell(V)).
 \label{eq:brocot-location-limit}
\end{equation}
Finally, if \(q_n=\operatorname{den}(x_n)\), then
\begin{equation}
 \frac{q_n}{M_n\mathcal K(U_n)\mathcal K(V_n)}
 \xrightarrow{\mathbb P}1.
 \label{eq:brocot-denominator-ratio}
\end{equation}
\end{theorem}

For a reduced fraction \(p/q\in(0,1)\), let \(c(p/q)=2d(p/q)+1\), where
\(d(p/q)\) is the sum of the shifted digits in its negative continued
fraction, and put
\[
 b_\ell(s)=\sum_{\substack{0<p<q,\ (p,q)=1\\c(p/q)=\ell}}q^{-s}.
\]

\begin{corollary}[Corrected denominator-layer asymptotic]
\label{thm:arithmetic-denominator-layer}
For every fixed \(s>2\),
\begin{equation}
 Z_n(s)\sim2\rho_s^2n^{-s},
 \qquad
 b_{2d+1}(s)=Z_{d+1}(s)
 \sim2\left(\frac{\zeta(s-1)}{\zeta(s)}\right)^2d^{-s}.
 \label{eq:low-temperature-local-layer}
\end{equation}
\end{corollary}

This argument determines the leading coefficient.  Determining the later
coefficients would require uniform higher-order expansions beyond the
dominated-convergence estimate used here.

Corollary~\ref{thm:arithmetic-denominator-layer} supplies exactly the
generator-cost asymptotic needed to obtain a second-order renewal expansion
for the classical Fibonacci partition function.  Let \(F_0=0,F_1=1\), and
let \(R(N)\) count representations of \(N\) as a sum of distinct members of
\(F_2,F_3,\ldots\).  On
\[
 \mathcal I_m=[F_{m+1}-1,F_{m+2}-1)\cap\ZZ,
 \qquad
 Z_m^R(t)=\sum_{N\in\mathcal I_m}R(N)^t,
\]
let \(u_0(s)=1\) and
\[
 u_j(s)=\sum_{\ell=1}^jb_\ell(s)u_{j-\ell}(s)\qquad(j\ge1).
\]

\begin{proposition}[Weighted generator renewal]
\label{prop:weighted-generator-renewal}
For every real \(s\) and \(m\ge1\),
\begin{equation}
 Z_m^R(-s)
 =1+u_m(s)+2\sum_{j=1}^{m-1}u_j(s)
 =2\sum_{j=0}^{m-1}u_j(s)+u_m(s)-1.
 \label{eq:standard-layer-renewal}
\end{equation}
Moreover, \(u_j(s)\) is exactly the total inverse-\(s\) weight of the
Weinstein generators of cost \(j\).
\end{proposition}

There is a unique \(\sigma_0>2\) such that
\[
 \frac{\zeta(\sigma_0-1)}{\zeta(\sigma_0)}=2.
\]
At this exponent \(b_j(\sigma_0)\) is a probability mass function.  Let
\(C\) have this law, set
\[
 \alpha=\sigma_0-1,\qquad \mu_C=\EE C,
\]
and define
\begin{equation}
 \begin{aligned}
  b_C&:=
  \sum_{\mathbf u\in\mathcal W_{\rm L}}
  \sum_{\mathbf v\in\mathcal W_{\rm R}}
  \{\mathcal K(\mathbf u)\mathcal K(\mathbf v)\}^{-\sigma_0}\\
  &=2\left(\frac{\zeta(\sigma_0-1)}{\zeta(\sigma_0)}\right)^2=8,\\
  K_C&=\frac{2^{\sigma_0-1}}{\sigma_0-1}b_C
  =\frac{2^{\sigma_0+2}}{\sigma_0-1}.
 \end{aligned}
 \label{eq:critical-tail-constants}
\end{equation}

\begin{theorem}[Critical asymptotic for the classical Fibonacci partition function]
\label{cor:classical-fibonacci-critical-asymptotic}
As \(m\to\infty\),
\begin{align}
 &\sum_{F_{m+1}-1\le N<F_{m+2}-1}R(N)^{-\sigma_0}
 \notag\\
 &\qquad=\frac{2m}{\mu_C}
 +\frac{2K_C}{\mu_C^2(\sigma_0-2)(3-\sigma_0)}
  m^{3-\sigma_0}+o(m^{3-\sigma_0}).
 \label{eq:classical-fibonacci-critical-asymptotic}
\end{align}
Equivalently,
\begin{equation}
 \sum_{N\in\mathcal I_m}R(N)^{-\sigma_0}
 =\frac{2m}{\mu_C}\left[
 1+\frac{K_C}{\mu_C(\sigma_0-2)(3-\sigma_0)}
 m^{2-\sigma_0}+o(m^{2-\sigma_0})\right].
 \label{eq:relative-fibonacci-critical-correction}
\end{equation}
The exponent \(3-\sigma_0\) lies strictly between zero and one, so the
second term diverges on the absolute partition-function scale while
remaining smaller than the leading linear term.
\end{theorem}

\begin{corollary}[Stable domain of attraction for the generator cost]
\label{cor:critical-cost-stable-domain}
The critical generator cost satisfies
\begin{equation}
 \Pr\{C>x\}\sim K_Cx^{1-\sigma_0}.
 \label{eq:critical-letter-tail}
\end{equation}
It belongs to the domain of attraction of a spectrally positive
\(\alpha\)-stable law: if \(C_1,C_2,\ldots\) are independent copies of
\(C\) and \(a_n>0\) satisfies \(n\Pr\{C>a_n\}\to1\), then
\begin{equation}
 \frac{C_1+\cdots+C_n-n\mu_C}{a_n}
 \xrightarrow{\mathrm d}\mathcal S_\alpha,
 \label{eq:critical-cost-stable-domain}
\end{equation}
where
\begin{equation}
 \log\EE e^{it\mathcal S_\alpha}
 =\int_0^\infty
   (e^{itx}-1-itx)\alpha x^{-\alpha-1}\,dx.
 \label{eq:centered-stable-characteristic-exponent}
\end{equation}
The explicit choice
\begin{equation}
 a_n=(K_Cn)^{1/\alpha}
 =\left(\frac{2^{\sigma_0+2}}{\sigma_0-1}n\right)^{1/(\sigma_0-1)}
 \label{eq:explicit-critical-stable-normalization}
\end{equation}
has the required tail normalization.  The limiting L\'evy measure has
density \(\alpha x^{-\alpha-1}\) on \((0,\infty)\) and zero mass on
\(( -\infty,0)\).
\end{corollary}

The stable law also governs a statistic of the critical Fibonacci ensemble
itself.  Give \(\mathcal I_m\) the probability measure
\begin{equation}
 \mathbb G_m\{N\}:=
 \frac{R(N)^{-\sigma_0}}{Z_m^R(-\sigma_0)},
 \qquad N\in\mathcal I_m.
 \label{eq:critical-fibonacci-gibbs-measure}
\end{equation}
If \(N\ne F_{m+1}-1\), Weinstein's orbit classification gives a unique
generator \(g(N)\), and the free-monoid bijection gives a unique letter word
\(w(N)=(x_1,\ldots,x_{H_m})\).  Set
\[
 J_m=c(w(N))=L(g(N)),
\]
and put \(J_m=H_m=0\) at the exceptional left endpoint.

\begin{theorem}[Critical Gibbs geometry of Fibonacci layers]
\label{thm:critical-fibonacci-gibbs-geometry}
Let \(N\) be sampled from \(\mathbb G_m\), let
\(a_m=(K_Cm)^{1/\alpha}\), and let \(U\) be uniform on \([0,1]\) and
independent of \(\mathcal S_\alpha\).  Then
\begin{equation}
 \left(\frac{J_m}{m},
       \frac{H_m-J_m/\mu_C}{a_m}\right)
 \xrightarrow{\mathrm d}
 \left(U,-\mu_C^{-1-1/\alpha}U^{1/\alpha}\mathcal S_\alpha\right).
 \label{eq:critical-fibonacci-gibbs-geometry}
\end{equation}
In particular, the generator cost of a critical Gibbs sample is
asymptotically uniform across the Fibonacci layer.  Conditional on its
macroscopic cost, the generator length has a stable fluctuation of the
opposite spectral sign.
\end{theorem}

The proof of Theorem~\ref{thm:brocot-condensation} rests on a summable
two-context domination.  Once a digit \(a>n/2\) is singled out,
multilinearity of the continuant gives
\[
 \mathcal K(\mathbf u,a,\mathbf v)
 =a\mathcal K(\mathbf u)\mathcal K(\mathbf v)
  +O_{\mathbf u,\mathbf v}(1).
\]
The inverse-continuant masses of both contexts are summable for \(s>2\).
A greedy block decomposition and a direct cut make words without a digit
above \(n/2\) negligible; dominated convergence then gives the product
context law, and the discrete Scheff\'e argument gives total-variation
convergence.

Moshchevitin and Zhigljavsky studied Farey-tree entropy sums
\cite{MoshchevitinZhigljavsky2004}, and Dushistova obtained precise
fixed-digit-sum denominator estimates for Brocot sequences
\cite{Dushistova2007}.  The scalar sum here is termwise her local sum after
the exponent change \(s=2\beta\).  The endpoint calculation changes its
printed leading coefficient from \(R_s+2R_s^2\) to \(2R_s^2\).
Total-variation one-big-jump results and Gibbs-partition condensation give
useful comparisons
\cite{ArmendarizLoulakis2011,Stufler2020}; their product-form hypotheses are
absent for continuant weights, and the two-sided continuant estimate supplies
the needed arithmetic factorization.

Fibonacci representations have a long history
\cite{Zeckendorf1972,Carlitz1968,Berstel2001}.  We use Weinstein's free
generator classification \cite{Weinstein2016}, with its Fibonacci indexing
and four orbit branches verified in the present notation.  Its letters are
reduced fractions in \((0,1)\): a letter \(p/q\) carries multiplicative
weight \(q\) and the odd cost from its negative continued fraction.
Proposition~\ref{prop:weighted-generator-renewal} converts their free-monoid
factorization into the exact renewal identity, and
Corollary~\ref{thm:arithmetic-denominator-layer} supplies the tail that
drives Theorems~\ref{cor:classical-fibonacci-critical-asymptotic} and
\ref{thm:critical-fibonacci-gibbs-geometry}.  Recent work on the Fibonacci
partition function gives exact mean-value asymptotics, variance and
positive-integral-moment estimates, and a local dynamical description
\cite{ChowSlattery2021,ChowJones2024,Sanna2025,Kempton2023}; these do not
concern the inverse-power Gibbs ensemble or the length of the Weinstein
factorization.

Section~\ref{sec:brocot-condensation} proves the condensation theorem and
the denominator-layer corollary.  Sections~\ref{sec:fibonacci-renewal} and
\ref{sec:arithmetic-critical} prove the exact generator renewal, the critical
Fibonacci expansion, and the stable Gibbs geometry.
