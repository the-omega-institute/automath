\section{The arithmetic critical law}\label{sec:arithmetic-critical}

Under the critical letter law,
\[
 \Pr\{C=c(p/q)\}=q^{-\sigma_0}
 \qquad(p/q\in\mathcal A),
\]
where equal costs are combined.  Thus
\(\Pr\{C=2d+1\}=b_{2d+1}(\sigma_0)\), and the normalization in
\eqref{eq:critical-renewal-normalization} makes this a probability law.

\begin{lemma}[From the local denominator layer to the cost tail]
\label{lem:critical-tail-summation}
At \(s=\sigma_0\), the context series defining \(b_C\) is absolutely
convergent and
\[
 b_C=2\left(\frac{\zeta(\sigma_0-1)}{\zeta(\sigma_0)}\right)^2=8.
\]
Moreover,
\begin{equation}
 b_{2d+1}(\sigma_0)\sim b_Cd^{-\sigma_0}.
 \label{eq:critical-letter-local-tail}
\end{equation}
If \(\alpha=\sigma_0-1\), then
\begin{equation}
 \overline F_C(x):=\Pr\{C>x\}
 \sim \frac{2^\alpha b_C}{\alpha}x^{-\alpha}
 =K_Cx^{-\alpha}.
 \label{eq:tail-summation-expanded}
\end{equation}
In particular, \(\overline F_C\) is regularly varying with index
\(-\alpha\).
\end{lemma}

\begin{proof}
The two context masses in \eqref{eq:brocot-context-masses} are finite for
every \(s>2\), so their product is the absolutely convergent double series
in \eqref{eq:critical-tail-constants}.  Their product is \(2\rho_s^2\).
At the critical point \(\rho_{\sigma_0}=2\), which proves \(b_C=8\).

By \Cref{thm:arithmetic-denominator-layer}, with \(d\) denoting the
negative-continuant depth,
\[
 \Pr\{C=2d+1\}=b_{2d+1}(\sigma_0)
 =b_Cd^{-\sigma_0}(1+o(1)).
\]
For real \(x\), the inequality \(2d+1>x\) is equivalent to
\(d>(x-1)/2\).  Discrete Karamata summation gives
\[
 \sum_{d>y}d^{-\sigma_0}
 \sim\frac{y^{1-\sigma_0}}{\sigma_0-1}.
\]
The same conclusion holds after multiplication by a sequence tending to
\(b_C\), by bounding all sufficiently large summands between
\((b_C-\varepsilon)d^{-\sigma_0}\) and
\((b_C+\varepsilon)d^{-\sigma_0}\).  Therefore
\begin{align*}
 \overline F_C(x)
 &=\sum_{d>(x-1)/2}b_{2d+1}(\sigma_0)\\
 &\sim\frac{b_C}{\sigma_0-1}
       \left(\frac{x-1}{2}\right)^{1-\sigma_0}
 =\frac{2^\alpha b_C}{\alpha}x^{-\alpha}.
\end{align*}
This is \eqref{eq:tail-summation-expanded}, and the last constant is
\(K_C=2^{\sigma_0+2}/(\sigma_0-1)\).
\end{proof}

\begin{lemma}[Second-order lattice renewal input]
\label{lem:second-order-lattice-renewal}
Let \((p_j)_{j\ge1}\) be a probability mass function on the positive
integers.  Suppose that
\begin{enumerate}
 \item \(\gcd\{j:p_j>0\}=1\), so the renewal law is aperiodic of span one;
 \item \(\mu=\sum_{j\ge1}jp_j<\infty\);
 \item its tail \(\overline F(j)=\sum_{k>j}p_k\) satisfies
 \(\overline F(j)\sim Kj^{-\alpha}\), where \(K>0\) and
 \(\alpha>1\).
\end{enumerate}
Define \(v_0=1\) and
\(v_j=\sum_{k=1}^jp_kv_{j-k}\).  Then
\begin{equation}
 v_j=\frac1\mu+\frac{K}{\mu^2(\alpha-1)}j^{1-\alpha}
      +o(j^{1-\alpha}).
 \label{eq:second-order-renewal-input}
\end{equation}
\end{lemma}

\begin{proof}
Put
\[
 p^{\rm e}_j=\frac{\overline F(j-1)}\mu,
 \qquad
 \overline F^{\rm e}(j)
 =\sum_{k>j}p^{\rm e}_k
 =\frac1\mu\sum_{r\ge j}\overline F(r).
\]
The first assertion following equation (8) in Section 3.2 of Omey and Van
Gulck \cite{OmeyVanGulck2015}, specialized to the preceding hypotheses, is
\begin{equation}
 v_j-\frac1\mu\sim\frac{\overline F^{\rm e}(j)}\mu.
 \label{eq:omey-vangulck-equilibrium-form}
\end{equation}
Their \(u_n\), \(\mu\), and \(\overline F_n\) are respectively \(v_j\),
\(\mu\), and \(\overline F(j)\) here.  Discrete Karamata summation gives
\[
 \overline F^{\rm e}(j)
 =\frac1\mu\sum_{r\ge j}\overline F(r)
 \sim\frac{K}{\mu(\alpha-1)}j^{1-\alpha}.
\]
Substitution in \eqref{eq:omey-vangulck-equilibrium-form} proves
\eqref{eq:second-order-renewal-input}.
\end{proof}

\begin{proposition}[Verification for the critical letter law]
\label{prop:critical-renewal-hypotheses}
The mass function \(p_j=b_j(\sigma_0)\) satisfies every hypothesis of
\Cref{lem:second-order-lattice-renewal}, with
\[
 \mu=\mu_C,\qquad \alpha=\sigma_0-1\in(1,2),\qquad K=K_C.
\]
Consequently \eqref{eq:critical-renewal-second-order} holds.
\end{proposition}

\begin{proof}
First, \eqref{eq:weighted-renewal-at-one} and
\eqref{eq:critical-renewal-normalization} give
\[
 \sum_{j\ge1}p_j=B_{\sigma_0}(1)=1,
\]
so this is a probability mass function on the positive integers.  More
particularly, the negative continued fractions
\(1/2=[0;2]_-\) and \(1/3=[0;3]_-\) have
\[
 c(1/2)=2(2-1)+1=3,\qquad
 c(1/3)=2(3-1)+1=5,
\]
and masses \(2^{-\sigma_0}>0\) and \(3^{-\sigma_0}>0\).  Hence
\[
 \gcd\{j:p_j>0\}\mid\gcd(3,5)=1,
\]
so the renewal span is one.

Second, the mean is exactly the series in
\eqref{eq:critical-cost-mean}.  The Panov--Liehl estimate recorded there
gives
\[
 \mu_C\ll\sum_{q\ge2}q^{1-\sigma_0}(\log q)^2<\infty
\]
because \(\sigma_0>2\).

Third, \(\rho_s=\zeta(s-1)/\zeta(s)\) decreases from infinity to one on
\((2,\infty)\), while \(\rho_3=\zeta(2)/\zeta(3)<2\).  The equation
\(\rho_{\sigma_0}=2\) therefore gives \(2<\sigma_0<3\), hence
\(1<\alpha<2\).  Finally, \Cref{lem:critical-tail-summation} gives
\(\overline F_C(j)\sim K_Cj^{-\alpha}\).  Applying
\Cref{lem:second-order-lattice-renewal} yields
\begin{equation}
 u_j(\sigma_0)=\frac1{\mu_C}
 +\frac{K_C}{\mu_C^2(\sigma_0-2)}j^{2-\sigma_0}
 +o(j^{2-\sigma_0}).
 \label{eq:critical-renewal-second-order}
\end{equation}
\end{proof}

\begin{proof}[Proof of \Cref{cor:classical-fibonacci-critical-asymptotic}]
Because \(2-\sigma_0\in(-1,0)\), summation of the renewal remainder in
\eqref{eq:critical-renewal-second-order} is legitimate: if
\(r_j=o(j^{2-\sigma_0})\), then for every \(\varepsilon>0\),
\[
 \left|\sum_{j=1}^{m-1}r_j\right|
 \le O_\varepsilon(1)+\varepsilon
      \sum_{j=1}^{m-1}j^{2-\sigma_0}
 =o(m^{3-\sigma_0}).
\]
Also,
\[
 \sum_{j=1}^{m-1}j^{2-\sigma_0}
 \sim\frac{m^{3-\sigma_0}}{3-\sigma_0}.
\]
Substitution in \eqref{eq:critical-weighted-renewal-identity} gives
\eqref{eq:classical-fibonacci-critical-asymptotic}; the terminal term
\(u_m=O(1)\) and the endpoint \(-1\) are
\(o(m^{3-\sigma_0})\).  Factoring out \(2m/\mu_C\) gives
\eqref{eq:relative-fibonacci-critical-correction}.  The already verified
inequalities \(2<\sigma_0<3\) give the stated scale comparison.
\end{proof}

\begin{lemma}[One-sided stable domain of attraction]
\label{lem:one-sided-stable-domain}
Let \(X\ge0\) have finite mean \(\mu\) and
\(\Pr\{X>x\}\sim Kx^{-\alpha}\), where \(K>0\) and
\(1<\alpha<2\).  Put \(a_n=(Kn)^{1/\alpha}\).  Then
\begin{equation}
 \frac{X_1+\cdots+X_n-n\mu}{a_n}
 \xrightarrow{\rm d}S_\alpha,
 \label{eq:generic-one-sided-stable-limit}
\end{equation}
where
\[
 \log\EE e^{itS_\alpha}
 =\int_0^\infty(e^{itx}-1-itx)\alpha x^{-\alpha-1}\,dx.
\]
The limiting L\'evy measure has density
\(\alpha x^{-\alpha-1}\) on \((0,\infty)\) and zero mass on
\(( -\infty,0)\).
\end{lemma}

\begin{proof}
Apply Feller's stable domain-of-attraction theorem
\cite[Ch.~XVII, Sec.~5, Thm.~2]{Feller1971}.  Since
\(a_n^\alpha=Kn\), for every \(x>0\),
\begin{equation}
 n\Pr\{X>a_nx\}\longrightarrow x^{-\alpha},
 \qquad
 n\Pr\{X<-a_nx\}=0.
 \label{eq:stable-tail-balance}
\end{equation}
Thus the positive and negative tail-balance constants are one and zero,
and the limiting L\'evy measure is
\(\alpha x^{-\alpha-1}\,dx\) on the positive half-line.  Because
\(1<\alpha<2\) and \(\mu<\infty\), the theorem permits centering by the
full mean \(n\mu\); with this centering its compensated exponent is the one
displayed above.  This proves \eqref{eq:generic-one-sided-stable-limit}.
\end{proof}

\begin{proof}[Proof of \Cref{cor:critical-cost-stable-domain}]
Take \(X=C\) in \Cref{lem:one-sided-stable-domain}.  The finite mean is
verified in \Cref{prop:critical-renewal-hypotheses}, and
\Cref{lem:critical-tail-summation} supplies the tail with \(K=K_C\).
Since \(\alpha=\sigma_0-1\), this proves
\eqref{eq:critical-letter-tail}, \eqref{eq:critical-cost-stable-domain}, and
\eqref{eq:centered-stable-characteristic-exponent}.  It also gives the
normalization \eqref{eq:explicit-critical-stable-normalization}.
\end{proof}
