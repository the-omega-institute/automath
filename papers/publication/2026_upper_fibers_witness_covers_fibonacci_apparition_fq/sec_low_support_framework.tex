We now introduce the Fibonacci-specific arithmetic input. The structural
results of Section~\ref{article-sec:upper-fibers} did not use the formulas below.

\begin{proposition}[Classical prime-power rank lifting]
\label{prop:prime-power-ladders}
Let $p$ be prime, and put $z_p:=\alpha(p)$. For $p\ne2,5$, put
$h_p:=\nu_p(F_{z_p})$. Then
\[
\alpha(2)=3,\qquad \alpha(4)=6,\qquad
\alpha(2^e)=3\cdot2^{e-2}\quad(e\ge3),
\]
\[
\alpha(5^e)=5^e\quad(e\ge1),
\]
and, for $p\ne2,5$,
\[
\alpha(p^e)=p^{\max\{e-h_p,0\}}z_p\quad(e\ge1).
\]
\end{proposition}

\begin{proof}
Wall's rank-divisibility theorem gives
\[
p\mid F_r\quad\Longleftrightarrow\quad z_p\mid r
\]
\cite{Wall1960FibonacciModm}. For $p\ne2,5$, the classical congruence
\[
F_{p-(5/p)}\equiv0\pmod p
\]
gives
\[
z_p\mid p-\left(\frac5p\right),
\]
so $p\nmid z_p$. Lengyel's valuation formula
\cite[Theorems~1 and~2]{Lengyel1995OrderFibonacciLucas}, in the modern
treatment of Medina--Rowland \cite{MedinaRowland2015PRegularity}, states that
when $z_p\mid r$,
\[
\nu_p(F_r)=\nu_p(F_{z_p})+\nu_p(r).
\]
Writing $r=z_pu$ and using $p\nmid z_p$ yields
\[
\nu_p(F_{z_pu})=h_p+\nu_p(u).
\]
Thus $p^e\mid F_{z_pu}$ if and only if
$\nu_p(u)\ge e-h_p$, and the least admissible $u$ is
$p^{\max\{e-h_p,0\}}$. This proves the formula for $p\ne2,5$ without an
implicit assumption on $z_p$.

For the exceptional primes, the cited formulas specialize to
\[
\nu_5(F_r)=\nu_5(r),
\qquad
\nu_2(F_r)=
\begin{cases}
0, & 3\nmid r,\\
1, & r\equiv3\pmod 6,\\
3, & r\equiv6\pmod {12},\\
\nu_2(r)+2, & 12\mid r.
\end{cases}
\]
The least index divisible by $5^e$ is $5^e$. For $2$, the least indices with
valuation at least $1$, $2$, and $3$ are $3$, $6$, and $6$; for $e\ge4$ the
least admissible multiple of $12$ is $3\cdot2^{e-2}$.
\end{proof}

\begin{corollary}\label{cor:rank-jump-classification}
The rank-jump prime powers are exactly the following:
\begin{enumerate}[label=(\roman*)]
\item $2\in\Atom(3)$, $4\in\Atom(6)$, and
$2^e\in\Atom(3\cdot2^{e-2})$ for every $e\ge4$; the power $8$ is not a rank-jump factor.
\item $5^e\in\Atom(5^e)$ for every $e\ge1$.
\item If $p\ne2,5$, $z_p=\alpha(p)$, and
$h_p=\nu_p(F_{z_p})$, then $p\in\Atom(z_p)$ and
$p^e\in\Atom(p^{e-h_p}z_p)$ for every $e>h_p$; no power $p^e$ with
$1<e\le h_p$ is rank-jump.
\end{enumerate}
Consequently every rank-jump factor is either a prime jump $p\in\Atom(\alpha(p))$ or a
ladder jump $p^e$ with $e>1$. Every ladder jump changes only the $p$-adic
coordinate of its apparition index when it is lowered to $p^{e-1}$.
\end{corollary}

\begin{proof}
Apply Proposition~\ref{prop:prime-power-ladders}. For $p=2$, the ranks are
$3,6,6,12,24,\ldots$ on $2,4,8,16,32,\ldots$. Hence $2$ and $4$ are rank-jump,
$8$ is not, and each $2^e$ with $e\ge4$ is rank-jump. For $p=5$, every step is
rank-jump. If $p\ne2,5$, the rank remains $z_p$ for $1\le e\le h_p$ and then
gains one factor of $p$ at each further level. This proves the list. At every
ladder step, $d(\theta)/d^-(\theta)$ is a power of the same prime $p$, which
proves the final assertion.
\end{proof}

This section keeps the low-support classification in the main theorem stream
while treating the slot mechanics as proof support. The principal conclusion is
Theorem~\ref{thm:low-support-classification-package}: for support three there
are exactly eight admissible families, and every remaining decision is a finite
slot-filling test.  The arithmetic tests for filling those slots are stated here;
the reader-facing checklist and the worked $n=42$ verification are collected in
Appendix~\ref{app:three-coordinate-checklist}.

\begin{lemma}[Prime--ladder alternative]\label{thm:primitive-ladder-dichotomy}
Let $\theta=p^e\in\Atom(d)$ and suppose $d\mid n$. If $e=1$, then
\[
E_n(\theta)=T_n(\theta).
\]
If $e>1$, then $E_n(\theta)$ is either empty or a singleton, and any
coordinate in $E_n(\theta)$ is the prime coordinate $p$. In particular no
Fibonacci rank-jump factor has
\[
1<\#E_n(\theta)<\#T_n(\theta).
\]
\end{lemma}

\begin{proof}
If $e=1$, then $d^-(\theta)=\alpha(1)=1$, so every coordinate of
$T_n(\theta)$ is lost after lowering $\theta$; hence
$E_n(\theta)=T_n(\theta)$.

Assume $e>1$. By Corollary~\ref{cor:rank-jump-classification}, $\theta$ is a
ladder jump. The prime-power formulas in
Proposition~\ref{prop:prime-power-ladders} show that
$d(\theta)$ and $d^-(\theta)$ differ only in the $p$-adic coordinate of the
apparition index: this is explicit for $p=2$ and $p=5$, and for
$p\ne2,5$ it follows from
$\alpha(p^e)=p^{e-h_p}z_p$ at the rank-jump ladder levels. Therefore the only
full coordinate that can be lost when
passing from $d(\theta)$ to $d^-(\theta)$ is the coordinate $p$ itself.
Thus $E_n(\theta)$ is empty or $\{p\}$.

The final assertion follows because either $E_n(\theta)=T_n(\theta)$ in
the prime jump case or $\#E_n(\theta)\le1$ in the ladder case.
\end{proof}

\begin{lemma}[Private-coordinate injection]\label{thm:extremal-coordinate-separation}
Let $\Theta$ be an $n$-witness cover. For each
$\theta\in\Theta$ one may choose a coordinate
\[
\ell(\theta)\in
E_n(\theta)\setminus\bigcup_{\eta\in\Theta,\ \eta\ne\theta}T_n(\eta),
\]
and the chosen coordinates are distinct. If
$|\Theta|=\omega(n)$, then after relabeling the elements of $\Theta$ by
the coordinates of $\mathcal P(n)$ one has
\[
T_n(\theta_\ell)=\{\ell\}\qquad(\ell\in\mathcal P(n)).
\]
In particular, when $\omega(n)=2$, every two-element $n$-witness cover has
one rank-jump factor supported on the first coordinate and one rank-jump factor supported on the
second.
\end{lemma}

\begin{proof}
The existence of $\ell(\theta)$ is exactly the irredundance condition in
Definition~\ref{article-def:witness-cover}. If
$\ell(\theta)=\ell(\eta)$ with $\theta\ne\eta$, then
$\ell(\theta)\in T_n(\eta)$ because
$E_n(\eta)\subseteq T_n(\eta)$, contradicting the choice of
$\ell(\theta)$. Hence the chosen coordinates are distinct.

If $|\Theta|=\omega(n)$, the injection
$\theta\mapsto\ell(\theta)$ is a bijection from $\Theta$ to
$\mathcal P(n)$. The coordinate $\ell(\theta)$ is not in
$T_n(\eta)$ for any $\eta\ne\theta$. Since the $T_n(\eta)$ cover
$\mathcal P(n)$, every coordinate other than $\ell(\theta)$ is the private
coordinate of some other rank-jump factor and therefore is not in $T_n(\theta)$. Thus
$T_n(\theta)=\{\ell(\theta)\}$ for every $\theta$, which gives the stated
separation after relabeling.
\end{proof}

\begin{corollary}[One-coordinate classification]
\label{cor:one-support-complete-classification}
If $\omega(n)=1$, then
\[
\mathcal M_n=\Atom(n).
\]
With the convention of Definition~\ref{article-def:birth-layer}, the exceptional
case $n=2$ gives $\mathcal M_2=\varnothing=\Atom(2)$.
\end{corollary}

\begin{proof}
Let $m\in\mathcal M_n$ and let $\Theta$ be the corresponding $n$-witness
cover from Theorem~\ref{article-thm:witness-classification}. By
Corollary~\ref{article-cor:support-bound}, $\#\Theta\le\omega(n)=1$. Since the
$T_n$-sets of a witness cover must cover $\mathcal P(n)$, the cover has one
rank-jump factor, say $\Theta=\{\theta\}$. The coverage condition forces
$T_n(\theta)=\mathcal P(n)$, and because $d(\theta)\mid n$ this is
equivalent to $d(\theta)=n$. Hence $\theta\in\Atom(n)$ and
$m=\theta$.

Conversely, if $\theta\in\Atom(n)$ and $n\ne2$, then
$d(\theta)=n$, so $T_n(\theta)=\mathcal P(n)$. Since
$d^-(\theta)<d(\theta)=n$ and $\mathcal P(n)$ has one element, the unique
prime coordinate of $n$ lies in $E_n(\theta)$. Thus $\{\theta\}$ is an
$n$-witness cover, and Theorem~\ref{article-thm:witness-classification} gives
$\theta\in\mathcal M_n$. For $n=2$, Proposition~\ref{prop:prime-power-ladders}
and Corollary~\ref{cor:rank-jump-classification} show that no prime power has
rank $2$, while Definition~\ref{article-def:birth-layer} gives
$\mathcal M_2=\varnothing$.
\end{proof}

\begin{corollary}[Two-coordinate classification]
\label{cor:two-support-complete-classification}
Assume $\mathcal P(n)=\{\ell_1,\ell_2\}$ and identify
$\{\ell_1,\ell_2\}$ with $\{1,2\}$ in the notation
$\mathcal A_{I,J}(n)$ from Definition~\ref{article-def:rank-jump-families}. Also put
\[
\mathcal P_i(n):=\{\theta\in\mathcal A_{\{i\},\{i\}}(n):
\theta\text{ is a prime jump}\},
\qquad
\mathcal L_i(n):=\{\theta\in\mathcal A_{\{i\},\{i\}}(n):
\theta\text{ is a ladder jump}\}.
\]
Then $\mathcal M_n$ is the disjoint union of the following two families:
\[
\Atom(n)
\]
and
\[
\{\theta_1\theta_2:\ \theta_i\in
\mathcal A_{\{i\},\{i\}}(n)=\mathcal P_i(n)\cup\mathcal L_i(n),\
\gcd(\theta_1,\theta_2)=1\}.
\]
Equivalently, every non-rank-jump minimal generator is obtained by choosing,
for each coordinate, one prime or ladder jump whose rank first jumps on that
coordinate, with the two chosen prime powers coprime.
\end{corollary}

\begin{proof}
Let $m\in\mathcal M_n$, and write its witness cover as $\Theta$. If
$\#\Theta=1$, then its sole rank-jump factor has rank $n$, so $m\in\Atom(n)$.

Assume $\#\Theta>1$. By Corollary~\ref{article-cor:support-bound},
$\#\Theta\le2$, hence $\#\Theta=2$. The second assertion of
Lemma~\ref{thm:extremal-coordinate-separation} says, after relabeling the
two factors, that one rank-jump factor has full support $\{1\}$ and the other has full
support $\{2\}$. Since each rank-jump factor in a witness cover has a nonempty essential
coordinate not supplied by the other rank-jump factor, the first has
$E_n=T_n=\{1\}$ and the second has $E_n=T_n=\{2\}$. Thus the two rank-jump factors lie in
the displayed families, and pairwise coprimeness is part of the witness-cover
definition. Corollary~\ref{cor:rank-jump-classification} gives the stated
prime-or-ladder decomposition of each one-coordinate family.

Conversely, a single rank-jump factor in $\Atom(n)$ is an $n$-witness cover by itself,
because lowering it lowers its rank. If
$\theta_i\in\mathcal A_{\{i\},\{i\}}(n)$ are coprime, then their
$T$-sets cover $\{1,2\}$ and each rank-jump factor has its own essential coordinate.
Therefore $\{\theta_1,\theta_2\}$ is an $n$-witness cover, and
Theorem~\ref{article-thm:witness-classification} gives
$\theta_1\theta_2\in\mathcal M_n$. The two displayed families are disjoint
because their elements have respectively one and two prime-power factors.
\end{proof}

We now classify the three-support case. The abstract $\Gamma_i$ record only
the pairs $(E_n(\theta),T_n(\theta))$ up to coordinate relabeling. The lemmas
below exclude the one shape incompatible with Fibonacci prime-power arithmetic,
identify finite exact-rank and ladder slots for the remaining shapes, and apply
the distinct-underlying-prime selection criterion.

\begin{lemma}[Abstract three-coordinate skeletons]\label{lem:abstract-three-types}
Let $X=\{1,2,3\}$. Consider finite collections
\[
\Delta=\{(I_1,J_1),\dots,(I_r,J_r)\},\qquad
\varnothing\ne I_a\subseteq J_a\subseteq X,
\]
up to permutation of $X$ and reordering of the pairs, subject to
\[
\bigcup_a J_a=X,\qquad
I_a\nsubseteq\bigcup_{b\ne a}J_b\quad(1\le a\le r).
\]
Then $r\le3$, and every such abstract three-coordinate witness type is
equivalent to exactly one of the following nine labelled types:
\[
\begin{array}{ll}
\Gamma_1:\ \{(123,123)\},&
\Gamma_2:\ \{(12,123)\},\\
\Gamma_3:\ \{(1,123)\},&
\Gamma_4:\ \{(12,12),(23,23)\},\\
\Gamma_5:\ \{(1,12),(23,23)\},&
\Gamma_6:\ \{(1,12),(3,23)\},\\
\Gamma_7:\ \{(1,1),(23,23)\},&
\Gamma_8:\ \{(1,1),(2,23)\},\\
\Gamma_9:\ \{(1,1),(2,2),(3,3)\}.
\end{array}
\]
Here, for instance, $123$ denotes $\{1,2,3\}$.
\end{lemma}

\begin{proof}
For each pair choose
$x_a\in I_a\setminus\bigcup_{b\ne a}J_b$. Then $x_a\in J_a$ and the
$x_a$ are distinct, so $r\le\#X=3$.

If $r=1$, then $J_1=X$. Up to relabeling, $I_1$ has size $3$, $2$, or $1$,
giving $\Gamma_1$, $\Gamma_2$, and $\Gamma_3$.

If $r=2$, neither $J_1$ nor $J_2$ is contained in the other, because each
has a private coordinate outside the other. Thus the unordered pair
$(J_1,J_2)$ is, up to relabeling, either $(12,23)$ or $(1,23)$. In the
overlapping case $(12,23)$, the first private coordinate is $1$ and the
second is $3$. Hence, again up to reversing the two pairs, the possible
essential sets are $(12,23)$, $(1,23)$, or $(1,3)$, giving
$\Gamma_4$, $\Gamma_5$, and $\Gamma_6$. In the disjoint case $(1,23)$, the
first essential set must be $1$, while the second is either $23$ or a
singleton; these are $\Gamma_7$ and $\Gamma_8$.

If $r=3$, the three chosen private coordinates exhaust $X$. Since each
coordinate is absent from the two $J$-sets to which it is not assigned, no
$J_a$ can contain any coordinate other than its own private coordinate.
Thus the only possibility is
$\{(1,1),(2,2),(3,3)\}$, namely $\Gamma_9$.
The cases are mutually exclusive by $r$, by the intersection pattern of the
$J$-sets, and by the sizes of the $I$-sets after those data are fixed.
\end{proof}

\begin{lemma}[Fibonacci three-coordinate types]\label{lem:fibonacci-three-coordinate-types}
Assume $\omega(n)=3$. Every $n$-witness cover has one of the abstract types
\[
\Gamma_1,\Gamma_3,\Gamma_4,\Gamma_5,\Gamma_6,\Gamma_7,\Gamma_8,\Gamma_9,
\]
and no $n$-witness cover has type $\Gamma_2$.
\end{lemma}

\begin{proof}
Let $\Theta$ be an $n$-witness cover. The collection of pairs
\[
\{(E_n(\theta),T_n(\theta)):\theta\in\Theta\}
\]
satisfies the abstract hypotheses of
Lemma~\ref{lem:abstract-three-types}; coverage and irredundance are exactly
the last two conditions in Definition~\ref{article-def:witness-cover}. Hence its
type is one of $\Gamma_1,\ldots,\Gamma_9$.

By Lemma~\ref{thm:primitive-ladder-dichotomy}, a Fibonacci rank-jump factor has either
$E_n(\theta)=T_n(\theta)$, in the prime jump case, or
$E_n(\theta)$ empty or a singleton, in the ladder case. In a witness cover
$E_n(\theta)$ cannot be empty. Therefore no rank-jump factor can have
$1<\#E_n(\theta)<\#T_n(\theta)$. The only abstract type in the displayed
list requiring such a rank-jump factor is $\Gamma_2$, represented by $(12,123)$, so
$\Gamma_2$ is excluded.

This proves the asserted containment in the eight displayed types and the
exclusion of $\Gamma_2$.
\end{proof}

\begin{lemma}[Prime and ladder jump families in three coordinates]
\label{lem:three-coordinate-rank-jump-families}
Assume $\mathcal P(n)=\{\ell_1,\ell_2,\ell_3\}$ and identify subsets of
$[3]=\{1,2,3\}$ with the corresponding coordinate subsets of
$\mathcal P(n)$. For nonempty $J\subseteq[3]$ define the prime jump family
\[
P_J(n):=\{\theta\in\mathcal A_{J,J}(n):\theta\text{ is a prime}\},
\]
and, for $i\in J$, define the ladder family
\[
L_{i,J}(n):=\{\theta\in\mathcal A_{\{i\},J}(n):
\theta=p^e\text{ with }e>1\}.
\]
Then, for every nonempty $I\subseteq J\subseteq[3]$,
\[
\mathcal A_{I,J}(n)=
\begin{cases}
P_J(n), & I=J,\ \#J>1,\\[1mm]
P_{\{i\}}(n)\cup L_{i,\{i\}}(n), & I=J=\{i\},\\[1mm]
L_{i,J}(n), & I=\{i\}\subsetneq J,\\[1mm]
\varnothing, & \text{otherwise.}
\end{cases}
\]
\end{lemma}

\begin{proof}
Let $\theta\in\mathcal A_{I,J}(n)$. By definition, $\theta$ is a rank-jump factor with
$E_n(\theta)=I$ and $T_n(\theta)=J$. If $\theta=p$ is a prime jump, then
$d^-(\theta)=1$, so every full coordinate is lost after lowering; hence
$E_n(\theta)=T_n(\theta)$ and $\theta\in P_J(n)$. Conversely, every
$\theta\in P_J(n)$ has $(E_n(\theta),T_n(\theta))=(J,J)$ by definition.

Now suppose $\theta=p^e$ with $e>1$. The prime--ladder alternative
(Lemma~\ref{thm:primitive-ladder-dichotomy}) gives that $E_n(\theta)$ is
either empty or a singleton. Since $\mathcal A_{I,J}(n)$ is defined only for
nonempty $I$, any such rank-jump factor in it has $I=\{i\}$ for some $i\in J$, and it
lies in $L_{i,J}(n)$. Conversely, every member of $L_{i,J}(n)$ is a ladder
rank-jump factor with $(E_n(\theta),T_n(\theta))=(\{i\},J)$ by definition.

Combining the prime jump case with the ladder case gives the four displayed
alternatives. If $\#I>1$ and $I\ne J$, the prime jump case is impossible
because prime jumps have $I=J$, while the ladder case is impossible
because ladder jumps have singleton essential support. Hence the remaining
families are empty.
\end{proof}

\begin{lemma}[Finite slot-selection test]
\label{lem:finite-slot-selection}
Let $X$ be a set of at most three labelled slots.  For each $x\in X$, let
$C_x$ be a finite set of admissible prime powers.  Define
\[
 \nu(C_x):=\{p:\ p\text{ is the underlying prime of some }p^e\in C_x\}.
\]
There are choices $\theta_x\in C_x$ whose underlying primes are pairwise
distinct if and only if, for every $Y\subseteq X$,
\[
\#\bigcup_{y\in Y}\nu(C_y)\ge \#Y .
\]
Consequently, in the three-coordinate classification below, a row is decided
by the finite bipartite graph whose left vertices are the row slots and whose
right vertices are the admissible underlying primes supplied by the exact-rank
rank windows and the explicit ladder candidates.
\end{lemma}

\begin{proof}
Apply Hall's marriage theorem to the finite family of underlying-prime sets
$N_x:=\nu(C_x)$, with the slots as the left vertex class
\cite[Chapter~2]{LovaszPlummer1986MatchingTheory}. Its neighborhood condition
is exactly the displayed inequality, and a matching saturating all slots is
equivalent to choosing representatives with distinct underlying primes.
\end{proof}

\begin{lemma}[Arithmetic tests for the three-coordinate rank-jump factor families]
\label{lem:three-coordinate-arithmetic-tests}
Assume $\mathcal P(n)=\{\ell_1,\ell_2,\ell_3\}$, and write
$a_i:=\nu_{\ell_i}(n)$. For nonempty $J\subseteq[3]$ put
\[
\mathcal R_J(n):=\{d\mid n:\ \{k:\nu_{\ell_k}(d)=a_k\}=J\},
\qquad
\mathcal R_J^\circ(n):=\mathcal R_J(n)\setminus\{1,2,6,12\}.
\]
Use the rank classes $\Pi_\alpha(d)$ from
Lemma~\ref{article-lem:primitive-prime-mobius-count}.
Then
\[
P_J(n)=\bigcup_{d\in\mathcal R_J^\circ(n)}\Pi_\alpha(d),
\]
where a prime $p$ is identified with the prime jump $p$. In particular
$P_J(n)\ne\varnothing$ if and only if $\mathcal R_J^\circ(n)\ne\varnothing$.
More generally, after forbidding a finite set $S$ of primes,
$P_J(n)\setminus S\ne\varnothing$ if and only if there exists
$d\in\mathcal R_J^\circ(n)$ such that
\[
\#\Pi_\alpha(d)>
\#\{p\in S:\alpha(p)=d\},
\qquad
\#\Pi_\alpha(d)=\sum_{c\mid d}\mu(d/c)\,\omega(F_c).
\]

For the ladder families, define a candidate set $\mathcal L_i^*(n)$ as follows.
If $\ell_i\notin\{2,5\}$, write
$z_i:=\alpha(\ell_i)$, $h_i:=\nu_{\ell_i}(F_{z_i})$, and
$s_i:=\nu_{\ell_i}(z_i)$. Then
\[
\mathcal L_i^*(n):=
\begin{cases}
\{4\}, & \ell_i=2,\ a_i=1,\\
\{2^{a_i+2}\}, & \ell_i=2,\ a_i\ge2,\\
\{5^{a_i}\}, & \ell_i=5,\ a_i\ge2,\\
\varnothing, & \ell_i=5,\ a_i=1,\\
\{\ell_i^{h_i+a_i-s_i}\}, &
   \ell_i\notin\{2,5\}\text{ and }a_i>s_i,\\
\varnothing, &
   \ell_i\notin\{2,5\}\text{ and }a_i\le s_i .
\end{cases}
\]
For each $i\in J$,
\[
L_{i,J}(n)=
\{\lambda\in\mathcal L_i^*(n):\alpha(\lambda)\mid n,\ 
\{k:\nu_{\ell_k}(\alpha(\lambda))=a_k\}=J\}.
\]
Thus every ladder family $L_{i,J}(n)$ is empty or consists of the single
displayed prime power.  Hence every labelled three-coordinate slot has an
explicit finite admissible set: a prime slot is the finite disjoint union
of the rank classes $\Pi_\alpha(d)$ with
$d\in\mathcal R_J^\circ(n)$, and a ladder slot is decided by the displayed
single candidate.  Pairwise-coprime filling of any row is therefore exactly
the finite slot-selection test of Lemma~\ref{lem:finite-slot-selection}; the
exact-rank prime multiplicities are
$\#\Pi_\alpha(d)=\sum_{c\mid d}\mu(d/c)\omega(F_c)$.
\end{lemma}

\begin{proof}
Let $p\in P_J(n)$ and put $d=\alpha(p)$. Since $p$ is a prime jump,
$E_n(p)=T_n(p)$, and the condition $T_n(p)=J$ says exactly that
$d\in\mathcal R_J(n)$. No prime has rank $1$ or $2$, because
$F_1=F_2=1$. Also no prime has rank $6$ or $12$: the prime divisors of
$F_6=8$ and $F_{12}=144$ are $2$ and $3$, with
$\alpha(2)=3$ and $\alpha(3)=4$. Hence $d\in\mathcal R_J^\circ(n)$.
Conversely, if $d\in\mathcal R_J^\circ(n)$ and $p\in\Pi_\alpha(d)$, then
$d\mid n$ and the full-coordinate set of $d$ is $J$; since $p$ is a prime jump,
$p\in P_J(n)$. This proves the displayed identity for $P_J(n)$.

The nonemptiness equivalence is precisely where the Fibonacci primitive-divisor
theorem enters. If $P_J(n)$ is nonempty, the identity just proved gives a rank
$d\in\mathcal R_J^\circ(n)$. Conversely, if
$\mathcal R_J^\circ(n)$ is nonempty and $d$ is any element of it, then by
definition $d\mid n$ and $d\notin\{1,2,6,12\}$. The only possible
exact-rank obstructions are exactly $1,2,6,12$: ranks $1$ and $2$ occur for
no prime, ranks $6$ and $12$ occur for no prime as checked above, and
Theorem~\ref{article-thm:primitive-divisors} supplies a prime $p$ with
$\alpha(p)=d$ for every remaining $d\ge3$. Thus
$\Pi_\alpha(d)$ is nonempty for every $d\in\mathcal R_J^\circ(n)$, and
$P_J(n)$ is nonempty exactly when $\mathcal R_J^\circ(n)$ is nonempty. The
version with a forbidden finite set $S$ is the same rank-by-rank argument after
removing the primes of $S$: some rank $d$ remains fillable exactly when the
number of available primes in $\Pi_\alpha(d)$ exceeds the number of forbidden
primes of that rank. The formula for $\#\Pi_\alpha(d)$ is
Lemma~\ref{article-lem:primitive-prime-mobius-count}.

It remains to identify the ladder families. Let
$\theta=p^e\in L_{i,J}(n)$. By
Lemma~\ref{thm:primitive-ladder-dichotomy}, the essential coordinate of a
ladder jump is the prime coordinate $p$ itself; hence $p=\ell_i$. The
prime-power formulas in Proposition~\ref{prop:prime-power-ladders} now leave
only one possible exponent once the $\ell_i$-adic coordinate of
$\alpha(p^e)$ is required to equal $a_i$. If $\ell_i=2$, the rank-jump ladder
levels are $4$ of rank $6$ and $2^e$ of rank $3\cdot2^{e-2}$ for $e\ge4$;
therefore the candidate is $4$ when $a_i=1$ and $2^{a_i+2}$ when $a_i\ge2$.
If $\ell_i=5$, the rank is $5^e$, so a nonprime ladder jump with full
$5$-coordinate $a_i$ exists only for $a_i\ge2$, and then it is $5^{a_i}$.
Finally suppose $\ell_i\notin\{2,5\}$. At rank-jump ladder levels,
\[
\alpha(\ell_i^e)=\ell_i^{e-h_i}z_i\qquad(e>h_i),
\]
so the full-coordinate equation
$\nu_{\ell_i}(\alpha(\ell_i^e))=a_i$ forces
$e=h_i+a_i-s_i$, and this is a rank-jump ladder exponent precisely when
$a_i>s_i$.

Thus every member of $L_{i,J}(n)$ lies in $\mathcal L_i^*(n)$. The remaining
conditions for membership in $L_{i,J}(n)$ are exactly that its rank divide
$n$ and that its full-coordinate set be $J$. Conversely, if
$\lambda\in\mathcal L_i^*(n)$ satisfies those two displayed conditions, the
same prime-power formulas show that $\lambda$ is a rank-jump ladder. Lowering
$\lambda$ reduces only the $\ell_i$-coordinate of its rank, so
$E_n(\lambda)=\{i\}$ and $T_n(\lambda)=J$. Hence
$\lambda\in L_{i,J}(n)$.
The displayed exact-rank windows and ladder candidate sets also exhaust the
exceptional cases: $2$ and $5$ are handled by their separate ladder lines,
the ranks $1,2,6,12$ are removed from exact-rank windows, and any
Wall--Sun--Sun plateau for $p\ne2,5$ is already absorbed in the parameter
$h_i=\nu_{\ell_i}(F_{z_i})$ before the single possible exponent
$h_i+a_i-s_i$ is formed.
\end{proof}
