\section{Connected-dominance criteria}
\label{sec:supp-connected-dominance}

\begin{lemma}[Refined private-coordinate upper bound]
\label{lem:refined-private-coordinate-upper-bound}
For $m\ge2$, put $s=\omega(m)$ and $r=R(m)$. Then
\begin{equation}\label{eq:refined-private-coordinate-upper-bound}
 \#\mathcal M_m\le U_s(r),\qquad
 U_s(r):=\sum_{t=1}^s\binom st
             \bigl(r2^{s-t}+1\bigr)^t.
\end{equation}
There is an absolute constant $K$ such that, for all $s,r\ge1$,
\begin{equation}\label{eq:refined-upper-versus-covers}
 U_s(r)\le Kr^sC_s.
\end{equation}
\end{lemma}

\begin{proof}
The case $m=2$ is immediate from $\mathcal M_2=\varnothing$ and $R(2)=1$.
Suppose $m\ge3$, and let a witness cover have $t$ atoms. Choose the canonical
private coordinate of each atom as in the proof of
Theorem~\ref{article-thm:minimal-generator-growth}; let $P$ be the resulting
$t$-set and $Q$ its complement. For a fixed $x\in P$, a prime atom has full
support $\{x\}\cup U$ for some $U\subseteq Q$, giving at most
$r2^{s-t}$ choices. A ladder atom with essential coordinate $x$ must be a
power of the index prime $x$, and the full-exponent equation in
Proposition~\ref{prop:prime-power-ladders} determines at most one such atom
over all its possible full supports. Thus the atom assigned to $x$ has at most
$r2^{s-t}+1$ choices. Summing over $P$ and $t$ proves
\eqref{eq:refined-private-coordinate-upper-bound}; ignoring coverage and
coprimeness only enlarges the count.

For $r\ge1$,
\[
 U_s(r)\le r^sU_s(1).
\]
If $t\le3s/4$, then, uniformly in this range,
\[
 (2^{s-t}+1)^t
 =2^{t(s-t)}(1+2^{-(s-t)})^t
 =(1+o(1))2^{t(s-t)}.
\]
The sum over $t>3s/4$ is
$2^{-s^2/16+O(s)}$ times a central term of
$b_s=\sum_t\binom st2^{t(s-t)}$. Hence $U_s(1)\sim b_s$. The published
asymptotic \eqref{article-eq:minimal-cover-theta} gives $C_s\sim b_s$; enlarging a
constant to cover the finitely many small values of $s$ proves
\eqref{eq:refined-upper-versus-covers}.
\end{proof}

\begin{theorem}[Blockwise criterion for connected dominance]
\label{thm:blockwise-connected-dominance}
Let $(n_j)$ be a sequence, put $V_j=\mathcal P(n_j)$ and
$k_j=\omega(n_j)\to\infty$, and for nonempty $C\subseteq V_j$ put
\[
 (n_j)_C:=\prod_{\ell\in C}\ell^{\nu_\ell(n_j)}.
\]
For a nontrivial partition $\Pi\in\mathsf{Part}(V_j)$ define
\[
 \Delta(\Pi):=\sum_{\substack{C,D\in\Pi\\C<D}}|C||D|
 =\frac{k_j^2-\sum_{C\in\Pi}|C|^2}{2}.
\]
Suppose
\begin{equation}\label{eq:blockwise-window-condition}
 \varepsilon_j:=
 \max_{\substack{\Pi\in\mathsf{Part}(V_j)\\|\Pi|\ge2}}
 \frac{\sum_{C\in\Pi}|C|\log R((n_j)_C)}{\Delta(\Pi)}
 \longrightarrow0\qquad\text{\rm(BW)}.
\end{equation}
Then
\begin{equation}\label{eq:connected-dominance-conclusion}
 \frac{\#(\mathcal M_{n_j}\setminus\mathcal M_{n_j}^{\rm conn})}
 {\#\mathcal M_{n_j}^{\rm conn}}
 \le\exp\left[-\left(\frac{\log2}{2}-o(1)\right)k_j\right]
 =o(1).
\end{equation}
A sufficient condition for \eqref{eq:blockwise-window-condition} is
\begin{equation}\label{eq:blockwise-window-sufficient-condition}
 \max_{\varnothing\ne C\subsetneq V_j}
 \frac{\log R((n_j)_C)}{k_j-|C|}\longrightarrow0
 \qquad\text{\rm(BW$'$)}.
\end{equation}
In particular, \eqref{eq:connected-dominance-conclusion} holds if
$R((n_j)_C)=1$ for every nonempty proper block $C$ and all sufficiently
large $j$.
\end{theorem}

\begin{proof}
The unique connected-block factorization in
Theorem~\ref{thm:mingenerator-connected-factorization} gives
\[
 \#(\mathcal M_{n_j}\setminus\mathcal M_{n_j}^{\rm conn})
 =\sum_{\substack{\Pi\in\mathsf{Part}(V_j)\\|\Pi|\ge2}}
   \prod_{C\in\Pi}M^{\rm conn}((n_j)_C).
\]
Apply Lemma~\ref{lem:refined-private-coordinate-upper-bound} to each block.
The theta asymptotic \eqref{article-eq:minimal-cover-theta} gives uniform two-sided
bounds
\[
 C_s\asymp(s+1)^{-1/2}2^{s^2/4+s}.
\]
Consequently there is an absolute constant $K_1$ such that every partition
$\Pi$ with block sizes $s_C=|C|$ contributes, after division by $C_{k_j}$,
at most
\begin{equation}\label{eq:partition-contribution-bound}
 K_1^{|\Pi|}\sqrt{k_j}\,
 \exp\left\{-\frac{\log2}{2}\Delta(\Pi)
 +\sum_{C\in\Pi}|C|\log R((n_j)_C)\right\}.
\end{equation}
Under \textup{(BW)}, this is
\[
 K_1^{|\Pi|}\sqrt{k_j}\,
 \exp\left\{-\left(\frac{\log2}{2}-o(1)\right)
 \Delta(\Pi)\right\},
\]
where the $o(1)$ is uniform over all nontrivial partitions.

If every block has size at most $k_j/2$, then
$\Delta(\Pi)\ge k_j^2/4$. There are at most $k_j^{k_j}$ set partitions,
so the total of these terms is $\exp(-\Omega(k_j^2))$. Otherwise there is a
unique largest block of size $k_j-q>k_j/2$. Then
$\Delta(\Pi)\ge q(k_j-q)$, and the remaining $q$ vertices can be selected and
partitioned in at most
\[
 \binom{k_j}{q}\mathsf B_q\le(ek_j)^q
\]
ways. Summing \eqref{eq:partition-contribution-bound} over
$1\le q<k_j/2$ therefore gives
\[
 \exp\left[-\left(\frac{\log2}{2}-o(1)\right)k_j\right]C_{k_j}.
\]
Corollary~\ref{cor:theta-refined-connected-entropy} gives
$M^{\rm conn}(n_j)\ge(1-o(1))C_{k_j}$, proving
\eqref{eq:connected-dominance-conclusion}.

Finally, if the maximum in \textup{(BW$'$)} is $\eta_j$, then every block of
a nontrivial partition is proper and
\[
 \sum_{C\in\Pi}|C|\log R((n_j)_C)
 \le\eta_j\sum_{C\in\Pi}|C|(k_j-|C|)
 =2\eta_j\Delta(\Pi).
\]
Thus \textup{(BW$'$)} implies \textup{(BW)}, and the last assertion is the
case $\eta_j=0$.
\end{proof}
