\section{Rank-pure secondary asymptotics}
\label{sec:supp-rank-pure-asymptotics}

\begin{lemma}[Connected minimal-cover concentration]
\label{lem:connected-minimal-cover-concentration}
If $C_k^{\rm conn}$ denotes the number of connected minimal covers, then
\begin{equation}\label{eq:connected-minimal-cover-asymptotic}
 C_k-C_k^{\rm conn}=O(kC_{k-1})=o(C_k).
\end{equation}
\end{lemma}

\begin{proof}
Connected components of a minimal cover are themselves minimal covers, so the
labelled exponential formula gives
\[
 C_k=\sum_{j=1}^k\binom{k-1}{j-1}C_j^{\rm conn}C_{k-j}.
\]
The terms $j=1$ and $j=k-1$ contribute $O(kC_{k-1})$. Choosing the least
private element in each cover member gives $C_j\le b_j\le2^j2^{j^2/4}$.
For the remaining terms, use this bound and the central lower bound for
$C_{k-1}$ furnished by the $s=\lfloor(k-1)/2\rfloor$ term of
\eqref{article-eq:hearne-wagner}. After division by $kC_{k-1}$, the term indexed by
$j$ is
\[
 \ll \binom{k}{j}
 2^{-(j-1)(k-j-1)/2}\qquad(2\le j\le k-2).
\]
The sum of these bounds is bounded (and in fact tends to zero), by symmetry
and separation into $2\le j\le k/2$ and its reflection. This proves the first
estimate in \eqref{eq:connected-minimal-cover-asymptotic}; the second equality
follows from \eqref{article-eq:minimal-cover-theta}.
\end{proof}

\begin{corollary}[Theta-refined connected support entropy]
\label{cor:theta-refined-connected-entropy}
Uniformly along every sequence of integers for which $k=\omega(n)\to\infty$,
\begin{equation}\label{eq:theta-connected-lower-bound}
 \#\mathcal M_n^{\rm conn}\ge
 (1-o(1))\,
 \vartheta_\varepsilon\binom{k}{\lfloor k/2\rfloor}2^{k^2/4},
 \qquad \varepsilon\equiv k\pmod2.
\end{equation}
Consequently
\begin{equation}\label{eq:theta-connected-log-lower-bound}
 \log\#\mathcal M_n^{\rm conn}\ge
 \frac{\log2}{4}k^2+k\log2-\frac12\log k
 +\log\vartheta_\varepsilon+\frac12\log\frac2\pi+o(1).
\end{equation}
The same lower bounds hold with $\mathcal M_n$ in place of
$\mathcal M_n^{\rm conn}$.
\end{corollary}

\begin{proof}
Fix an exceptional support $S_0\in\mathcal E_n$. If a minimal cover contains
$S_0$, choose a canonical private vertex $x\in S_0$ of that member. Every
other member avoids $x$ and is recovered from its canonical private vertex and
an arbitrary subset of the remaining nonprivate vertices. The number of such
encodings is at most $b_{k-1}$ for each choice of $x$. Therefore the number of
minimal covers meeting $\mathcal E_n$ is at most
\[
 \sum_{S_0\in\mathcal E_n}|S_0|b_{k-1}\le4b_{k-1}=o(C_k).
\]
Lemma~\ref{lem:connected-minimal-cover-concentration} also shows that only
$o(C_k)$ minimal covers are disconnected. Thus $(1-o(1))C_k$ covers are simultaneously
connected and disjoint from $\mathcal E_n$. Theorem
\ref{article-thm:rank-pure-universality} embeds them in
$\mathcal M_n^{\rm conn}$, proving \eqref{eq:theta-connected-lower-bound}.
Taking logarithms and using
\[
 \log\binom{k}{\lfloor k/2\rfloor}
 =k\log2-\frac12\log k+\frac12\log\frac2\pi+o(1)
\]
gives \eqref{eq:theta-connected-log-lower-bound}.
\end{proof}

\begin{corollary}[Discrete local limit in a fixed-prime rank-pure sector]
\label{cor:rank-pure-local-limit}
Let $(n_j)$ be any sequence with $k_j=\omega(n_j)\to\infty$. For every
$j$ and every nonexceptional nonempty support
$S\subseteq\mathcal P(n_j)$, fix one prime
$\pi_{j,S}\in\Pi_\alpha((n_j)_S)$. Let $\mathcal Q_{n_j}^{\rm rp}$ consist of the
products $\prod_{S\in\mathcal C}\pi_{j,S}$ over all
$\mathcal C\in\operatorname{MinCov}(\mathcal P(n_j))$ disjoint from
$\mathcal E_{n_j}$, and choose $m$ uniformly from $\mathcal Q_{n_j}^{\rm rp}$. Put
$W_j=\omega(m)$. Along the subsequence $k_j=2q_j$, for every fixed
$d\in\mathbb Z$,
\begin{equation}\label{eq:even-rank-pure-local-limit}
 \Pr(W_j=q_j+d)\longrightarrow\frac{2^{-d^2}}{\vartheta_0}.
\end{equation}
Along the subsequence $k_j=2q_j+1$,
\begin{equation}\label{eq:odd-rank-pure-local-limit}
 \Pr(W_j=q_j+d)\longrightarrow
 \frac{2^{-(d-1/2)^2}}{\vartheta_1}.
\end{equation}
In particular, $W_j=k_j/2+O_{\mathbb P}(1)$.
\end{corollary}

\begin{proof}
Unique factorization and distinct support ranks show that $W_j$ is the number
of members of the corresponding cover. By the proof of
Corollary~\ref{cor:theta-refined-connected-entropy}, deleting covers meeting
$\mathcal E_{n_j}$ changes the uniform distribution in total variation by
$o(1)$. For $s=\lfloor k/2\rfloor+d$ with fixed $d$, the $j=s$ term of
\eqref{article-eq:hearne-wagner} and the canonical-private-element encoding give
\[
 \binom{k}{s}(2^s-s-1)^{k-s}
 \le C_{k,s}\le \binom{k}{s}2^{s(k-s)},
 \qquad
 C_{k,s}\sim\binom{k}{s}2^{s(k-s)}.
\]
Dividing this by \eqref{article-eq:minimal-cover-theta} gives
\eqref{eq:even-rank-pure-local-limit} and
\eqref{eq:odd-rank-pure-local-limit}. The quadratic limiting tails are
summable, which is equivalent to tightness on the integer scale.
\end{proof}

\begin{remark}\label{rem:connected-lower-not-dominance}
Corollary~\ref{cor:theta-refined-connected-entropy} proves that connected
minimal generators alone realize the theta-refined lower bound supplied by the
unweighted minimal-cover count in the fixed-choice rank-pure sector, including
its linear and logarithmic corrections. It does not compare the total connected
and disconnected parts of $\mathcal M_n$, nor does it assert that this sector
exhausts the consequences of fibotomic information. Such a comparison would
require upper control of the arithmetic multiplicities in $R(n)$ and is not
presently available.
\end{remark}
