\section{Rank level sets and witness covers}\label{sec:upper-fibers}

Let $D_n$ denote the divisor set of $F_n$.

\begin{lemma}[Upper-fiber identity]\label{thm:upper-fiber}
For every $n\geq2$,
\[
 B_n=D_n\setminus\bigcup_{p\mid n}D_{n/p}.
\]
Moreover, $B_n$ is an upper set in $D_n$, and therefore
\[
 B_n=\bigcup_{m\in\mathcal M_n}\{q\in D_n:m\mid q\}.
\]
\end{lemma}

\begin{proof}
By Proposition~\ref{prop:rank-basics}, $q\in D_n$ lies outside the union
exactly when $\alpha(q)\mid n$ but $\alpha(q)\nmid n/p$ for every
$p\mid n$, which is equivalent to $\alpha(q)=n$.  If $q\mid r$ with
$q\in B_n$ and $r\in D_n$, then
$n=\alpha(q)\mid\alpha(r)\mid n$, so $r\in B_n$.  The last display follows
because the finite upper set $B_n$ is generated by its minimal elements.
\end{proof}

\begin{lemma}[Minimal-lcm and coordinate criterion]
\label{thm:minimal-generators}
Let $n\geq2$, and let $m=\prod_{i=1}^r p_i^{e_i}$ divide $F_n$, where the
$p_i$ are distinct.  Put
\[
 \lambda_i:=\alpha(p_i^{e_i}),\qquad
 \lambda_i^-:=\alpha(p_i^{e_i-1}).
\]
Then $m\in\mathcal M_n$ if and only if
\begin{equation}\label{eq:lcm-criterion}
 \lcmop(\lambda_1,\ldots,\lambda_r)=n
\quad\text{and}\quad
 \lcmop(\lambda_1,\ldots,\lambda_i^-,\ldots,\lambda_r)<n
 \quad(1\leq i\leq r).
\end{equation}
If the $p_i^{e_i}$ are rank jumps and the first lcm is $n$, then for each
$i$ the second lcm equals $n$ if and only if
\[
 E_n(p_i^{e_i})\subseteq
 \bigcup_{j\ne i}T_n(p_j^{e_j}).
\]
\end{lemma}

\begin{proof}
Repeated use of Proposition~\ref{prop:rank-basics} gives
\[
 \alpha(m)=\lcmop(\lambda_1,\ldots,\lambda_r),\qquad
 \alpha(m/p_i)=
 \lcmop(\lambda_1,\ldots,\lambda_i^-,\ldots,\lambda_r).
\]
If $m$ is minimal in $B_n$, the first rank is $n$ and every second rank is
smaller.  Conversely, if \eqref{eq:lcm-criterion} holds, every proper divisor
of $m$ divides some $m/p_i$ and hence has rank smaller than $n$.

For the coordinate assertion, all ranks in the lowered lcm divide $n$.
The lcm remains $n$ precisely when every prime coordinate lost on replacing
$\lambda_i$ by $\lambda_i^-$ is supplied at full exponent by one of the
other $\lambda_j$.  This is exactly the displayed containment.
\end{proof}

\begin{proposition}[Wagner comparison]\label{prop:wagner-comparison}
Let $n\geq2$, and let $\Theta=\{p_1,\ldots,p_r\}$ be distinct prime rank
jumps with $d_i=\alpha(p_i)\mid n$.  Then $\Theta$ is an $n$-witness cover
if and only if
\begin{equation}\label{eq:wagner-minimal-multiplicative-cover}
 \lcmop(d_1,\ldots,d_r)=n,\qquad
 \lcmop(d_1,\ldots,\widehat{d_i},\ldots,d_r)<n
 \quad(1\leq i\leq r).
\end{equation}
Thus the prime-jump case is the unordered form of Wagner's minimal
multiplicative cover condition.  For arbitrary pairwise coprime rank jumps
$\Theta=\{\theta_1,\ldots,\theta_r\}$, deletion is replaced by
\begin{equation}\label{eq:lowered-label-multiplicative-cover}
 \lcmop(d(\theta_1),\ldots,d^-(\theta_i),\ldots,d(\theta_r))<n
 \quad(1\leq i\leq r).
\end{equation}
\end{proposition}

\begin{proof}
For a prime jump $p_i$, one has $d^-(p_i)=\alpha(1)=1$, so lowering the
label is the same as omitting it.  Lemma~\ref{thm:minimal-generators}
identifies the witness condition with
\eqref{eq:wagner-minimal-multiplicative-cover}.  The same lemma gives
\eqref{eq:lowered-label-multiplicative-cover} for arbitrary rank jumps.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:witness-classification}]
Let $m\in\mathcal M_n$, and write
$m=\prod_{i=1}^r\theta_i$ with
$\theta_i=p_i^{e_i}$.  Each $\theta_i$ is a rank jump: otherwise lowering
its exponent would preserve the apparition lcm, contrary to
Lemma~\ref{thm:minimal-generators}.  The identity
\[
 n=\lcmop(d(\theta_1),\ldots,d(\theta_r))
\]
says that the $T_n(\theta_i)$ cover $\mathcal P(n)$.  The lowered lcm is
smaller for every $i$, so the coordinate criterion supplies
\[
 E_n(\theta_i)\nsubseteq\bigcup_{j\ne i}T_n(\theta_j).
\]
The prime-power factors are pairwise coprime, and hence form an
$n$-witness cover.

Conversely, let $\Theta$ be an $n$-witness cover and put
$m=\prod_{\theta\in\Theta}\theta$.  Coverage gives
$\alpha(m)=\lcmop(d(\theta):\theta\in\Theta)=n$.  For each
$\theta\in\Theta$, a coordinate in
\[
 E_n(\theta)\setminus
 \bigcup_{\eta\in\Theta\setminus\{\theta\}}T_n(\eta)
\]
is lost when $\theta$ is lowered.  Lemma~\ref{thm:minimal-generators}
therefore gives $m\in\mathcal M_n$.

If two witness covers have the same product, unique factorization gives the
same underlying primes with the same exponents.  Their pairwise coprime
prime-power factors are consequently identical.  This proves injectivity and
identifies the inverse as stated.
\end{proof}

\begin{corollary}\label{cor:support-bound}
Every $m\in\mathcal M_n$ satisfies $\omega(m)\leq\omega(n)$.
\end{corollary}

\begin{proof}
For each factor $\theta$ in the witness cover of $m$, choose a coordinate in
\[
 E_n(\theta)\setminus
 \bigcup_{\eta\ne\theta}T_n(\eta).
\]
These coordinates are distinct, giving an injection from the prime-power
factors of $m$ into $\mathcal P(n)$.
\end{proof}
