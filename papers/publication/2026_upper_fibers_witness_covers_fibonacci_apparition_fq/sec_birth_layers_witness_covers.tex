\section{Birth layers and witness covers}\label{sec:upper-fibers}
%% ====================================================================

The first two lemmas in this section use only rank divisibility and the lcm
identity. They are included to fix the structural framework and are not
presented as new Fibonacci identities. The witness-cover theorem packages
their minimality conditions for the subsequent Fibonacci analysis.
The underlying deletion-one-factor condition is classical: Wagner called
divisor tuples satisfying it minimal multiplicative covers of an integer
\cite{Wagner1978MultiplicativeCovers}, as a multiplicative generalization of
the minimal set covers studied by Hearne--Wagner
\cite{HearneWagner1973MinimalCovers}. Proposition~
\ref{prop:wagner-comparison} states the exact boundary between that skeleton
and the lowered-label structure used here.

\begin{definition}\label{def:birth-layer}
For $n\ge2$ let
\[
D_n:=\operatorname{Div}(F_n),\qquad
B_n:=\{q\in D_n:\ \alpha(q)=n\},\qquad
A(n):=\#B_n.
\]
The set $B_n$ is partially ordered by divisibility. Its set of minimal
elements is
\[
\mathcal M_n:=\{m\in B_n:\ \text{there is no }q\in B_n
\text{ with }q\mid m\text{ and }q<m\},
\]
and
\[
\widthop(B_n):=\max\{\#\mathcal A:\ \mathcal A\subseteq B_n
\text{ is an antichain}\}.
\]
We write $\mathcal P(n)$ for the set of prime divisors of $n$. Since
$F_2=1$, we use the boundary convention
\[
B_2=\mathcal M_2=\varnothing,\qquad A(2)=\widthop(B_2)=0.
\]
\end{definition}

\begin{lemma}[Upper-fiber identity]\label{thm:upper-fiber}
For every $n\ge2$,
\[
B_n=D_n\setminus\bigcup_{p\mid n}D_{n/p}.
\]
Moreover $B_n$ is an upper set in $D_n$: if $q\in B_n$, $r\in D_n$, and
$q\mid r$, then $r\in B_n$. Consequently
\[
B_n=\bigcup_{m\in\mathcal M_n}\{q\in D_n:\ m\mid q\}.
\]
\end{lemma}

\begin{proof}
By Proposition~\ref{prop:rank-basics}, an integer $q$ lies in $D_m$ if and
only if $\alpha(q)\mid m$. Thus $q\in D_n$ lies outside
$\bigcup_{p\mid n}D_{n/p}$ precisely when $\alpha(q)\mid n$ but
$\alpha(q)\nmid n/p$ for every prime $p\mid n$.

If $\alpha(q)=n$, then clearly $\alpha(q)\nmid n/p$ for every $p\mid n$, so
$q$ is outside the union. Conversely, suppose $\alpha(q)\mid n$ and
$\alpha(q)<n$. Then $n/\alpha(q)>1$; choose a prime $p$ dividing
$n/\alpha(q)$. Since $\alpha(q)\mid n/p$, Proposition~\ref{prop:rank-basics}
gives $q\in D_{n/p}$, contrary to being outside the union. Hence the
complement is exactly the set of $q\in D_n$ with $\alpha(q)=n$, namely
$B_n$. For $n=2$ this also gives
$B_2=D_2\setminus D_1=\varnothing$ under the boundary convention in
Definition~\ref{def:birth-layer}.

Now let $q\in B_n$, $r\in D_n$, and $q\mid r$. Since $r\in D_n$,
$\alpha(r)\mid n$. Since $q\mid r$, every Fibonacci number divisible by $r$
is divisible by $q$; applying the divisibility criterion to
$F_{\alpha(r)}$ gives $\alpha(q)\mid\alpha(r)$. But $\alpha(q)=n$, so
$n\mid\alpha(r)$ and $\alpha(r)\mid n$, whence $\alpha(r)=n$. Therefore
$r\in B_n$.

Finally $D_n$ is finite, so every element $q\in B_n$ has at least one
minimal element $m$ of $B_n$ below it in the divisibility order. The upper-set
property gives the reverse containment: every element of $D_n$ divisible by
some $m\in\mathcal M_n$ lies again in $B_n$. This proves the displayed
decomposition.
\end{proof}

\begin{lemma}[Minimal-lcm criterion]\label{thm:minimal-generators}
Let $n\ge2$, and let
$m=\prod_{i=1}^r p_i^{e_i}$ be a divisor of $F_n$, with the $p_i$ distinct
and $e_i\ge1$. Put
\[
\lambda_i:=\alpha(p_i^{e_i}),\qquad
\lambda_i^-:=\alpha(p_i^{e_i-1}),
\]
where $\alpha(1)=1$. Then $m\in\mathcal M_n$ if and only if
\begin{equation}\label{eq:lcm-criterion}
\lcmop(\lambda_1,\ldots,\lambda_r)=n
\quad\text{and}\quad
\lcmop(\lambda_1,\ldots,\lambda_{i-1},\lambda_i^-,
\lambda_{i+1},\ldots,\lambda_r)<n
\quad(1\le i\le r).
\end{equation}
\end{lemma}

\begin{proof}
Because $m\mid F_n$, Proposition~\ref{prop:rank-basics} gives
$\lambda_i\mid n$ for every $i$. Applying the lcm identity in
Proposition~\ref{prop:rank-basics} repeatedly gives
\[
\alpha(m)=\lcmop(\lambda_1,\ldots,\lambda_r).
\]
Thus the first condition in \eqref{eq:lcm-criterion} is equivalent to
$m\in B_n$.

Assume first that $m\in\mathcal M_n$. Then $m\in B_n$, so the first lcm is
$n$. For a fixed $i$, the integer $m/p_i$ is a proper divisor of $m$ and is
still a divisor of $F_n$. Minimality of $m$ in $B_n$ implies
$m/p_i\notin B_n$. Since $\alpha(m/p_i)\mid n$, this is equivalent to
$\alpha(m/p_i)<n$. A second application of the lcm identity gives
\[
\alpha(m/p_i)=
\lcmop(\lambda_1,\ldots,\lambda_{i-1},\lambda_i^-,
\lambda_{i+1},\ldots,\lambda_r),
\]
which is the lowered prime-power condition in
\eqref{eq:lcm-criterion}.

Conversely, suppose \eqref{eq:lcm-criterion} holds. The first condition gives
$m\in B_n$. Let $q$ be any proper divisor of $m$. Then for at least one
index $i$ the exponent of $p_i$ in $q$ is at most $e_i-1$. Hence
\[
\alpha(q)\mid
\lcmop(\lambda_1,\ldots,\lambda_{i-1},\lambda_i^-,
\lambda_{i+1},\ldots,\lambda_r),
\]
again by the lcm identity and by monotonicity under divisibility of prime
powers. The right-hand side is $<n$ by
\eqref{eq:lcm-criterion}. Since $q\mid m\mid F_n$, we also have
$\alpha(q)\mid n$, and therefore $\alpha(q)<n$. Thus no proper divisor of
$m$ lies in $B_n$, so $m$ is minimal in $B_n$.
\end{proof}

\subsection*{Witness covers}

\begin{definition}\label{def:witness-cover}
An \emph{$n$-witness cover} is a finite set
$\Theta=\{\theta_1,\ldots,\theta_r\}$ of atomic prime powers satisfying:
\begin{enumerate}[label=(\roman*)]
\item $d(\theta)\mid n$ for every $\theta\in\Theta$;
\item the elements of $\Theta$ are pairwise coprime;
\item the full-coordinate supports cover all prime coordinates of $n$,
\[
\bigcup_{\theta\in\Theta}T_n(\theta)=\mathcal P(n);
\]
\item each atom has an essential coordinate not supplied by the other atoms,
\[
E_n(\theta)\nsubseteq
\bigcup_{\eta\in\Theta,\ \eta\ne\theta}T_n(\eta)
\qquad(\theta\in\Theta).
\]
\end{enumerate}
The last condition is the irredundance condition. It is stated with
$E_n$ rather than $T_n$ because overlapping full-coordinate support is
allowed; what is forbidden is that every coordinate lost by lowering
$\theta$ is already supplied at full exponent by the remaining atoms.
\end{definition}

\begin{lemma}[Coordinate test for lowering one atom]
\label{lem:coordinate-lowering-test}
Let $n\ge2$, and let
$\Theta=\{\theta_1,\ldots,\theta_r\}$ be a pairwise coprime set of atomic
prime powers with $d(\theta_i)\mid n$ and
\[
\lcmop(d(\theta_1),\ldots,d(\theta_r))=n.
\]
Fix $\theta_i=p_i^{e_i}$, and put
\[
N_i^-:=\lcmop\bigl(d^-(\theta_i),\ d(\theta_j):j\ne i\bigr).
\]
Then $N_i^-=n$ if and only if
\[
E_n(\theta_i)\subseteq\bigcup_{j\ne i}T_n(\theta_j).
\]
Equivalently, lowering $\theta_i$ to $p_i^{e_i-1}$ keeps the apparition lcm
equal to $n$ exactly when every coordinate lost by the lowering is supplied at
full exponent by some other atom. If
$\ell\in E_n(\theta_i)\setminus\bigcup_{j\ne i}T_n(\theta_j)$, then
$\nu_\ell(N_i^-)<\nu_\ell(n)$, so $N_i^-<n$.
\end{lemma}

\begin{proof}
Since $d^-(\theta_i)\mid d(\theta_i)\mid n$ and $d(\theta_j)\mid n$ for
$j\ne i$, the integer $N_i^-$ divides $n$. Hence $N_i^-=n$ is equivalent to
saying that every prime coordinate $\ell\in\mathcal P(n)$ occurs in $N_i^-$
with valuation $\nu_\ell(n)$.

Assume first that
$E_n(\theta_i)\subseteq\bigcup_{j\ne i}T_n(\theta_j)$, and fix
$\ell\in\mathcal P(n)$. Because the original lcm is $n$, at least one of the
numbers $d(\theta_1),\ldots,d(\theta_r)$ supplies the full
$\ell$-adic valuation of $n$. If such a supplier is some $d(\theta_j)$ with
$j\ne i$, then $\ell\in T_n(\theta_j)$ and the same full valuation appears in
$N_i^-$. The only remaining case is that $\theta_i$ itself supplies the full
valuation and no other atom does, so $\ell\in T_n(\theta_i)$ and
$\ell\notin\bigcup_{j\ne i}T_n(\theta_j)$. The assumed containment then gives
$\ell\notin E_n(\theta_i)$. By the definition of $E_n(\theta_i)$,
$\nu_\ell(d^-(\theta_i))=\nu_\ell(n)$, so the lowered factor still supplies the
full $\ell$-coordinate in $N_i^-$. Thus every coordinate of $n$ is still
present at full valuation, and $N_i^-=n$.

Conversely, suppose that some coordinate
$\ell\in E_n(\theta_i)$ is not in $\bigcup_{j\ne i}T_n(\theta_j)$. The
membership $\ell\in E_n(\theta_i)$ says
$\nu_\ell(d^-(\theta_i))<\nu_\ell(n)$. For every $j\ne i$, the exclusion
$\ell\notin T_n(\theta_j)$ says
$\nu_\ell(d(\theta_j))<\nu_\ell(n)$. Therefore the lcm defining $N_i^-$ also
has $\ell$-adic valuation strictly below $\nu_\ell(n)$. Since $N_i^-\mid n$,
this proves $N_i^-<n$.
\end{proof}

\begin{proposition}[Wagner comparison and lowered labels]
\label{prop:wagner-comparison}
Let $n\ge2$, and let $\Theta=\{p_1,\ldots,p_r\}$ be a set of distinct prime
atoms with $d_i:=\alpha(p_i)\mid n$. Then $\Theta$ is an $n$-witness cover
if and only if, after forgetting the order of the tuple
$(d_1,\ldots,d_r)$,
\begin{equation}\label{eq:wagner-minimal-multiplicative-cover}
 \lcmop(d_1,\ldots,d_r)=n,
 \qquad
 \lcmop(d_1,\ldots,\widehat{d_i},\ldots,d_r)<n
 \quad(1\le i\le r).
\end{equation}
Thus the prime-atom case is precisely the unordered form of Wagner's minimal
multiplicative cover condition.

For arbitrary pairwise coprime atoms
$\Theta=\{\theta_1,\ldots,\theta_r\}$, the witness condition replaces the
deletion in \eqref{eq:wagner-minimal-multiplicative-cover} by
\begin{equation}\label{eq:lowered-label-multiplicative-cover}
 \lcmop\bigl(d(\theta_1),\ldots,d^-(\theta_i),\ldots,
 d(\theta_r)\bigr)<n
 \quad(1\le i\le r).
\end{equation}
In particular, ladder atoms supply the genuinely additional lowered-label
case: their label is lowered to $d^-(\theta_i)$ rather than deleted.
\end{proposition}

\begin{proof}
For a prime atom $p_i$, one has $d^-(p_i)=\alpha(1)=1$. Hence replacing
$d_i$ by $d^-(p_i)$ in an lcm is identical to omitting $d_i$. Lemma~
\ref{thm:minimal-generators} and Lemma~\ref{lem:coordinate-lowering-test}
then identify the witness-cover conditions with
\eqref{eq:wagner-minimal-multiplicative-cover}. Since the labels form a
minimal cover, no repeated label can occur; forgetting tuple order therefore
produces the corresponding unordered cover without a multiplicity ambiguity.

For arbitrary atoms, the same two lemmas say exactly that the full apparition
lcm is $n$ and that lowering each atom produces
\eqref{eq:lowered-label-multiplicative-cover}. The final assertion is the
distinction between $d^-(p)=1$ for prime atoms and the nontrivial lower rank
of a ladder atom.
\end{proof}

Wagner's condition and its set-theoretic predecessor supply the classical
cover skeleton. The Fibonacci-specific content below is the arithmetic
realization of its labels by exact-rank primes, the lowered-label refinement
forced by prime-power ladders, and the unique correspondence with
$\mathcal M_n$. The canonical squarefree slice and its exact weighted count
in Section~\ref{sec:rank-pure} are further arithmetic consequences of that
realization.

\begin{theorem}\label{thm:witness-classification}
For every $n\ge 2$, multiplication gives a bijection
\[
\Theta\longmapsto \prod_{\theta\in\Theta}\theta
\]
from the set of $n$-witness covers onto $\mathcal M_n$. Its inverse sends
$m\in\mathcal M_n$ to the set
\[
\Theta(m):=\{p^{\nu_p(m)}:p\mid m\}
\]
of prime-power factors occurring in the prime factorization of $m$. In
particular,
\[
\mathcal M_n=\bigl\{\prod_{\theta\in\Theta}\theta:
\Theta\text{ is an $n$-witness cover}\bigr\},
\]
and each element of $\mathcal M_n$ has a unique witness-cover factorization.
\end{theorem}

\begin{proof}
Let $m\in\mathcal M_n$, and write its prime factorization as
$m=\prod_{i=1}^r\theta_i$ with $\theta_i=p_i^{e_i}$ and the $p_i$ distinct.
Each $\theta_i$ is atomic: if $\alpha(p_i^{e_i-1})=\alpha(p_i^{e_i})$, then
replacing $\theta_i$ by $p_i^{e_i-1}$ preserves the lcm, contradicting
the minimal-generator criterion in
Lemma~\ref{thm:minimal-generators}. Since
$n=\lcmop(d(\theta_i))$, the sets $T_n(\theta_i)$ cover $\mathcal P(n)$. The
condition $\alpha(m/p_i)<n$ forces
$E_n(\theta_i)\nsubseteq\bigcup_{j\neq i}T_n(\theta_j)$; otherwise the lost
coordinates would be supplied by other factors, keeping the lcm equal to $n$.
Thus $\{\theta_1,\dots,\theta_r\}$ is an $n$-witness cover.

Conversely, let $\Theta$ be an $n$-witness cover and
$m=\prod_{\theta\in\Theta}\theta$. By
the lcm property recorded in Proposition~\ref{prop:rank-basics},
$\alpha(m)=\lcmop(d(\theta):\theta\in\Theta)=n$. For each
$\theta=p^e\in\Theta$, the witness condition
provides $\ell\in E_n(\theta)\setminus\bigcup_{\eta\neq\theta}T_n(\eta)$; at
this coordinate, $\alpha(m/p)<n$.
Lemma~\ref{thm:minimal-generators} then gives $m\in\mathcal M_n$.

It remains to record injectivity and the stated inverse.  If two witness covers
$\Theta$ and $\Theta'$ have the same product, then unique factorization in
$\mathbb Z$ gives the same underlying primes with the same exponents in the two
products.  Since the elements of each witness cover are pairwise coprime prime
powers, those prime-power factors are exactly the elements of the cover.  Hence
$\Theta=\Theta'$.  Therefore multiplication is injective, while the first
paragraph shows that the prime-power factor set $\Theta(m)$ of every
$m\in\mathcal M_n$ is an $n$-witness cover and the second paragraph shows that
every $n$-witness cover maps into $\mathcal M_n$.  Thus multiplication is the
claimed bijection, with inverse $m\mapsto\Theta(m)$.
\end{proof}

\begin{corollary}\label{cor:support-bound}
Every $m\in\mathcal M_n$ satisfies $\omega(m)\le\omega(n)$.
\end{corollary}

\begin{proof}
The witness condition provides an injection
$\theta\mapsto\ell(\theta)$ from $\Theta$ into $\mathcal P(n)$: choose
$\ell(\theta)\in E_n(\theta)\setminus\bigcup_{\eta\neq\theta}T_n(\eta)$.
Hence $\omega(m)=\#\Theta\le\omega(n)$.
\end{proof}
