\section{Secondary arithmetic bounds}
\label{sec:supp-arithmetic-bounds}

\begin{lemma}[Known baseline for the arithmetic interface]
\label{lem:R-baseline}
For every $n\ge3$,
\[
 R(n)\le 1+\omega(F_n)=O(n),
 \qquad
 \log R(n)\le\log n+O(1).
\]
Moreover, Stroi\'nski's cumulative estimate gives
\[
 \limsup_{x\to\infty}\frac{\log x}{x^2}
 \#\{p:\alpha(p)\le x\}
 \le \frac{3\log\varphi}{2\pi^2}.
\]
Neither estimate implies the conjectures below.
\end{lemma}

\begin{proof}
Every prime atom in a family $\mathcal A_{I,J}(n)$ divides $F_n$, and a fixed
ladder family contains at most one atom. Thus a family contains at most all
distinct prime divisors of $F_n$ together with one ladder atom, proving
$R(n)\le1+\omega(F_n)$. Since
$2^{\omega(F_n)}\le\operatorname{rad}(F_n)\le F_n$ and
$\log F_n=n\log\varphi+O(1)$, this gives $R(n)=O(n)$ and the logarithmic
bound. The cumulative estimate is Stroi\'nski's Theorem~7
\cite{Stroinski2016AlphaContraction}.

Quantitatively, the normal-order conjecture below asks for
$\log R(n)=o(\log\log n)$ for almost all $n$, whereas the elementary estimate
gives only $O(\log n)$. The cumulative estimate controls all primes of rank at
most $x$ together and likewise gives at best a logarithmic bound of order
$\log n$ for a single window. For the mean conjecture, the elementary bound
and the mean order of $\omega$ give only
$O(x\log x\log\log x)$, not $o(x(\log\log x)^2)$.
\end{proof}

\begin{remark}[Cumulative rank estimates do not imply \textup{(H2)}]
\label{rem:cumulative-rank-estimates-insufficient}
This nonimplication can be made quantitative without asserting that the model
below is realized by Fibonacci ranks. Fix $0<c<\log\varphi/2$ and put
\[
 a_0(d):=\max\left\{1,
 \left\lfloor\frac{c\phi_{\rm E}(d)}{\log(2d)}\right\rfloor\right\}.
\]
Elementary integral comparison gives
\[
 \sum_{d\le y}a_0(d)\ll\frac{y^2}{\log y},\qquad
 a_0(d)\ll1+\frac{\phi_{\rm E}(d)}{\log(2d)},\qquad
 \sum_{d\le y}a_0(d)\log d\ll y^2.
\]
Thus this formal multiplicity sequence has the cumulative, pointwise, and
logarithmic-mass scales supplied by the presently known estimates, including
Theorem~\ref{article-thm:fibotomic-rank-entropy}. It is not asserted to arise from
Fibonacci exact-rank primes, and therefore it neither proves sharpness of the
fibotomic inequality nor supplies a realizable extremal model for it. On the
other hand, the standard
lower bound for Euler's function gives
\[
 \log a_0(n)=\log n-\log\log n+O(\log\log\log(3n))
\]
for large $n$.
Partial summation in \eqref{article-eq:mean-omega-input} therefore gives
\[
 \sum_{n\le x}\omega(n)\log a_0(n)
 \sim x\log x\log\log x,
\]
which is larger than the scale in \textup{(H2)}. Hence cumulative exact-rank
counts alone cannot establish the required visible-divisor maximum estimate.
\end{remark}

\begin{corollary}[Unconditional lower growth]
\label{cor:unconditional-lower-growth}
For almost all $n$,
\[
 \log\#\mathcal M_n\ge
 \left(\frac{\log2}{4}+o(1)\right)(\log\log n)^2.
\]
If $N_y:=\prod_{p\le y}p$, then
\[
 \log\#\mathcal M_{N_y}\ge
 \left(\frac{\log2}{4}+o(1)\right)
 \frac{(\log N_y)^2}{(\log\log N_y)^2}.
\]
\end{corollary}

\begin{proof}
The Hardy--Ramanujan normal-order theorem states that
$\omega(n)=(1+o(1))\log\log n$ for almost all $n$
\cite{HardyRamanujan1917NormalPrimeFactors,%
Tenenbaum2015AnalyticProbabilistic}. Substitution in the lower half of
\eqref{article-eq:private-cover-entropy-order} proves the first assertion. The prime
number theorem gives
$\omega(N_y)=\pi(y)\sim y/\log y\sim\log N_y/\log\log N_y$; substituting this
in the same lower bound proves the second assertion.
\end{proof}
