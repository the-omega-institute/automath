\section{Visible single-rank bounds}
\label{sec:supp-visible-rank-bounds}

\begin{corollary}[Bounds for the visible single-rank contribution]
\label{cor:visible-rank-sum-bounds}
Put
\[
 S(x):=\sum_{3\leq n\leq x}\omega(n)\log A^*(n).
\]
Then
\begin{equation}\label{eq:visible-rank-sum-upper}
 S(x)\leq
 \sum_{3\leq n\leq x}\omega(n)\log\phi_{\rm E}(n)
 -x(\log\log x)^2+O(x\log\log x),
\end{equation}
whereas
\begin{equation}\label{eq:visible-rank-sum-lower}
 S(x)\geq
 \left(\frac{2\log2}{25}+o(1)\right)x\log\log x.
\end{equation}
Neither estimate settles \eqref{article-eq:visible-single-rank-interface}.
\end{corollary}

\begin{proof}
Taking logarithms in \eqref{article-eq:visible-exact-rank-bound}, uniformly for
large $n$, gives
\[
 \log A^*(n)\leq
 \log\phi_{\rm E}(n)-\log\log n
 +\log\frac{\log\varphi}{2}+o(1).
\]
In addition to \eqref{article-eq:mean-omega-input}, partial summation gives
\[
 \sum_{3\leq n\leq x}\omega(n)\log\log n
 =x(\log\log x)^2+O(x\log\log x).
\]
For example, on $\sqrt x<n\leq x$ one may replace $\log\log n$ by
$\log\log x$ with $O(1)$ error, while the complementary range contributes
$o(x)$. Summation proves \eqref{eq:visible-rank-sum-upper}.

For the lower bound, Jarden proved that $L_{5p}$ has at least two distinct
primitive prime divisors for every prime $p>5$
\cite{Jarden1968LucasTwoPrimitive}; this published theorem is used as an
input and is not asserted as a result of the present paper. If $q$ is one of
them, then
$q\mid L_{5p}$ and
\[
 F_{10p}=F_{5p}L_{5p},\qquad \gcd(F_{5p},L_{5p})\mid2.
\]
Here $q$ is odd, so the gcd identity gives $q\nmid F_{5p}$. Thus
$\alpha(q)=2r$ for some $r\mid5p$. If $r<5p$, then
$F_{2r}=F_rL_r$ and $q\nmid F_r$, so $q\mid L_r$, contrary to the
primitivity of $q$ for $L_{5p}$.
Consequently $\alpha(q)=10p$, and therefore
\begin{equation}\label{eq:jarden-exact-rank-consequence}
 a(10p)\geq2\qquad(p>5\ \hbox{prime}).
\end{equation}

Let $n=10m$, where $5\nmid m$ and $m$ has a prime factor $p>5$. Then
$d=10p$ divides $n$ and
$\nu_5(d)=\nu_5(n)=1$, so $T_n(d)\ne\varnothing$. By
\eqref{eq:jarden-exact-rank-consequence}, $A^*(n)\geq2$. The excluded
integers $m$ having no prime factor greater than $5$ are of the form
$2^a3^b$ and contribute only $O((\log x)^3)$ after weighting by $\omega$.
Writing $y=x/10$ and interchanging two finite sums gives
\[
 \sum_{\substack{m\leq y\\5\nmid m}}\omega(m)
 =\sum_{\substack{q\leq y\\q\ \mathrm{prime},\ q\ne5}}
 \left(\left\lfloor\frac yq\right\rfloor
 -\left\lfloor\frac y{5q}\right\rfloor\right)
 =\frac45y\log\log y+O(y).
\]
Since $\omega(10m)=\omega(m)+O(1)$, the total $\omega$-weight of the
constructed integers is
$(2/25)x\log\log x+O(x)$. Multiplication by $\log2$ proves
\eqref{eq:visible-rank-sum-lower}.
\end{proof}


For a finite set $V$, let $\operatorname{MinCov}(V)$ denote the set of
minimal, or irredundant, covers of $V$ by nonempty subsets:
\[
 \operatorname{MinCov}(V):=
 \left\{\mathcal C\subseteq 2^V\setminus\{\varnothing\}:
 \bigcup_{S\in\mathcal C}S=V,\quad
 S\nsubseteq\bigcup_{T\in\mathcal C\setminus\{S\}}T
 \ \text{for every }S\in\mathcal C\right\}.
\]
