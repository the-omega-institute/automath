\section{Preliminaries}\label{sec:prelim}

\begin{proposition}[Standard rank facts]\label{prop:rank-basics}
Let $q,u,v,a,b,m$ be positive integers.  Then
\begin{enumerate}[label=(\roman*)]
\item $q\mid F_m$ if and only if $\alpha(q)\mid m$;
\item $\gcd(F_a,F_b)=F_{\gcd(a,b)}$;
\item $\alpha(\lcmop(u,v))=\lcmop(\alpha(u),\alpha(v))$.
\end{enumerate}
\end{proposition}

\begin{proof}
The first two statements are the classical rank-divisibility and strong
divisibility identities; see
\cite{Wall1960FibonacciModm,Vinson1963RankApparition}.  The third is also
standard \cite{LucaTron2015SelfFibonacciDivisors}.  Indeed, $m$ is divisible
by $\alpha(\lcmop(u,v))$ exactly when both $u$ and $v$ divide $F_m$, which
by (i) is equivalent to
$\lcmop(\alpha(u),\alpha(v))\mid m$.
\end{proof}

A prime $p$ is an \emph{exact-rank prime of rank $d$} if $\alpha(p)=d$.
We reserve \emph{primitive divisor of $F_d$} for the standard convention in
which a prime $p\mid F_d$ is primitive when
\[
 p\nmid 5F_1F_2\cdots F_{d-1}.
\]
Thus every primitive divisor is an exact-rank prime, while the exact-rank
prime $5$ at rank $5$ is excluded because it divides the discriminant.

\begin{theorem}[Classical exact-rank existence]
\label{thm:primitive-divisors}
For every $n\geq3$ with $n\notin\{6,12\}$, there is a prime $p_n$ such that
$\alpha(p_n)=n$.
\end{theorem}

\begin{proof}
For $n=3,4,5$, take $p_n=2,3,5$, respectively.  For $n\geq7$,
$n\ne12$, the primitive-divisor theorem
\cite{Carmichael1913PrimitiveDivisors,Yabuta2001PrimitiveDivisorFibonacci,
BiluHanrotVoutier2001PrimitiveDivisorsLucas} gives a prime dividing $F_n$
and no earlier Fibonacci number, hence a prime of exact rank $n$.
\end{proof}

For $d\geq1$, put
\[
 \Pi_\alpha(d):=\{p\text{ prime}:\alpha(p)=d\},
 \qquad a(d):=\#\Pi_\alpha(d).
\]

\begin{lemma}[Exact-rank prime count]
\label{lem:primitive-prime-mobius-count}
For every $d\geq1$, the set $\Pi_\alpha(d)$ is finite and
\[
 a(d)=\sum_{c\mid d}\mu(d/c)\omega(F_c).
\]
\end{lemma}

\begin{proof}
Proposition~\ref{prop:rank-basics} partitions the prime divisors of $F_d$
according to their exact ranks:
\[
 \omega(F_d)=\sum_{c\mid d}a(c).
\]
The assertion follows by M\"obius inversion.
\end{proof}

For a rank jump $\theta=p^e$, write $\theta\in\Atom(d(\theta))$, where
\[
 \Atom(d):=\{p^e:\alpha(p^e)=d,\ \alpha(p^{e-1})<d\}.
\]
For nonempty $I\subseteq J\subseteq\mathcal P(n)$, the family used in
Theorem~\ref{thm:minimal-generator-growth} is
\[
 \mathcal A_{I,J}(n):=
 \{\theta\in\bigcup_{d\mid n}\Atom(d):
 E_n(\theta)=I,\ T_n(\theta)=J\}.
\]
