\subsection{Rank of apparition and Fibonacci divisibility}\label{subsec:fib-div}

\begin{lemma}[Finite rank of apparition]\label{lem:finite-rank}
For every integer $q\ge1$, the rank of apparition $\alpha(q)$ is finite.  More
precisely, $\alpha(1)=1$, and for every $q\ge2$ there is a positive integer
$r$ such that $q\mid F_r$.
\end{lemma}

\begin{proof}
The case $q=1$ is the stated convention.  Let $q\ge2$.  Work modulo $q$ with
the Fibonacci state vector $(F_m,F_{m+1})$.  The transition
\[
(x,y)\longmapsto (y,x+y)
\]
has inverse $(x,y)\mapsto (y-x,x)$ modulo $q$, so it is a permutation of the
finite set $(\ZZ/q\ZZ)^2$.  Therefore the orbit of the initial state $(0,1)$
is periodic and returns to $(0,1)$ after some positive number $r$ of steps.
Thus $(F_r,F_{r+1})\equiv(0,1)\pmod q$, in particular $q\mid F_r$.  The
least such positive $r$ exists by well ordering.
\end{proof}

\begin{proposition}\label{prop:rank-basics}
Let $q,u,v\ge1$ and let $a,b,m\ge1$. Then:
\begin{enumerate}[label=(\roman*)]
\item $q\mid F_m$ if and only if $\alpha(q)\mid m$.
\item $\gcd(F_a,F_b)=F_{\gcd(a,b)}$.
\item $\alpha(\lcmop(u,v))=\lcmop(\alpha(u),\alpha(v))$.
\end{enumerate}
\end{proposition}

\begin{proof}
The divisibility criterion and the strong-divisibility identity are classical;
we use them in the forms proved by Wall \cite{Wall1960FibonacciModm} and
Vinson \cite{Vinson1963RankApparition}. The convention $\alpha(1)=1$ makes
the assertion for $q=1$ immediate.

The Fibonacci lcm identity is recorded by Luca--Tron
\cite{LucaTron2015SelfFibonacciDivisors}; Renault gives the corresponding
identity for the more general $(a,b)$-Fibonacci sequences
\cite{Renault2013PeriodRankOrder}. For completeness, an integer $m$ is
divisible by
$\alpha(\lcmop(u,v))$ if and only if $\lcmop(u,v)\mid F_m$, by (i). This is
equivalent to $u\mid F_m$ and $v\mid F_m$, hence to
$\alpha(u)\mid m$ and $\alpha(v)\mid m$, again by (i). Thus the multiples of
$\alpha(\lcmop(u,v))$ are exactly the multiples of
$\lcmop(\alpha(u),\alpha(v))$, and the two positive integers are equal.
\end{proof}

\subsection{Exact-rank primes and primitive divisors}\label{subsec:primitive}

A prime $p$ is an \emph{exact-rank prime of rank $d$} if $\alpha(p)=d$;
when it is viewed as an atomic prime power, we also call it a \emph{prime
atom}. We reserve \emph{primitive divisor of $F_d$} for the standard
Carmichael--Bilu--Hanrot--Voutier convention: a prime $p$ dividing $F_d$ is
primitive when
\[
 p\nmid 5F_1F_2\cdots F_{d-1}.
\]
Thus every primitive divisor is an exact-rank prime, but the exact-rank prime
$5$ at rank $5$ is excluded from the primitive-divisor terminology because it
divides the discriminant. This distinction will be maintained throughout.

\begin{theorem}[Classical exact-rank existence]\label{thm:primitive-divisors}
For every $n\ge 3$ with $n\notin\{6,12\}$, there exists a prime $p_n$ with
$\alpha(p_n)=n$.
\end{theorem}

\begin{proof}
For $n=3,4,5$ one takes $p_3=2$, $p_4=3$, and $p_5=5$; the last is the
discriminant-related exact-rank exception just described. For $n\ge 7$,
$n\neq 12$, the primitive divisor theorem
\cite{Carmichael1913PrimitiveDivisors,Yabuta2001PrimitiveDivisorFibonacci,%
BiluHanrotVoutier2001PrimitiveDivisorsLucas}
gives a prime dividing $F_n$ but no earlier Fibonacci number. Since
$\alpha(p_n)\mid n$ and $p_n$ divides no $F_d$ with $d<n$, necessarily
$\alpha(p_n)=n$.
\end{proof}

\subsection{Counting corollaries}\label{subsec:counting}

The standard M\"obius-inversion formulas for first-appearance counts follow
from Stroi\'nski's $\alpha$-contraction identity
\cite{Stroinski2016AlphaContraction}: for any $f:\NN\to\mathbb{C}$ with
$f(1)=0$,
\begin{equation}\label{eq:stroinski}
\sum_{d\mid F_n}f(d)=\sum_{d\mid n}f_\alpha(d),
\qquad
f_\alpha(d):=\sum_{\alpha(m)=d}f(m).
\end{equation}
Taking $f$ to be the indicator of integers at least $2$, of primes, or of
prime powers, and applying M\"obius inversion, one obtains
\begin{equation}\label{eq:mobius-birth}
A(n)=\sum_{d\mid n}\mu(n/d)\,\tau(F_d),
\qquad
A_{\mathrm{pr}}(n)=\sum_{d\mid n}\mu(n/d)\,\omega(F_d).
\end{equation}
We regard these as known corollaries; the witness-cover classification below
is independent of them.

\begin{lemma}[Exact-rank prime M\"obius count]\label{lem:primitive-prime-mobius-count}
For $d\ge1$, put
\[
\Pi_\alpha(d):=\{p\text{ prime}:\alpha(p)=d\}.
\]
Then $\Pi_\alpha(d)$ is finite and
\[
\#\Pi_\alpha(d)=\sum_{c\mid d}\mu(d/c)\,\omega(F_c).
\]
\end{lemma}

\begin{proof}
For each $d\ge1$, the primes with rank dividing $d$ are exactly the prime
divisors of $F_d$: indeed, for a prime $p$, Proposition~\ref{prop:rank-basics}
gives $p\mid F_d$ if and only if $\alpha(p)\mid d$.  Hence
Proposition~\ref{prop:rank-basics} partitions the distinct prime divisors of
$F_d$ according to their exact ranks $c\mid d$, including the exceptional
primes $2$ and $5$ with ranks $3$ and $5$ respectively; a prime divisor of
$F_c$ can occur earlier only through the condition $\alpha(p)\mid e$ for the
earlier index $e$.
\[
\omega(F_d)=\sum_{c\mid d}\#\Pi_\alpha(c).
\]
The right-hand side is finite because $F_d$ has only finitely many prime
divisors.  Applying M\"obius inversion on the divisor lattice of $d$ gives the
displayed formula.
\end{proof}

\subsection{Atomic notation}\label{subsec:prime-power}

\begin{definition}\label{def:atomic-families}
For a prime power $\theta=p^e$ with $e\ge1$, put
\[
d(\theta):=\alpha(\theta),\qquad
d^-(\theta):=\alpha(p^{e-1}),
\]
where $\alpha(1)=1$. For $d\ge1$ define
\[
\Atom(d):=\{p^e:\ \alpha(p^e)=d,\ \alpha(p^{e-1})<d\}.
\]
Thus $\Atom(d)$ consists of the prime-power factors that are born exactly at
rank $d$.

If $d(\theta)\mid n$, define the full-coordinate support and the
essential-coordinate support of $\theta$ relative to $n$ by
\[
T_n(\theta):=\{\ell\in\mathcal P(n):\
\nu_\ell(d(\theta))=\nu_\ell(n)\},
\]
and
\[
E_n(\theta):=\{\ell\in T_n(\theta):\
\nu_\ell(d^-(\theta))<\nu_\ell(n)\}.
\]
Equivalently, $T_n(\theta)$ records the prime coordinates of $n$ supplied at
full exponent by $d(\theta)$, while $E_n(\theta)$ records the full coordinates
lost when $\theta=p^e$ is lowered to $p^{e-1}$.

For nonempty $I\subseteq J\subseteq\mathcal P(n)$, put
\[
\mathcal A_{I,J}(n):=
\{\theta\in\bigcup_{d\mid n}\Atom(d):\
E_n(\theta)=I,\ T_n(\theta)=J\}.
\]
\end{definition}

\begin{definition}\label{def:support-hypergraph}
Let $q=\prod_i p_i^{e_i}\in D_n$, and put
$\lambda_i:=\alpha(p_i^{e_i})$. The \emph{support hypergraph} $H_n(q)$ has
vertex set $\mathcal P(n)$ and one edge
\[
 \operatorname{supp}_n(\lambda_i):=
 \{\ell\in\mathcal P(n):\nu_\ell(\lambda_i)>0\}
\]
for each nonempty such support. We call $q$ \emph{connected} if
$H_n(q)$ is connected on its vertex set, and write
\[
 \mathcal M_n^{\mathrm{conn}}:=
 \{m\in\mathcal M_n:\ H_n(m)\text{ is connected}\}.
\]
\end{definition}
