\section{A fixed-total-weight comparison}

For a finite set $V$, let $\operatorname{MinCov}(V)$ be its irredundant
covers by distinct nonempty subsets, and let $C_k$ be the number of such
covers when $|V|=k$.

\begin{proposition}[Sharp cumulative-data bound]
\label{prop:sharp-cumulative-data-obstruction}
Let $V$ be a labelled $k$-set, $k\geq1$, and assign a real weight
$w_S\geq1$ to every nonempty $S\subseteq V$.  Put
\[
 Z_V(w):=\sum_{\mathcal C\in\operatorname{MinCov}(V)}
              \prod_{S\in\mathcal C}w_S,
 \qquad A:=\sum_{\varnothing\ne S\subseteq V}w_S.
\]
Then
\begin{equation}\label{eq:sharp-total-mass-cover-bound}
 Z_V(w)\geq C_k+A-(2^k-1).
\end{equation}
This is best possible for every $A\geq2^k-1$: equality is attained by
\[
 w_V=A-(2^k-2),\qquad w_S=1\quad(S\ne V).
\]
If $k\geq3$, this is the only equality profile.

Consequently, let $n\geq3$ be odd and squarefree, put
$V=\mathcal P(n)$ and $k=\omega(n)$, and define
\[
 n_S:=\prod_{\ell\in S}\ell,\qquad
 a(d):=\#\{p\text{ prime}:\alpha(p)=d\},
\]
and
\[
 \mathcal W(n):=\frac1{C_k}
 \sum_{\mathcal C\in\operatorname{MinCov}(V)}
 \prod_{S\in\mathcal C}a(n_S).
\]
Then
\begin{equation}\label{eq:squarefree-cumulative-rank-pure-bound}
 \#\{m\in\mathcal M_n:\mu^2(m)=1\}
 \geq C_k+\omega(F_n)-(2^k-1),
\end{equation}
and
\begin{equation}\label{eq:squarefree-cumulative-weighted-excess}
 \mathcal W(n)\geq
 1+\frac{\omega(F_n)-(2^k-1)}{C_k}.
\end{equation}
Among positive weight profiles with the same total
$\sum_{d\mid n,\ d>1}a(d)=\omega(F_n)$, this conclusion is exact.
\end{proposition}

\begin{proof}
Write $w_S=1+x_S$ with $x_S\geq0$.  For every minimal cover $\mathcal C$,
\[
 \prod_{S\in\mathcal C}(1+x_S)
 \geq1+\sum_{S\in\mathcal C}x_S.
\]
Every nonempty $S\subseteq V$ occurs in a minimal cover: use $\{V\}$ when
$S=V$, and otherwise use
$\{S\}\cup\{\{v\}:v\in V\setminus S\}$.  Summing gives
\[
 Z_V(w)\geq C_k+\sum_{\varnothing\ne S\subseteq V}x_S
 =C_k+A-(2^k-1).
\]
The only minimal cover containing $V$ is $\{V\}$.  The displayed
full-support profile therefore attains equality.

Suppose $k\geq3$ and $S$ is proper and nonempty.  If $|S|=1$, both the
singleton partition and $\{S,V\setminus S\}$ are minimal covers containing
$S$.  If $|S|\geq2$, choose $s\in S$ and $t\in V\setminus S$; another such
cover consists of $S$, $\{s,t\}$, and the singletons
$\{v\}$ for $v\in V\setminus(S\cup\{t\})$.  Thus every proper $S$ has
coefficient at least two in the summed linear bound, and equality forces
$x_S=0$.  This proves uniqueness.

For odd squarefree $n$, nonempty subsets of $\mathcal P(n)$ correspond to
divisors $d>1$ of $n$, and exact ranks partition the prime divisors of $F_n$:
\[
 \sum_{\varnothing\ne S\subseteq\mathcal P(n)}a(n_S)
 =\sum_{d\mid n}a(d)=\omega(F_n).
\]
All weights are positive by exact-rank prime existence.  Apply
\eqref{eq:sharp-total-mass-cover-bound}; the rank-pure squarefree-sector
theorem in the article identifies its weighted sum with the left side of
\eqref{eq:squarefree-cumulative-rank-pure-bound}.
\end{proof}
