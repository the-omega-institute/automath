\begin{definition}[Three-coordinate slot construction]
\label{def:three-coordinate-slot-construction}
Assume $\mathcal P(n)=\{\ell_1,\ell_2,\ell_3\}$ and identify the prime
coordinates with $[3]=\{1,2,3\}$.  For every nonempty
$J\subseteq[3]$ let
\[
P_J(n)=\{\theta\in\mathcal A_{J,J}(n):\theta\text{ is a prime}\}
\]
be the exact-rank prime slot. For $i\in J$ let
\[
L_{i,J}(n)=\{\theta\in\mathcal A_{\{i\},J}(n):
\theta=p^e\text{ with }e>1\}
\]
be the ladder slot with essential coordinate $i$ and full support $J$, and put
\[
S_i(n):=P_{\{i\}}(n)\cup L_{i,\{i\}}(n).
\]
The following eight slot families, together with all coordinate permutations,
are called the admissible three-coordinate rows:
\[
\begin{array}{ccl}
\Gamma_1&:& P_{123},\\
\Gamma_3&:& L_{i,123},\\
\Gamma_4&:& P_{ij}P_{jk},\\
\Gamma_5&:& L_{i,ij}P_{jk},\\
\Gamma_6&:& L_{i,ij}L_{k,jk},\\
\Gamma_7&:& S_iP_{jk},\\
\Gamma_8&:& S_iL_{j,jk},\\
\Gamma_9&:& S_iS_jS_k,
\end{array}
\]
where $(i,j,k)$ ranges over ordered triples of distinct coordinates and braces
are suppressed in subscripts.  A \emph{labelled filling} of one of these rows
means choosing one prime power from each displayed slot after the coordinates
$1,2,3$ have been fixed; it is \emph{coprime} if the chosen prime powers have
distinct underlying primes.
\end{definition}

\begin{lemma}[Three-coordinate nonemptiness criterion]
\label{lem:three-coordinate-nonemptiness}
Assume $\mathcal P(n)=\{\ell_1,\ell_2,\ell_3\}$ and use the slots and rows of
Definition~\ref{def:three-coordinate-slot-construction}. Fix one labelled
coordinate permutation of one of the eight rows
\[
\Gamma_1,
\Gamma_3,
\Gamma_4,
\Gamma_5,
\Gamma_6,
\Gamma_7,
\Gamma_8,
\Gamma_9.
\]
Let $(I_j,J_j)$ be the essential and full coordinate supports attached to the
$j$th slot of the labelled row. The row is realized by an element of
$\mathcal M_n$ if and only if the following arithmetic selection test succeeds:
\begin{enumerate}[label=(\alph*)]
\item for every nonsingleton prime slot $I_j=J_j$, $\#J_j>1$, choose a prime
$\theta_j\in\Pi_\alpha(d_j)$ for some
$d_j\in\mathcal R_{J_j}^\circ(n)$;
\item for every proper ladder slot $I_j=\{i\}\subsetneq J_j$, choose the
unique candidate in $L_{i,J_j}(n)$ given by
Lemma~\ref{lem:three-coordinate-arithmetic-tests};
\item for every singleton slot $I_j=J_j=\{i\}$, choose either an exact-rank
prime from $\bigcup_{d\in\mathcal R_{\{i\}}^\circ(n)}\Pi_\alpha(d)$ or the
unique ladder candidate in $L_{i,\{i\}}(n)$;
\item the chosen prime powers have distinct underlying primes.
\end{enumerate}
Equivalently, form for each slot its finite admissible set $C_j$: for a
prime slot use
$\bigcup_{d\in\mathcal R_{J_j}^\circ(n)}\Pi_\alpha(d)$, for a proper ladder
slot use $L_{i,J_j}(n)$, and for a singleton slot use
$P_{\{i\}}(n)\cup L_{i,\{i\}}(n)$.  The row succeeds precisely when these
sets satisfy the Hall inequalities of Lemma~\ref{lem:finite-slot-selection},
with underlying-prime multiplicities in each exact-rank window given by
$\#\Pi_\alpha(d)=\sum_{c\mid d}\mu(d/c)\omega(F_c)$.

In that case $m=\theta_1\cdots\theta_r$ is the realizing minimal generator.
Its labelled rank-jump factor set
\[
 \{(\theta,E_n(\theta),T_n(\theta)):\theta\text{ is a maximal prime-power
 factor of }m\}
\]
is unique. The associated unlabelled shape is unique only as the $S_3$-orbit
of the labelled support set $\mathsf T(m)$.

Equivalently, the rows in
Definition~\ref{def:three-coordinate-slot-construction} are unlabelled
representatives modulo coordinate relabeling: an unlabelled type $\Gamma_i$ is
realized if and only if the preceding labelled arithmetic selection test
succeeds for one of its coordinate permutations. The excluded type $\Gamma_2$
is never realized.
\end{lemma}

\begin{proof}
Suppose first that $m\in\mathcal M_n$ realizes the fixed labelled row. Write its witness cover as
$\Theta=\{\theta_1,\dots,\theta_r\}$ with the factors ordered so that
$(E_n(\theta_j),T_n(\theta_j))=(I_j,J_j)$. By definition
$\theta_j\in\mathcal A_{I_j,J_j}(n)$, and pairwise coprimeness is part of
Definition~\ref{article-def:witness-cover}. Lemma~\ref{lem:three-coordinate-rank-jump-families}
places each such factor in the corresponding prime or ladder family, and
Lemma~\ref{lem:three-coordinate-arithmetic-tests} translates those
families into the displayed rank-window and ladder-candidate conditions. Thus
the arithmetic selection test succeeds.

Conversely, assume that the arithmetic selection test succeeds. By
Lemma~\ref{lem:three-coordinate-arithmetic-tests}, each selected
prime power lies in the required prime or ladder family, and
Lemma~\ref{lem:three-coordinate-rank-jump-families} gives
$\theta_j\in\mathcal A_{I_j,J_j}(n)$. Distinct underlying primes are exactly
pairwise coprimeness for prime powers. Because the fixed row is an abstract
witness type, the sets $J_j$ cover $[3]$, and for every $j$ one has
$I_j\nsubseteq\bigcup_{h\ne j}J_h$. Translating back to $T_n$ and $E_n$, the
set $\Theta=\{\theta_1,\dots,\theta_r\}$ satisfies the coverage and
irredundance conditions in Definition~\ref{article-def:witness-cover}; coprimeness
gives the remaining condition. Hence $\Theta$ is an $n$-witness cover, so
Theorem~\ref{article-thm:witness-classification} gives
$m=\prod_j\theta_j\in\mathcal M_n$ with the prescribed row type.

The factorization of $m$ into maximal prime powers is unique, so it recovers
the displayed labelled rank-jump factor set uniquely. A raw enumeration over all
coordinate permutations may nevertheless repeat a labelled row or a product
when a $\Gamma_i$ has a nontrivial stabilizer. Accordingly, unlabelled shapes
are identified only at the level of $S_3$-orbits, and all enumerated products
are deduplicated. The final exclusion of $\Gamma_2$ is
Lemma~\ref{lem:fibonacci-three-coordinate-types}.
\end{proof}

\begin{lemma}[Three-coordinate realization]\label{lem:three-coordinate-realization}
Assume $\mathcal P(n)=\{\ell_1,\ell_2,\ell_3\}$. After the labelled coordinate
supports $E_n,T_n\subseteq\{1,2,3\}$ are fixed, every element of
$\mathcal M_n$ arises from exactly one nonempty arithmetic selection in the
rank windows $\mathcal R_J^\circ(n)$ and the explicit ladder candidates
$\mathcal L_i^*(n)$ belonging to one fixed labelled coordinate permutation of
one of the eight admissible types in
Lemma~\ref{lem:three-coordinate-nonemptiness}; the selected prime powers
must have distinct underlying primes. The unlabelled $\Gamma$-rows are only
representatives modulo coordinate relabeling.
\end{lemma}

\begin{proof}
This is Theorem~\ref{article-thm:witness-classification} together with
Lemmas~\ref{lem:fibonacci-three-coordinate-types} and
\ref{lem:three-coordinate-nonemptiness}.
\end{proof}

The finite slot-selection algorithm behind Table~\ref{tab:three-coordinate-classification}
is recorded in Appendix~\ref{app:three-coordinate-checklist}; it is used below
only to justify that the displayed table is an effective closed classification.

\begin{theorem}[Complete low-support classification: eight support-three families]
\label{thm:low-support-classification-package}
Let $n\ge2$.  The minimal generators $\mathcal M_n$ are classified for
$\omega(n)\le3$ as follows.
\begin{enumerate}[label=(\roman*)]
\item If $\omega(n)=1$, then $\mathcal M_n=\Atom(n)$.
\item If $\mathcal P(n)=\{\ell_1,\ell_2\}$, then $\mathcal M_n$ is the
 disjoint union of $\Atom(n)$ and the products
 $\theta_1\theta_2$ with
 $\theta_i\in\mathcal A_{\{i\},\{i\}}(n)$ and
 $\gcd(\theta_1,\theta_2)=1$.
\item If $\mathcal P(n)=\{\ell_1,\ell_2,\ell_3\}$, then $\mathcal M_n$ is the
 set of distinct products admitted by Table~\ref{tab:three-coordinate-classification},
 after applying all coordinate permutations and the finite slot-selection
 criterion of Lemma~\ref{lem:finite-slot-selection}.  Equivalently, for each
 permuted row one chooses one prime power from each displayed slot set, with
 distinct underlying primes. Every such product occurs. Each element of
 $\mathcal M_n$ has a unique labelled support set $\mathsf T(m)$, while its
 unlabelled $\Gamma_i$ is determined only up to the $S_3$-orbit; a product
 produced more than once by the raw permuted enumeration is retained once.
\end{enumerate}
For the three-coordinate rows,
$P_J(n)$ denotes the exact-rank prime slot, $L_{i,J}(n)$ the ladder slot with
essential coordinate $i$ and full support $J$, and
$S_i(n)=P_{\{i\}}(n)\cup L_{i,\{i\}}(n)$ the singleton slot.  Each slot is the
following explicit finite set:
\[
\begin{aligned}
\mathfrak C(P_J)&:=\bigcup_{d\in\mathcal R_J^\circ(n)}\Pi_\alpha(d),\\
\mathfrak C(L_{i,J})&:=\{\lambda\in\mathcal L_i^*(n):\alpha(\lambda)\mid n,
\ \{k:\nu_{\ell_k}(\alpha(\lambda))=\nu_{\ell_k}(n)\}=J\},\\
\mathfrak C(S_i)&:=\mathfrak C(P_{\{i\}})\cup \mathfrak C(L_{i,\{i\}}).
\end{aligned}
\]
Here $\mathcal R_J^\circ(n)$ is the finite exact-rank window and
$\mathcal L_i^*(n)$ is the empty-or-singleton ladder candidate set of
Lemma~\ref{lem:three-coordinate-arithmetic-tests}. A row succeeds
exactly when its displayed finite slot sets admit a filling by distinct
underlying primes, equivalently when the Hall test of
Lemma~\ref{lem:finite-slot-selection} holds.  The displayed rows are
representatives modulo relabeling of the coordinates $1,2,3$.
Equivalently, writing $[3]=\{1,2,3\}$, suppressing braces in prime-slot
subscripts, and letting $(i,j,k)$ range over ordered triples of distinct
coordinates, the eight orbit representatives are
\[
P_{[3]},\quad L_{i,[3]},\quad P_{i,j}P_{j,k},\quad
L_{i,\{i,j\}}P_{j,k},\quad
L_{i,\{i,j\}}L_{k,\{j,k\}},\quad S_iP_{j,k},
\quad S_iL_{j,\{j,k\}},\quad S_iS_jS_k,
\]
respectively for $\Gamma_1,\Gamma_3,\Gamma_4,\Gamma_5,\Gamma_6,\Gamma_7,
\Gamma_8,\Gamma_9$.
Let $\mathcal G_n$ denote the finite set displayed in (i), (ii), or (iii), with
the convention $\mathcal G_2=\varnothing$.  Then, for every $n\ge2$ with
$\omega(n)\le3$,
\[
\mathcal G_n=\mathcal M_n,
\]
and the displayed list is a finite minimal-generator classification. The
table should be read in three passes.  First choose one of the eight
unlabelled support skeletons and then apply an arbitrary permutation of the
coordinate labels $1,2,3$.  Next replace every displayed slot by its explicit
finite candidate set $\mathfrak C(\cdot)$, so prime slots are tested through
the rank windows $\mathcal R_J^\circ(n)$ and ladder slots through the candidate
sets $\mathcal L_i^*(n)$. Finally apply the distinct-prime Hall test and insert
the resulting products into a set: a row
contributes exactly the products obtained by choosing one prime power from each
of its candidate slots with no underlying prime repeated.  The lemmas preceding
the theorem prove that these three finite steps are equivalent to the abstract
witness-cover conditions, so the table is not an additional heuristic case
split but the effective normal form for all support-three minimal generators.

\end{theorem}

\begin{proof}
The one-coordinate assertion is Corollary~\ref{cor:one-support-complete-classification}.
The two-coordinate assertion is Corollary~\ref{cor:two-support-complete-classification}.
Assume now that $\omega(n)=3$. Lemma~\ref{lem:fibonacci-three-coordinate-types}
reduces every witness cover to the eight listed abstract types and excludes
$\Gamma_2$. Lemma~\ref{lem:three-coordinate-rank-jump-families} translates
an abstract slot $(I,J)$ into exactly one of the displayed prime, ladder,
or singleton families: nonsingleton slots with $I=J$ are $P_J(n)$, proper
ladder slots $I=\{i\}\subsetneq J$ are $L_{i,J}(n)$, and singleton slots are
$S_i(n)=P_{\{i\}}(n)\cup L_{i,\{i\}}(n)$. Substituting these translations into
the eight admissible abstract types of Lemma~\ref{lem:abstract-three-types}
gives precisely the rows of Table~\ref{tab:three-coordinate-classification}.
The same substitution, with braces suppressed in prime-slot subscripts,
gives the displayed canonical-family list: the first two rows are the full
prime and full ladder families, the next three are the two-slot overlapping
families, the next two attach a singleton slot to a nonsingleton slot, and the
last row is the three-singleton family.

It remains only to identify the tests in the last column. For prime slots,
Lemma~\ref{lem:three-coordinate-arithmetic-tests} gives
$P_J(n)=\bigcup_{d\in\mathcal R_J^\circ(n)}\Pi_\alpha(d)$ and, after deleting
previously used underlying primes, the stated finite multiplicity criterion.
For ladder slots, the same proposition gives the displayed singleton-candidate
formula for $L_{i,J}(n)$. Pairwise coprimeness of the chosen prime powers is
exactly the distinct-underlying-prime condition because all slots are filled by
prime powers. Hence the table's row test is the arithmetic selection test of
Lemma~\ref{lem:three-coordinate-nonemptiness}. That theorem also gives the
converse construction and uniqueness, up to reordering slots with identical
labelled data, while Theorem~\ref{article-thm:witness-classification} identifies the
resulting products with the elements of $\mathcal M_n$.

Thus the displayed support-dependent set $\mathcal G_n$ equals
$\mathcal M_n$ in all cases $\omega(n)\le3$, including the boundary convention
$\mathcal G_2=\varnothing=\mathcal M_2$.  In the three-coordinate case the
finiteness of the prime and ladder slot sets and the exact slot enumeration
are supplied by Proposition~\ref{prop:three-coordinate-table-checklist}.
\end{proof}

\begin{corollary}[Support-three obstruction certificates]
\label{cor:support-three-obstruction-certificates}
Assume $\omega(n)=3$ and let a proposed support-three minimal-generator shape be
specified by an irredundant finite list of rank-jump slots whose full-support sets
cover the labelled coordinates $\{1,2,3\}$.  Exactly one of the following
alternatives occurs.
\begin{enumerate}[label=(\alph*)]
\item The shape is the excluded skeleton $\Gamma_2$, equivalently it requires an
rank-jump factor with $1<\#E_n(\theta)<\#T_n(\theta)=3$; then no Fibonacci minimal
generator has that shape.
\item After relabeling it is one of the eight rows in
Table~\ref{tab:three-coordinate-classification}.  The row contributes exactly
the products obtained from the finite candidate sets
$\mathfrak C(P_J)$, $\mathfrak C(L_{i,J})$, and $\mathfrak C(S_i)$ by choosing
one prime power in each row slot with distinct underlying primes.
\item The row is one of the eight admissible skeletons but the finite
distinct-prime Hall inequalities of Lemma~\ref{lem:finite-slot-selection} fail;
then the row contributes no element of $\mathcal M_n$.
\end{enumerate}
Thus every failure in support three has a certificate of one of two kinds: the
single structural Fibonacci obstruction $\Gamma_2$, or a finite Hall deficiency
inside an otherwise admissible slot row.
\end{corollary}

\begin{proof}
By Lemma~\ref{lem:abstract-three-types}, every irredundant three-coordinate
witness skeleton is one of $\Gamma_1,\ldots,\Gamma_9$. The prime--ladder
dichotomy, Lemma~\ref{thm:primitive-ladder-dichotomy}, excludes precisely the
case in which a rank-jump factor has a proper nonsingleton essential set inside a larger
full-support set; among the nine skeletons this is exactly $\Gamma_2$, proving
(a).

For every remaining skeleton, Lemma~\ref{lem:three-coordinate-rank-jump-families}
translates each abstract slot into one of the prime slots $P_J(n)$, ladder
slots $L_{i,J}(n)$, or singleton slots $S_i(n)$.  Lemma~\ref{lem:three-coordinate-arithmetic-tests}
identifies these as the finite candidate sets displayed in
Theorem~\ref{thm:low-support-classification-package}.  A selected row therefore
forms a witness cover exactly when its chosen prime powers are pairwise
coprime, and for prime powers this is exactly distinctness of underlying primes.
Lemma~\ref{lem:finite-slot-selection} is precisely the Hall criterion for such a
selection.  If the criterion holds, Theorem~\ref{article-thm:witness-classification}
turns the selected witness cover into the corresponding element of
$\mathcal M_n$; if it fails, no coprime filling of that row exists.  These are
(b) and (c), and the alternatives exhaust all support-three possibilities.
\end{proof}

\begin{corollary}[Low-support rank level sets and counting]
\label{cor:low-support-normal-form}
\label{cor:low-support-birth-layer-package}
\label{prop:low-support-cardinality-formula}
Let $n\ge2$ with $\omega(n)\le3$, and let $\mathcal G_n$ be the finite
minimal-generator set described in
Theorem~\ref{thm:low-support-classification-package}.  Then $\mathcal G_n$ is
the canonical antichain of minimal elements of $B_n$, and for every $q\in D_n$,
\[
q\in B_n\quad\Longleftrightarrow\quad
\text{there exists }g\in\mathcal G_n\text{ with }g\mid q.
\]
Equivalently,
\[
B_n=\bigcup_{g\in\mathcal G_n}\{q\in D_n:\ g\mid q\},
\]
where the union is taken inside the divisor lattice $D_n$ and is not asserted
to be disjoint.  The rank-level-set cardinality is the closed finite formula
\begin{equation}\label{eq:low-support-cardinality-formula}
A(n)=
\sum_{\varnothing\ne\mathcal S\subseteq\mathcal G_n}
(-1)^{|\mathcal S|+1}
\tau\left(\frac{F_n}{\lcmop(g:g\in\mathcal S)}\right),
\end{equation}
with the convention that the empty sum is $0$.
\end{corollary}

\begin{proof}
Theorem~\ref{thm:low-support-classification-package} gives
$\mathcal G_n=\mathcal M_n$ for $\omega(n)\le3$, including the boundary
convention $\mathcal G_2=\varnothing=\mathcal M_2$. Minimal elements of an
upper set are pairwise incomparable, so $\mathcal G_n$ is an antichain in
$D_n$.  The upper-fiber identity, Lemma~\ref{article-thm:upper-fiber}, gives for every
$n\ge2$ the principal-upper-ideal decomposition
\[
B_n=\bigcup_{m\in\mathcal M_n}\{q\in D_n:m\mid q\}.
\]
Replacing $\mathcal M_n$ by $\mathcal G_n$ gives the displayed union and the
membership criterion.  The minimal elements of this union are exactly the
generators in $\mathcal G_n$, because no two elements of $\mathcal G_n$ are
comparable.

It remains to count the union.  If $\mathcal G_n=\varnothing$, then
$B_n=\varnothing$, so $A(n)=0$ and the empty-sum convention gives
\eqref{eq:low-support-cardinality-formula}.  Otherwise, for $g\in\mathcal G_n$
put $U_g:=\{q\in D_n:g\mid q\}$.  For a nonempty subset
$\mathcal S\subseteq\mathcal G_n$, a divisor $q$ of $F_n$ lies in
$\bigcap_{g\in\mathcal S}U_g$ if and only if
$\lcmop(g:g\in\mathcal S)$ divides $q$.  Since every $g\in\mathcal G_n$ divides
$F_n$, this least common multiple also divides $F_n$, and division by it gives a
bijection
\[
\bigcap_{g\in\mathcal S}U_g
\longleftrightarrow
\operatorname{Div}\left(F_n/\lcmop(g:g\in\mathcal S)\right).
\]
Hence
\[
\#\bigcap_{g\in\mathcal S}U_g=
\tau\left(\frac{F_n}{\lcmop(g:g\in\mathcal S)}\right).
\]
Ordinary finite inclusion--exclusion applied to
$B_n=\bigcup_{g\in\mathcal G_n}U_g$ proves
\eqref{eq:low-support-cardinality-formula}.
\end{proof}

\begin{corollary}[Prime-only three-coordinate skeleton]
\label{cor:primitive-three-skeleton}
Assume $\mathcal P(n)=\{\ell_1,\ell_2,\ell_3\}$ and that
$n_S:=\prod_{i\in S}\ell_i^{\nu_{\ell_i}(n)}$ satisfies
$n_S\ge 3$ and $n_S\notin\{6,12\}$ for every nonempty
$S\subseteq[3]$. Then the four prime-only three-coordinate
types $\Gamma_1$, $\Gamma_4$, $\Gamma_7$, and $\Gamma_9$ are realized in
$\mathcal M_n$. More precisely, after choosing exact-rank primes
$\pi_S$ with $\alpha(\pi_S)=n_S$, the following minimal generators realize
these types:
\[
\pi_{123},\qquad
\pi_{12}\pi_{23},\qquad
\pi_1\pi_{23},\qquad
\pi_1\pi_2\pi_3 .
\]
\end{corollary}

\begin{proof}
The hypothesis and Theorem~\ref{article-thm:primitive-divisors} provide the indicated
exact-rank primes. An exact-rank prime for $n_S$ is a prime jump with
$E_n(\pi_S)=T_n(\pi_S)=S$. The four displayed collections of coordinate
sets are precisely the prime-only types $\Gamma_1$, $\Gamma_4$,
$\Gamma_7$, and $\Gamma_9$ in Lemma~\ref{lem:abstract-three-types}, and
the chosen primes are pairwise distinct. The
pairwise-coprime selection criterion of
Lemma~\ref{lem:three-coordinate-nonemptiness} therefore gives the four
minimal generators.
\end{proof}

\begin{remark}\label{rem:wss}
The only Wall--Sun--Sun input needed here is local.  By
Proposition~\ref{prop:prime-power-ladders}, for $p\neq 2,5$ one has
$\alpha(p^2)=\alpha(p)$ if and only if $\nu_p(F_{\alpha(p)})\ge 2$, the
Wall--Sun--Sun condition for $p$ \cite{SunSun1992FibonacciFLT}.  Thus the
condition merely determines whether the first odd-prime ladder step is delayed;
we do not use any distributional assertion about such primes.
\end{remark}
