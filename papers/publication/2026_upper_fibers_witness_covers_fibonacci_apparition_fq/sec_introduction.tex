\section{Introduction}\label{sec:intro}

Let $F_0=0$, $F_1=1$, and $F_{r+1}=F_r+F_{r-1}$.  For an integer
$q\geq1$, its \emph{rank of apparition} is
\[
 \alpha(q):=\min\{r\geq1:q\mid F_r\},
\]
with $\alpha(1)=1$.  The rank is finite for every $q$; see
\cite{Wall1960FibonacciModm,Vinson1963RankApparition}.  For $n\geq2$ put
\[
 B_n:=\{q:\alpha(q)=n\},\qquad
 \mathcal M_n:=\min_{\mid}B_n.
\]
Thus $\mathcal M_n$ consists of the exact-rank moduli having no proper
divisor of the same rank.  We write $\mathcal P(n)$ for the set of prime
divisors of $n$, $\omega(n)=\#\mathcal P(n)$, and use natural logarithms.
The boundary fiber $B_2$ is empty.

The layer $n=15$ already shows the two basic cover shapes.  Since
\[
 \alpha(2)=3,\qquad \alpha(5)=5,\qquad \alpha(61)=15,
\]
the products $2\cdot5=10$ and $61$ are the two squarefree elements of
$\mathcal M_{15}$.  They correspond, respectively, to the irredundant covers
$\{\{3\},\{5\}\}$ and $\{\{3,5\}\}$ of the prime coordinates of $15$.

For a prime power $\theta=p^e$, set
\[
 d(\theta):=\alpha(\theta),\qquad
 d^-(\theta):=\alpha(p^{e-1}).
\]
Call $\theta$ a \emph{rank jump} if $d^-(\theta)<d(\theta)$.  When
$d(\theta)\mid n$, define
\[
 T_n(\theta):=\{\ell\in\mathcal P(n):
 \nu_\ell(d(\theta))=\nu_\ell(n)\}
\]
and
\[
 E_n(\theta):=\{\ell\in T_n(\theta):
 \nu_\ell(d^-(\theta))<\nu_\ell(n)\}.
\]
An \emph{$n$-witness cover} is a finite, pairwise coprime set $\Theta$ of
rank-jump prime powers such that $d(\theta)\mid n$ for every $\theta$, the
sets $T_n(\theta)$ cover $\mathcal P(n)$, and
\[
 E_n(\theta)\nsubseteq
 \bigcup_{\eta\in\Theta\setminus\{\theta\}}T_n(\eta)
 \qquad(\theta\in\Theta).
\]
The last condition gives every factor a coordinate that is lost on lowering
and is not supplied by the other factors.

\begin{theorem}[Witness-cover classification]\label{thm:witness-classification}
For every $n\geq2$, multiplication gives a bijection
\[
 \Theta\longmapsto\prod_{\theta\in\Theta}\theta
\]
from the $n$-witness covers onto $\mathcal M_n$.  Its inverse sends
$m\in\mathcal M_n$ to
$\{p^{\nu_p(m)}:p\mid m\}$.  In particular, each element of
$\mathcal M_n$ has a unique witness-cover factorization.
\end{theorem}

For a finite set $V$, let $\operatorname{MinCov}(V)$ denote its irredundant
covers by distinct nonempty subsets.  For $n\geq3$, set $V=\mathcal P(n)$
and, for nonempty $S\subseteq V$, put
\[
 n_S:=\prod_{\ell\in S}\ell^{\nu_\ell(n)},\qquad
 a_n(S):=\#\{p\text{ prime}:\alpha(p)=n_S\}.
\]
Let
\[
 \mathcal E_n:=\{S\ne\varnothing:n_S\in\{2,6,12\}\}
\]
and define
\[
 \mathcal R_n^{\rm rp}:=
 \left\{\prod_{S\in\mathcal C}p_S:
 \begin{array}{l}
 \mathcal C\in\operatorname{MinCov}(V),\quad
 \mathcal C\cap\mathcal E_n=\varnothing,\\
 p_S\text{ prime and }\alpha(p_S)=n_S
 \end{array}\right\}.
\]

\begin{theorem}[Rank-pure and squarefree sectors]
\label{thm:rank-pure-universality}
For every $n\geq3$, one has $\#\mathcal E_n\leq2$, each member of
$\mathcal E_n$ has at most two elements, and
$\mathcal R_n^{\rm rp}\subseteq\mathcal M_n$.  The displayed representation
is unique, its support family is $\mathcal C$, and
\begin{equation}\label{eq:rank-pure-weighted-sector}
 \#\mathcal R_n^{\rm rp}=
 \sum_{\substack{\mathcal C\in\operatorname{MinCov}(V)\\
                   \mathcal C\cap\mathcal E_n=\varnothing}}
 \prod_{S\in\mathcal C}a_n(S),
 \qquad
 a_n(S)=\sum_{d\mid n_S}\mu(n_S/d)\omega(F_d).
\end{equation}
If $n$ is squarefree, then
\begin{equation}\label{eq:canonical-squarefree-slice}
 \mathcal R_n^{\rm rp}
 =\mathcal M_n\cap\{q:\mu^2(q)=1\}
 =\min_{\mid}\{q:\alpha(q)=n,\ \mu^2(q)=1\}.
\end{equation}
\end{theorem}

To state the principal bound, for nonempty $I\subseteq J\subseteq\mathcal
P(n)$ let $\mathcal A_{I,J}(n)$ be the rank jumps with
$(E_n,T_n)=(I,J)$, and put
\[
 R(n):=\max\left\{1,
 \max_{\varnothing\ne I\subseteq J\subseteq\mathcal P(n)}
 \#\mathcal A_{I,J}(n)\right\}.
\]

\begin{theorem}[Support bounds]\label{thm:minimal-generator-growth}
For $n\geq3$, the set $\mathcal M_n$ is nonempty.  If
$k=\omega(n)\geq3$ and $r=\lfloor k/2\rfloor$, then
\begin{equation}\label{eq:private-cover-finite-bounds}
 (2^r-1)^{k-r}\leq\#\mathcal M_n
 \leq\sum_{s=1}^{k}\binom{k}{s}(2R(n))^s2^{s(k-s)}.
\end{equation}
Consequently, uniformly when $\omega(n)\to\infty$,
\begin{equation}\label{eq:private-cover-entropy-order}
 \frac{\log2}{4}\omega(n)^2-O(1)
 \leq\log\#\mathcal M_n
 \leq\frac{\log2}{4}\omega(n)^2+
 \omega(n)\log R(n)+O(\omega(n)).
\end{equation}
\end{theorem}

Wagner's minimal multiplicative covers
\cite{Wagner1978MultiplicativeCovers} provide the cover skeleton of
Theorem~\ref{thm:witness-classification}.  For a prime jump, lowering deletes
the rank label, so the witness condition is Wagner's irredundance condition.
Fibonacci prime-power lifting also produces ladder jumps; lowering such a
factor changes one prime coordinate rather than deleting its label.  This is
the arithmetic distinction used by the classification.

Exact-rank prime existence follows from the classical primitive-divisor
theorems of Carmichael and Bilu--Hanrot--Voutier
\cite{Carmichael1913PrimitiveDivisors,
BiluHanrotVoutier2001PrimitiveDivisorsLucas}.  Their primes supply the
weights $a_n(S)$ in Theorem~\ref{thm:rank-pure-universality}.  Thus the
squarefree part of an exact fiber is governed by a familiar minimal-cover
enumeration with arithmetic multiplicities attached to its supports.

Section~\ref{sec:prelim} records the standard Fibonacci facts and exact-rank
prime input.  Sections~\ref{sec:upper-fibers} and
\ref{sec:prime-power-realization} prove the classification and the required
lifting formulas.  Section~\ref{sec:rank-pure} proves the weighted and
squarefree statements, and Section~\ref{sec:growth} proves the support bounds.
