\subsection{Large primitive divisors or exceptional lifting}
\label{subsec:large-primitive-alternative}

We now apply Theorem~\ref{thm:fibotomic-rank-entropy} to the standard
primitive part of a Fibonacci number.  This requires retaining multiplicities
in the classical Lucas cyclotomic-factor theorem, rather than only the radical
divisibility in Lemma~\ref{lem:fibotomic-integer-radical}.

For $n\geq13$, define
\[
 U_n^{\rm prim}:=
 \prod_{\substack{p\mid F_n\\p\nmid 5F_1\cdots F_{n-1}}}
 p^{\nu_p(F_n)}.
\]
Thus the product uses the discriminant-excluding Lucas-sequence convention
for primitive prime divisors.  Write $z=\alpha$ for the rank of apparition in
the standard notation used in the Wall--Sun--Sun literature.

\begin{lemma}[Classical primitive part of the Fibonacci cyclotomic factor]
\label{lem:classical-primitive-part}
For every $n\geq13$ there is an integer $c_n$ such that
\begin{equation}\label{eq:primitive-part-factorization}
 \Psi_n(1)=c_nU_n^{\rm prim},
 \qquad
 c_n=1\quad\text{or}\quad c_n\text{ is a prime divisor of }n.
\end{equation}
In particular,
\begin{equation}\label{eq:primitive-part-ratio}
 1\leq\frac{\Psi_n(1)}{U_n^{\rm prim}}\leq n
\end{equation}
and
\begin{equation}\label{eq:primitive-part-log-mass}
 \log U_n^{\rm prim}
 =\phi_{\rm E}(n)\log\varphi+O(\log n).
\end{equation}
\end{lemma}

\begin{proof}
The Lucas cyclotomic factor denoted $\phi_n$ by Granville is
\[
 \prod_{d\mid n}F_d^{\mu(n/d)}=\Psi_n(1)
\]
in the present Fibonacci specialization.  Granville's cyclotomic-factor
theorem states that a characteristic prime of $F_n$ occurs in this factor
with the same exponent as in $F_n$.  Every noncharacteristic prime $p$ in
the factor occurs only to the first power and satisfies
$n/z(p)=p^a$ for some $a\geq1$; apart from the indices $6$ and $12$, there
is at most one such prime
\cite[Corollaries~3 and~4]{Granville2012PrimitivePrimeFactors}.

At $n\geq13$, the characteristic primes are exactly the primes satisfying
$p\mid F_n$ and $p\nmid F_1\cdots F_{n-1}$.  None is $5$, since $z(5)=5$,
so they are exactly the primitive primes in the convention above.  Any
remaining prime $p$ in $\Psi_n(1)$ satisfies $p\mid n$ by
$n/z(p)=p^a$; this also covers the possible discriminant prime $5$ at an
index divisible by a higher power of $5$.  The cited theorem therefore gives
\eqref{eq:primitive-part-factorization}, including equality of all primitive
exponents.  The bound \eqref{eq:primitive-part-ratio} follows immediately.
Finally, \eqref{eq:fibotomic-binet-error} and
$1\leq c_n\leq n$ give \eqref{eq:primitive-part-log-mass}.
\end{proof}

\begin{theorem}[Large primitive divisor--Fibonacci--Wieferich alternative]
\label{thm:large-primitive-wieferich-alternative}
Let $F_0=0$ and $F_1=1$.  Call a prime $p$ a primitive prime divisor of
$F_n$ when
\[
 p\mid F_n\qquad\text{and}\qquad p\nmid 5F_1F_2\cdots F_{n-1}.
\]
For every real $\varepsilon>0$, there is an integer
$N_\varepsilon\geq13$ such that, for every $n\geq N_\varepsilon$, at least
one of the following holds:
\begin{enumerate}[label=\textup{(\roman*)}]
\item $F_n$ has a primitive prime divisor $p\geq n^{2-\varepsilon}$;
\item $F_n$ has a primitive prime divisor $q$ such that $q^2\mid F_n$.
\end{enumerate}
In the second alternative, $z(q)=n$ and $q$ is a Wall--Sun--Sun,
equivalently Fibonacci--Wieferich, prime.

Consequently, if no Wall--Sun--Sun primes exist and
$P_{\rm prim}(F_n)$ denotes the largest primitive prime divisor of $F_n$,
then
\[
 \liminf_{n\to\infty}
 \frac{\log P_{\rm prim}(F_n)}{\log n}\geq2.
\]
Equivalently, $P_{\rm prim}(F_n)\geq n^{2-o(1)}$.
\end{theorem}

\begin{proof}
Fix $n\geq13$ and suppose alternative \textup{(ii)} fails.  Every primitive
prime then occurs in $F_n$ to exponent one.  The classical primitive-divisor
theorem gives at least one such prime, and hence
\begin{equation}\label{eq:primitive-part-max-bound}
 \log U_n^{\rm prim}
 \leq a(n)\log P_{\rm prim}(F_n).
\end{equation}
For $n\geq13$, the exact-rank primes counted by $a(n)$ are precisely the
primitive prime divisors in the convention of the theorem.  By
\eqref{eq:pointwise-exact-rank-bound} and
\eqref{eq:primitive-part-log-mass},
\[
 a(n)\leq
 \left(\frac{\log\varphi}{2}+o(1)\right)
 \frac{\phi_{\rm E}(n)}{\log n},
 \qquad
 \log U_n^{\rm prim}
 =(\log\varphi+o(1))\phi_{\rm E}(n).
\]
Here $O(\log n)=o(\phi_{\rm E}(n))$ follows from the standard lower bound
$\phi_{\rm E}(n)\gg n/\log\log(3n)$.  Substitution in
\eqref{eq:primitive-part-max-bound} gives, uniformly as $n\to\infty$,
\[
 \log P_{\rm prim}(F_n)\geq(2-o(1))\log n.
\]
The first alternative therefore holds for all sufficiently large $n$, with
the threshold depending only on $\varepsilon$.

It remains to identify the other alternative.  A primitive $q$ has
$q>5$ and $z(q)=n$.  Put $h_q=\nu_q(F_{z(q)})$.  Proposition~
\ref{prop:prime-power-ladders} gives $q^2\mid F_{z(q)}$ exactly when
$h_q\geq2$.  Since $z(q)\mid q-(5/q)$ and $q\nmid q-(5/q)$, the same
valuation formula shows that this is equivalent to
\[
 q^2\mid F_{q-(5/q)},
\]
the standard Wall--Sun--Sun or Fibonacci--Wieferich condition
\cite{Klaska2018WallConjecture}.  The conditional conclusion follows by
excluding alternative \textup{(ii)} for every $n$.
\end{proof}

\begin{remark}[Comparison with known large-divisor results]
\label{rem:large-primitive-comparison}
Stewart proved the unconditional lower bound
$n\exp(\log n/(104\log\log n))$ for the greatest prime factor of the relevant
Lucas cyclotomic factor, beyond an effective sequence-dependent threshold
\cite[Theorem~1]{Stewart2013DivisorsLucasLehmer}.  Hong gives an explicit
Fibonacci formulation: for each fixed positive integer $\kappa$, every
sufficiently large $F_n$ has a primitive divisor outside
$n\pm1,\ldots,\kappa n\pm1$, and hence one at least
$(\kappa+1)n-1$ \cite[Theorem~1.1]{Hong2025BigPrimitiveDivisors}.
The exponent $2-o(1)$ above is much stronger, but only after excluding the
lifting anomaly.

Kiss proved an earlier, different bridge to Wieferich-type primes: if, for a
fixed $0<\delta<1$, the greatest primitive divisor of a Lucas term is below
$n^{1+\delta}$ for almost all indices, then a positive-density set of indices
has a term divisible by a Wieferich-type prime
\cite{Kiss1987WieferichLucas}.  That result is neither pointwise nor an
exact-rank alternative.  The theorem above does not resolve Wall's question;
it says that failure of a near-quadratic primitive divisor cannot occur in an
ordinary rank-lifting layer, but forces a Fibonacci--Wieferich anomaly at that
exact rank.
\end{remark}
