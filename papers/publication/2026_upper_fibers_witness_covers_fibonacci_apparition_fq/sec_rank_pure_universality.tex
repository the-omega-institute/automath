\section{Rank-pure covers and the canonical squarefree slice}
\label{sec:rank-pure}

For a finite set $V$, write $\operatorname{MinCov}(V)$ for the set of
irredundant covers of $V$ by distinct nonempty subsets: a family
$\mathcal C$ belongs to $\operatorname{MinCov}(V)$ when
$\bigcup_{S\in\mathcal C}S=V$ and every $S\in\mathcal C$ contains an element
outside $\bigcup_{T\in\mathcal C\setminus\{S\}}T$.

\begin{theorem}[Rank-pure universality]\label{thm:rank-pure-universality}
Let $n\ge3$, put $V=\mathcal P(n)$, and, for every nonempty $S\subseteq V$,
define
\[
 n_S:=\prod_{\ell\in S}\ell^{\nu_\ell(n)},\qquad
 a_n(S):=\#\Pi_\alpha(n_S),
\]
and
\[
 \mathcal E_n:=\{S\subseteq V:S\ne\varnothing,\ n_S\in\{2,6,12\}\}.
\]
Then $\#\mathcal E_n\le2$, and every member of $\mathcal E_n$ has cardinality
at most two. Define the rank-pure prime sector
\[
 \mathcal R_n^{\rm rp}:=
 \left\{
 \prod_{S\in\mathcal C}p_S:
 \begin{array}{l}
  \mathcal C\in\operatorname{MinCov}(V),\quad
  \mathcal C\cap\mathcal E_n=\varnothing,\\
  p_S\in\Pi_\alpha(n_S)\quad(S\in\mathcal C)
 \end{array}
 \right\}.
\]
Then $\mathcal R_n^{\rm rp}\subseteq\mathcal M_n$, the displayed representation is
unique, and
\begin{equation}\label{eq:rank-pure-weighted-sector}
 \#\mathcal R_n^{\rm rp}=
 \sum_{\substack{\mathcal C\in\operatorname{MinCov}(V)\\
                   \mathcal C\cap\mathcal E_n=\varnothing}}
 \prod_{S\in\mathcal C}a_n(S),
 \qquad
 a_n(S)=\sum_{d\mid n_S}\mu(n_S/d)\,\omega(F_d).
\end{equation}
Moreover, the support hypergraph of the product attached to $\mathcal C$ has
edge set exactly $\mathcal C$. In particular, connected minimal covers map to
$\mathcal M_n^{\rm conn}$, with their complete hypergraph structure preserved.
\end{theorem}

\begin{proof}
The map $S\mapsto n_S$ is injective. If $n_S=2$, then
$S=\{2\}$ and $\nu_2(n)=1$. If $n_S=6$, then
$S=\{2,3\}$ and $\nu_2(n)=\nu_3(n)=1$; if $n_S=12$, then
$S=\{2,3\}$, $\nu_2(n)=2$, and $\nu_3(n)=1$. Thus the ranks $6$ and $12$
cannot both occur, while rank $2$ can coexist only with rank $6$. This proves
the assertions about $\mathcal E_n$.

Fix $\mathcal C\in\operatorname{MinCov}(V)$ avoiding $\mathcal E_n$. For
every $S\in\mathcal C$, the integer $n_S$ is at least $3$ and is not $6$ or
$12$. Theorem~\ref{thm:primitive-divisors} therefore gives a prime
$p_S\in\Pi_\alpha(n_S)$. Distinct supports have distinct ranks, so the primes
$p_S$ are distinct. Since $p_S$ is a prime atom and $d^-(p_S)=1$,
\[
 E_n(p_S)=T_n(p_S)=S.
\]
Consequently the coverage and private-coordinate conditions defining
$\operatorname{MinCov}(V)$ are exactly the witness-cover conditions for
$\{p_S:S\in\mathcal C\}$. Theorem~\ref{thm:witness-classification} places
$\prod_{S\in\mathcal C}p_S$ in $\mathcal M_n$.

Unique factorization recovers every selected prime from the product; its rank
$\alpha(p_S)=n_S$ then recovers $S$. Hence both the cover and all prime choices
are recovered, proving injectivity and the weighted count in
\eqref{eq:rank-pure-weighted-sector}. The formula for $a_n(S)$ is
Lemma~\ref{lem:primitive-prime-mobius-count}. Finally,
\[
 \operatorname{supp}_n(\alpha(p_S))=
 \operatorname{supp}_n(n_S)=S,
\]
so Definition~\ref{def:support-hypergraph} gives precisely the edge set
$\mathcal C$ and therefore preserves connected components.
\end{proof}

\begin{corollary}[Canonical minimal-transversal model]
\label{cor:rank-pure-minimal-transversals}
Retain the notation of Theorem~\ref{thm:rank-pure-universality}. Define a
hypergraph $\mathcal H_n^{\rm rp}$ with vertex set
\[
 U_n:=\{(S,p):\varnothing\ne S\subseteq V,\ S\notin\mathcal E_n,
                    \ p\in\Pi_\alpha(n_S)\}
\]
and, for each $\ell\in V$, the coordinate edge
\[
 X_\ell:=\{(S,p)\in U_n:\ell\in S\}.
\]
Multiplication induces a bijection
\begin{equation}\label{eq:rank-pure-minimal-transversal-bijection}
 \operatorname{MinTr}(\mathcal H_n^{\rm rp})
 \longrightarrow \mathcal R_n^{\rm rp},
 \qquad Q\longmapsto\prod_{(S,p)\in Q}p,
\end{equation}
where $\operatorname{MinTr}$ denotes the inclusion-minimal transversals.
Hence the weighted cover polynomial
\eqref{eq:rank-pure-weighted-sector} is exactly a minimal-transversal count of
a canonically defined hypergraph. By the standard split-incidence
correspondence \cite[Proposition~30]{KanteLimouzyMaryNourine2014DominatingSets},
it also counts the minimal connected dominating sets of the associated split
graph.
\end{corollary}

\begin{proof}
A set $Q\subseteq U_n$ is a transversal precisely when its support labels
cover $V$. If a minimal transversal contained $(S,p)$ and $(S,q)$ with
$p\ne q$, removing either vertex would preserve every coordinate edge hit by
it, a contradiction. Thus its support labels are distinct. Minimality also
says that every $(S,p)\in Q$ has a coordinate $\ell\in S$ belonging to no
other selected support. The support labels therefore form an irredundant
cover. Conversely, selected prime choices over an irredundant cover form a
minimal transversal, since every selected vertex has such a private
coordinate edge. Theorem~\ref{thm:rank-pure-universality} and unique
factorization now prove that multiplication is the asserted bijection.
\end{proof}

The rank-pure sector has a canonical description on squarefree indices.  In
particular, it is not then an auxiliary class defined only through the cover
construction.

\begin{corollary}[Canonical squarefree exact-fiber slice]
\label{cor:canonical-squarefree-slice}
Let $n\ge3$ be squarefree.  Then
\begin{equation}\label{eq:canonical-squarefree-slice}
 \mathcal R_n^{\rm rp}
 =\mathcal M_n\cap\{q:\mu^2(q)=1\}
 =\min_{\mid}\{q:\alpha(q)=n,\ \mu^2(q)=1\}.
\end{equation}
Thus the weighted sum in
\eqref{eq:rank-pure-weighted-sector} counts exactly the divisibility-minimal
squarefree elements of the standard exact fiber $\alpha^{-1}(n)$.
\end{corollary}

\begin{proof}
Proposition~\ref{prop:rank-basics} gives
$B_n=\{q:\alpha(q)=n\}=\alpha^{-1}(n)$.  If a squarefree member of $B_n$
has a proper divisor in $B_n$, that divisor is again squarefree.  Hence
minimality in the squarefree part of the exact fiber is equivalent to
membership in $\mathcal M_n$.

Every element of $\mathcal R_n^{\rm rp}$ is a product of distinct primes and
belongs to $\mathcal M_n$, which proves one inclusion in the first equality.
Conversely, let $m\in\mathcal M_n$ be squarefree.  Its unique atomic
factorization from Theorem~\ref{thm:witness-classification} consists only of
prime atoms $p$.  Put $S=T_n(p)$.  Since $\alpha(p)\mid n$ and $n$ is
squarefree,
\[
 \alpha(p)=\prod_{\ell\in S}\ell=n_S.
\]
Moreover $d^-(p)=1$, so $E_n(p)=T_n(p)=S$.  The witness-cover conditions are
therefore precisely that the distinct supports $S$ form an irredundant cover
of $\mathcal P(n)$.  No such support lies in $\mathcal E_n$, because the
corresponding prime $p$ itself has exact rank $n_S$.  Thus the atomic
factorization of $m$ is one of the uniquely represented products defining
$\mathcal R_n^{\rm rp}$.  This proves the reverse inclusion and the result.
\end{proof}

\subsection*{Worked exact-fiber layers}

\begin{example}[An odd squarefree layer without exceptions]
Let $n=15$ and label its prime coordinates by $V=\{3,5\}$. The relevant
Fibonacci factorizations are
\[
 F_3=2,\qquad F_5=5,\qquad F_{15}=2\cdot5\cdot61,
\]
so
\[
 \Pi_\alpha(3)=\{2\},\qquad
 \Pi_\alpha(5)=\{5\},\qquad
 \Pi_\alpha(15)=\{61\}.
\]
There is no exceptional support. For the prime atoms one has
\[
 T_{15}(2)=E_{15}(2)=\{3\},\quad
 T_{15}(5)=E_{15}(5)=\{5\},\quad
 T_{15}(61)=E_{15}(61)=\{3,5\}.
\]
The two irredundant covers of $V$ are
$\{\{3,5\}\}$ and $\{\{3\},\{5\}\}$. They give the products
\[
 61\qquad\text{and}\qquad2\cdot5=10,
\]
respectively. Thus the canonical squarefree slice is
$\mathcal R_{15}^{\rm rp}=\{10,61\}$, with each atom, support, essential
support, cover, and product visible in the displays.
\end{example}

\begin{example}[An even layer with exceptional supports]
Let $n=6$ and $V=\{2,3\}$. Here the supports $\{2\}$ and $\{2,3\}$ carry
the exceptional ranks $2$ and $6$, respectively. The only nonexceptional
support $\{3\}$ cannot cover $V$, so $\mathcal R_6^{\rm rp}=\varnothing$.
On the other hand, $4$ is a ladder atom with
\[
 d(4)=\alpha(4)=6,\qquad d^-(4)=\alpha(2)=3,
\]
and hence
\[
 T_6(4)=\{2,3\},\qquad E_6(4)=\{2\}.
\]
The singleton $\{4\}$ is a witness cover and its product is $4$. Directly,
$F_6=8$ and the ranks of its divisors $1,2,4,8$ are $1,3,6,6$, so
$\mathcal M_6=\{4\}$. This shows exactly how the exceptional support removes
the prime realization while a lowered-label ladder remains.
\end{example}

\begin{example}[The ladder boundary at $n=91$]
The coordinates are $V=\{7,13\}$. Since $F_7=13$, one has
$\alpha(13)=7$ and $\nu_{13}(F_7)=1$. Proposition~
\ref{prop:prime-power-ladders} therefore gives
\[
 d(13^2)=\alpha(13^2)=13\alpha(13)=91,
 \qquad d^-(13^2)=7.
\]
Consequently
\[
 T_{91}(169)=\{7,13\},\qquad E_{91}(169)=\{13\}.
\]
The singleton cover $\{169\}$ is irredundant, and its product is $169$, so
$169\in\mathcal M_{91}$. It is a squareful ladder atom and therefore lies
outside the prime-only rank-pure sector:
\[
 169\in\mathcal M_{91}\setminus\mathcal R_{91}^{\rm rp}.
\]
\end{example}

\begin{corollary}[Odd-layer cover universality]
\label{cor:odd-rank-pure-universality}
Let $n\ge3$ be odd and put $k=\omega(n)$. Then $\mathcal E_n=\varnothing$.
After fixing one prime $\pi_S\in\Pi_\alpha(n_S)$ for every nonempty
$S\subseteq\mathcal P(n)$, the map
\[
 \mathcal C\longmapsto\prod_{S\in\mathcal C}\pi_S
\]
embeds $\operatorname{MinCov}(\mathcal P(n))$ into $\mathcal M_n$ and
preserves the support hypergraph. Hence, if $C_k$ denotes the number of minimal
covers of a labelled $k$-set, then
\begin{equation}\label{eq:odd-minimal-cover-bound}
 \#\mathcal M_n\ge C_k,
 \qquad
 \widthop(B_n)\ge C_k.
\end{equation}
Through Royle's correspondence
\cite{Royle2000SetCoversSplitGraphs}, every isomorphism type of split graph on
$k$ vertices consequently has a rank-pure Fibonacci realization on the prime
coordinates of $n$. This last sentence concerns unlabelled isomorphism types
only and is not used to enumerate the labelled covers below.
\end{corollary}

\begin{proof}
Every nonempty $n_S$ is odd and at least $3$, so no support is exceptional.
Theorem~\ref{thm:rank-pure-universality} gives the embedding and support
preservation. Minimal generators form an antichain, which proves the width
bound. Royle's theorem is a correspondence between isomorphism types of split
graphs and minimal covers; choosing a labelling of a representative cover and
then applying the embedding gives the final assertion.
\end{proof}

We next fix the classical enumeration input. Let $C_{k,s}$ count the minimal
covers of a labelled $k$-set having
$s$ members, put $C_k=\sum_s C_{k,s}$, and put
\[
 b_k:=\sum_{s=0}^k\binom{k}{s}2^{s(k-s)}.
\]
Hearne and Wagner \cite{HearneWagner1973MinimalCovers} proved
\begin{equation}\label{eq:hearne-wagner}
 C_{k,s}=\sum_{j=s}^k\binom{k}{j}
 \genfrac{\{}{\}}{0pt}{}{j}{s}
 (2^s-s-1)^{k-j}.
\end{equation}
The prime coordinates of $n$ are specific primes, so the arithmetic embedding
uses these labelled covers: there is no quotient by $S_k$ and no division by
$k!$. Royle's correspondence above concerns isomorphism classes and supplies
only the realization statement. The labelled central sum $b_k$ is the object
in the split-graph enumeration of
Bender--Richmond--Wormald \cite{BenderRichmondWormald1985ChordalSplit}, in the
form recorded by Troyka \cite[Section~3]{Troyka2019SplitGraphs}. The following
direct comparison closes the conversion from the Hearne--Wagner labelled
formula to that labelled central sum.

\begin{lemma}[Labelled minimal-cover comparison]
\label{lem:labelled-minimal-cover-comparison}
As $k\to\infty$,
\[
 \frac{C_k}{b_k}\longrightarrow1.
\]
\end{lemma}

\begin{proof}
Fix an order on the labelled ground set. In a minimal cover with $s$ members,
select the least private element of each member. The selected elements form an
$s$-set $P$; no selected element can occur in another member, and each member
is recovered from its selected element and a subset of the remaining
$k-s$ elements. This canonical encoding is injective, so
\[
 C_{k,s}\le\binom{k}{s}2^{s(k-s)}.
\]
The $j=s$ term of \eqref{eq:hearne-wagner} gives the matching lower bound
\[
 C_{k,s}\ge\binom{k}{s}(2^s-s-1)^{k-s}.
\]
Uniformly for $k/3\le s\le2k/3$,
\[
 \left(1-\frac{s+1}{2^s}\right)^{k-s}=1-o(1).
\]
The contribution to $b_k$ from the complementary values of $s$ is at most
\[
 2^k2^{2k^2/9}=o\bigl(2^{\lfloor k^2/4\rfloor}\bigr)=o(b_k).
\]
Summing the two bounds over the central range therefore gives
$(1-o(1))b_k\le C_k\le b_k$, which proves the claim.
\end{proof}

Consequently, with
\[
 \vartheta_0:=\sum_{r\in\mathbb Z}2^{-r^2},\qquad
 \vartheta_1:=\sum_{r\in\mathbb Z}2^{-(r+1/2)^2},\qquad
 \varepsilon\equiv k\pmod2,
\]
the published labelled asymptotic for $b_k$ and
Lemma~\ref{lem:labelled-minimal-cover-comparison} give
\begin{equation}\label{eq:minimal-cover-theta}
 C_k\sim b_k\sim
 \vartheta_\varepsilon\binom{k}{\lfloor k/2\rfloor}2^{k^2/4}.
\end{equation}
In particular, $b_{k-1}/b_k=O(2^{-k/2})$.

\begin{corollary}[Weighted-cover obstruction on odd layers]
\label{cor:odd-weighted-cover-obstruction}
Let $n\ge3$ be odd, put $V=\mathcal P(n)$ and $k=\omega(n)$, and retain the
weights $a_n(S)=\#\Pi_\alpha(n_S)$ from
Theorem~\ref{thm:rank-pure-universality}. Then
\begin{equation}\label{eq:odd-weighted-rank-pure-sector}
 \#\mathcal R_n^{\rm rp}
 =\sum_{\mathcal C\in\operatorname{MinCov}(V)}
       \prod_{S\in\mathcal C}a_n(S)
 \le\#\mathcal M_n.
\end{equation}
Every weight in this sum is positive. Consequently, with
\begin{equation}\label{eq:weighted-cover-excess}
 \mathcal W(n):=\frac1{C_k}
 \sum_{\mathcal C\in\operatorname{MinCov}(V)}
       \prod_{S\in\mathcal C}a_n(S)\ge1,
\end{equation}
one has
\begin{equation}\label{eq:odd-rank-pure-log-decomposition}
 \log\#\mathcal R_n^{\rm rp}
 =\frac{\log2}{4}k^2+\log\mathcal W(n)+O(k).
\end{equation}
Therefore an almost-all asymptotic
\[
 \log\#\mathcal M_n\sim\frac{\log2}{4}(\log\log n)^2
\]
would necessarily imply
\begin{equation}\label{eq:weighted-cover-dispersion-interface}
 \log\mathcal W(n)=o\bigl((\log\log n)^2\bigr)
\end{equation}
outside an $o(x)$ exceptional set of odd $n\le x$.
The converse is not asserted: ladder atoms and other non-rank-pure witness
covers still require an upper bound.  This distinction occurs already at the
odd squarefree index $91$, where $\alpha(13)=7$ and
$\alpha(13^2)=91$, so $169\in\mathcal M_{91}\setminus\mathcal R_{91}^{\rm rp}$.
\end{corollary}

\begin{proof}
For odd $n$, every $n_S$ is odd and at least $3$, so
$\mathcal E_n=\varnothing$ and the exact-rank existence theorem gives
$a_n(S)\ge1$.  The weighted identity and inclusion are
\eqref{eq:rank-pure-weighted-sector}.  The asymptotic
\eqref{eq:minimal-cover-theta} gives
$\log C_k=(\log2)k^2/4+O(k)$ and proves
\eqref{eq:odd-rank-pure-log-decomposition}.

If the displayed almost-all formula for $\mathcal M_n$ held, then it would
hold outside an $o(x)$ subset of the odd integers.  Intersecting with the
Hardy--Ramanujan normal-order set and using
$\#\mathcal R_n^{\rm rp}\le\#\mathcal M_n$ forces
\eqref{eq:weighted-cover-dispersion-interface}.  Finally, the classical
lifting formula gives $\alpha(13^2)=13\alpha(13)=91$ and
$\alpha(13)=7<91$.  Thus $13^2$ is an atom of rank $91$ and the singleton
witness cover $\{13^2\}$ gives the stated strict inclusion.
\end{proof}

The following sharp extremal calculation separates total primitive-prime
mass from the dispersion information retained by the cover polynomial.

\begin{proposition}[Sharp cumulative-data obstruction]
\label{prop:sharp-cumulative-data-obstruction}
Let $V$ be a labelled $k$-set, $k\ge1$, and assign a real weight
$w_S\ge1$ to every nonempty $S\subseteq V$.  Put
\[
 Z_V(w):=\sum_{\mathcal C\in\operatorname{MinCov}(V)}
              \prod_{S\in\mathcal C}w_S,
 \qquad A:=\sum_{\varnothing\ne S\subseteq V}w_S.
\]
Then
\begin{equation}\label{eq:sharp-total-mass-cover-bound}
 Z_V(w)\ge C_k+A-(2^k-1).
\end{equation}
This is best possible for every $A\ge2^k-1$: equality is attained by
\[
 w_V=A-(2^k-2),\qquad w_S=1\quad(S\ne V).
\]
If $k\ge3$, this is the only equality profile.

Consequently, if $n\ge3$ is odd and squarefree and $k=\omega(n)$, then
\begin{equation}\label{eq:squarefree-cumulative-rank-pure-bound}
 \#\{m\in\mathcal M_n:\mu^2(m)=1\}
 \ge C_k+\omega(F_n)-(2^k-1),
\end{equation}
and
\begin{equation}\label{eq:squarefree-cumulative-weighted-excess}
 \mathcal W(n)\ge
 1+\frac{\omega(F_n)-(2^k-1)}{C_k}.
\end{equation}
Among all positive weight profiles with the same total
$\sum_{d\mid n,\ d>1}a(d)=\omega(F_n)$, this conclusion is exact.  The
extremizing profile is an abstract information-theoretic model; it is not
asserted to be realized by Fibonacci exact-rank primes.
\end{proposition}

\begin{proof}
Write $w_S=1+x_S$ with $x_S\ge0$.  For every minimal cover $\mathcal C$,
\[
 \prod_{S\in\mathcal C}(1+x_S)
 \ge1+\sum_{S\in\mathcal C}x_S.
\]
Every nonempty $S\subseteq V$ occurs in a minimal cover: use $\{V\}$ when
$S=V$, and otherwise use
$\{S\}\cup\{\{v\}:v\in V\setminus S\}$.  Summing the preceding inequality
over all minimal covers therefore gives
\[
 Z_V(w)\ge C_k+\sum_{\varnothing\ne S\subseteq V}x_S
 =C_k+A-(2^k-1).
\]
The only minimal cover containing $V$ is $\{V\}$.  Hence the displayed
full-support profile has weighted sum
$C_k-1+w_V=C_k+A-(2^k-1)$, proving sharpness.

Suppose $k\ge3$ and $S$ is proper and nonempty. If $|S|=1$, both the
singleton partition of $V$ and $\{S,V\setminus S\}$ are minimal covers
containing $S$. If $|S|\ge2$, choose $s\in S$ and $t\in V\setminus S$; in
addition to the preceding singleton-complement cover, there is the minimal
cover consisting of $S$, $\{s,t\}$, and the singletons
$\{v\}$ for $v\in V\setminus(S\cup\{t\})$. Thus every proper $S$ occurs in
at least two minimal covers. In the summed linear lower bound its coefficient
is therefore at least two, so equality forces $x_S=0$ for every proper $S$.
This proves the equality classification.

For odd squarefree $n$, the nonempty subsets of $\mathcal P(n)$ correspond
to the divisors $d>1$ of $n$.  Exact ranks partition the prime divisors of
$F_n$, and $a(1)=0$, so
\[
 \sum_{\varnothing\ne S\subseteq\mathcal P(n)}a(n_S)
 =\sum_{d\mid n}a(d)=\omega(F_n).
\]
Every weight is positive by exact-rank prime existence.  Apply
\eqref{eq:sharp-total-mass-cover-bound}, then use
Corollary~\ref{cor:canonical-squarefree-slice} and
\eqref{eq:weighted-cover-excess}.
\end{proof}
