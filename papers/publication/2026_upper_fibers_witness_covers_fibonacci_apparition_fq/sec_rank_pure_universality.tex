\begin{theorem}[Rank-pure universality]\label{thm:rank-pure-universality}
Let $n\ge3$, put $V=\mathcal P(n)$, and, for every nonempty $S\subseteq V$,
define
\[
 n_S:=\prod_{\ell\in S}\ell^{\nu_\ell(n)},\qquad
 a_n(S):=\#\Pi_\alpha(n_S),
\]
and
\[
 \mathcal E_n:=\{S\subseteq V:S\ne\varnothing,\ n_S\in\{2,6,12\}\}.
\]
Then $\#\mathcal E_n\le2$, and every member of $\mathcal E_n$ has cardinality
at most two. Define the rank-pure prime sector
\[
 \mathcal R_n^{\rm rp}:=
 \left\{
 \prod_{S\in\mathcal C}p_S:
 \begin{array}{l}
  \mathcal C\in\operatorname{MinCov}(V),\quad
  \mathcal C\cap\mathcal E_n=\varnothing,\\
  p_S\in\Pi_\alpha(n_S)\quad(S\in\mathcal C)
 \end{array}
 \right\}.
\]
Then $\mathcal R_n^{\rm rp}\subseteq\mathcal M_n$, the displayed representation is
unique, and
\begin{equation}\label{eq:rank-pure-weighted-sector}
 \#\mathcal R_n^{\rm rp}=
 \sum_{\substack{\mathcal C\in\operatorname{MinCov}(V)\\
                   \mathcal C\cap\mathcal E_n=\varnothing}}
 \prod_{S\in\mathcal C}a_n(S),
 \qquad
 a_n(S)=\sum_{d\mid n_S}\mu(n_S/d)\,\omega(F_d).
\end{equation}
Moreover, the support hypergraph of the product attached to $\mathcal C$ has
edge set exactly $\mathcal C$. In particular, connected minimal covers map to
$\mathcal M_n^{\rm conn}$, with their complete hypergraph structure preserved.
\end{theorem}

\begin{proof}
The map $S\mapsto n_S$ is injective. If $n_S=2$, then
$S=\{2\}$ and $\nu_2(n)=1$. If $n_S=6$, then
$S=\{2,3\}$ and $\nu_2(n)=\nu_3(n)=1$; if $n_S=12$, then
$S=\{2,3\}$, $\nu_2(n)=2$, and $\nu_3(n)=1$. Thus the ranks $6$ and $12$
cannot both occur, while rank $2$ can coexist only with rank $6$. This proves
the assertions about $\mathcal E_n$.

Fix $\mathcal C\in\operatorname{MinCov}(V)$ avoiding $\mathcal E_n$. For
every $S\in\mathcal C$, the integer $n_S$ is at least $3$ and is not $6$ or
$12$. Theorem~\ref{thm:primitive-divisors} therefore gives a prime
$p_S\in\Pi_\alpha(n_S)$. Distinct supports have distinct ranks, so the primes
$p_S$ are distinct. Since $p_S$ is a prime atom and $d^-(p_S)=1$,
\[
 E_n(p_S)=T_n(p_S)=S.
\]
Consequently the coverage and private-coordinate conditions defining
$\operatorname{MinCov}(V)$ are exactly the witness-cover conditions for
$\{p_S:S\in\mathcal C\}$. Theorem~\ref{thm:witness-classification} places
$\prod_{S\in\mathcal C}p_S$ in $\mathcal M_n$.

Unique factorization recovers every selected prime from the product; its rank
$\alpha(p_S)=n_S$ then recovers $S$. Hence both the cover and all prime choices
are recovered, proving injectivity and the weighted count in
\eqref{eq:rank-pure-weighted-sector}. The formula for $a_n(S)$ is
Lemma~\ref{lem:primitive-prime-mobius-count}. Finally,
\[
 \operatorname{supp}_n(\alpha(p_S))=
 \operatorname{supp}_n(n_S)=S,
\]
so Definition~\ref{def:support-hypergraph} gives precisely the edge set
$\mathcal C$ and therefore preserves connected components.
\end{proof}

\begin{corollary}[Odd-layer cover universality]
\label{cor:odd-rank-pure-universality}
Let $n\ge3$ be odd and put $k=\omega(n)$. Then $\mathcal E_n=\varnothing$.
After fixing one prime $\pi_S\in\Pi_\alpha(n_S)$ for every nonempty
$S\subseteq\mathcal P(n)$, the map
\[
 \mathcal C\longmapsto\prod_{S\in\mathcal C}\pi_S
\]
embeds $\operatorname{MinCov}(\mathcal P(n))$ into $\mathcal M_n$ and
preserves the support hypergraph. Hence, if $C_k$ denotes the number of minimal
covers of a labelled $k$-set, then
\begin{equation}\label{eq:odd-minimal-cover-bound}
 \#\mathcal M_n\ge C_k,
 \qquad
 \widthop(B_n)\ge C_k.
\end{equation}
Through Royle's correspondence
\cite{Royle2000SetCoversSplitGraphs}, every isomorphism type of split graph on
$k$ vertices consequently has a rank-pure Fibonacci realization on the prime
coordinates of $n$.
\end{corollary}

\begin{proof}
Every nonempty $n_S$ is odd and at least $3$, so no support is exceptional.
Theorem~\ref{thm:rank-pure-universality} gives the embedding and support
preservation. Minimal generators form an antichain, which proves the width
bound. Royle's theorem is a correspondence between isomorphism types of split
graphs and minimal covers; choosing a labelling of a representative cover and
then applying the embedding gives the final assertion.
\end{proof}

We next fix the classical enumeration input, without claiming it as a result of
this paper. Let $C_{k,s}$ count the minimal covers of a labelled $k$-set having
$s$ members, put $C_k=\sum_s C_{k,s}$, and put
\[
 b_k:=\sum_{s=0}^k\binom{k}{s}2^{s(k-s)}.
\]
Hearne and Wagner \cite{HearneWagner1973MinimalCovers} proved
\begin{equation}\label{eq:hearne-wagner}
 C_{k,s}=\sum_{j=s}^k\binom{k}{j}
 \genfrac{\{}{\}}{0pt}{}{j}{s}
 (2^s-s-1)^{k-j}.
\end{equation}
The central-sum asymptotic occurring in the labelled split-graph enumeration of
Bender--Richmond--Wormald \cite{BenderRichmondWormald1985ChordalSplit}, in the
form recorded also by Troyka \cite[Section~3]{Troyka2019SplitGraphs}, gives,
with
\[
 \vartheta_0:=\sum_{r\in\mathbb Z}2^{-r^2},\qquad
 \vartheta_1:=\sum_{r\in\mathbb Z}2^{-(r+1/2)^2},\qquad
 \varepsilon\equiv k\pmod2,
\]
the published asymptotic
\begin{equation}\label{eq:minimal-cover-theta}
 C_k\sim b_k\sim
 \vartheta_\varepsilon\binom{k}{\lfloor k/2\rfloor}2^{k^2/4}.
\end{equation}
In particular, $b_{k-1}/b_k=O(2^{-k/2})$.
