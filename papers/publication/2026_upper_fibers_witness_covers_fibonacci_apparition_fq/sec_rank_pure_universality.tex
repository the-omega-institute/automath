\section{Rank-pure covers and the squarefree slice}\label{sec:rank-pure}

\begin{proof}[Proof of Theorem~\ref{thm:rank-pure-universality}]
The map $S\mapsto n_S$ is injective.  The exceptional ranks have supports
\[
 n_S=2:\ S=\{2\},\qquad
 n_S\in\{6,12\}:\ S=\{2,3\}.
\]
Ranks $6$ and $12$ cannot both occur, while rank $2$ can coexist only with
rank $6$.  Hence $\#\mathcal E_n\leq2$, and every exceptional support has at
most two elements.

Fix $\mathcal C\in\operatorname{MinCov}(V)$ avoiding $\mathcal E_n$.
For each $S\in\mathcal C$, Theorem~\ref{thm:primitive-divisors} gives a
prime $p_S$ with $\alpha(p_S)=n_S$.  Distinct supports give distinct ranks
and therefore distinct primes.  Since $p_S$ is a prime jump,
\[
 E_n(p_S)=T_n(p_S)=S.
\]
Thus the cover and private-coordinate conditions on $\mathcal C$ are exactly
the witness-cover conditions for $\{p_S:S\in\mathcal C\}$.  Theorem~
\ref{thm:witness-classification} places its product in $\mathcal M_n$.

Unique factorization recovers each selected prime, its rank recovers $S$, and
therefore the representation is unique.  Summing the independent prime
choices gives \eqref{eq:rank-pure-weighted-sector}; the formula for $a_n(S)$
is Lemma~\ref{lem:primitive-prime-mobius-count}.  The support family of the
product is exactly $\mathcal C$, so connected covers also give connected
support hypergraphs.

Now suppose $n$ is squarefree.  Every member of $\mathcal R_n^{\rm rp}$ is
a squarefree element of $\mathcal M_n$.  Conversely, the witness cover of a
squarefree $m\in\mathcal M_n$ consists only of primes.  If
$S=T_n(p)$, then squarefreeness of $n$ gives
\[
 \alpha(p)=\prod_{\ell\in S}\ell=n_S,
 \qquad E_n(p)=T_n(p)=S.
\]
The supports form an irredundant cover and none is exceptional, since it is
realized by the exact-rank prime $p$.  Hence $m\in\mathcal R_n^{\rm rp}$.
Finally, a proper divisor of a squarefree exact-rank modulus is squarefree,
so minimality in the squarefree part of the fiber is equivalent to membership
in $\mathcal M_n$.  This proves \eqref{eq:canonical-squarefree-slice}.
\end{proof}

\begin{corollary}[Transversal interpretation]
\label{cor:rank-pure-minimal-transversals}
Let $U_n$ consist of the pairs $(S,p)$ with
$\varnothing\ne S\subseteq\mathcal P(n)$, $S\notin\mathcal E_n$, and
$\alpha(p)=n_S$.  For each $\ell\mid n$, let
\[
 X_\ell:=\{(S,p)\in U_n:\ell\in S\}.
\]
Multiplication gives a bijection from the inclusion-minimal transversals of
the hypergraph with edges $X_\ell$ onto $\mathcal R_n^{\rm rp}$.  The
standard split-incidence correspondence therefore also identifies these
products with the minimal connected dominating sets of the associated split
graph \cite[Proposition~30]{KanteLimouzyMaryNourine2014DominatingSets}.
\end{corollary}

\begin{proof}
A transversal is minimal precisely when its support labels form an
irredundant cover.  It cannot contain two vertices with the same label $S$,
because either could then be removed.  The conclusion follows from
Theorem~\ref{thm:rank-pure-universality} and unique factorization.
\end{proof}

This corollary is only an encoding of the weighted sum; no assertion about
general transversal enumeration is needed here.

\begin{example}[The squarefree layer $n=15$]
The relevant exact-rank prime sets are
\[
 \Pi_\alpha(3)=\{2\},\qquad
 \Pi_\alpha(5)=\{5\},\qquad
 \Pi_\alpha(15)=\{61\}.
\]
The two irredundant covers of $\{3,5\}$ are
$\{\{3,5\}\}$ and $\{\{3\},\{5\}\}$.  They give
\[
 \mathcal R_{15}^{\rm rp}=\mathcal M_{15}=\{61,10\}.
\]
\end{example}

\begin{example}[The ladder boundary at $n=91$]
Since $F_7=13$, one has $\alpha(13)=7$ and
$\nu_{13}(F_7)=1$.  Proposition~\ref{prop:prime-power-ladders} gives
\[
 \alpha(13^2)=91,\qquad d^-(13^2)=7,
\]
so
\[
 T_{91}(169)=\{7,13\},\qquad E_{91}(169)=\{13\}.
\]
The singleton $\{169\}$ is a witness cover.  Therefore
\[
 169\in\mathcal M_{91}\setminus\mathcal R_{91}^{\rm rp}.
\]
\end{example}

\begin{corollary}[Odd-layer cover universality]
\label{cor:odd-rank-pure-universality}
Let $n\geq3$ be odd and $k=\omega(n)$.  After choosing one prime
$\pi_S\in\Pi_\alpha(n_S)$ for every nonempty
$S\subseteq\mathcal P(n)$, the map
\[
 \mathcal C\longmapsto\prod_{S\in\mathcal C}\pi_S
\]
embeds $\operatorname{MinCov}(\mathcal P(n))$ into $\mathcal M_n$ and
preserves the support family.  If $C_k$ is the number of minimal covers of a
labelled $k$-set, then
\begin{equation}\label{eq:odd-minimal-cover-bound}
 \#\mathcal M_n\geq C_k,
 \qquad \operatorname{width}(B_n)\geq C_k.
\end{equation}
\end{corollary}

\begin{proof}
Every $n_S$ is odd and at least $3$, so no support is exceptional.
Theorem~\ref{thm:rank-pure-universality} gives the embedding.  The minimal
elements of $B_n$ form an antichain, proving the width bound.
\end{proof}

Let $C_{k,s}$ count the minimal covers of a labelled $k$-set having $s$
members, put $C_k=\sum_s C_{k,s}$, and set
\[
 b_k:=\sum_{s=0}^k\binom{k}{s}2^{s(k-s)}.
\]
Hearne and Wagner \cite{HearneWagner1973MinimalCovers} proved
\begin{equation}\label{eq:hearne-wagner}
 C_{k,s}=\sum_{j=s}^k\binom{k}{j}
 \genfrac{\{}{\}}{0pt}{}{j}{s}(2^s-s-1)^{k-j}.
\end{equation}

\begin{lemma}[Labelled minimal-cover comparison]
\label{lem:labelled-minimal-cover-comparison}
As $k\to\infty$, one has $C_k/b_k\to1$.
\end{lemma}

\begin{proof}
Fix an order on the ground set.  In a minimal cover with $s$ members, select
the least private element of each member.  The selected elements form an
$s$-set, and each cover member is recovered from its selected element and a
subset of the remaining $k-s$ elements.  Hence
\[
 C_{k,s}\leq\binom{k}{s}2^{s(k-s)}.
\]
The term $j=s$ in \eqref{eq:hearne-wagner} gives
\[
 C_{k,s}\geq\binom{k}{s}(2^s-s-1)^{k-s}.
\]
Uniformly for $k/3\leq s\leq2k/3$, the ratio of these bounds tends to one.
The complementary contribution to $b_k$ is at most
$2^k2^{2k^2/9}=o(2^{\lfloor k^2/4\rfloor})=o(b_k)$, and summation proves
the result.
\end{proof}

With
\[
 \vartheta_0:=\sum_{r\in\mathbb Z}2^{-r^2},\qquad
 \vartheta_1:=\sum_{r\in\mathbb Z}2^{-(r+1/2)^2},\qquad
 \varepsilon\equiv k\pmod2,
\]
the labelled split-graph asymptotic of
Bender--Richmond--Wormald and Troyka
\cite{BenderRichmondWormald1985ChordalSplit,Troyka2019SplitGraphs}, together
with the lemma, gives
\begin{equation}\label{eq:minimal-cover-theta}
 C_k\sim b_k\sim
 \vartheta_\varepsilon\binom{k}{\lfloor k/2\rfloor}2^{k^2/4}.
\end{equation}

\begin{corollary}[Weighted odd layers]
\label{cor:odd-weighted-cover-obstruction}
Let $n\geq3$ be odd, $V=\mathcal P(n)$, and $k=\omega(n)$.  Then
\begin{equation}\label{eq:odd-weighted-rank-pure-sector}
 \#\mathcal R_n^{\rm rp}
 =\sum_{\mathcal C\in\operatorname{MinCov}(V)}
       \prod_{S\in\mathcal C}a_n(S)
 \leq\#\mathcal M_n.
\end{equation}
All weights are positive.  Consequently, if
\begin{equation}\label{eq:weighted-cover-excess}
 \mathcal W(n):=\frac1{C_k}
 \sum_{\mathcal C\in\operatorname{MinCov}(V)}
       \prod_{S\in\mathcal C}a_n(S),
\end{equation}
then $\mathcal W(n)\geq1$ and
\begin{equation}\label{eq:odd-rank-pure-log-decomposition}
 \log\#\mathcal R_n^{\rm rp}
 =\frac{\log2}{4}k^2+\log\mathcal W(n)+O(k).
\end{equation}
In particular, an almost-all asymptotic
$\log\#\mathcal M_n\sim(\log2)(\log\log n)^2/4$ would force
\begin{equation}\label{eq:weighted-cover-dispersion-interface}
 \log\mathcal W(n)=o((\log\log n)^2)
\end{equation}
outside an $o(x)$ subset of the odd integers up to $x$.
\end{corollary}

\begin{proof}
For odd $n$, exact-rank existence gives $a_n(S)\geq1$ for every nonempty
$S$.  Equation~\eqref{eq:rank-pure-weighted-sector} and
\eqref{eq:minimal-cover-theta} give the first three assertions.  For the last,
intersect the asserted almost-all set with the Hardy--Ramanujan normal-order
set and use $\#\mathcal R_n^{\rm rp}\leq\#\mathcal M_n$.
\end{proof}
