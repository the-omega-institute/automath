\section{Supplementary witness-cover consequences}
\label{sec:supp-witness-consequences}

\begin{corollary}[Finite Fibonacci support grammar]
\label{cor:finite-witness-cover-types}
Fix $k\ge1$. Up to relabeling of the $k$ prime coordinates of $n$, only
finitely many witness-cover support types can occur for minimal generators
$m\in\mathcal M_n$ with $\omega(n)=k$. Put
\[
\mathcal P_k:=\{(I,J):\varnothing\ne I\subseteq J\subseteq[k],\
 I=J\text{ or }|I|=1\},
\qquad
N'_k:=\#\mathcal P_k=(2^k-1)+k(2^{k-1}-1).
\]
Every minimal generator determines a unique labelled support set
\[
\mathsf T(m):=\{(E_n(\theta),T_n(\theta)):\theta\in\Theta(m)\}
 \subseteq\mathcal P_k,
\]
with $1\le|\mathsf T(m)|\le k$, satisfying coverage and irredundance:
\[
\bigcup_{(I,J)\in\mathsf T(m)}J=[k],\qquad
I\nsubseteq\bigcup_{(I',J')\in\mathsf T(m)\setminus\{(I,J)\}}J'.
\]
Consequently the number of possible labelled support types is at most
\[
\sum_{r=1}^k\binom{N'_k}{r}.
\]
The converse is not asserted: arithmetic realization additionally requires
nonempty atom families and representatives with distinct underlying primes.
Unlabelled types are $S_k$-orbits of the labelled support sets. For $k=3$, the
realization layer is the support-three slot test of
Lemma~\ref{lem:three-coordinate-nonemptiness} and
Theorem~\ref{thm:low-support-classification-package}.
\end{corollary}

\begin{proof}
By Theorem~\ref{article-thm:witness-classification}, every $m\in\mathcal M_n$ is the
product attached to a unique witness cover $\Theta(m)$, and
Corollary~\ref{article-cor:support-bound} gives $\#\Theta(m)\le k$. Irredundance makes
each $E_n(\theta)$ nonempty. After labeling $\mathcal P(n)$ by $[k]$, the
prime--ladder alternative in Lemma~\ref{thm:primitive-ladder-dichotomy} shows
that each atom contributes a pair in $\mathcal P_k$: a prime atom has $I=J$,
whereas a ladder atom with nonempty essential support has $|I|=1$.

Two distinct atoms cannot contribute the same pair $(I,J)$, since then the
essential support of either would be contained in the full support of the
other. Hence $\mathsf T(m)$ is a genuine set and is uniquely determined by
the labelled prime-power factorization. There are $2^k-1$ diagonal pairs and,
for each singleton $I=\{i\}$, there are $2^{k-1}-1$ proper supersets
$J\supsetneq I$, proving the formula for $N'_k$. Coverage and irredundance are
exactly the two displayed conditions. Counting nonempty subsets of
$\mathcal P_k$ of size at most $k$ proves the bound. Passing to unlabelled
types amounts to taking $S_k$-orbits. The atom-family nonemptiness and
coprimeness requirements explain why the converse need not hold.
\end{proof}

\begin{corollary}[Best possible support budget]
\label{cor:support-budget-best-possible}
Let $g:\NN\to\NN$ be any function such that
\[
\omega(m)\le g(\omega(n))
\]
for every $n\ge2$ and every $m\in\mathcal M_n$. Then $g(k)\ge k$ for every
$k\ge1$. Equivalently, Corollary~\ref{article-cor:support-bound} is sharp as a
uniform bound depending only on the number of prime coordinates of $n$.
Theorem~\ref{article-thm:exact-support-spectra} gives the exact layer-by-layer
equality criterion and the complete support spectrum.
\end{corollary}

\begin{proof}
Fix $k\ge1$ and choose any product $n$ of $k$ distinct odd primes; for
example, take the product of the first $k$ odd primes. Then $n$ is odd,
$n\ge3$, and $\omega(n)=k$. Corollary~\ref{article-cor:support-bound-sharp} gives an
element $m\in\mathcal M_n$ with $\omega(m)=\omega(n)=k$. Applying the assumed
uniform inequality to this pair gives $k=\omega(m)\le g(k)$. Since $k$ was
arbitrary, no uniform support budget smaller than $\omega(n)$ can hold.
\end{proof}
