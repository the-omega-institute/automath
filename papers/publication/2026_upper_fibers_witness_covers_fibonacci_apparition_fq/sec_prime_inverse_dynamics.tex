\section{Prime inverse rays and two dynamics problems}\label{sec:inverse-dynamics}
%% ====================================================================

Write $\operatorname{Fix}(\alpha)$ for the fixed points of the rank map.  For
$m\geq1$, define its \emph{fixed-point order} by
\[
 h(m):=\min\{k\geq0:\alpha^{\circ k}(m)\in
 \operatorname{Fix}(\alpha)\},
 \qquad
 \Lambda_k:=\{m\geq1:h(m)=k\}.
\]
For $x\in\operatorname{Fix}(\alpha)$, define its basin by
\[
 \Omega_x:=\{m\geq1:\alpha^{\circ j}(m)=x
 \text{ for some }j\geq0\}.
\]
These are the conventions of FitzGibbons--Javaheri--Miller--Verga
\cite{FitzGibbonsJavaheriMillerVerga2025Dynamics}.  We use their recalled
classification
\[
 \operatorname{Fix}(\alpha)=\{5^a:a\geq0\}
 \cup\{12\cdot5^a:a\geq0\}
\]
and the fact that every orbit reaches a fixed point.  In particular, the rank
map has no cycles of length greater than one.

Let $\varphi=(1+\sqrt5)/2$ and, for $t>0$, set
\[
 E_0(t)=t,\qquad E_{r+1}(t)=\varphi^{E_r(t)}\quad(r\geq0).
\]

\begin{theorem}[Prime inverse rays]\label{thm:prime-inverse-rays}
Let $y\geq7$ be an integer with $y\neq12$.  There are pairwise distinct
primes $p_1,p_2,\ldots$ such that, on putting $p_0=y$,
\[
 \alpha(p_r)=p_{r-1}\qquad(r\geq1).
\]
The primes can be chosen so that
\[
 p_r<E_r(y)\qquad(r\geq1).
\]
\end{theorem}

\begin{proof}
Theorem~\ref{thm:primitive-divisors} gives an exact-rank prime $p_1$ with
$\alpha(p_1)=y$.  Since $y\geq7$, this prime is not $2$, $3$, or $5$, and
hence $p_1\geq7$.  Inductively, if $p_r\geq7$ is chosen, then
$p_r\notin\{6,12\}$, so the same theorem supplies a prime $p_{r+1}$ with
$\alpha(p_{r+1})=p_r$.  This constructs the ray indefinitely.

No $p_r$ equals $5$: for $r=1$ this follows from $\alpha(p_1)=y\neq5$, and
for $r>1$ the equality $p_r=5$ would propagate backwards under $\alpha$ to
$p_1=5$.  If two primes in the ray were equal, that repeated prime would be
periodic under $\alpha$.  The absence of cycles of length greater than one
would make it a fixed prime, whereas the fixed-point classification shows
that the only fixed prime is $5$.  Thus the $p_r$ are pairwise distinct.

Finally, $\alpha(p_r)=p_{r-1}$ implies $p_r\mid F_{p_{r-1}}$.  Since
$F_d<\varphi^d$ for $d\geq1$, we have
\[
 p_r\leq F_{p_{r-1}}<\varphi^{p_{r-1}}.
\]
Induction on $r$ now gives $p_r<E_r(y)$.
\end{proof}

\begin{corollary}[Primes in every nontrivial fixed-point basin]
\label{cor:prime-basins}
For every fixed point $x>5$, the basin $\Omega_x$ contains infinitely many
primes.  More precisely, put
\[
 y_x=\begin{cases}7,&x=12,\\ x,&x\neq12.\end{cases}
\]
Then, for every $X>0$,
\[
 \#\{p\leq X:p\text{ prime},\ p\in\Omega_x\}
 \geq \max\bigl(\{r\geq1:E_r(y_x)\leq X\}\cup\{0\}\bigr).
\]
In particular, the lower bound is $\log^*X-O_x(1)$.
\end{corollary}

\begin{proof}
If $x\neq12$, then $x\geq7$ and Theorem~\ref{thm:prime-inverse-rays}
applied with $y=x$ gives distinct primes whose forward orbits reach $x$.
For the exceptional fixed point $12$, use
\[
 \alpha(7)=8,\qquad \alpha(8)=6,\qquad \alpha(6)=12
\]
and apply Theorem~\ref{thm:prime-inverse-rays} above $y=7$.  In either case,
the bound $p_r<E_r(y_x)$ gives the displayed counting estimate.  The
iterated-logarithm form follows by inverting the fixed-base exponential
recursion; changing the initial value changes the inverse height by
$O_x(1)$.
\end{proof}

\begin{corollary}[Primes of every positive fixed-point order]
\label{cor:prime-fixed-order}
For every $k\geq1$, the class $\Lambda_k$ contains infinitely many primes.
More precisely, for every $X>0$,
\[
 \#\{p\leq X:p\text{ prime},\ h(p)=k\}
 \geq
 \#\{b\geq1:E_k(12\cdot5^b)\leq X\}.
\]
\end{corollary}

\begin{proof}
Fix $b\geq1$ and put $p_{0,b}=12\cdot5^b$.  Repeating the construction in
Theorem~\ref{thm:prime-inverse-rays} for $k$ steps gives primes
$p_{1,b},\ldots,p_{k,b}$ such that
\[
 \alpha(p_{j,b})=p_{j-1,b}\quad(1\leq j\leq k),
 \qquad p_{k,b}<E_k(12\cdot5^b).
\]
The points $p_{1,b},\ldots,p_{k,b}$ are not fixed, so
$h(p_{k,b})=k$.  If $b\neq b'$, then $p_{k,b}\neq p_{k,b'}$, since applying
$\alpha^{\circ k}$ would otherwise give
$12\cdot5^b=12\cdot5^{b'}$.  The asserted bound and infinitude follow.
\end{proof}

FitzGibbons--Javaheri--Miller--Verga asked whether every $\Lambda_k$ with
$k\geq1$ contains infinitely many relatively prime integers, and conjectured
that every $\Omega_x$ with fixed $x>5$ does so
\cite{FitzGibbonsJavaheriMillerVerga2025Dynamics}.  Corollaries
\ref{cor:prime-fixed-order} and \ref{cor:prime-basins} answer both questions
affirmatively in the stronger form of infinitely many distinct primes.  The
input from the present paper needed for these conclusions is only the
classical exact-rank prime existence recorded in
Theorem~\ref{thm:primitive-divisors}; the witness-cover and rank-pure
machinery developed below is not used in their proofs.
