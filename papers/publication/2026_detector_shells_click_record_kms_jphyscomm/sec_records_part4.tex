It remains to treat the double root.  Put \(\theta=-\log z\) and set
\[
b=\theta z,\qquad a=1-(1+\theta)z .
\]
The coefficient \(a\) is nonnegative because
\((1+\theta)e^{-\theta}\le1\), equivalently \(\log y\le y-1\) with
\(y=e^\theta\).  Thus
\[
T_0(z,z)=
\begin{pmatrix}
z&0\\ \theta z&z
\end{pmatrix},
\qquad
T_1(z,z)=
\begin{pmatrix}
0&1-z\\0&1-(1+\theta)z
\end{pmatrix}
\]
again define a stochastic triangular labelled kernel.  Here
\[
T_0(z,z)^k=
\begin{pmatrix}
z^k&0\\ k\theta z^k&z^k
\end{pmatrix},
\]
so
\[
e_D^\top T_0(z,z)^kT_1(z,z)\mathbf1
=z^k\{(1-z)(1+\theta k)-\theta z\}.
\]
The same spectral-radius argument gives total mass one.  Hence the diagonal
formula is also a nonnegative normalized gap law.
\end{proof}

\begin{corollary}[Three inclusion probabilities are law-determining quotient coordinates]\label{cor:three_inclusion_complete_coordinates}
For the stationary visible law of the constrained D-MAP kernels
in \Cref{thm:toeplitz_image_inverse_duality}, with positive
\(\Gamma,\kappa_{\mathrm r},\Delta\tau\), \(T_1(R,R)=T_1(D,R)=0\), and
equal-rate formulas interpreted continuously, the three inclusion probabilities
\((r_0,r_1,r_2)=(\rho,\rho a,\rho u_2)\) determine the full stationary visible
law up to the exchange
\((\Gamma,\kappa_{\mathrm r})\leftrightarrow(\kappa_{\mathrm r},\Gamma)\).
Here ``law-determining'' means sufficient to reconstruct every finite-word
probability; no cardinality-minimality assertion is made.
\end{corollary}

\begin{proof}
\Cref{lem:algorithmic-complete-law-reconstruction-main} gives the explicit
reconstruction: from \((r_0,r_1,r_2)\) compute \((p+s,ps)\), recover the
unordered roots, compute \(g_k\), compute \(u_n\), and then apply the renewal
product and inclusion--exclusion.  The constrained D-MAP population satisfies
the retained denominator condition by
\Cref{thm:toeplitz_image_inverse_duality}.  The procedure therefore determines
the whole finite-word visible law on the exchange quotient, and the only
remaining ambiguity is the exchange of the two roots.
\end{proof}

\begin{lemma}[Diagonal common-rate estimand]\label{lem:diagonal-common-rate-main}
On the exchange diagonal
\[
\sigma=(\sigma_1,\sigma_2)=(2z,z^2),\qquad 0<z<1,
\]
the visible quotient law identifies the common labelled numerical rate
\[
R_{\rm diag}:=-\frac{\log z}{\Delta\tau}.
\]
Equivalently, it identifies the invariant ordered pair
\[
(R_{\rm diag},R_{\rm diag}).
\]
It does not identify a separated ordered branch, because the
separated-root map has derivative denominator \(p-s=0\).  Thus separated-root
Wald intervals are invalid on the diagonal, but reporting
\((R_{\rm diag},R_{\rm diag})\) is valid as a quotient invariant.
\end{lemma}

\begin{proof}
The polynomial with symmetric coordinate \((2z,z^2)\) is
\[
t^2-2zt+z^2=(t-z)^2,
\]
so both unordered roots equal \(z\).  Applying the fixed sampling map
\(w\mapsto-\log w/\Delta\tau\) to both entries gives the ordered pair
\((R_{\rm diag},R_{\rm diag})\).  Because the two entries are equal,
exchanging the two branch labels leaves this pair unchanged, so it is a
visible-law quotient invariant.  By contrast, the separated-root
differential
\[
dw=\frac{w\,d\sigma_1-d\sigma_2}{2w-\sigma_1}
\]
has denominator \(2z-2z=0\) at the double root, equivalently \(p-s=0\).
Therefore the ordinary separated-root delta method and its Wald intervals do
not apply on the exchange diagonal; the available first-order finite-record
limit is the projected symmetric-coordinate limit in
\Cref{cor:diagonal-projected-gaussian-exact-branch-main}.
\end{proof}

\begin{proposition}[Identifiable estimands]\label{thm:identifiable-finite-record-estimands-main}
In the fixed-setting constrained D-MAP model of
\Cref{thm:toeplitz_image_inverse_duality}, a stationary visible record
identifies only functionals of its stationary visible law.  The finite-record
CLT and covariance statements below therefore concern
\[
(r_0,r_1,r_2),\qquad
\Phi(r)=(p+s,ps),\qquad
\gamma_2=\operatorname{Cov}(A_0,A_2),
\]
and their regenerative covariance matrices.  These quantities are invariant
under the hidden-label exchange
\((p,\Gamma)\leftrightarrow(s,\kappa_{\mathrm r})\).  All later separated
labelled-rate finite-record reports are to be read through the single branch
box \Cref{conv:branch-map-labelled-rates-main}.
\end{proposition}

\begin{proof}
\Cref{thm:toeplitz_image_inverse_duality} identifies the quotient
coordinates \((p+s,ps)\) from \((r_0,r_1,r_2)\) and proves that the stationary
visible law is unchanged by the exchange
\((p,\Gamma)\leftrightarrow(s,\kappa_{\mathrm r})\).  Hence no statistic of
one visible record can distinguish which separated root is the attempt
survival factor and which is the recovery survival factor.  The listed
estimands are functions of the visible law and are therefore identifiable.
At separated roots, an ordered labelled-rate vector is obtained from the
unordered roots only after an externally supplied branch map chooses the
ordering, as in \Cref{conv:branch-map-labelled-rates-main}.  At a double root the
ordering is immaterial, and \Cref{lem:diagonal-common-rate-main} records the
identified common-rate vector.
\end{proof}

\begin{lemma}[Separated-root labelled-rate differential]\label{lem:separated-root-rate-delta-main}
Let \(r=(r_0,r_1,r_2)\) lie in the population inverse domain of
\Cref{thm:toeplitz_image_inverse_duality}, let
\[
\Phi(r)=\sigma_{\rm sym}=(\sigma_1,\sigma_2)=(p+s,ps),
\]
and suppose the two roots \(p\ne s\) both lie in \((0,1)\).  A labelled root
chart is the deterministic oracle branch map of
\Cref{conv:branch-map-labelled-rates-main}, or an estimated branch map only on
an event whose probability tends to one and on which it agrees with that
oracle as in the estimated-map reduction following
\Cref{conv:branch-map-labelled-rates-main}.  On any
neighbourhood where \(|p-s|\) is bounded away from zero and the oracle map is
fixed, this chart is \(C^1\), and its Jacobian with respect to the
symmetric coordinates is
\begin{equation}\label{eq:separated-root-jacobian-main}
J_{\mathrm{root}}
=
\begin{pmatrix}
\dfrac{p}{p-s} & -\dfrac1{p-s}\\[1.1em]
\dfrac{s}{s-p} & -\dfrac1{s-p}
\end{pmatrix}.
\end{equation}
Equivalently, for either labelled root \(z\in\{p,s\}\),
\[
dz=\frac{z\,d\sigma_1-d\sigma_2}{2z-\sigma_1}.
\]
For the fixed sampling interval \(\Delta\tau\), the labelled rate map
\[
R(p,s)=\left(-\frac{\log p}{\Delta\tau},
              -\frac{\log s}{\Delta\tau}\right)
=(\Gamma,\kappa_{\mathrm r})
\]
has Jacobian
\begin{equation}\label{eq:separated-rate-jacobian-main}
J_{\mathrm{rate}}
=
\begin{pmatrix}
-\dfrac1{\Delta\tau p} & 0\\[0.8em]
0 & -\dfrac1{\Delta\tau s}
\end{pmatrix}
J_{\mathrm{root}} .
\end{equation}
The full Jacobian from inclusion coordinates to labelled rates is therefore
\begin{equation}\label{eq:inclusion-to-rate-jacobian-main}
J_{\Gamma,\kappa}(r)
=J_{\mathrm{rate}}\,J_\Phi(r),
\qquad
J_\Phi(r)=D\Phi(r).
\end{equation}
Writing
\[
\Delta_{\mathrm{inv}}=r_1-r_0^2,\qquad E=r_2-r_0^2,\qquad
a=\frac{r_1}{r_0},\qquad \lambda=\frac{E}{\Delta_{\mathrm{inv}}},
\]
the entries of \(J_\Phi\) are explicit:
\begin{align}
\frac{\partial\lambda}{\partial r_0}
&=-\frac{2r_0(r_1-r_2)}{\Delta_{\mathrm{inv}}^2},&
\frac{\partial\lambda}{\partial r_1}
&=-\frac{E}{\Delta_{\mathrm{inv}}^2},&
\frac{\partial\lambda}{\partial r_2}
&=\frac1{\Delta_{\mathrm{inv}}}, \label{eq:lambda-gradient-main}\\
D\Phi_1
&=
\left(
\frac{a}{r_0}+\lambda_{0},\
-\frac1{r_0}+\lambda_{1},\
\lambda_{2}
\right),\nonumber\\
D\Phi_2
&=
\left(
1-\lambda+\frac{a}{r_0}+(1-r_0)\lambda_{0},\
-\frac1{r_0}+(1-r_0)\lambda_{1},\
(1-r_0)\lambda_{2}
\right), \label{eq:phi-jacobian-main}
\end{align}
where \(\lambda_i=\partial\lambda/\partial r_i\).

Thus the labelled-rate chart has covariance push-forward
\[
\Sigma_{\Gamma,\kappa}
=J_{\Gamma,\kappa}(r)\,\Sigma_r\,J_{\Gamma,\kappa}(r)^\top
\]
whenever a separate side-data hypothesis supplies a valid ordered chart and a
Gaussian limit for \(\widehat r_N\).  The visible record alone supplies only
the symmetric input \(J_\Phi(r)\) and the unordered roots; the ordered
labelled-rate stochastic statement is
\Cref{cor:side-data-labelled-rate-wald-main}.
\end{lemma}

\begin{proof}
The implicit equation for a root is
\(z^2-\sigma_1z+\sigma_2=0\).  Differentiating gives
\[
(2z-\sigma_1)\,dz-z\,d\sigma_1+d\sigma_2=0.
\]
At separated roots \(2p-\sigma_1=p-s\) and \(2s-\sigma_1=s-p\), which gives
\eqref{eq:separated-root-jacobian-main}.  Thus the labelled-root differential
itself has a pole at \(p=s\); the quotient map \(\Phi\) remains the symmetric
coordinate map there, but the labelled-root and labelled-rate charts do not
extend through the exchange diagonal.  The logarithmic rate Jacobian is
the ordinary derivative of
\((p,s)\mapsto(-\log p/\Delta\tau,-\log s/\Delta\tau)\), proving
\eqref{eq:separated-rate-jacobian-main}.  The displayed formula for
\(\lambda=E/\Delta_{\mathrm{inv}}\) is the quotient rule, and
\eqref{eq:phi-jacobian-main} follows by differentiating
\[
\Phi_1=1-a+\lambda,\qquad
\Phi_2=r_0-a+(1-r_0)\lambda .
\]
All denominators are bounded away from zero on the stated neighbourhood, so
the maps and plug-in Jacobians are continuous there.  The displayed covariance
push-forward is only the Jacobian sandwich for this externally supplied chart;
the probability limit and Wald statistic are stated separately in
\Cref{cor:side-data-labelled-rate-wald-main}.
\end{proof}

\subsection{Fixed-setting retained-record asymptotics and the exchange diagonal}

The finite-record argument has three layers: the complete-cycle protocol, the
D-MAP-specific reward identities, and the quotient/projection consequences.
Every confidence statement in this subsection is conditional on the model
contract in \Cref{ass:model-inference-contract-main} and on the
pre-specified reporting gates stated below.  The gates are withholding rules
inside that fixed experiment; they are not tests that certify the model and
they do not make post-diagnostic branch choices coverage-valid.
The model-specific content in this subsection is the reduction of the
inclusion coordinates to the adjacent-gap rewards
\[
U_j^{(0)}=1,\qquad
U_j^{(1)}=\mathbf1_{\{G_j=0\}},\qquad
U_j^{(2)}
=\mathbf1_{\{G_j=1\}}+\mathbf1_{\{G_j=0,\ G_{j+1}=0\}},
\]
followed by the quotient inverse and the exchange-diagonal projection.  The
inverse statements are delta-method chart statements for identifiable
symmetric coordinates, with labelled rates only on a retained separated-root
branch satisfying \Cref{conv:branch-map-labelled-rates-main}.

\begin{assumption}[Standing retained-record experiment]\label{ass:fixed-sampled-boundary-record-main}
All D-MAP finite-record statements in this subsection and in
\Cref{subsec:studentized-lag-two-main} are relative to the following single
record experiment, unless a statement explicitly says otherwise.  The model
and pointwise retained-record interpretation are those of
\Cref{ass:model-inference-contract-main}.  In the stationary version,
\((A_n)_{n\in\mathbb Z}\) is the stationary visible binary process generated by
the contracted model kernels.  In the finite-start version, an arbitrary hidden
initial law is run through a deterministic burn-in \(L_N\to\infty\); the
observed record is the retained post-burn-in block of length \(N\), and every
complete-cycle index, batch, projection, Wald statistic, and finite-record report is
formed only from that retained visible word.  Equal-rate formulas are
interpreted by the continuous dictionary
\eqref{eq:equal-rate-dictionary-main}.  The same-bin recovery leakage
comparator excluded by
\Cref{thm:two-state-renewal-characterization-main} is outside this retained-record
experiment; replacing the killed entries by positive one-labelled leakage would
require a different complete-cycle law, inverse map, and covariance estimator.
\end{assumption}

\paragraph{Generic regenerative tools.}
The model-independent renewal and separated-batch facts are collected in
\Cref{app:finite-record-regenerative-tools-main}.  They enter here only after
the killed-leakage renewal gap law has verified the required exponential tail
\cite{GlynnIglehart1990STS,AsmussenGlynn2007,
VatsFlegalJones2019Multivariate}.

The diagonal statistic below is not a likelihood-ratio statistic and the
feasible half-space is not estimated as a nuisance cone of a parametric
likelihood.  Its cone enters after the visible inclusion CLT is pushed through
the quotient inverse and then projected onto the local hypograph
\(\{h_2\le zh_1\}\).  The local projection step is proved below by the
explicit exchange-parabola normal equation, while Shapiro
metric-projection differentiability, Geyer constrained-estimation limits, and
Fang--Santos directional-delta inference remain the surrounding boundary and
directional-delta background
\cite{Shapiro1987MetricProjection,Geyer1994ConstrainedM,
FangSantos2019DirectionalDifferentiability}.  The classical boundary
likelihood-ratio theory of Chernoff and Self--Liang is still the relevant
nearby benchmark for cone inference and for the rank-one
\(\tfrac12\delta_0+\tfrac12\chi^2_1\) mixture, but here that mixture arises
from the retained projected-Gaussian quadratic form rather than from a
parametric cone LRT
\cite{Chernoff1954LikelihoodRatio,SelfLiang1987NonstandardLRT}.

The covariance estimator below is therefore a regenerative complete-cycle
estimator, not an MCMC theorem imported wholesale.  MCMC batch-means papers
supply useful terminology for long-run covariance estimation and retained-rank
numerical reporting, but the proof here uses the observed click cycles as the
regeneration blocks.

\begin{definition}[Cycle-index inclusion estimator]\label{def:cycle-index-inclusion-estimator-main}
For a finite observed record \(A_0,\ldots,A_{N-1}\), write the complete click
epochs in the observed window as
\[
\tau_0<\tau_1<\cdots<\tau_{K_N},\qquad K_N=\max\{k:\tau_k<N\}.
\]
If the observed window contains no click, set \(K_N=-1\) and every
all-complete-cycle index set below to the empty set.  Thus \(K_N+1\) is the
number of observed clicks, and \(K_N\) is one less than the number of click
epochs from which complete observed cycles can be formed.  Set
\(L_j=\tau_{j+1}-\tau_j\) whenever both endpoints are observed.  For
\(m=0,1,2\), put
\[
X_n^{(m)}=A_nA_{n+m},\qquad X_n^{(0)}=A_n,
\]
and define the complete-cycle uncentered reward
\[
U_j^{(m)}=\sum_{n=\tau_j}^{\tau_{j+1}-1}X_n^{(m)}.
\]
Set \(c_0=c_1=0\) and \(c_2=1\).  The admissible all-complete-cycle index set is
\[
\begin{aligned}
J_{m,N}
=\{0\le j\le K_N-c_m-1:\;&
\tau_j,\tau_{j+1}\in[0,N-1],\\
&\text{and every }n+m\text{ entering }U_j^{(m)}
\text{ lies in }[0,N-1]\}.
\end{aligned}
\]
The cycle-index estimator of the three inclusion coordinates
\[
\Rinc=(r_0,r_1,r_2)=(\PP(A_0=1),\PP(A_0=A_1=1),\PP(A_0=A_2=1))
\]
is
\[
\Rinch_N^{\mathrm{cyc}}
=\bigl(\widehat r_{0,N}^{\mathrm{cyc}},
\widehat r_{1,N}^{\mathrm{cyc}},
\widehat r_{2,N}^{\mathrm{cyc}}\bigr)^\top,\qquad
\widehat r_{m,N}^{\mathrm{cyc}}
=\frac1N\sum_{j\in J_{m,N}}U_j^{(m)} .
\]
The displayed sum is an operational estimator only after the relevant
complete-cycle set is nonempty.  Records with no click, too few complete
cycles, or no admissible terminal look-ahead cycles are not assigned the
numerical value \(0\) as an inferential estimate; the finite-record protocol
below instead stops with the message ``insufficient complete cycles; no
finite-record report.''  In asymptotic proofs only, we use the harmless auxiliary
extension in which an empty sum is interpreted as \(0\); that extension is not
part of the finite-record report.
For covariance estimation the three coordinates are evaluated on the common
complete-cycle set \(J_N=J_{0,N}\cap J_{1,N}\cap J_{2,N}\).  With plug-in
centering, the vector reward used in the separated batches is
\[
\widehat Y_j=
\bigl(U_j^{(0)}-\widehat r_{0,N}^{\mathrm{cyc}}L_j,\,
U_j^{(1)}-\widehat r_{1,N}^{\mathrm{cyc}}L_j,\,
U_j^{(2)}-\widehat r_{2,N}^{\mathrm{cyc}}L_j\bigr)^\top .
\]
\end{definition}

\paragraph{Master adjacent-gap covariance theorem.}
The covariance estimator used in Theorem~C is not assembled from several
asymptotic reductions.  It is the single adjacent-gap separated-batch object
below: the calendar stopping rule, the all-complete-cycle set \(J_N\), the
separator cycle, the empirical batch mean, the multiplier \(K_N/N\), and the
plug-in centering perturbation are all part of the same statement.
Theorems~C--E cite this result through the D-MAP reward identity and the
quotient sandwich statements below.

\begin{theorem}[Master adjacent-gap retained-cycle covariance with plug-in centering]\label{thm:main-calendar-stopped-separated-batch-plugin}
Let \((G_j)_{j\ge0}\) be i.i.d. nonnegative integer-valued gaps with
\(\EE e^{\eta G_0}<\infty\) for some \(\eta>0\), put \(L_j=G_j+1\),
\(\mu=\EE L_0\), and
\[
S_k=\sum_{j=0}^{k-1}L_j,\qquad
K_N=\max\{k:S_k<N\}.
\]
Let \(Y_j=f(G_j,G_{j+1})\in\RR^d\) be a centered adjacent-gap reward:
\[
\EE Y_j=0,\qquad \EE\|Y_j\|^4<\infty .
\]
Define
\[
T_1=\Var(Y_0)+\operatorname{Cov}(Y_0,Y_1)
     +\operatorname{Cov}(Y_1,Y_0),
\qquad
\Sigma=T_1/\mu .
\]
The complete adjacent-gap retained cycle set is
\[
J_N=\{j:0\le j\le K_N-2\},
\]
with \(J_N=\varnothing\) if \(K_N<2\).  This is exactly the set of cycle
indices whose reward \(f(G_j,G_{j+1})\) and one-cycle look-ahead are complete
before the calendar stop \(N\).

For \(b_N=\lfloor N^\beta\rfloor\), \(0<\beta<1\), put \(d_N=b_N+1\) and
\[
I_{\ell,N}=\{(\ell-1)d_N,\ldots,(\ell-1)d_N+b_N-1\}.
\]
The omitted cycle \(\ell d_N-1\) is the separator between the retained
batches \(I_{\ell,N}\) and \(I_{\ell+1,N}\).  Define
\[
B_{\ell,N}=\sum_{j\in I_{\ell,N}}Y_j
\]
for retained batches, and let the all-complete-cycle stopped batch count be
\[
M_N=\max\{M\ge0:I_{\ell,N}\subseteq J_N
        \text{ for every }1\le\ell\le M\}.
\]
Equivalently,
\[
M_N=\max\{M\ge0:(M-1)(b_N+1)+b_N\le K_N-1\},
\]
with \(M_N=0\) if the set is empty.  On \(M_N\ge2\), write
\(\bar B_N=M_N^{-1}\sum_{\ell=1}^{M_N}B_{\ell,N}\).  Then
\[
M_N=\frac{N}{\mu(b_N+1)}
    +O_{\PP}\!\left(\frac{\sqrt N}{b_N}+1\right),
\]
so \(M_N\ge2\) with probability tending to one, and, with the harmless
auxiliary zero definition on \(M_N<2\),
\[
\left\|
\frac{K_N}{N}\frac1{b_N(M_N-1)}
\sum_{\ell=1}^{M_N}(B_{\ell,N}-\bar B_N)(B_{\ell,N}-\bar B_N)^\top
-\Sigma
\right\|_{\operatorname{op}}
=O_{\PP}\left(b_N^{-1}+\sqrt{\frac{b_N}{N}}\right).
\]
Operational reporting uses the stronger gate \(M_N\ge m_N\), where
\(m_N\to\infty\) and \(m_N=o(N/b_N)\); the displayed expansion implies that
this gate also has probability tending to one.  The condition \(M_N\ge2\) is
retained only to define the auxiliary covariance formula.
Moreover, if \(\Delta_N=O_{\PP}(N^{-1/2})\) in \(\RR^d\) and the reported
plug-in rewards are
\[
\widehat Y_j=Y_j-\Delta_NL_j,\qquad
\widehat B_{\ell,N}=\sum_{j\in I_{\ell,N}}\widehat Y_j,\qquad
\overline{\widehat B}_N=M_N^{-1}\sum_{\ell=1}^{M_N}\widehat B_{\ell,N},
\]
then
\[
\max_{\ell\le M_N}\|\widehat B_{\ell,N}-B_{\ell,N}\|
=O_{\PP}(b_NN^{-1/2})
\]
and the centered quadratic perturbation obeys
\[
\left\|
\frac{K_N}{N}\frac1{b_N(M_N-1)}
\sum_{\ell=1}^{M_N}
\left\{
(\widehat B_{\ell,N}-\overline{\widehat B}_N)
(\widehat B_{\ell,N}-\overline{\widehat B}_N)^\top
-(B_{\ell,N}-\bar B_N)(B_{\ell,N}-\bar B_N)^\top
\right\}
\right\|_{\operatorname{op}}
=O_{\PP}\!\left(\sqrt{\frac{b_N}{N}}\right).
\]
Consequently the plug-in covariance estimator
\[
\widehat\Sigma_N:=
\frac{K_N}{N}\frac1{b_N(M_N-1)}
\sum_{\ell=1}^{M_N}
(\widehat B_{\ell,N}-\overline{\widehat B}_N)
(\widehat B_{\ell,N}-\overline{\widehat B}_N)^\top
\]
satisfies, again with auxiliary zero completion on \(M_N<2\),
\[
\|\widehat\Sigma_N-\Sigma\|_{\operatorname{op}}
=O_{\PP}\left(b_N^{-1}+\sqrt{\frac{b_N}{N}}\right).
\]
Thus, for \(b_N=\lfloor N^\beta\rfloor\),
\[
\|\widehat\Sigma_N-\Sigma\|_{\operatorname{op}}
=O_{\PP}\left(N^{-\beta}+N^{-(1-\beta)/2}\right).
\]
For the protocol choice \(b_N=\lfloor N^{1/3}\rfloor\), the rate is
\(O_{\PP}(N^{-1/3})\).
\end{theorem}

\begin{proof}
The exponential moment gives the renewal and maximal-cycle estimates
\[
K_N=\frac{N}{\mu}+O_{\PP}(\sqrt N),\qquad
\max_{0\le j\le K_N+2}L_j=O_{\PP}(\log N).
\]
Thus the fixed one-cycle look-ahead deletion from
\(\{0,\ldots,K_N-1\}\) to \(J_N=\{0,\ldots,K_N-2\}\) removes only one cycle,
of calendar length \(O_{\PP}(\log N)\).  For the stopped batch count above,
\[
M_N=\frac{K_N}{b_N+1}+O(1)
=\frac{N}{\mu(b_N+1)}+O_{\PP}\!\left(\frac{\sqrt N}{b_N}+1\right),
\qquad
M_N\asymp_{\PP}\frac{N}{b_N}.
\]
In particular \(M_N\ge2\) with probability tending to one.

The retained batch \(B_{\ell,N}\) depends only on
\[
G_{(\ell-1)d_N},\ldots,G_{(\ell-1)d_N+b_N}.
\]
The next retained batch starts at \(\ell d_N\).  The omitted separator index
\(\ell d_N-1\) is not batched, but it supplies the last gap
\(G_{\ell d_N-1}\) needed by the final reward in the previous batch; hence the
previous batch uses gaps only through \(G_{\ell d_N-1}\), while the next batch
uses gaps starting at \(G_{\ell d_N}\).  These gap sets are disjoint for
distinct \(\ell\), so the retained batch sums are independent.  Stationarity
and \(1\)-dependence give
\[
b_N^{-1}\Var(B_{\ell,N})=T_1+O(b_N^{-1}),
\]
where the \(O(b_N^{-1})\) term is the finite number of missing boundary
covariance layers inside a block.  Since \((Y_j)\) is \(1\)-dependent and
\(\EE\|Y_j\|^4<\infty\), the usual fourth-moment expansion for
\(1\)-dependent sums gives
\[
\EE\|B_{\ell,N}\|^4\le Cb_N^2 .
\]

Put
\[
X_{\ell,N}
=b_N^{-1}\{B_{\ell,N}B_{\ell,N}^\top
-\EE(B_{\ell,N}B_{\ell,N}^\top)\}.
\]
The preceding fourth-moment estimate gives
\(\sup_{\ell,N}\EE\|X_{\ell,N}\|_{\operatorname F}^2<\infty\).  Applying
Kolmogorov's maximal inequality entrywise to all deterministic prefixes and
then taking a union bound over the matrix entries gives, for a constant
\(C_d<\infty\),
\[
\PP\!\left[
\max_{m\le C_1N/b_N}
\left\|\sum_{\ell=1}^mX_{\ell,N}\right\|_F
>\eta N/b_N
\right]
\le \frac{C_d b_N}{\eta^2N}.
\]
Together with
\(\PP\{c_1N/b_N\le M_N\le C_1N/b_N\}\to1\), the same inequality at the
natural \(\sqrt{N/b_N}\) scale yields
\[
\left\|
\frac1{M_N}\sum_{\ell=1}^{M_N}X_{\ell,N}
\right\|_{\operatorname{op}}
=O_{\PP}\left(\sqrt{\frac{b_N}{N}}\right).
\]
The empirical batch mean contributes only
\[
b_N^{-1}\bar B_N\bar B_N^\top
=O_{\PP}(M_N^{-1})
=O_{\PP}\left(\frac{b_N}{N}\right),
\]
and \(K_N/N=\mu^{-1}+O_{\PP}(N^{-1/2})\).  Combining these displays proves the
oracle covariance bound.

For the plug-in rewards,
\[
\widehat B_{\ell,N}-B_{\ell,N}
=-\Delta_N\sum_{j\in I_{\ell,N}}L_j .
\]
A Chernoff bound for \(\sum_{j=1}^{b_N}L_j\), followed by a union bound over
the deterministic prefix, gives constants \(c_L,C_0>0\) such that
\[
\PP\!\left(
\max_{\ell\le C_1N/b_N}\sum_{j\in I_{\ell,N}}L_j>C_0b_N
\right)
\le \frac{C_1N}{b_N}e^{-c_Lb_N}=o(1).
\]
Since \(M_N\le C_1N/b_N\) with probability tending to one, this proves
\[
\max_{\ell\le M_N}b_N^{-1}\sum_{j\in I_{\ell,N}}L_j=O_{\PP}(1),
\]
so the displayed block perturbation follows from
\(\Delta_N=O_{\PP}(N^{-1/2})\).  Expanding the centered quadratic covariance in
\(\widehat B_{\ell,N}=B_{\ell,N}+(\widehat B_{\ell,N}-B_{\ell,N})\), the cross
term is \(O_{\PP}(\sqrt{b_N/N})\) by Cauchy's inequality and the oracle
covariance bound, while the pure plug-in quadratic term is
\(O_{\PP}(b_N/N)\).  Since \(b_N/N\le\sqrt{b_N/N}\) eventually, the plug-in
covariance has the same rate as the oracle covariance.
\end{proof}

\paragraph{Inclusion-coordinate covariance estimation.}
Algorithm~1 reports \(\widehat\Sigma_{r,N}\) only on the retained reporting
event; it does not use a zero matrix as an operational replacement.  Whenever
the main text writes an \(O_{\PP}\) or \(o_{\PP}\) statement for a covariance
matrix as a random element on the full probability space, the object is instead
the auxiliary completed matrix
\[
\widehat\Sigma_{r,N}^{\rm tot}
:=\mathbf 1_{\mathcal E_N}\widehat\Sigma_{r,N}
+\mathbf 1_{\mathcal E_N^c}0_{3\times3}.
\]
This completion is only proof notation; on \(\mathcal E_N^c\) the finite-record
procedure withholds the covariance report and all finite-record reports.  The
master adjacent-gap theorem
\Cref{thm:main-calendar-stopped-separated-batch-plugin} is invoked only after
the following D-MAP reward identity has reduced the record coordinates to the
specific reward \(Y_j=f(G_j,G_{j+1})\).

\begin{proposition}[Exact inclusion-coordinate separated-batch covariance rate]\label{prop:exact-inclusion-separated-batch-covariance-main}
Work under \Cref{ass:fixed-sampled-boundary-record-main} and the deterministic
threshold protocol \ref{fr:protocol-main} at a population point in the stable
quotient chart \(r_0>0\), \(\Delta_{\mathrm{inv}}=r_1-r_0^2\ne0\).  Let
\[
U_j=
\bigl(1,\mathbf1_{\{G_j=0\}},
\mathbf1_{\{G_j=1\}}+\mathbf1_{\{G_j=0,G_{j+1}=0\}}\bigr)^\top,
\qquad
L_j=G_j+1,
\]
and put \(Y_j=U_j-\Rinc L_j=f(G_j,G_{j+1})\), where
\(\Rinc=(r_0,r_1,r_2)^\top\) and \(r_m=\PP(A_0=A_m=1)\).  Then
\(\EE Y_j=0\), \(\EE\|Y_j\|^4<\infty\), and \((Y_j)\) is \(1\)-dependent.
Let
\[
T_1^{(r)}
=\operatorname{Var}(Y_0)+\operatorname{Cov}(Y_0,Y_1)
 +\operatorname{Cov}(Y_1,Y_0),
\qquad
\Sigma_r=T_1^{(r)}/\mu,\qquad \mu=\EE L_0 .
\]
Let \(\widehat\Sigma_{r,N}\) be Algorithm~1's all-complete-cycle separated
batch covariance, formed from the plug-in rewards
\(\widehat Y_j=Y_j-(\Rinch_N^{\mathrm{cyc}}-\Rinc)L_j\) with
\(b_N=\lfloor N^\beta\rfloor\), \(0<\beta<1\).  Explicitly, with
\[
S_k=\sum_{j=0}^{k-1}L_j,\qquad K_N=\max\{k:S_k<N\},\qquad
d_N=b_N+1,
\]
the separated batch blocks are
\[
I_{\ell,N}=\{(\ell-1)d_N,\ldots,(\ell-1)d_N+b_N-1\},
\]
and the all-complete-cycle stopped batch count is
\[
M_N=\max\{M\ge0:(M-1)d_N+b_N\le K_N-1\},
\]
with \(M_N=0\) when the set is empty.  This is the same stopped count as
Algorithm~1's condition \(I_{\ell,N}\subseteq J_N\): the final batched reward
uses \(G_{j+1}\), so one terminal look-ahead complete cycle is required, and
the separator cycle between adjacent blocks is omitted.  With the auxiliary
completion
\[
\widehat\Sigma_{r,N}^{\rm tot}
:=\mathbf 1_{\mathcal E_N}\widehat\Sigma_{r,N}
+\mathbf 1_{\mathcal E_N^c}0_{3\times3},
\]
one has
\[
\|\widehat\Sigma_{r,N}^{\rm tot}-\Sigma_r\|_{\operatorname{op}}
=
O_{\PP}\left(b_N^{-1}+\sqrt{\frac{b_N}{N}}\right).
\]
The displayed rate is for Algorithm~1's exact reported covariance object:
the random calendar-stopped prefix, fixed endpoint/look-ahead deletion,
empirical batch mean, multiplier \(K_N/N\), and plug-in centering by
\(\Rinch_N^{\mathrm{cyc}}\) are all included before the auxiliary
completion is introduced.
In particular, for the paper's fixed protocol
\[
b_N=\lfloor N^{1/3}\rfloor,
\]
the plug-in covariance estimator itself satisfies, on the all-complete-cycle
reported event and with
\(\overline{\widehat B}_N=M_N^{-1}\sum_{\ell=1}^{M_N}\widehat B_{\ell,N}\),
\[
\left\|
\frac{K_N}{N}\frac{1}{b_N(M_N-1)}
\sum_{\ell=1}^{M_N}
(\widehat B_{\ell,N}-\overline{\widehat B}_N)
(\widehat B_{\ell,N}-\overline{\widehat B}_N)^\top
-\Sigma_r
\right\|_{\operatorname{op}}
=O_{\PP}(N^{-1/3}).
