Bickel--Kwon identify the interaction tangent for a stationary Markov model
with known marginal as the doubly centred space and obtain its orthogonal
projection away from additive marginal scores
\cite{BickelKwon2001Semiparametric}.  That tangent and projection are inputs
here.  The result below intersects their space with the sampled-counter
three-inclusion constraint and supplies model-specific positive paths,
calendar-time scaling, and exchange-local calibration.

\begin{theorem}[Atom-constrained section of the fixed-marginal Markov tangent and pointwise omnibus score]
\label{thm:complete-markov-palm-tangent-omnibus-main}
Fix $0<\alpha<1$, an interior exchange point $\theta_0=(x_0,x_0)$, and let
$g=(g_k)_{k\geq0}$ be its Palm zero-gap law.  Thus $g_k>0$,
$\mu=\sum_{k\geq0}(k+1)g_k<\infty$, and, for this sampled-counter law,
$g$ has an exponential tail.  Put
\[
 \mathsf H=L^2(g\otimes g),\qquad
 \mathsf H_{\rm int}=\{q\in\mathsf H:
 \EE(q\mid G_0)=\EE(q\mid G_1)=0\},
\]
where $G_0,G_1$ are independent with law $g$, and define
\[
 \mathsf T=\{q\in\mathsf H_{\rm int}:q(0,0)=0\}.
 \tag{H}
\]
Consider all quadratic-mean differentiable Markov-gap paths
\[
 Q_t(\ell\mid k)=g_\ell w_t(k,\ell),\qquad w_0=1,
 \qquad \left\|\frac{2(\sqrt{w_t}-1)}t-q\right\|_{\mathsf H}
 \longrightarrow0,
\tag{I}
\]
whose rows sum to one, for which $gQ_t=g$, and which preserve
$(r_0,r_1,r_2)$ to first order.  Then the following statements hold.

\begin{enumerate}[label=\textup{(\roman*)}]
\item Relative to the declared Markov-gap path class, the closed tangent
space is exactly $\mathsf T$; in
particular it is infinite dimensional.  If
\[
 u_0(k)=\mathbf1_{\{k=0\}}-g_0,\qquad
 v_0(k,\ell)=u_0(k)u_0(\ell),
\]
then
\[
 \mathsf T=\mathsf H_{\rm int}\ominus\operatorname{span}\{v_0\}.
 \tag{J}
\]
Every $q\in\mathsf T$ is realized by a two-sided path in \textup{(I)} whose
kernels satisfy a Doeblin minorization by $(1-o(1))g$ and preserve $g$ and
all three inclusion coordinates exactly.  A nonzero such path is nonrenewal
for all sufficiently small nonzero $t$.

\item Every adjacent-pair score generated by a quadratic-mean differentiable
renewal marginal path belongs to the closed span of
$s(G_0)+s(G_1)$, with $s\in L^2_0(g)$.  Hence the sampled-counter renewal
nuisance space is orthogonal to $\mathsf T$.  On the ambient space
$\mathsf H$, the efficient information operator for the Markov interaction
is
\[
 \mathcal I_{\rm eff}=\mu^{-1}\Pi_{\mathsf T},
 \tag{K}
\]
where
\[
 \Pi_{\mathsf T}f=f-\EE(f\mid G_0)-\EE(f\mid G_1)+\EE f
 -\frac{\langle f,v_0\rangle_{\mathsf H}}
 {\|v_0\|_{\mathsf H}^2}v_0.
 \tag{L}
\]
In particular,
\[
 \ker\mathcal I_{\rm eff}
 =\mathsf H_{\rm int}^{\perp}\oplus\operatorname{span}\{v_0\},
 \qquad
\left.\mathcal I_{\rm eff}\right|_{\mathsf T}
 =\mu^{-1}I_{\mathsf T}.
 \tag{M}
\]
The ambient renewal-marginal nuisance and its two-dimensional
sampled-counter parametric subspace are distinct model classes; both are
additive and therefore orthogonal to the interaction space.

\item Delete the two incomplete endpoint cycles from a stationary record and
write its complete gaps as $G_1,\ldots,G_{K_N}$, where
$K_N/N\to_P\mu^{-1}$.  For $q\in\mathsf T$, let
\[
 \Delta_N(q)=N^{-1/2}\sum_{i=1}^{K_N-1}q(G_i,G_{i+1}).
\]
For each fixed finite collection $q_1,\ldots,q_d\in\mathsf T$, the null
joint limit is centred Gaussian with covariance
\[
 \lim_{N\to\infty}\operatorname{Cov}
 \{\Delta_N(q_a),\Delta_N(q_b)\}
 =\mu^{-1}\langle q_a,q_b\rangle_{\mathsf H}.
 \tag{N}
\]
Along the realizing paths constructed in part~\textup{(i)}, or any other
such path whose retained-record endpoint log likelihood is $o_{P_0}(1)$,
the score $q$, evaluated at $t_N=N^{-1/2}$, satisfies
\[
 \log\frac{dP_{N,q}}{dP_{N,0}}
 =\Delta_N(q)-\frac1{2\mu}\|q\|_{\mathsf H}^2+o_{P_0}(1).
 \tag{O}
\]
Thus the limit is the isonormal process on $\mathsf T$ with covariance
\textup{(N)}, rather than a $\mathsf T$-valued normal random element with
identity covariance.  No uniform-LAN assertion over arbitrary norm-compact
subsets of $\mathsf T$ is made.

\item Put
\[
 e_0(k)=\frac{\mathbf1_{\{k=0\}}-g_0}{\sqrt{g_0(1-g_0)}}
\]
and choose a bounded orthonormal basis $e_1,e_2,\ldots$ of
$L^2_0(g)\cap\{e_0\}^{\perp}$.  Enumerate
\[
 \{e_a\otimes e_b:a,b\geq0,\ (a,b)\neq(0,0)\}
 =\{\phi_1,\phi_2,\ldots\};
\]
this is an orthonormal basis of $\mathsf T$.  Fix strictly positive weights
$\lambda_j$ satisfying
\[
 \sum_{j\geq1}\lambda_j<\infty,
 \qquad
 \sum_{j\geq1}\lambda_j\|\phi_j\|_\infty^2<\infty,
 \tag{P}
\]
and any deterministic $J_N\uparrow\infty$.  Put
\[
 \widehat\mu_N=
 \begin{cases}
 K_N^{-1}\sum_{i=1}^{K_N}(G_i+1),&K_N\geq1,\\
 1,&K_N=0,
 \end{cases}
\]
and define, when $K_N\geq2$,
\[
 Z_{j,N}=\sqrt{\frac{\widehat\mu_N}{N}}
 \sum_{i=1}^{K_N-1}\phi_j(G_i,G_{i+1}),\qquad
 Q_N=\sum_{j=1}^{J_N}\lambda_jZ_{j,N}^2,
 \tag{Q}
\]
and put $Q_N=0$ otherwise.  If $\zeta_j$ are independent standard normals,
then
\[
 Q_N\Rightarrow Q_0:=\sum_{j\geq1}\lambda_j\zeta_j^2.
 \tag{R}
\]
Let $c_{\alpha,\lambda}$ be the $(1-\alpha)$ quantile of $Q_0$.  Under the
local direction $q=\sum_jq_j\phi_j\in\mathsf T$ along one of the paths
admitted in \textup{(O)},
\[
 Q_N\Rightarrow
 \sum_{j\geq1}\lambda_j
 \left(\zeta_j+\frac{q_j}{\sqrt\mu}\right)^2.
 \tag{S}
\]
Consequently the test
$\psi_N=\mathbf1_{\{Q_N>c_{\alpha,\lambda}\}}$ has limiting power strictly
larger than $\alpha$ for every fixed $q\in\mathsf T\setminus\{0\}$.

\item The null calibration in \textup{(R)} is stable along the exchange
boundary in the following explicit local sense.  Let
$\sigma=(p+s,ps)$ be the symmetric sampled-counter coordinate and, for fixed
$C<\infty$, let $\sigma_N$ range over null sequences satisfying
\[
 \|\sqrt N(\sigma_N-\sigma_0)\|\leq C.
 \tag{T}
\]
Use in \textup{(Q)} the basis fixed at $\sigma_0$ and the displayed
$\widehat\mu_N$.  Then
\[
 \sup_{\sigma_N:\,\textup{(T)}}
 \left|\PP_{\sigma_N}(Q_N>c_{\alpha,\lambda})-\alpha\right|
 \longrightarrow0.
 \tag{U}
\]
This class includes exchange-root splittings
$N^{1/4}|p_N-s_N|=O(1)$ provided their symmetric coordinates obey
\textup{(T)}.
\end{enumerate}

The algebraic assertions in parts~\textup{(i)}--\textup{(ii)} use only full
support of $g$, finite $\mu$, and square-integrability of each local score.
For the explicit realizing paths, the uniform density-ratio bound controls
the endpoint terms in parts~\textup{(iii)}--\textup{(iv)}.  The
sampled-counter exponential tail and analytic dependence on $\sigma$ are used
in part~\textup{(v)} for its uniform triangular-array statement.  No claim
that finite mean alone controls the endpoints of every quadratic-mean
differentiable stationary-record path is made.

The double-centring and additive-orthogonality assertions are the
Bickel--Kwon known-marginal Markov result.  Generic basis tests of serial independence, Hilbert-space central limit
theorems, and Gaussian-shift arguments are established methods
\cite{SkaugTjostheim1993SerialIndependence,
DedeckerMerlevede2003HilbertCLT,vanDerVaart1998AsymptoticStatistics}.
The additional assertions are restricted to the sampled-counter
Markov--Palm interaction: the atom constraint in \textup{(H)}, constructive
positivity-compatible paths, calendar information, and the exchange-boundary
calibration of \textup{(Q)}.
\end{theorem}

\begin{proof}
The row and stationary-marginal identities for \textup{(I)} are
\[
 \sum_\ell g_\ell w_t(k,\ell)=1,
 \qquad
 \sum_k g_kw_t(k,\ell)=1.
\]
Differentiating in quadratic mean gives the two conditional-mean constraints
in \textup{(H)}.  Fixed $g$ fixes $r_0=\mu^{-1}$ and
$r_1=\mu^{-1}g_0$, whereas a stationary Markov gap law satisfies
\[
 r_2=\mu^{-1}\{g_1+g_0Q_t(0\mid0)\}.
\]
Since point evaluation at $(0,0)$ is continuous in $\mathsf H$,
$Q_t(0\mid0)=g_0\{1+tq(0,0)+o(t)\}$; hence preservation of $r_2$ to first
order is equivalent to $q(0,0)=0$.

For $q\in\mathsf H_{\rm int}$, direct expansion and double centring give
\[
 \langle q,v_0\rangle_{\mathsf H}=g_0^2q(0,0),
\]
which proves \textup{(J)} and \textup{(L)}.  To prove realizability, truncate
$q$ at levels $m$ and write
$q_m=\Pi_{\mathsf T}\{(-m)\vee(q\wedge m)\}$.  Then $q_m$ is bounded and
$q_m\to q$ in $\mathsf H$.  Choose $m(t)\uparrow\infty$ so slowly that
$|t|\|q_{m(t)}\|_\infty\to0$ and set
$w_t=1+tq_{m(t)}$.  Its weighted row and column sums are exactly one and
$w_t(0,0)=1$; hence all three inclusion coordinates are exact.  Moreover,
$Q_t(\ell\mid k)\geq\{1-o(1)\}g_\ell$, proving the Doeblin assertion.  If
a nonzero score generated a renewal kernel, it would have the form
$a(\ell)$; the two centring identities would then force $a=0$.
The inequality
\[
 \left|2\{\sqrt{1+x}-1\}-x\right|\leq Cx^2,
 \qquad |x|\leq\tfrac12,
\]
together with $|t|\|q_{m(t)}\|_\infty\to0$ proves the quadratic-mean
differentiability in \textup{(I)}.

The Hoeffding row--column decomposition of $L^2(g\otimes g)$ is
\[
 \mathbb R\oplus L^2_0(G_0)\oplus L^2_0(G_1)
 \oplus\mathsf H_{\rm int}.
\]
A renewal marginal perturbation contributes only its one-gap score at the two
ends of an adjacent pair.  It is therefore orthogonal to $\mathsf T$.
Under the null, successive interaction rewards have zero lag-one covariance:
\[
 \EE\{q_a(G_0,G_1)q_b(G_1,G_2)\}=0.
\]
The number of complete cycles per calendar bin tends to $\mu^{-1}$, so the
information quadratic form is $\mu^{-1}\|q\|_{\mathsf H}^2$.  Projection
onto the admissible interaction scores proves \textup{(K)}--\textup{(M)}.

For each fixed finite collection of scores, the one-dependent central limit
theorem and random-index replacement prove \textup{(N)}.  For the realizing
paths in part~\textup{(i)}, the bound
$|t|\|q_{m(t)}\|_\infty\to0$ also makes the endpoint log likelihood
negligible.  Quadratic-mean differentiability then gives \textup{(O)}; the
same conclusion holds under the alternative endpoint hypothesis stated in
part~\textup{(iii)}.  This proves the finite-dimensional local experiment.
Identity covariance on the
infinite-dimensional space is not trace class, so the limit can only be
interpreted cylindrically as the displayed isonormal process.

The normalized $e_0$ is proportional to $u_0$.  Tensoring an orthonormal
basis of $L^2_0(g)$ gives an orthonormal basis of
$\mathsf H_{\rm int}$, and deleting $e_0\otimes e_0$ imposes \textup{(J)}.
Condition \textup{(P)} makes
\[
 \bigl(\sqrt{\lambda_j}\phi_j(G_i,G_{i+1})\bigr)_{j\geq1}
\]
a square-integrable, one-dependent $\ell^2$-valued random variable.  Its
lag-one covariance vanishes by double centring.  The Hilbert-space central
limit theorem, random-index replacement, and truncation in $J_N$ therefore
give \textup{(R)}.  Joint convergence with the likelihood score and the third
lemma give \textup{(S)}.  If $q\neq0$, some $q_j\neq0$; for every positive
threshold the square of a normal with nonzero mean strictly stochastically
dominates a central normal square.  Conditioning on the remaining independent
coordinates proves strict power because every $\lambda_j$ is positive.

It remains to prove the uniform calibration in \textup{(U)}.  We give the
truncation argument because covariance convergence alone is not a
triangular-array Hilbert-space CLT.  Write
\[
 g_N=g_{\sigma_N},\qquad \delta_N=g_N-g .
\]
Analyticity in the symmetric coordinate and the locally uniform exponential
tail give, uniformly under \textup{(T)},
\[
 \|\delta_N\|_1=O(N^{-1/2}),\qquad
 \mu_N-\mu=O(N^{-1/2}),
\tag{U1}
\]
together with a common exponential moment for the gap laws \(g_N\).
Introduce the weighted feature
\[
 \Psi(k,\ell)=
 \bigl(\sqrt{\lambda_j}\phi_j(k,\ell)\bigr)_{j\ge1}\in\ell^2,
\qquad
 B_\lambda^2=\sum_{j\ge1}\lambda_j\|\phi_j\|_\infty^2<\infty .
\tag{U2}
\]
Thus \(\|\Psi(k,\ell)\|_{\ell^2}\le B_\lambda\) for every pair.  Since every
\(\phi_j\) is double centred under \(g\),
\[
\begin{split}
 m_N:=\EE_{g_N\otimes g_N}\Psi
 &=\sum_{j\ge1}\sqrt{\lambda_j}
 \langle\phi_j,\delta_N\otimes\delta_N\rangle\,\mathbf e_j,\\
 \|m_N\|_{\ell^2}
 &\le B_\lambda\|\delta_N\|_1^2=O(N^{-1}).
\end{split}
\tag{U3}
\]
This displays, rather than assumes, the cancellation of the two first-order
marginal terms.

Fix \(J\).  The first \(J\) coordinates of
\(\Psi(G_i,G_{i+1})-m_N\) form a uniformly bounded one-dependent triangular
array.  Its lag-zero covariance tends to
\(\operatorname{diag}(\lambda_1,\ldots,\lambda_J)\), while its two lag-one
covariances tend to zero: at \(g\) they vanish exactly by double centring, and
\textup{(U1)} controls the perturbation.  The fixed-dimensional
one-dependent triangular-array CLT therefore gives
\[
 (Z_{1,N},\ldots,Z_{J,N})
 \Longrightarrow(\zeta_1,\ldots,\zeta_J)
\tag{U4}
\]
uniformly under \textup{(T)}.  For completeness, uniformity follows by the
subsequence criterion: from every sequence satisfying \textup{(T)}, extract
a convergent subsequence of
\(\sqrt N(\sigma_N-\sigma_0)\); boundedness gives Lindeberg, \textup{(U1)}
gives the covariance limit, and the usual even--odd blocking proves the
one-dependent CLT.  The uniform renewal law of large numbers replaces the
deterministic number of pairs by \(K_N-1\), and
\(\widehat\mu_N-\mu_N=o_P(1)\).  These replacements are uniform because the
common exponential moment controls both endpoint cycles and the renewal-count
deviation.  The mean contribution in \textup{(U3)} is
\(O(\sqrt N\,\|m_N\|_{\ell^2})=o(1)\).

The passage from fixed \(J\) to the full statistic is also uniform.  Put
\[
 L_J=\sum_{j>J}\lambda_j\|\phi_j\|_\infty^2\longrightarrow0 .
\]
The even--odd maximal inequality for the centred one-dependent tail feature,
followed by the renewal-count bound \(K_N\le N\), gives
\[
 \sup_{\sigma_N:\,\textup{(T)}}
 \EE_{\sigma_N}\!\left[
 \sum_{j>J}\lambda_jZ_{j,N}^2;\,
 K_N\ge2\right]
 \le C L_J+o(1).
\tag{U5}
\]
Here the \(o(1)\) contains the squared mean from \textup{(U3)} and the event
on which \(\widehat\mu_N\) is outside a fixed compact interval.  The latter
event is exponentially small by the common exponential renewal bound; on its
complement \(\widehat\mu_N\) is uniformly bounded.  This proves
\textup{(U5)} without treating the random number of overlapping pairs as a
conditionally binomial count.

For the limit variable,
\[
 \EE\sum_{j>J}\lambda_j\zeta_j^2
 =\sum_{j>J}\lambda_j\longrightarrow0.
\tag{U6}
\]
Equations \textup{(U4)}--\textup{(U6)} and Markov's inequality give uniform
bounded-Lipschitz convergence of \(Q_N\) to
\(Q_0=\sum_j\lambda_j\zeta_j^2\).  The law of \(Q_0\) is continuous because
\(\lambda_1>0\), so evaluation at \(c_{\alpha,\lambda}\) proves
\textup{(U)}.  Finally,
\[
 (p_N+s_N,p_Ns_N)
 =(2z,z^2-h^2N^{-1/2}),
\]
so the stated \(N^{-1/4}\) root-splitting sequences satisfy \textup{(T)}.
\end{proof}
