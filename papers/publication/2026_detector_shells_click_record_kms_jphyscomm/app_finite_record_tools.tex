\section{Finite-Record Regenerative Tools}\label{app:finite-record-regenerative-tools-main}

This appendix collects the model-independent renewal and separated-batch
arguments used in \Cref{sec:records}.  The main retained-record asymptotic theorems use
these results only after the constrained D-MAP law has been reduced to the
explicit adjacent-gap rewards displayed in the main text.

\begin{theorem}[Separated-batch covariance for fixed-range gap rewards]\label{thm:regenerative-separated-batch-covariance-main}
Fix an integer \(m\ge0\) and a dimension \(d\).  Let
\((G_j)_{j\in\mathbb Z}\) be i.i.d. with
\(\EE e^{\eta G_0}<\infty\) for some \(\eta>0\), and let
\[
Y_j=f(G_j,\ldots,G_{j+m})\in\RR^d,\qquad
\EE Y_j=0,\qquad \EE\|Y_j\|^4<\infty .
\]
Set
\[
T_m=\operatorname{Var}(Y_0)+
\sum_{h=1}^m\{\operatorname{Cov}(Y_0,Y_h)+\operatorname{Cov}(Y_h,Y_0)\}.
\]
Let \(b_N\to\infty\), \(b_N/N\to0\),
\(M_N=\lfloor N/(b_N+m)\rfloor\), and define the separated batches
\[
I_{\ell,N}=\{(\ell-1)(b_N+m),\ldots,
(\ell-1)(b_N+m)+b_N-1\},\qquad
B_{\ell,N}=\sum_{j\in I_{\ell,N}}Y_j ,
\]
for \(1\le\ell\le M_N\).  On \(M_N\ge2\), put
\(\bar B_N=M_N^{-1}\sum_{\ell=1}^{M_N}B_{\ell,N}\).  Then
\begin{equation}\label{eq:regenerative-separated-batch-covariance-main}
\left\|
\frac{1}{b_N(M_N-1)}
\sum_{\ell=1}^{M_N}(B_{\ell,N}-\bar B_N)
(B_{\ell,N}-\bar B_N)^\top
-T_m
\right\|_{\operatorname{op}}
=O_{\PP}\left(b_N^{-1}+\sqrt{\frac{b_N}{N}}\right).
\end{equation}
\end{theorem}

\begin{proof}
The sequence \((Y_j)\) is \(m\)-dependent: if two index sets are separated by
more than \(m\), then the corresponding \(Y\)'s are functions of disjoint
families of i.i.d. gaps.  The retained batch \(B_{\ell,N}\) depends only on
\[
G_{(\ell-1)(b_N+m)},\ldots,
G_{(\ell-1)(b_N+m)+b_N-1+m}.
\]
The next retained batch starts with
\(G_{\ell(b_N+m)}\).  Since the separator has length \(m\), these gap sets are
disjoint for different \(\ell\), and therefore the retained sums
\((B_{\ell,N})_{\ell\le M_N}\) are independent and identically distributed.

Write \(\Gamma_h=\operatorname{Cov}(Y_0,Y_h)\).  Stationarity and
\(m\)-dependence give
\[
\operatorname{Var}(B_{\ell,N})
=b_N\Gamma_0+
\sum_{h=1}^{m}(b_N-h)(\Gamma_h+\Gamma_h^\top).
\]
Hence
\[
\frac{1}{b_N}\operatorname{Var}(B_{\ell,N})
=T_m+R_{b_N},\qquad
R_{b_N}=-\frac1{b_N}\sum_{h=1}^{m}h(\Gamma_h+\Gamma_h^\top).
\]
The matrices \(\Gamma_h\) are fixed finite matrices because
\(\EE\|Y_0\|^2<\infty\).  Thus
\begin{equation}\label{eq:regenerative-boundary-covariance-main}
\|R_{b_N}\|_{\operatorname{op}}\le Cb_N^{-1}.
\end{equation}
Equivalently, in a block of length \(b_N\) the infinite \(m\)-dependent
covariance sum misses only the \(h\) boundary pairs at each lag
\(1\le h\le m\), an \(O(1)\) number of covariance-pair layers after division
by \(b_N\).

It remains to bound the stochastic part.  The standard fourth-moment bound
for fixed-range-dependent sums gives
\begin{equation}\label{eq:regenerative-batch-fourth-main}
\EE\|B_{\ell,N}\|^4\le Cb_N^2 .
\end{equation}
Indeed,
\(\|B_{\ell,N}\|^4\le d^2\sum_{a,c=1}^d
(\sum_{j\in I_{\ell,N}}Y_{j,a})^2(\sum_{j\in I_{\ell,N}}Y_{j,c})^2\);
after expanding, any nonzero centered contribution can involve only index
quadruples whose dependency graph has connected components of diameter at
most \(m\).  There are \(O(b_N^2)\) such quadruples, and H\"older's inequality
bounds each expectation by a constant depending only on
\(\EE\|Y_0\|^4\), \(m\), and \(d\).

Define
\[
V_N:=b_N^{-1}\operatorname{Var}(B_{1,N}),\qquad
W_{\ell,N}:=
b_N^{-1}\{B_{\ell,N}B_{\ell,N}^\top-\EE(B_{\ell,N}B_{\ell,N}^\top)\}.
\]
The \(W_{\ell,N}\)'s are independent, mean zero, and
\[
\EE\|W_{\ell,N}\|_{\operatorname{F}}^2
\le b_N^{-2}\EE\|B_{\ell,N}B_{\ell,N}^\top\|_{\operatorname{F}}^2
= b_N^{-2}\EE\|B_{\ell,N}\|^4
\le C
\]
by \eqref{eq:regenerative-batch-fourth-main}.  Therefore
\begin{equation}\label{eq:regenerative-uncentered-fluctuation-main}
\EE\left\|
\frac{1}{b_NM_N}\sum_{\ell=1}^{M_N}
\{B_{\ell,N}B_{\ell,N}^\top-\EE(B_{\ell,N}B_{\ell,N}^\top)\}
\right\|_{\operatorname{op}}^2
\le
\EE\left\|
\frac{1}{M_N}\sum_{\ell=1}^{M_N}W_{\ell,N}
\right\|_{\operatorname{F}}^2
\le \frac{C}{M_N}.
\end{equation}
Consequently the uncentered empirical second moment satisfies
\[
\frac{1}{b_NM_N}\sum_{\ell=1}^{M_N}B_{\ell,N}B_{\ell,N}^\top
=V_N+O_{\PP}(M_N^{-1/2}).
\]
Since \(M_N=\lfloor N/(b_N+m)\rfloor\) and \(b_N/N\to0\),
\(M_N^{-1/2}=O(\sqrt{b_N/N})\).

The subtraction of the empirical batch mean changes only a smaller-order
term.  Algebra gives
\[
\frac{1}{b_N(M_N-1)}\sum_{\ell=1}^{M_N}
(B_{\ell,N}-\bar B_N)(B_{\ell,N}-\bar B_N)^\top
=
\frac{M_N}{M_N-1}
\left\{
\frac1{b_NM_N}\sum_{\ell=1}^{M_N}B_{\ell,N}B_{\ell,N}^\top
-\frac1{b_N}\bar B_N\bar B_N^\top
\right\}.
\]
Moreover
\[
\EE\left\|\frac1{\sqrt{b_N}}\bar B_N\right\|^2
=\frac1{b_NM_N^2}\sum_{\ell=1}^{M_N}\operatorname{tr}\Var(B_{\ell,N})
\le \frac{C}{M_N},
\]
so \(b_N^{-1}\bar B_N\bar B_N^\top=O_{\PP}(M_N^{-1})\).  Also
\((M_N/(M_N-1))-1=O(M_N^{-1})\) and \(V_N=O(1)\).  Combining this display
with \eqref{eq:regenerative-boundary-covariance-main} and
\eqref{eq:regenerative-uncentered-fluctuation-main} yields
\[
\frac{1}{b_N(M_N-1)}\sum_{\ell=1}^{M_N}
(B_{\ell,N}-\bar B_N)(B_{\ell,N}-\bar B_N)^\top
=T_m+O_{\PP}\left(b_N^{-1}+M_N^{-1/2}\right),
\]
which is \eqref{eq:regenerative-separated-batch-covariance-main}.
\end{proof}

\begin{proposition}[Standard complete-cycle and separated-batch facts]\label{lem:local-window-regenerative-decomposition-main}\label{prop:generic-complete-cycle-regenerative-clt-main}\label{lem:calendar-stopped-batch-cov-main}\label{lem:separated-batch-covariance-estimator-main}\label{lem:fixed-low-cycle-event-main}\label{lem:uniform-calendar-block-length-main}
Let \((A_n)_{n\in\mathbb Z}\) be a stationary binary renewal process with
click epochs \((\tau_j)\), i.i.d. gaps \(G_j\), cycle lengths
\(L_j=G_j+1\), mean \(\mu=\EE L_0\in(0,\infty)\), and
\(\EE e^{\eta G_0}<\infty\) for some \(\eta>0\).  Let
\(K_N=\max\{k:\tau_k<N\}\), with \(K_N=-1\) if no click is observed.  The
following standard regenerative conclusions are used in the main text
\cite{GlynnIglehart1990STS,AsmussenGlynn2007,GlynnWhitt1991BatchMeans,
VatsFlegalJones2019Multivariate}.
\begin{enumerate}[label=\textup{(\alph*)},leftmargin=2.4em,itemsep=0.25em]
\item Every bounded local window
\[
Z_n=\psi(A_{n-r},\ldots,A_{n+\ell}),\qquad \bar z=\EE Z_0,
\]
has a complete-cycle reward
\[
U_j=\sum_{n=\tau_j}^{\tau_{j+1}-1}Z_n,\qquad
Y_j=U_j-\bar zL_j,
\]
with Palm centering
\begin{equation}\label{eq:local-window-palm-inversion-main}
\EE U_j=\mu\bar z,\qquad \EE Y_j=0 .
\end{equation}
Moreover \(Y_j\) is a measurable function of finitely many adjacent gaps and
is therefore fixed-range dependent.  Endpoint and fixed look-ahead/look-back
trimming changes the calendar sum by
\begin{equation}\label{eq:local-window-endpoint-main}
\sum_{n=0}^{N-1}Z_n-\sum_j U_j=O_{\PP}(\log N),
\end{equation}
where the cycle sum is over the complete cycles intersecting the observed
calendar window, with the same \(O_{\PP}(\log N)\) bound after any fixed
endpoint trimming.

\item If \(Y_j=f(G_j,\ldots,G_{j+m_{\rm dep}})\in\RR^{d_{\rm rew}}\),
\(\EE Y_j=0\), and
\[
\|f(g_0,\ldots,g_{m_{\rm dep}})\|
\le C_f\left(1+\sum_{i=0}^{m_{\rm dep}}g_i\right),
\]
then, for any all-complete-cycle index set \(J_N\) obtained from
\(\{0,\ldots,K_N-1\}\) by deleting only fixed endpoint/look-ahead indices,
\[
\sqrt N\,\frac1N\sum_{j\in J_N}Y_j
\Rightarrow
\mathcal N\left(0,\frac{\mathcal T_{m_{\rm dep}}}{\mu}\right),
\]
where
\[
\mathcal T_{m_{\rm dep}}
=\Var(Y_0)+
\sum_{h=1}^{m_{\rm dep}}
\{\operatorname{Cov}(Y_0,Y_h)+\operatorname{Cov}(Y_h,Y_0)\}.
\]

\item For \(b_N\to\infty\), \(b_N/N\to0\), \(d_N=b_N+m_{\rm dep}\), and
separated batches
\[
I_{\ell,N}:=\{(\ell-1)d_N,\ldots,(\ell-1)d_N+b_N-1\},
\qquad
C_{\ell,N}=\sum_{j=(\ell-1)d_N}^{(\ell-1)d_N+b_N-1}Y_j ,
\]
let \(M_N\) be the number of all-complete-cycle batches retained before
calendar time \(N\), and put
\(\bar C_N=M_N^{-1}\sum_{\ell=1}^{M_N}C_{\ell,N}\) on \(M_N\ge2\).  Then the
retained batch covariance satisfies
\begin{equation}\label{eq:separated-batch-covariance-estimator-rate-main}
\left\|
\frac{K_N}{N}\frac1{b_N(M_N-1)}
\sum_{\ell=1}^{M_N}(C_{\ell,N}-\bar C_N)(C_{\ell,N}-\bar C_N)^\top
-\frac{\mathcal T_{m_{\rm dep}}}{\mu}
\right\|_{\operatorname{op}}
=O_{\PP}\left(b_N^{-1}+\sqrt{b_N/N}\right).
\end{equation}
Equivalently, before multiplying by \(K_N/N\), the cycle-time sample covariance
estimates \(\mathcal T_{m_{\rm dep}}\) at the same rate.

\item If plug-in rewards have the form
\[
\widehat Y_j=Y_j-(\widehat r_N-r)L_j,\qquad
\widehat r_N-r=O_{\PP}(N^{-1/2}),
\]
and \(b_N\) is polynomial, then
\begin{equation}\label{eq:separated-batch-plugin-block-main}
\max_{\ell\le M_N}
\left\|\sum_{j\in I_{\ell,N}}(\widehat Y_j-Y_j)\right\|
=O_{\PP}(b_NN^{-1/2})
\end{equation}
and replacing \(Y_j\) by \(\widehat Y_j\) in the separated-batch covariance
changes it by
\begin{equation}\label{eq:separated-batch-plugin-covariance-main}
O_{\PP}\!\left(\sqrt{\frac{b_N}{N}}\right)
+O_{\PP}\!\left(\frac{b_N}{N}\right).
\end{equation}

\item For every fixed \(C\), \(\PP(K_N<C)\to0\), and the same conclusion holds
after deleting any fixed number of endpoint, look-back, or look-ahead cycles.

\item If \(b_N=\lfloor N^\beta\rfloor\), \(0<\beta<1\), and
\(M_N=O_{\PP}(N/b_N)\), then
\begin{equation}\label{eq:uniform-calendar-block-length-main}
\max_{1\le \ell\le M_N}
\frac1{b_N}\sum_{j\in I_{\ell,N}}L_j
=O_{\PP}(1).
\end{equation}
\end{enumerate}
\end{proposition}

\begin{proof}
The only model-specific input to this proposition is that the click epochs
form a stationary renewal process with an exponential gap moment.  The
Palm-cycle identity gives
\[
\EE Z_0=\frac{\EE U_j}{\EE L_j},\qquad
U_j=\sum_{n=\tau_j}^{\tau_{j+1}-1}Z_n,
\]
and hence \(\EE U_j=\mu\bar z\) and \(\EE Y_j=0\).  Because \(Z_n\) is a fixed
local window, \(Y_j\) depends on only finitely many adjacent gaps; call this
range \(m_{\rm dep}\).  Exponential tails imply that the largest cycle
intersecting any fixed enlargement of \([0,N-1]\) is \(O_{\PP}(\log N)\), so
endpoint, look-ahead, and look-back deletions give
\eqref{eq:local-window-endpoint-main}.

The CLT and the calendar-time scale are the standard regenerative-reward
facts: apply the ordinary finite-range-dependent CLT to the stationary
sequence \(Y_j=f(G_j,\ldots,G_{j+m_{\rm dep}})\), then use the renewal law of
large numbers \(K_N/N\to1/\mu\) and the usual random-index replacement
\cite[Chs.~IV.3 and VI.3]{AsmussenGlynn2007}.  This gives the covariance
\(\mathcal T_{m_{\rm dep}}/\mu\).

For the separated-batch estimator, apply
\Cref{thm:regenerative-separated-batch-covariance-main} to the fixed-range
gap reward \(Y_j=f(G_j,\ldots,G_{j+m_{\rm dep}})\).  This gives the
cycle-index covariance estimate
\[
\frac1{b_N(M-1)}
\sum_{\ell=1}^{M}(C_{\ell,N}-\bar C_N)(C_{\ell,N}-\bar C_N)^\top
=\mathcal T_{m_{\rm dep}}
+O_{\PP}\left(b_N^{-1}+\sqrt{b_N/N}\right)
\]
for deterministic prefixes \(M\asymp N/b_N\).  The all-complete-cycle retained
prefix is a random prefix of the same independent separated-batch sequence.
The renewal law gives
\[
M_N=\frac{K_N}{b_N+m_{\rm dep}}+O(1),\qquad K_N/N\to\mu^{-1},
\]
in probability after the fixed endpoint and look-ahead deletions.  Replacing
the deterministic prefix by this calendar-stopped prefix changes the
normalized covariance by \(O_{\PP}(b_N/N)+O_{\PP}(N^{-1/2})\), using the same
fourth-moment bound as in
\Cref{thm:regenerative-separated-batch-covariance-main}; this is absorbed by
\(O_{\PP}(b_N^{-1}+\sqrt{b_N/N})\).  Multiplication by
\(K_N/N=\mu^{-1}+O_{\PP}(N^{-1/2})\) converts the cycle-time covariance
\(\mathcal T_{m_{\rm dep}}\) to the calendar-time covariance
\(\mathcal T_{m_{\rm dep}}/\mu\) and contributes only another absorbed
\(O_{\PP}(N^{-1/2})\) term.  This proves
\eqref{eq:separated-batch-covariance-estimator-rate-main}.

For the plug-in perturbation, write
\(\Delta_N=\widehat r_N-r=O_{\PP}(N^{-1/2})\).  Each batch changes by
\[
\widehat C_{\ell,N}-C_{\ell,N}
=-\Delta_N\sum_{j\in I_{\ell,N}}L_j .
\]
The same exponential-moment Chernoff bound gives
\[
\max_{\ell\le M_N}b_N^{-1}\sum_{j\in I_{\ell,N}}L_j=O_{\PP}(1),
\]
which proves \eqref{eq:separated-batch-plugin-block-main}.  To see the
covariance perturbation explicitly, put
\[
E_{\ell,N}:=\widehat C_{\ell,N}-C_{\ell,N},\qquad
\bar E_N:=M_N^{-1}\sum_{\ell=1}^{M_N}E_{\ell,N}.
\]
On the high-probability event where
\(\max_{\ell\le M_N}\|E_{\ell,N}\|\le C b_NN^{-1/2}\), the centered covariance
difference is bounded by
\[
\begin{aligned}
&\frac1{b_N(M_N-1)}
\left\|\sum_{\ell=1}^{M_N}
(\widehat C_{\ell,N}-\bar{\widehat C}_N)
(\widehat C_{\ell,N}-\bar{\widehat C}_N)^\top
-\sum_{\ell=1}^{M_N}
(C_{\ell,N}-\bar C_N)(C_{\ell,N}-\bar C_N)^\top
\right\|_{\operatorname{op}}\\
&\qquad\le
\frac{2}{b_N(M_N-1)}
\sum_{\ell=1}^{M_N}\|C_{\ell,N}-\bar C_N\|\,\|E_{\ell,N}-\bar E_N\|
+\frac{1}{b_N(M_N-1)}
\sum_{\ell=1}^{M_N}\|E_{\ell,N}-\bar E_N\|^2 .
\end{aligned}
\]
By Cauchy's inequality,
\(\sum_\ell\|E_{\ell,N}-\bar E_N\|^2=O_{\PP}(M_Nb_N^2/N)\).  Also
\[
\frac1{b_N(M_N-1)}
\sum_{\ell=1}^{M_N}\|C_{\ell,N}-\bar C_N\|^2
=O_{\PP}(1),
\]
because the separated-batch covariance estimator already has bounded
operator norm in probability by
\eqref{eq:separated-batch-covariance-estimator-rate-main}.  Therefore the
cross term contributes \(O_{\PP}(\sqrt{b_N/N})\) after the outer normalization.
The second sum contributes
\(O_{\PP}(b_N/N)\).  Multiplication by \(K_N/N=O_{\PP}(1)\) leaves the same
orders, proving \eqref{eq:separated-batch-plugin-covariance-main}.  Finally,
\((K_N+1)/N\to1/\mu\) gives the fixed-low-cycle statement, and the preceding
uniform block-length bound is exactly \eqref{eq:uniform-calendar-block-length-main}.
\end{proof}

\subsection{Finite-domain diagonal certification for implementation extensions}
\label{app:finite-domain-diagonal-certification-main}

The submitted manuscript uses only the point certificate at \(z=1/2\).  The
following interval-cover verifier is retained as implementation machinery for a
future extension that prints additional point or interval transcripts.  It is
not a paper-level theorem claim for any candidate outside \(z=1/2\).

\begin{algorithmiclemma}[Finite interval-cover diagonal branch certification]\label{lem:diagonal-certified-interval-cover-main}
Let \(I=[z^-,z^+]\Subset(0,1)\) be a compact diagonal candidate interval that
is reported as coverage-certified in an implementation extension.  The
deterministic branch verifier is as follows.

\begin{enumerate}[label=\textup{(\arabic*)},leftmargin=2.4em,itemsep=0.25em]
\item Start with the queue containing \(I\) and with an empty certified list
\(\mathcal C\).
\item Remove one interval \(J\) from the queue.  Apply the outward-rounded
evaluation in \Cref{lem:diagonal-interval-branch-certificate-main} on the
whole interval \(J\).
\item If exactly one of the four branch tests in
\Cref{def:dmap-diagonal-rank-certificate-main} succeeds on \(J\), append
\((J,\mathrm{branch})\) to \(\mathcal C\).
\item If no branch test succeeds and the prescribed maximum subdivision depth,
precision budget, or exact-symbolic fallback has not been exhausted, bisect
\(J\) and return the two children to the queue.
\item If no branch test succeeds after the declared resources are exhausted,
append \(J\) to the undecided list \(\mathcal U\).
\end{enumerate}

The extension transcript contains the input interval \(I\), the precision and
rounding mode, every terminal interval, the branch label and margin for every
interval in \(\mathcal C\), the undecided list \(\mathcal U\), and an exact
rational endpoint check proving
\[
I=\bigcup_{(J,\mathrm{branch})\in\mathcal C}J\ \cup\
\bigcup_{J\in\mathcal U}J
\]
with disjoint interiors.  The interval \(I\) is a certified extension interval
only when this verifier returns a finite terminal list and
\(\mathcal U=\varnothing\).  In that case
\[
I=\bigcup_{j=1}^M I_j
\]
and directed interval evaluation proves exactly one branch condition on each
\(I_j\).  Therefore every candidate \(z\in I\) belongs to a certified
subinterval on which the population branch of
\(\Sigma_\sigma(z)\) is fixed.  If \(\mathcal U\ne\varnothing\), the union of
the undecided intervals is displayed and those candidates remain diagnostic.
\end{algorithmiclemma}

\begin{proof}
All branch assertions made on a terminal interval are consequences of
\Cref{lem:diagonal-interval-branch-certificate-main}, so each accepted
terminal interval has a proved population branch for every \(z\) in that
interval.  The four branch outcomes in
\Cref{def:dmap-diagonal-rank-certificate-main} are mutually exclusive for the
actual nonnegative definite matrix \(\Sigma_\sigma(z)\) by
\Cref{lem:dmap-diagonal-covariance-alternatives-main}.  Hence two distinct
branch labels cannot both be valid at the same population point; the verifier
therefore accepts a terminal interval only when its directed enclosures prove a
single outcome uniformly over the interval.

The equality of the endpoint union is a finite rational check in the transcript,
independent of floating-point arithmetic.  When the undecided list is empty,
the accepted terminal intervals cover all of \(I\), and every point in \(I\)
inherits the unique branch proved on the terminal interval containing it.  If
the list is nonempty, no coverage-certified extension is invoked on that
undecided subunion.
\end{proof}

\subsection{Finite-record implementation protocols}\label{app:finite-record-implementation-protocols-main}

The following protocols implement the reporting conventions used by
Theorems~C--E.  They are computational recipes and certificate checks, not
additional assumptions or paper-level theorem statements.

\phantomsection\label{alg:all-complete-cycle-separated-batch-main}
\paragraph{Algorithm 1: all-complete-cycle separated-batch covariance.}

\begin{enumerate}[label=\textup{\arabic*.},leftmargin=2.0em,itemsep=0.35em]
\item From the retained record \(A_0,\ldots,A_{N-1}\), form the complete-cycle
sets \(J_{m,N}\) of
\Cref{def:cycle-index-inclusion-estimator-main} and use only the common set
\[
J_N:=J_{0,N}\cap J_{1,N}\cap J_{2,N}.
\]
Initial incomplete cycles, terminal incomplete cycles, and the fixed terminal
look-ahead cycles required by the local windows are omitted.

\item For each \(j\in J_N\), compute the observable centered reward
\[
\widehat Y_j=
\bigl(U_j^{(0)}-\widehat r_{0,N}^{\mathrm{cyc}}L_j,\,
U_j^{(1)}-\widehat r_{1,N}^{\mathrm{cyc}}L_j,\,
U_j^{(2)}-\widehat r_{2,N}^{\mathrm{cyc}}L_j\bigr)^\top .
\]

\item With \(b_N=\lfloor N^{1/3}\rfloor\), set \(d_N=b_N+1\) and use separated
batch grids
\[
I_{\ell,N}:=\{(\ell-1)d_N,\ldots,(\ell-1)d_N+b_N-1\}.
\]
The omitted index \(\ell d_N-1\) is the separator.

\item Let
\[
M_N:=\max\{m\ge0:\ I_{\ell,N}\subseteq J_N
\text{ for every }1\le\ell\le m\},
\]
with \(M_N=0\) if the set is empty.  A partial terminal batch is omitted.
With \(K_N=\max\{k:\tau_k<N\}\), and with \(K_N=-1\) if no click is observed,
set
\[
\widehat D_N=
\widehat r_{1,N}^{\mathrm{cyc}}-(\widehat r_{0,N}^{\mathrm{cyc}})^2,
\qquad
\mathcal E_N=
\{K_N\ge K_{\min},\ M_N\ge2,
\widehat r_{0,N}^{\mathrm{cyc}}\ge\varepsilon_N,
|\widehat D_N|\ge\varepsilon_N\}.
\]

\item Define
\[
\widehat B_{\ell,N}:=\sum_{j\in I_{\ell,N}}\widehat Y_j,\qquad
\overline{\widehat B}_N:=M_N^{-1}\sum_{\ell=1}^{M_N}\widehat B_{\ell,N}
\quad\text{on }\mathcal E_N.
\]
Report the covariance estimator only on \(\mathcal E_N\):
\[
\widehat\Sigma_{r,N}
=
\frac{K_N}{N}\,
\frac1{b_N(M_N-1)}
\sum_{\ell=1}^{M_N}
(\widehat B_{\ell,N}-\overline{\widehat B}_N)
(\widehat B_{\ell,N}-\overline{\widehat B}_N)^\top .
\]
On \(\mathcal E_N^c\), do not form a zero replacement; report
``insufficient complete cycles; no confidence set.''
\end{enumerate}

\paragraph{Algorithm 2: quotient inverse, projection, and chart retention.}

Compute \(\widehat D_N\) and apply the denominator thresholds
\(\widehat r_{0,N}^{\mathrm{cyc}}\ge\varepsilon_N\) and
\(|\widehat D_N|\ge\varepsilon_N\).  If \(\mathcal E_N^c\) occurs, stop
retained-record inference and report ``insufficient complete cycles; no
confidence set.''  On \(\mathcal E_N\), compute
\(\Phi(\widehat r_N^{\mathrm{cyc}})\) and project the resulting raw symmetric
coordinate by
\(\widetilde\sigma_N=\Pi_F^{\rm tb}(\Phi(\widehat r_N^{\mathrm{cyc}}))\).
Report separated labelled rates only when
\[
\widehat\Delta_{\sigma,N}\ge\delta_N^2,
\qquad
\widehat p_N,\widehat s_N\in[\eta_N,1-\eta_N],
\]
and the deterministic oracle branch map \(\mathcal B\) is unique on the two
projected roots.  If an estimated branch map is used, require the
high-probability oracle-agreement condition from
\Cref{conv:branch-map-labelled-rates-main}; otherwise withhold separated
labels.  At a population exchange diagonal, use only the pointwise common-rate
acceptance event calibrated by Algorithm~3; a simultaneous diagonal band needs
a separate certified inversion rule.

\paragraph{Algorithm 3: fixed-candidate projected-Gaussian diagonal calibration.}

Fix \(0<\alpha<1\).  In this manuscript the coverage-certified candidate set
is the singleton
\[
Z_{\rm cert}^{\rm paper}=\{1/2\},
\]
because \(\Cref{lem:z-half-diagonal-branch-certificate-main}\) is the only
printed terminal branch certificate.  Thus the paper-level operational
algorithm is run at \(z=1/2\), with \(n_z=(-z,1)^\top\).  A finite set or
finite interval union \(Z_{\rm cert}\) from
\Cref{def:dmap-diagonal-rank-certificate-main,lem:diagonal-certified-interval-cover-main}
is only an implementation extension: every displayed candidate in that set
must carry its own finite branch transcript and a certified conditional
quantile transcript before it can be used for coverage reporting.  The finite
Monte Carlo add-on, when used for a coverage-certified report rather than a
plot, has the fixed theorem-level settings
\[
R_N=N,\qquad \delta_N=N^{-2},\qquad
\varepsilon_N^{\rm MC}:=\sqrt{\frac{\log(2N^2)}{2N}} .
\]
The algorithm has two outputs: an ideal conditional projected-Gaussian
calculation and, only when all certificates below are present, a
coverage-certified finite report.  In the submitted manuscript, if
\(z\ne1/2\), continue only for diagnostic plotting or diagnostic
ideal-quantile evaluation and report ``diagonal candidate outside the printed
paper certificate; no coverage-certified acceptance event.''  In an
implementation extension, the same diagnostic-only rule applies to every
\(z\notin Z_{\rm cert}\).  For a certified candidate, compute the population diagonal
covariance \(\Sigma_\sigma(z)\) from the finite D-MAP formula
\eqref{eq:dmap-diagonal-sigma-finite-main} and attach the outward-rounded
branch certificate specified in
\Cref{def:dmap-diagonal-rank-certificate-main,lem:diagonal-interval-branch-certificate-main}.
The certificate transcript must display the intervals for
\(\operatorname{tr}\Sigma_\sigma(z)\), \(\det\Sigma_\sigma(z)\), and
\(\Sigma_\sigma(z)(1,z)^\top\).  If the branch certificate is inconclusive,
continue only if a diagnostic projected-Gaussian plot or diagnostic ideal
quantile is desired, and report ``diagonal covariance branch not certified; no
coverage-certified acceptance event.''  A finite-sample eigenvalue calculation of
\(\widehat\Sigma_{\sigma,N}\) is not a substitute for this population
certificate.

Compute
\[
\widehat\Sigma_{\sigma,N}
=D\Phi(\widehat r_N^{\mathrm{cyc}})
 \widehat\Sigma_{r,N}
 D\Phi(\widehat r_N^{\mathrm{cyc}})^\top
\]
on the reporting event \(\mathcal E_N\); if \(\mathcal E_N^c\) occurs, stop
and report ``insufficient complete cycles; no confidence set.''  Symmetrize
the matrix for numerical reporting and truncate only roundoff-negative
eigenvalues to zero.

Diagonalize
\[
\widehat\Sigma_{\sigma,N}
=\sum_{i=1}^2\widehat\lambda_{i,N}
\widehat v_{i,N}\widehat v_{i,N}^{\top},
\qquad
\widehat\lambda_{1,N}\ge\widehat\lambda_{2,N}\ge0,
\]
retain exactly
\[
\widehat I_N=\{i:\widehat\lambda_{i,N}>t_N\},
\qquad \widehat k_N=|\widehat I_N|,
\]
and use
\[
\widehat\Sigma_{\sigma,N}^{+}
=\sum_{i\in\widehat I_N}
\widehat\lambda_{i,N}^{-1}
\widehat v_{i,N}\widehat v_{i,N}^{\top}.
\]
If \(\widehat k_N=0\), report ``retained rank zero'' and use only the
buffered conservative acceptance event from
\Cref{thm:e1-ideal-fixed-candidate-projection-main} when the population
certificate is the zero-covariance branch; otherwise this finite retained-rank
event is an asymptotic pseudoinverse event, not a population branch label.  If
the population certificate is ``rank-one normal direction,'' then, with
\(p_c=1-\alpha\),
the limiting law satisfies
\[
\PP(T_z=0)=\frac12,\qquad
\PP(T_z\le t)=\frac12+\frac12F_{\chi^2_1}(t),\quad t>0.
\]
Thus \(c_{p_c}=0\) for \(p_c\le1/2\), in which case use the buffered
conservative acceptance event from
\Cref{thm:e1-ideal-fixed-candidate-projection-main}.  For \(p_c>1/2\),
\(c_{p_c}=F_{\chi^2_1}^{-1}(2p_c-1)>0\), and the unbuffered exact branch of
\Cref{thm:e1-ideal-fixed-candidate-projection-main} applies.  If the population certificate is full rank or
rank-one non-normal, use the exact projected-Gaussian branch.  In the
rank-one non-normal case the branch certificate must also display
\([a_{\rm rn}]\) from
\eqref{eq:diagonal-rank-one-nonnormal-scale-certificate-main}; the ideal
conditional law is
\[
F^{(1)}_{z,[a_{\rm rn}]}(c)
=\frac12F_{\chi^2_1}(c)
 +\frac12F_{\chi^2_1}\!\left(\frac{c}{a_{\rm rn}}\right),
\]
with \(a_{\rm rn}\) interpreted through its certified interval enclosure and
the lower quantile bracketed by the one-dimensional directed bisection in
\Cref{lem:certified-rank-one-nonnormal-projected-gaussian-quantile-main}.  If
the population certificate is zero covariance, use the retained-rank-zero
buffered branch.  If it is inconclusive, no coverage-certified branch is
selected.

Generate conditionally independent auxiliary Gaussian draws
\[
Z_{N,m}^{\ast}\sim\mathcal N(0,\widehat\Sigma_{\sigma,N}),
\qquad 1\le m\le N,
\]
when a finite simulation diagnostic or DKW-certified simulated critical value
is requested.  These are plug-in draws from the limiting projected Gaussian
law, not resampled records, and their count is unrelated to the separated-
batch count \(M_N\).  Project each draw onto
\[
H_z=\{h:h_2\le zh_1\},\qquad
\Pi_{H_z}(u)
=u-\frac{(u_2-zu_1)_+}{1+z^2}\binom{-z}{1},
\]
and sort the quadratic values
\[
T_{N,m}^{\ast}(z)
=\Pi_{H_z}(Z_{N,m}^{\ast})^\top
\widehat\Sigma_{\sigma,N}^{+}
\Pi_{H_z}(Z_{N,m}^{\ast}).
\]
The ideal critical value \(c^{-}_{1-\alpha,N}(z)\) is the lower conditional
\((1-\alpha)\)-quantile of this projected-Gaussian law.  In the retained
full-rank branch it is constructed from the conditional analogue of
\[
F_{z,\Sigma}(c)=
\PP\{Z\in H_z,\ Z^\top\Sigma^{-1}Z\le c\}
+\PP\{Z\notin H_z,\
(\Pi_{H_z}Z)^\top\Sigma^{-1}(\Pi_{H_z}Z)\le c\}
\]
in \Cref{lem:deterministic-full-rank-projected-gaussian-quantile-main}, with
\(\Sigma\) replaced by the retained \(\widehat\Sigma_{\sigma,N}\).  In the
rank-one non-normal branch it is constructed from the scalar mixture
\eqref{eq:rank-one-nonnormal-cdf-main}, with \(a\) replaced by the certified
conditional enclosure obtained from the retained rank-one covariance.  A
coverage-certified report must evaluate the applicable ideal conditional
quantile directly, either analytically or by the appropriate deterministic
certified enclosure routine:
\Cref{thm:certified-projected-gaussian-quantile-algorithm-main,lem:certified-deterministic-projected-gaussian-quantile-main}
for full rank and
\Cref{lem:certified-rank-one-nonnormal-projected-gaussian-quantile-main}
for rank-one non-normal.  The full-rank
routine must use a certified Gaussian tail truncation, interval quadrature over
the two projected regions in the full-rank display above, and bisection
stopping rules.  In the rank-one non-normal branch, the one-dimensional
routine of
\Cref{lem:certified-rank-one-nonnormal-projected-gaussian-quantile-main}
instead uses directed evaluations of \(F_{\chi^2_1}(x)=2\Phi(\sqrt x)-1\)
and bisection on the scalar mixture.  In either case the transcript must
produce a quantile bracket \((L,U]\) narrow enough to decide the inequality
\[
Q_N(z)\le c^{-}_{1-\alpha,N}(z)
\]
or, on a buffered branch, the inequality with
\(c^{-}_{1-\alpha,N}(z)+\varepsilon\).  If no such certified ideal-quantile
evaluation is available, the branch certificate alone is insufficient: report
``ideal quantile only; Monte Carlo diagnostic not coverage-certified'' for
this candidate.

When optional Monte Carlo draws are generated, the raw simulation quantile
estimate is the order statistic
\[
\widehat c^{-}_{1-\alpha,N}(z)
=T_{N,(\lceil(1-\alpha)N\rceil)}^{\ast}(z),
\]
after sorting increasingly.  It may be displayed as a diagnostic for the ideal
conditional quantile.  To turn finite simulation into an operational
coverage-certified critical value, the algorithm uses only the canonical DKW
choice \(R_N=N\), \(\delta_N=N^{-2}\).  If
\(\varepsilon_N^{\rm MC}\ge\alpha\), the simulated quantile remains
diagnostic.  If \(\varepsilon_N^{\rm MC}<\alpha\), the DKW one-sided simulated
critical value is
\[
\widehat c^{\mathrm{DKW},-}_{1-\alpha,N}(z)
=T_{N,(\lceil(1-\alpha+\varepsilon_N^{\rm MC})N\rceil)}^\ast(z),
\]
equivalently
\[
\widehat c^{\mathrm{DKW},-}_{1-\alpha,N}(z)
=\widehat c^{-}_{1-\alpha,N}(z)
+\Bigl[
T_{N,(\lceil(1-\alpha+\varepsilon_N^{\rm MC})N\rceil)}^\ast(z)
-T_{N,(\lceil(1-\alpha)N\rceil)}^\ast(z)
\Bigr].
\]
The event
\[
G_{\mathrm{MC},N}^{\mathrm{DKW}}(z)
=\{\Delta_{N,N}(z)\le\varepsilon_N^{\rm MC}\},
\qquad
\Delta_{N,N}(z)=
\sup_t|\widehat F^\ast_{N,N}(t;z)-F_N^\ast(t;z)|,
\]
has conditional probability at least \(1-N^{-2}\), and on it
\[
\widehat c^{\mathrm{DKW},-}_{1-\alpha,N}(z)
\ge c^{-}_{1-\alpha,N}(z).
\]
Equivalently, before conditioning on the auxiliary simulation,
\[
\PP^\ast\{\widehat c^{\mathrm{DKW},-}_{1-\alpha,N}(z)
\ge c^{-}_{1-\alpha,N}(z)\}\ge1-N^{-2}.
\]
This is the default certified finite-Monte-Carlo choice in
\Cref{cor:exchange-diagonal-dkw-monte-carlo-main}.  The
reported simulated event is then
\[
Q_N(z)\le
\widehat c^{\mathrm{DKW},-}_{1-\alpha,N}(z),
\]
with coverage transferred from the ideal Theorem~E1 statement, the
reported-event corollary
\Cref{cor:exchange-diagonal-reported-event-main}, and the DKW transfer
Corollary~E3 \Cref{cor:exchange-diagonal-dkw-monte-carlo-main}.  No other finite
simulation choice is treated as an operational certificate.  Without the
canonical DKW event above, all simulated quantiles are diagnostic only: report
``ideal quantile only; Monte Carlo diagnostic not coverage-certified'' for
this candidate.

\phantomsection\label{alg:projected-plugin-word-law-report-main}
\paragraph{Algorithm 4: projected plug-in word-law report.}

This protocol implements
\Cref{def:projected-plugin-word-law-report-main,prop:finite-record-complete-law-renewal-consistency-main}.
It is a numerical reporting recipe for a retained projected renewal law, not a
confidence procedure for word probabilities.

\begin{enumerate}[label=\textup{\arabic*.},leftmargin=2.0em,itemsep=0.35em]
\item Start with the finite-record reporting gate \(\mathcal E_N\), the
quotient inverse map \(\Phi\), the feasible projection \(\Pi_F^{\rm tb}\),
and the thresholds \(\varepsilon_N,\eta_N\) of the fixed protocol
\eqref{prot:finite-record-main}.  If \(\mathcal E_N^c\) occurs, stop and
report ``insufficient complete cycles; no projected plug-in word-law report.''

\item On \(\mathcal E_N\), compute
\[
\widetilde\sigma_N
:=\Pi_F^{\rm tb}\{\Phi(\widehat r_N^{\mathrm{cyc}})\},
\]
and let \(\{\widetilde z_{1,N},\widetilde z_{2,N}\}\) be the unordered roots of
\[
z^2-\widetilde\sigma_{1,N}z+\widetilde\sigma_{2,N}=0 .
\]
Retain the word-law branch only on
\[
\mathcal E_N^{\rm law}
=\mathcal E_N\cap
\{\widetilde z_{1,N},\widetilde z_{2,N}\in[\eta_N,1-\eta_N]\}.
\]
On \((\mathcal E_N^{\rm law})^c\), withhold all projected plug-in word
probabilities.

\item On \(\mathcal E_N^{\rm law}\), compute the projected root gap law
\(\widetilde g_{k,N}\) from the reconstruction formulas in
\Cref{lem:algorithmic-complete-law-reconstruction-main}.  Define the projected
renewal intensity by
\[
\widetilde\rho_N
=\rho_*(\widetilde\sigma_N)
:=\left(1+\sum_{k\ge0}k\,\widetilde g_{k,N}\right)^{-1}.
\]
Do not insert the noisy empirical intensity
\(\widehat r_{0,N}^{\mathrm{cyc}}\) into the final renewal inclusion product.

\item For a requested finite window \(I\), put \(d_I=0\) if \(|I|\le1\), and
otherwise put \(d_I=\max I-\min I\).  The report uses only
\(\widetilde u_{0,N},\ldots,\widetilde u_{d_I,N}\), obtained from
\[
\widetilde U_N(t):=\sum_{n\ge0}\widetilde u_{n,N}t^n
:=\frac{1}{1-\sum_{k\ge0}\widetilde g_{k,N}t^{k+1}},
\]
or equivalently from the finite recursion
\[
\widetilde u_{0,N}=1,\qquad
\widetilde u_{n,N}
=\sum_{\ell=1}^{n}\widetilde g_{\ell-1,N}\,
  \widetilde u_{n-\ell,N},\qquad 1\le n\le d_I .
\]
If the implementation truncates the gap law after \(K\ge d_I-1\), the renewal
masses needed for this finite window are unchanged; the truncation error in
\(\widetilde u_{0,N},\ldots,\widetilde u_{d_I,N}\) is exactly zero.

\item The only infinite-tail numerical quantity in a finite-window report is
the scalar mean in \(\widetilde\rho_N\).  Evaluate it either from the closed
root formula in
\Cref{rem:complete-law-numerical-implementation-main} or from a truncated
sum accompanied by the certified tail bound stated there.  If the declared
tail tolerance certificate fails, withhold the projected plug-in word-law
report instead of replacing the failed quantity by an uncertified number.

\item For every nonempty \(B=\{i_1<\cdots<i_m\}\subset I\), report the projected
renewal inclusion probability
\[
\widetilde\PP_N(B\subset\varXi)
:=\widetilde\rho_N
\prod_{\ell=1}^{m-1}\widetilde u_{i_{\ell+1}-i_\ell,N},
\]
with empty inclusion value \(1\), and obtain the word probabilities on \(I\)
by finite inclusion--exclusion.  The reported object is the projected plug-in
word-law report of \(\Cref{def:projected-plugin-word-law-report-main}\).  It
contains projected word probabilities only, not confidence intervals for those
word probabilities.
\end{enumerate}


\paragraph{Estimated branch-map reduction.}\label{app:branch-map-oracle-reduction-main}
Work at a separated population point \(p\ne s\), and let \(\mathcal B\) be the
deterministic oracle branch map from
\Cref{conv:branch-map-labelled-rates-main}.  Choose disjoint open
neighbourhoods \(U_p,U_s\subset(0,1)\) of the two population roots on which the
oracle labels are constant.  Let \(\widehat{\mathcal B}_N\) be any branch map
formed from calibration or side data, and let
\[
\mathcal C_N=\{\widehat{\mathcal B}_N
\text{ agrees with }\mathcal B\text{ on }U_p\cup U_s\}.
\]
If \(\PP(\mathcal C_N)\to1\), then every retained labelled-root or
labelled-rate estimator obtained by replacing \(\mathcal B\) with
\(\widehat{\mathcal B}_N\) is asymptotically equivalent to the oracle-labelled
estimator.  In particular, all Gaussian limits, plug-in covariance
consistency statements, and retained-rank Wald limits proved conditionally on
\(\mathcal B\) remain valid for the estimated branch map.  If only the weaker
bound \(\PP(\mathcal C_N^c)\le\epsilon_N\) is available, each corresponding
one-sided coverage lower bound is reduced by at most \(\epsilon_N+o(1)\).

The retained empirical roots converge in probability to the separated
population roots.  Hence, with probability tending to one, the retained roots
lie in \(U_p\cup U_s\).  On the intersection of this event with
\(\mathcal C_N\), the estimated-map and oracle-map labelled estimators are
identical.  Therefore their difference is zero with probability tending to
one, and Slutsky's theorem transfers the oracle Gaussian, covariance, and Wald
limits.  For coverage, the reported event can fail relative to the oracle
event only on \(\mathcal C_N^c\), up to the \(o(1)\) probability that the
retained roots have not entered \(U_p\cup U_s\).  This gives the stated loss
bound.
