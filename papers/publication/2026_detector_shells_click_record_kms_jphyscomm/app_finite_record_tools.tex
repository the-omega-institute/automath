\section{Finite-Record Regenerative Tools}\label{app:finite-record-regenerative-tools-main}

This appendix collects the model-independent renewal and separated-batch
arguments used in \Cref{sec:records}.  The main retained-record asymptotic theorems use
these results only after the constrained D-MAP law has been reduced to the
explicit adjacent-gap rewards displayed in the main text.

\begin{theorem}[Separated-batch covariance for fixed-range gap rewards]\label{thm:regenerative-separated-batch-covariance-main}
Fix an integer \(m\ge0\) and a dimension \(d\).  Let
\((G_j)_{j\in\mathbb Z}\) be i.i.d. with
\(\EE e^{\eta G_0}<\infty\) for some \(\eta>0\), and let
\[
Y_j=f(G_j,\ldots,G_{j+m})\in\RR^d,\qquad
\EE Y_j=0,\qquad \EE\|Y_j\|^4<\infty .
\]
Set
\[
T_m=\operatorname{Var}(Y_0)+
\sum_{h=1}^m\{\operatorname{Cov}(Y_0,Y_h)+\operatorname{Cov}(Y_h,Y_0)\}.
\]
Let \(b_N\to\infty\), \(b_N/N\to0\),
\(M_N=\lfloor N/(b_N+m)\rfloor\), and define the separated batches
\[
I_{\ell,N}=\{(\ell-1)(b_N+m),\ldots,
(\ell-1)(b_N+m)+b_N-1\},\qquad
B_{\ell,N}=\sum_{j\in I_{\ell,N}}Y_j ,
\]
for \(1\le\ell\le M_N\).  On \(M_N\ge2\), put
\(\bar B_N=M_N^{-1}\sum_{\ell=1}^{M_N}B_{\ell,N}\).  Then
\begin{equation}\label{eq:regenerative-separated-batch-covariance-main}
\left\|
\frac{1}{b_N(M_N-1)}
\sum_{\ell=1}^{M_N}(B_{\ell,N}-\bar B_N)
(B_{\ell,N}-\bar B_N)^\top
-T_m
\right\|_{\operatorname{op}}
=O_{\PP}\left(b_N^{-1}+\sqrt{\frac{b_N}{N}}\right).
\end{equation}
\end{theorem}

\begin{proof}
The sequence \((Y_j)\) is \(m\)-dependent: if two index sets are separated by
more than \(m\), then the corresponding \(Y\)'s are functions of disjoint
families of i.i.d. gaps.  The retained batch \(B_{\ell,N}\) depends only on
\[
G_{(\ell-1)(b_N+m)},\ldots,
G_{(\ell-1)(b_N+m)+b_N-1+m}.
\]
The next retained batch starts with
\(G_{\ell(b_N+m)}\).  Since the separator has length \(m\), these gap sets are
disjoint for different \(\ell\), and therefore the retained sums
\((B_{\ell,N})_{\ell\le M_N}\) are independent and identically distributed.

Write \(\Gamma_h=\operatorname{Cov}(Y_0,Y_h)\).  Stationarity and
\(m\)-dependence give
\[
\operatorname{Var}(B_{\ell,N})
=b_N\Gamma_0+
\sum_{h=1}^{m}(b_N-h)(\Gamma_h+\Gamma_h^\top).
\]
Hence
\[
\frac{1}{b_N}\operatorname{Var}(B_{\ell,N})
=T_m+R_{b_N},\qquad
R_{b_N}=-\frac1{b_N}\sum_{h=1}^{m}h(\Gamma_h+\Gamma_h^\top).
\]
The matrices \(\Gamma_h\) are fixed finite matrices because
\(\EE\|Y_0\|^2<\infty\).  Thus
\begin{equation}\label{eq:regenerative-boundary-covariance-main}
\|R_{b_N}\|_{\operatorname{op}}\le Cb_N^{-1}.
\end{equation}
Equivalently, in a block of length \(b_N\) the infinite \(m\)-dependent
covariance sum misses only the \(h\) boundary pairs at each lag
\(1\le h\le m\), an \(O(1)\) number of covariance-pair layers after division
by \(b_N\).

It remains to bound the stochastic part.  The standard fourth-moment bound
for fixed-range-dependent sums gives
\begin{equation}\label{eq:regenerative-batch-fourth-main}
\EE\|B_{\ell,N}\|^4\le Cb_N^2 .
\end{equation}
Indeed,
\(\|B_{\ell,N}\|^4\le d^2\sum_{a,c=1}^d
(\sum_{j\in I_{\ell,N}}Y_{j,a})^2(\sum_{j\in I_{\ell,N}}Y_{j,c})^2\);
after expanding, any nonzero centered contribution can involve only index
quadruples whose dependency graph has connected components of diameter at
most \(m\).  There are \(O(b_N^2)\) such quadruples, and H\"older's inequality
bounds each expectation by a constant depending only on
\(\EE\|Y_0\|^4\), \(m\), and \(d\).

Define
\[
V_N:=b_N^{-1}\operatorname{Var}(B_{1,N}),\qquad
W_{\ell,N}:=
b_N^{-1}\{B_{\ell,N}B_{\ell,N}^\top-\EE(B_{\ell,N}B_{\ell,N}^\top)\}.
\]
The \(W_{\ell,N}\)'s are independent, mean zero, and
\[
\EE\|W_{\ell,N}\|_{\operatorname{F}}^2
\le b_N^{-2}\EE\|B_{\ell,N}B_{\ell,N}^\top\|_{\operatorname{F}}^2
= b_N^{-2}\EE\|B_{\ell,N}\|^4
\le C
\]
by \eqref{eq:regenerative-batch-fourth-main}.  Therefore
\begin{equation}\label{eq:regenerative-uncentered-fluctuation-main}
\EE\left\|
\frac{1}{b_NM_N}\sum_{\ell=1}^{M_N}
\{B_{\ell,N}B_{\ell,N}^\top-\EE(B_{\ell,N}B_{\ell,N}^\top)\}
\right\|_{\operatorname{op}}^2
\le
\EE\left\|
\frac{1}{M_N}\sum_{\ell=1}^{M_N}W_{\ell,N}
\right\|_{\operatorname{F}}^2
\le \frac{C}{M_N}.
\end{equation}
Consequently the uncentered empirical second moment satisfies
\[
\frac{1}{b_NM_N}\sum_{\ell=1}^{M_N}B_{\ell,N}B_{\ell,N}^\top
=V_N+O_{\PP}(M_N^{-1/2}).
\]
Since \(M_N=\lfloor N/(b_N+m)\rfloor\) and \(b_N/N\to0\),
\(M_N^{-1/2}=O(\sqrt{b_N/N})\).

The subtraction of the empirical batch mean changes only a smaller-order
term.  Algebra gives
\[
\frac{1}{b_N(M_N-1)}\sum_{\ell=1}^{M_N}
(B_{\ell,N}-\bar B_N)(B_{\ell,N}-\bar B_N)^\top
=
\frac{M_N}{M_N-1}
\left\{
\frac1{b_NM_N}\sum_{\ell=1}^{M_N}B_{\ell,N}B_{\ell,N}^\top
-\frac1{b_N}\bar B_N\bar B_N^\top
\right\}.
\]
Moreover
\[
\EE\left\|\frac1{\sqrt{b_N}}\bar B_N\right\|^2
=\frac1{b_NM_N^2}\sum_{\ell=1}^{M_N}\operatorname{tr}\Var(B_{\ell,N})
\le \frac{C}{M_N},
\]
so \(b_N^{-1}\bar B_N\bar B_N^\top=O_{\PP}(M_N^{-1})\).  Also
\((M_N/(M_N-1))-1=O(M_N^{-1})\) and \(V_N=O(1)\).  Combining this display
with \eqref{eq:regenerative-boundary-covariance-main} and
\eqref{eq:regenerative-uncentered-fluctuation-main} yields
\[
\frac{1}{b_N(M_N-1)}\sum_{\ell=1}^{M_N}
(B_{\ell,N}-\bar B_N)(B_{\ell,N}-\bar B_N)^\top
=T_m+O_{\PP}\left(b_N^{-1}+M_N^{-1/2}\right),
\]
which is \eqref{eq:regenerative-separated-batch-covariance-main}.
\end{proof}

\begin{proposition}[Calendar-stopped separated batches with plug-in centering]\label{prop:calendar-stopped-separated-batch-plugin-main}
Let \((G_j)_{j\ge0}\) be i.i.d. nonnegative integer-valued gaps with
\(\EE e^{\eta G_0}<\infty\) for some \(\eta>0\), put
\(L_j=G_j+1\), \(\mu=\EE L_0\), and
\[
S_k=\sum_{j=0}^{k-1}L_j,\qquad
K_N=\max\{k:S_k<N\}.
\]
Fix \(m\ge0\), \(d\ge1\), and
\[
Y_j=f(G_j,\ldots,G_{j+m})\in\RR^d,\qquad
\EE Y_j=0,
\]
where
\[
\|f(g_0,\ldots,g_m)\|\le C_f\left(1+\sum_{i=0}^m g_i\right).
\]
Let
\[
\mathcal T_m=\Var(Y_0)+
\sum_{h=1}^{m}\{\operatorname{Cov}(Y_0,Y_h)+\operatorname{Cov}(Y_h,Y_0)\}.
\]
For \(b_N=\lfloor N^\beta\rfloor\), \(0<\beta<1\), set
\(d_N=b_N+m\) and
\[
I_{\ell,N}=\{(\ell-1)d_N,\ldots,(\ell-1)d_N+b_N-1\},
\qquad
B_{\ell,N}=\sum_{j\in I_{\ell,N}}Y_j .
\]
Fix an integer \(q\ge0\), representing the fixed terminal endpoint and
look-ahead deletion.  The all-complete-cycle stopped batch count is
\[
M_N=\max\{M\ge0:\ (M-1)d_N+b_N-1+m+q\le K_N-1\},
\]
with \(M_N=0\) if the set is empty.  On \(M_N\ge2\), define
\(\bar B_N=M_N^{-1}\sum_{\ell=1}^{M_N}B_{\ell,N}\) and
\[
\widehat\Sigma_N^{Y}
=
\frac{K_N}{N}\frac1{b_N(M_N-1)}
\sum_{\ell=1}^{M_N}(B_{\ell,N}-\bar B_N)
(B_{\ell,N}-\bar B_N)^\top .
\]
Then \(M_N\ge2\) with probability tending to one and, with the harmless
auxiliary definition \(\widehat\Sigma_N^{Y}=0_{d\times d}\) on \(M_N<2\),
\begin{equation}\label{eq:calendar-stopped-separated-batch-rate-main}
\left\|\widehat\Sigma_N^{Y}-\frac{\mathcal T_m}{\mu}
\right\|_{\operatorname{op}}
=O_{\PP}\left(b_N^{-1}+\sqrt{\frac{b_N}{N}}\right).
\end{equation}

Moreover, let \(\Delta_N=O_{\PP}(N^{-1/2})\) be an \(\RR^d\)-valued
finite-record plug-in error and set
\[
\widehat Y_j=Y_j-\Delta_N L_j,\qquad
\widehat B_{\ell,N}=\sum_{j\in I_{\ell,N}}\widehat Y_j,\qquad
\overline{\widehat B}_N=M_N^{-1}\sum_{\ell=1}^{M_N}\widehat B_{\ell,N}.
\]
The mean \(\overline{\widehat B}_N\) is used only on \(M_N\ge2\).
Then
\begin{equation}\label{eq:calendar-stopped-plugin-block-main}
\max_{1\le\ell\le M_N}
\left\|\widehat B_{\ell,N}-B_{\ell,N}\right\|
=O_{\PP}(b_NN^{-1/2})
\end{equation}
and the Algorithm~1 covariance object with plug-in rewards satisfies
\begin{equation}\label{eq:calendar-stopped-plugin-covariance-rate-main}
\left\|
\frac{K_N}{N}\frac1{b_N(M_N-1)}
\sum_{\ell=1}^{M_N}
(\widehat B_{\ell,N}-\overline{\widehat B}_N)
(\widehat B_{\ell,N}-\overline{\widehat B}_N)^\top
-\frac{\mathcal T_m}{\mu}
\right\|_{\operatorname{op}}
=O_{\PP}\left(b_N^{-1}+\sqrt{\frac{b_N}{N}}\right),
\end{equation}
again with the auxiliary zero completion on \(M_N<2\).
\end{proposition}

\begin{proof}
First record the two renewal bounds used below.  Since \(L_0\) has an
exponential moment, the renewal central limit theorem gives
\begin{equation}\label{eq:calendar-stopped-k-rate-main}
K_N=\frac{N}{\mu}+O_{\PP}(\sqrt N),
\qquad
\frac{K_N}{N}=\frac1\mu+O_{\PP}(N^{-1/2}).
\end{equation}
The same exponential moment gives the maximal-cycle estimate
\begin{equation}\label{eq:calendar-stopped-max-cycle-main}
\max_{0\le j\le K_N+q+m+1}L_j=O_{\PP}(\log N).
\end{equation}
Indeed, for \(C_0>\mu^{-1}\),
\[
\PP\!\left(\max_{0\le j\le K_N+q+m+1}L_j>a\log N\right)
\le \PP(K_N>C_0N)
 +(C_0N+q+m+2)\PP(L_0>a\log N),
\]
and the right side tends to zero for \(a\) large enough.  Fixed endpoint or
look-ahead deletions therefore remove only \(O(1)\) cycles and at most
\(O_{\PP}(\log N)\) calendar observations.  For the separated-batch
covariance they enter only through the fixed integer \(q\) in the definition
of \(M_N\).

The stopped batch count obeys
\begin{equation}\label{eq:calendar-stopped-batch-count-main}
M_N=\frac{K_N}{d_N}+O(1)
=\frac{N}{\mu d_N}+O_{\PP}\!\left(\frac{\sqrt N}{b_N}+1\right).
\end{equation}
Consequently \(M_N\asymp_{\PP}N/b_N\), so \(M_N\ge2\) with probability
tending to one and \(M_N^{-1/2}=O_{\PP}(\sqrt{b_N/N})\).

The retained batches are independent.  More explicitly, \(B_{\ell,N}\)
depends only on
\[
G_{(\ell-1)d_N},\ldots,G_{(\ell-1)d_N+b_N-1+m},
\]
and these gap sets are disjoint for different \(\ell\).  Let
\[
V_N=b_N^{-1}\Var(B_{1,N}),\qquad
X_{\ell,N}=b_N^{-1}\{B_{\ell,N}B_{\ell,N}^\top
-\EE(B_{\ell,N}B_{\ell,N}^\top)\}.
\]
As in \Cref{thm:regenerative-separated-batch-covariance-main},
\begin{equation}\label{eq:calendar-stopped-batch-variance-main}
V_N=\mathcal T_m+O(b_N^{-1}),\qquad
\EE\|B_{\ell,N}\|^4\le Cb_N^2,\qquad
\EE\|X_{\ell,N}\|_{\operatorname F}^2\le C .
\end{equation}

The point needing care is that \(M_N\) is random and is determined by the same
gaps used to decide whether a terminal batch is complete.  The proof avoids
optional-stopping shorthand by controlling all deterministic prefixes at once.
For each matrix entry, the independent centered variables
\(\{X_{\ell,N}^{ab}\}_{\ell\ge1}\) have uniformly bounded second moments by
\eqref{eq:calendar-stopped-batch-variance-main}.  Kolmogorov's maximal
inequality therefore gives, for every fixed \(C_1<\infty\),
\[
\EE\max_{1\le r\le C_1N/b_N}
\left|\sum_{\ell=1}^rX_{\ell,N}^{ab}\right|^2
\le C\,\frac{N}{b_N}.
\]
Summing over the finitely many matrix entries yields
\[
\max_{1\le r\le C_1N/b_N}
\left\|\sum_{\ell=1}^rX_{\ell,N}\right\|_{\operatorname F}
=O_{\PP}\!\left(\sqrt{\frac{N}{b_N}}\right).
\]
By \eqref{eq:calendar-stopped-batch-count-main}, for any small
\(\epsilon>0\) there are constants \(0<c_1<C_1<\infty\) such that
\[
\PP\!\left(c_1\frac{N}{b_N}\le M_N\le C_1\frac{N}{b_N}\right)>1-\epsilon
\]
for all sufficiently large \(N\).  On this event the random stopped prefix is
one of the deterministic prefixes already controlled, so
\begin{equation}\label{eq:calendar-stopped-random-prefix-fluctuation-main}
\left\|
\frac1{M_N}\sum_{\ell=1}^{M_N}X_{\ell,N}
\right\|_{\operatorname{op}}
=O_{\PP}\left(\sqrt{\frac{b_N}{N}}\right).
\end{equation}
Thus
\[
\frac1{b_NM_N}\sum_{\ell=1}^{M_N}B_{\ell,N}B_{\ell,N}^\top
=V_N+O_{\PP}\left(\sqrt{\frac{b_N}{N}}\right).
\]
The empirical mean subtraction is of the same or smaller order.  The stopped
martingale \(R_{r,N}=\sum_{\ell=1}^rB_{\ell,N}\) satisfies
\(\EE\|B_{\ell,N}\|^2\le Cb_N\), and the same maximal argument gives
\[
\frac1{b_N}\bar B_N\bar B_N^\top
=O_{\PP}(M_N^{-1})
=O_{\PP}\left(\frac{b_N}{N}\right).
\]
Since \(M_N/(M_N-1)=1+O_{\PP}(b_N/N)\), the centered cycle-time covariance is
\[
\frac1{b_N(M_N-1)}
\sum_{\ell=1}^{M_N}(B_{\ell,N}-\bar B_N)
(B_{\ell,N}-\bar B_N)^\top
=\mathcal T_m+
O_{\PP}\left(b_N^{-1}+\sqrt{\frac{b_N}{N}}\right).
\]
Multiplying by \(K_N/N=\mu^{-1}+O_{\PP}(N^{-1/2})\) in
\eqref{eq:calendar-stopped-k-rate-main} proves
\eqref{eq:calendar-stopped-separated-batch-rate-main}.

It remains to check that the plug-in rewards used in Algorithm~1 do not change
the rate.  A Chernoff bound for the exponential-moment variables \(L_j\) gives
\begin{equation}\label{eq:calendar-stopped-uniform-block-length-main}
\max_{1\le\ell\le M_N}
\frac1{b_N}\sum_{j\in I_{\ell,N}}L_j=O_{\PP}(1).
\end{equation}
Indeed, after intersecting with \(M_N\le C_1N/b_N\), the union bound is
\[
\frac{C_1N}{b_N}\,
\PP\!\left(\sum_{j=1}^{b_N}L_j>Cb_N\right),
\]
which tends to zero for \(C\) large because \(b_N=N^\beta\).  Since
\(\Delta_N=O_{\PP}(N^{-1/2})\),
\eqref{eq:calendar-stopped-uniform-block-length-main} proves
\eqref{eq:calendar-stopped-plugin-block-main}.  Put
\[
E_{\ell,N}=\widehat B_{\ell,N}-B_{\ell,N}
=-\Delta_N\sum_{j\in I_{\ell,N}}L_j,\qquad
\bar E_N=M_N^{-1}\sum_{\ell=1}^{M_N}E_{\ell,N}.
\]
On a high-probability event,
\(\max_{\ell\le M_N}\|E_{\ell,N}\|\le Cb_NN^{-1/2}\).  The difference between
the centered plug-in and oracle cycle-time covariance matrices is bounded by
\[
\begin{aligned}
&\frac{2}{b_N(M_N-1)}
\sum_{\ell=1}^{M_N}\|B_{\ell,N}-\bar B_N\|\,\|E_{\ell,N}-\bar E_N\|\\
&\qquad
+\frac{1}{b_N(M_N-1)}
\sum_{\ell=1}^{M_N}\|E_{\ell,N}-\bar E_N\|^2 .
\end{aligned}
\]
The oracle centered covariance just proved has bounded operator norm in
probability, hence
\[
\frac1{b_N(M_N-1)}
\sum_{\ell=1}^{M_N}\|B_{\ell,N}-\bar B_N\|^2=O_{\PP}(1).
\]
Also
\(\sum_{\ell\le M_N}\|E_{\ell,N}-\bar E_N\|^2
=O_{\PP}(M_Nb_N^2/N)\).  Cauchy's inequality therefore makes the cross term
\(O_{\PP}(\sqrt{b_N/N})\), while the quadratic plug-in term is
\(O_{\PP}(b_N/N)\).  Since \(b_N/N\le\sqrt{b_N/N}\) eventually and
\(K_N/N=O_{\PP}(1)\), the plug-in covariance has the same rate as the oracle
covariance.  This proves
\eqref{eq:calendar-stopped-plugin-covariance-rate-main}.
\end{proof}

\begin{proposition}[Standard complete-cycle and separated-batch facts]\label{prop:standard-complete-cycle-separated-batch-facts-main}
Let \((A_n)_{n\in\mathbb Z}\) be a stationary binary renewal process with
click epochs \((\tau_j)\), i.i.d. gaps \(G_j\), cycle lengths
\(L_j=G_j+1\), mean \(\mu=\EE L_0\in(0,\infty)\), and
\(\EE e^{\eta G_0}<\infty\) for some \(\eta>0\).  Let
\(K_N=\max\{k:\tau_k<N\}\), with \(K_N=-1\) if no click is observed.  The
following standard regenerative conclusions are used in the main text
\cite{GlynnIglehart1990STS,AsmussenGlynn2007,GlynnWhitt1991BatchMeans,
VatsFlegalJones2019Multivariate}.
\begin{enumerate}[label=\textup{(\alph*)},leftmargin=2.4em,itemsep=0.25em]
\item Every bounded local window
\[
Z_n=\psi(A_{n-r},\ldots,A_{n+\ell}),\qquad \bar z=\EE Z_0,
\]
has a complete-cycle reward
\[
U_j=\sum_{n=\tau_j}^{\tau_{j+1}-1}Z_n,\qquad
Y_j=U_j-\bar zL_j,
\]
with Palm centering
\begin{equation}\label{eq:local-window-palm-inversion-main}
\EE U_j=\mu\bar z,\qquad \EE Y_j=0 .
\end{equation}
Moreover \(Y_j\) is a measurable function of finitely many adjacent gaps and
is therefore fixed-range dependent.  Endpoint and fixed look-ahead/look-back
trimming changes the calendar sum by
\begin{equation}\label{eq:local-window-endpoint-main}
\sum_{n=0}^{N-1}Z_n-\sum_j U_j=O_{\PP}(\log N),
\end{equation}
where the cycle sum is over the complete cycles intersecting the observed
calendar window, with the same \(O_{\PP}(\log N)\) bound after any fixed
endpoint trimming.

\item If \(Y_j=f(G_j,\ldots,G_{j+m_{\rm dep}})\in\RR^{d_{\rm rew}}\),
\(\EE Y_j=0\), and
\[
\|f(g_0,\ldots,g_{m_{\rm dep}})\|
\le C_f\left(1+\sum_{i=0}^{m_{\rm dep}}g_i\right),
\]
then, for any all-complete-cycle index set \(J_N\) obtained from
\(\{0,\ldots,K_N-1\}\) by deleting only fixed endpoint/look-ahead indices,
\[
\sqrt N\,\frac1N\sum_{j\in J_N}Y_j
\Rightarrow
\mathcal N\left(0,\frac{\mathcal T_{m_{\rm dep}}}{\mu}\right),
\]
where
\[
\mathcal T_{m_{\rm dep}}
=\Var(Y_0)+
\sum_{h=1}^{m_{\rm dep}}
\{\operatorname{Cov}(Y_0,Y_h)+\operatorname{Cov}(Y_h,Y_0)\}.
\]

\item For \(b_N=\lfloor N^\beta\rfloor\), \(0<\beta<1\),
\(d_N=b_N+m_{\rm dep}\), and separated batches
\[
I_{\ell,N}:=\{(\ell-1)d_N,\ldots,(\ell-1)d_N+b_N-1\},
\qquad
C_{\ell,N}=\sum_{j=(\ell-1)d_N}^{(\ell-1)d_N+b_N-1}Y_j ,
\]
let \(M_N\) be the number of all-complete-cycle batches retained before
calendar time \(N\), and put
\(\bar C_N=M_N^{-1}\sum_{\ell=1}^{M_N}C_{\ell,N}\) on \(M_N\ge2\).  Then the
retained batch covariance satisfies
\begin{equation}\label{eq:separated-batch-covariance-estimator-rate-main}
\left\|
\frac{K_N}{N}\frac1{b_N(M_N-1)}
\sum_{\ell=1}^{M_N}(C_{\ell,N}-\bar C_N)(C_{\ell,N}-\bar C_N)^\top
-\frac{\mathcal T_{m_{\rm dep}}}{\mu}
\right\|_{\operatorname{op}}
=O_{\PP}\left(b_N^{-1}+\sqrt{b_N/N}\right).
\end{equation}
Equivalently, before multiplying by \(K_N/N\), the cycle-time sample covariance
estimates \(\mathcal T_{m_{\rm dep}}\) at the same rate.

\item If plug-in rewards have the form
\[
\widehat Y_j=Y_j-(\widehat r_N-r)L_j,\qquad
\widehat r_N-r=O_{\PP}(N^{-1/2}),
\]
and \(b_N\) is polynomial, then
\begin{equation}\label{eq:separated-batch-plugin-block-main}
\max_{\ell\le M_N}
\left\|\sum_{j\in I_{\ell,N}}(\widehat Y_j-Y_j)\right\|
=O_{\PP}(b_NN^{-1/2})
\end{equation}
and replacing \(Y_j\) by \(\widehat Y_j\) in the separated-batch covariance
changes it by
\begin{equation}\label{eq:separated-batch-plugin-covariance-main}
O_{\PP}\!\left(\sqrt{\frac{b_N}{N}}\right)
+O_{\PP}\!\left(\frac{b_N}{N}\right).
\end{equation}

\item For every fixed \(C\), \(\PP(K_N<C)\to0\), and the same conclusion holds
after deleting any fixed number of endpoint, look-back, or look-ahead cycles.

\item If \(b_N=\lfloor N^\beta\rfloor\), \(0<\beta<1\), and
\(M_N=O_{\PP}(N/b_N)\), then
\begin{equation}\label{eq:uniform-calendar-block-length-main}
\max_{1\le \ell\le M_N}
\frac1{b_N}\sum_{j\in I_{\ell,N}}L_j
=O_{\PP}(1).
\end{equation}
\end{enumerate}
\end{proposition}

\begin{proof}
The only model-specific input to this proposition is that the click epochs
form a stationary renewal process with an exponential gap moment.  The
Palm-cycle identity gives
\[
\EE Z_0=\frac{\EE U_j}{\EE L_j},\qquad
U_j=\sum_{n=\tau_j}^{\tau_{j+1}-1}Z_n,
\]
and hence \(\EE U_j=\mu\bar z\) and \(\EE Y_j=0\).  Because \(Z_n\) is a fixed
local window, \(Y_j\) depends on only finitely many adjacent gaps; call this
range \(m_{\rm dep}\).  Exponential tails imply that the largest cycle
intersecting any fixed enlargement of \([0,N-1]\) is \(O_{\PP}(\log N)\), so
endpoint, look-ahead, and look-back deletions give
\eqref{eq:local-window-endpoint-main}.

The CLT and the calendar-time scale are the standard regenerative-reward
facts: apply the ordinary finite-range-dependent CLT to the stationary
sequence \(Y_j=f(G_j,\ldots,G_{j+m_{\rm dep}})\), then use the renewal law of
large numbers \(K_N/N\to1/\mu\) and the usual random-index replacement
\cite[Chs.~IV.3 and VI.3]{AsmussenGlynn2007}.  This gives the covariance
\(\mathcal T_{m_{\rm dep}}/\mu\).

For the separated-batch estimator, apply
\Cref{prop:calendar-stopped-separated-batch-plugin-main} to the fixed-range
gap reward \(Y_j=f(G_j,\ldots,G_{j+m_{\rm dep}})\), with \(q\) equal to the
fixed number of endpoint and look-ahead cycles deleted by the particular
complete-cycle set.  That proposition proves the random calendar-stopped
batch count bound
\(M_N=K_N/(b_N+m_{\rm dep})+O_{\PP}(\sqrt N/b_N+1)\), controls the maximal
deleted endpoint cycle, subtracts the empirical batch mean, and gives exactly
\eqref{eq:separated-batch-covariance-estimator-rate-main}.

For the plug-in perturbation, write
\(\Delta_N=\widehat r_N-r=O_{\PP}(N^{-1/2})\).  Each batch changes by
\[
\widehat C_{\ell,N}-C_{\ell,N}
=-\Delta_N\sum_{j\in I_{\ell,N}}L_j .
\]
\Cref{prop:calendar-stopped-separated-batch-plugin-main} gives the uniform
block-length bound, the block perturbation
\eqref{eq:separated-batch-plugin-block-main}, and the covariance perturbation
\eqref{eq:separated-batch-plugin-covariance-main}.  Finally,
\((K_N+1)/N\to1/\mu\) gives the fixed-low-cycle statement, and the preceding
uniform block-length bound is exactly \eqref{eq:uniform-calendar-block-length-main}.
\end{proof}

\subsection{Finite-domain diagonal certification for implementation extensions}
\label{app:finite-domain-diagonal-certification-main}

The submitted manuscript uses only the point certificate at \(z=1/2\).  The
following interval-cover verifier is retained as implementation machinery for a
future extension that prints additional point or interval transcripts.  It is
not a paper-level theorem claim for any candidate outside \(z=1/2\).

\begin{algorithmiclemma}[Finite interval-cover diagonal branch certification]\label{lem:diagonal-certified-interval-cover-main}
Let \(I=[z^-,z^+]\Subset(0,1)\) be a compact diagonal candidate interval that
is reported as coverage-certified in an implementation extension.  The
deterministic branch verifier is as follows.

\begin{enumerate}[label=\textup{(\arabic*)},leftmargin=2.4em,itemsep=0.25em]
\item Start with the queue containing \(I\) and with an empty certified list
\(\mathcal C\).
\item Remove one interval \(J\) from the queue.  Apply the outward-rounded
evaluation in \Cref{lem:diagonal-interval-branch-certificate-main} on the
whole interval \(J\).
\item If exactly one of the four branch tests in
\Cref{def:dmap-diagonal-rank-certificate-main} succeeds on \(J\), append
\((J,\mathrm{branch})\) to \(\mathcal C\).
\item If no branch test succeeds and the prescribed maximum subdivision depth,
precision budget, or exact-symbolic fallback has not been exhausted, bisect
\(J\) and return the two children to the queue.
\item If no branch test succeeds after the declared resources are exhausted,
append \(J\) to the undecided list \(\mathcal U\).
\end{enumerate}

The extension transcript contains the input interval \(I\), the precision and
rounding mode, every terminal interval, the branch label and margin for every
interval in \(\mathcal C\), the undecided list \(\mathcal U\), and an exact
rational endpoint check proving
\[
I=\bigcup_{(J,\mathrm{branch})\in\mathcal C}J\ \cup\
\bigcup_{J\in\mathcal U}J
\]
with disjoint interiors.  The interval \(I\) is a certified extension interval
only when this verifier returns a finite terminal list and
\(\mathcal U=\varnothing\).  In that case
\[
I=\bigcup_{j=1}^M I_j
\]
and directed interval evaluation proves exactly one branch condition on each
\(I_j\).  Therefore every candidate \(z\in I\) belongs to a certified
subinterval on which the population branch of
\(\Sigma_\sigma(z)\) is fixed.  If \(\mathcal U\ne\varnothing\), the union of
the undecided intervals is displayed and those candidates remain diagnostic.
\end{algorithmiclemma}

\begin{proof}
All branch assertions made on a terminal interval are consequences of
\Cref{lem:diagonal-interval-branch-certificate-main}, so each accepted
terminal interval has a proved population branch for every \(z\) in that
interval.  The four branch outcomes in
\Cref{def:dmap-diagonal-rank-certificate-main} are mutually exclusive for the
actual nonnegative definite matrix \(\Sigma_\sigma(z)\) by
\Cref{lem:dmap-diagonal-covariance-alternatives-main}.  Hence two distinct
branch labels cannot both be valid at the same population point; the verifier
therefore accepts a terminal interval only when its directed enclosures prove a
single outcome uniformly over the interval.

The equality of the endpoint union is a finite rational check in the transcript,
independent of floating-point arithmetic.  When the undecided list is empty,
the accepted terminal intervals cover all of \(I\), and every point in \(I\)
inherits the unique branch proved on the terminal interval containing it.  If
the list is nonempty, no coverage-certified extension is invoked on that
undecided subunion.
\end{proof}

\subsection{Diagonal branch certificates}\label{app:diagonal-branch-certificate-protocol-main}

This subsection contains the finite branch-certificate convention and
outward-rounded verifier used by Theorem~E.  It is implementation certificate
machinery for the analytic covariance alternatives of
\Cref{lem:dmap-diagonal-covariance-alternatives-main}.

\begin{definition}[Certified diagonal covariance branch]\label{def:dmap-diagonal-rank-certificate-main}
Fix \(0<z<1\) and form the actual D-MAP covariance
\(\Sigma_\sigma(z)\) by \eqref{eq:dmap-diagonal-sigma-finite-main}.  Put
\[
n_z=(-z,1)^\top,\qquad t_z=(1,z)^\top .
\]
For each reported diagonal candidate \(z\), a diagonal branch certificate at
\(z\) is the following finite object.
It contains:
\begin{enumerate}[label=\textup{(\arabic*)},leftmargin=2.4em,itemsep=0.2em]
\item an exact description of the input \(z\), or an interval enclosure
\([z]\Subset(0,1)\) if the reported branch is asserted uniformly on that
input interval;
\item an outward-rounded enclosure \([\theta]\) of
\(\theta=-\log z\), or a symbolic identity for \(\theta\);
\item outward-rounded interval enclosures for every scalar arithmetic operation
in \eqref{eq:dmap-diagonal-sigma-r-finite-main} and
\eqref{eq:dmap-diagonal-sigma-finite-main};
\item the three outward-rounded branch quantities
\[
[I_{\tau}]\supset \operatorname{tr}\Sigma_\sigma(z),\qquad
[I_{\det}]\supset \det\Sigma_\sigma(z),\qquad
[w]\supset \Sigma_\sigma(z)t_z,
\]
computed from the enclosed matrix by the directed-rounding formulas in
\Cref{lem:diagonal-interval-branch-certificate-main}.
\item in the rank-one non-normal branch, an outward-rounded enclosure
\[
[a_{\rm rn}]
\supset
\left(
1-\frac{n_z^\top\Sigma_\sigma(z)n_z}
{\operatorname{tr}\Sigma_\sigma(z)\,\|n_z\|^2}
\right)^2
\]
of the scalar projected-line scale used in
\Cref{lem:certified-rank-one-nonnormal-projected-gaussian-quantile-main}.
\end{enumerate}
The certificate transcript printed in the manuscript, or an explicitly
included appendix transcript, must specify the precision, rounding mode, and
the displayed branch margin \(\eta>0\) whenever it asserts a strict positive
inequality.  A repository script may reproduce the transcript, but the
coverage-certified status is attached to the printed interval transcript, not
to an undisplayed local file.  Equalities used to assert a rank-deficient
branch must be exact equalities, either symbolic identities or interval
enclosures of width zero in the certified arithmetic; a small floating-point
residual is not an equality certificate.

The four admissible certified outcomes are:
\begin{enumerate}[label=\textup{(\alph*)},leftmargin=2.4em,itemsep=0.25em]
\item \textbf{Full rank.}  The certificate proves
\[
[I_{\det}]\subset[\eta,\infty).
\]
\item \textbf{Rank one, normal range.}  The certificate proves
\[
[I_{\det}]=[0,0],\qquad
[I_{\tau}]\subset[\eta,\infty),\qquad
[w]=\{(0,0)^\top\}.
\]
\item \textbf{Rank one, non-normal range.}  The certificate proves
\[
[I_{\det}]=[0,0],\qquad
[I_{\tau}]\subset[\eta,\infty),\qquad
[q]\subset[\eta,\infty),
\]
where \([q]\) is the outward-rounded enclosure of
\(\|w\|^2=w_1^2+w_2^2\) obtained from \([w]\), and also supplies
\([a_{\rm rn}]\Subset(0,1]\).
\item \textbf{Zero covariance.}  The certificate proves
\[
\Sigma_\sigma(z)=0
\]
entrywise, equivalently \(\operatorname{tr}\Sigma_\sigma(z)=0\) for this
nonnegative definite matrix.
\end{enumerate}
If none of these certificates is available, the diagonal rank branch is
declared inconclusive.  An inconclusive branch may still be used to display the
projected-Gaussian diagnostic \(Q_N(z)\) and the ideal conditional quantile,
but it is not a coverage-certified report in Theorem~E.

The certification convention is deliberately finite.  This paper does not use
an unproved global analytic sign classification of
\(\det\Sigma_\sigma(z)\) on \(0<z<1\), and a floating-point eigenvalue
calculation at a candidate is not a substitute for a certificate.  The worked
certificate in \Cref{rem:worked-diagonal-branch-certificate-main} supplies only
the single submitted candidate \(z=1/2\); no continuum branch classification
and no neighbouring-point certificate is inferred from it.  The finite-domain
interval-cover machinery needed for a later implementation extension is not
part of the paper-level theorem and is recorded in
\Cref{app:finite-domain-diagonal-certification-main}.
\end{definition}

\begin{algorithmiclemma}[Outward-rounded diagonal branch certificate]\label{lem:diagonal-interval-branch-certificate-main}
For a fixed point \(0<z<1\), or for a closed input interval
\([z]\Subset(0,1)\), the branch certificate in
\Cref{def:dmap-diagonal-rank-certificate-main} is reproducibly obtained from
the finite formulas of \Cref{prop:computable-diagonal-covariance-main} as
follows.
Work in an interval arithmetic system with directed rounding.  Enclose the
input by \([z]\), enclose \(\theta=-\log z\) over that whole input enclosure by
\([\theta]\), and then evaluate the displayed formulas
\[
[g_0]=(1-[z])-[\theta][z],
\qquad
[g_1]=[z]\{(1-[z])(1+[\theta])-[\theta][z]\},
\]
\[
[\mu]=\frac1{1-[z]}+\frac{[\theta][z]}{(1-[z])^2},
\qquad
[\nu]=\frac{1+[z]}{(1-[z])^2}
     +\frac{[\theta][z](3+[z])}{(1-[z])^3},
\qquad
[m_2]=[g_1]+[g_0]^2.
\]
Set
\[
[r]=\frac1{[\mu]}(1,[g_0],[m_2])^\top
\]
and form the interval matrices \([A_z]\), \([B_z]\) and interval vectors
\([e_z]\), \([d_z]\) from the displayed formulas immediately preceding
\eqref{eq:dmap-diagonal-sigma-r-finite-main}.  Directed evaluation of
\[
[\Sigma_r]=\frac1{[\mu]}\left\{
[A_z]-[e_z][r]^\top-[r][e_z]^\top+[\nu][r][r]^\top
+[B_z]-[d_z][r]^\top+[B_z]^\top-[r][d_z]^\top
\right\}
\]
encloses the regenerative covariance.
Next compute the quotient-Jacobian interval \([J_\Phi]\) from
\eqref{eq:lambda-gradient-main}--\eqref{eq:phi-jacobian-main}, with every
division checked to have a denominator interval not containing zero, and put
\[
[\Sigma]=[J_\Phi]\,[\Sigma_r]\,[J_\Phi]^\top .
\]
Writing
\[
[\Sigma]=\begin{pmatrix}
[a]&[b]\\
[c]&[d]
\end{pmatrix},
\]
the branch quantities required by \Cref{def:dmap-diagonal-rank-certificate-main}
are not optional diagnostics but the certificate data:
\begin{equation}\label{eq:diagonal-branch-interval-quantities-main}
[I_{\tau}]=[a]+[d],\qquad
[I_{\det}]=[a][d]-[b][c],\qquad
[w]=
\binom{[a]+[z][b]}{[c]+[z][d]},
\qquad
[q]=[w_1]^2+[w_2]^2 .
\end{equation}
For a rank-one non-normal outcome, also compute the scalar-law enclosure
\begin{equation}\label{eq:diagonal-rank-one-nonnormal-scale-certificate-main}
[a_{\rm rn}]
=\left(
1-\frac{[n]^\top[\Sigma][n]}{[I_{\tau}]\,([z]^2+1)}
\right)^2,
\qquad [n]=(-[z],1)^\top .
\end{equation}
If the intervals in \eqref{eq:diagonal-branch-interval-quantities-main}
satisfy one of the four certified outcomes in
\Cref{def:dmap-diagonal-rank-certificate-main}, then the corresponding
population branch of \(\Sigma_\sigma(z)\) is proved for the fixed candidate,
or uniformly for every candidate in the certified input interval.  If none of
the interval separations succeeds, the branch is reported as inconclusive.
\end{algorithmiclemma}

\begin{proof}
The formulas for \(g_0,g_1,\mu,\nu,m_2,r,A_z,e_z,d_z,B_z\) are exactly the
finite D-MAP covariance formulas of
\Cref{prop:computable-diagonal-covariance-main}.  Directed interval arithmetic
is inclusion isotone for addition, subtraction, multiplication, and division
by an interval excluding zero.  Therefore each displayed interval contains the
corresponding real scalar or matrix entry.  The quotient denominator is
nonzero at every fixed diagonal point by
\eqref{eq:diagonal-quotient-denominator-main}, so sufficiently sharp input
enclosures make the Jacobian divisions certified.  The sandwich product then
contains the real covariance
\(\Sigma_\sigma(z)=D\Phi(r(z))\Sigma_r(z)D\Phi(r(z))^\top\).  Finally,
\eqref{eq:diagonal-branch-interval-quantities-main} is precisely the directed
interval evaluation of
\(\operatorname{tr}\Sigma_\sigma(z)\), \(\det\Sigma_\sigma(z)\),
\(\Sigma_\sigma(z)(1,z)^\top\), and its squared norm.  Interval exclusion from
zero proves the stated strict inequalities, while a zero-width interval or a
symbolic identity proves the asserted equalities.  The branch conclusion is
then the \(2\times2\) nonnegative-definite rank/range classification of
\Cref{lem:dmap-diagonal-covariance-alternatives-main}.  In the rank-one
non-normal case, write
\(\Sigma_\sigma(z)=\lambda vv^\top\), \(\lambda>0\), \(\|v\|=1\).  Since
\(\operatorname{tr}\Sigma_\sigma(z)=\lambda\),
\[
\frac{n_z^\top\Sigma_\sigma(z)n_z}
{\operatorname{tr}\Sigma_\sigma(z)\|n_z\|^2}
=\frac{(n_z^\top v)^2}{\|n_z\|^2},
\]
so \eqref{eq:diagonal-rank-one-nonnormal-scale-certificate-main} encloses the
scale \(a(z,v)\) in
\Cref{lem:certified-rank-one-nonnormal-projected-gaussian-quantile-main}.
\end{proof}

\begin{lemma}[D-MAP diagonal covariance alternatives]\label{lem:dmap-diagonal-covariance-alternatives-main}
Work at a fixed constrained D-MAP exchange-diagonal population point
\(\sigma_z=(2z,z^2)\), \(0<z<1\), and form the actual diagonal covariance
\(\Sigma_\sigma(z)\) by the finite formula
\eqref{eq:dmap-diagonal-sigma-finite-main}.  Let \(n_z=(-z,1)^\top\).  Exactly
one of the following mutually exclusive
alternatives holds:
\[
\operatorname{rank}\Sigma_\sigma(z)=2,
\]
\[
\operatorname{rank}\Sigma_\sigma(z)=1
\quad\text{and}\quad
\operatorname{Range}\Sigma_\sigma(z)\ne\operatorname{span}\{n_z\},
\]
\[
\operatorname{rank}\Sigma_\sigma(z)=1
\quad\text{and}\quad
\operatorname{Range}\Sigma_\sigma(z)=\operatorname{span}\{n_z\},
\]
or
\[
\Sigma_\sigma(z)=0 .
\]
The first two alternatives have a continuous projected-Gaussian statistic and
therefore supply exact-level projected-Gaussian cases for all confidence
levels.  In the third alternative the limiting statistic has law
\(\frac12\delta_0+\frac12\chi^2_1\): its distribution function is
\(1/2+F_{\chi^2_1}(t)/2\) for \(t>0\).  Hence its lower \(p_c\)-quantile is
zero for \(p_c\le1/2\), but for \(p_c>1/2\) it is
\(F_{\chi^2_1}^{-1}(2p_c-1)\) and is continuous.  The
fourth alternative has an identically zero limiting statistic.  The occurrence
of the covariance alternative is not a modelling choice: it is obtained by
evaluating the finite matrix
\eqref{eq:dmap-diagonal-sigma-finite-main}.  The paper does not prove a
closed-form analytic sign classification of this determinant for every
\(0<z<1\).  Therefore the operational diagonal coverage branches in
Theorem~E are conditional on the certified population branch in
\Cref{def:dmap-diagonal-rank-certificate-main}, operationally generated by
\Cref{lem:diagonal-interval-branch-certificate-main}, not on a global
closed-form \(z\)-classification and not on an uncertified floating-point rank
label.  In
the rank-one-normal covariance
alternative, the theorem-level branch still depends on the requested quantile
level \(p_c=1-\alpha\): exact unbuffered coverage for \(p_c>1/2\),
and buffered conservative coverage for \(p_c\le1/2\).

Equivalently, because \(\Sigma_\sigma(z)\) is \(2\times2\) and nonnegative
definite, these cases are checked directly from the actual D-MAP covariance by
\[
\operatorname{rank}\Sigma_\sigma(z)=2
\quad\Longleftrightarrow\quad
\det\Sigma_\sigma(z)>0,
\]
\[
\Sigma_\sigma(z)=0
\quad\Longleftrightarrow\quad
\operatorname{tr}\Sigma_\sigma(z)=0,
\]
and, with \(t_z=(1,z)^\top\) tangent to the exchange diagonal,
\[
\operatorname{Range}\Sigma_\sigma(z)=\operatorname{span}\{n_z\}
\quad\Longleftrightarrow\quad
\det\Sigma_\sigma(z)=0,\quad
\operatorname{tr}\Sigma_\sigma(z)>0,\quad
\Sigma_\sigma(z)t_z=0 .
\]
The remaining case
\[
\det\Sigma_\sigma(z)=0,\quad
\operatorname{tr}\Sigma_\sigma(z)>0,\quad
\Sigma_\sigma(z)t_z\ne0
\]
is the rank-one non-normal exact-level alternative.
\end{lemma}

\begin{proof}
The matrix \(\Sigma_\sigma(z)\) is a \(2\times2\) nonnegative definite matrix
because it is obtained by sandwiching the nonnegative definite regenerative
covariance \(\Sigma_r(z)\) with the deterministic Jacobian \(D\Phi(r(z))\).
Thus its rank is \(0\), \(1\), or \(2\).  In rank one its range is a single
line, which is either the normal line \(\operatorname{span}\{n_z\}\) or is not
that line.  These alternatives are exhaustive and mutually exclusive.

For a \(2\times2\) nonnegative definite matrix, full rank is equivalent to
positive determinant, and the zero matrix is equivalent to zero trace.  In the
remaining rank-one case, \(\det\Sigma_\sigma(z)=0\) and
\(\operatorname{tr}\Sigma_\sigma(z)>0\).  Since
\(t_z=(1,z)^\top\) spans the tangent line orthogonal to \(n_z=(-z,1)^\top\),
the range is \(\operatorname{span}\{n_z\}\) exactly when the kernel is the
tangent line, equivalently \(\Sigma_\sigma(z)t_z=0\).  If this kernel test
fails, the rank-one range is not the normal line.  The statements about
continuity, the half-mass at zero, and the zero statistic are precisely
\Cref{thm:projected-gaussian-continuity-degeneracy-main} applied to
\(\Sigma=\Sigma_\sigma(z)\).  The certificate definition is only the same
case analysis written as a reproducible branch-selection rule: each positive
inequality gives the displayed margin, and each asserted rank deficiency is
accepted only when the corresponding equality has been proved exactly.
\end{proof}

\subsection{Finite projected-Gaussian certificates}\label{app:finite-projected-gaussian-certificates-main}

This subsection contains the mechanical quantile brackets, midpoint-undecided
outputs, DKW simulation domination, and finite diagonal report conventions
cross-referenced from Theorem~E.  These are executable certificate protocols,
not additional paper-level theorem targets.

\begin{algorithmiclemma}[Certified deterministic projected-Gaussian quantile bracket]\label{lem:certified-deterministic-projected-gaussian-quantile-main}
Fix \(0<z<1\), \(0<p<1\), and a positive definite \(2\times2\) covariance
matrix \(\Sigma\).  Suppose the data used by the numerical report are given by
outward-rounded interval enclosures \([z]\), \([\Sigma]\), and \([Q]\), where
\([Q]\) contains the computed statistic \(Q\).  Then the lower quantile
\[
c_p(z,\Sigma)=\inf\{c\ge0:F_{z,\Sigma}(c)\ge p\}
\]
from \Cref{lem:deterministic-full-rank-projected-gaussian-quantile-main} has
a reproducible deterministic \emph{bracket} certificate of the following form.
The routine certifies endpoint containment and leaves unresolved midpoint
comparisons explicitly undecided.  For
any requested abscissa tolerance \(\eta_c>0\), probability tolerance
schedule \((\eta_k)_{k\ge0}\) of positive rationals with \(\eta_k\downarrow0\),
and finite midpoint refinement budget \(J\ge1\), the outward-rounded
computation terminates after finitely many rational split points and returns
one of two transcripts.

The first, called a \emph{resolved bracket}, consists of numbers
\[
L<U,\qquad U-L\le\eta_c,
\]
and certified distribution-function enclosures at the two endpoints satisfying
\[
F^+(L)<p\le F^-(U).
\]
The second, called an \emph{undecided but bracketed midpoint}, consists of a
current bracket \(L<U\), a rational split point \(m\in(L,U)\), and certified
distribution-function enclosures satisfying
\[
F^+(L)<p\le F^-(U),\qquad
F^-(m)<p\le F^+(m).
\]
In either case the transcript certifies the quantile bracket
\[
L<c_p(z,\Sigma)\le U.
\]
The undecided transcript is a legitimate finite output: it records that the
available outward-rounded calculation has not decided on which side of the
ideal lower quantile the split point lies, and it does not assert equality or
force an accept/reject decision at \(m\).

Moreover any resolved or undecided bracket gives a rigorous comparison of
\(Q\) with the ideal quantile whenever the certified intervals are separated:
\[
[Q]^+\le L\quad\Longrightarrow\quad Q<c_p(z,\Sigma),
\]
and
\[
[Q]^->U\quad\Longrightarrow\quad Q>c_p(z,\Sigma).
\]
If neither separation holds, the routine returns the certified statement
\[
Q-c_p(z,\Sigma)\in
\bigl[[Q]^--U,\,[Q]^+-L\bigr],
\]
so the only undecided cases are those lying within the declared numerical
quantile/statistic enclosure.

The computation is the following one.  First compute an outward-rounded
Cholesky enclosure \([C]\) for \(\Sigma=C C^\top\), together with interval
enclosures for \(\Sigma^{-1}\).  Write \(Y\sim\mathcal N(0,I_2)\) and
\(Z=CY\).  For a requested CDF tolerance \(\eta>0\), choose \(R=R(\eta)>0\)
so that the certified Mills-ratio bound
\begin{equation}\label{eq:certified-gaussian-tail-truncation-main}
4\,\frac{\exp(-R^2/2)}{R\sqrt{2\pi}}\le\eta_{\rm tail}
\end{equation}
holds, with \(0<\eta_{\rm tail}<\eta/4\).  Then
\[
\PP\{\|Y\|_\infty>R\}\le\eta_{\rm tail},
\]
so all probability outside the square \(S_R=[-R,R]^2\) is charged only to the
upper distribution-function bound.

For a candidate value \(c\ge0\), partition \(S_R\) into finitely many closed
axis-parallel boxes \(B\).  On each box use directed interval arithmetic to
enclose
\[
s_B=n_{[z]}^\top [C]B,\qquad
q_B^{\rm in}=([C]B)^\top[\Sigma^{-1}]([C]B),
\]
and, for the exterior projected region,
\[
[P_z u]
=u-\frac{(u_2-[z]u_1)_+}{1+[z]^2}\binom{-[z]}{1},
\qquad
q_B^{\rm out}=([P_z C B])^\top[\Sigma^{-1}]([P_z C B]).
\]
The lower sum \(F^-(c)\) adds the exact product-normal mass
\[
\gamma(B)=
\{\Phi(b_1)-\Phi(a_1)\}\{\Phi(b_2)-\Phi(a_2)\},
\qquad B=[a_1,b_1]\times[a_2,b_2],
\]
for every box certified to lie in the first projected region and satisfy
\((q_B^{\rm in})^+\le c\), and for every box certified to lie in the second
projected region and satisfy \((q_B^{\rm out})^+\le c\).  The upper sum
\(F^+(c)\) adds \(\gamma(B)\) for every box not certified to be outside the
corresponding event, namely for boxes whose interval tests do not prove
\(s_B>0\) or \((q_B^{\rm in})^->c\) in the inside integral, and do not prove
\(s_B\le0\) or \((q_B^{\rm out})^->c\) in the outside integral.  Finally set
\[
F^-(c)=\sum_{\text{certified inside/outside accepted boxes}}\gamma(B),
\qquad
F^+(c)=\eta_{\rm tail}+\sum_{\text{not certified rejected boxes}}\gamma(B).
\]
Then
\[
F^-(c)\le F_{z,\Sigma}(c)\le F^+(c).
\]
For each rational \(c\ge0\) and each rational tolerance \(\eta>0\), choose
the corresponding \(R(\eta)\), build the square \(S_{R(\eta)}\), and refine
the partition until \(F^+(c)-F^-(c)\le\eta\).  Denote this finite subroutine
by \(\mathsf{CDF}(c,\eta)\).

The bracket construction is mechanical.  First run
\(\mathsf{CDF}(0,\eta_j)\), \(j=0,1,\ldots\), until \(F^+(0)<p\), and set
\(L=0\).  Next dovetail the dyadic upper search: for stages
\(s=0,1,2,\ldots\), compute \(\mathsf{CDF}(2^r,\eta_s)\) for every
\(0\le r\le s\), using rational directed arithmetic at each dyadic point.
Stop at the first stage \(s\) for which some inspected dyadic point satisfies
\(p\le F^-(2^r)\), and choose the smallest such \(r\) as \(U=2^r\).  The
stored endpoint certificates give the invariant
\[
F^+(L)<p\le F^-(U).
\]

Let
\[
K=\max\left\{0,\left\lceil\log_2\frac{U-L}{\eta_c}\right\rceil\right\}.
\]
For \(k=1,\ldots,K\), form the rational midpoint \(m=(L+U)/2\).  Run
\(\mathsf{CDF}(m,\eta_0),\ldots,\mathsf{CDF}(m,\eta_J)\).  If some certified
enclosure has \(F^+(m)<p\), replace \(L\) by \(m\) and keep the certificate
for the new lower endpoint.  If some certified enclosure has
\(p\le F^-(m)\), replace \(U\) by \(m\) and keep the certificate for the new
upper endpoint.  If, after the \(J+1\) prescribed refinements,
\[
F^-(m)<p\le F^+(m),
\]
return the undecided-but-bracketed transcript with the current endpoint
certificates and the midpoint certificate.  If no undecided midpoint is
returned during the \(K\) passes, return the resolved bracket \((L,U]\); its
length is at most \(\eta_c\) and its endpoint certificates still satisfy
\(F^+(L)<p\le F^-(U)\).
\end{algorithmiclemma}

\begin{proof}
The transformation \(Z=CY\) reduces the Gaussian density to the product
standard-normal density.  The bound
\eqref{eq:certified-gaussian-tail-truncation-main} and the union bound give
the displayed square-truncation error.  On a retained box \(B\), directed
interval arithmetic encloses every value of the half-space sign, the
unprojected quadratic form on \(H_z\), and the projected quadratic form on
\(H_z^c\).  A box added to the lower sum is therefore wholly contained in one
of the two defining events for \(F_{z,\Sigma}(c)\), and a box removed from the
upper sum is wholly contained in the complement of the relevant event.  The
remaining boxes are charged to the upper sum, so
\[
F^-(c)\le F_{z,\Sigma}(c)\le F^+(c)
\]
after adding the tail only to the upper side.

It remains to justify finite termination of each distribution-function
enclosure.  For fixed rational \(c\), the ambiguous set
is contained, up to an arbitrarily small rectangular neighbourhood, in the
union of the line \(n_z^\top C y=0\) and the two quadratic level curves
\((Cy)^\top\Sigma^{-1}(Cy)=c\) and
\((\Pi_{H_z}Cy)^\top\Sigma^{-1}(\Pi_{H_z}Cy)=c\).  These sets have
two-dimensional Lebesgue measure zero.  Since the standard-normal density is
bounded on \(S_R\) and the Gaussian measure is regular, sufficiently fine
rectangular partitions make the total Gaussian mass of undecided boxes below
any prescribed positive value.  Hence the refinement loop achieves
\(F^+(c)-F^-(c)\le\eta\) in finitely many steps for every requested
\(\eta>0\).  This proves finite termination of each call to
\(\mathsf{CDF}(c,\eta)\).

The initial endpoint search is also finite.  At \(c=0\), the no-atom
statement in \Cref{lem:deterministic-full-rank-projected-gaussian-quantile-main}
gives \(F_{z,\Sigma}(0)=\PP\{T_{z,\Sigma}=0\}=0<p\); once
\(\eta<p\), any certified enclosure with width at most \(\eta\) has
\(F^+(0)<p\).  For the upper endpoint,
\(F_{z,\Sigma}(c)\to1\) as \(c\to\infty\), so some dyadic point \(2^r\) has
\(F_{z,\Sigma}(2^r)>p\).  Because the same lemma proves continuity and
strict increase, this inequality is strict at that dyadic point; once the CDF
enclosure is refined below the positive margin
\(F_{z,\Sigma}(2^r)-p\), the certificate has \(F^-(2^r)\ge p\).  Thus the
mechanical endpoint search reaches a certified pair \(L<U\) after finitely
many rational CDF calls.

The certified endpoint inequalities \(F^+(L)<p\) and \(F^-(U)\ge p\) imply
\[
F_{z,\Sigma}(L)<p,\qquad F_{z,\Sigma}(U)\ge p,
\]
and therefore \(L<c_p(z,\Sigma)\le U\).  The bisection loop preserves this
same invariant by construction.  If the midpoint certificate proves
\(F^+(m)<p\), then \(F_{z,\Sigma}(m)<p\), so \(m<c_p(z,\Sigma)\) and \(m\) is
a valid new lower endpoint.  If it proves \(p\le F^-(m)\), then
\(p\le F_{z,\Sigma}(m)\), so \(c_p(z,\Sigma)\le m\) and \(m\) is a valid new
upper endpoint.  If neither side is certified after the prescribed finite
refinement budget, the algorithm returns the explicitly allowed undecided
midpoint transcript; the stored endpoint certificates still imply
\(L<c_p(z,\Sigma)\le U\), while the midpoint certificate records only
\(F^-(m)<p\le F^+(m)\).  This is a finite certificate of a bracket, not a
claim that the exact midpoint comparison has been resolved.

There are at most \(K\) midpoint passes before the resolved output, and each
pass makes at most \(J+1\) finite CDF calls.  Hence the whole procedure has a
finite transcript.  The final comparison with \([Q]\) follows immediately
from \(L<c_p(z,\Sigma)\le U\) and the fact that \(Q\in[Q]\).
\end{proof}


\begin{lemma}[Certified rank-one non-normal projected-Gaussian quantile]\label{lem:certified-rank-one-nonnormal-projected-gaussian-quantile-main}
Fix \(0<z<1\), put
\[
n_z=(-z,1)^\top,\qquad H_z=\{u:n_z^\top u\le0\},
\]
and let \(\Sigma=\lambda vv^\top\), where \(\lambda>0\),
\(\|v\|=1\), and
\(\operatorname{span}\{v\}\ne\operatorname{span}\{n_z\}\).  Let
\(Z\sim\mathcal N(0,\Sigma)\) and
\[
T_{z,\Sigma}
=(\Pi_{H_z}Z)^\top\Sigma^+(\Pi_{H_z}Z).
\]
Define
\begin{equation}\label{eq:rank-one-nonnormal-scale-main}
c_v(z)=1-\frac{(n_z^\top v)^2}{\|n_z\|^2},
\qquad
a(z,v)=c_v(z)^2 .
\end{equation}
Then \(0<c_v(z)\le1\), hence \(0<a(z,v)\le1\).  One support half-line gives a
\(\chi^2_1\) variate and the other gives \(c_v(z)^2\chi^2_1\), with weights
\(1/2\) and \(1/2\).  Equivalently, for \(c\ge0\)
\begin{equation}\label{eq:rank-one-nonnormal-cdf-main}
F^{(1)}_{z,v}(c)
:=\PP\{T_{z,\Sigma}\le c\}
=\frac12F_{\chi^2_1}(c)
 +\frac12F_{\chi^2_1}\!\left(\frac{c}{a(z,v)}\right).
\end{equation}
The formula is independent of the scale \(\lambda\).  The distribution is
continuous and strictly increasing on \([0,\infty)\), so for \(0<p<1\)
\[
c^{(1)}_p(z,v):=\inf\{c\ge0:F^{(1)}_{z,v}(c)\ge p\}
\]
is the unique exact lower \(p\)-quantile.

Moreover this quantile has a certified one-dimensional construction.  Suppose
a diagonal branch certificate proves the rank-one non-normal branch and gives
an outward-rounded enclosure \([a]\Subset(0,1]\) of the scale
\(a(z,v)\), together with an outward-rounded enclosure \([Q]\) of a reported
statistic.  For every abscissa tolerance \(\eta_c>0\), directed-evaluation
tolerance schedule, and finite midpoint refinement budget, a deterministic
rational bisection returns either a resolved bracket \(L<U\), \(U-L\le\eta_c\),
or an undecided-but-bracketed midpoint transcript.  In both cases the endpoint
certificates satisfy
\[
F^{(1),+}_{[a]}(L)<p\le F^{(1),-}_{[a]}(U),
\]
where, for \(c\ge0\),
\[
F^{(1),-}_{[a]}(c)
=\frac12F_{\chi^2_1}^{-}(c)
 +\frac12F_{\chi^2_1}^{-}\!\left(\frac{c}{[a]^+}\right),
\qquad
F^{(1),+}_{[a]}(c)
=\frac12F_{\chi^2_1}^{+}(c)
 +\frac12F_{\chi^2_1}^{+}\!\left(\frac{c}{[a]^-}\right).
\]
Here \(F_{\chi^2_1}^{-}(x)\le F_{\chi^2_1}(x)\le
F_{\chi^2_1}^{+}(x)\) are directed interval evaluations of
\[
F_{\chi^2_1}(x)=2\Phi(\sqrt x)-1,\qquad x\ge0 .
\]
An undecided midpoint transcript also contains a rational \(m\in(L,U)\) with
\[
F^{(1),-}_{[a]}(m)<p\le F^{(1),+}_{[a]}(m).
\]
Consequently, for either output,
\[
L<c^{(1)}_p(z,v)\le U.
\]
The same transcript certifies the comparison with the ideal quantile:
\[
[Q]^+\le L\Rightarrow Q<c^{(1)}_p(z,v),\qquad
[Q]^->U\Rightarrow Q>c^{(1)}_p(z,v),
\]
and otherwise certifies only
\[
Q-c^{(1)}_p(z,v)\in[[Q]^--U,[Q]^+-L].
\]
\end{lemma}

\begin{proof}
Write \(Z=\xi v\), where \(\xi\sim\mathcal N(0,\lambda)\).  Since
\(\Sigma^+=\lambda^{-1}vv^\top\), the inside half-line
\(\{n_z^\top(\xi v)\le0\}\) contributes
\[
T_{z,\Sigma}=\xi^2/\lambda .
\]
On the other half-line,
\[
\Pi_{H_z}(\xi v)
=\xi\left(v-\frac{n_z^\top v}{\|n_z\|^2}n_z\right),
\]
and therefore
\[
T_{z,\Sigma}
=\frac{\xi^2}{\lambda}
\left(1-\frac{(n_z^\top v)^2}{\|n_z\|^2}\right)^2 .
\]
The factor in parentheses is strictly positive because \(v\) is not parallel
to \(n_z\), and it is at most \(1\) by Cauchy's inequality.  The sign of
\(\xi\) is independent of \(\xi^2\), so the two half-lines have equal
probability whenever \(n_z^\top v\ne0\), and the displayed mixture follows.
If \(n_z^\top v=0\), the whole support lies on the boundary of \(H_z\);
then \(c_v(z)=a(z,v)=1\), and the same display reduces to the single
\(\chi^2_1\) law.  This proves
\eqref{eq:rank-one-nonnormal-cdf-main}, continuity, strict increase, and the
exact lower-quantile statement.

For the certificate, monotonicity of \(F_{\chi^2_1}\) and
\([a]\Subset(0,1]\) give the displayed lower and upper enclosures of
\(F^{(1)}_{z,v}(c)\).  The endpoint \(0\) has certified upper value \(0<p\),
and sufficiently large dyadic \(b\) has certified lower value at least \(p\)
because \(F^{(1)}_{z,v}(c)\to1\).  The bisection is exactly the mechanical
rule used above: split the current bracket at a rational midpoint, keep the
left half when the directed upper CDF bound is \(<p\), keep the right half
when the directed lower CDF bound is \(\ge p\), and otherwise stop after the
prescribed finite refinement budget with the displayed undecided midpoint
certificate.  Every accepted endpoint preserves
\(F^{(1),+}_{[a]}(L)<p\le F^{(1),-}_{[a]}(U)\), and therefore
\(L<c^{(1)}_p(z,v)\le U\).  The procedure is finite because it performs only
the prescribed endpoint search and finitely many midpoint evaluations; it
certifies the bracket and does not require a total decision at a midpoint
where the interval comparison straddles \(p\).  The comparison statements
follow from \(Q\in[Q]\) and \(L<c^{(1)}_p(z,v)\le U\).
\end{proof}


\begin{theorem}[Certified projected-Gaussian quantile bracketing algorithm]\label{thm:certified-projected-gaussian-quantile-algorithm-main}
Fix \(0<z<1\), a positive definite \(2\times2\) covariance matrix
\(\Sigma\), and \(0<p<1\).  Put
\[
H_z=\{u\in\RR^2:u_2\le zu_1\},\qquad
\Pi_{H_z}u=u-\frac{(u_2-zu_1)_+}{1+z^2}(-z,1)^\top ,
\]
and, with \(\phi_\Sigma\) denoting the density of
\(\mathcal N(0,\Sigma)\), define
\begin{equation}\label{eq:certified-projected-gaussian-cdf-main}
F_{z,\Sigma}(c)
:=\int_{H_z}\mathbf1\{u^\top\Sigma^{-1}u\le c\}\phi_\Sigma(u)\,du
+\int_{H_z^c}
\mathbf1\{(\Pi_{H_z}u)^\top\Sigma^{-1}(\Pi_{H_z}u)\le c\}
\phi_\Sigma(u)\,du .
\end{equation}
The lower projected-Gaussian quantile
\[
c_p(z,\Sigma):=\inf\{c\ge0:F_{z,\Sigma}(c)\ge p\}
\]
is a reproducible mathematical object in the following finite-certificate
sense.  The algorithm certifies an enclosing bracket for the lower quantile
and a comparison with a reported statistic only when the statistic interval is
separated from that bracket; it does not decide all finite-precision
midpoints.

For every abscissa tolerance \(\eta_c>0\), probability tolerance schedule
\((\eta_k)_{k\ge0}\), and finite midpoint refinement budget \(J\), the
deterministic procedure of
\Cref{lem:certified-deterministic-projected-gaussian-quantile-main} produces
a finite transcript consisting of outward-rounded enclosures for \(z\),
\(\Sigma\), \(\Sigma^{-1}\), a Cholesky factor \(C\), a certified tail square
\([-R,R]^2\), finite box partitions of that square, interval lower and upper
distribution sums \(F^-(c)\le F_{z,\Sigma}(c)\le F^+(c)\), rational split
points, and certified endpoints \(L<U\) satisfying the invariant
\[
F^+(L)<p\le F^-(U).
\]
The transcript is of one of the following two forms.
\begin{enumerate}[label=\textup{(\alph*)},leftmargin=2.4em,itemsep=0.2em]
\item A resolved bracket, where \(U-L\le\eta_c\).
\item An undecided-but-bracketed midpoint, where the transcript also prints a
rational \(m\in(L,U)\) satisfying
\[
F^-(m)<p\le F^+(m).
\]
\end{enumerate}
Both outputs certify the same theorem-level statement,
\[
L<c_p(z,\Sigma)\le U .
\]
The theorem does not require the finite computation to decide a midpoint whose
outward-rounded distribution-function enclosure straddles \(p\); such a
midpoint is reported as undecided, while the endpoint bracket remains
certified.
If a computed statistic \(Q\) is given by an outward-rounded interval
\([Q]\), the same transcript certifies the comparison with the ideal quantile:
\[
[Q]^+\le L\Rightarrow Q<c_p(z,\Sigma),\qquad
[Q]^->U\Rightarrow Q>c_p(z,\Sigma),
\]
and otherwise certifies only the enclosure
\[
Q-c_p(z,\Sigma)\in[[Q]^--U,[Q]^+-L].
\]

Moreover, conditional finite Monte Carlo simulation is coverage-certified only
after a one-sided domination certificate is attached.  Given conditionally
independent draws \(T_1^\ast,\ldots,T_R^\ast\) from the law with distribution
function \(F_{z,\Sigma}\), let
\[
\widehat F_R(t)=R^{-1}\sum_{m=1}^R\mathbf1_{\{T_m^\ast\le t\}},\qquad
q_R=T_{(\lceil pR\rceil)}^\ast ,
\]
with the draws sorted increasingly, and let
\[
\varepsilon_{R,\delta}:=\sqrt{\frac{\log(2/\delta)}{2R}},\qquad
d_{R,\delta}:=
\inf\{d\ge0:\widehat F_R(q_R+d)-\varepsilon_{R,\delta}\ge p\}.
\]
If \(p+\varepsilon_{R,\delta}<1\), then
\[
\PP^\ast\{q_R+d_{R,\delta}\ge c_p(z,\Sigma)\}\ge1-\delta .
\]
Thus a raw simulated quantile is diagnostic by itself; the reproducible
coverage-certified simulated critical value is the inflated value
\(q_R+d_{R,\delta}\), or another value accompanied by an explicitly proved
one-sided domination event of the same form.
\end{theorem}

\begin{proof}
The identity \eqref{eq:certified-projected-gaussian-cdf-main} is the
distribution function of
\((\Pi_{H_z}Z)^\top\Sigma^{-1}\Pi_{H_z}Z\),
\(Z\sim\mathcal N(0,\Sigma)\), by
\Cref{lem:deterministic-full-rank-projected-gaussian-quantile-main}.  The
same lemma proves that \(F_{z,\Sigma}\) is continuous and strictly increasing
on \([0,\infty)\).  The interval-integration and bisection procedure in
\Cref{lem:certified-deterministic-projected-gaussian-quantile-main} constructs
the displayed \(F^-,F^+\), rational split points, and endpoint inequalities.
The endpoint inequalities imply
\[
F_{z,\Sigma}(L)\le F^+(L)<p,\qquad
F_{z,\Sigma}(U)\ge F^-(U)\ge p,
\]
so the lower quantile lies in \((L,U]\).  If the output is an undecided
midpoint, the additional displayed inequality is only a record of the
midpoint's unresolved interval comparison and is not used to infer equality or
coverage.  The three comparison statements with \([Q]\) are then immediate
from \(Q\in[Q]\) and \(L<c_p(z,\Sigma)\le U\).

For the simulation statement, the Dvoretzky--Kiefer--Wolfowitz inequality gives
\[
\PP^\ast\{\sup_t|\widehat F_R(t)-F_{z,\Sigma}(t)|
\le\varepsilon_{R,\delta}\}\ge1-\delta .
\]
On this event the defining inequality for \(d_{R,\delta}\) gives
\[
F_{z,\Sigma}(q_R+d_{R,\delta})
\ge \widehat F_R(q_R+d_{R,\delta})-\varepsilon_{R,\delta}
\ge p .
\]
By the definition of the lower quantile,
\(q_R+d_{R,\delta}\ge c_p(z,\Sigma)\).  The stated conditional probability
bound follows.
\end{proof}


\begin{definition}[Finite diagonal report]\label{def:finite-diagonal-report-main}
Fix \(0<\alpha<1\).  The manuscript's finite diagonal report is a
record-level statement only at
\[
z_0=\frac12,\qquad \sigma_{z_0}=(2z_0,z_0^2)=(1,1/4),
\]
and only on the reporting event \(\mathcal E_N\).  On \(\mathcal E_N\), compute
\[
Q_N(z_0)=
N(\widehat\sigma_N-\sigma_{z_0})^\top
\widehat\Sigma_{\sigma,N}^{+}
(\widehat\sigma_N-\sigma_{z_0}),
\]
where \(\widehat\Sigma_{\sigma,N}^{+}\) is the retained-rank pseudoinverse
used in \Cref{thm:e-fixed-candidate-exchange-diagonal-main}.  The report must
also carry the population branch certificate for the computable diagonal
covariance \(\Sigma_\sigma(z_0)\) in the sense of
\Cref{def:dmap-diagonal-rank-certificate-main}.  For the submitted manuscript
this certificate is the full-rank certificate
\Cref{lem:z-half-diagonal-branch-certificate-main}.

Given this branch certificate, the critical value must be supplied by exactly
one of the following two routes.  These two routes are exhaustive for the
finite-record certificate interfaces cross-referenced after Theorem~E: a raw
simulated quantile, a finite-sample eigenvalue branch guess, or an undocumented
numerical CDF value is a diagnostic, not a coverage-certified critical value.

\emph{Certified ideal-CDF route.}  Let \(F_N^\ast\) be the conditional CDF of
the ideal projected-Gaussian statistic \(T_N^\ast(z_0)\).  A certified ideal
conditional CDF routine returns outward-rounded functions
\[
F_N^-(c)\le F_N^\ast(c)\le F_N^+(c).
\]
A certified critical-value transcript consists of this CDF bracket, a pair
\(L_N<U_N\) satisfying
\[
F_N^+(L_N)<1-\alpha\le F_N^-(U_N),\qquad
U_N-L_N=o(1).
\]
The statistic computation also returns an outward-rounded interval
\([Q_N]=[[Q_N]^-,[Q_N]^+]\) containing the realized value \(Q_N(z_0)\).
The transcript is converted into a realized finite-record decision only by
\Cref{thm:certified-diagonal-inference-main}.  Without that complete
transcript, no accept/reject decision is asserted; the numerical values are
diagnostic only.
The deterministic routines in
\Cref{thm:certified-projected-gaussian-quantile-algorithm-main,lem:certified-deterministic-projected-gaussian-quantile-main,lem:certified-rank-one-nonnormal-projected-gaussian-quantile-main}
produce exactly such endpoint inequalities on their certified branches.  A
rank-one-normal zero-atom branch or a zero-covariance branch uses the buffered
coverage statement of \Cref{thm:e-fixed-candidate-exchange-diagonal-main};
without the corresponding printed buffered inequality the finite output is
diagnostic only.

\emph{Canonical DKW simulation route.}  Conditional on the observed record,
let \(\PP^\ast\) denote the auxiliary simulation law of
\(T_N^\ast(z_0)\).  This route certifies only a conditional domination
property under \(\PP^\ast\) for the simulated critical value; record
probabilities remain governed by the separate law \(\PP\) and by the
fixed-candidate coverage theorem.  Draw \(N\) projected-Gaussian values from
\(T_N^\ast(z_0)\), sort them increasingly, and set
\[
\widehat c_{N,\mathrm{DKW}}
=T^*_{N,(\lceil(1-\alpha+\epsilon_N^{\rm MC})N\rceil)},\qquad
\epsilon_N^{\rm MC}=\sqrt{\frac{\log(2N^2)}{2N}} .
\]
This route is available only when \(\epsilon_N^{\rm MC}<\alpha\).  Then
\[
\PP^\ast\{\widehat c_{N,\mathrm{DKW}}
\ge c_{1-\alpha,N}^{-,*}\}\ge1-N^{-2},
\]
where \(c_{1-\alpha,N}^{-,*}\) is the ideal lower conditional
\((1-\alpha)\)-quantile of \(T_N^\ast(z_0)\).  The finite simulation report
uses this critical value in the accept/reject comparison with the interval
\([Q_N]\).  Without the displayed DKW domination statement, simulated
quantiles are diagnostics only.
\end{definition}

\begin{proposition}[Master rule for finite diagonal reports]\label{prop:finite-diagonal-report-master-main}
Under \Cref{ass:finite-record-master-main}, every coverage-certified
finite-record exchange-diagonal acceptance statement in this manuscript is a
finite diagonal report in the sense of
\Cref{def:finite-diagonal-report-main}.  Equivalently, the operational
certificate is
\[
\text{branch certificate}
\quad+\quad
\bigl(\text{ideal quantile bracket deciding }[Q_N]\bigr),
\]
or
\[
\text{branch certificate}
\quad+\quad
\bigl(R_N=N,\ \delta_N^{\rm MC}=N^{-2}\text{ DKW-inflated empirical quantile}\bigr).
\]
If either the branch certificate or the critical-value certificate is absent,
the projected-Gaussian calculation remains an asymptotic diagnostic and no
coverage-certified finite diagonal report is formed.
\end{proposition}

\begin{proof}
The exhaustive alternatives are part of
\Cref{def:finite-diagonal-report-main}; it remains only to verify that the two
allowed critical-value routes dominate or bracket the same ideal conditional
quantile used in the coverage theorem.
For the certified ideal-CDF route, the transcript inequalities
\(F_N^+(L_N)<1-\alpha\le F_N^-(U_N)\) imply
\[
F_N^\ast(L_N)<1-\alpha,\qquad F_N^\ast(U_N)\ge1-\alpha.
\]
Hence the ideal lower conditional quantile
\[
c_{1-\alpha,N}^{-,*}:=\inf\{c:F_N^\ast(c)\ge1-\alpha\}
\]
satisfies \(L_N<c_{1-\alpha,N}^{-,*}\le U_N\).  The realized finite-record
decision is then exactly the interval-separation decision in
\Cref{thm:certified-diagonal-inference-main}: separation certifies the
corresponding side of the ideal inequality, and absence of separation leaves
the finite transcript undecided.

For the DKW route, let \(\widehat F_N^\ast\) be the empirical CDF of the \(N\)
conditional projected-Gaussian draws.  The record is fixed throughout this
paragraph.  Massart's DKW inequality under the auxiliary simulation law
\(\PP^\ast\) gives
\[
\PP^\ast\left\{\sup_c|\widehat F_N^\ast(c)-F_N^\ast(c)|
\le \epsilon_N^{\rm MC}\right\}
\ge
1-2e^{-2N(\epsilon_N^{\rm MC})^2}
=1-N^{-2}.
\]
On this event, at
\(\widehat c_{N,\mathrm{DKW}}
=T^*_{N,(\lceil(1-\alpha+\epsilon_N^{\rm MC})N\rceil)}\),
the empirical CDF satisfies
\[
\widehat F_N^\ast(\widehat c_{N,\mathrm{DKW}})
\ge1-\alpha+\epsilon_N^{\rm MC}.
\]
Therefore
\[
F_N^\ast(\widehat c_{N,\mathrm{DKW}})
\ge
\widehat F_N^\ast(\widehat c_{N,\mathrm{DKW}})
-\epsilon_N^{\rm MC}
\ge1-\alpha.
\]
By the definition of the lower conditional quantile,
\(\widehat c_{N,\mathrm{DKW}}\ge c_{1-\alpha,N}^{-,*}\).  This proves the
displayed conditional domination probability.  The restriction
\(\epsilon_N^{\rm MC}<\alpha\) ensures that the requested empirical level
\(1-\alpha+\epsilon_N^{\rm MC}\) is strictly below one.
\end{proof}

\paragraph{Algorithm 1: all-complete-cycle separated-batch covariance.}

\begin{enumerate}[label=\textup{\arabic*.},leftmargin=2.0em,itemsep=0.35em]
\item From the retained record \(A_0,\ldots,A_{N-1}\), form the complete-cycle

look-ahead cycles required by the local windows are omitted.

U_j^{(1)}-\widehat r_{1,N}^{\mathrm{cyc}}L_j,\,
U_j^{(2)}-\widehat r_{2,N}^{\mathrm{cyc}}L_j\bigr)^\top .

\text{ for every }1\le\ell\le m\},

\widehat r_{1,N}^{\mathrm{cyc}}-(\widehat r_{0,N}^{\mathrm{cyc}})^2,

\paragraph{Estimated branch-map reduction.}\label{app:branch-map-oracle-reduction-main}
Work at a separated population point \(p\ne s\), and let \(\mathcal B\) be the
deterministic oracle branch map from
\Cref{conv:branch-map-labelled-rates-main}.  Choose disjoint open
neighbourhoods \(U_p,U_s\subset(0,1)\) of the two population roots on which the
oracle labels are constant.  Let \(\widehat{\mathcal B}_N\) be any branch map
formed from calibration or side data, and let
\[
\mathcal C_N=\{\widehat{\mathcal B}_N
\text{ agrees with }\mathcal B\text{ on }U_p\cup U_s\}.
\]
If \(\PP(\mathcal C_N)\to1\), then every retained labelled-root or
labelled-rate estimator obtained by replacing \(\mathcal B\) with
\(\widehat{\mathcal B}_N\) is asymptotically equivalent to the oracle-labelled
estimator.  In particular, all Gaussian limits, plug-in covariance
consistency statements, and retained-rank Wald limits proved conditionally on
\(\mathcal B\) remain valid for the estimated branch map.  If only the weaker
bound \(\PP(\mathcal C_N^c)\le\epsilon_N\) is available, each corresponding
one-sided coverage lower bound is reduced by at most \(\epsilon_N+o(1)\).

The retained empirical roots converge in probability to the separated
population roots.  Hence, with probability tending to one, the retained roots
lie in \(U_p\cup U_s\).  On the intersection of this event with
\(\mathcal C_N\), the estimated-map and oracle-map labelled estimators are
identical.  Therefore their difference is zero with probability tending to
one, and Slutsky's theorem transfers the oracle Gaussian, covariance, and Wald
limits.  For coverage, the reported event can fail relative to the oracle
event only on \(\mathcal C_N^c\), up to the \(o(1)\) probability that the
retained roots have not entered \(U_p\cup U_s\).  This gives the stated loss
bound.

\begin{lemma}[D-MAP diagonal covariance alternatives]\label{lem:dmap-diagonal-covariance-alternatives-main}
At each fixed exchange-diagonal population point
\(\sigma_z=(2z,z^2)\), \(0<z<1\), the nonnegative definite matrix
\(\Sigma_\sigma(z)\) satisfies exactly one of four alternatives: full rank,
rank one with range different from \(\operatorname{span}\{n_z\}\), rank one
with range equal to \(\operatorname{span}\{n_z\}\), or zero covariance.  The
tests are
\[
\operatorname{rank}\Sigma_\sigma(z)=2
\Longleftrightarrow \det\Sigma_\sigma(z)>0,
\qquad
\Sigma_\sigma(z)=0
\Longleftrightarrow \operatorname{tr}\Sigma_\sigma(z)=0,
\]
and, in the remaining rank-one case,
\[
\operatorname{Range}\Sigma_\sigma(z)=\operatorname{span}\{n_z\}
\Longleftrightarrow \Sigma_\sigma(z)t_z=0 .
\]
The rank-one-normal branch has limiting law
\(\frac12\delta_0+\frac12\chi^2_1\); the zero-covariance branch has an
identically zero limiting statistic.  The full-rank and rank-one non-normal
branches have continuous projected-Gaussian laws and exact lower quantiles.
\end{lemma}

\begin{algorithmiclemma}[Certified deterministic projected-Gaussian quantile bracket]\label{lem:certified-deterministic-projected-gaussian-quantile-main}
For a certified full-rank covariance branch, the lower conditional
projected-Gaussian quantile is accepted in the main theorem chain only through
a finite bracket \(L<U\) with outward-rounded CDF inequalities
\[
F^+(L)<p\le F^-(U).
\]
The transcript may leave midpoint comparisons undecided; the endpoint
inequalities alone certify \(L<c_p\le U\) and the corresponding comparison
with an outward-rounded statistic interval \([Q]\).
\end{algorithmiclemma}

\begin{lemma}[Certified rank-one non-normal projected-Gaussian quantile]\label{lem:certified-rank-one-nonnormal-projected-gaussian-quantile-main}
In the rank-one non-normal branch,
\(\Sigma=\lambda vv^\top\) with
\(\operatorname{span}\{v\}\ne\operatorname{span}\{n_z\}\), the projected
statistic has distribution
\begin{equation}\label{eq:rank-one-nonnormal-cdf-main}
\frac12F_{\chi^2_1}(c)
+\frac12F_{\chi^2_1}\!\left(\frac{c}{a(z,v)}\right),
\qquad
a(z,v)=
\left(1-\frac{(n_z^\top v)^2}{\|n_z\|^2}\right)^2\in(0,1].
\end{equation}
For the certificate transcript, the scale enclosure is the directed-rounded
version of
\begin{equation}\label{eq:diagonal-rank-one-nonnormal-scale-certificate-main}
a_{\rm rn}=
\left(
1-\frac{n_z^\top\Sigma_\sigma(z)n_z}
{\operatorname{tr}\Sigma_\sigma(z)\,\|n_z\|^2}
\right)^2 .
\end{equation}
A coverage-certified report for this branch must attach an interval enclosure
of \(a(z,v)\) and a one-dimensional directed-rounding quantile bracket.
\end{lemma}

\begin{theorem}[Certified projected-Gaussian quantile bracketing algorithm]\label{thm:certified-projected-gaussian-quantile-algorithm-main}
The deterministic quantile transcript for a certified projected-Gaussian
branch consists of finite outward-rounded enclosures proving
\[
L<c_p(z,\Sigma)\le U
\]
for the lower \(p\)-quantile of the ideal projected-Gaussian law.  Conditional
Monte Carlo output is coverage-certified only when it is inflated by a
one-sided DKW domination certificate; raw simulated quantiles are diagnostics.
\end{theorem}

\begin{definition}[Finite diagonal report]\label{def:finite-diagonal-report-main}
The finite diagonal report covered by the main article is the fixed candidate
\(z_0=1/2\) on the retained reporting event \(\mathcal E_N\).  It consists of
one theorem-level certificate tuple
\[
\mathcal C(z_0,\alpha,N)
=\bigl([z],[\theta],[\Sigma_\sigma],[I_\tau],
[I_{\det}],[w],[L,U],[Q]\bigr).
\]
Here \([z]\) contains only the fixed candidate, \([\theta]\) encloses
\(-\log z_0\), \([\Sigma_\sigma]\), \([I_\tau]\), \([I_{\det}]\), and \([w]\)
are the branch-certificate intervals for the retained covariance used in
Theorem~E, \([L,U]\) is the certified critical-value bracket or dominated
simulated critical value, and \([Q]\) contains the realized statistic
\(Q_N(z_0)\).  The submitted branch certificate is
\Cref{lem:z-half-diagonal-branch-certificate-main}; in the full-rank submitted
case the tuple must prove
\[
[I_{\det}]\subset[\eta,\infty),\qquad
F_N^+(L)<1-\alpha\le F_N^-(U).
\]
The critical-value component is exactly one of two certificates: either an
ideal-CDF bracket deciding the statistic interval or the canonical
DKW-inflated projected-Gaussian simulated critical value with
\[
\epsilon_N^{MC}=\sqrt{\frac{\log(2N^2)}{2N}}.
\]
Without both a branch certificate and a critical-value transcript, the
projected-Gaussian output is diagnostic and no finite accept/reject report is
asserted.
\end{definition}

\begin{proposition}[Master rule for finite diagonal reports]\label{prop:finite-diagonal-report-master-main}
Every coverage-certified exchange-diagonal finite-record acceptance statement
in the main article has the form
\[
\text{branch certificate}+\text{critical-value domination/bracket}.
\]
The branch certificate fixes the population covariance case; the
critical-value certificate either brackets the ideal conditional quantile or
dominates it by the DKW rule.  If either component is absent, no
coverage-certified finite diagonal report is formed.
\end{proposition}
