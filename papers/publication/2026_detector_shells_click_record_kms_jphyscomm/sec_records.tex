\section{Four-block record theorem chain}\label{sec:records}

This section gives the record-side theorem chain in compressed form.  The
standing boundary is the boxed statement in the introduction: the inputs are
the sampled labelled kernels \(T_0,T_1\), and all probabilities below are
probabilities of their stationary visible binary record.  Throughout this
section \((A_n)_{n\in\mathbb Z}\) denotes that two-sided stationary extension.
One-sided strings such as \(A_0,\ldots,A_{N-1}\) are finite restrictions used
only for observed-record statistics; when a proof conditions on a past event
or shifts a renewal epoch, it is using the same stationary two-sided process.

\subsection{Model, benchmark, and renewal law}

\begin{proposition}[Killed-leakage sampled-boundary one-recovery-opportunity instrument and labelled kernels]\label{prop:instrument-kernels-main}
Fix \(\Gamma,\kappa_{\mathrm r},\Delta\tau>0\).  With hidden states ordered as
\((R,D)\), let
\begin{equation}\label{eq:t0-t1-main}
T_0=
\begin{pmatrix}
p&0\\ b&s
\end{pmatrix},
\qquad
T_1=
\begin{pmatrix}
0&1-p\\0&a
\end{pmatrix},
\end{equation}
where
\begin{equation}\label{eq:p-c-main}
p=e^{-\Gamma\Delta\tau},
\qquad
s=e^{-\kappa_{\mathrm r}\Delta\tau},
\end{equation}
and
\begin{equation}\label{eq:a-b-main}
a=\int_0^{\Delta\tau}\kappa_{\mathrm r}e^{-\kappa_{\mathrm r}r}
\bigl(1-e^{-\Gamma(\Delta\tau-r)}\bigr)\,dr,
\qquad
b=\int_0^{\Delta\tau}\kappa_{\mathrm r}e^{-\kappa_{\mathrm r}r}
e^{-\Gamma(\Delta\tau-r)}\,dr .
\end{equation}
For \(\kappa_{\mathrm r}\ne\Gamma\),
\[
a=\frac{\kappa_{\mathrm r}(1-p)-\Gamma(1-s)}
        {\kappa_{\mathrm r}-\Gamma},
\qquad
b=\frac{\kappa_{\mathrm r}(p-s)}
        {\kappa_{\mathrm r}-\Gamma},
\]
with continuous equal-rate limits.  The labelled-kernel convention is
\((T_i)_{XY}=\PP(A_n=i,H_{n+1}=Y\mid H_n=X)\), and word probabilities are
\(\pi T_{u_0}\cdots T_{u_{m-1}}\mathbf1\) from the stationary hidden law.
\end{proposition}

\begin{proof}
From \(R\), no attempt occurs with probability \(p\), while any attempt causes
one visible click and resets the next hidden state to \(D\).  From \(D\), no
recovery before the boundary gives the \(s\) entry; recovery at time \(r\)
followed by no attempt gives the integrand for \(b\), and recovery followed
by at least one attempt gives the integrand for \(a\).  These disjoint cases
are exactly \eqref{eq:t0-t1-main}.  The displayed unequal-rate formulas are
the elementary evaluations of the two integrals.
\end{proof}

\begin{definition}[Reference criteria]\label{def:thermality-notions-main}
Detailed-balance thermality is the rate-ratio condition
\(\Gamma_+/\Gamma_-=e^{-\beta\Omega}\).  Gibbs thermalization is convergence
of the reduced detector state to \(e^{-\beta H_{\mathrm{det}}}/Z\).  The
record-side Parry benchmark is the Parry measure on the golden-mean shift,
the maximum-entropy law under \(11\notin\operatorname{supp}\mu\)
\cite{Parry1964,LindMarcus1995}.  Only the third criterion is a benchmark for
the binary record studied here.
\end{definition}

\begin{lemma}[Standard Parry benchmark for the stipulated ideal counter]\label{prop:ideal-main}
For the ideal one-step blanking counter with transition matrix
\[
P_\beta=
\begin{pmatrix}
p_\beta&1-p_\beta\\
1&0
\end{pmatrix},
\qquad p_\beta=e^{-\Gamma\Delta\tau},
\]
the sampled output is the Parry law on the golden-mean shift iff
\begin{equation}\label{eq:parry-threshold-main}
p_\beta=\varphi^{-1}
\quad\Longleftrightarrow\quad
\Gamma\Delta\tau=\log\varphi .
\end{equation}
\end{lemma}

\begin{proof}
The standard golden-mean Parry calculation gives
\(P^{\mathrm P}(0,0)=\varphi^{-1}\) and \(P^{\mathrm P}(1,0)=1\)
\cite{Parry1964,LindMarcus1995}.  The stipulated blanking counter has the
same support and the same forced transition from \(1\), so equality is
equivalent to \(p_\beta=\varphi^{-1}\).
\end{proof}

\begin{lemma}[Equal-rate and critical notation]\label{lem:equal-rate-critical-convention-main}
The survival factors are always
\[
p=e^{-\Gamma\Delta\tau},\qquad
s=e^{-\kappa_{\mathrm r}\Delta\tau},
\]
and the hidden eigenvalue is
\begin{equation}\label{eq:lambda-hid-main}
\lambda_{\mathrm{hid}}
=p-b
=\frac{\kappa_{\mathrm r}s-\Gamma p}{\kappa_{\mathrm r}-\Gamma}
\quad(\kappa_{\mathrm r}\ne\Gamma),
\end{equation}
with continuous equal-rate value
\(\lambda_{\mathrm{hid}}=e^{-y}(1-y)\), where
\(y=\kappa_{\mathrm r}\Delta\tau=\Gamma\Delta\tau\) on the equal-rate
diagonal.  Put
\[
x=\Gamma\Delta\tau,\qquad y=\kappa_{\mathrm r}\Delta\tau .
\]
For \(x\ne y\),
\[
\lambda_{\mathrm{hid}}=0
\quad\Longleftrightarrow\quad
x e^{-x}=y e^{-y}.
\]
The equation \(x e^{-x}=y e^{-y}\) always has the formal solution \(x=y\).
For \(y\ne1\) this diagonal solution is only the removable zero of the
multiplied quotient numerator, not a zero of the continuous eigenvalue
\(\lambda_{\mathrm{hid}}\).  The critical rate is therefore the non-removable
branch
\begin{equation}\label{eq:gamma-crit-piecewise-main}
\Gamma_{\mathrm{crit}}(\kappa_{\mathrm r},\Delta\tau)
=\Delta\tau^{-1}
\begin{cases}
x_+(y),&0<y<1,\quad x_+(y)>1,\quad x_+(y)e^{-x_+(y)}=y e^{-y},\\
1,&y=1,\\
x_-(y),&y>1,\quad 0<x_-(y)<1,\quad x_-(y)e^{-x_-(y)}=y e^{-y}.
\end{cases}
\end{equation}
Equivalently, for \(y\ne1\) it is the unique positive solution of
\(x e^{-x}=y e^{-y}\) on the opposite side of \(1\) from \(y\), and the
removable equal-rate branch \(x=y\) is explicitly excluded.  Only at \(y=1\)
does the equal-rate branch become a genuine zero, giving
\(\Gamma_{\mathrm{crit}}=\kappa_{\mathrm r}=1/\Delta\tau\).
\end{lemma}

\begin{proof}
The identity \(p-b\) and the quotient form follow from
\eqref{eq:a-b-main}.  The equal-rate value is l'Hopital's rule, equivalently
\(b=y e^{-y}\) when \(\kappa_{\mathrm r}=\Gamma\), and hence
\(\lambda_{\mathrm{hid}}=e^{-y}(1-y)\).  For \(x\ne y\), multiplying the
quotient numerator by \(\Delta\tau\) gives the scalar equation
\(x e^{-x}=y e^{-y}\).  The function \(x\mapsto xe^{-x}\) is strictly
increasing on \((0,1)\), strictly decreasing on \((1,\infty)\), and has its
only maximum at \(1\).  Hence the level through \(y\ne1\) has exactly one
positive solution on the opposite side of \(1\), in addition to the removable
solution \(x=y\).  This gives the two non-removable branches in
\eqref{eq:gamma-crit-piecewise-main}.  When \(y=1\), the opposite branch
coalesces with the diagonal and the equal-rate limit \(e^{-1}(1-1)=0\) gives
\(\Gamma_{\mathrm{crit}}=1/\Delta\tau=\kappa_{\mathrm r}\).
\end{proof}

\begin{lemma}[Regeneration and zero-run law]\label{lem:renewal-gap-main}
Let \(G\) be the number of zeros between consecutive clicks in the stationary
sampled boundary-reset record.  Click times are regeneration times.  For
\(k\ge0\),
\begin{equation}\label{eq:gap-main}
g_k=\PP(G=k)=e_D^\top T_0^kT_1\mathbf1 .
\end{equation}
If \(\kappa_{\mathrm r}\ne\Gamma\), then
\begin{equation}\label{eq:gap-closed-main}
g_k=
\frac{\kappa_{\mathrm r}(1-p)p^k-\Gamma(1-s)s^k}
     {\kappa_{\mathrm r}-\Gamma}.
\end{equation}
The gap law has full support on \(\mathbb N_0\) and an exponential tail.
\end{lemma}

\begin{proof}
After a click the next hidden state is \(D\), independently of the earlier
record.  Thus a gap of \(k\) zeros followed by a click has probability
\eqref{eq:gap-main}.  Since \(T_0\) is lower triangular,
\[
e_D^\top T_0^kT_1\mathbf1=b\sum_{j=0}^{k-1}s^jp^{k-1-j}(1-p)+s^ka,
\]
which reduces to \eqref{eq:gap-closed-main}; the equal-rate case is the
continuous limit.  All terms are probabilities of disjoint paths, so they are
positive, and the two survival factors lie in \((0,1)\).
\end{proof}

\begin{corollary}[Support of finite binary words]\label{cor:word-support-main}
Every finite binary word has positive probability under the stationary
sampled boundary-reset record.
\end{corollary}

\begin{proof}
The gap law in \Cref{lem:renewal-gap-main} has \(g_k>0\) for every
\(k\ge0\).  Decompose any finite word into the observed portions of successive
zero runs between clicks, extending at the two ends if needed.
\end{proof}

\subsection{Exact dependence, hazard, and count diagnostics}

\begin{lemma}[Stationary boundary-reset recovery law]\label{lem:stationary-main}
For \(P=T_0+T_1=\begin{psmallmatrix}p&1-p\\ b&1-b\end{psmallmatrix}\), the
stationary hidden law is
\[
\pi=\left(\frac b{1-p+b},\frac{1-p}{1-p+b}\right).
\]
Writing \(q_0=1-p\), the stationary click density is
\begin{equation}\label{eq:rho-main}
\rho=\pi T_1\mathbf1=\frac{q_0(1-s)}{q_0+b}.
\end{equation}
Moreover \(a<q_0(1-s)<q_0\), and therefore
\(a-\rho=-b(q_0-a)/(q_0+b)<0\).
\end{lemma}

\begin{proof}
Solving \(\pi P=\pi\) gives the displayed law.  Since
\(u=T_1\mathbf1=(q_0,a)^\top\) and \(a+b=1-s\), \(\rho=\pi u\) gives
\eqref{eq:rho-main}.  For \(r>0\),
\[
1-e^{-\Gamma(\Delta\tau-r)}<1-e^{-\Gamma\Delta\tau}=q_0,
\]
so the defining integral for \(a\) gives
\[
a<q_0\int_0^{\Delta\tau}\kappa_{\mathrm r}e^{-\kappa_{\mathrm r}r}\,dr
=q_0(1-s)<q_0 .
\]
The sign follows from
\[
a-\rho=-\frac{b(q_0-a)}{q_0+b}
\]
and the strict positivity of \(b\) and \(q_0-a\).
\end{proof}

\begin{lemma}[Covariance transfer through the hidden mode]\label{lem:covariance-transfer-main}
For the stationary sampled record,
\begin{equation}\label{eq:cov-transfer-main}
\PP(A_n=1\mid A_0=1)
=\rho+(a-\rho)\lambda_{\mathrm{hid}}^{\,n-1},
\qquad
\operatorname{Cov}(A_0,A_n)=\rho(a-\rho)\lambda_{\mathrm{hid}}^{\,n-1}
\end{equation}
for \(n\ge1\).
\end{lemma}

\begin{proof}
With \(q_0=1-p\) and \(u=(q_0,a)^\top\),
\[
u-\rho\mathbf1=\frac{q_0-a}{q_0+b}\binom {q_0}{-b},
\qquad
P\binom {q_0}{-b}=(p-b)\binom {q_0}{-b}.
\]
Thus \(P^{n-1}u=\rho\mathbf1+
\lambda_{\mathrm{hid}}^{n-1}(u-\rho\mathbf1)\).  A click at time \(0\) resets
the next hidden state to \(D\), so the \(D\)-component gives the conditional
probability in \eqref{eq:cov-transfer-main}; multiplying by \(\rho\) and
subtracting \(\rho^2\) gives the covariance.
\end{proof}

\begin{lemma}[Critical-rate uniqueness and covariance phase]\label{lem:critical-rate-main}
For fixed \(\kappa_{\mathrm r},\Delta\tau>0\), the equation
\(\lambda_{\mathrm{hid}}(\Gamma)=0\) has the unique positive solution
\(\Gamma_{\mathrm{crit}}(\kappa_{\mathrm r},\Delta\tau)\) defined by the
piecewise non-removable branch
\eqref{eq:gamma-crit-piecewise-main}.  When
\(\Gamma_{\mathrm{crit}}\ne\kappa_{\mathrm r}\), this is equivalently
\begin{equation}\label{eq:gamma-c-main}
\Gamma_{\mathrm{crit}}e^{-\Gamma_{\mathrm{crit}}\Delta\tau}
=\kappa_{\mathrm r}e^{-\kappa_{\mathrm r}\Delta\tau}.
\end{equation}
The displayed scalar equation also has the diagonal solution
\(\Gamma_{\mathrm{crit}}=\kappa_{\mathrm r}\), but that solution is removable
for \(\lambda_{\mathrm{hid}}\) unless
\(\kappa_{\mathrm r}\Delta\tau=1\).  Thus on the diagonal
\(\Gamma=\kappa_{\mathrm r}\), the zero occurs only at
\(\Gamma=\kappa_{\mathrm r}=1/\Delta\tau\).
The sign of \(\operatorname{Cov}(A_0,A_n)\) is the sign of
\(-\lambda_{\mathrm{hid}}^{\,n-1}\).
\end{lemma}

\begin{proof}
The uniqueness is the branch statement in
\Cref{lem:equal-rate-critical-convention-main}.  The covariance sign follows
from \(\rho>0\) and \(a-\rho<0\) in \Cref{lem:stationary-main}.
\end{proof}

\begin{lemma}[Hazard obstruction to finite symbolic memory]\label{lem:hazard-obstruction-main}
A renewal binary process with full gap support is a finite-order Markov chain
only if its gap hazard is eventually constant.
\end{lemma}

\begin{proof}
For a renewal process, after observing \(k\) consecutive zeros since the last
click, the conditional click probability is the zero-run hazard \(h_k\).
If an \(r\)-step Markov rule generated the same full-support law, then all
histories with \(k\ge r\) trailing zeros would have the same last \(r\) symbols
and hence the same click probability.
\end{proof}

\begin{lemma}[Cylinder factorization at the memory threshold]\label{lem:kernel-factorization-main}
If \(\lambda_{\mathrm{hid}}=0\), then \(P=T_0+T_1=\mathbf1\pi\).  More
explicitly, let \(u=(u_0,\ldots,u_{r-1})\) and
\(v=(v_0,\ldots,v_{\ell-1})\) be arbitrary finite binary words and write
\(T_u=T_{u_0}\cdots T_{u_{r-1}}\), \(T_v=T_{v_0}\cdots T_{v_{\ell-1}}\).
If the word \(u\) is observed on one block and \(v\) is observed on a second
block beginning two samples later, with the intervening sample unobserved,
then
\[
\PP(A_0^{r-1}=u,\ A_{r+1}^{r+\ell}=v)
=\pi T_u P T_v\mathbf1
=\bigl(\pi T_u\mathbf1\bigr)\bigl(\pi T_v\mathbf1\bigr).
\]
Consequently every two visible blocks separated by one unobserved sample are
independent, so the visible record is exact \(1\)-dependent.
\end{lemma}

\begin{proof}
The second eigenvalue of \(P\) is \(\lambda_{\mathrm{hid}}\), while the first is
\(1\).  At \(\lambda_{\mathrm{hid}}=0\), the primitive stochastic matrix
\(P\) has rank one and both rows equal \(\pi\), hence \(P=\mathbf1\pi\).
For fixed words \(u\) and \(v\), summing over the unobserved sample between
the two displayed blocks inserts \(T_0+T_1=P\).  The stationary cylinder
probability is therefore
\[
\pi T_u(T_0+T_1)T_v\mathbf1=\pi T_uPT_v\mathbf1.
\]
At the threshold,
\[
\pi T_uPT_v\mathbf1
=\pi T_u\mathbf1\,\pi T_v\mathbf1
=\PP(A_0^{r-1}=u)\PP(A_0^{\ell-1}=v),
\]
where the last equality is stationarity applied to the two single-block
cylinders.  Thus every pair of finite cylinder events supported on two
finite blocks with one omitted sample between them factorizes.  Translating
the indices gives the same statement for any blocks
\(\{i,\ldots,m\}\) and \(\{m+2,\ldots,j\}\).  Since finite cylinders generate
the corresponding finite-block sigma algebras, the two finite-block sigma
algebras are independent.  Letting the left and right block lengths increase
and using the monotone-class theorem gives independence of
\(\sigma(A_j:j\le m)\) and \(\sigma(A_j:j\ge m+2)\) for every \(m\).
\end{proof}

\begin{lemma}[Exact \texorpdfstring{\(1\)}{1}-dependence criterion]\label{lem:one-dep-criterion-main}
The sampled boundary-reset record is exact \(1\)-dependent iff
\(\lambda_{\mathrm{hid}}=0\), equivalently
\(\Gamma=\Gamma_{\mathrm{crit}}(\kappa_{\mathrm r},\Delta\tau)\).
\end{lemma}

\begin{proof}
The forward direction at the threshold is
\Cref{lem:kernel-factorization-main}.  Conversely, exact \(1\)-dependence
would make \(A_0\) and \(A_2\) independent.  But
\[
\operatorname{Cov}(A_0,A_2)=\rho(a-\rho)\lambda_{\mathrm{hid}},
\]
and the prefactor is nonzero by \Cref{lem:stationary-main}.
\end{proof}

\begin{lemma}[Survival tails and hazard monotonicity]\label{lem:survival-hazard-main}
For \(\kappa_{\mathrm r}\ne\Gamma\),
\begin{equation}\label{eq:survival-tail-main}
S_k=\PP(G\ge k)=
\frac{\kappa_{\mathrm r}p^k-\Gamma s^k}{\kappa_{\mathrm r}-\Gamma},
\end{equation}
and
\begin{equation}\label{eq:hazard-main}
h_k=\PP(G=k\mid G\ge k)=
\frac{\kappa_{\mathrm r}(1-p)p^k-\Gamma(1-s)s^k}
     {\kappa_{\mathrm r}p^k-\Gamma s^k}.
\end{equation}
The equal-rate limit is
\begin{equation}\label{eq:equal-rate-survival-hazard-main}
\PP(G\ge k)=p^k(1+\Gamma\Delta\tau\,k),
\qquad
h_k=
\frac{(1-p)+\Gamma\Delta\tau(k(1-p)-p)}
     {1+\Gamma\Delta\tau\,k}.
\end{equation}
In all cases \(h_{k+1}>h_k\).
\end{lemma}

\begin{proof}
Summing \eqref{eq:gap-closed-main} over \(j\ge k\) gives
\eqref{eq:survival-tail-main}, and \(h_k=g_k/S_k\) gives
\eqref{eq:hazard-main}.  The equal-rate display is the continuous limit.
For the unequal-rate case, a direct subtraction gives
\[
h_{k+1}-h_k
=\frac{\Gamma\kappa_{\mathrm r}(p-s)^2p^ks^k}
        {(\kappa_{\mathrm r}p^k-\Gamma s^k)
         (\kappa_{\mathrm r}p^{k+1}-\Gamma s^{k+1})}
\]
and the denominator has positive product sign.  Indeed
\[
\kappa_{\mathrm r}p^m-\Gamma s^m
=(\kappa_{\mathrm r}-\Gamma)S_m,\qquad m=k,k+1,
\]
by \eqref{eq:survival-tail-main}.  Since \(S_m=\PP(G\ge m)>0\), both factors
have the sign of \(\kappa_{\mathrm r}-\Gamma\): they are both positive when
\(\kappa_{\mathrm r}>\Gamma\) and both negative when
\(\kappa_{\mathrm r}<\Gamma\).  Their product is therefore
\((\kappa_{\mathrm r}-\Gamma)^2S_kS_{k+1}>0\).  The numerator is also
positive because \(\Gamma,\kappa_{\mathrm r}>0\), \(p,s\in(0,1)\), and
\(p\ne s\) in the unequal-rate case.

On the equal-rate diagonal, write
\(y=\Gamma\Delta\tau=\kappa_{\mathrm r}\Delta\tau\) and
\(p=e^{-y}\).  The displayed equal-rate hazard can be rewritten as
\[
h_k
=1-p-\frac{y p}{1+y k}.
\]
Thus
\[
h_{k+1}-h_k
=yp\left(\frac1{1+yk}-\frac1{1+y(k+1)}\right)
=\frac{y^2p}{(1+yk)(1+y(k+1))}>0.
\]
\end{proof}

\begin{theorem}[Classical killed-leakage sampled-boundary record theorem for the one-recovery-opportunity hidden Markov model]\label{thm:recovery-main}
For the sampled kernels \eqref{eq:t0-t1-main}, the stationary visible process
is a full-support renewal process.  Its zero-run law is
\eqref{eq:gap-main}-\eqref{eq:gap-closed-main}, its hazard is
\eqref{eq:hazard-main} with the equal-rate convention
\eqref{eq:equal-rate-survival-hazard-main}, and its covariance is
\[
\operatorname{Cov}(A_0,A_n)=\rho(a-\rho)\lambda_{\mathrm{hid}}^{\,n-1},
\qquad n\ge1.
\]
It is exact \(1\)-dependent iff
\(\lambda_{\mathrm{hid}}=0\), equivalently
\(\Gamma=\Gamma_{\mathrm{crit}}(\kappa_{\mathrm r},\Delta\tau)\).  In all
parameter cases it is not a finite-order Markov chain.
\end{theorem}

\begin{proof}
Regeneration, full support, and the gap law are
\Cref{lem:renewal-gap-main,cor:word-support-main}.  The hazard formula and
strict monotonicity are \Cref{lem:survival-hazard-main}; together with
\Cref{lem:hazard-obstruction-main} they exclude every finite visible Markov
order.  The covariance formula is \Cref{lem:covariance-transfer-main}, and
the exact \(1\)-dependence criterion is \Cref{lem:one-dep-criterion-main}.
\end{proof}

\begin{lemma}[Discrete renewal count-variance identity]\label{lem:discrete-renewal-fano-main}
Let \((A_n)_{n\in\mathbb Z}\) be a stationary binary renewal process on the
sample lattice, with click epochs separated by i.i.d. integer cycle lengths
\(L_j\ge1\), mean \(\mu=\EE L_j<\infty\), variance
\(\sigma_L^2=\Var(L_j)<\infty\), and exponentially decaying tails.  For
\(N_m=\sum_{n=0}^{m-1}A_n\),
\begin{equation}\label{eq:discrete-renewal-fano-identity-main}
\lim_{m\to\infty}\frac{\Var(N_m)}{\EE N_m}
=\frac{\sigma_L^2}{\mu^2}.
\end{equation}
If, in addition, the autocovariances are summable and \(f(0)\) denotes the
zero-frequency spectral density
\(\operatorname{Cov}(A_0,A_0)+2\sum_{h\ge1}\operatorname{Cov}(A_0,A_h)\),
then the same limit equals \(f(0)/\rho\), where \(\rho=\EE A_0=1/\mu\).
\end{lemma}

\begin{proof}
Work first in Palm coordinates with a click at the left endpoint and write
\(S_k=L_1+\cdots+L_k\) and
\(K_m=\max\{k:S_k\le m\}\).  The stationary count \(N_m\) differs from the
corresponding Palm count only through the initial delayed cycle and the
terminal residual cycle.  The stationary residual cycle has the size-biased
cycle tail, which is still exponentially decaying, and the maximum of the
complete cycles intersecting \([0,m)\) is \(O_{\PP}(\log m)\).  Hence this
endpoint replacement changes \(m^{-1/2}(N_m-\EE N_m)\) by \(o_{\PP}(1)\), and
the squared normalized family is uniformly integrable.

At the renewal stopping index,
\[
S_{K_m}\le m<S_{K_m+1}.
\]
Thus, with \(W_j=L_j-\mu\),
\[
\mu\left(K_m-\frac m\mu\right)
=-\sum_{j=1}^{K_m}W_j+O_{\PP}(\log m).
\]
Since \(K_m/m\to1/\mu\) and the partial sums of the centered i.i.d. sequence
\((W_j)\) satisfy the usual random-time-change estimate,
\[
\sum_{j=1}^{K_m}W_j
=\sum_{j=1}^{\lfloor m/\mu\rfloor}W_j+o_{\PP}(\sqrt m).
\]
Therefore
\[
\frac{K_m-m/\mu}{\sqrt m}
=-\frac1\mu\,
\frac{\sum_{j=1}^{\lfloor m/\mu\rfloor}W_j}{\sqrt m}
+o_{\PP}(1),
\]
and uniform integrability yields
\[
\frac{\Var(N_m)}m
\longrightarrow
\frac{\sigma_L^2}{\mu^3}.
\]
Stationarity gives \(\EE N_m=m\rho=m/\mu\), which proves
\eqref{eq:discrete-renewal-fano-identity-main}.

For the spectral form, stationarity gives the exact sampled-window identity
\[
\Var(N_m)=
m\gamma_0+2\sum_{h=1}^{m-1}(m-h)\gamma_h,\qquad
\gamma_h=\operatorname{Cov}(A_0,A_h).
\]
Summability of \((\gamma_h)\) implies
\[
\frac{\Var(N_m)}m\longrightarrow
\gamma_0+2\sum_{h\ge1}\gamma_h=f(0).
\]
Dividing by \(\EE N_m/m=\rho\) gives the equivalent limit \(f(0)/\rho\).
\end{proof}

\begin{proposition}[Spectral density and classical sub-Poisson count noise for the killed-leakage sampled-boundary renewal record]\label{prop:spectrum-main}
The stationary covariance spectral density is
\begin{equation}\label{eq:spectrum-main}
f(\theta)=
\rho(1-\rho)+
2\rho(a-\rho)
\frac{\cos\theta-\lambda_{\mathrm{hid}}}
     {1-2\lambda_{\mathrm{hid}}\cos\theta+\lambda_{\mathrm{hid}}^2}.
\end{equation}
If \(N_m=\sum_{n=0}^{m-1}A_n\), then
\begin{equation}\label{eq:fano-main}
0\le
\lim_{m\to\infty}\frac{\Var(N_m)}{\EE N_m}
=\frac{\Var(G+1)}{\bigl(\EE(G+1)\bigr)^2}
=1-\rho+\frac{2(a-\rho)}{1-\lambda_{\mathrm{hid}}}
<1,
\end{equation}
with \(G\) distributed as in \Cref{lem:renewal-gap-main}.
\end{proposition}

\begin{proof}
Insert the covariance tail from \Cref{thm:recovery-main} into the Herglotz
Fourier series:
\[
f(\theta)
=\rho(1-\rho)
+2\rho(a-\rho)\sum_{n\ge1}
\lambda_{\mathrm{hid}}^{\,n-1}\cos(n\theta).
\]
The exponent is \(n-1\), not \(n\), because the covariance tail begins with
\(\operatorname{Cov}(A_0,A_1)=\rho(a-\rho)\).  Thus the relevant geometric
series is the shifted series below; using
\(\sum_{n\ge1}\lambda_{\mathrm{hid}}^{n}e^{in\theta}\) would insert an
extraneous factor of \(\lambda_{\mathrm{hid}}\) and give the wrong numerator.
Since \(|\lambda_{\mathrm{hid}}|<1\),
\[
\sum_{n\ge1}\lambda_{\mathrm{hid}}^{\,n-1}e^{in\theta}
=
\frac{e^{i\theta}}{1-\lambda_{\mathrm{hid}}e^{i\theta}},
\]
and taking real parts gives
\[
\sum_{n\ge1}\lambda_{\mathrm{hid}}^{\,n-1}\cos(n\theta)
=
\frac{\cos\theta-\lambda_{\mathrm{hid}}}
       {1-2\lambda_{\mathrm{hid}}\cos\theta+\lambda_{\mathrm{hid}}^2}.
\]
This includes the exact \(1\)-dependent case
\(\lambda_{\mathrm{hid}}=0\), where only the adjacent covariance contributes.
For this sampled count convention,
\Cref{lem:discrete-renewal-fano-main} gives, with \(L=G+1\),
\[
\lim_{m\to\infty}\frac{\Var(N_m)}{\EE N_m}
=\frac{\Var(L)}{(\EE L)^2}.
\]
The same lemma identifies this renewal-count limit with \(f(0)/\rho\), since
\(\rho=1/\EE L\) for the stationary renewal record.  Evaluating the displayed
spectral density at \(\theta=0\) gives the direct cycle-variance computation
\[
\frac{\Var(G+1)}{\bigl(\EE(G+1)\bigr)^2}
=\frac{f(0)}{\rho}
=1-\rho+\frac{2(a-\rho)}{1-\lambda_{\mathrm{hid}}}.
\]
The lower bound in \eqref{eq:fano-main} is therefore inherited from the
nonnegative variance quotient, not from the sign of the closed algebraic
correction.  The upper bound is direct: \(\rho>0\), \(a-\rho<0\) by
\Cref{lem:stationary-main}, and \(\lambda_{\mathrm{hid}}<1\) because
\(\lambda_{\mathrm{hid}}=p-b\) with \(p<1\) and \(b>0\).  Hence
\[
\frac{f(0)}{\rho}-1
=-\rho+\frac{2(a-\rho)}{1-\lambda_{\mathrm{hid}}}<0.
\]
\end{proof}

\begin{proposition}[Killed-leakage sampled-boundary fixed-setting long-record diagnostics]\label{prop:finite-diagnostics-main}
For fixed \((\kappa_{\mathrm r},\Delta\tau)\), the lag-two covariance
\(\gamma_2=\operatorname{Cov}(A_0,A_2)\) has the same zero set as
\(\lambda_{\mathrm{hid}}\).  Therefore \(\gamma_2=0\) is the population
fixed-setting diagnostic for the exact \(1\)-dependence threshold
\(\Gamma=\Gamma_{\mathrm{crit}}(\kappa_{\mathrm r},\Delta\tau)\).
\end{proposition}

\begin{proof}
\(\gamma_2=\rho(a-\rho)\lambda_{\mathrm{hid}}\), and the prefactor is nonzero.
\end{proof}

\begin{remark}[Scope of the displayed diagnostics]\label{rem:diagnostics-consistency-only-main}
The hazard curve, the closed Fano factor, and the finite-record plots derived
from \Cref{lem:survival-hazard-main,prop:spectrum-main} are population formula
checks for the sampled boundary-reset law.  In this manuscript they are not
studentized fixed-setting long-record procedures.  The only diagnostic here with a complete
finite-record studentization theorem is the lag-two covariance statistic in
\Cref{cor:boundary-reset-cycle-clt-main}; hazard and Fano diagnostics should
therefore be read only as consistency checks unless a separate long-run
variance theory is added for the chosen empirical hazard or block-count
functional.
\end{remark}

\subsection{Renewal LU product determinant and quotient inverse}

In Markovian-arrival-process language, the pair \((T_0,T_1)\) in
\eqref{eq:t0-t1-main} is a constrained two-state discrete-time MAP:
\(T_0+T_1\) is stochastic, \(T_0\) records a no-arrival transition, and
\(T_1\) records a one-arrival transition.  This places the model inside the
MAP/D-MAP family introduced and developed for correlated arrival streams
\cite{Neuts1979MAP,Lucantoni1991BMAP,AlfaNeuts1995DMAP}.  The inverse result
below is therefore not a generic MAP identifiability theorem.  Generic MAP and
MAP2 parametrizations have nonunique representations, and MAP2
non-identifiability and canonical-form issues are well documented
\cite{BodrogHeindlHorvathTelek2008MAP2Canonical,RamirezCoboLilloWiper2010MAP2Nonidentifiability,
RamirezCoboLillo2012WeakEquivalentMAP}.  The point here is narrower: for this
sampled killed-leakage D-MAP(2) subfamily, the renewal formulas reduce the
observable law to the explicit quotient coordinates \((p+s,ps)\), so the
visible law identifies the unordered pair \(\{p,s\}\) and no label beyond it.

\begin{lemma}[Renewal LU product determinant for inclusion products]\label{prop:renewal-principal-minor-lu-main}\label{lem:renewal-principal-minor-main}
Let \(u_n\) be a scalar sequence and \(\rho>0\).  For a nonempty finite
ordered set \(B=\{i_1<\cdots<i_m\}\), define the finite matrix
\begin{equation}\label{eq:renewal-lu-product-matrix-main}
M_u(B)_{rs}=
\begin{cases}
\rho,&s\le r,\\
\rho-u_{i_s-i_r},&s>r,
\end{cases}
\qquad 1\le r,s\le m,
\end{equation}
and set \(D_u(\varnothing)=1\).  Its renewal LU product determinant is
\begin{equation}\label{eq:renewal-lu-product-determinant-main}
D_u(B):=\det M_u(B)
=\rho\prod_{r=1}^{m-1}u_{i_{r+1}-i_r}.
\end{equation}
\end{lemma}

\begin{proof}
Write \(M_u(B)=\rho L+U\), where \(L_{rs}=\mathbf1_{\{s\le r\}}\) and
\(U_{rs}=(\rho-u_{i_s-i_r})\mathbf1_{\{s>r\}}\).  Then \(L\) is unit lower
triangular, \(L^{-1}\) has diagonal \(1\) and subdiagonal \(-1\), and
\[
M_u(B)=L(\rho I+L^{-1}U).
\]
The second factor is upper triangular with diagonal
\(\rho,u_{i_2-i_1},\ldots,u_{i_m-i_{m-1}}\).  Taking determinants gives the
displayed product.
\end{proof}

\begin{remark}[DPP comparison and noncanonical determinant language]\label{rem:finite-determinant-language-main}\label{lem:principal_minor_gauge_nonuniqueness_main}
\Cref{prop:renewal-principal-minor-lu-main} is only a renewal LU product
determinant for inclusion products already constructed by renewal conditioning
and finite inclusion-exclusion.  It is a set-indexed finite determinant
bookkeeping device, not a canonical point-process correlation object.  No
Hermitian, positive-contraction, operator, nonsymmetric, or sampling DPP
structure is asserted for \(M_u(B)\) or \(D_u(B)\).
\end{remark}

\begin{corollary}[Renewal inclusion and finite word law]\label{cor:renewal-principal-minor-word-law-main}
For a stationary renewal click set \(\varXi\) with intensity \(\rho\) and Palm
renewal masses \(u_n=\PP(n\in\varXi\mid0\in\varXi)\),
\begin{equation}\label{eq:renewal-inclusion-product-main}
\PP(\{i_1<\cdots<i_m\}\subset\varXi)
=\rho\prod_{r=1}^{m-1}u_{i_{r+1}-i_r}
=D_u(\{i_1,\ldots,i_m\}).
\end{equation}
Exact finite configuration probabilities follow by finite
inclusion-exclusion:
\begin{equation}\label{eq:finite-exact-configuration-probabilities-main}
\PP(\varXi\cap I=S)
=\sum_{C\subset I\setminus S}(-1)^{|C|}D_u(S\cup C).
\end{equation}
\end{corollary}

\begin{proposition}[Boundary-reset finite-word determinant formula]\label{cor:boundary_reset_toeplitz_renewal_specialization}\label{cor:finite_window_principal_minor_word_law}\label{prop:finite_inclusion_tail_phase}
For the sampled boundary-reset law,
\begin{equation}\label{eq:finite-inclusion-u-n-main}
u_n=\PP(A_n=1\mid A_0=1)
=\rho+(a-\rho)\lambda_{\mathrm{hid}}^{\,n-1},
\qquad n\ge1.
\end{equation}
Thus the boundary-reset renewal LU product determinants \(D_\Gamma(B)\) are
obtained from \eqref{eq:renewal-lu-product-matrix-main} and
\eqref{eq:renewal-lu-product-determinant-main} by this \(u_n\), and finite word
laws are given by \eqref{eq:finite-exact-configuration-probabilities-main}.
\end{proposition}

\begin{proof}
The formula for \(u_n\) is \Cref{lem:covariance-transfer-main}; the renewal LU
product determinant and word formulas are
\Cref{prop:renewal-principal-minor-lu-main,cor:renewal-principal-minor-word-law-main}.
\end{proof}

\begin{lemma}[Symmetric-coordinate algebra for the quotient inverse]\label{lem:quotient-inverse-symmetric-algebra-main}
Let \(p\in(0,1)\), \(q_0=1-p\), and let the sampled-boundary parameters satisfy
\[
a+b=1-s,\qquad
\lambda_{\mathrm{hid}}=p-b,\qquad
\rho=\frac{q_0(1-s)}{q_0+b}.
\]
Then
\begin{equation}\label{eq:quotient-inverse-s1-algebra-main}
1-a+\lambda_{\mathrm{hid}}=p+s
\end{equation}
and
\begin{equation}\label{eq:quotient-inverse-s2-algebra-main}
\rho(1-\lambda_{\mathrm{hid}})-a+\lambda_{\mathrm{hid}}=ps .
\end{equation}
\end{lemma}

\begin{proof}
Since \(a+b=1-s\), we have \(a=1-s-b\).  Therefore
\[
1-a+\lambda_{\mathrm{hid}}
=1-(1-s-b)+(p-b)
=p+s,
\]
which proves \eqref{eq:quotient-inverse-s1-algebra-main}.  For the second
identity, first note that
\[
1-\lambda_{\mathrm{hid}}=1-p+b=q_0+b.
\]
Thus
\[
\rho(1-\lambda_{\mathrm{hid}})
=\frac{q_0(1-s)}{q_0+b}\,(q_0+b)
=q_0(1-s).
\]
Combining this with \(a=1-s-b\) and \(\lambda_{\mathrm{hid}}=p-b\) gives
\[
\rho(1-\lambda_{\mathrm{hid}})-a+\lambda_{\mathrm{hid}}
=q_0(1-s)-(1-s-b)+(p-b)
=q_0(1-s)+p+s-1.
\]
Because \(p+q_0=1\), the last expression is
\[
q_0(1-s)+p+s-1
=s-q_0s
=ps,
\]
as claimed.
\end{proof}

\begin{theorem}[Killed-leakage sampled D-MAP visible-law quotient]\label{thm:toeplitz_image_inverse_duality}
Suppose the visible law is generated by the sampled-boundary kernels with
\(\Gamma,\kappa_{\mathrm r},\Delta\tau>0\).  Let
\[
r_0=\rho,\qquad r_1=\rho a,\qquad r_2=\rho u_2 .
\]
The population inverse is defined on the domain
\begin{equation}\label{eq:population-inverse-domain-main}
r_0>0,\qquad a-r_0\ne0
\quad
\left(\text{equivalently } r_0>0,\ r_1-r_0^2\ne0\right).
\end{equation}
We write the inverse-map denominator as
\begin{equation}\label{eq:inverse-map-denominator-main}
D:=r_1-r_0^2=r_0(a-r_0).
\end{equation}
For the sampled-boundary kernels this domain condition holds:
\(r_0=\rho>0\) and \(a-r_0=a-\rho<0\) by
\Cref{lem:stationary-main}.  On this domain,
\[
a=\frac{r_1}{r_0},\qquad
u_2=\frac{r_2}{r_0},\qquad
\lambda_{\mathrm{hid}}=\frac{u_2-r_0}{a-r_0}.
\]
The symmetric coordinates are
\begin{equation}\label{eq:quotient-symmetric-coordinates-main}
s_1=1-a+\lambda_{\mathrm{hid}},\qquad
s_2=r_0(1-\lambda_{\mathrm{hid}})-a+\lambda_{\mathrm{hid}}.
\end{equation}
They equal
\[
s_1=p+s,\qquad s_2=ps,
\]
so the visible law determines the unordered survival-factor pair
\[
\{p,s\}=\{e^{-\Gamma\Delta\tau},e^{-\kappa_{\mathrm r}\Delta\tau}\},
\]
or equivalently the roots of \(z^2-s_1z+s_2=0\).  Within this constrained
sampled D-MAP(2) subfamily, the stationary visible law generated by
\((\Gamma,\kappa_{\mathrm r})\) equals that generated by
\((\kappa_{\mathrm r},\Gamma)\), and the visible law does not label which root
is the attempt survival factor and which is the recovery survival factor.  No
claim is made that arbitrary MAP2 representations are identified by these
three coordinates.
For finite-record use of this inverse chart, restrict to stability
neighbourhoods where, for some \(\epsilon,\delta>0\),
\[
r_0\ge\epsilon,\qquad |a-r_0|\ge\epsilon,\qquad
0<p,s<1,\qquad |p-s|\ge\delta,
\]
unless only the projected symmetric coordinates are reported at the exchange
diagonal.  These hypotheses keep the inverse denominator away from zero and
separate the two roots when labelled rate charts are used.
\end{theorem}

\begin{proof}
The gap formula \eqref{eq:gap-closed-main}, the covariance formula, and the
renewal inclusion products are symmetric under exchanging the two rates after
rewriting in \(p,s\).  Conversely, from
\[
r_0=\rho,\qquad r_1=\PP(A_0=A_1=1)=\rho a,\qquad
r_2=\PP(A_0=A_2=1)=\rho u_2
\]
the domain condition \eqref{eq:population-inverse-domain-main} permits the
divisions below.  Hence one obtains \(a=r_1/r_0\), \(u_2=r_2/r_0\), and
\[
\lambda_{\mathrm{hid}}=\frac{u_2-r_0}{a-r_0}.
\]
Then the identities
\[
s_1=1-a+\lambda_{\mathrm{hid}},
\qquad
s_2=r_0(1-\lambda_{\mathrm{hid}})-a+\lambda_{\mathrm{hid}}
\]
recover \(p+s\) and \(ps\) by
\Cref{lem:quotient-inverse-symmetric-algebra-main}, since \(r_0=\rho\) for
the sampled-boundary law.  The unordered roots of
\(z^2-s_1z+s_2=0\) are therefore identified, but their labels are not.
\end{proof}

\begin{corollary}[Three inclusion probabilities are complete quotient coordinates]\label{cor:three_inclusion_complete_coordinates}
The three inclusion probabilities
\((r_0,r_1,r_2)=(\rho,\rho a,\rho u_2)\) determine the full stationary visible
law up to the exchange
\((\Gamma,\kappa_{\mathrm r})\leftrightarrow(\kappa_{\mathrm r},\Gamma)\).
\end{corollary}

\begin{proof}
They determine \(p+s\) and \(ps\) by the proof of
\Cref{thm:toeplitz_image_inverse_duality}; substituting the resulting
unordered pair into \eqref{eq:gap-closed-main} determines the renewal law and
therefore all finite-dimensional distributions.
\end{proof}

\begin{lemma}[Separated-root labelled-rate delta method]\label{lem:separated-root-rate-delta-main}
Let \(r=(r_0,r_1,r_2)\) lie in the population inverse domain of
\Cref{thm:toeplitz_image_inverse_duality}, let
\[
\Phi(r)=s=(s_1,s_2)=(p+s,ps),
\]
and suppose the two roots \(p\ne s\) both lie in \((0,1)\).  A labelled root
chart is a choice, supplied outside the visible record, of which local root is
called the attempt survival factor \(p\) and which is called the recovery
survival factor \(s\).  On any neighbourhood where \(|p-s|\) is bounded away
from zero this chart is \(C^1\), and its Jacobian with respect to the
symmetric coordinates is
\begin{equation}\label{eq:separated-root-jacobian-main}
J_{\mathrm{root}}
=
\begin{pmatrix}
\dfrac{p}{p-s} & -\dfrac1{p-s}\\[1.1em]
\dfrac{s}{s-p} & -\dfrac1{s-p}
\end{pmatrix}.
\end{equation}
Equivalently, for either labelled root \(z\in\{p,s\}\),
\[
dz=\frac{z\,ds_1-ds_2}{2z-s_1}.
\]
For the fixed sampling interval \(\Delta\tau\), the labelled rate map
\[
R(p,s)=\left(-\frac{\log p}{\Delta\tau},
              -\frac{\log s}{\Delta\tau}\right)
=(\Gamma,\kappa_{\mathrm r})
\]
has Jacobian
\begin{equation}\label{eq:separated-rate-jacobian-main}
J_{\mathrm{rate}}
=
\begin{pmatrix}
-\dfrac1{\Delta\tau p} & 0\\[0.8em]
0 & -\dfrac1{\Delta\tau s}
\end{pmatrix}
J_{\mathrm{root}} .
\end{equation}
The full Jacobian from inclusion coordinates to labelled rates is therefore
\begin{equation}\label{eq:inclusion-to-rate-jacobian-main}
J_{\Gamma,\kappa}(r)
=J_{\mathrm{rate}}\,J_\Phi(r),
\qquad
J_\Phi(r)=D\Phi(r).
\end{equation}
Writing
\[
D=r_1-r_0^2,\qquad E=r_2-r_0^2,\qquad
a=\frac{r_1}{r_0},\qquad \lambda=\frac{E}{D},
\]
the entries of \(J_\Phi\) are explicit:
\begin{align}
\frac{\partial\lambda}{\partial r_0}
&=-\frac{2r_0(r_1-r_2)}{D^2},&
\frac{\partial\lambda}{\partial r_1}
&=-\frac{E}{D^2},&
\frac{\partial\lambda}{\partial r_2}
&=\frac1D, \label{eq:lambda-gradient-main}\\
D\Phi_1
&=
\left(
\frac{a}{r_0}+\lambda_{0},\
-\frac1{r_0}+\lambda_{1},\
\lambda_{2}
\right),\nonumber\\
D\Phi_2
&=
\left(
1-\lambda+\frac{a}{r_0}+(1-r_0)\lambda_{0},\
-\frac1{r_0}+(1-r_0)\lambda_{1},\
(1-r_0)\lambda_{2}
\right), \label{eq:phi-jacobian-main}
\end{align}
where \(\lambda_i=\partial\lambda/\partial r_i\).

Consequently, if
\[
\sqrt N(\widehat r_N-r)\Rightarrow \mathcal N(0,\Sigma_r),
\qquad
\widehat\Sigma_{r,N}\to\Sigma_r
\quad\text{in probability},
\]
and \(\widehat r_N\) remains in the same separated-root chart with probability
tending to one, then
\[
\sqrt N\{(\widehat p_N,\widehat s_N)-(p,s)\}
\Rightarrow
\mathcal N(0,\Sigma_{\mathrm{root}}),
\qquad
\Sigma_{\mathrm{root}}
=J_{\mathrm{root}}\,\Sigma_s\,J_{\mathrm{root}}^\top,
\]
\[
\sqrt N\{(\widehat\Gamma_N,\widehat\kappa_{\mathrm r,N})
-(\Gamma,\kappa_{\mathrm r})\}
\Rightarrow
\mathcal N(0,\Sigma_{\Gamma,\kappa}),
\qquad
\Sigma_{\Gamma,\kappa}
=J_{\Gamma,\kappa}(r)\,\Sigma_r\,J_{\Gamma,\kappa}(r)^\top,
\]
where \(\Sigma_s=J_\Phi(r)\Sigma_rJ_\Phi(r)^\top\).  The plug-in covariance
estimators obtained by replacing \(r,\Sigma_r,p,s\) with their separated-chart
estimates are consistent.

For retained-rank Wald reporting in the labelled rate chart, let
\(k=\operatorname{rank}\Sigma_{\Gamma,\kappa}\).  If the thresholded covariance
\(\widehat\Sigma_{\Gamma,\kappa,N}\) satisfies
\[
\|\widehat\Sigma_{\Gamma,\kappa,N}-\Sigma_{\Gamma,\kappa}\|_{\operatorname{op}}
=o_{\PP}(t_N),\qquad t_N\downarrow0,
\]
then the threshold retains exactly the \(k\) positive limiting eigenvalues
with probability tending to one, and
\[
N\,
\{(\widehat\Gamma_N,\widehat\kappa_{\mathrm r,N})
-(\Gamma,\kappa_{\mathrm r})\}^{\top}
\widehat\Sigma_{\Gamma,\kappa,N}^{+}
\{(\widehat\Gamma_N,\widehat\kappa_{\mathrm r,N})
-(\Gamma,\kappa_{\mathrm r})\}
\Rightarrow \chi_k^2 .
\]
This labelled-rate conclusion is conditional on the externally chosen branch;
the visible law alone still identifies only the unordered pair.
\end{lemma}

\begin{proof}
The implicit equation for a root is
\(z^2-s_1z+s_2=0\).  Differentiating gives
\[
(2z-s_1)\,dz-z\,ds_1+ds_2=0.
\]
At separated roots \(2p-s_1=p-s\) and \(2s-s_1=s-p\), which gives
\eqref{eq:separated-root-jacobian-main}.  The logarithmic rate Jacobian is
the ordinary derivative of
\((p,s)\mapsto(-\log p/\Delta\tau,-\log s/\Delta\tau)\), proving
\eqref{eq:separated-rate-jacobian-main}.  The displayed formula for
\(\lambda=E/D\) is the quotient rule, and
\eqref{eq:phi-jacobian-main} follows by differentiating
\[
\Phi_1=1-a+\lambda,\qquad
\Phi_2=r_0-a+(1-r_0)\lambda .
\]
All denominators are bounded away from zero on the stated neighbourhood, so
the maps and plug-in Jacobians are continuous there.  The two Gaussian limits
are then the multivariate delta method, and covariance consistency follows by
continuous mapping.  The retained-rank statement is Weyl's eigenvalue
inequality plus Slutsky's theorem on the limiting range of
\(\Sigma_{\Gamma,\kappa}\).
\end{proof}

\subsection{Finite-record inference and the exchange diagonal}

\begin{definition}[Cycle-index inclusion estimator]\label{def:cycle-index-inclusion-estimator-main}
For a finite observed record \(A_0,\ldots,A_{N-1}\), write the complete click
epochs in the observed window as
\[
\tau_0<\tau_1<\cdots,\qquad K_N=\max\{k:\tau_k<N\},
\]
and set \(L_j=\tau_{j+1}-\tau_j\) whenever both endpoints are observed.  For
\(m=0,1,2\), put
\[
X_n^{(m)}=A_nA_{n+m},\qquad X_n^{(0)}=A_n,
\]
and define the complete-cycle uncentered reward
\[
U_j^{(m)}=\sum_{n=\tau_j}^{\tau_{j+1}-1}X_n^{(m)}.
\]
The admissible all-complete-cycle index set is
\[
\begin{aligned}
J_{m,N}
=\{0\le j\le K_N-m-1:\;&
\tau_j,\tau_{j+1}\in[0,N-1],\\
&\text{and every }n+m\text{ entering }U_j^{(m)}
\text{ lies in }[0,N-1]\}.
\end{aligned}
\]
The cycle-index estimator of the three inclusion coordinates
\[
r=(r_0,r_1,r_2)=(\PP(A_0=1),\PP(A_0=A_1=1),\PP(A_0=A_2=1))
\]
is
\[
\widehat r_N^{\mathrm{cyc}}
=\bigl(\widehat r_{0,N}^{\mathrm{cyc}},
\widehat r_{1,N}^{\mathrm{cyc}},
\widehat r_{2,N}^{\mathrm{cyc}}\bigr)^\top,\qquad
\widehat r_{m,N}^{\mathrm{cyc}}
=\frac1N\sum_{j\in J_{m,N}}U_j^{(m)} .
\]
For covariance estimation the three coordinates are evaluated on the common
complete-cycle set \(J_N=J_{0,N}\cap J_{1,N}\cap J_{2,N}\).  With plug-in
centering, the vector reward used in the separated batches is
\[
\widehat Y_j=
\bigl(U_j^{(0)}-\widehat r_{0,N}^{\mathrm{cyc}}L_j,\,
U_j^{(1)}-\widehat r_{1,N}^{\mathrm{cyc}}L_j,\,
U_j^{(2)}-\widehat r_{2,N}^{\mathrm{cyc}}L_j\bigr)^\top .
\]
\end{definition}

\begin{lemma}[Calendar-stopped triangular-array batch covariance rate]\label{lem:calendar-stopped-batch-cov-main}
Let \((Y_j,L_j)_{j\in\mathbb Z}\) be a stationary cycle sequence with
\(\EE Y_0=0\) in a fixed finite dimension,
\(\mu=\EE L_0\in(0,\infty)\), exponentially decaying tails for
\(L_0\), and a fixed nonnegative look-ahead \(q_{\rm lag}\) such that
\(Y_j=f(G_j,\ldots,G_{j+q_{\rm lag}})\), where the gaps \(G_j\) are i.i.d.
and
\[
\|f(g_0,\ldots,g_{q_{\rm lag}})\|
\le C_f\left(1+\sum_{i=0}^{q_{\rm lag}}g_i\right).
\]
Let \(b_N\to\infty\), \(b_N/N\to0\), \(d_N=b_N+q_{\rm lag}\), and set
\[
C_{\ell,N}=\sum_{j=(\ell-1)d_N}^{(\ell-1)d_N+b_N-1}Y_j,\qquad
V_{\ell,N}=b_N^{-1/2}C_{\ell,N}.
\]
Then \((V_{\ell,N})_{\ell\ge1}\) are independent within each row,
\(\sup_{\ell,N}\EE\|V_{\ell,N}\|^4<\infty\), and
\[
\EE[V_{\ell,N}V_{\ell,N}^{\top}]\longrightarrow
\mathcal T_{q_{\rm lag}}:=\Var(Y_0)+
\sum_{h=1}^{q_{\rm lag}}\{\operatorname{Cov}(Y_0,Y_h)+\operatorname{Cov}(Y_h,Y_0)\}.
\]
Let \(R_N\) be any positive integer-valued random variable, possibly depending
on the same gaps and rewards, such that
\[
n_N:=\left\lfloor\frac{N}{\mu d_N}\right\rfloor\to\infty,\qquad
\left|\frac{R_N}{n_N}-1\right|
=O_{\PP}\left(\sqrt{\frac{d_N}{N}}\right).
\]
On the event \(R_N\ge2\), define
\[
S_N(R_N)=
\frac1{R_N-1}\sum_{\ell=1}^{R_N}
(V_{\ell,N}-\bar V_{R_N,N})(V_{\ell,N}-\bar V_{R_N,N})^{\top},
\qquad
\bar V_{R_N,N}=R_N^{-1}\sum_{\ell=1}^{R_N}V_{\ell,N}.
\]
Then
\begin{equation}\label{eq:calendar-stopped-batch-rate-main}
\left\|S_N(R_N)-\mathcal T_{q_{\rm lag}}\right\|_{\operatorname{op}}
=
O_{\PP}\left(b_N^{-1}+\sqrt{\frac{d_N}{N}}\right).
\end{equation}
Consequently, if
\(K_N=\max\{k:\tau_k<N\}\) and \(R_N=M_N\) is the number of all-complete-cycle
batches selected before the calendar time \(N\), with
\(\bar C_N=M_N^{-1}\sum_{\ell=1}^{M_N}C_{\ell,N}\), then
\[
\left\|
\frac{K_N}{N}\,\frac1{b_N(M_N-1)}
\sum_{\ell=1}^{M_N}(C_{\ell,N}-\bar C_N)(C_{\ell,N}-\bar C_N)^{\top}
-\frac{\mathcal T_{q_{\rm lag}}}{\mu}
\right\|_{\operatorname{op}}
=
O_{\PP}\left(b_N^{-1}+\sqrt{\frac{b_N}{N}}\right).
\]
\end{lemma}

\begin{proof}
The separation by \(q_{\rm lag}\) omitted cycle rewards makes the gap sets entering
distinct \(C_{\ell,N}\)'s disjoint, so the row variables \(V_{\ell,N}\) are
independent.  The displayed growth bound and the exponential gap tail give the
uniform fourth-moment bound after the \(b_N^{-1/2}\) normalization.  Since
\((Y_j)\) is \(q_{\rm lag}\)-dependent,
\[
\frac1{b_N}\Var(C_{\ell,N})
=\Var(Y_0)+\sum_{h=1}^{q_{\rm lag}}\left(1-\frac{h}{b_N}\right)
\{\operatorname{Cov}(Y_0,Y_h)+\operatorname{Cov}(Y_h,Y_0)\}
\]
and therefore
\begin{equation}\label{eq:batch-second-moment-bias-main}
\left\|\EE[V_{\ell,N}V_{\ell,N}^{\top}]
-\mathcal T_{q_{\rm lag}}\right\|_{\operatorname{op}}
=O(b_N^{-1}).
\end{equation}
For the deterministic prefix \(n_N\), the fixed dimension and the uniform
fourth-moment bound give
\[
\left\|
\frac1{n_N}\sum_{\ell=1}^{n_N}
\{V_{\ell,N}V_{\ell,N}^{\top}
-\EE[V_{\ell,N}V_{\ell,N}^{\top}]\}
\right\|_{\operatorname{op}}
=O_{\PP}(n_N^{-1/2})
=O_{\PP}\left(\sqrt{\frac{d_N}{N}}\right),
\]
and similarly
\[
\left\|\frac1{n_N}\sum_{\ell=1}^{n_N}V_{\ell,N}\right\|
=O_{\PP}(n_N^{-1/2})
=O_{\PP}\left(\sqrt{\frac{d_N}{N}}\right).
\]
The deterministic-prefix sample covariance therefore differs from
\(\mathcal T_{q_{\rm lag}}\) by
\[
O_{\PP}\left(b_N^{-1}+\sqrt{\frac{d_N}{N}}\right).
\]

It remains to show that the same rate survives the calendar-stopped prefix.
Put \(a_N=\sqrt{d_N/N}\).  Since
\(|R_N/n_N-1|=O_{\PP}(a_N)\), for each fixed large \(C\) the event
\[
\mathcal E_{N,C}=\{|R_N-n_N|\le Ca_Nn_N,\ n_N/2\le R_N\le2n_N\}
\]
has probability tending to one as \(C\to\infty\), uniformly for large \(N\).
On \(\mathcal E_{N,C}\), the terms added to or removed from the deterministic
prefix are contained in the deterministic window
\[
\{\ell:\ n_N-\lceil Ca_Nn_N\rceil\le \ell\le
n_N+\lceil Ca_Nn_N\rceil\}.
\]
Hence the difference between the \(R_N\)- and \(n_N\)-prefix second-moment
averages is bounded by the normalization change times
\(n_N^{-1}\sum_{\ell\le n_N}\|V_{\ell,N}\|^2\), plus \(2n_N^{-1}\) times the
sum of \(\|V_{\ell,N}\|^2\) over this deterministic window.  The first term is
\(O_{\PP}(a_N)\).  The second term is \(O_{\PP}(a_N)\) by Markov's inequality,
because the window has \(O(a_Nn_N+1)\) indices and
\(\sup_{\ell,N}\EE\|V_{\ell,N}\|^2<\infty\).  The same argument, with
\(\|V_{\ell,N}\|\) in place of \(\|V_{\ell,N}\|^2\), shows that the
empirical mean changes by \(O_{\PP}(a_N)\).  Thus the random-prefix sample
covariance has the rate in
\eqref{eq:calendar-stopped-batch-rate-main}; no independence between \(R_N\)
and the row variables is used.

For the renewal calendar stop, the renewal law gives \(K_N/N\to1/\mu\) in
probability, and the renewal CLT gives
\(K_N=N/\mu+O_{\PP}(\sqrt N)\).  The endpoint convention defining \(J_N\)
removes only the initial incomplete cycle, the terminal incomplete cycle,
\(O(q_{\rm lag})\) local look-ahead cycles, and one unfinished terminal batch of length
less than \(d_N\).  Exponential tails give the largest endpoint cycle length
\(O_{\PP}(\log N)\), so
\[
M_Nd_N=K_N+O_{\PP}(d_N+\log N+q_{\rm lag}).
\]
Consequently
\[
\left|\frac{M_N}{N/(\mu d_N)}-1\right|
=O_{\PP}\left(N^{-1/2}+\frac{d_N}{N}+\frac{\log N}{N}\right)
=O_{\PP}\left(\sqrt{\frac{d_N}{N}}\right),
\]
because \(d_N\to\infty\) and \(d_N=o(N)\).  The preceding random-prefix rate
applied with \(R_N=M_N\) gives
\[
\left\|
\frac1{b_N(M_N-1)}
\sum_{\ell=1}^{M_N}(C_{\ell,N}-\bar C_N)(C_{\ell,N}-\bar C_N)^\top
-\mathcal T_{q_{\rm lag}}
\right\|_{\operatorname{op}}
=O_{\PP}\left(b_N^{-1}+\sqrt{\frac{d_N}{N}}\right).
\]
Multiplication by
\[
K_N/N=1/\mu+O_{\PP}(N^{-1/2})
\]
changes the target to \(\mathcal T_{q_{\rm lag}}/\mu\) and adds only
\(O_{\PP}(N^{-1/2})\), which is dominated by \(\sqrt{b_N/N}\).  Since
\(d_N=b_N+q_{\rm lag}\) and \(q_{\rm lag}\) is fixed, this proves the calendar-time covariance
rate.
\end{proof}

\begin{lemma}[Projected-Gaussian bootstrap at a moving half-space]\label{lem:projected-gaussian-bootstrap-halfspace-main}
For \(t\in(0,1)\), write
\[
H_t=\{h\in\RR^2:h_2\le th_1\}.
\]
Let \(z_N\to z\in(0,1)\) in probability and let
\(\widehat\Sigma_N\) be a symmetric nonnegative definite covariance estimator
such that
\[
\|\widehat\Sigma_N-\Sigma\|_{\operatorname{op}}=o_{\PP}(t_N),
\qquad t_N\downarrow0,
\]
for a fixed nonnegative definite \(2\times2\) matrix \(\Sigma\).  Let
\[
\widehat\Sigma_N^+
=\sum_{\widehat\lambda_{i,N}>t_N}
\widehat\lambda_{i,N}^{-1}
\widehat v_{i,N}\widehat v_{i,N}^{\top}
\]
be the thresholded Moore--Penrose inverse.  Suppose that
\[
X_N\Rightarrow X:=\Pi_{H_z}(Z),
\qquad Z\sim\mathcal N(0,\Sigma).
\]
Define
\[
T_N=X_N^\top\widehat\Sigma_N^+X_N
\]
and, conditionally on the observed record,
\[
Z_N^\ast\sim \mathcal N(0,\widehat\Sigma_N),\qquad
X_N^\ast=\Pi_{H_{z_N}}(Z_N^\ast),\qquad
T_N^\ast=(X_N^\ast)^\top\widehat\Sigma_N^+X_N^\ast .
\]
Then
\[
T_N\Rightarrow T:=X^\top\Sigma^+X,
\]
and the conditional law of \(T_N^\ast\) converges weakly in probability to
the law of \(T\).  Consequently, if \(c_{1-\alpha,N}^\ast\) is the lower
conditional \((1-\alpha)\)-quantile of \(T_N^\ast\) and the limiting quantile
\[
c_{1-\alpha}:=\inf\{c:\PP(T\le c)\ge1-\alpha\}
\]
is a continuity point of the distribution function of \(T\), then
\[
c_{1-\alpha,N}^\ast\to c_{1-\alpha}
\quad\text{in probability}.
\]
If in addition \(\PP(T\le c_{1-\alpha})=1-\alpha\), then
\[
\PP\{T_N\le c_{1-\alpha,N}^\ast\}\to1-\alpha .
\]
\end{lemma}

\begin{proof}
Weyl's inequality and the condition
\(\|\widehat\Sigma_N-\Sigma\|_{\operatorname{op}}=o_{\PP}(t_N)\) imply that,
with probability tending to one, every positive eigenvalue of \(\Sigma\) is
retained and every zero eigenvalue is removed by the threshold.  On that event
the retained spectral projection converges to the range projection of
\(\Sigma\), and
\[
\|\widehat\Sigma_N^+-\Sigma^+\|_{\operatorname{op}}\to0
\]
in probability.  Slutsky's theorem therefore gives
\[
X_N^\top\widehat\Sigma_N^+X_N
\Rightarrow X^\top\Sigma^+X .
\]

The remaining issue is conditional Gaussian convergence after the moving
projection.  The half-space projection has the explicit formula
\[
\Pi_{H_t}(u)
=u-\frac{(u_2-tu_1)_+}{1+t^2}\binom{-t}{1}.
\]
Thus \(u\mapsto\Pi_{H_t}(u)\) is one-Lipschitz and
\((t,u)\mapsto\Pi_{H_t}(u)\) is continuous, uniformly for \(t\) in compact
subintervals of \((0,1)\) and \(u\) in bounded sets.  Since
\(\widehat\Sigma_N\to\Sigma\) in probability, the conditional Gaussian laws
\(\mathcal N(0,\widehat\Sigma_N)\) converge weakly in probability to
\(\mathcal N(0,\Sigma)\).  Combining this Gaussian continuity with
\(z_N\to z\) and \(\widehat\Sigma_N^+\to\Sigma^+\) shows that, for every
bounded continuous \(\varphi:\RR\to\RR\),
\[
\EE^\ast\varphi(T_N^\ast)
\to
\EE\varphi\!\left((\Pi_{H_z}Z)^\top\Sigma^+\Pi_{H_z}Z\right)
\]
in probability, where \(\EE^\ast\) denotes conditional expectation given the
record.  This is conditional weak convergence in probability.  The quantile
conclusion is the standard consequence of weak convergence at a continuity
point of the limiting distribution function; the final coverage statement
then follows from the convergence of \(T_N\) and the random quantile
\(c_{1-\alpha,N}^\ast\).
\end{proof}

\begin{theorem}[Consolidated finite-record asymptotics for the killed-leakage sampled-boundary record]\label{thm:finite-record-inference-audit-main}\label{prop:three-coordinate-vector-batch-main}\label{lem:three-coordinate-batch-rate-main}\label{cor:finite_inverse_delta_boundary_main}\label{cor:exchange-diagonal-projected-gaussian-confidence-main}\label{cor:projected-symmetric-wald-main}\label{cor:finite-separated-root-rate-chart-main}
Let \((A_n)\) be one fixed sampled boundary-reset renewal record, observed
either in stationarity or after discarding a deterministic finite-start burn-in
\(L_N\to\infty\), \(L_N=o(\sqrt N)\).  Assume the following conditions.
\begin{enumerate}[label=\textup{(\roman*)},leftmargin=2.4em,itemsep=0.2em]
\item The gap law has an exponential moment:
\[
\EE e^{\eta G}<\infty
\]
for some \(\eta>0\).
\item The reward dimension \(d_{\rm rew}\) and look-ahead \(q_{\rm lag}\) are fixed.  The centered
Palm cycle reward has the form
\[
Y_j=f(G_j,\ldots,G_{j+q_{\rm lag}})\in\RR^{d_{\rm rew}},\qquad
\EE Y_j=0,
\]
with
\[
\|f(g_0,\ldots,g_{q_{\rm lag}})\|
\le C_f\left(1+\sum_{i=0}^{q_{\rm lag}}g_i\right).
\]
\item The separated batch length is deterministic,
\[
b_N=\lfloor N^\beta\rfloor,\qquad 0<\beta<1,
\]
and \(q_{\rm lag}\) complete cycle rewards are omitted between adjacent batches.
\item If estimated centering is used, the rewards entering the batches satisfy,
uniformly over all complete cycles used by the estimator,
\[
\widehat Y_j-Y_j=\Delta_NL_j,\qquad
L_j=G_j+1,\qquad
\Delta_N=O_{\PP}(N^{-1/2}).
\]
For vector plug-in centering this condition is understood componentwise.
\end{enumerate}
Write \(K_N=\max\{k:\tau_k<N\}\) for the number of complete observed cycles,
put \(d_N=b_N+q_{\rm lag}\), and use only all-complete-cycle indices \(J_N\) for which
every local window entering \(Y_j\) is observed.  Let
\[
M_N=\max\{m\ge0: (m-1)d_N+b_N-1\in J_N\},
\qquad
I_{\ell,N}=\{(\ell-1)d_N,\ldots,(\ell-1)d_N+b_N-1\}.
\]
For \(B_{\ell,N}=\sum_{j\in I_{\ell,N}}Y_j\) and
\(\bar B_N=M_N^{-1}\sum_{\ell=1}^{M_N}B_{\ell,N}\), define on
\(\{M_N\ge2\}\)
\begin{equation}\label{eq:all-cycle-batch-cov-main}
\widehat\Sigma_N^{\mathrm{all}}
=\frac{K_N}{N}\,
\frac1{b_N(M_N-1)}
\sum_{\ell=1}^{M_N}(B_{\ell,N}-\bar B_N)(B_{\ell,N}-\bar B_N)^\top .
\end{equation}
Then \(M_N\ge2\) with probability tending to one and
\[
\sqrt N\,\widehat\theta_N
\Rightarrow
\mathcal N(0,\Sigma),
\qquad
\Sigma=\frac{\mathcal T_{q_{\rm lag}}}{\mu},\qquad
\mu=\EE(G+1),
\]
where \(\widehat\theta_N=N^{-1}\sum_{j\in J_N}Y_j+o_{\PP}(N^{-1/2})\) is the
corresponding complete-cycle average and
\[
\mathcal T_{q_{\rm lag}}
=\Var(Y_0)+
\sum_{h=1}^{q_{\rm lag}}\{\operatorname{Cov}(Y_0,Y_h)+\operatorname{Cov}(Y_h,Y_0)\}.
\]
The all-complete-cycle covariance estimator with plug-in centering is
consistent.  Define the covariance-estimator rate exponent
\[
\omega_\beta:=\min\{\beta,(1-\beta)/2\}.
\]
Equivalently, the following displayed bound is
\(O_{\PP}(N^{-\omega_\beta})\):
\[
\|\widehat\Sigma_N^{\mathrm{all}}-\Sigma\|_{\operatorname{op}}
=
O_{\PP}\left(N^{-\beta}+N^{-(1-\beta)/2}\right).
\]
Consequently, if a rank-adaptive covariance is reported, the deterministic
threshold \(t_N\) must satisfy
\[
\|\widehat\Sigma_N^{\mathrm{all}}-\Sigma\|_{\operatorname{op}}=o_{\PP}(t_N).
\]
A concrete admissible choice is
\[
t_N=N^{-\alpha},
\qquad
0<\alpha<\omega_\beta;
\]
for the balanced choice \(b_N=\lfloor N^{1/3}\rfloor\), this becomes any
\(0<\alpha<1/3\), for instance \(t_N=N^{-1/4}\).

For the three inclusion coordinates in
\Cref{def:cycle-index-inclusion-estimator-main}, take \(d_{\rm rew}=3\), \(q_{\rm lag}=2\), and
\[
Y_j^{(m)}=U_j^{(m)}-r_mL_j,\qquad
r_m=\PP(A_0=A_m=1),\qquad m=0,1,2 .
\]
Then
\[
\sqrt N(\widehat r_N^{\mathrm{cyc}}-r)
\Rightarrow \mathcal N(0,\Sigma_r),
\]
and the same separated-batch estimator, using the plug-in rewards from
\Cref{def:cycle-index-inclusion-estimator-main}, estimates \(\Sigma_r\) at the
rate above.

On the population domain
\[
r_0>0,\qquad r_1/r_0\ne r_0,
\]
define
\[
a(r)=\frac{r_1}{r_0},\qquad
u_2(r)=\frac{r_2}{r_0},\qquad
\lambda(r)=\frac{u_2(r)-r_0}{a(r)-r_0},
\]
and
\begin{equation}\label{eq:finite-inverse-delta-map-main}
\Phi(r)=
\bigl(1-a(r)+\lambda(r),\,
r_0(1-\lambda(r))-a(r)+\lambda(r)\bigr).
\end{equation}
If the inverse is evaluated on a stability neighbourhood where
\(r_0\) and \(|a(r)-r_0|\) are bounded away from zero and the same bounds hold
for \(\widehat r_N\) with probability tending to one, then
\[
\sqrt N(\Phi(\widehat r_N)-\Phi(r))
\Rightarrow \mathcal N(0,\Sigma_s),
\qquad
\Sigma_s=D\Phi(r)\Sigma_rD\Phi(r)^\top .
\]
The reported symmetric-coordinate estimator is the feasible projection
\[
\widetilde s_N=\Pi_F(\Phi(\widehat r_N)),
\qquad
F=\{(x+y,xy):0\le x,y\le1\}.
\]
At separated roots the projection is asymptotically inactive.  If, in
addition, an external label chooses which separated root is the attempt root
and which is the recovery root, then the labelled roots and rates are obtained
from the local chart
\[
(\widehat p_N,\widehat s_N)
=\mathcal R_{\mathrm{root}}(\Phi(\widehat r_N)),\qquad
(\widehat\Gamma_N,\widehat\kappa_{\mathrm r,N})
=\left(-\frac{\log\widehat p_N}{\Delta\tau},
       -\frac{\log\widehat s_N}{\Delta\tau}\right).
\]
With \(J_{\Gamma,\kappa}\) as in
\Cref{lem:separated-root-rate-delta-main},
\[
\sqrt N\{(\widehat\Gamma_N,\widehat\kappa_{\mathrm r,N})
-(\Gamma,\kappa_{\mathrm r})\}
\Rightarrow
\mathcal N(0,\Sigma_{\Gamma,\kappa}),
\qquad
\Sigma_{\Gamma,\kappa}
=J_{\Gamma,\kappa}(r)\Sigma_rJ_{\Gamma,\kappa}(r)^\top .
\]
The plug-in covariance
\[
\widehat\Sigma_{\Gamma,\kappa,N}
=\widehat J_{\Gamma,\kappa,N}
\,\widehat\Sigma_{r,N}^{\mathrm{all}}\,
\widehat J_{\Gamma,\kappa,N}^{\top}
\]
is consistent with the same operator-norm rate on every retained
separated-root chart.  If a deterministic rank threshold \(t_N\) satisfies
\(\|\widehat\Sigma_{\Gamma,\kappa,N}-\Sigma_{\Gamma,\kappa}\|_{\operatorname{op}}
=o_{\PP}(t_N)\), then the labelled-rate retained-rank Wald statistic converges
to \(\chi_k^2\), \(k=\operatorname{rank}\Sigma_{\Gamma,\kappa}\).  No labelled
rate interval is asserted without this separated-root chart and external
branch label.

At an exchange-diagonal point \(s_z=(2z,z^2)\), \(0<z<1\),
\[
\sqrt N(\widetilde s_N-s_z)
\Rightarrow
\Pi_{H_z}(Z),\qquad
Z\sim\mathcal N(0,\Sigma_s),\qquad
H_z=\{h\in\RR^2:h_2\le zh_1\}.
\]
Thus ordinary chi-square reporting is asserted only at separated interior
points and only on the nondegenerate retained covariance range.  At a true
exchange-diagonal point the usable boundary confidence construction is the
projected-Gaussian half-space inversion of
\Cref{lem:projected-gaussian-bootstrap-halfspace-main}: with
\(\widehat\Sigma_{s,N}^{+}\) formed after the deterministic threshold \(t_N\),
define
\begin{equation}\label{eq:boundary-projected-statistic-main}
Q_N(z)=
N(\widetilde s_N-s_z)^\top\widehat\Sigma_{s,N}^{+}
(\widetilde s_N-s_z),
\end{equation}
and take \(c_{1-\alpha,N}(z)\) as the lower conditional \((1-\alpha)\)-quantile
of
\[
\Pi_{H_z}(Z_N^\ast)^\top\widehat\Sigma_{s,N}^{+}\Pi_{H_z}(Z_N^\ast),
\qquad
Z_N^\ast\mid A_0,\ldots,A_{N-1}\sim
\mathcal N(0,\widehat\Sigma_{s,N}).
\]
The diagonal set
\begin{equation}\label{eq:boundary-projected-confidence-set-main}
\{(2z,z^2):0<z<1,\ Q_N(z)\le c_{1-\alpha,N}(z)\}
\end{equation}
has first-order projected-Gaussian coverage at the true diagonal point
\((2z,z^2)\) whenever the limiting projected statistic has a continuous
\((1-\alpha)\)-quantile.

Studentized scalar intervals are claimed only under a nondegenerate limiting
studentization variance.  For example, the lag-two covariance interval below
requires \(\sigma_{\gamma_2}^2>0\); if the relevant limiting variance is zero,
only the degenerate Gaussian limit is asserted.
\end{theorem}

\begin{proof}
The exponential moment gives all moments of \(L_j=G_j+1\) and, for each fixed
\(c\),
\[
\max_{0\le j\le cN}L_j=O_{\PP}(\log N).
\]
The initial incomplete cycle, terminal incomplete cycle, and the finitely many
look-ahead cycles removed by the all-complete-cycle convention therefore
contribute \(O_{\PP}(\log N)=o_{\PP}(\sqrt N)\) to any centered
\(\sqrt N\)-scaled complete-cycle sum.  Since \(Y_j\) is a function of
\((G_j,\ldots,G_{j+q_{\rm lag}})\), the sequence \((Y_j)\) is centered and
\(q_{\rm lag}\)-dependent.  The growth bound and the exponential moment give a finite
second moment.  The multivariate CLT for fixed-range dependent sequences gives
\[
m^{-1/2}\sum_{j=0}^{m-1}Y_j
\Rightarrow \mathcal N(0,\mathcal T_{q_{\rm lag}})
\]
for deterministic \(m\to\infty\).  The renewal law gives
\(K_N/N\to1/\mu\).  If \(m_N=\lfloor N/\mu\rfloor\), then
\(|K_N-m_N|=O_{\PP}(\sqrt N)\); the variance of the extra centered
\(q_{\rm lag}\)-dependent partial sum is \(O_{\PP}(|K_N-m_N|)\), so divided by \(N\) it
vanishes in probability.  Anscombe's replacement of \(m_N\) by \(K_N\) yields
\[
\sqrt N\,\widehat\theta_N
\Rightarrow \mathcal N(0,\mathcal T_{q_{\rm lag}}/\mu).
\]

For covariance estimation, first use true centering and the deterministic
cycle prefix \(n_N=\lfloor N/(\mu(b_N+q_{\rm lag}))\rfloor\).  With
\[
V_{\ell,N}=b_N^{-1/2}B_{\ell,N},
\]
the omitted \(q_{\rm lag}\) rewards make the \(V_{\ell,N}\)'s independent within each
triangular row.  The fourth moments are uniformly bounded by the exponential
moment and the fixed reward dimension.  Hence the empirical second moment of
the deterministic-prefix row differs from its expectation by
\[
O_{\PP}(n_N^{-1/2})=O_{\PP}\!\left(\sqrt{\frac{b_N}{N}}\right),
\]
and the empirical-mean subtraction contributes only \(O_{\PP}(n_N^{-1})\).
The fixed \(q_{\rm lag}\)-dependence calculation gives
\[
\EE[V_{\ell,N}V_{\ell,N}^{\top}]
=
\Var(Y_0)+\sum_{h=1}^{q_{\rm lag}}
\left(1-\frac{h}{b_N}\right)
\{\operatorname{Cov}(Y_0,Y_h)+\operatorname{Cov}(Y_h,Y_0)\},
\]
so the finite-batch bias relative to \(\mathcal T_{q_{\rm lag}}\) is \(O(b_N^{-1})\).

The random number \(M_N\) of used batches is not external: it is determined by
the same gaps and rewards.  \Cref{lem:calendar-stopped-batch-cov-main} is used
precisely at this point.  Its calendar-stopped conclusion, applied with
\(R_N=M_N\) and \(d_N=b_N+q_{\rm lag}\), gives directly
\[
\|\widehat\Sigma_N^{\mathrm{all}}-\Sigma\|_{\operatorname{op}}
=O_{\PP}\left(b_N^{-1}+\sqrt{\frac{b_N}{N}}\right).
\]
This use of the lemma includes the rate
\(M_N/(N/\{\mu(b_N+q_{\rm lag})\})-1=O_{\PP}(\sqrt{b_N/N})\), the endpoint cycles
omitted by the all-complete-cycle rule, the random-prefix change in the
empirical second moment, and the empirical-mean subtraction.  No independence
between \(M_N\) and the batch sums is assumed.

It remains to control the plug-in replacement.  In each used batch,
\[
\sum_{j\in I_{\ell,N}}(\widehat Y_j-Y_j)
=\Delta_N\sum_{j\in I_{\ell,N}}L_j .
\]
Because \(b_N^{-1}\sum_{j\in I_{\ell,N}}L_j=O_{\PP}(1)\) uniformly in the
empirical average and \(\Delta_N=O_{\PP}(N^{-1/2})\), the cross term in the
sample covariance after the \(1/b_N\) normalization is
\(O_{\PP}(\sqrt{b_N/N})\), and the square term is \(O_{\PP}(b_N/N)\).  These
terms are no larger than the already displayed rate when \(b_N=\lfloor
N^\beta\rfloor\), \(0<\beta<1\).  Substituting \(b_N=\lfloor N^\beta\rfloor\)
therefore gives
\[
O_{\PP}\left(N^{-\beta}+N^{-(1-\beta)/2}\right).
\]
The threshold claim is immediate from Weyl's eigenvalue inequality: every
positive limiting eigenvalue stays above \(t_N\), and every zero limiting
eigenvalue falls below \(t_N\), with probability tending to one whenever
\(\|\widehat\Sigma_N^{\mathrm{all}}-\Sigma\|_{\operatorname{op}}=o_{\PP}(t_N)\).

For the inclusion coordinates, the half-open cycle
\([\tau_j,\tau_{j+1})\) contains only the click at \(\tau_j\).  Thus
\(U_j^{(m)}\) counts the cycle times whose \(m\)-step look-ahead is also a
click.  For \(m=0,1,2\), this is a function of
\((G_j,\ldots,G_{j+m})\), and
\[
|U_j^{(m)}-r_mL_j|\le 1+r_mL_j\le2+G_j.
\]
Palm inversion gives \(\EE U_j^{(m)}=r_m\EE L_j\), so these rewards are
centered.  Hence the preceding vector theorem applies with
\(d_{\rm rew}=3\), \(q_{\rm lag}=2\).
The plug-in condition follows from
\(\widehat r_{m,N}^{\mathrm{cyc}}-r_m=O_{\PP}(N^{-1/2})\), proved by the same
CLT just applied to the \(m\)th coordinate.

On the inverse stability neighbourhood, \(\Phi\) is \(C^1\).  The delta method
therefore gives the displayed Gaussian limit for \(\Phi(\widehat r_N)\), and
the sandwich covariance inherits the same operator-norm rate because
\(D\Phi(\widehat r_N)\) is bounded and converges in probability to
\(D\Phi(r)\).  At separated roots, \(\Phi(r)\) lies in the relative interior
of the feasible set \(F\), so the projection is locally inactive with
probability tending to one.  On a labelled separated-root branch,
\Cref{lem:separated-root-rate-delta-main} applies to this same Gaussian limit
and gives the labelled-root, labelled-rate, plug-in covariance, and retained-rank
Wald conclusions.  At \(s_z=(2z,z^2)\), the local projection and half-space
limit are exactly \Cref{lem:local-projection-exchange-diagonal-main,lem:exchange-diagonal-projection-main}.
\Cref{lem:projected-gaussian-bootstrap-halfspace-main}, applied with
\(X_N=\sqrt N(\widetilde s_N-s_z)\), \(z_N=z\), and
\(\widehat\Sigma_N=\widehat\Sigma_{s,N}\), gives the conditional
projected-Gaussian quantile convergence and the displayed boundary coverage
claim.  The same spectral-threshold argument gives the rank-adaptive
chi-square statement at separated interior points, where the feasible
projection is asymptotically inactive.  Scalar studentization divides by a
consistent positive variance estimate; hence it is asserted only when the
limiting variance is nondegenerate.

The finite-start case follows because the hidden chain is primitive.  After
discarding \(L_N\to\infty\), its law is \(o(1)\) in total variation from
stationarity, and deleting \(L_N=o(\sqrt N)\) deterministic observations is
negligible on the centered \(\sqrt N\) scale.
\end{proof}

\begin{lemma}[Local projection at the exchange diagonal]\label{lem:local-projection-exchange-diagonal-main}
Let
\[
F=\{(x+y,xy):0\le x,y\le1\},\qquad s_z=(2z,z^2),\quad 0<z<1.
\]
There is \(\varepsilon>0\) such that Euclidean projection onto \(F\) is
single-valued on \(B_\varepsilon(s_z)\).  In local coordinates
\(v=s_z+h/\sqrt N\), feasibility is
\[
h_2\le zh_1+\frac{h_1^2}{4\sqrt N}.
\]
Hence the scaled feasible sets converge locally to the half-space
\[
T_{s_z}F=\{h:h_2\le zh_1\}.
\]
\end{lemma}

\begin{proof}
Choose \(\delta>0\) so that \(2z-\delta>0\), \(2z+\delta<2\), and the other
box boundaries of \(F\) are outside \(B_\delta(s_z)\).  In this ball the root
constraints \(0\le x,y\le1\) are inactive except for the exchange
discriminant.  Hence
\[
F\cap B_\delta(s_z)
=\{(s_1,s_2)\in B_\delta(s_z):s_2\le s_1^2/4\}.
\]
Let \(\Gamma_z=\{(t,t^2/4):|t-2z|<\delta\}\).  This \(C^2\) arc has a tubular
neighbourhood on which nearest-point projection onto \(\Gamma_z\) is
single-valued; equivalently, the arc has positive local reach
\cite{Shapiro1987MetricProjection}.  Shrinking \(\delta\) if necessary, a
point below the parabola is already in \(F\), while a point above the parabola
and sufficiently close to \(s_z\) has its nearest feasible point on
\(\Gamma_z\), not on any remote part of \(F\).  This gives
\(\varepsilon>0\) such that the Euclidean projection onto the full, globally
nonconvex set \(F\) is single-valued on \(B_\varepsilon(s_z)\).

For \(v=s_z+h/\sqrt N\), the local feasibility condition is
\[
z^2+\frac{h_2}{\sqrt N}
\le
\frac{(2z+h_1/\sqrt N)^2}{4},
\]
which is exactly
\[
h_2\le zh_1+\frac{h_1^2}{4\sqrt N}.
\]
On every bounded set of \(h\)'s the last term converges uniformly to \(0\),
so the scaled feasible sets converge locally to
\(T_{s_z}F=\{h:h_2\le zh_1\}\).
\end{proof}

\begin{lemma}[Projection and root splitting at the exchange diagonal]\label{lem:exchange-diagonal-projection-main}
If \(y_N=s_z+Z_N/\sqrt N+o_{\PP}(N^{-1/2})\) and
\(Z_N\Rightarrow Z\), then
\[
\sqrt N\{\Pi_F(y_N)-s_z\}\Rightarrow \Pi_{T_{s_z}F}(Z).
\]
For the unordered roots of \(t^2-s_1t+s_2=0\), the corresponding root
splitting is \(N^{-1/4}\) at the diagonal unless the half-space limit lies on
the diagonal direction.
\end{lemma}

\begin{proof}
\Cref{lem:local-projection-exchange-diagonal-main} gives local uniqueness and
Painleve-Kuratowski convergence of the scaled feasible sets to the closed
convex half-space \(T_{s_z}F\).  Continuity of projection under this local
positive-reach limit gives the displayed weak convergence by the continuous
mapping theorem.  The discriminant of the projected coordinate is
\((s_1-2z)^2-4(s_2-z^2)\); after projection it is nonnegative and is typically
of order \(N^{-1/2}\), so the two roots split at order \(N^{-1/4}\).
\end{proof}

\begin{corollary}[Exchange-diagonal self-dual inverse]\label{cor:critical_self_dual_inverse}
At the exchange diagonal \(p=s\), the visible quotient inverse returns the
self-dual unordered pair \(\{p,p\}\), and root splitting is governed by
\Cref{lem:exchange-diagonal-projection-main}.  This exchange-diagonal
self-duality is not, by itself, the exact-dependence critical point: along
\(p=s=e^{-\Gamma\Delta\tau}=e^{-\kappa_{\mathrm r}\Delta\tau}\),
\(\lambda_{\mathrm{hid}}=0\) only at
\(\Gamma\Delta\tau=\kappa_{\mathrm r}\Delta\tau=1\), equivalently
\(p=s=e^{-1}\).
\end{corollary}

\begin{proposition}[Hidden-label exchange non-identifiability]\label{prop:hidden_label_exchange_nonidentifiability}
No statistic of one visible sampled record can distinguish
\((\Gamma,\kappa_{\mathrm r})\) from
\((\kappa_{\mathrm r},\Gamma)\) without external labelling information.
\end{proposition}

\begin{proof}
The two parameter points induce the same finite-dimensional visible law by
\Cref{thm:toeplitz_image_inverse_duality}; hence every visible-record
statistic has the same distribution under them.  This is the exchange
specialization inside the present sampled D-MAP(2) family, consistent with the
broader MAP2 lesson that a hidden arrival representation need not be uniquely
identified from the stationary observable arrival law
\cite{RamirezCoboLilloWiper2010MAP2Nonidentifiability}.
\end{proof}

\subsection{Fixed-setting long-record lag-two covariance CLT}

\begin{theorem}[Finite-lag renewal-reward CLT and separated-batch variance]\label{thm:finite-lag-renewal-reward-main}\label{def:cycle-block-main}\label{lem:cycle-one-dep-main}\label{lem:renewal-reward-boundary-main}\label{lem:renewal-local-clt-main}\label{lem:batch-var-random-count-main}\label{lem:deterministic-prefix-main}\label{lem:lag2-boundary-main}\label{thm:cycle-block-main}
Let \((A_n)\) be a stationary renewal indicator process with i.i.d. gaps
satisfying \(\PP(G\ge k)\le Ce^{-\eta k}\).  Let
\(Z_n\in\RR^{d_{\rm rew}}\) be a bounded centered local observable, with fixed
reward dimension \(d_{\rm rew}\) and fixed nonnegative look-ahead
\(q_{\rm lag}\), whose Palm cycle reward is
\[
Y_j=f(G_j,\ldots,G_{j+q_{\rm lag}})\in\RR^{d_{\rm rew}},\qquad
\|f\|\le C_f\left(1+\sum_{i=0}^{q_{\rm lag}}G_{j+i}\right).
\]
Put \(\mu=\EE(G+1)\) and
\[
\mathcal T_{q_{\rm lag}}=
\Var(Y_0)+
\sum_{h=1}^{q_{\rm lag}}\{\operatorname{Cov}(Y_0,Y_h)+\operatorname{Cov}(Y_h,Y_0)\}.
\]
Then the exact calendar-time empirical vector average satisfies
\[
\sqrt N\left(\frac1N\sum_{n=0}^{N-1}Z_n\right)
=\frac1{\sqrt N}\sum_{n=0}^{N-1}Z_n
\Rightarrow
\mathcal N\left(0,\frac{\mathcal T_{q_{\rm lag}}}{\mu}\right).
\]
The all-complete-cycle separated-batch estimator
\eqref{eq:all-cycle-batch-cov-main}, with \(b_N\to\infty\), \(b_N/N\to0\), and
\(q_{\rm lag}\) omitted rewards between batches, is consistent for the same
covariance \(\mathcal T_{q_{\rm lag}}/\mu\).  In the scalar case this reduces
to \(\tau_{q_{\rm lag}}^2/\mu\), where
\[
\tau_{q_{\rm lag}}^2=\Var(Y_0)+2\sum_{h=1}^{q_{\rm lag}}\operatorname{Cov}(Y_0,Y_h).
\]
The same conclusion holds after the finite-start burn-in in
\Cref{thm:finite-record-inference-audit-main}.
\end{theorem}

\begin{proof}
Write the click epochs as \(\tau_j\), put
\(L_j=\tau_{j+1}-\tau_j=G_j+1\), and decompose the calendar sum into complete
cycles.  By the definition of the Palm reward,
\[
Y_j=\sum_{n=\tau_j}^{\tau_{j+1}-1}Z_n
\]
after the finitely many endpoint terms needed by the \(q_{\rm lag}\)-cycle look-ahead
have been assigned to adjacent cycles.  The record segment not covered by
complete observed cycles consists of an initial incomplete cycle, a terminal
incomplete cycle, and at most \(q_{\rm lag}\) additional look-ahead cycles.  Since the
gap tail is exponential,
\(\max_{0\le j\le cN}L_j=O_{\PP}(\log N)\) for each fixed \(c\); hence those
endpoint contributions are \(O_{\PP}(\log N)=o_{\PP}(\sqrt N)\).  Therefore
\[
\sum_{n=0}^{N-1}Z_n
=\sum_{j=0}^{K_N-q_{\rm lag}-1}Y_j+o_{\PP}(\sqrt N),
\qquad
K_N=\max\{k:\tau_k<N\}.
\]

The variables \(Y_j=f(G_j,\ldots,G_{j+q_{\rm lag}})\) are centered and \(q_{\rm lag}\)-dependent,
because the gaps are i.i.d. and disjoint gap blocks are independent once their
cycle indices are more than \(q_{\rm lag}\) apart.  The displayed growth bound and the
exponential tail give a finite second moment.  The multivariate CLT for
centered \(q_{\rm lag}\)-dependent sequences, equivalently the scalar CLT applied by
Cramer--Wold, gives, for deterministic \(m\to\infty\),
\[
\frac1{\sqrt m}\sum_{j=0}^{m-1}Y_j
\Rightarrow
\mathcal N(0,\mathcal T_{q_{\rm lag}}),
\qquad
\mathcal T_{q_{\rm lag}}=\Var(Y_0)+
\sum_{h=1}^{q_{\rm lag}}\{\operatorname{Cov}(Y_0,Y_h)+\operatorname{Cov}(Y_h,Y_0)\}.
\]
The renewal law gives \(K_N/N\to1/\EE L\), where \(L=G+1\).  Anscombe's
random-index theorem, or the equivalent maximal inequality for
\(q_{\rm lag}\)-dependent partial sums, then replaces \(m\) by \(K_N-q_{\rm lag}\) and yields
\[
\frac1{\sqrt N}\sum_{n=0}^{N-1}Z_n
\Rightarrow
\mathcal N\left(0,\frac{\mathcal T_{q_{\rm lag}}}{\EE L}\right).
\]

For the variance estimator, form cycle-index batches of length \(b_N\) and
leave \(q_{\rm lag}\) rewards between consecutive batches.  The omitted rewards make the
gap blocks used by different batches disjoint, so the batch sums are
independent within each triangular row.  Their normalized second moments
converge to \(\mathcal T_{q_{\rm lag}}\) by the same \(q_{\rm lag}\)-dependent covariance calculation.
The number of all-complete-cycle batches is calendar-stopped, but
\Cref{lem:calendar-stopped-batch-cov-main} proves that replacing the
deterministic prefix by this random all-complete-cycle prefix changes the
sample covariance by \(o_{\PP}(1)\).  Multiplication by \(K_N/N\to1/\EE L\)
therefore gives consistency for the calendar-time covariance
\(\mathcal T_{q_{\rm lag}}/\EE L\).  The finite-start burn-in case is the final assertion of
\Cref{thm:finite-record-inference-audit-main}: after discarding
\(L_N\to\infty\), \(L_N=o(\sqrt N)\), the remaining initial-law error and the
deleted deterministic prefix are negligible on the \(\sqrt N\) scale.
\end{proof}

\begin{lemma}[Observed lag-two windows in the all-complete-cycle batches]\label{lem:all-cycle-lag-two-windows-main}
For lag-two rewards, using all-complete-cycle batches and leaving one complete
cycle after the last batched cycle ensures every term \(A_nA_{n+2}\) used by
the statistic is observed.
\end{lemma}

\begin{proof}
If \(n\in[\tau_j,\tau_{j+1}-1]\), then \(A_{n+2}\) is determined by the
click pattern up to time \(\tau_{j+1}+1\).  This time lies no later than the
end of cycle \(j+1\): if \(G_j\ge1\), then \(\tau_{j+1}+1\le\tau_{j+2}\);
if \(G_j=0\), then the cycle \(j\) interval contains only \(n=\tau_j\), and
\(A_{\tau_j+2}\) is determined by whether \(G_{j+1}=0\).  Thus every lag-two
term assigned to cycle \(j\) is determined by \(G_j\) and \(G_{j+1}\).  The
all-complete-cycle convention uses only batched indices for which cycle
\(j+1\) is complete and, at the terminal edge, leaves one complete cycle after
the last batched cycle.  Hence no term \(A_nA_{n+2}\) in a used batch reaches
outside the observed record.
\end{proof}

\begin{lemma}[Lag-two reward identity, centering, and one-gap dependence]\label{lem:lag-two-reward-main}
For \(Z_n=(A_n-\rho)(A_{n+2}-\rho)-\gamma_2\), the Palm cycle reward has the
explicit form
\begin{equation}\label{eq:lag-two-cycle-reward-main}
Y_j=f(G_j,G_{j+1}),
\end{equation}
where
\[
f(g,h)=
\mathbf1_{\{g=1\}}+\mathbf1_{\{g=0,\ h=0\}}
-\rho\bigl(1+\mathbf1_{\{g\ge1\}}+\mathbf1_{\{h=0\}}\bigr)
+(\rho^2-\gamma_2)(g+1).
\]
In particular \(\EE Y_j=0\), and \((Y_j)\) is \(1\)-dependent in the cycle
index.
\end{lemma}

\begin{proof}
Work in Palm coordinates with \(\tau_j=0\).  The cycle \(j\) contains the
times \(0,1,\ldots,G_j\), the next click is at \(G_j+1\), and the following
click is at \(G_j+G_{j+1}+2\).  Hence
\[
\sum_{n=0}^{G_j}A_n=1,
\]
while the shifted count in the same cycle is
\[
\sum_{n=0}^{G_j}A_{n+2}
=\mathbf1_{\{G_j\ge1\}}+\mathbf1_{\{G_{j+1}=0\}}.
\]
The lag-two product count is nonzero only when the click at the cycle start
is followed by another click exactly two sampled steps later:
\[
\sum_{n=0}^{G_j}A_nA_{n+2}
=\mathbf1_{\{G_j=1\}}+\mathbf1_{\{G_j=0,\ G_{j+1}=0\}}.
\]
Summing
\((A_n-\rho)(A_{n+2}-\rho)-\gamma_2\) over the cycle therefore gives
\[
\begin{aligned}
Y_j
&=\sum_{n=0}^{G_j}A_nA_{n+2}
-\rho\sum_{n=0}^{G_j}A_n
-\rho\sum_{n=0}^{G_j}A_{n+2}
+(\rho^2-\gamma_2)(G_j+1)  \\
&=f(G_j,G_{j+1}),
\end{aligned}
\]
which is \eqref{eq:lag-two-cycle-reward-main}.

The centering is not an additional assumption.  The stationary renewal record
has \(\EE[(A_0-\rho)(A_2-\rho)-\gamma_2]=0\), and Palm inversion gives
\[
\EE Y_j
=\EE L\,
\EE\bigl[(A_0-\rho)(A_2-\rho)-\gamma_2\bigr]
=0.
\]
Finally, \(Y_j\) is a function only of the adjacent pair
\((G_j,G_{j+1})\).  If \(|j-k|>1\), the pairs
\((G_j,G_{j+1})\) and \((G_k,G_{k+1})\) are disjoint functions of i.i.d. gaps,
so \(Y_j\) and \(Y_k\) are independent.  Thus the reward sequence is
\(1\)-dependent.
\end{proof}

\begin{corollary}[Killed-leakage sampled-boundary fixed-setting lag-two covariance CLT]\label{cor:boundary-reset-cycle-clt-main}\label{thm:sampled-finite-record-main}
For one fixed sampled boundary-reset setting,
define the observed complete-cycle lag-two covariance estimator by
\[
\widehat\gamma_{2,N}
=\frac1N\sum_{j\in J_{2,N}}\sum_{n=\tau_j}^{\tau_{j+1}-1}
(A_n-\widehat\rho_N)(A_{n+2}-\widehat\rho_N),
\qquad
\widehat\rho_N=\frac1N\sum_{n=0}^{N-1}A_n,
\]
where \(J_{2,N}\) contains exactly those all-complete-cycle indices for which
the lag-two windows lie in the observed record.  Then
\[
\sqrt N(\widehat\gamma_{2,N}-\gamma_2)
\Rightarrow \mathcal N(0,\sigma_{\gamma_2}^2),
\]
where \(\sigma_{\gamma_2}^2\) is the regenerative long-run variance of the
lag-two cycle reward.  The all-complete-cycle separated-batch estimator is
consistent.  The additional hypothesis
\[
\sigma_{\gamma_2}^2>0
\]
is required for studentization.  Under this nondegeneracy hypothesis, the
usual studentized fixed-setting interval is valid; if
\(\sigma_{\gamma_2}^2=0\), the Gaussian limit is degenerate and no
nondegenerate studentized interval is asserted.
\end{corollary}

\begin{proof}
\Cref{lem:lag-two-reward-main} identifies the complete-cycle reward for the
true-centered statistic as a centered \(1\)-dependent reward, and
\Cref{lem:all-cycle-lag-two-windows-main} verifies that every lag-two term
used by \(J_{2,N}\) is observed.  Thus the \(q_{\rm lag}=1\) case of
\Cref{thm:finite-lag-renewal-reward-main} applies with
\[
\sigma_{\gamma_2}^2
=\frac{\Var(Y_0)+2\operatorname{Cov}(Y_0,Y_1)}{\EE(G+1)}.
\]
Replacing \(\rho\) by \(\widehat\rho_N\) in the displayed estimator changes
the true-centered average by
\[
-(\widehat\rho_N-\rho)\frac1N
\sum_{j\in J_{2,N}}\sum_{n=\tau_j}^{\tau_{j+1}-1}
\bigl((A_n-\rho)+(A_{n+2}-\rho)\bigr)
+(\widehat\rho_N-\rho)^2\frac1N\sum_{j\in J_{2,N}}L_j .
\]
The first factor \(\widehat\rho_N-\rho\) is \(O_{\PP}(N^{-1/2})\), the
centered average in the middle is \(O_{\PP}(N^{-1/2})\), and
\(N^{-1}\sum_{j\in J_{2,N}}L_j=O_{\PP}(1)\).  Hence the plug-in replacement is
\(o_{\PP}(N^{-1/2})\).  The CLT therefore has the same variance as the
true-centered reward.  The separated-batch consistency is the \(q_{\rm lag}=1\) batch
conclusion of \Cref{thm:finite-lag-renewal-reward-main}.  Studentization
requires the stated additional nondegeneracy
\(\sigma_{\gamma_2}^2>0\); otherwise the Gaussian limit is degenerate and no
nondegenerate studentized interval follows.
\end{proof}

\begin{theorem}[What one killed-leakage sampled-boundary fixed record estimates]\label{thm:one-record-estimates-main}
A single fixed-setting record estimates only quantities of that stationary
visible law: the inclusion coordinates, the projected symmetric coordinates
\((p+s,ps)\), the fixed-setting lag-two covariance, and their regenerative
covariances under the assumptions above.  It does not label the two rates and
does not identify any object outside the visible stationary record law.
\end{theorem}

\begin{proof}
The positive estimands are exactly the CLT targets in
\Cref{thm:finite-record-inference-audit-main,thm:sampled-finite-record-main}.
The non-identifiability statement is
\Cref{prop:hidden_label_exchange_nonidentifiability}.  The theorem chain
starts from \(T_0,T_1\) and therefore supplies no estimator for any externally
defined scalar map or its inverse images.
\end{proof}
