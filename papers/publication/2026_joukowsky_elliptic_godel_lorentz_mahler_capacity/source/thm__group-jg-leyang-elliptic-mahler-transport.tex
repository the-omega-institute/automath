\begin{theorem}[Two-scale logarithmic mean identity on the Joukowsky elliptic boundary]\label{thm:elliptic-two-scale-stokes-mean}
Specialize the algebraic notation of Theorem \ref{thm:pullback-factorization} to complex coefficients, and let \(r>1\). Let \(P(z)\in\CC[z]\) have degree \(n\ge 1\) and satisfy \(P(0)\neq 0\). Set
\[
J_r(e^{i\theta})=r e^{i\theta}+r^{-1}e^{-i\theta},
\qquad
E_r:=J_r(\{e^{i\theta}:\theta\in[0,2\pi)\}),
\]
and let \(\mu_r\) be the pushforward of Haar measure \(\frac{\dd\theta}{2\pi}\) under \(J_r\). Define the circular logarithmic mean functional of radius \(\rho>0\) by
\[
\mathcal M_\rho(P):=\frac1{2\pi}\int_{0}^{2\pi}\log|P(\rho e^{i\theta})|\,\dd\theta.
\]
These circular logarithmic means are interpreted by the global lower-truncated/Lebesgue convention fixed in Section \ref{sec:preliminaries}.
Then one has the exact identity
\[
\boxed{\;
\int_{E_r}\log|Q_r(w)|\,\dd\mu_r(w)
=
\mathcal M_{1}(P)+\mathcal M_{r^{-2}}(P)+n\log r-\log|a_n|.
\;}
\]
\end{theorem}
\begin{proof}
By Theorem \ref{thm:pullback-factorization}, for \(|z|=1\) one has
\[
r^n z^n Q_r\!\left(r z+r^{-1}z^{-1}\right)=P(z)\,P^{\vee}(r^2 z).
\]
Set \(z=e^{i\theta}\) and take logarithmic absolute values, using the lower-truncated logarithmic convention from Section \ref{sec:preliminaries}. Since \(\log|z^n|=0\), this gives
\[
\log\bigl|Q_r(J_r(e^{i\theta}))\bigr|
=
\log|P(e^{i\theta})|+\log|P^\vee(r^2 e^{i\theta})|-n\log r.
\]
Average over \(\theta\in[0,2\pi)\), and identify the left-hand side with \(\int_{E_r}\log|Q_r(w)|\,\dd\mu_r(w)\) by definition of \(\mu_r\). We obtain
\[
\int_{E_r}\log|Q_r(w)|\,\dd\mu_r(w)
=\mathcal M_1(P)+\frac1{2\pi}\int_{0}^{2\pi}\log|P^\vee(r^2 e^{i\theta})|\,\dd\theta-n\log r.
\]
On the other hand, from \(P^\vee(u)=a_n^{-1}u^nP(1/u)\) we have
\[
P^\vee(r^2 e^{i\theta})=a_n^{-1}r^{2n}e^{in\theta}P(r^{-2}e^{-i\theta}),
\qquad
\bigl|P^\vee(r^2 e^{i\theta})\bigr|=|a_n|^{-1}r^{2n}\bigl|P(r^{-2}e^{-i\theta})\bigr|.
\]
Hence
\[
\frac1{2\pi}\int_{0}^{2\pi}\log|P^\vee(r^2 e^{i\theta})|\,\dd\theta
=2n\log r+\mathcal M_{r^{-2}}(P)-\log|a_n|,
\]
and substituting this into the previous identity gives the claim.
\end{proof}

\begin{remark}[Supporting multiplicative form of the elliptic logarithmic mean]\label{rem:elliptic-two-scale-mahler-multiplicative}
Under the hypotheses of Theorem \ref{thm:elliptic-two-scale-stokes-mean},
\[
\exp\!\left(\int_{E_r}\log|Q_r(w)|\,\dd\mu_r(w)\right)
=
|a_n|^{-1}r^n\,\exp(\mathcal M_1(P))\,\exp(\mathcal M_{r^{-2}}(P)).
\]
\end{remark}
\begin{proof}
Exponentiating both sides of Theorem \ref{thm:elliptic-two-scale-stokes-mean} gives the result.
\end{proof}

\begin{theorem}[Transported Jensen shell count for Joukowsky ellipses]\label{thm:transported-jensen-shell-count}
In the setting of Theorem \ref{thm:elliptic-two-scale-stokes-mean}, write the zeros of \(P\) as \(z_1,\dots,z_n\), counted with multiplicity, and set \(s:=\log r>0\). Define
\[
\mathcal F(s):=\int_{E_{e^s}}\log|Q_{e^s}(w)|\,\dd\mu_{e^s}(w).
\]
Then
\[
\mathcal F(s)=\log|a_n|+ns+
\sum_{j=1}^{n}\log\max\{1,|z_j|\}+
\sum_{j=1}^{n}\log\max\{e^{-2s},|z_j|\}.
\]
Consequently, at every \(s>0\) with \(e^{-2s}\ne |z_j|\) for all \(j\),
\[
\frac{\dd\mathcal F}{\dd s}(s)
=n-2\#\{j:\ |z_j|<e^{-2s}\}.
\]
In particular the elliptic transport logarithmic mean detects the radial zero-counting function inside the moving circle \(|z|=e^{-2s}\). If, in addition, \(|a_0|=|a_n|\), then all zeros of \(P\) lie on the unit circle if and only if \(\mathcal F'(s)=n\) for every regular value \(s>0\).
\end{theorem}

\begin{proof}
Jensen's root formula for circular logarithmic means, applied with multiplicities to \(P(z)=a_n\prod_j(z-z_j)\) (see, e.g., \citet[Chap.~1]{EverestWard1999} or \citet{Mahler1962}), gives for every \(\rho>0\),
\[
\mathcal M_\rho(P)=\log|a_n|+
\sum_{j=1}^{n}\log\max\{\rho,|z_j|\}.
\]
Substituting \(\rho=1\) and \(\rho=e^{-2s}\) into Theorem \ref{thm:elliptic-two-scale-stokes-mean} gives the displayed expression for \(\mathcal F(s)\). Away from the finitely many breakpoints \(s=-\frac12\log|z_j|\), the only terms depending nonlinearly on \(s\) are
\[
\log\max\{e^{-2s},|z_j|\},
\]
whose derivative is \(-2\) precisely when \(|z_j|<e^{-2s}\), and is \(0\) precisely when \(|z_j|>e^{-2s}\). This proves the derivative formula and the zero-counting assertion.

If all zeros lie on \(|z|=1\), then \(|a_0|=|a_n|\) and no zero lies inside \(|z|=e^{-2s}\) for \(s>0\), so \(\mathcal F'(s)=n\) at every regular value. Conversely, assume \(|a_0|=|a_n|\) and \(\mathcal F'(s)=n\) for every regular value \(s>0\). The derivative formula implies that no zero satisfies \(|z_j|<e^{-2s}\) for any \(s>0\). Letting \(s\downarrow0\), we obtain \(|z_j|\ge1\) for every \(j\). Since
\[
|a_0|=|a_n|\prod_{j=1}^{n}|z_j|,
\]
the hypothesis \(|a_0|=|a_n|\) forces \(\prod_j|z_j|=1\). With every factor at least \(1\), this is possible only when \(|z_j|=1\) for all \(j\).
\end{proof}

\begin{theorem}[Mahler identity under the unit-circle zero hypothesis]\label{thm:leyang-mahler-transport}
Specialize the algebraic notation of Theorem \ref{thm:pullback-factorization} to complex coefficients, and let \(r>1\). Let \(P(z)\in\CC[z]\) have degree \(n\ge 1\), and assume the classical unit-circle zero condition that all zeros satisfy \(|z_j|=1\). Set
\[
J_r(e^{i\theta})=r e^{i\theta}+r^{-1}e^{-i\theta},
\qquad
E_r:=J_r(\{e^{i\theta}:\theta\in[0,2\pi)\}),
\]
and let \(\mu_r\) be the pushforward of Haar measure \(\frac{\dd\theta}{2\pi}\) under \(J_r\). Then the transported Jensen shell count of Theorem \ref{thm:transported-jensen-shell-count} has no interior shell contribution, and one has the exact identity
\[
\boxed{\;
\int_{E_r}\log|Q_r(w)|\,\dd\mu_r(w)=\log|a_0|+n\log r\; }.
\]
In particular, if \(P\) is monic, then
\[
\int_{E_r}\log|Q_r(w)|\,\dd\mu_r(w)=n\log r.
\]
\end{theorem}

\begin{proof}
By Theorem \ref{thm:pullback-factorization}, for \(|z|=1\) one has
\[
r^n z^n Q_r\!\left(r z+r^{-1}z^{-1}\right)=P(z)\,P^{\vee}(r^2 z).
\]
Taking logarithmic absolute values and integrating over \(|z|=1\) against \(\frac{\dd\theta}{2\pi}\), while observing that \(\log|z|=0\), gives
\[
\int_{E_r}\log|Q_r(w)|\,\dd\mu_r(w)
=\int_{0}^{2\pi}\log|P(e^{i\theta})|\,\frac{\dd\theta}{2\pi}
\ +\ \int_{0}^{2\pi}\log|P^{\vee}(r^2 e^{i\theta})|\,\frac{\dd\theta}{2\pi}
\ -\ n\log r.
\]
Since all zeros of \(P\) lie on the unit circle, Jensen's formula for Mahler measure \citep{Mahler1962,EverestWard1999} yields
\[
\int_{0}^{2\pi}\log|P(e^{i\theta})|\,\frac{\dd\theta}{2\pi}=\log|a_n|.
\]
On the other hand, the leading coefficient of \(P^\vee(u)=a_n^{-1}u^n P(1/u)\) is \(a_0/a_n\), and the zeros of \(P^\vee(r^2 z)\) are \((r^2 z_j)^{-1}\), all of modulus \(r^{-2}<1\).
Applying Jensen's formula once more, we find
\[
\int_{0}^{2\pi}\log|P^{\vee}(r^2 e^{i\theta})|\,\frac{\dd\theta}{2\pi}
=\log|(a_0/a_n) r^{2n}|=\log|a_0/a_n|+2n\log r.
\]
Substituting this into the previous identity gives the stated formula.
\end{proof}

\begin{remark}[Differential reading of the unit-circle Mahler identity]\label{rem:leyang-linear-log-response}
Under the hypotheses of Theorem \ref{thm:leyang-mahler-transport}, define
\[
\mathcal F(r):=\int_{E_r}\log|Q_r(w)|\,\dd\mu_r(w).
\]
Then \(\mathcal F(r)=\log|a_0|+n\log r\), and hence
\[
\frac{\dd \mathcal F}{\dd(\log r)}\equiv n.
\]
This derivative is only a convenient reading of the Mahler identity in Theorem \ref{thm:leyang-mahler-transport}.  The structural assertion is the Jensen shell-counting obstruction in Theorem \ref{thm:transported-jensen-shell-count}.
\end{remark}

\begin{theorem}[Mahler--Jensen unit-circle criterion and off-circle obstruction]\label{thm:mahler-jensen-leyang-rigidity}
\emergencystretch=2em
Let \(P(z)=a_n\prod_{j=1}^{n}(z-z_j)\) be a degree-\(n\) polynomial with
\(P(0)\ne0\), and let \(Q_r\) be its Joukowsky transport polynomial.  For
\(s>0\) put
\[
\mathcal F(s):=\int_{E_{e^s}}\log|Q_{e^s}(w)|\,\dd\mu_{e^s}(w)
\]
in the lower-truncated logarithmic sense of Theorem \ref{thm:elliptic-two-scale-stokes-mean}.  Assume \(|a_0|=|a_n|\).  Then the unit-circle zero condition is characterized purely by the transported Mahler--Jensen increments: the following are equivalent:
\begin{enumerate}[label=\textup{(\roman*)}]
\item every zero of \(P\) lies on \(\TT\);
\item \(\mathcal F(s)-ns\) is constant on \((0,\infty)\);
\item for every \(0<s_1<s_2\),
\[
\mathcal F(s_2)-\mathcal F(s_1)=n(s_2-s_1).
\]
\end{enumerate}
Equivalently, the Jensen deficit measure of the function \(s\mapsto\mathcal F(s)-ns\) is exactly twice the radial zero-counting measure of the reciprocal interior shells.  More concretely, if \(0<s_1<s_2\), then
\[
\boxed{\;
\mathcal F(s_2)-\mathcal F(s_1)-n(s_2-s_1)
=-2\sum_{j=1}^{n}\operatorname{meas}\{s\in(s_1,s_2): |z_j|<e^{-2s}\}.
\;}
\]
Consequently any zero with \(|z_j|<1\) forces a strict negative Jensen--Mahler deficit on every interval crossing the breakpoint \(-\frac12\log|z_j|\).  Thus the Mahler identity of Theorem \ref{thm:leyang-mahler-transport} has no off-circle false positives under the reciprocal modulus condition \(|a_0|=|a_n|\).
\end{theorem}

\begin{proof}
Theorem \ref{thm:transported-jensen-shell-count} gives, with multiplicities,
\[
\mathcal F(s)=\log|a_n|+ns+
\sum_{j=1}^{n}\log\max\{1,|z_j|\}+
\sum_{j=1}^{n}\log\max\{e^{-2s},|z_j|\}.
\]
For each root define
\[
\ell_j(s_1,s_2):=\operatorname{meas}\{s\in(s_1,s_2): |z_j|<e^{-2s}\}.
\]
The function \(s\mapsto\log\max\{e^{-2s},|z_j|\}\) is absolutely continuous away from its single breakpoint and has derivative \(-2\) exactly on the set \(|z_j|<e^{-2s}\) and derivative \(0\) exactly on the complementary regular set.  Integrating this derivative from \(s_1\) to \(s_2\) yields
\[
\mathcal F(s_2)-\mathcal F(s_1)-n(s_2-s_1)=-2\sum_{j=1}^{n}\ell_j(s_1,s_2),
\]
which is the displayed deficit formula.

If every zero lies on \(\TT\), then \(|z_j|=1\) for all \(j\), so \(e^{-2s}<1=|z_j|\) for every \(s>0\).  Hence every \(\ell_j(s_1,s_2)\) vanishes, proving both constancy of \(\mathcal F(s)-ns\) and all finite-interval equalities.  Conversely, if all finite-interval equalities hold, then every \(\ell_j(s_1,s_2)\) must vanish on every interval.  Therefore there is no index with \(|z_j|<1\), since such an index would give positive length on intervals meeting \(( -\frac12\log|z_j|,\infty)\).  Thus \(|z_j|\ge1\) for every \(j\).  The hypothesis \(|a_0|=|a_n|\) says \(\prod_j |z_j|=1\), and the inequalities \(|z_j|\ge1\) force \(|z_j|=1\) for all \(j\).  This proves \textup{(iii)} implies \textup{(i)}; \textup{(ii)} and \textup{(iii)} are equivalent because a function is constant exactly when all of its finite increments vanish.  The strict-deficit statement is the same interval formula applied to a zero with \(|z_j|<1\).
\end{proof}

\begin{theorem}[Unit-circle shell separation from one elliptic transport]\label{thm:leyang-elliptic-shell-separation}
Let \(r>1\), and let \(P(z)=\prod_{j=1}^{n}(z-z_j)\) be monic with \(|z_j|=1\), counted with multiplicity.  Let
\[
Q_r(w):=a_0\prod_{j=1}^{n}\bigl(w-rz_j-r^{-1}z_j^{-1}\bigr),\qquad a_0=P(0)=(-1)^n\prod_{j=1}^{n}z_j
\]
be the associated Joukowsky transport polynomial, and put
\[
S_r(z):=z^nQ_r\!\left(rz+r^{-1}z^{-1}\right).
\]
Then
\[
\boxed{\;
S_r(z)=r^nP(z)\,z^nP\!\left(r^{-2}z^{-1}\right).
\;}
\]
Consequently the zero divisor of \(S_r\) is the disjoint union
\[
\sum_{j=1}^{n}[z_j]+\sum_{j=1}^{n}[r^{-2}z_j^{-1}],
\]
with the first summand supported on \(|z|=1\) and the second on \(|z|=r^{-2}\).  For every \(\rho\) satisfying \(r^{-2}<\rho<1\), the monic degree-\(n\) factor of \(S_r\) whose zeros lie in \(\{|z|>\rho\}\) is exactly \(P\).  Thus the same elliptic transport polynomial that satisfies the unit-circle Mahler identity above also supplies the shell-selection input for the one-scale inverse theorem, Theorem \ref{thm:leyang-single-scale-reconstruction}.
\end{theorem}

\begin{proof}
By the pullback factorization Theorem \ref{thm:pullback-factorization}, specialized to a monic source polynomial with the stated transport normalization, one has
\[
r^n z^n Q_r\!\left(rz+r^{-1}z^{-1}\right)=P(z)P^{\vee}(r^2z).
\]
Since \(P(z)=\prod_j(z-z_j)\) is monic, its constant coefficient is
\[
a_0=(-1)^n\prod_{j=1}^{n}z_j,
\]
and the reciprocal polynomial in the normalization used here is
\[
P^{\vee}(u)=u^nP(1/u)=\prod_{j=1}^{n}(1-uz_j).
\]
Substituting \(u=r^2z\) gives
\[
P^{\vee}(r^2z)=\prod_{j=1}^{n}(1-r^2zz_j)
=(-1)^n r^{2n}z^n\left(\prod_{j=1}^{n}z_j\right)\prod_{j=1}^{n}\left(1-r^{-2}z^{-1}z_j^{-1}\right).
\]
Using the displayed value of \(a_0\), this is the same as
\[
P^{\vee}(r^2z)=r^{2n}z^nP\!\left(r^{-2}z^{-1}\right).
\]
Dividing the pullback factorization by \(r^nz^n\) and then multiplying by \(z^n\) yields
\[
S_r(z)=z^nQ_r(J_r(z))=r^nP(z)\,z^nP\!\left(r^{-2}z^{-1}\right),
\]
which proves the boxed identity.

The zeros of the first factor are exactly \(z_1,\dots,z_n\), with multiplicity.  The zeros of the second factor satisfy \(r^{-2}z^{-1}=z_j\), hence are exactly \(r^{-2}z_j^{-1}\), with the same multiplicities.  Because \(|z_j|=1\) and \(r>1\), these two zero clusters lie on the disjoint circles \(|z|=1\) and \(|z|=r^{-2}\).  Any \(\rho\) with \(r^{-2}<\rho<1\) separates the two clusters, so the unique monic factor determined by the exterior zero divisor is \(\prod_j(z-z_j)=P\).  The final assertion is this constructive shell separation read as the input required by Theorem \ref{thm:leyang-single-scale-reconstruction}.
\end{proof}
