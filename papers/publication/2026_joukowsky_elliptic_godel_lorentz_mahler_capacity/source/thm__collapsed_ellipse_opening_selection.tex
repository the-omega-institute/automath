At the excluded scale (r=1), the reciprocal collision locus meets
\(\TT^2\), and the ellipse collapses to the segment \([-2,2]\).  The
resulting loss of injectivity changes the equilibrium equality class and
also determines the first-order selection of that class when the ellipse
is reopened.

\begin{theorem}[Collapsed-ellipse equilibrium fiber and opening
selection]\label{thm:collapsed-ellipse-opening-selection}
Let
\[
J_1(z)=z+z^{-1},\qquad \iota(z)=\overline z,
\qquad m_{\TT}=\frac{\dd\theta}{2\pi},
\]
and, for \(\eta\in\mathcal P(\TT)\), define its reflection
symmetrization by
\[
\mathsf S\eta:=\frac12(\eta+\iota_\ast\eta).
\]
Write \(I_{\TT}\) for circle logarithmic energy and use the
lower-truncated extended-energy convention of
Section~\ref{sec:preliminaries}.  Then the following assertions hold.

\begin{enumerate}[label=\textup{(\alph*)},leftmargin=*]
\item \emph{Collapsed energy and complete equality fiber.}  For every
\(\eta\in\mathcal P(\TT)\),
\[
\boxed{\quad I(J_{1\ast}\eta)=2I_{\TT}(\mathsf S\eta).\quad}
\]
Consequently
\[
I(J_{1\ast}\eta)\le 0,
\]
with equality if and only if \(\mathsf S\eta=m_{\TT}\).  Equivalently,
the complete equality fiber consists of the measures
\[
\dd\eta=(1+h)\,\dd m_{\TT},
\qquad h\in L^\infty(m_{\TT};\RR),
\qquad h(\overline z)=-h(z),
\qquad |h|\le1\quad\text{a.e.}
\]
Every member of this fiber has the same image, namely the arcsine
equilibrium measure
\[
J_{1\ast}m_{\TT}
=\frac{\mathbf 1_{(-2,2)}(x)}{\pi\sqrt{4-x^2}}\,\dd x
\]
on \([-2,2]\).

\item \emph{Endpoint Fourier identity.}  If
\(I_{\TT}(\eta)>-\infty\), then
\[
-I(J_{1\ast}\eta)
=2\sum_{k=1}^{\infty}
\frac{\bigl(\operatorname{Re}\widehat\eta(k)\bigr)^2}{k}.
\]
Thus the collapsed segment detects exactly the reflection-symmetric
Fourier component; the reflection-antisymmetric component is precisely
the equality fiber in part \textup{(a)}.

\item \emph{Sharp opening selection.}  Suppose that
\(\mathsf S\eta=m_{\TT}\), and write
\(\dd\eta=(1+h)\dd m_{\TT}\) as in part \textup{(a)}.  For \(s>0\)
put \(r=e^s\) and
\[
\varphi(x):=\frac{1-e^{-x}}{x},\qquad x>0.
\]
Then
\[
\frac{s-I(J_{e^s\ast}\eta)}{2s}
=\sum_{k=1}^{\infty}\varphi(2ks)
\left|\widehat\eta(k)\right|^2,
\]
and hence
\[
\boxed{\quad
\lim_{s\downarrow0}
\frac{s-I(J_{e^s\ast}\eta)}{2s}
=\sum_{k=1}^{\infty}|\widehat\eta(k)|^2
=\frac12\int_{\TT}|h|^2\,\dd m_{\TT}.
\quad}
\]
The limiting selection functional has the sharp range
\[
0\le \frac12\|h\|_{L^2(m_{\TT})}^2\le\frac12.
\]
Its minimum is attained only by \(\eta=m_{\TT}\), while its maximum is
attained, for example, by
\(h(e^{i\theta})=\operatorname{sgn}(\sin\theta)\).

\item \emph{Finite-cover endpoint.}  Let \(B\) be a finite Blaschke
product of positive degree.  For every \(\mu\in\mathcal P(\TT)\),
\[
I((J_1\circ B)_\ast\mu)\le0,
\]
and equality holds if and only if
\[
\mathsf S(B_\ast\mu)=m_{\TT}.
\]
If equality holds and
\(\dd(B_\ast\mu)=(1+h)\dd m_{\TT}\), then
\[
\lim_{s\downarrow0}
\frac{s-I((J_{e^s}\circ B)_\ast\mu)}{2s}
=\frac12\|h\|_{L^2(m_{\TT})}^2.
\]
In particular, the members of the collapsed equality fiber that remain
maximizers at every scale \(r>1\) are exactly the Haar lifts
\(B_\ast\mu=m_{\TT}\).
\end{enumerate}
\end{theorem}

\begin{proof}
For \(z,t\in\TT\), the endpoint factorization is
\[
J_1(z)-J_1(t)=(z-t)(1-z^{-1}t^{-1}),
\]
and \(|1-z^{-1}t^{-1}|=|z-\overline t|\).  Therefore
\[
\log|J_1(z)-J_1(t)|
=\log|z-t|+\log|z-\iota(t)|.
\]
Both summands are bounded above on \(\TT^2\), so the identity may be
integrated in the lower-truncated extended sense without an
indeterminate difference of infinities.  Invariance of the circle
kernel under \(\iota\) and expansion of
\(I_{\TT}(\mathsf S\eta)\) give
\[
I(J_{1\ast}\eta)
=I_{\TT}(\eta)+I_{\TT}(\eta,\iota_\ast\eta)
=2I_{\TT}(\mathsf S\eta).
\]
Here \(I_{\TT}(\alpha,\beta)\) denotes the mutual logarithmic energy
of two circle measures.  Since \(J_1\circ\iota=J_1\), one also has
\(J_{1\ast}\eta=J_{1\ast}(\mathsf S\eta)\).
The circle has capacity one and unique equilibrium measure
\(m_{\TT}\).  Hence \(I_{\TT}(\mathsf S\eta)\le0\), with equality
exactly when \(\mathsf S\eta=m_{\TT}\), proving the energy and equality
claims in part \textup{(a)}.

If \(\mathsf S\eta=m_{\TT}\), then
\(\eta(A)\le2m_{\TT}(A)\) for every Borel set \(A\).  Thus
\(\eta=fm_{\TT}\) for a function \(0\le f\le2\), and the
symmetrization identity is equivalent to
\(f(z)+f(\overline z)=2\) almost everywhere.  Setting \(h=f-1\) gives
the displayed density class.  The converse follows by direct
symmetrization.  Finally, \(J_1(e^{i\theta})=2\cos\theta\), so the
change of variables on the two semicircles gives the stated arcsine
density.

Under the finite-energy hypothesis in part \textup{(b)}, the circle
Fourier identity in Lemma
\ref{lem:circle-log-kernel-fourier-truncation}, applied to
\(\mathsf S\eta\), yields
\[
I_{\TT}(\mathsf S\eta)
=-\sum_{k=1}^{\infty}
\frac{|\widehat{\mathsf S\eta}(k)|^2}{k}.
\]
Since
\[
\widehat{\mathsf S\eta}(k)
=\frac12\left(\widehat\eta(k)+\overline{\widehat\eta(k)}\right)
=\operatorname{Re}\widehat\eta(k),
\]
part \textup{(b)} follows from part \textup{(a)}.

Now assume \(\mathsf S\eta=m_{\TT}\).  The density obtained in part
\textup{(a)} is bounded, so \(\eta\) has finite circle logarithmic
energy, and \(\operatorname{Re}\widehat\eta(k)=0\) for every \(k\ge1\).
Theorem~\ref{thm:finite-energy-joukowsky-fourier-identity}, with
\(r=e^s\), therefore reduces to
\[
s-I(J_{e^s\ast}\eta)
=\sum_{k=1}^{\infty}
\frac{1-e^{-2ks}}{k}|\widehat\eta(k)|^2.
\]
Division by \(2s\) gives the exact formula in part \textup{(c)}.  For
each \(k\), \(\varphi(2ks)\) increases to one as \(s\downarrow0\).
Monotone convergence for the counting measure on \(\NN\) gives the
first equality in the boxed limit.  Since the nonzero Fourier
coefficients of \(h\) are those of \(\eta\), Parseval's identity and
the reality of \(h\) give
\[
\int_{\TT}|h|^2\,\dd m_{\TT}
=2\sum_{k=1}^{\infty}|\widehat\eta(k)|^2.
\]
The bound \(|h|\le1\) proves the upper endpoint \(1/2\); equality is
attained by the displayed odd sign function.  The limit vanishes only
when every nonzero Fourier coefficient of \(h\) vanishes, equivalently
when \(h=0\) almost everywhere.

For part \textup{(d)}, set \(\eta=B_\ast\mu\) and use
\((J_r\circ B)_\ast\mu=J_{r\ast}\eta\) for \(r\ge1\).  Parts
\textup{(a)} and \textup{(c)} give the endpoint equality criterion and
the opening limit.  For every \(r>1\), Theorem
\ref{thm:finite-blaschke-joukowsky-cover} says that equality in the
elliptic equilibrium bound holds exactly when \(B_\ast\mu=m_{\TT}\).
This is the condition \(h=0\), completing the proof.
\end{proof}
