The inverse part of the paper proceeds in visible modules, always from
exact complete coefficient vectors when multiscale reconstruction is
asserted.  First we pull one transported polynomial back to the source
variable and expose its two concentric zero shells.  Then we classify
all one-scale algebraic fibers by the involution \(z\mapsto r^{-2}/z\).
This is a scaled reciprocal zero-flipping ambiguity in the same
structural family as the reciprocal-conjugate choices in classical
one-dimensional phase retrieval and spectral factorization
\citep{Akutowicz1956,Walther1963,Hofstetter1964,Hayes1982,OppenheimSchafer2010}.
Here the additional content is not the existence of a zero-flip
ambiguity by itself, but its exact normalization under the Joukowsky
transport identity, the product resonance imposed by the fully
normalized transport polynomial, and the way two distinct scales turn
the residual choices into a finite matching graph.
Next, two distinct scales give a finite candidate-graph matching
problem whose perfect matchings are exactly the common algebraic
source divisors, with generic injectivity outside an explicit
resonance locus.  Exact complete coefficient-vector Laurent
interpolation then gives the uniform \(n+1\)-scale inverse theorem,
and the unit-circle separation gives one-scale injectivity.  Finally,
Rouch\'e contours provide stability of the annular divisor after the
exact model \((r,P)\) is known.

\subsection{One-Scale Shell Recognition}\label{subsec:one-scale-shell-recognition}

\begin{corollary}[One-scale transport identity under the unit-circle zero hypothesis]\label{cor:leyang-transport-identity}
In the setting of Theorem \ref{thm:pullback-factorization}, assume that \(K\subseteq\CC\), that every zero \(z_j\) of \(P\) satisfies \(|z_j|=1\), and that \(r\in\RR_{>0}\).  Write \(P(z)=a_n\prod_{j=1}^{n}(z-z_j)\), with \(P^\vee\) normalized as in Table~\ref{tab:normalization-ledger}.
Then for every \(z\in\CC^\times\),
\[
Q_r\!\bigl(J_r(z)\bigr)=a_n^{-1}r^n\,P(z)\,P\!\bigl(r^{-2}z^{-1}\bigr).
\]
Although the algebraic identity itself does not use the unit-circle hypothesis, we retain that assumption here because it is needed in the geometric corollaries below.
\end{corollary}

\begin{proof}
By Theorem \ref{thm:pullback-factorization},
\[
r^n z^n\,Q_r\!\bigl(J_r(z)\bigr)=P(z)\,P^{\vee}(r^2 z).
\]
Using Table~\ref{tab:normalization-ledger} and the roots of \(P\),
\[
P^{\vee}(r^2 z)
=\prod_{j=1}^{n}(1-r^2zz_j)
=a_n^{-1}r^{2n}z^n P(r^{-2}z^{-1}).
\]
Cancelling the common factor \(z^n\) yields the claim.
\end{proof}

\begin{corollary}[Double-resultant elimination and concentric spectral shells]\label{cor:leyang-double-resultant}
Retain the hypotheses of Corollary \ref{cor:leyang-transport-identity}, and define
\[
S_r(z):=\operatorname{Res}_w\!\bigl(Q_r(w),\,r z^2-w z+r^{-1}\bigr)\in K(r)[z].
\]
Then
\[
S_r(z)=a_n^{-1} r^{n}\,P(z)\,\Bigl[z^{n}P\!\bigl(r^{-2}z^{-1}\bigr)\Bigr].
\]
Consequently,
\[
\mathrm{Zero}(S_r)
=\{z_1,\dots,z_n\}\cup\{r^{-2}z_1^{-1},\dots,r^{-2}z_n^{-1}\},
\]
where the two clusters lie on the circles \(|z|=1\) and \(|z|=r^{-2}\), respectively.
\end{corollary}

\begin{proof}
Because \(r z^2-w z+r^{-1}=z(J_r(z)-w)\) is linear in \(w\), the Sylvester resultant convention fixed in Section \ref{sec:preliminaries} and used in the standard one-variable resultant normalization \citep[Chap.~12]{GKZ1994} gives
\[
\operatorname{Res}_w\!\bigl(Q_r(w),\,r z^2-w z+r^{-1}\bigr)
=z^n\,Q_r\!\left(\frac{r z^2+r^{-1}}{z}\right)
=z^n Q_r\!\bigl(J_r(z)\bigr).
\]
Substituting Corollary \ref{cor:leyang-transport-identity} yields
\[
S_r(z)=z^n a_n^{-1}r^n P(z)P(r^{-2}z^{-1})
=a_n^{-1}r^{n}P(z)\,\Bigl[z^{n}P\!\bigl(r^{-2}z^{-1}\bigr)\Bigr],
\]
which is the required factorization. The description of the zero set follows immediately from the two factors.
\end{proof}


\begin{definition}[Pulled-back transport shell]\label{def:noiseless-leyang-reconstruction-data}
Fix \(r>1\) and a polynomial \(Q\in\CC[w]\) of degree \(n\ge1\).  Its pulled-back doubled polynomial is
\[
S_{r,Q}(z):=\operatorname{Res}_w\!\bigl(Q(w),\,r z^2-w z+r^{-1}\bigr)=z^nQ(J_r(z)).
\]
A one-scale unit-circle source realization of \((r,Q)\) is a monic polynomial \(P\in\CC[z]\), all of whose zeros lie on \(\TT\), such that \(Q=Q_r(P)\) in the normalization of Table~\ref{tab:normalization-ledger}.  The polynomial \(S_{r,Q}\) is the shell object tested by Theorem \ref{thm:intrinsic-leyang-shell-recognition} and by the Rouch\'e theorem below.  The complete-vector inverse data interface is the ledger convention formalized in Definition \ref{def:transport-coefficient-observations}.
\end{definition}

\begin{theorem}[Exact one-scale unit-circle factor recovery from an exact complete coefficient vector]\label{thm:intrinsic-leyang-shell-recognition}
Let \(e\in\RR\), let \(Q\in\CC[w]\) be known as a complete exact coefficient vector of degree \(n\ge1\), and put \(r=\exp(e)\).  If \(e>0\), define
\[
S_{e,Q}(z):=z^nQ\!\left(rz+r^{-1}z^{-1}\right).
\]
Then a monic unit-circle source polynomial at the scale \(r\) is algebraically identifiable from the exact pair \((e,Q)\) as follows.  The following conditions are equivalent.
\begin{enumerate}[label=\textup{(\roman*)}]
\item \(e>0\), and there exists a monic polynomial \(P\in\CC[z]\), all of whose zeros lie on \(\TT\), such that \(Q=Q_r(P)\) in the normalization of Table~\ref{tab:normalization-ledger}.
\item \(e>0\), and for some \(\rho\) with \(r^{-2}<\rho<1\), the polynomial \(S_{e,Q}\) has exactly \(n\) zeros in \(\{|z|>\rho\}\), counted with multiplicity; if \(P_\rho\) is the monic factor determined by those zeros, then all zeros of \(P_\rho\) lie on \(\TT\) and
\[
\boxed{\;
S_{e,Q}(z)=r^n P_\rho(z)\,z^nP_\rho\!\left(r^{-2}z^{-1}\right).
\;}
\]
\end{enumerate}
When these conditions hold, the factor \(P_\rho\) is independent of \(\rho\in(r^{-2},1)\), and is the unique monic unit-circle source polynomial whose transport at scale \(r\) is \(Q\).  Equivalently, non-realizable pairs are detected by failure of the positive-scale condition, failure of annular zero-shell separation, failure of the unit-circle location of the outer shell, or failure of the displayed compact reciprocal-polynomial identity, whose reflected-shell expansion includes the scalar from Lemma \ref{lem:unit-circle-reciprocal-shell-scalar}.
\end{theorem}

\begin{proof}
Assume first that \(Q=Q_r(P)\) for a monic polynomial \(P\) whose zeros lie on \(\TT\).  Corollary \ref{cor:leyang-double-resultant}, with \(a_n=1\), gives
\[
S_{e,Q}(z)=r^nP(z)\,z^nP\!\left(r^{-2}z^{-1}\right).
\]
The zeros of the first factor lie on \(|z|=1\), while the zeros of the second factor are \(r^{-2}z_j^{-1}\) as \(z_j\) runs through the zeros of \(P\); hence they lie on \(|z|=r^{-2}\).  Therefore every \(\rho\in(r^{-2},1)\) separates the two zero clusters, and the outer monic factor is exactly \(P\).  This proves \textup{(ii)}, with independence of \(\rho\), from \textup{(i)}.

Conversely assume \textup{(ii)}.  Put \(P=P_\rho\).  The hypotheses give a monic polynomial \(P\) whose zeros lie on \(\TT\), and the displayed identity says
\[
z^nQ(J_r(z))=r^nP(z)\,z^nP(r^{-2}z^{-1}).
\]
For the transport polynomial \(Q_r(P)\), Corollary \ref{cor:leyang-transport-identity} gives the same identity:
\[
z^nQ_r(P)(J_r(z))=r^nP(z)\,z^nP(r^{-2}z^{-1}).
\]
Thus
\[
Q(J_r(z))=Q_r(P)(J_r(z))
\]
for every \(z\ne0\).  The set \(J_r(\CC^\times)\) is infinite, so the two polynomials in \(w\) are equal.  Hence \(Q=Q_r(P)\), proving \textup{(i)}.

The uniqueness assertion follows from the same separated-shell argument: any monic unit-circle source producing \(Q\) must be the outer factor of \(S_{e,Q}\) for every \(\rho\in(r^{-2},1)\).  The final sentence is the contrapositive of the equivalence.
\end{proof}

\begin{proposition}[Exact reconstruction from an exact complete one-scale coefficient vector by shell-factor selection]\label{prop:exact-shell-reconstruction}
Fix a one-scale unit-circle reconstruction datum \((r,Q)\) in which \(Q\) is known as its complete exact coefficient vector, and let \(P\) be the corresponding monic unit-circle source polynomial.  Put \(n=\deg_w Q\).  The exact algebraic procedure
\[
S_{r,Q}(z)=\operatorname{Res}_w\!\bigl(Q(w),rz^2-wz+r^{-1}\bigr),
\]
followed by selection of the degree-\(n\) zero sub-divisor supported on \(|z|=1\), equivalently supported in \(|z|>\rho\) for any exact \(\rho\) satisfying \(r^{-2}<\rho<1\), returns the original polynomial \(P\).  Repeated roots are recovered with multiplicity.
\end{proposition}

\begin{proof}
By Corollary \ref{cor:leyang-double-resultant},
\[
S_{r,Q}(z)=r^nP(z)\Bigl[z^nP(r^{-2}z^{-1})\Bigr]
\]
because the source is monic.  The first factor has all zeros on \(|z|=1\), and the second has all zeros on \(|z|=r^{-2}\).  The two shells are disjoint, so the selected degree-\(n\) divisor is exactly the zero divisor of \(P\).  A monic polynomial is uniquely determined by its zero divisor, proving the claim.
\end{proof}

The preceding factor-recovery theorem tests a pair \((e,Q)\) without referring to a hidden source polynomial.  The next stage drops the unit-circle hypothesis and describes the full one-scale ambiguity.  This ambiguity is finite and completely controlled by the reflected pairs of the involution \(\tau_r(z)=r^{-2}/z\).

\subsection{One-Scale Fiber Classification}\label{subsec:one-scale-fiber-classification}

\begin{lemma}[One-scale fiber matching with multiplicities]\label{lem:one-scale-orbit-fiber-matching}
Fix \(r>1\) and put
\[
\tau_r(z):=r^{-2}z^{-1},
\qquad
T_r(z):=rz+r^{-1}z^{-1}
\qquad(z\in\CC^\times).
\]
Let \(P\) be monic of degree \(n\) with zero divisor
\[
D=\sum_{x\in\CC^\times}m_x[x]
\]
and let \(Q_r\) be its transport polynomial with the normalization of Table~\ref{tab:normalization-ledger}.  A monic polynomial \(\widetilde P\) of degree \(n\), with zero divisor
\[
\widetilde D=\sum_{x\in\CC^\times}\widetilde m_x[x],
\]
has the same one-scale transport polynomial \(Q_r\) if and only if the following two conditions hold:
\begin{enumerate}[label=\textup{(\alph*)}]
  \item for every orbit \(O\) of the involution \(\tau_r\) on \(\CC^\times\),
  \[
  \sum_{x\in O}\widetilde m_x=\sum_{x\in O}m_x;
  \]
  \item the source-root product is unchanged,
  \[
  \prod_{x\in\CC^\times}x^{\widetilde m_x-m_x}=1.
  \]
\end{enumerate}
In particular, if the \(2n\) points \(z_j\) and \(r^{-2}/z_j\) are pairwise distinct, then the fiber is obtained by choosing, for each indexed pair \(\{z_j,r^{-2}/z_j\}\), exactly one point, and the choices indexed by \(S\subset\{1,\dots,n\}\) preserve \(Q_r\) exactly when
\[
\prod_{j\in S}z_j^2=r^{-2|S|}.
\]
\end{lemma}

\begin{proof}
For \(u,v\in\CC^\times\), the equation \(T_r(u)=T_r(v)\) is equivalent to
\[
0=r(u-v)+r^{-1}(u^{-1}-v^{-1})=(u-v)\left(r-\frac{r^{-1}}{uv}\right).
\]
Hence \(u=v\) or \(v=r^{-2}/u=\tau_r(u)\).  Thus the fibers of \(T_r\) on \(\CC^\times\) are exactly the \(\tau_r\)-orbits, with the fixed points \(u=\pm r^{-1}\) allowed as one-point orbits.

Writing the actual normalized transport polynomial in divisor form gives
\[
Q_r(w)=(-1)^n\left(\prod_x x^{m_x}\right)
\prod_x\bigl(w-T_r(x)\bigr)^{m_x},
\]
and the same formula with \(m_x\) replaced by \(\widetilde m_x\) holds for \(\widetilde P\).  Equality of the monic factors in \(w\) is therefore exactly equality of the pushed-forward divisors \((T_r)_\ast D=(T_r)_\ast\widetilde D\), which is condition \textup{(a)} by the fiber computation above.  After the monic factors agree, equality of the full normalized polynomials is exactly equality of their leading coefficients, namely
\[
(-1)^n\prod_xx^{m_x}=(-1)^n\prod_xx^{\widetilde m_x},
\]
which is condition \textup{(b)}.  This proves the equivalence.

Under the pairwise-distinct hypothesis every \(\tau_r\)-orbit meeting the source divisor contains exactly one original point \(z_j\) and exactly one reflected point \(r^{-2}/z_j\), each with total orbit multiplicity one.  Condition \textup{(a)} says that a competitor chooses exactly one point from each indexed pair.  If the chosen reflected indices are \(S\), then the root product changes by
\[
\prod_{j\in S}\frac{r^{-2}/z_j}{z_j}
=r^{-2|S|}\prod_{j\in S}z_j^{-2}.
\]
Condition \textup{(b)} says that this factor is one, which is precisely the displayed subset-resonance equation.
\end{proof}


\subsection{Two-Scale Candidate Graphs}\label{subsec:two-scale-candidate-graphs}

From this point through Theorem
\ref{thm:algebraic-mahler-transport-inverse}, all root multisets are
exact divisors in a fixed algebraic closure of the coefficient field.
Constructing \(\Gamma_{r,s}\) means factoring the exact monic transported
polynomials and copying roots with algebraic multiplicity.  No numerical
root-finding, root clustering, or multiplicity-stability assertion is
made here; the only perturbative statement is the later local Rouch\'e
selector after a model and contours have been fixed.

\begin{definition}[Two-scale candidate graph]\label{def:two-scale-candidate-graph}
Let \(r,s>1\) be distinct, and let \(Q_r,Q_s\in\CC[w]\) be complete
degree-\(n\) transported coefficient vectors with nonzero leading
coefficients.  Write \(\widehat Q_r\) and \(\widehat Q_s\) for their
monic normalizations.  The vertices of the two-scale candidate graph
\[
\Gamma_{r,s}(Q_r,Q_s)
\]
are copied root vertices.  Explicitly, write the root divisor of
\(\widehat Q_r\) as \(\sum_{\alpha}m_r(\alpha)[\alpha]\) and the root
divisor of \(\widehat Q_s\) as
\(\sum_{\beta}m_s(\beta)[\beta]\).  The left vertices are the labelled
copies
\[
L_r=\{(\alpha,i):\widehat Q_r(\alpha)=0,\ 1\le i\le m_r(\alpha)\},
\]
and the right vertices are the labelled copies
\[
R_s=\{(\beta,j):\widehat Q_s(\beta)=0,\ 1\le j\le m_s(\beta)\}.
\]
Thus repeated transported roots are not collapsed before the matching
step; only their copy labels record multiplicity.  Put
\[
\Delta_{r,s}:=\frac r s-\frac s r .
\]
For each copied pair \((\alpha,i)\in L_r\) and
\((\beta,j)\in R_s\), define
\[
x(\alpha,\beta):=
\frac{\alpha/s-\beta/r}{\Delta_{r,s}},
\qquad
y(\alpha,\beta):=
\frac{r\beta-s\alpha}{\Delta_{r,s}}.
\]
An edge is drawn between the copied vertices \((\alpha,i)\) and
\((\beta,j)\) exactly when
\[
x(\alpha,\beta)y(\alpha,\beta)=1.
\]
Equivalently, the nonsingular two-scale linear system has recovered
coordinates \((x,y)\) that are mutual inverses, so the pair
\((\alpha,\beta)\) can come from a single nonzero source root.  Thus
the edge condition is a compatibility condition for the two recovered
coordinates, not an additional algebraic assumption on a source that
has already been found.
That edge is labelled by the candidate source root
\[
x=x(\alpha,\beta).
\]
A perfect matching of \(\Gamma_{r,s}(Q_r,Q_s)\) means a set of \(n\)
edges covering every copied left and right root vertex exactly once.
The algebraic output of the graph construction is the multiset
\[
\left\{\sum_{e\in M}[x_e]:
  M\text{ is a perfect matching of }\Gamma_{r,s}(Q_r,Q_s)\right\},
\]
with the final multiset of divisors taken modulo permutations of the
copy labels over each fixed transported value \(\alpha\) and \(\beta\).
Consequently two copied perfect matchings that differ only by
renumbering equal vertices \((\alpha,i)\) or \((\beta,j)\) represent
the same output divisor, while distinct edge-label multisets represent
distinct candidate source divisors.
\end{definition}

\begin{theorem}[Exact two-scale matching inverse from exact complete coefficient vectors]\label{thm:two-scale-matching-inverse}
Let the exact root divisors of the monic normalizations be known in the
algebraic closure.  More explicitly, let \(r,s>1\) be distinct and let
\(Q_r,Q_s\in\CC[w]\) be degree-\(n\) polynomials with nonzero leading
coefficients.  Let \(\widehat Q_r,\widehat Q_s\) be their monic
normalizations, and form
\(\Gamma_{r,s}(Q_r,Q_s)\) as in Definition
\ref{def:two-scale-candidate-graph}.
This is the algebraic-oracle setting of the paragraph preceding
Definition \ref{def:two-scale-candidate-graph}: complete coefficient
vectors are factored exactly over an algebraic closure, with no numerical
root-finding, clustering, or multiplicity-stability claim.

In the projective monic-normalized setting, the common source zero
divisors whose monic transports are \(\widehat Q_r\) and
\(\widehat Q_s\) are exactly the divisors
\[
D_M:=\sum_{e\in M}[x_e]
\]
obtained from perfect matchings \(M\) of
\(\Gamma_{r,s}(Q_r,Q_s)\), where \(x_e\) is the edge label.  These
divisors are listed with the output convention of Definition
\ref{def:two-scale-candidate-graph}: copied matchings are quotiented
by permutations of equal transported root labels before the final
candidate divisor multiset is recorded.  For such a matching the full
transport scalar recovered from the source divisor is
\[
a_0(M)=(-1)^n\prod_{e\in M}x_e .
\]
If transported roots coincide, the copied labels are a bookkeeping
device rather than additional divisor data: several perfect matchings
may differ only by relabelling equal \((\alpha,i)\) or \((\beta,j)\)
vertices and then represent the same source divisor.
Equivalently, the monic source polynomial
\[
P_M(z)=\prod_{e\in M}(z-x_e)
\]
has nonzero constant term and has monic transports
\(\widehat Q_r,\widehat Q_s\).  In this monic-normalized statement
there is no further scalar constraint: covering every copied root
vertex once already makes the transported root multisets agree at both
scales.

For fully normalized transported data, a monic polynomial \(P\) with
nonzero constant term has transported polynomials exactly \(Q_r\) and
\(Q_s\) if and only if there is a perfect matching \(M\) of
\(\Gamma_{r,s}(Q_r,Q_s)\) such that the two observed leading
coefficients are equal and
\[
\operatorname{lc}(Q_r)=\operatorname{lc}(Q_s)
=(-1)^n\prod_{e\in M}x_e .
\]
\end{theorem}

\begin{proof}
We first solve the two-scale equations.  Suppose a source root \(z\)
contributes the roots
\[
\alpha=rz+r^{-1}z^{-1},
\qquad
\beta=sz+s^{-1}z^{-1}
\]
to the two monic transported polynomials.  If \(u=z^{-1}\), then
\[
\begin{pmatrix}
r&r^{-1}\\
s&s^{-1}
\end{pmatrix}
\begin{pmatrix}
z\\ u
\end{pmatrix}
=
\begin{pmatrix}
\alpha\\ \beta
\end{pmatrix}.
\]
The determinant is
\(\Delta_{r,s}=r/s-s/r\), which is nonzero because \(r\ne s\).
Cramer's rule gives
\[
z=\frac{\alpha/s-\beta/r}{\Delta_{r,s}},
\qquad
u=\frac{r\beta-s\alpha}{\Delta_{r,s}}.
\]
Thus the two displayed quantities in Definition
\ref{def:two-scale-candidate-graph} are forced to be \(z\) and
\(z^{-1}\), and in particular their product is \(1\).  Conversely,
for any transported pair \((\alpha,\beta)\), the same Cramer formulas
are the only possible values of a source coordinate and its inverse;
therefore the condition \(x(\alpha,\beta)y(\alpha,\beta)=1\) is
exactly the condition that the recovered coordinates are
\((z,z^{-1})\) for the single source root \(z=x(\alpha,\beta)\).

Now assume that \(P(z)=\prod_{j=1}^{n}(z-z_j)\), with every
\(z_j\ne0\), has monic transports \(\widehat Q_r,\widehat Q_s\).  The
copied left roots are
\[
\alpha_j=rz_j+r^{-1}z_j^{-1},
\]
with multiplicity, and the copied right roots are
\[
\beta_j=sz_j+s^{-1}z_j^{-1},
\]
with multiplicity.  Pair each copied \(\alpha_j\) with the copied
\(\beta_j\) coming from the same indexed source root.  By the preceding
linear-system computation this pair is a valid edge and its label is
\(z_j\).  The \(n\) such pairs cover all copied vertices, so they form
a perfect matching whose divisor is the zero divisor of \(P\).

Conversely let \(M\) be a perfect matching of
\(\Gamma_{r,s}(Q_r,Q_s)\), and write its edge labels as \(x_e\).  Since
each edge satisfies \(x_ey_e=1\), no \(x_e\) is zero and
\(y_e=x_e^{-1}\).  Thus every edge contributes a well-defined nonzero
linear source root.  If the edge \(e\) joins copied vertices
\((\alpha_e,i_e)\) and \((\beta_e,j_e)\), the definitions of
\(x_e,y_e\) use only the underlying transported values
\(\alpha_e,\beta_e\) and solve the same nonsingular linear system, so
substitution back
gives
\[
\alpha_e=rx_e+r^{-1}x_e^{-1},
\qquad
\beta_e=sx_e+s^{-1}x_e^{-1}.
\]
Because \(M\) covers every copied left root exactly once, the multiset
of roots of the monic transport of
\[
P_M(z)=\prod_{e\in M}(z-x_e)
\]
at scale \(r\) is exactly the root multiset of \(\widehat Q_r\).  The
two monic degree-\(n\) polynomials are therefore equal.  The same
argument on the right vertices gives equality with \(\widehat Q_s\) at
scale \(s\).  This proves the projective monic-normalized
classification and also shows that the full scalar determined by the
source divisor is
\[
a_0(M)=(-1)^n\prod_{e\in M}x_e,
\]
as in Table~\ref{tab:normalization-ledger}.

It remains only to record the full-normalized condition.  A genuine
full transport \(Q_t\) has leading coefficient
\((-1)^n\prod z_j\), independent of the scale \(t\).  Hence any full
two-scale source must have equal observed leading coefficients, and
that common coefficient must equal
\((-1)^n\prod_{e\in M}x_e\) for the matching attached to its roots.
Conversely, if a perfect matching satisfies this product condition,
then multiplying both monic transported polynomials of \(P_M\) by the
common scalar gives exactly the observed full polynomials \(Q_r\) and
\(Q_s\).  This proves the fully normalized equivalence.
\end{proof}

\begin{corollary}[Generic squarefree two-scale injectivity]\label{cor:generic-two-scale-injectivity}
Fix \(r,s>1\) with \(r\ne s\), and let
\[
P(z)=\prod_{i=1}^{n}(z-z_i)
\]
be monic with nonzero constant term.  Assume the following explicit
generic squarefree nonresonance hypothesis:
\[
\tag{\(\mathrm{GS}_{r,s}\)}
z_i\ne z_j
\quad\text{and}\quad
z_i z_j\ne r^{-2},\quad
z_i z_j\ne s^{-2},\quad
z_j\ne \left(\frac r s\right)^2 z_i
\qquad (i\ne j).
\]
Equivalently, the ordered source coordinate
\((z_1,\dots,z_n)\in(\CC^\times)^n\) lies outside the finite union of
the squarefree diagonals \(z_i=z_j\) and the algebraic hypersurfaces
\[
z_i z_j=r^{-2},
\qquad
z_i z_j=s^{-2},
\qquad
z_j=\left(\frac r s\right)^2 z_i
\qquad (i\ne j).
\]
Then the graph
\(\Gamma_{r,s}(Q_r(P),Q_s(P))\) has the diagonal perfect matching as
its unique perfect matching.  Consequently the two monic transported
coefficient vectors determine the source zero divisor uniquely, and
the full-normalized vectors determine the full source polynomial
normalization by
\[
a_0=(-1)^n\prod_{i=1}^{n}z_i .
\]
\end{corollary}

\begin{proof}
For a source root \(z_i\), set
\[
\alpha_i=rz_i+r^{-1}z_i^{-1},
\qquad
\beta_i=sz_i+s^{-1}z_i^{-1}.
\]
The edge joining \(\alpha_i\) to \(\beta_i\) is valid and has label
\(z_i\), so the diagonal matching exists.  We show that under the
stated exclusions there is no off-diagonal edge.

If an edge joins \(\alpha_i\) to \(\beta_j\), then the candidate
\[
x=\frac{\alpha_i/s-\beta_j/r}{\Delta_{r,s}}
\]
and its partner \(y\) solve
\[
\alpha_i=rx+r^{-1}y,
\qquad
\beta_j=sx+s^{-1}y,
\qquad
xy=1.
\]
Thus \(x\) is a source value whose \(r\)-transport equals
\(\alpha_i\).  By the one-scale fiber computation in the proof of
Lemma \ref{lem:one-scale-orbit-fiber-matching},
\[
x\in\{z_i,\tau_r(z_i)\},
\qquad
\tau_r(z)=r^{-2}/z.
\]
The same argument at scale \(s\) gives
\[
x\in\{z_j,\tau_s(z_j)\},
\qquad
\tau_s(z)=s^{-2}/z.
\]
Hence any valid edge from \(\alpha_i\) to \(\beta_j\) forces one of the
four equalities
\[
z_i=z_j,\qquad
z_i=\tau_s(z_j),\qquad
\tau_r(z_i)=z_j,\qquad
\tau_r(z_i)=\tau_s(z_j).
\]
For \(i\ne j\), squarefreeness rules out \(z_i=z_j\).  The next two
equalities are exactly
\[
z_i z_j=s^{-2},
\qquad
z_i z_j=r^{-2},
\]
and the last equality is
\[
\frac{r^{-2}}{z_i}=\frac{s^{-2}}{z_j},
\quad\text{equivalently}\quad
z_j=\left(\frac r s\right)^2z_i .
\]
These are precisely the forbidden off-diagonal resonances.  Therefore
every edge of the candidate graph is diagonal, and the diagonal edges
are the only possible perfect matching.

The uniqueness of the recovered source divisor now follows from
Theorem \ref{thm:two-scale-matching-inverse}.  The formula for the
full scalar is the product normalization in the same theorem.  The
exceptional set is the displayed finite union of algebraic
hypersurfaces because there are finitely many ordered pairs
\((i,j)\) with \(i\ne j\).
\end{proof}

\begin{corollary}[Two-scale nonresonance common fiber]\label{cor:two-scale-nonresonance-common-fiber}
Let \(r,s>1\) be distinct, and let
\[
P(z)=\prod_{i=1}^{n}(z-z_i)
\]
be squarefree with nonzero constant term.  Suppose that for all
distinct \(i,j\),
\[
z_i z_j\ne r^{-2},\qquad
z_i z_j\ne s^{-2},\qquad
z_j\ne \left(\frac r s\right)^2 z_i .
\]
Then the common inverse fiber determined by the two transported
divisors at the scales \(r\) and \(s\) consists of the original source
divisor \(\sum_i[z_i]\) only.
\end{corollary}

\begin{proof}
At scale \(r\), a transported root coming from \(z_i\) has exactly the
two possible one-scale lifts
\[
z_i,\qquad r^{-2}/z_i .
\]
At scale \(s\), a transported root coming from \(z_j\) has exactly the
two possible one-scale lifts
\[
z_j,\qquad s^{-2}/z_j .
\]
Thus a false matching between an \(r\)-transported root indexed by
\(i\) and an \(s\)-transported root indexed by \(j\ne i\) can occur
only if these two two-point lift sets intersect.  The four possible
intersections give
\[
z_i=z_j,\qquad z_i z_j=s^{-2},\qquad
z_i z_j=r^{-2},\qquad
z_j=\left(\frac r s\right)^2z_i .
\]
Squarefreeness excludes the first equality, and the hypotheses exclude
the other three.  Hence the perfect-matching graph of Definition
\ref{def:two-scale-candidate-graph} has no off-diagonal edge.  Its
identity matching is therefore the unique perfect matching, and Theorem
\ref{thm:two-scale-matching-inverse} identifies the common inverse
fiber with the original source divisor.
\end{proof}

\begin{remark}[Repeated roots and nongeneric fibers]\label{rem:two-scale-repeated-root-caveat}
Corollary \ref{cor:generic-two-scale-injectivity} is deliberately a
squarefree generic statement.  If repeated roots are allowed, the
copied vertices in Definition \ref{def:two-scale-candidate-graph} may
carry multiplicities; Theorem \ref{thm:two-scale-matching-inverse}
still classifies the exact common fiber, but uniqueness must be read
as uniqueness of copied perfect matchings modulo the
indistinguishability of equal transported labels.  Repeated-source
coincidences are precisely the additional nongeneric diagonals
\(z_i=z_j\).  For squarefree sources, no other genericity assumption
is hidden: the only excluded off-diagonal two-scale resonances are
the three displayed hypersurface families
\[
z_i z_j=r^{-2},\qquad
z_i z_j=s^{-2},\qquad
z_j=\left(\frac r s\right)^2z_i
\qquad (i\ne j),
\]
with the reverse ordered equation obtained by interchanging \(i\) and
\(j\).
\end{remark}

\begin{corollary}[Dihedral common-fiber normal form]\label{cor:dihedral-common-fiber}
Let \(r,s>1\) be distinct, and put
\[
\tau_r(z)=r^{-2}/z,\qquad \tau_s(z)=s^{-2}/z .
\]
For any monic source divisor \(D\) on \(\CC^\times\), the common
two-scale fiber over the monic transports at \(r\) and \(s\) is the
intersection of the one-scale fibers cut out by \(\tau_r\) and
\(\tau_s\).  Equivalently, every competing source root is forced to
lie in the orbit of an original root under the group generated by the
two reciprocal involutions, and
\[
\tau_s\circ\tau_r=\left(\frac r s\right)^2\operatorname{id},
\qquad
\tau_r\circ\tau_s=\left(\frac s r\right)^2\operatorname{id}.
\]
Thus exceptional competitors are finite matching choices inside
finite pieces of the infinite dihedral orbit generated by
\(\tau_r\) and \(\tau_s\); the data-level inverse is exactly the finite
candidate graph of Definition \ref{def:two-scale-candidate-graph}.

If the original source roots all lie on \(\TT\), then for \(r\ne s\)
this common two-scale fiber collapses automatically to the original
source divisor.  In particular, the possible off-diagonal resonance
relations in Corollary \ref{cor:generic-two-scale-injectivity} cannot
hold for distinct unit-circle roots.
\end{corollary}

\begin{proof}
By Lemma \ref{lem:one-scale-orbit-fiber-matching}, a source divisor
with the same monic transport at scale \(r\) must preserve total
multiplicity on each \(\tau_r\)-orbit.  The same statement with \(s\)
in place of \(r\) holds at the second scale.  Therefore a common
two-scale competitor must lie in the intersection of these two
one-scale orbit-matching constraints.  Theorem
\ref{thm:two-scale-matching-inverse} gives the equivalent finite
description: a common competitor is obtained by choosing a perfect
matching between the copied \(r\)-transport and \(s\)-transport roots.

The compositions are immediate:
\[
\tau_s(\tau_r(z))
=s^{-2}\left(r^{-2}z^{-1}\right)^{-1}
=\left(\frac r s\right)^2z,
\]
and similarly
\[
\tau_r(\tau_s(z))=\left(\frac s r\right)^2z.
\]
Thus the two involutions generate the dihedral action consisting of
reciprocal reflections and the two multiplicative translations by
\((r/s)^2\) and \((s/r)^2\).  A polynomial has only finitely many
transported roots, so the actual common fiber is not the whole orbit
but the finite matching problem cut out by the observed copied root
multisets.

Finally suppose every original root lies on \(\TT\).  The one-scale
fiber at the single scale \(r\) already collapses to the original
divisor by Theorem
\ref{thm:algebraic-mahler-transport-inverse}\textup{(iii)}, so the
common two-scale fiber is a singleton.  Equivalently, the explicit
resonances from Corollary \ref{cor:generic-two-scale-injectivity} are
excluded by moduli: \(z_i z_j\) has modulus \(1\), whereas
\(r^{-2}\) and \(s^{-2}\) have modulus \(<1\), and
\((r/s)^2\ne1\) is a positive real number different from \(1\), so it
cannot carry one unit-circle point to a distinct unit-circle point.
\end{proof}

\subsection{Multiscale Laurent Coefficient Interpolation}\label{subsec:multiscale-coefficient-vector-inverse}

\begin{lemma}[Parity-restricted Laurent interpolation]\label{lem:parity-restricted-laurent-interpolation}
Fix \(k\ge0\) and distinct positive numbers \(r_0,
\dots,r_k\).  The evaluation map
\[
\sum_{j=0}^{k}\alpha_j r^{k-2j}
\longmapsto
\left(\sum_{j=0}^{k}\alpha_j r_i^{k-2j}\right)_{i=0}^{k}
\]
from the \((k+1)\)-dimensional space spanned by \(r^k,r^{k-2},\dots,r^{-k}\) to \(\CC^{k+1}\) is an isomorphism.  Hence any coefficient family supported on this parity class is determined by its values at \(k+1\) distinct scales; in particular \(n+1\) distinct scales determine it for every \(0\le k\le n\).
\end{lemma}

\begin{proof}
Multiplication by \(r^k\) sends the displayed Laurent polynomial to
\[
\sum_{j=0}^{k}\alpha_j r^{2k-2j}
=\sum_{j=0}^{k}\alpha_j (r^2)^{k-j},
\]
a polynomial of degree at most \(k\) in \(u=r^2\).  The evaluation matrix at \(u_i=r_i^2\) is the Vandermonde matrix \((u_i^{k-j})_{0\le i,j\le k}\), whose determinant is, up to sign, \(\prod_{0\le i<j\le k}(u_j-u_i)\).  This determinant is nonzero because the scales \(r_i\) are distinct and positive.  Therefore the evaluations determine the coefficients \(\alpha_j\) uniquely.  If \(n+1\) scales are available and \(k\le n\), restricting to any \(k+1\) of them already gives the stated interpolation.
\end{proof}

\begin{lemma}[Full and monic multiscale normalization]\label{lem:full-monic-multiscale-normalization}
Let \(P(z)=\prod_{j=1}^{n}(z-z_j)\) be monic with \(z_1\cdots z_n\ne0\), let
\[
\begin{gathered}
Q_r(w)=\sum_{k=0}^{n}A_k(r)w^{n-k},
\qquad
a_0=(-1)^nz_1\cdots z_n,\\
\widetilde Q_r(w):=a_0^{-1}Q_r(w)
=\sum_{k=0}^{n}\widetilde A_k(r)w^{n-k}.
\end{gathered}
\]
and let \(r_0,
\dots,r_n>0\) be distinct.  The source polynomial is reconstructed by the following two algebraic normalization procedures.
\begin{enumerate}[label=\textup{(\roman*)}]
\item \emph{Full-normalized data.}  If the coefficient lists of \(Q_{r_i}\) are known, then the common leading coefficient gives \(a_0\).  For each \(k\), interpolate \(A_k(r)\) in the parity-supported Laurent span \(r^k,r^{k-2},
\dots,r^{-k}\); then
\[
e_k(z_1,
\dots,z_n)=(-1)^k a_0^{-1}[r^k]A_k(r).
\]
\item \emph{Monic-normalized data.}  If only nonzero scalar multiples of the coefficient lists are known, each scale may carry its own unknown nonzero scalar.  Because the underlying exact object is a transported degree-\(n\) zero divisor, the leading entry of every complete list is nonzero: before rescaling it is \(a_0\ne0\), and after rescaling it is multiplied by a nonzero scalar.  Divide each list by its own leading entry, obtaining the monic lists of \(\widetilde Q_{r_i}\).  For each \(k\), interpolate \(\widetilde A_k(r)\) in the same parity-supported Laurent span; then
\[
e_k(z_1,
\dots,z_n)=(-1)^k[r^k]\widetilde A_k(r),
\qquad
a_0=(-1)^ne_n(z_1,
\dots,z_n)=[r^n]\widetilde A_n(r).
\]
\end{enumerate}
In both cases the recovered elementary symmetric functions determine the original monic source polynomial \(P\).
\end{lemma}

\begin{proof}
The leading coefficient of \(Q_r\) is the nonzero scalar \(a_0\), independent of \(r\), so full-normalized data immediately identify it.  If a known list is multiplied by an unknown nonzero scalar, its leading entry is multiplied by the same scalar; division by that leading entry therefore gives the coefficient list of the monic transport \(\widetilde Q_r\).

For fixed \(k\), the coefficient of \(w^{n-k}\) in \(Q_r\) is
\[
A_k(r)=(-1)^k a_0\sum_{|I|=k}\prod_{j\in I}(rz_j+r^{-1}z_j^{-1}),
\]
and hence is supported on the exponents \(k,k-2,
\dots,-k\).  Lemma \ref{lem:parity-restricted-laurent-interpolation} reconstructs this Laurent polynomial from any \(k+1\) of the available scale values, and therefore from the displayed \(n+1\) scales.  Its highest coefficient is \((-1)^ka_0e_k(z_1,
\dots,z_n)\), proving the full-normalized formula.  Dividing the same identity by \(a_0\) gives
\[
\widetilde A_k(r)=(-1)^k\sum_{|I|=k}\prod_{j\in I}(rz_j+r^{-1}z_j^{-1}),
\]
whose highest coefficient is \((-1)^ke_k(z_1,
\dots,z_n)\), proving the monic-normalized formula.  For \(k=n\), the highest coefficient of \(\widetilde A_n\) is \((-1)^ne_n=a_0\), so the leading coefficient of the full transport is also recovered from monic-normalized data.  Finally, the monic polynomial \(P(z)=z^n-e_1z^{n-1}+\cdots+(-1)^ne_n\) is determined by the recovered elementary symmetric functions.
\end{proof}

The data model for the following theorem is the complete-vector entry of
Table~\ref{tab:normalization-ledger}, with the formal observation
interface in Definition \ref{def:transport-coefficient-observations}.
The normalization step is Lemma
\ref{lem:full-monic-multiscale-normalization}.

\begin{theorem}[Exact complete-coefficient-vector inverse theorem]\label{thm:algebraic-mahler-transport-inverse}
Let \(n\ge1\), and let \(P(z)=\prod_{j=1}^{n}(z-z_j)\) be monic with \(z_1\cdots z_n\ne0\).  For each \(r>1\), let \(Q_r\) be the Joukowsky transport polynomial normalized as in Table~\ref{tab:normalization-ledger}, so
\[
Q_r(w)=\sum_{k=0}^{n}A_k(r)w^{n-k},
\qquad
A_0(r)=a_0=(-1)^n z_1\cdots z_n\ne0.
\]
Assume the exact data are complete coefficient-vector data in one of the
two reconstruction models of Definition
\ref{def:transport-coefficient-observations}.

The interpolation-budget assertion in this theorem is titled
\emph{sharpness for Laurent coefficient interpolation}: it is only a
sharpness statement for unrestricted parity-supported Laurent coefficient
interpolation, not a lower-bound claim for source recovery in every
algebraically constrained source class.  Then the following inverse
statements hold.
\begin{enumerate}[label=\textup{(\roman*)}]
  \item \leavevmode\par\noindent\textbf{Sharpness for Laurent coefficient interpolation.}

  If \(r_0,\dots,r_n>1\) are distinct and exact complete coefficient-vector data are observed at these scales, then \(P\) is reconstructed uniquely by the full or monic algebraic normalization procedure of Lemma \ref{lem:full-monic-multiscale-normalization}.  Equivalently, the coefficient \(A_k(r)\) of \(w^{n-k}\) lies in the parity-supported Laurent span \(r^k,r^{k-2},\dots,r^{-k}\), and its top Laurent coefficient recovers \((-1)^ka_0e_k(z_1,\dots,z_n)\) in the full normalization and \((-1)^ke_k(z_1,\dots,z_n)\) in the monic normalization.  The uniform budget \(n+1\) is sufficient for exact reconstruction by this coefficient-interpolation method, and its sharpness is only a sharp interpolation budget for Laurent coefficient interpolation, not a source-recovery lower bound for every algebraically constrained source class.
  In particular, this is not a lower-bound theorem for every
  algebraically constrained source-recovery problem; the lower bound is
  only on the unrestricted parity-supported Laurent coefficient space
  used by the interpolation proof.

  \item \textbf{One-scale resonance fibers.}  Fix one scale \(r>1\), and let \(\tau_r(z)=r^{-2}/z\).  The full one-scale fiber, including repeated roots, overlapping reflected pairs, and the fixed critical points \(z=\pm r^{-1}\), is the \(\tau_r\)-orbit matching set described in Lemma \ref{lem:one-scale-orbit-fiber-matching}: a competitor must preserve the total multiplicity on every \(\tau_r\)-orbit and must preserve the product of all source roots.  Equivalently, on each two-point orbit \(O=\{x,\tau_r(x)\}\) with total multiplicity \(N_O\), a competitor chooses an integer \(c_O\in\{0,\dots,N_O\}\) as the multiplicity at \(x\), while a one-point fixed orbit \(x=\tau_r(x)\) has no choice.  These choices are allowed exactly when
  \[
  \prod_{O=\{x,\tau_r(x)\}}
  \left(\frac{x}{\tau_r(x)}\right)^{c_O-m_x}=1,
  \]
  the product being over the two-point orbits and with the original divisor corresponding to \(c_O=m_x\) for every \(O\).  In the separated simple case where the \(2n\) points \(z_j\) and \(r^{-2}/z_j\) are pairwise distinct, this reduces to choosing one element from each indexed pair \(\{z_j,r^{-2}/z_j\}\).  If \(S\subset\{1,\dots,n\}\) is the set of indices for which the reflected point \(r^{-2}/z_j\) is chosen, then the choice is in the fiber exactly when
  \[
  \prod_{j\in S}z_j^2=r^{-2|S|}.
  \]

  \item \textbf{Unit-circle zero and separation injectivity.}  If \(|z_j|=1\) for all \(j\), then the one-scale map is injective on the monic unit-circle zero class, including repeated roots.  More generally, one-scale injectivity holds for every source divisor \(D\) for which the orbitwise integer choice \((c_O)_O=(m_x)_O\) is the only choice satisfying the preceding resonance product.  In the separated simple case this criterion is exactly the absence of an alternative subset choice \(S\ne\varnothing\) satisfying the displayed resonance equation; the empty subset is the original divisor and is not a competitor.
\end{enumerate}
\end{theorem}

\begin{proof}
For (i), the data restriction in Definition \ref{def:transport-coefficient-observations} is used at the first step: at each scale the proof knows the entire ordered coefficient vector of the degree-\(n\) polynomial \(Q_{r_i}\), or a projective multiple of that entire vector with its nonzero leading coordinate visible.  Thus coefficient values may be interpolated coefficient-by-coefficient.  Write
\[
\widetilde Q_r(w)=\prod_{j=1}^{n}(w-rz_j-r^{-1}z_j^{-1}).
\]
for the monic transport.  If the exact data are scalar multiples of the coefficient lists, we first divide each list by its leading coefficient; this gives \(\widetilde Q_{r_i}\).  The constant coefficient of \(\widetilde Q_r\), as a parity-restricted Laurent polynomial in \(r\), has bottom coefficient \((-1)^n(z_1\cdots z_n)^{-1}\), so interpolation from the distinct scales recovers this nonzero number and hence recovers
\[
a_0=(-1)^n z_1\cdots z_n.
\]
For fully normalized transport data this same scalar is read off directly as the leading coefficient of any known polynomial \(Q_{r_i}\).  The actual transport polynomial is multiplied by
\[
a_0=(-1)^n z_1\cdots z_n,
\]
and the coefficient \(A_k(r)\) of \(w^{n-k}\) in the actual polynomial
\[
Q_r(w)=a_0\prod_{j=1}^{n}(w-rz_j-r^{-1}z_j^{-1})
\]
is
\[
(-1)^k a_0\sum_{|I|=k}\prod_{j\in I}(rz_j+r^{-1}z_j^{-1}).
\]
It is a Laurent polynomial in \(r\) with exponents \(k,k-2,\dots,-k\).  Lemma \ref{lem:parity-restricted-laurent-interpolation} applies to this parity-restricted family: after multiplication by \(r^k\), its coefficient values form a degree-at-most-\(k\) polynomial in \(r^2\), and the Vandermonde matrix on any \(k+1\) of the available \(n+1\) distinct scales has full rank.  Hence the scale values determine this Laurent polynomial uniquely, for every \(k\le n\).  Its highest \(r\)-power coefficient is \((-1)^k a_0 e_k(z_1,\dots,z_n)\), so division by the already known nonzero coefficient \(a_0\) recovers \(e_k(z_1,
\dots,z_n)\).  In the monic-normalized observation convention the same conclusion follows from the highest coefficient of \(\widetilde A_k(r)=a_0^{-1}A_k(r)\), namely \((-1)^ke_k(z_1,\dots,z_n)\); for \(k=n\) this highest coefficient is \((-1)^ne_n(z_1,\dots,z_n)=a_0\), so the full transport scalar is recovered after the monic interpolation.  For \(k=n\), the full-normalized constant term has highest coefficient \((-1)^n a_0(z_1\cdots z_n)=(z_1\cdots z_n)^2\), and its lowest coefficient is \((-1)^n a_0(z_1\cdots z_n)^{-1}=1\).  Dividing by \(a_0\) gives the displayed top and bottom coefficients for the monic transport.  Thus all elementary symmetric functions of \(z_1,
\dots,z_n\) are recovered, and these are precisely the coefficients of the monic polynomial \(P\).

The sufficiency of the uniform scale count is Lemma \ref{lem:parity-restricted-laurent-interpolation} applied coefficient by coefficient.  The largest coefficient level has parity-restricted Laurent support
\[
r^n,r^{n-2},\dots,r^{-n},
\]
a vector space of dimension \(n+1\).  Values at \(n+1\) distinct positive scales give a nonsingular Vandermonde system in \(r^2\), while values at only \(n\) prescribed scales leave a nonzero kernel on this unrestricted coefficient space.  Hence \(n+1\) is the exact uniform budget for the unrestricted Laurent interpolation used here.  This proves the advertised sharpness for Laurent coefficient interpolation itself; it does not assert that every algebraically constrained subfamily of source polynomials realizes every vector in the interpolation kernel.

For (ii), apply Lemma \ref{lem:one-scale-orbit-fiber-matching}.  It gives the necessary and sufficient orbit-multiplicity matching condition for arbitrary repeated and overlapping reflected roots.  On a two-point orbit \(O=\{x,\tau_r(x)\}\), condition \textup{(a)} says only that the total multiplicity \(N_O\) is fixed, so every competitor is obtained by choosing an integer \(c_O\in\{0,\dots,N_O\}\) at \(x\) and \(N_O-c_O\) at \(\tau_r(x)\).  The quotient of the new root product by the old one on this orbit is \((x/\tau_r(x))^{c_O-m_x}\).  Multiplying over the two-point orbits gives the displayed orbitwise resonance product.  If \(x=\tau_r(x)=\pm r^{-1}\), then the orbit has one point and condition \textup{(a)} fixes its multiplicity, so fixed critical orbits contribute no choice and no resonance factor.  The final separated-case clause of Lemma \ref{lem:one-scale-orbit-fiber-matching} is the special case \(N_O=1\) for every indexed pair, where choosing the reflected point for precisely the indices in \(S\) gives the stated subset equation.

For (iii), assume first that all source roots lie on \(\TT\).  Let \(\widetilde D\) satisfy the two orbit-matching conditions of Lemma \ref{lem:one-scale-orbit-fiber-matching}.  Moving one unit of multiplicity from a unit-circle point \(x\) to its reflected point \(\tau_r(x)=r^{-2}/x\) changes the modulus of the source-root product by the factor \(r^{-2}\); moving in the reverse direction changes it by \(r^2\).  Since the original divisor has no mass on the inner shell \(|z|=r^{-2}\), every nontrivial orbit redistribution away from the original unit-circle divisor decreases the product modulus by a positive power of \(r^2\).  It therefore cannot satisfy the product-preservation condition.  Hence \(\widetilde D=D\), proving unit-circle injectivity even with repeated roots.  The stated general criterion is precisely the uniqueness of the orbitwise integer choice satisfying the resonance product in (ii).  In the separated simple case the original choice is \(S=\varnothing\); an actual competitor must have \(S\ne\varnothing\), so injectivity is equivalent to the nonexistence of such a nonempty resonant subset.  Repeated roots and overlapping reflected pairs are not indexed by subsets: they are governed by the orbit totals \(N_O\), while fixed orbits remain forced.
\end{proof}

\begin{remark}[Sharpness for Laurent coefficient interpolation]\label{rem:multiscale-sharpness-scope}
For the coefficient \(A_k(r)\) of \(w^{n-k}\), Lemma \ref{lem:parity-restricted-laurent-interpolation} shows that any \(k+1\) distinct positive scales determine the parity-supported Laurent polynomial in
\[
\operatorname{span}\{r^k,r^{k-2},\dots,r^{-k}\}.
\]
This number is sharp for that unrestricted vector space, because evaluation at only \(k\) scales has a nontrivial kernel.  The theorem therefore uses the uniform budget \(n+1\) to cover all coefficient levels \(0\le k\le n\) at once.  This sharpness claim is only about unrestricted parity-supported Laurent interpolation.  It is not a lower-bound theorem for every algebraically constrained subfamily of source polynomials.
\end{remark}

\begin{theorem}[Exact one-scale unit-circle zero inverse theorem from an exact complete coefficient vector]\label{thm:leyang-single-scale-reconstruction}
Fix an exact real number \(r>1\).  Let \(\mathcal L\) be the class of all monic polynomials \(P\in\CC[z]\) of positive degree with every zero on \(\TT\), counted with multiplicity.  For \(P\in\mathcal L\), let \(Q_r(P)\), known as a complete exact coefficient vector, be its transport polynomial from Table~\ref{tab:normalization-ledger}.  Then the one-scale transport map
\[
\mathcal T_r:\ P\longmapsto Q_r(P)
\]
is injective on \(\mathcal L\), and the original source is recovered from the exact pair \((r,Q_r(P))\) by the shell selection of Proposition \ref{prop:exact-shell-reconstruction}.  The unit-circle hypothesis \(P\in\mathcal L\) is essential to this one-scale collapse; outside it, the complete one-scale fiber is the finite orbit-redistribution fiber described in Theorem \ref{thm:algebraic-mahler-transport-inverse}\textup{(ii)}.
\end{theorem}

\begin{proof}
This is the unit-circle zero case of Theorem \ref{thm:algebraic-mahler-transport-inverse}(iii), with the constructive shell recovery supplied by Proposition \ref{prop:exact-shell-reconstruction}.
\end{proof}

\begin{corollary}[No-resonance unit-circle fiber collapse]\label{cor:no-resonance-leyang-fiber-collapse}
Fix \(r>1\), and let \(P\in\mathcal L\) have degree \(n\) and zero divisor \(D=\sum_xm_x[x]\).  Among all monic degree-\(n\) polynomials with nonzero constant term, the complete one-scale fiber over \(Q_r(P)\) consists of the orbitwise redistributions described in Theorem \ref{thm:algebraic-mahler-transport-inverse}\textup{(ii)}.  Its intersection with the unit-circle class \(\mathcal L\) is the singleton \(\{P\}\).  In the separated simple-root case this says that every nonempty reflected-choice subset \(S\) fails the resonance equation
\[
\prod_{j\in S}z_j^2=r^{-2|S|}
\]
for the elementary reason that the two sides have moduli \(1\) and \(r^{-2|S|}<1\), respectively.

Consequently the annular shell selection below is not an additional hypothesis: it is the geometric form of the same fiber collapse after the two shells \(|z|=1\) and \(|z|=r^{-2}\) have been separated.
\end{corollary}

\begin{proof}
The first sentence is exactly the fiber classification in Theorem \ref{thm:algebraic-mahler-transport-inverse}\textup{(ii)}.  The same theorem, part \textup{(iii)}, proves that if the original divisor lies on \(\TT\), then no nontrivial redistribution inside the reflected orbits can both preserve the transport polynomial and keep the source-root product required by the normalization; hence the only unit-circle divisor in the fiber is \(D\) itself.  In the separated simple-root case the reflected choices are indexed by subsets \(S\).  If \(S\ne\varnothing\), then \(|\prod_{j\in S}z_j^2|=1\), whereas \(|r^{-2|S|}|=r^{-2|S|}<1\), so the resonance equation cannot hold.  The shell-selection statement is Corollary \ref{cor:leyang-annular-shell-projection}: any radius \(\rho\) with \(r^{-2}<\rho<1\) separates the forced outer unit-circle roots from their reflected inner partners, so the unique outer monic factor is precisely \(P\).
\end{proof}

\begin{corollary}[Annular shell projection for exact unit-circle data]\label{cor:leyang-annular-shell-projection}
Under the hypotheses of Theorem \ref{thm:leyang-single-scale-reconstruction}, let \(\rho\) be any real number satisfying
\[
r^{-2}<\rho<1.
\]
Then \(P\) is equivalently recovered as the unique monic degree-\(n\) factor of \(S_{r,Q_r}\) whose zeros lie in the exterior annulus \(\{|z|>\rho\}\).  The complementary monic factor has all its zeros in \(\{|z|<\rho\}\).
\end{corollary}

\begin{proof}
The zeros of \(S_{r,Q_r}\) are the disjoint union
\[
\{z_1,
\dots,z_n\}\cup\{r^{-2}z_1^{-1},
\dots,r^{-2}z_n^{-1}\}.
\]
The first set lies on \(|z|=1\), and the second lies on \(|z|=r^{-2}\).  The chosen \(\rho\) separates the two shells, so the exterior factor is exactly \(P\).
\end{proof}

\subsection{Annular Rouch\'e Stability}\label{subsec:annular-rouche-stability}

\begin{lemma}[Outer Rouch\'e count prevents degree loss]\label{lem:outer-rouche-prevents-degree-loss}
Let \(S\) be a polynomial of degree \(2n\), let \(R>0\), and assume that every zero of \(S\) lies in \(|z|<R\).  Let
\[
\widetilde Q(w)=\sum_{\ell=0}^{n}\widetilde b_\ell w^\ell
\]
with coefficients allowed to satisfy \(\widetilde b_n=0\), and put
\[
\widetilde S(z):=z^n\widetilde Q(J_r(z)),
\qquad J_r(z)=rz+r^{-1}z^{-1}.
\]
Then \(\widetilde S\) is a polynomial with expansion
\[
\widetilde S(z)=
\sum_{\ell=0}^{n}\widetilde b_\ell
\sum_{j=0}^{\ell}\binom{\ell}{j}r^{2j-\ell}z^{n-\ell+2j}.
\]
Its trailing coefficient is \(r^{-n}\widetilde b_n\), and its leading coefficient is \(r^n\widetilde b_n\).  If
\[
|\widetilde S(z)-S(z)|<|S(z)|\qquad(|z|=R),
\]
then \(\widetilde b_n\ne0\), \(\deg\widetilde S=2n\), and \(\widetilde S\) has exactly \(2n\) finite zeros in \(|z|<R\), counted with multiplicity.  Thus no part of the Rouch\'e count is lost through a vanished leading coefficient or a zero at infinity.
\end{lemma}

\begin{proof}
For \(0\le\ell\le n\), the binomial expansion gives
\[
z^nJ_r(z)^\ell
=z^n\sum_{j=0}^{\ell}\binom{\ell}{j}(rz)^j(r^{-1}z^{-1})^{\ell-j}
=\sum_{j=0}^{\ell}\binom{\ell}{j}r^{2j-\ell}z^{n-\ell+2j}.
\]
This proves the displayed expansion and shows that \(\widetilde S\) is a polynomial of degree at most \(2n\).  The only term capable of contributing to degree \(0\) is \(\ell=n,j=0\), so the trailing coefficient is \(r^{-n}\widetilde b_n\).  The only term capable of contributing to degree \(2n\) is \(\ell=n,j=n\), so the leading coefficient is \(r^n\widetilde b_n\).

Rouch\'e's theorem on \(|z|=R\) gives the same number of zeros for \(\widetilde S\) and \(S\) in the disk \(|z|<R\).  Since all \(2n\) zeros of the degree-\(2n\) polynomial \(S\) lie in this disk, \(\widetilde S\) has \(2n\) finite zeros there, counted with multiplicity.  A nonzero polynomial of degree at most \(2n\) with \(2n\) finite zeros must have degree at least \(2n\), hence exactly \(2n\).  The leading coefficient formula then forces \(\widetilde b_n\ne0\), and the trailing coefficient is nonzero as well.  This rules out both the possible degree loss from \(\deg\widetilde Q<n\) and any missing zero at infinity.
\end{proof}

\begin{lemma}[Reciprocal shell scalar for monic unit-circle sources]\label{lem:unit-circle-reciprocal-shell-scalar}
Let
\[
P(z)=\prod_{j=1}^{n}(z-z_j)
\]
be monic, assume \(|z_j|=1\) for all \(j\), let \(r>0\), and put
\[
a_0=P(0)=(-1)^n\prod_{j=1}^{n}z_j .
\]
Then, for \(z\ne0\),
\[
z^nP(r^{-2}z^{-1})
=a_0\prod_{j=1}^{n}(z-r^{-2}z_j^{-1}).
\]
Consequently \(|a_0|=1\), and the modulus of this reciprocal-shell
factor is
\[
\left|z^nP(r^{-2}z^{-1})\right|
=\prod_{j=1}^{n}|z-r^{-2}z_j^{-1}|.
\]
Moreover, for every \(\rho>0\) and every \(z\) with \(|z|=\rho\),
\[
\prod_{j=1}^{n}|z-z_j|
\prod_{j=1}^{n}|z-r^{-2}z_j^{-1}|
\ge |1-\rho|^n|\rho-r^{-2}|^n .
\]
Thus the reciprocal-shell scalar \(a_0\) never changes any
absolute-value shell estimate for monic unit-circle sources.
\end{lemma}

\begin{proof}
For each root,
\[
r^{-2}-zz_j=-z_j(z-r^{-2}z_j^{-1}).
\]
Multiplying these identities gives
\[
z^nP(r^{-2}z^{-1})
=\prod_{j=1}^{n}(r^{-2}-zz_j)
=(-1)^n\left(\prod_{j=1}^{n}z_j\right)
\prod_{j=1}^{n}(z-r^{-2}z_j^{-1})
=a_0\prod_{j=1}^{n}(z-r^{-2}z_j^{-1}).
\]
Since the roots lie on \(\TT\), the displayed formula for \(a_0\) gives
\(|a_0|=1\), and taking absolute values proves the final identity.
If \(|z|=\rho\), then
\[
|z-z_j|\ge \bigl||z|-|z_j|\bigr|=|1-\rho|
\]
and
\[
|z-r^{-2}z_j^{-1}|
\ge \bigl||z|-r^{-2}|z_j|^{-1}\bigr|
=|\rho-r^{-2}|.
\]
Multiplying these inequalities over \(j\) proves the displayed shell
bound.  Since \(|a_0|=1\), inserting the reciprocal-shell scalar in
\(S_r\) leaves these absolute-value estimates unchanged.
\end{proof}

The next theorem is the core annular-divisor stability interface used in the inverse section.  It supplies the analytic perturbation step that is later coupled to the exact capacity scale and the exact unit-circle shell factorization in the fixed-scale recognition statements.  The result is local: exact recognition from complete coefficient-vector data first supplies the model pair \((r,P)\); after that model and the separating contours are fixed, the number \(\Delta(r,P;\rho_-,\rho_+)\) gives a Rouch\'e radius inside which the annular divisor persists.  The analytic input is Rouch\'e's theorem in its usual zero-counting form for a contour and its interior \citep[pp.~152--153]{Ahlfors1979}.  The unperturbed polynomial
\(S_r(z)=z^nQ_r(J_r(z))\) has one zero shell on \(|z|=r^{-2}\) and one on \(|z|=1\); the reciprocal-shell scalar is the \(a_0\) of Lemma \ref{lem:unit-circle-reciprocal-shell-scalar}, whose modulus is one.  The inner contour \(\Gamma_-\) surrounds the reflected shell, the middle contour \(\Gamma_+\) separates it from the unit-circle shell, and the outer contour \(\Gamma_0\) encloses both shells while preventing degree loss at infinity.  A perturbation whose pullback is smaller than the displayed Rouch\'e envelope on these three contours therefore preserves the annular divisor in \(\rho_+<|z|<\rho_+^{-1}\); this divisor is the contour-selected annular factor associated with the perturbed transported polynomial and is not intrinsic before \((r,P,\rho_-,\rho_+)\) are fixed.

\begin{theorem}[Core quantitative annular-divisor Rouch\'e stability]\label{thm:quantitative-annular-rouche-stability}
Let \(P\in\CC[z]\) be a monic polynomial of degree \(n\ge1\), all of whose zeros lie on \(\TT\), let \(r>1\), let \(Q_r\) be its transport polynomial from Table~\ref{tab:normalization-ledger}, and define
\[
S_r(z):=S_{r,Q_r}(z)=z^nQ_r(J_r(z)).
\]
Choose radii
\[
r^{-2}<\rho_-<\rho_+<1
\]
and set
\[
\Gamma_-:=\{|z|=\rho_-\},\qquad
\Gamma_+:=\{|z|=\rho_+\},\qquad
\Gamma_0:=\{|z|=\rho_+^{-1}\},
\]
so that the bounded annulus \(\rho_+<|z|<\rho_+^{-1}\) contains precisely the \(n\) unit-circle zeros of \(S_r\), while the disk \(|z|<\rho_-\) contains precisely the \(n\) inner zeros.  Define the Rouch\'e envelope
\[
\Delta(r,P;\rho_-,\rho_+):=\frac12\min\left\{
\inf_{z\in\Gamma_-}|S_r(z)|,
\inf_{z\in\Gamma_+}|S_r(z)|,
\inf_{z\in\Gamma_0}|S_r(z)|
\right\}>0.
\]
Equivalently, writing \(P(z)=\prod_{j=1}^{n}(z-z_j)\), one has the
exact factorization
\[
S_r(z)
=a_0r^n\prod_{j=1}^{n}(z-z_j)
\prod_{j=1}^{n}(z-r^{-2}z_j^{-1})
=r^nP(z)\,z^nP(r^{-2}z^{-1}),
\]
where
\[
a_0=P(0)=(-1)^n\prod_{j=1}^{n}z_j .
\]
Thus the scalar \(a_0\) is not being discarded in the exact identity; it
is exactly the reciprocal-shell scalar in \(z^nP(r^{-2}z^{-1})\).
Since \(|a_0|=1\) for unit-circle roots, taking absolute values gives
\[
|S_r(z)|=r^n\prod_{j=1}^{n}|z-z_j|
\prod_{j=1}^{n}|z-r^{-2}z_j^{-1}|,
\qquad |z_j|=1,
\]
which is the absolute-value equality used for the Rouch\'e lower bounds below.  The
displayed algebraic factorization above still includes the scalar
\(a_0\); only the absolute-value estimate removes it.  Lemma
\ref{lem:unit-circle-reciprocal-shell-scalar} gives the
equivalent reciprocal-shell modulus identity
\[
\left|z^nP(r^{-2}z^{-1})\right|
=\prod_{j=1}^{n}|z-r^{-2}z_j^{-1}|
\]
for the last factor.  Hence the known two-shell geometry gives the
explicit product lower bound
\[
\begin{aligned}
\Delta(r,P;\rho_-,\rho_+)
&\ge
\frac{r^n}{2}\min\Bigl\{
(1-\rho_-)^n(\rho_- - r^{-2})^n,\\
&\hspace{5.7em}
(1-\rho_+)^n(\rho_+ - r^{-2})^n,\,
(\rho_+^{-1}-1)^n(\rho_+^{-1}-r^{-2})^n
\Bigr\}.
\end{aligned}
\]
The geometry and degree preservation are fixed before perturbing: the contours \(\Gamma_-\), \(\Gamma_+\), and \(\Gamma_0\) are pairwise disjoint circles, \(\Gamma_-\) encloses exactly the reflected shell, \(\Gamma_+\) separates that shell from the unit-circle shell, and \(\Gamma_0\) encloses both shells.  The outer-contour count is used to prevent loss of degree at infinity.  If \(\widetilde Q\in\CC[w]\) has \(\deg \widetilde Q\le n\) and
\[
\sup_{z\in\Gamma_-\cup\Gamma_+\cup\Gamma_0}
\left|z^n\widetilde Q(J_r(z))-S_r(z)\right|<\Delta(r,P;\rho_-,\rho_+),
\]
then \(\widetilde S(z):=z^n\widetilde Q(J_r(z))\) is in fact a degree-\(2n\) polynomial, even if the input polynomial was initially allowed to have \(\deg\widetilde Q<n\), and it has exactly \(n\) zeros, counted with multiplicity, in the annulus
\[
\rho_+<|z|<\rho_+^{-1}
\]
and no zeros on any of \(\Gamma_-\), \(\Gamma_+\), or \(\Gamma_0\).  The monic degree-\(n\) polynomial formed from the zeros in \(\rho_+<|z|<\rho_+^{-1}\) is denoted
\[
P_{\mathrm{ann}}^{\rho_-,\rho_+}(\widetilde Q;r,P).
\]
It is the unique contour-selected annular factor associated with \(\widetilde Q\) after the model \((r,P)\) and contours \((\rho_-,\rho_+)\) have been fixed.  For \(\widetilde Q=Q_r\) this factor is exactly \(P\).

The contour-selected annular factor is quantitatively stable in the following local sense.  Write the zero divisor of \(P\) as \(\sum_{a\in Z(P)}m_a[a]\), where \(Z(P)\) is the finite set of distinct zeros.  Let \(U_a\) \((a\in Z(P))\) be pairwise disjoint Jordan domains whose closures lie in \(\rho_+<|z|<\rho_+^{-1}\), with \(a\in U_a\) and with no zero of \(P\) other than \(a\) in \(U_a\).  Put
\[
\Delta_{\mathcal U}(r,P)
:=\frac12\min_{a\in Z(P)}\inf_{z\in\partial U_a}|S_r(z)|>0.
\]
If, in addition,
\[
\sup_{z\in\bigcup_{a\in Z(P)}\partial U_a}
\left|z^n\widetilde Q(J_r(z))-S_r(z)\right|<\Delta_{\mathcal U}(r,P),
\]
then \(P_{\mathrm{ann}}^{\rho_-,\rho_+}(\widetilde Q;r,P)(z)=\prod_{j=1}^{n}(z-\widetilde z_j)\) has exactly \(m_a\) zeros in each \(U_a\).  Consequently, if the domains are chosen so that \(U_a\subset\{|z-a|<\eta\}\) for all \(a\in Z(P)\), then the zeros can be labelled against the zeros \(z_1,\dots,z_n\) of \(P\) so that \(|\widetilde z_j-z_j|<\eta\) for every \(j\).  Writing
\[
P(z)=z^n+\sum_{k=1}^{n}p_k z^{n-k},
\qquad
P_{\mathrm{ann}}^{\rho_-,\rho_+}(\widetilde Q;r,P)(z)=z^n+\sum_{k=1}^{n}\widetilde p_k z^{n-k},
\]
one has the explicit coefficient estimate
\[
|\widetilde p_k-p_k|
\le \binom{n}{k}k\eta(1+\eta)^{k-1},
\qquad 1\le k\le n.
\]
After the model \((r,P)\) and the three contours have been selected, the theorem produces and stabilizes the annular divisor determined by that contour envelope.  Exact unit-circle zeros and reciprocal transport factorization remain the algebraic realizability conditions stated separately in the fixed-scale criteria.
\end{theorem}

\begin{proof}
The zeros of \(S_r\) are on \(|z|=r^{-2}\) and \(|z|=1\), so none of \(\Gamma_-\), \(\Gamma_+\), or \(\Gamma_0\) meets the zero set.  More explicitly, if the zeros of \(P\) are \(z_1,\dots,z_n\), then the shell factorization gives the exact identity
\[
S_r(z)
=a_0r^n\prod_{j=1}^n(z-z_j)
\prod_{j=1}^n(z-r^{-2}z_j^{-1})
=r^nP(z)\,z^nP(r^{-2}z^{-1}),
\]
where Lemma \ref{lem:unit-circle-reciprocal-shell-scalar} identifies
\(a_0=P(0)=(-1)^n\prod_jz_j\) as the reciprocal-shell scalar.  Therefore
\[
|S_r(z)|
=|a_0|r^n\prod_{j=1}^n |z-z_j|\prod_{j=1}^n |z-r^{-2}z_j^{-1}|.
\]
The scalar \(a_0\), the last product, and the following circle estimate
are exactly those in Lemma \ref{lem:unit-circle-reciprocal-shell-scalar};
in particular \(|a_0|=1\), because \(P\) is monic and all \(z_j\) lie on
\(\TT\).  Thus, on any circle \(|z|=\rho\),
\[
|S_r(z)|\ge r^n |1-\rho|^n|\rho-r^{-2}|^n.
\]
Applying this with \(\rho=\rho_-\), \(\rho=\rho_+\), and
\(\rho=\rho_+^{-1}\) gives the displayed product lower bound for
\(\Delta(r,P;\rho_-,\rho_+)\), in particular the three infima in the
definition of \(\Delta\) are positive.  On each contour the perturbation is strictly smaller than \(|S_r|\), so Rouch\'e's theorem implies that \(\widetilde S\) and \(S_r\) have the same number of zeros in the disks bounded by these contours.  The outer contour case is exactly Lemma \ref{lem:outer-rouche-prevents-degree-loss} with \(R=\rho_+^{-1}\), since all zeros of \(S_r\) lie in \(|z|<\rho_+^{-1}\); therefore \(\widetilde S\) is degree \(2n\) and the count on \(\Gamma_0\) is a finite zero count rather than a count with missing zeros at infinity.  Thus \(\widetilde S\) has \(n\) zeros in \(|z|<\rho_-\), \(n\) zeros in \(|z|<\rho_+\), and \(2n\) zeros in \(|z|<\rho_+^{-1}\).  Therefore no zeros lie in \(\rho_-<|z|<\rho_+\), and exactly \(n\) zeros lie in \(\rho_+<|z|<\rho_+^{-1}\).  A finite zero divisor determines a unique monic polynomial, so those \(n\) annular zeros define the unique contour-selected annular factor \(P_{\mathrm{ann}}^{\rho_-,\rho_+}(\widetilde Q;r,P)\).  In the unperturbed case the annular zeros are precisely the zeros of \(P\).

For the local stability statement, the boundaries \(\partial U_a\) are compact and avoid the zero set of \(S_r\), while each \(U_a\) contains exactly the multiplicity \(m_a\) zero of \(S_r\) at \(a\).  Hence \(\Delta_{\mathcal U}(r,P)>0\).  The additional perturbation bound and Rouch\'e's theorem on every \(\partial U_a\) show that \(\widetilde S\) has exactly \(m_a\) zeros in \(U_a\), counted with multiplicity.  These zeros are among the annular zeros already selected, because every \(\overline{U_a}\) lies inside \(\rho_+<|z|<\rho_+^{-1}\), and their total number is \(\sum_a m_a=n\); therefore they are precisely the zeros of \(P_{\mathrm{ann}}^{\rho_-,\rho_+}(\widetilde Q;r,P)\).  If \(U_a\subset\{|z-a|<\eta\}\), label the \(m_a\) zeros in \(U_a\) against the \(m_a\) copies of the zero \(a\) of \(P\).  This gives \(|\widetilde z_j-z_j|<\eta\).  Since \(|z_j|=1\), the labelled zeros satisfy \(|\widetilde z_j|\le1+\eta\).  For each subset \(I\) of cardinality \(k\), the telescoping identity gives
\[
\left|\prod_{j\in I}\widetilde z_j-
\prod_{j\in I}z_j\right|
\le k\eta(1+\eta)^{k-1}.
\]
Summing over the \(\binom nk\) subsets yields the coefficient estimate, because
\[
p_k=(-1)^ke_k(z_1,
\dots,z_n),\qquad
\widetilde p_k=(-1)^ke_k(\widetilde z_1,
\dots,
\widetilde z_n).
\]
\end{proof}

\begin{corollary}[Coefficient-level annular stability]\label{cor:coefficient-annular-stability}
Retain the hypotheses and notation of Theorem \ref{thm:quantitative-annular-rouche-stability}, and write
\[
Q_r(w)=\sum_{\ell=0}^{n}b_\ell w^\ell,
\qquad
\widetilde Q(w)=\sum_{\ell=0}^{n}\widetilde b_\ell w^\ell,
\]
allowing \(\widetilde b_n=0\) if \(\deg\widetilde Q<n\).  Put
\[
B_n(r;\rho_-,\rho_+)
:=
\max_{z\in\Gamma_-\cup\Gamma_+\cup\Gamma_0}
|z|^n\sum_{\ell=0}^{n}|J_r(z)|^\ell.
\]
The denominator \(\Delta(r,P;\rho_-,\rho_+)\) is the Rouch\'e envelope
from Theorem \ref{thm:quantitative-annular-rouche-stability}; its
explicit product lower bound uses Lemma
\ref{lem:unit-circle-reciprocal-shell-scalar} to remove the
reciprocal-shell scalar in absolute value.
If the coefficient errors satisfy
\[
\max_{0\le \ell\le n}|\widetilde b_\ell-b_\ell|
<
\frac{\Delta(r,P;\rho_-,\rho_+)}{B_n(r;\rho_-,\rho_+)},
\]
then, with \(\widetilde S(z):=z^n\widetilde Q(J_r(z))\),
\[
\sup_{\Gamma_-\cup\Gamma_+\cup\Gamma_0}
\left|z^n\widetilde Q(J_r(z))-S_r(z)\right|
<\Delta(r,P;\rho_-,\rho_+).
\]
Consequently \(\widetilde S\) satisfies the Rouch\'e envelope in Theorem
\ref{thm:quantitative-annular-rouche-stability}.  The associated
factor is the contour-selected annular factor associated with
\((\widetilde Q;r,P,\rho_-,\rho_+)\), namely
\(P_{\mathrm{ann}}^{\rho_-,\rho_+}(\widetilde Q;r,P)\), and it has
degree \(n\); for \(\widetilde Q=Q_r\) this factor is exactly \(P\).
More generally, the same conclusion holds under the weighted \(\ell^1\)
coefficient condition
\[
\max_{z\in\Gamma_-\cup\Gamma_+\cup\Gamma_0}
|z|^n\sum_{\ell=0}^{n}|\widetilde b_\ell-b_\ell|\,|J_r(z)|^\ell
<\Delta(r,P;\rho_-,\rho_+).
\]
\end{corollary}

\begin{proof}
For each point of the three contours,
\[
\left|z^n\widetilde Q(J_r(z))-S_r(z)\right|
=
\left|z^n\sum_{\ell=0}^{n}(\widetilde b_\ell-b_\ell)J_r(z)^\ell\right|
\le
|z|^n\sum_{\ell=0}^{n}|\widetilde b_\ell-b_\ell|\,|J_r(z)|^\ell.
\]
The weighted \(\ell^1\) hypothesis is exactly the assertion that the right side is uniformly smaller than \(\Delta(r,P;\rho_-,\rho_+)\).  The simpler maximum-coefficient hypothesis implies it because the last display is at most
\[
\max_{0\le \ell\le n}|\widetilde b_\ell-b_\ell|\,
B_n(r;\rho_-,\rho_+).
\]
Taking the supremum over the three contours gives the displayed usable
perturbation bound.  Thus the contour hypothesis of Theorem
\ref{thm:quantitative-annular-rouche-stability} holds, and the announced
contour-selected annular factor statement follows from that theorem.
\end{proof}
