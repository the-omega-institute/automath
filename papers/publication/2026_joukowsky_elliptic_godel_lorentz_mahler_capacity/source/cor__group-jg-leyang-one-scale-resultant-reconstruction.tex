\begin{corollary}[One-scale transport identity under the Lee--Yang unit-circle hypothesis]\label{cor:leyang-transport-identity}
In the setting of Theorem \ref{thm:pullback-factorization}, assume that \(K\subseteq\CC\), that every zero \(z_j\) of \(P\) satisfies \(|z_j|=1\), and that \(r\in\RR_{>0}\). Then for every \(z\in\CC^\times\),
\[
Q_r\!\bigl(J_r(z)\bigr)=a_n^{-1}r^n\,P(z)\,P\!\bigl(r^{-2}z^{-1}\bigr).
\]
Although the algebraic identity itself does not use the unit-circle hypothesis, we retain that assumption here because it is needed in the geometric corollaries below.
\end{corollary}

\begin{proof}
By Theorem \ref{thm:pullback-factorization},
\[
r^n z^n\,Q_r\!\bigl(J_r(z)\bigr)=P(z)\,P^{\vee}(r^2 z).
\]
Since \(P^{\vee}(u)=a_n^{-1}u^nP(1/u)\), substituting \(u=r^2 z\) gives
\[
P^{\vee}(r^2 z)=a_n^{-1}r^{2n}z^n P(r^{-2}z^{-1}).
\]
Cancelling the common factor \(z^n\) yields the claim.
\end{proof}

\begin{corollary}[Double-resultant elimination and concentric spectral shells]\label{cor:leyang-double-resultant}
Retain the hypotheses of Corollary \ref{cor:leyang-transport-identity}, and define
\[
S_r(z):=\operatorname{Res}_w\!\bigl(Q_r(w),\,r z^2-w z+r^{-1}\bigr)\in K(r)[z].
\]
Then
\[
S_r(z)=a_n^{-1} r^{n}\,P(z)\,\Bigl[z^{n}P\!\bigl(r^{-2}z^{-1}\bigr)\Bigr].
\]
Consequently,
\[
\mathrm{Zero}(S_r)
=\{z_1,\dots,z_n\}\cup\{r^{-2}z_1^{-1},\dots,r^{-2}z_n^{-1}\},
\]
where the two clusters lie on the circles \(|z|=1\) and \(|z|=r^{-2}\), respectively.
\end{corollary}

\begin{proof}
Because \(r z^2-w z+r^{-1}=z(J_r(z)-w)\) is linear in \(w\), the resultant convention fixed in Section \ref{sec:preliminaries} gives
\[
\operatorname{Res}_w\!\bigl(Q_r(w),\,r z^2-w z+r^{-1}\bigr)
=z^n\,Q_r\!\left(\frac{r z^2+r^{-1}}{z}\right)
=z^n Q_r\!\bigl(J_r(z)\bigr).
\]
Substituting Corollary \ref{cor:leyang-transport-identity} yields
\[
S_r(z)=z^n a_n^{-1}r^n P(z)P(r^{-2}z^{-1})
=a_n^{-1}r^{n}P(z)\,\Bigl[z^{n}P\!\bigl(r^{-2}z^{-1}\bigr)\Bigr],
\]
which is the required factorization. The description of the zero set follows immediately from the two factors.
\end{proof}


\begin{definition}[Full-polynomial transport inverse data]\label{def:noiseless-leyang-reconstruction-data}
Fix \(r>1\) and a polynomial \(Q\in\CC[w]\) of degree \(n\ge1\).  Its pulled-back doubled polynomial is
\[
S_{r,Q}(z):=\operatorname{Res}_w\!\bigl(Q(w),\,r z^2-w z+r^{-1}\bigr)=z^nQ(J_r(z)).
\]
A noiseless one-scale Lee--Yang datum is a pair \((r,Q)\) for which there exists a monic polynomial \(P\in\CC[z]\), all of whose zeros lie on \(\TT\), such that \(Q=Q_r\) for \(P\).  A multiscale transport datum consists of distinct exact scales \(r_0,
\dots,r_n>1\) and the full coefficient lists of \(Q_{r_0},\dots,Q_{r_n}\), all known to arise from the same monic degree-\(n\) source polynomial with nonzero constant term.
\end{definition}

\begin{proposition}[Exact reconstruction interface and shell-factor selection]\label{prop:exact-reconstruction-interface}
Fix a noiseless one-scale Lee--Yang reconstruction datum \((r,Q)\), and let \(P\) be the certified monic Lee--Yang source polynomial.  Put \(n=\deg_w Q\).  The exact algebraic procedure
\[
S_{r,Q}(z)=\operatorname{Res}_w\!\bigl(Q(w),rz^2-wz+r^{-1}\bigr),
\]
followed by selection of the degree-\(n\) zero sub-divisor supported on \(|z|=1\), equivalently supported in \(|z|>\rho\) for any exact \(\rho\) satisfying \(r^{-2}<\rho<1\), returns the original polynomial \(P\).  Repeated roots are recovered with multiplicity.
\end{proposition}

\begin{proof}
By Corollary \ref{cor:leyang-double-resultant},
\[
S_{r,Q}(z)=r^nP(z)\Bigl[z^nP(r^{-2}z^{-1})\Bigr]
\]
because the source is monic.  The first factor has all zeros on \(|z|=1\), and the second has all zeros on \(|z|=r^{-2}\).  The two shells are disjoint, so the selected degree-\(n\) divisor is exactly the zero divisor of \(P\).  A monic polynomial is uniquely determined by its zero divisor, proving the claim.
\end{proof}

\begin{lemma}[One-scale fiber matching with multiplicities]\label{lem:one-scale-orbit-fiber-matching}
Fix \(r>1\) and put
\[
\tau_r(z):=r^{-2}z^{-1},
\qquad
T_r(z):=rz+r^{-1}z^{-1}
\qquad(z\in\CC^\times).
\]
Let \(P\) be monic of degree \(n\) with zero divisor
\[
D=\sum_{x\in\CC^\times}m_x[x]
\]
and let \(Q_r\) be its transport polynomial with the normalization of Definition \ref{def:reciprocal-transport-normalization}.  A monic polynomial \(\widetilde P\) of degree \(n\), with zero divisor
\[
\widetilde D=\sum_{x\in\CC^\times}\widetilde m_x[x],
\]
has the same one-scale transport polynomial \(Q_r\) if and only if the following two conditions hold:
\begin{enumerate}[label=\textup{(\alph*)}]
  \item for every orbit \(O\) of the involution \(\tau_r\) on \(\CC^\times\),
  \[
  \sum_{x\in O}\widetilde m_x=\sum_{x\in O}m_x;
  \]
  \item the source-root product is unchanged,
  \[
  \prod_{x\in\CC^\times}x^{\widetilde m_x-m_x}=1.
  \]
\end{enumerate}
In particular, if the \(2n\) points \(z_j\) and \(r^{-2}/z_j\) are pairwise distinct, then the fiber is obtained by choosing, for each indexed pair \(\{z_j,r^{-2}/z_j\}\), exactly one point, and the choices indexed by \(S\subset\{1,\dots,n\}\) preserve \(Q_r\) exactly when
\[
\prod_{j\in S}z_j^2=r^{-2|S|}.
\]
\end{lemma}

\begin{proof}
For \(u,v\in\CC^\times\), the equation \(T_r(u)=T_r(v)\) is equivalent to
\[
0=r(u-v)+r^{-1}(u^{-1}-v^{-1})=(u-v)\left(r-\frac{r^{-1}}{uv}\right).
\]
Hence \(u=v\) or \(v=r^{-2}/u=\tau_r(u)\).  Thus the fibers of \(T_r\) on \(\CC^\times\) are exactly the \(\tau_r\)-orbits, with the fixed points \(u=\pm r^{-1}\) allowed as one-point orbits.

Writing the actual normalized transport polynomial in divisor form gives
\[
Q_r(w)=(-1)^n\left(\prod_x x^{m_x}\right)
\prod_x\bigl(w-T_r(x)\bigr)^{m_x},
\]
and the same formula with \(m_x\) replaced by \(\widetilde m_x\) holds for \(\widetilde P\).  Equality of the monic factors in \(w\) is therefore exactly equality of the pushed-forward divisors \((T_r)_\ast D=(T_r)_\ast\widetilde D\), which is condition \textup{(a)} by the fiber computation above.  After the monic factors agree, equality of the full normalized polynomials is exactly equality of their leading coefficients, namely
\[
(-1)^n\prod_xx^{m_x}=(-1)^n\prod_xx^{\widetilde m_x},
\]
which is condition \textup{(b)}.  This proves the equivalence.

Under the pairwise-distinct hypothesis every \(\tau_r\)-orbit meeting the source divisor contains exactly one original point \(z_j\) and exactly one reflected point \(r^{-2}/z_j\), each with total orbit multiplicity one.  Condition \textup{(a)} says that a competitor chooses exactly one point from each indexed pair.  If the chosen reflected indices are \(S\), then the root product changes by
\[
\prod_{j\in S}\frac{r^{-2}/z_j}{z_j}
=r^{-2|S|}\prod_{j\in S}z_j^{-2}.
\]
Condition \textup{(b)} says that this factor is one, which is precisely the displayed subset-resonance equation.
\end{proof}


\begin{lemma}[Parity-restricted Laurent interpolation]\label{lem:parity-restricted-laurent-interpolation}
Fix \(k\ge0\) and distinct positive numbers \(r_0,
\dots,r_k\).  The evaluation map
\[
\sum_{j=0}^{k}\alpha_j r^{k-2j}
\longmapsto
\left(\sum_{j=0}^{k}\alpha_j r_i^{k-2j}\right)_{i=0}^{k}
\]
from the \((k+1)\)-dimensional space spanned by \(r^k,r^{k-2},\dots,r^{-k}\) to \(\CC^{k+1}\) is an isomorphism.  Hence any coefficient family supported on this parity class is determined by its values at \(k+1\) distinct scales; in particular \(n+1\) distinct scales determine it for every \(0\le k\le n\).
\end{lemma}

\begin{proof}
Multiplication by \(r^k\) sends the displayed Laurent polynomial to
\[
\sum_{j=0}^{k}\alpha_j r^{2k-2j}
=\sum_{j=0}^{k}\alpha_j (r^2)^{k-j},
\]
a polynomial of degree at most \(k\) in \(u=r^2\).  The evaluation matrix at \(u_i=r_i^2\) is the Vandermonde matrix \((u_i^{k-j})_{0\le i,j\le k}\), whose determinant is, up to sign, \(\prod_{0\le i<j\le k}(u_j-u_i)\).  This determinant is nonzero because the scales \(r_i\) are distinct and positive.  Therefore the evaluations determine the coefficients \(\alpha_j\) uniquely.  If \(n+1\) scales are available and \(k\le n\), restricting to any \(k+1\) of them already gives the stated interpolation.
\end{proof}

\begin{theorem}[Algebraic Mahler transport inverse package]\label{thm:algebraic-mahler-transport-inverse-package}
Let \(n\ge1\), and let \(P(z)=\prod_{j=1}^{n}(z-z_j)\) be monic with \(z_1\cdots z_n\ne0\).  For each \(r>1\), let \(Q_r\) be the Joukowsky transport polynomial.  The following inverse statements hold.
\begin{enumerate}[label=\textup{(\roman*)}]
  \item \textbf{Multiscale Laurent interpolation.}  If \(r_0,
  \dots,r_n>1\) are distinct and the full polynomials \(Q_{r_0},
  \dots,Q_{r_n}\) are known with the normalization of Definition \ref{def:reciprocal-transport-normalization}, then \(P\) is reconstructed uniquely.  More explicitly, the leading coefficient of every \(Q_r\) is
  \[
  a_0=(-1)^n z_1\cdots z_n,
  \]
  and the coefficient \(A_k(r)\) of \(w^{n-k}\) in \(Q_r(w)\) is a Laurent polynomial in \(r\) with exponents \(k,k-2,\dots,-k\) whose highest coefficient is \((-1)^k a_0 e_k(z_1,
  \dots,z_n)\).  In particular, the constant term is
  \[
  Q_r(0)=\sum_{m=-n}^{n}c_m r^m,
  \]
  with no powers congruent to \(n-1\) modulo \(2\), top coefficient \((z_1\cdots z_n)^2\), and bottom coefficient \(1\).  Equivalently, the monic transport \(a_0^{-1}Q_r\) has constant-term top coefficient \((-1)^n z_1\cdots z_n\) and bottom coefficient \((-1)^n(z_1\cdots z_n)^{-1}\).  Interpolating the Laurent coefficients \(A_k(r)\) from the \(n+1\) scales and dividing their top coefficients by the known leading coefficient \(a_0\) recovers all elementary symmetric functions \(e_k(z_1,
  \dots,z_n)\), hence the original monic source polynomial.

  \item \textbf{One-scale resonance fibers.}  Fix one scale \(r>1\), and let \(\tau_r(z)=r^{-2}/z\).  The full one-scale fiber, including repeated roots, overlapping reflected pairs, and the fixed critical points \(z=\pm r^{-1}\), is the \(\tau_r\)-orbit matching set described in Lemma \ref{lem:one-scale-orbit-fiber-matching}: a competitor must preserve the total multiplicity on every \(\tau_r\)-orbit and must preserve the product of all source roots.  Equivalently, on each two-point orbit \(O=\{x,\tau_r(x)\}\) with total multiplicity \(N_O\), a competitor chooses an integer \(c_O\in\{0,\dots,N_O\}\) as the multiplicity at \(x\), while a one-point fixed orbit \(x=\tau_r(x)\) has no choice.  These choices are allowed exactly when
  \[
  \prod_{O=\{x,\tau_r(x)\}}
  \left(\frac{x}{\tau_r(x)}\right)^{c_O-m_x}=1,
  \]
  the product being over the two-point orbits and with the original divisor corresponding to \(c_O=m_x\) for every \(O\).  In the separated simple case where the \(2n\) points \(z_j\) and \(r^{-2}/z_j\) are pairwise distinct, this reduces to choosing one element from each indexed pair \(\{z_j,r^{-2}/z_j\}\).  If \(S\subset\{1,\dots,n\}\) is the set of indices for which the reflected point \(r^{-2}/z_j\) is chosen, then the choice is in the fiber exactly when
  \[
  \prod_{j\in S}z_j^2=r^{-2|S|}.
  \]

  \item \textbf{Lee--Yang and separation injectivity.}  If \(|z_j|=1\) for all \(j\), then the one-scale map is injective on the monic Lee--Yang class, including repeated roots.  More generally, one-scale injectivity holds for every source divisor \(D\) for which the orbitwise integer choice \((c_O)_O=(m_x)_O\) is the only choice satisfying the preceding resonance product.  In the separated simple case this criterion is exactly the absence of an alternative subset choice \(S\ne\varnothing\) satisfying the displayed resonance equation; the empty subset is the original divisor and is not a competitor.
\end{enumerate}
\end{theorem}

\begin{proof}
For (i), write
\[
\widetilde Q_r(w)=\prod_{j=1}^{n}(w-rz_j-r^{-1}z_j^{-1}).
\]
for the monic transport and remember that the actual transport polynomial is multiplied by
\[
a_0=(-1)^n z_1\cdots z_n,
\]
which is read off as the leading coefficient of any one of the known polynomials \(Q_{r_i}\).  The coefficient \(A_k(r)\) of \(w^{n-k}\) in the actual polynomial
\[
Q_r(w)=a_0\prod_{j=1}^{n}(w-rz_j-r^{-1}z_j^{-1})
\]
is
\[
(-1)^k a_0\sum_{|I|=k}\prod_{j\in I}(rz_j+r^{-1}z_j^{-1}).
\]
It is a Laurent polynomial in \(r\) with exponents \(k,k-2,\dots,-k\).  Lemma \ref{lem:parity-restricted-laurent-interpolation} applies to this parity-restricted family: after multiplication by \(r^k\), its coefficient values form a degree-at-most-\(k\) polynomial in \(r^2\), and the Vandermonde matrix on any \(k+1\) of the available \(n+1\) distinct scales has full rank.  Hence the scale values determine this Laurent polynomial uniquely, for every \(k\le n\).  Its highest \(r\)-power coefficient is \((-1)^k a_0 e_k(z_1,\dots,z_n)\), so division by the already known nonzero coefficient \(a_0\) recovers \(e_k(z_1,
\dots,z_n)\).  For \(k=n\), this also gives the stated check on the constant term: its highest coefficient is \((-1)^n a_0(z_1\cdots z_n)=(z_1\cdots z_n)^2\), and its lowest coefficient is \((-1)^n a_0(z_1\cdots z_n)^{-1}=1\).  Dividing by \(a_0\) gives the displayed top and bottom coefficients for the monic transport.  Thus all elementary symmetric functions of \(z_1,
\dots,z_n\) are recovered, and these are precisely the coefficients of the monic polynomial \(P\).

For (ii), apply Lemma \ref{lem:one-scale-orbit-fiber-matching}.  It gives the necessary and sufficient orbit-multiplicity matching condition for arbitrary repeated and overlapping reflected roots.  On a two-point orbit \(O=\{x,\tau_r(x)\}\), condition \textup{(a)} says only that the total multiplicity \(N_O\) is fixed, so every competitor is obtained by choosing an integer \(c_O\in\{0,\dots,N_O\}\) at \(x\) and \(N_O-c_O\) at \(\tau_r(x)\).  The quotient of the new root product by the old one on this orbit is \((x/\tau_r(x))^{c_O-m_x}\).  Multiplying over the two-point orbits gives the displayed orbitwise resonance product.  If \(x=\tau_r(x)=\pm r^{-1}\), then the orbit has one point and condition \textup{(a)} fixes its multiplicity, so fixed critical orbits contribute no choice and no resonance factor.  The final separated-case clause of Lemma \ref{lem:one-scale-orbit-fiber-matching} is the special case \(N_O=1\) for every indexed pair, where choosing the reflected point for precisely the indices in \(S\) gives the stated subset equation.

For (iii), assume first that all source roots lie on \(\TT\).  Let \(\widetilde D\) satisfy the two orbit-matching conditions of Lemma \ref{lem:one-scale-orbit-fiber-matching}.  Moving one unit of multiplicity from a unit-circle point \(x\) to its reflected point \(\tau_r(x)=r^{-2}/x\) changes the modulus of the source-root product by the factor \(r^{-2}\); moving in the reverse direction changes it by \(r^2\).  Since the original divisor has no mass on the inner shell \(|z|=r^{-2}\), every nontrivial orbit redistribution away from the original Lee--Yang divisor decreases the product modulus by a positive power of \(r^2\).  It therefore cannot satisfy the product-preservation condition.  Hence \(\widetilde D=D\), proving Lee--Yang injectivity even with repeated roots.  The stated general criterion is precisely the uniqueness of the orbitwise integer choice satisfying the resonance product in (ii).  In the separated simple case the original choice is \(S=\varnothing\); an actual competitor must have \(S\ne\varnothing\), so injectivity is equivalent to the nonexistence of such a nonempty resonant subset.  Repeated roots and overlapping reflected pairs are not indexed by subsets: they are governed by the orbit totals \(N_O\), while fixed orbits remain forced.
\end{proof}

\begin{corollary}[Sharp coefficient budgets inside the uniform multiscale reconstruction]\label{cor:sharp-coefficient-scale-budgets}
In the setting of Theorem \ref{thm:algebraic-mahler-transport-inverse-package}\textup{(i)}, let \(A_k(r)\) denote the coefficient of \(w^{n-k}\) in \(Q_r(w)\).  For each fixed \(k\), the values of \(A_k(r)\) at any \(k+1\) distinct positive scales determine the full Laurent polynomial \(A_k(r)\) and hence determine \(e_k(z_1,
\dots,z_n)\), once the common leading coefficient \(a_0\) is known.  The number \(k+1\) is sharp for unrestricted parity-supported Laurent families: values at only \(k\) distinct scales do not determine all coefficients in the span of \(r^k,r^{k-2},\dots,r^{-k}\).  Consequently the \(n+1\) scales in Theorem \ref{thm:algebraic-mahler-transport-inverse-package}\textup{(i)} are a uniform sufficient budget that simultaneously covers every coefficient level \(0\le k\le n\); the coefficient \(A_k\) itself uses only the smaller sharp budget \(k+1\).
\end{corollary}

\begin{proof}
The interpolation assertion is exactly Lemma \ref{lem:parity-restricted-laurent-interpolation} applied to the coefficient family \(A_k(r)\), together with the coefficient computation in Theorem \ref{thm:algebraic-mahler-transport-inverse-package}\textup{(i)}: the coefficient of the highest power \(r^k\) in \(A_k(r)\) is \((-1)^k a_0e_k(z_1,
\dots,z_n)\), so division by the known nonzero \(a_0\) recovers \(e_k\).

It remains to prove sharpness for the stated unrestricted coefficient class.  Fix any \(k\) distinct positive scales \(r_1,
\dots,r_k\).  The vector space spanned by \(r^k,r^{k-2},\dots,r^{-k}\) has dimension \(k+1\).  Evaluation at the chosen \(k\) scales is a linear map from this \((k+1)\)-dimensional space to \(\CC^k\), so its kernel is nonzero.  Equivalently, there is a nonzero parity-supported Laurent polynomial vanishing at all the chosen scales.  Adding a nonzero scalar multiple of this kernel element changes the Laurent coefficients without changing those \(k\) values.  Thus fewer than \(k+1\) scale values cannot determine an arbitrary member of the parity-supported class.  Since the largest coefficient level in a degree-\(n\) transport is \(k=n\), the single budget \(n+1\) is precisely the uniform budget that works for all levels at once.
\end{proof}

\begin{theorem}[Exact one-scale Lee--Yang inverse theorem]\label{thm:leyang-single-scale-reconstruction}
Fix an exact real number \(r>1\).  Let \(\mathcal L\) be the class of all monic polynomials \(P\in\CC[z]\) of positive degree with every zero on \(\TT\), counted with multiplicity.  For \(P\in\mathcal L\), let \(Q_r\) be its transport polynomial from Definition \ref{def:reciprocal-transport-normalization}.  Then the one-scale transport map
\[
\mathcal T_r:\ P\longmapsto Q_r
\]
is injective on \(\mathcal L\), and the original source is recovered from \((r,Q_r)\) by the shell selection of Proposition \ref{prop:exact-reconstruction-interface}.
\end{theorem}

\begin{proof}
This is the Lee--Yang case of Theorem \ref{thm:algebraic-mahler-transport-inverse-package}(iii), with the constructive shell recovery supplied by Proposition \ref{prop:exact-reconstruction-interface}.
\end{proof}

\begin{corollary}[Annular shell projection certificate]\label{cor:leyang-annular-shell-projection}
Under the hypotheses of Theorem \ref{thm:leyang-single-scale-reconstruction}, let \(\rho\) be any real number satisfying
\[
r^{-2}<\rho<1.
\]
Then \(P\) is equivalently recovered as the unique monic degree-\(n\) factor of \(S_{r,Q_r}\) whose zeros lie in the exterior annulus \(\{|z|>\rho\}\).  The complementary monic factor has all its zeros in \(\{|z|<\rho\}\).
\end{corollary}

\begin{proof}
The zeros of \(S_{r,Q_r}\) are the disjoint union
\[
\{z_1,
\dots,z_n\}\cup\{r^{-2}z_1^{-1},
\dots,r^{-2}z_n^{-1}\}.
\]
The first set lies on \(|z|=1\), and the second lies on \(|z|=r^{-2}\).  The chosen \(\rho\) separates the two shells, so the exterior factor is exactly \(P\).
\end{proof}

\begin{lemma}[Outer Rouch\'e count prevents degree loss]\label{lem:outer-rouche-prevents-degree-loss}
Let \(S\) be a polynomial of degree \(2n\), let \(R>0\), and assume that every zero of \(S\) lies in \(|z|<R\).  Let
\[
\widetilde Q(w)=\sum_{\ell=0}^{n}\widetilde b_\ell w^\ell
\]
with coefficients allowed to satisfy \(\widetilde b_n=0\), and put
\[
\widetilde S(z):=z^n\widetilde Q(J_r(z)),
\qquad J_r(z)=rz+r^{-1}z^{-1}.
\]
Then \(\widetilde S\) is a polynomial with expansion
\[
\widetilde S(z)=
\sum_{\ell=0}^{n}\widetilde b_\ell
\sum_{j=0}^{\ell}\binom{\ell}{j}r^{2j-\ell}z^{n-\ell+2j}.
\]
Its trailing coefficient is \(r^{-n}\widetilde b_n\), and its leading coefficient is \(r^n\widetilde b_n\).  If
\[
|\widetilde S(z)-S(z)|<|S(z)|\qquad(|z|=R),
\]
then \(\widetilde b_n\ne0\), \(\deg\widetilde S=2n\), and \(\widetilde S\) has exactly \(2n\) finite zeros in \(|z|<R\), counted with multiplicity.  Thus no part of the Rouch\'e count is lost through a vanished leading coefficient or a zero at infinity.
\end{lemma}

\begin{proof}
For \(0\le\ell\le n\), the binomial expansion gives
\[
z^nJ_r(z)^\ell
=z^n\sum_{j=0}^{\ell}\binom{\ell}{j}(rz)^j(r^{-1}z^{-1})^{\ell-j}
=\sum_{j=0}^{\ell}\binom{\ell}{j}r^{2j-\ell}z^{n-\ell+2j}.
\]
This proves the displayed expansion and shows that \(\widetilde S\) is a polynomial of degree at most \(2n\).  The only term capable of contributing to degree \(0\) is \(\ell=n,j=0\), so the trailing coefficient is \(r^{-n}\widetilde b_n\).  The only term capable of contributing to degree \(2n\) is \(\ell=n,j=n\), so the leading coefficient is \(r^n\widetilde b_n\).

Rouch\'e's theorem on \(|z|=R\) gives the same number of zeros for \(\widetilde S\) and \(S\) in the disk \(|z|<R\).  Since all \(2n\) zeros of the degree-\(2n\) polynomial \(S\) lie in this disk, \(\widetilde S\) has \(2n\) finite zeros there, counted with multiplicity.  A nonzero polynomial of degree at most \(2n\) with \(2n\) finite zeros must have degree at least \(2n\), hence exactly \(2n\).  The leading coefficient formula then forces \(\widetilde b_n\ne0\), and the trailing coefficient is nonzero as well.  This rules out both the possible degree loss from \(\deg\widetilde Q<n\) and any missing zero at infinity.
\end{proof}

\begin{theorem}[Quantitative annular Rouch\'e stability]\label{thm:quantitative-annular-rouche-stability}
Let \(P\in\mathcal L\) have degree \(n\), let \(r>1\), let \(Q_r\) be its transport polynomial, and define
\[
S_r(z):=S_{r,Q_r}(z)=z^nQ_r(J_r(z)).
\]
Choose radii
\[
r^{-2}<\rho_-<\rho_+<1
\]
and set
\[
\Gamma_-:=\{|z|=\rho_-\},\qquad
\Gamma_+:=\{|z|=\rho_+\},\qquad
\Gamma_0:=\{|z|=\rho_+^{-1}\},
\]
so that the annulus \(\rho_+<|z|<\infty\) contains precisely the \(n\) unit-circle zeros of \(S_r\), while the disk \(|z|<\rho_-\) contains precisely the \(n\) inner zeros.  Define the Rouch\'e envelope
\[
\Delta(r,P;\rho_-,\rho_+):=\frac12\min\left\{
\inf_{z\in\Gamma_-}|S_r(z)|,
\inf_{z\in\Gamma_+}|S_r(z)|,
\inf_{z\in\Gamma_0}|S_r(z)|
\right\}>0.
\]
If \(\widetilde Q\in\CC[w]\) has \(\deg \widetilde Q\le n\) and
\[
\sup_{z\in\Gamma_-\cup\Gamma_+\cup\Gamma_0}
\left|z^n\widetilde Q(J_r(z))-S_r(z)\right|<\Delta(r,P;\rho_-,\rho_+),
\]
then \(\widetilde S(z):=z^n\widetilde Q(J_r(z))\) has exactly \(n\) zeros, counted with multiplicity, in the annulus
\[
\rho_+<|z|<\rho_+^{-1}
\]
and no zeros on \(\Gamma_+\) or \(\Gamma_0\).  The monic degree-\(n\) polynomial formed from the zeros in \(\rho_+<|z|<\rho_+^{-1}\) is the unique canonical annular factor associated with \(\widetilde Q\).  For \(\widetilde Q=Q_r\) this factor is exactly \(P\).

The canonical factor is quantitatively stable in the following local sense.  Write the zero divisor of \(P\) as \(\sum_{a\in Z(P)}m_a[a]\), where \(Z(P)\) is the finite set of distinct zeros.  Let \(U_a\) \((a\in Z(P))\) be pairwise disjoint Jordan domains whose closures lie in \(\rho_+<|z|<\rho_+^{-1}\), with \(a\in U_a\) and with no zero of \(P\) other than \(a\) in \(U_a\).  Put
\[
\Delta_{\mathcal U}(r,P)
:=\frac12\min_{a\in Z(P)}\inf_{z\in\partial U_a}|S_r(z)|>0.
\]
If, in addition,
\[
\sup_{z\in\bigcup_{a\in Z(P)}\partial U_a}
\left|z^n\widetilde Q(J_r(z))-S_r(z)\right|<\Delta_{\mathcal U}(r,P),
\]
then the canonical annular factor \(\widetilde P_{\rm ann}(z)=\prod_{j=1}^{n}(z-\widetilde z_j)\) has exactly \(m_a\) zeros in each \(U_a\).  Consequently, if the domains are chosen so that \(U_a\subset\{|z-a|<\eta\}\) for all \(a\in Z(P)\), then the zeros can be labelled against the zeros \(z_1,\dots,z_n\) of \(P\) so that \(|\widetilde z_j-z_j|<\eta\) for every \(j\).  Writing
\[
P(z)=z^n+\sum_{k=1}^{n}p_k z^{n-k},
\qquad
\widetilde P_{\rm ann}(z)=z^n+\sum_{k=1}^{n}\widetilde p_k z^{n-k},
\]
one has the explicit coefficient estimate
\[
|\widetilde p_k-p_k|
\le \binom{n}{k}k\eta(1+\eta)^{k-1},
\qquad 1\le k\le n.
\]
\end{theorem}

\begin{proof}
The zeros of \(S_r\) are on \(|z|=r^{-2}\) and \(|z|=1\), so none of \(\Gamma_-\), \(\Gamma_+\), or \(\Gamma_0\) meets the zero set.  Hence the three infima in the definition of \(\Delta\) are positive.  On each contour the perturbation is strictly smaller than \(|S_r|\), so Rouch\'e's theorem implies that \(\widetilde S\) and \(S_r\) have the same number of zeros in the disks bounded by these contours.  The outer contour case is exactly Lemma \ref{lem:outer-rouche-prevents-degree-loss} with \(R=\rho_+^{-1}\), since all zeros of \(S_r\) lie in \(|z|<\rho_+^{-1}\); therefore \(\widetilde S\) is degree \(2n\) and the count on \(\Gamma_0\) is a finite zero count rather than a count with missing zeros at infinity.  Thus \(\widetilde S\) has \(n\) zeros in \(|z|<\rho_-\), \(n\) zeros in \(|z|<\rho_+\), and \(2n\) zeros in \(|z|<\rho_+^{-1}\).  Therefore no zeros lie in \(\rho_-<|z|<\rho_+\), and exactly \(n\) zeros lie in \(\rho_+<|z|<\rho_+^{-1}\).  A finite zero divisor determines a unique monic polynomial, so those \(n\) annular zeros define a unique canonical annular factor.  In the unperturbed case the annular zeros are precisely the zeros of \(P\).

For the local stability statement, the boundaries \(\partial U_a\) are compact and avoid the zero set of \(S_r\), while each \(U_a\) contains exactly the multiplicity \(m_a\) zero of \(S_r\) at \(a\).  Hence \(\Delta_{\mathcal U}(r,P)>0\).  The additional perturbation bound and Rouch\'e's theorem on every \(\partial U_a\) show that \(\widetilde S\) has exactly \(m_a\) zeros in \(U_a\), counted with multiplicity.  These zeros are among the annular zeros already selected, because every \(\overline{U_a}\) lies inside \(\rho_+<|z|<\rho_+^{-1}\), and their total number is \(\sum_a m_a=n\); therefore they are precisely the zeros of \(\widetilde P_{\rm ann}\).  If \(U_a\subset\{|z-a|<\eta\}\), label the \(m_a\) zeros in \(U_a\) against the \(m_a\) copies of the zero \(a\) of \(P\).  This gives \(|\widetilde z_j-z_j|<\eta\).  Since \(|z_j|=1\), the labelled zeros satisfy \(|\widetilde z_j|\le1+\eta\).  For each subset \(I\) of cardinality \(k\), the telescoping identity gives
\[
\left|\prod_{j\in I}\widetilde z_j-
\prod_{j\in I}z_j\right|
\le k\eta(1+\eta)^{k-1}.
\]
Summing over the \(\binom nk\) subsets and using \(p_k=(-1)^ke_k(z_1,\dots,z_n)\), \(\widetilde p_k=(-1)^ke_k(\widetilde z_1,\dots,\widetilde z_n)\), yields the coefficient estimate.
\end{proof}

\begin{corollary}[Coefficient-level annular stability certificate]\label{cor:coefficient-annular-rouche-stability}
Retain the hypotheses and notation of Theorem \ref{thm:quantitative-annular-rouche-stability}, and write
\[
Q_r(w)=\sum_{\ell=0}^{n}b_\ell w^\ell,
\qquad
\widehat Q(w)=\sum_{\ell=0}^{n}\widehat b_\ell w^\ell,
\]
allowing \(\widehat b_n=0\) if \(\deg\widehat Q<n\).  Put
\[
\mathfrak B_n(r;\rho_-,\rho_+)
:=
\max_{z\in\Gamma_-\cup\Gamma_+\cup\Gamma_0}
|z|^n\sum_{\ell=0}^{n}|J_r(z)|^\ell.
\]
If the coefficient errors satisfy
\[
\max_{0\le \ell\le n}|\widehat b_\ell-b_\ell|
<
\frac{\Delta(r,P;\rho_-,\rho_+)}{\mathfrak B_n(r;\rho_-,\rho_+)},
\]
then \(\widehat S(z):=z^n\widehat Q(J_r(z))\) satisfies the Rouch\'e envelope in Theorem \ref{thm:quantitative-annular-rouche-stability}.  Consequently \(\widehat Q\) has a unique canonical annular factor of degree \(n\), and for \(\widehat Q=Q_r\) this factor is exactly \(P\).  More generally, the same conclusion holds under the weighted \(\ell^1\) coefficient condition
\[
\max_{z\in\Gamma_-\cup\Gamma_+\cup\Gamma_0}
|z|^n\sum_{\ell=0}^{n}|\widehat b_\ell-b_\ell|\,|J_r(z)|^\ell
<\Delta(r,P;\rho_-,\rho_+).
\]
\end{corollary}

\begin{proof}
For each point of the three contours,
\[
\left|z^n\widehat Q(J_r(z))-S_r(z)\right|
=
\left|z^n\sum_{\ell=0}^{n}(\widehat b_\ell-b_\ell)J_r(z)^\ell\right|
\le
|z|^n\sum_{\ell=0}^{n}|\widehat b_\ell-b_\ell|\,|J_r(z)|^\ell.
\]
The weighted \(\ell^1\) hypothesis is exactly the assertion that the right side is uniformly smaller than \(\Delta(r,P;\rho_-,\rho_+)\).  The simpler maximum-coefficient hypothesis implies it because the last display is at most
\[
\max_{0\le \ell\le n}|\widehat b_\ell-b_\ell|\,
\mathfrak B_n(r;\rho_-,\rho_+).
\]
Thus the contour hypothesis of Theorem \ref{thm:quantitative-annular-rouche-stability} holds with \(\widetilde Q=\widehat Q\), and the announced annular factor statement follows from that theorem.
\end{proof}
