\begin{lemma}[Joukowsky boundary kernel diagonalization]\label{lem:joukowsky-boundary-kernel-diagonalization}
Let \(r>1\), let \(J_r(z)=rz+r^{-1}z^{-1}\), and let
\(\mu\in\mathcal P(\TT)\) have finite circle logarithmic energy.  Then
\[
I(J_{r\ast}\mu)
=
\log r-\sum_{k=1}^{\infty}
\frac{|\widehat\mu(k)|^2+r^{-2k}\operatorname{Re}(\widehat\mu(k)^2)}{k}.
\]
The identity is a finite-energy identity only: it is obtained by applying
the lower-truncated circle Fourier formula to the diagonal singularity and
ordinary uniform Fourier integration to the remaining continuous factor.
\end{lemma}

\begin{proof}
For \(z=e^{i\theta}\) and \(w=e^{i\phi}\), the degree-two boundary
mechanism is the factorization
\[
J_r(z)-J_r(w)
=(z-w)\left(r-r^{-1}z^{-1}w^{-1}\right).
\]
Equivalently,
\[
\log|J_r(e^{i\theta})-J_r(e^{i\phi})|
=\log r+\log|e^{i\theta}-e^{i\phi}|
 +\log|1-r^{-2}e^{-i(\theta+\phi)}|.
\]
The first term after \(\log r\) is the only singular term.  By Lemma
\ref{lem:circle-log-kernel-fourier-truncation},
\[
\iint\log|e^{i\theta}-e^{i\phi}|\,\dd\mu(\theta)\dd\mu(\phi)
=-\sum_{k=1}^{\infty}\frac{|\widehat\mu(k)|^2}{k}.
\]
The second factor is continuous because \(r^{-2}<1\), and its Fourier
series is uniformly and absolutely convergent:
\[
\log|1-r^{-2}e^{-i(\theta+\phi)}|
=-\sum_{k=1}^{\infty}
\frac{r^{-2k}\operatorname{Re}(e^{-ik(\theta+\phi)})}{k}.
\]
Termwise integration gives
\[
\iint\log|1-r^{-2}e^{-i(\theta+\phi)}|
\,\dd\mu(\theta)\dd\mu(\phi)
=-\sum_{k=1}^{\infty}
\frac{r^{-2k}\operatorname{Re}(\widehat\mu(k)^2)}{k}.
\]
Combining the singular diagonal contribution with the continuous
reciprocal-scale contribution proves the formula.
\end{proof}

\begin{theorem}[Theorem A: finite-energy Joukowsky Fourier identity]\label{thm:finite-energy-joukowsky-fourier-identity}
Let \(r>1\), let \(J_r(z)=rz+r^{-1}z^{-1}\), and let
\[
E_r=J_r(\TT),\qquad \mu_r=J_{r\ast}\!\left(\frac{\dd\theta}{2\pi}\right).
\]
For a probability measure \(\mu\) on \(\TT\), write
\[
\widehat\mu(k)=\int_{0}^{2\pi}e^{-ik\theta}\,\dd\mu(\theta),\qquad k\in\ZZ,
\]
and let
\[
I(\sigma):=\iint\log|x-y|\,\dd\sigma(x)\dd\sigma(y)
\]
denote logarithmic energy as the lower-truncated extended logarithmic integral.
If \(I_{\TT}(\mu)>-\infty\), equivalently \(J_{r\ast}\mu\) has finite logarithmic energy on \(E_r\), then
\[
\boxed{\;
\log r-I(J_{r\ast}\mu)
=\sum_{k=1}^{\infty}
\frac{|\widehat\mu(k)|^2+r^{-2k}\operatorname{Re}(\widehat\mu(k)^2)}{k}.
\;}
\]
This displayed equality is a finite-energy identity in the sense of the
standing conventions, not an assertion obtained by termwise integration for
arbitrary probability measures.
Consequently,
\[
\sum_{k=1}^{\infty}\frac{(1-r^{-2k})|\widehat\mu(k)|^2}{k}
\le \log r-I(J_{r\ast}\mu)
\le
\sum_{k=1}^{\infty}\frac{(1+r^{-2k})|\widehat\mu(k)|^2}{k}.
\]
For example, \(\mu=\delta_1\) falls outside this identity:
\(I(J_{r\ast}\delta_1)=-\infty\), while \(\widehat\mu(k)=1\) for all
\(k\ge1\) and the displayed Fourier series diverges.
For each \(k\ge1\) the constants \(1+r^{-2k}\) and \(1-r^{-2k}\) are sharp on the \(k\)-th Fourier mode: for \(|\varepsilon|\le1/2\),
\[
\dd\mu^c_{k,\varepsilon}=(1+2\varepsilon\cos k\theta)\frac{\dd\theta}{2\pi},
\qquad
\dd\mu^s_{k,\varepsilon}=(1+2\varepsilon\sin k\theta)\frac{\dd\theta}{2\pi}
\]
satisfy
\[
\log r-I(J_{r\ast}\mu^c_{k,\varepsilon})=\frac{(1+r^{-2k})\varepsilon^2}{k},
\qquad
\log r-I(J_{r\ast}\mu^s_{k,\varepsilon})=\frac{(1-r^{-2k})\varepsilon^2}{k}.
\]
\end{theorem}

\begin{proof}
The exact finite-energy diagonalization is Lemma
\ref{lem:joukowsky-boundary-kernel-diagonalization}.  By Lemma
\ref{lem:joukowsky-boundary-energy-equivalence}, the same hypothesis is
equivalently the condition that the pushed-forward boundary measure
\(J_{r\ast}\mu\) have finite energy on \(E_r\).  The identity is not used
for arbitrary measures because the singular circle Fourier series need not
represent a finite energy outside that class.

Writing \(\widehat\mu(k)=x_k+i y_k\), the \(k\)-th summand numerator is
\[
|\widehat\mu(k)|^2+r^{-2k}\operatorname{Re}(\widehat\mu(k)^2)
=(1+r^{-2k})x_k^2+(1-r^{-2k})y_k^2,
\]
which proves the two modewise bounds.  Equality in the left or right coefficient for a fixed mode is attained by purely imaginary or purely real \(\widehat\mu(k)\), respectively.  The densities \(\mu^c_{k,\varepsilon}\) and \(\mu^s_{k,\varepsilon}\) are nonnegative probability measures for \(|\varepsilon|\le1/2\), have no nonzero Fourier coefficients except in modes \(\pm k\), and have \(\widehat\mu^c_{k,\varepsilon}(k)=\varepsilon\) and \(\widehat\mu^s_{k,\varepsilon}(k)=-i\varepsilon\).  The displayed extremal identities follow.

Since every term in the deficit is nonnegative and \(1-r^{-2k}>0\), one has \(I(J_{r\ast}\mu)\le\log r\) for finite-energy source measures, with equality only when \(\widehat\mu(k)=0\) for all \(k\ne0\), that is, when \(\mu\) is Haar measure on \(\TT\).
\end{proof}

\begin{lemma}[Abel passage and Frostman extremality are separate]\label{lem:abel-passage-frostman-separation}
Let \(r>1\), let \(J_r(z)=rz+r^{-1}z^{-1}\), and put
\(E_r=J_r(\TT)\).  For \(0<\rho<1\), set
\[
K_{\rho,r}(\theta,\phi)
:=\log r+\log|1-\rho e^{i(\theta-\phi)}|
  +\log|1-r^{-2}e^{-i(\theta+\phi)}|.
\]
Then the following two assertions have different hypotheses.
\begin{enumerate}[label=\textup{(\alph*)}]
  \item If \(\mu\in\mathcal P(\TT)\) has finite circle logarithmic energy, then
  \[
  \lim_{\rho\uparrow1}
  \iint K_{\rho,r}(\theta,\phi)\,\dd\mu(\theta)\dd\mu(\phi)
  =
  I(J_{r\ast}\mu),
  \]
  with the limit equal to the lower-truncated logarithmic energy on \(E_r\).  Equivalently the Abel-regularized identity passes to
  \[
  I(J_{r\ast}\mu)
  =
  \log r-\sum_{k\ge1}
  \frac{|\widehat\mu(k)|^2+r^{-2k}\operatorname{Re}(\widehat\mu(k)^2)}{k}.
  \]
  No such Fourier identity is asserted outside the finite-energy class.

  \item For every \(\eta\in\mathcal P(\TT)\), with no finite-energy assumption,
  \[
  I(J_{r\ast}\eta)\le \log r,
  \]
  and equality holds if and only if \(\eta\) is Haar measure on \(\TT\).
\end{enumerate}
\end{lemma}

\begin{proof}
For \(\textup{(a)}\), the boundary factorization gives, for
\(z=e^{i\theta}\) and \(w=e^{i\phi}\),
\[
\log|J_r(z)-J_r(w)|
=\log r+\log|1-e^{i(\theta-\phi)}|
 +\log|1-r^{-2}e^{-i(\theta+\phi)}|.
\]
The first singular term is exactly the Abel passage proved under the
finite-energy hypothesis in Lemma
\ref{lem:circle-log-kernel-fourier-truncation}.  The second term is a
continuous kernel, since \(r^{-2}<1\), and its Fourier series is uniformly
and absolutely convergent on \(\TT^2\).  Thus its integral is obtained by
ordinary termwise integration, while the singular term is obtained only as
the lower-truncated Abel limit supplied by that lemma.  Combining the two
terms gives the displayed identity, and the proof uses finite energy
precisely to identify the Abel limit with the lower-truncated singular
circle energy.

For \(\textup{(b)}\), first compute the capacity normalization by the
exterior map itself.  Since
\[
J_r'(z)=r-r^{-1}z^{-2}
\]
has its critical points \(z=\pm r^{-1}\) inside \(\{|z|<1\}\), the branch
of \(J_r\) on \(\{|z|>1\}\) maps the exterior disk conformally onto
\(\CC\setminus E_r\).  Its inverse branch \(z_+(w)\), characterized by
\(|z_+(w)|>1\), is obtained from
\[
r z^2-wz+r^{-1}=0
\]
and has the expansion
\[
z_+(w)=\frac{w}{r}+O(w^{-1}),\qquad w\to\infty .
\]
Pulling back the Green function of the exterior unit disk therefore gives
\[
g_{E_r}(w)=\log|z_+(w)|
=\log|w|-\log r+o(1),\qquad w\to\infty .
\]
With the standard normalization
\[
g_K(w)=\log|w|-\log\operatorname{cap}(K)+o(1)
\]
for a compact nonpolar set \(K\), the constant term is \(-\log r\), hence
\(\operatorname{cap}(E_r)=r\).  Harmonic measure from infinity is
\[
\mu_r=J_{r\ast}\!\left(\frac{\dd\theta}{2\pi}\right).
\]
Frostman's variational principle for the compact nonpolar set \(E_r\)
therefore gives, for every \(\nu\in\mathcal P(E_r)\),
\[
I(\nu)\le\log\operatorname{cap}(E_r)=\log r,
\]
with equality precisely for the equilibrium measure \(\nu=\mu_r\).  Apply
this to \(\nu=J_{r\ast}\eta\).  Since \(J_r:\TT\to E_r\) is a homeomorphism
by Corollary \ref{cor:degree-two-boundary-operativity}, the equality
condition \(J_{r\ast}\eta=\mu_r\) is equivalent to
\(\eta=\dd\theta/(2\pi)\).  This proof of the all-measure extremal
statement does not use the Abel--Fourier identity from \(\textup{(a)}\).
\end{proof}

\begin{theorem}[Theorem B: all-measure Joukowsky equilibrium bound]\label{thm:sharp-joukowsky-equilibrium-stability}\label{thm:all-measure-joukowsky-equilibrium-bound}
Let \(r>1\), let \(J_r(z)=rz+r^{-1}z^{-1}\), put
\(E_r=J_r(\TT)\), and interpret logarithmic energies by lower
truncation.  For every \(\eta\in\mathcal P(\TT)\),
\[
I(J_{r\ast}\eta)\le \log r,
\]
with equality if and only if \(\eta\) is Haar measure.  Equivalently,
\(\operatorname{cap}(E_r)=r\) and
\[
\mu_r=J_{r\ast}\!\left(\frac{\dd\theta}{2\pi}\right)
\]
is the unique equilibrium measure of \(E_r\).
As in the standing conventions, this is the all-measure variational bound
only; the Fourier identity remains the finite-energy statement of Theorem
\ref{thm:finite-energy-joukowsky-fourier-identity}.
\end{theorem}

\begin{proof}
The all-measure assertion is Lemma
\ref{lem:abel-passage-frostman-separation}\textup{(b)}.  That lemma also
records separately that the Abel--Fourier passage belongs only to the
finite-energy statement in Lemma
\ref{lem:abel-passage-frostman-separation}\textup{(a)} and Theorem
\ref{thm:finite-energy-joukowsky-fourier-identity}.
\end{proof}

\begin{corollary}[Sharp boundary descent for Joukowsky covers]\label{cor:sharp-boundary-descent-joukowsky-covers}
Let \(r>1\), put \(E_r=J_r(\TT)\), and interpret logarithmic energies by lower truncation.  The exact descent theorem and the sharp equilibrium theorem combine as follows.
\begin{enumerate}[label=\textup{(\alph*)}]
  \item The boundary variational problem has the sharp extended-energy form
  \[
  \sup_{\mu\in\mathcal P(\TT)} I(J_{r\ast}\mu)=\log r,
  \qquad
  I(J_{r\ast}\mu)=\log r
  \Longleftrightarrow
  \mu=\frac{\dd\theta}{2\pi}.
  \]

  \item If \(B\) is a finite Blaschke product and \(\eta=B_\ast\mu\), then
  \[
  I((J_r\circ B)_\ast\mu)=I(J_{r\ast}\eta)
  \]
  for every \(\mu\in\mathcal P(\TT)\).  Consequently
  \[
  \sup_{\mu\in\mathcal P(\TT)}I((J_r\circ B)_\ast\mu)=\log r,
  \qquad
  I((J_r\circ B)_\ast\mu)=\log r
  \Longleftrightarrow
  B_\ast\mu=\frac{\dd\theta}{2\pi}.
  \]
\end{enumerate}
Thus the exact boundary descent is a kernel and pushforward statement, while the sharp value \(\log r\) and the Haar-lift equality class enter only after Theorem \ref{thm:sharp-joukowsky-equilibrium-stability}.
\end{corollary}

\begin{proof}
The sharp Joukowsky equilibrium theorem gives
\[
I(J_{r\ast}\eta)\le\log r
\]
for every probability measure \(\eta\) on \(\TT\), with equality exactly when \(\eta=\dd\theta/(2\pi)\).  Taking \(\eta=\mu\) proves \textup{(a)}.  Since \(J_r:\TT\to E_r\) is a homeomorphism by Theorem \ref{thm:exact-boundary-descent-joukowsky-covers}, this is also the unique equilibrium statement on \(E_r\).

For a finite Blaschke product \(B\), Theorem \ref{thm:exact-boundary-descent-joukowsky-covers} gives the energy identity
\[
I((J_r\circ B)_\ast\mu)=I(J_{r\ast}(B_\ast\mu)).
\]
Applying the sharp equilibrium bound to \(\eta=B_\ast\mu\) gives the covered supremum and the equality condition \(B_\ast\mu=\dd\theta/(2\pi)\).  Haar lifts exist for every finite Blaschke product by Lemma \ref{lem:finite-blaschke-boundary-covering-formula}, so the supremum is attained.
Equivalently, Lemma \ref{lem:finite-blaschke-haar-lift-fiber} identifies the whole maximizer set as the Haar-lift fiber through \(B\).
\end{proof}

\begin{corollary}[Calibrated two-test identity for the Joukowsky scale]\label{cor:two-mode-joukowsky-scale-rigidity}
Fix an integer \(k\ge1\) and a real number \(\varepsilon\) with \(0<|\varepsilon|\le1/2\).  For \(r>1\), let
\[
\dd\mu^c_{k,\varepsilon}=(1+2\varepsilon\cos k\theta)\frac{\dd\theta}{2\pi},
\qquad
\dd\mu^s_{k,\varepsilon}=(1+2\varepsilon\sin k\theta)\frac{\dd\theta}{2\pi}.
\]
Then, within this fixed probe family and not as a general inverse invariant, the calibrated ordered pair of boundary pushforward energies
\[
\mathcal A_k(r,\varepsilon):=
\left(
I(J_{r\ast}\mu^c_{k,\varepsilon}),
I(J_{r\ast}\mu^s_{k,\varepsilon})
\right)
\]
determines the Joukowsky scale \(r\) uniquely within this prescribed two-test family.  More precisely, if
\[
E_c=I(J_{r\ast}\mu^c_{k,\varepsilon}),
\qquad
E_s=I(J_{r\ast}\mu^s_{k,\varepsilon}),
\]
then
\[
E_s-E_c=\frac{2\varepsilon^2}{k}r^{-2k},
\qquad
\log r=\frac{E_c+E_s}{2}+\frac{\varepsilon^2}{k},
\]
and hence
\[
\boxed{\;
r=\left(\frac{2\varepsilon^2}{k(E_s-E_c)}\right)^{1/(2k)}
=\exp\!\left(\frac{E_c+E_s}{2}+\frac{\varepsilon^2}{k}\right).
\;}
\]
Consequently, for \(r,s>1\), equality
\[
\mathcal A_k(r,\varepsilon)=\mathcal A_k(s,\varepsilon)
\]
forces \(r=s\).  The same conclusion holds after a finite Blaschke cover \(B\) whenever the pushed-forward probe measures \(B_\ast\nu^c\) and \(B_\ast\nu^s\) are \(\mu^c_{k,\varepsilon}\) and \(\mu^s_{k,\varepsilon}\), respectively; in that case the two covered energies
\[
I((J_r\circ B)_\ast\nu^c),
\qquad
I((J_r\circ B)_\ast\nu^s)
\]
  recover the same \(r\).  This corollary is not an inverse theorem for arbitrary energy observations: two unspecified energies, even if taken from measures on the same ellipse, need not determine \(r\) without the prescribed probe family and its normalization.
\end{corollary}

\begin{proof}
The measures \(\mu^c_{k,\varepsilon}\) and \(\mu^s_{k,\varepsilon}\) have bounded densities on \(\TT\), hence finite logarithmic energy.  The sharp mode computation in Theorem \ref{thm:finite-energy-joukowsky-fourier-identity} therefore applies and gives
\[
\log r-I(J_{r\ast}\mu^c_{k,\varepsilon})=\frac{(1+r^{-2k})\varepsilon^2}{k},
\qquad
\log r-I(J_{r\ast}\mu^s_{k,\varepsilon})=\frac{(1-r^{-2k})\varepsilon^2}{k}.
\]
With \(E_c\) and \(E_s\) as in the statement, subtracting the first energy identity from the second gives
\[
E_s-E_c=
\left(\log r-\frac{(1-r^{-2k})\varepsilon^2}{k}\right)
-
\left(\log r-\frac{(1+r^{-2k})\varepsilon^2}{k}\right)
=\frac{2\varepsilon^2}{k}r^{-2k}.
\]
This quantity is positive because \(\varepsilon\ne0\) and \(r>1\), so the first displayed formula for recovering \(r\) follows.  Adding the two identities gives
\[
E_c+E_s=2\log r-\frac{2\varepsilon^2}{k},
\]
which is the logarithmic recovery formula.  Either formula is injective on \((1,\infty)\), so equality of the ordered pairs for two scales \(r,s>1\) implies \(r=s\).

For the covered assertion, the descent identity in Theorem \ref{thm:exact-boundary-descent-joukowsky-covers} gives
\[
I((J_r\circ B)_\ast\nu)=I(J_{r\ast}(B_\ast\nu))
\]
for every probability measure \(\nu\) on \(\TT\).  Applying this first to \(\nu=\nu^c\) and then to \(\nu=\nu^s\) reduces the two covered energies to the two already treated probe energies for \(\mu^c_{k,\varepsilon}\) and \(\mu^s_{k,\varepsilon}\).  The same recovery formulas therefore recover \(r\) after the finite Blaschke lift.
\end{proof}

\begin{proposition}[Probe calibration is necessary]\label{prop:two-mode-calibration-necessary}
Fix \(k\ge1\).  The scale recovery in Corollary
\ref{cor:two-mode-joukowsky-scale-rigidity} is a statement about the
prescribed ordered cosine--sine probe family and not about arbitrary
boundary energy observations.  More precisely, for any two scales
\(1<r<s\) there are finite-energy probability measures
\(\mu_r,\nu_r,\mu_s,\nu_s\) on \(\TT\), all with smooth positive
density, such that
\[
I(J_{r\ast}\mu_r)=I(J_{s\ast}\mu_s),
\qquad
I(J_{r\ast}\nu_r)=I(J_{s\ast}\nu_s).
\]
Thus an unordered or uncalibrated pair of finite-energy observations
does not determine the scale \(r\); the recovery formulas in Corollary
\ref{cor:two-mode-joukowsky-scale-rigidity} use the fixed probes and
the known normalization \((k,\varepsilon)\).
\end{proposition}

\begin{proof}
Fix a scale \(\rho>1\).  Let \(h_N(\theta)\dd\theta/(2\pi)\) be any
smooth positive approximate identity on \(\TT\), normalized to have
total mass one, and put
\[
\dd\lambda_{\rho,N,a}
=\bigl((1-a)+a h_N(\theta)\bigr)\frac{\dd\theta}{2\pi},
\qquad 0\le a\le1.
\]
For fixed \(N\), the density is smooth and positive, and
the energy \(a\mapsto I(J_{\rho\ast}\lambda_{\rho,N,a})\) is continuous
on \([0,1]\), because the pushed-forward density is smooth on the
analytic Jordan curve \(E_\rho\) and the logarithmic kernel is locally
integrable against such densities.  At \(a=0\) the measure is Haar, so
Theorem \ref{thm:sharp-joukowsky-equilibrium-stability} gives the value
\(\log\rho\).

As \(N\to\infty\), the self-interaction of \(h_N(\theta)\dd\theta\)
concentrates on the diagonal.  Since \(J_\rho:\TT\to E_\rho\) is a
homeomorphism and the local boundary kernel is
\(\log|\theta-\phi|+O(1)\), these smooth measures have
\[
I(J_{\rho\ast}(h_N\,\dd\theta/(2\pi)))\longrightarrow -\infty.
\]
Consequently, for every \(E<\log\rho\) there are \(N\) and \(a\in[0,1]\) with
\[
I(J_{\rho\ast}\lambda_{\rho,N,a})=E.
\]

Choose two distinct real numbers \(E,E'<\log r\).  Since \(r<s\), both
also lie below \(\log s\).  Applying the preceding paragraph at the two
scales \(\rho=r\) and \(\rho=s\) gives smooth positive source measures
whose pushed-forward Joukowsky energies are \(E\), and then again whose
energies are \(E'\).  Naming these four measures
\(\mu_r,\mu_s,\nu_r,\nu_s\) proves the displayed equalities.  The
measures depend on the scale and on the target energy; this is exactly
the freedom excluded by the calibrated probe family in Corollary
\ref{cor:two-mode-joukowsky-scale-rigidity}.
\end{proof}

\begin{proposition}[All-measure Joukowsky equilibrium bound]\label{prop:extended-energy-joukowsky-equilibrium-bound}
Let \(r>1\) and let \(\mu\) be an arbitrary Borel probability measure on \(\TT\).  With logarithmic energies interpreted by lower truncation,
\[
I(J_{r\ast}\mu)\le \log r.
\]
Equality holds if and only if \(\mu\) is Haar measure on \(\TT\).  Equivalently, for every probability measure \(\eta\) on \(\TT\), finite source energy is not needed for the sharp equilibrium bound
\[
I(J_{r\ast}\eta)\le \log r,
\]
and the unique maximizer is \(\eta=\dd\theta/(2\pi)\).
This proposition records only the all-measure variational inequality.  The
Fourier deficit formula remains restricted to the finite-energy class in
Theorem \ref{thm:finite-energy-joukowsky-fourier-identity}.
\end{proposition}

\begin{proof}
This is Theorem \ref{thm:all-measure-joukowsky-equilibrium-bound},
repeated here for later reference in the finite-cover argument.  The
final formulation is the same assertion with \(\eta\) in place of
\(\mu\).
\end{proof}
