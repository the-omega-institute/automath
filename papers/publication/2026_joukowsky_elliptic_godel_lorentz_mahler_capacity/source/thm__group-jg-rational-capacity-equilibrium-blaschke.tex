We use the following standard potential-theoretic tools throughout this section.  For compact nonpolar plane sets, logarithmic capacity is the variational supremum of the lower-truncated energy \(I\), and the supremum is attained by a unique equilibrium measure; see the compact-set equilibrium discussion in \citet[Chap.~5, pp.~105--107]{Ransford1995}, or equivalently the classical Frostman equilibrium theorem in Tsuji's terminology \citep[Chap.~III, Secs.~15--17]{Tsuji1959}.  The remaining set-theoretic inputs needed for the rational pullback formula are isolated in the next three lemmas: nonpolarity of rational images, equality of lower-truncated pullback and pushforward energies, and a compact-fiber Borel selector.  Thus the new content of Theorem \ref{thm:finite-degree-rational-image-capacity} is the explicit rational pullback kernel and its branch/collision bookkeeping, not the classical existence theory.

\begin{lemma}[Nonpolarity is preserved by nonconstant rational images]\label{lem:nonpolar-rational-image}
Let \(K\subset\CC\) be compact and nonpolar, and let \(R\) be a nonconstant rational map with no pole on \(K\).  Then \(R(K)\) is compact and nonpolar.
\end{lemma}

\begin{proof}
Compactness follows from continuity of \(R\) on \(K\).  We next give the standard pullback argument in full.  Suppose, for a contradiction, that \(R(K)\) is polar.  By the subharmonic characterization of polar sets, in the form complementary to Cartan's theorem \citep[Thm.~3.5.1, p.~65]{Ransford1995}, there is a subharmonic function \(u\not\equiv -\infty\) on a neighbourhood of \(R(K)\) such that \(u=-\infty\) on \(R(K)\).  Since \(R\) has no pole on \(K\), choose an open neighbourhood \(U\) of \(K\), with no component on which \(R\) is constant after discarding components disjoint from \(K\) if necessary, on which \(R\) is holomorphic.  The composition \(u\circ R\) is subharmonic on \(U\) unless it is identically \(-\infty\) on a connected component.  On no component meeting \(K\) can it be identically \(-\infty\): otherwise the open mapping theorem would force \(u\equiv -\infty\) on a nonempty open subset of the domain of \(u\), hence everywhere on the relevant component, contrary to the choice of \(u\).  Thus \(u\circ R\) is a genuine subharmonic function on a neighbourhood of \(K\) and equals \(-\infty\) on \(K\).  This makes \(K\) polar, contradicting the hypothesis.
\end{proof}

\begin{lemma}[Extended pullback energy equals pushforward energy]\label{lem:extended-pullback-energy-equals-pushforward-energy}
Let \(X\) be a compact metric space, let \(F:X\to\CC\) be continuous, and let \(\mu\in\mathcal P(X)\).  Define
\[
L_F(x,y):=
\begin{cases}
\log|F(x)-F(y)|,&F(x)\ne F(y),\\
-\infty,&F(x)=F(y),
\end{cases}
\]
and let \(\nu=F_\ast\mu\).  If both logarithmic integrals are interpreted by lower truncation, namely as the limit of the kernels \(\max\{\log|\cdot|,-M\}\) as \(M\to\infty\), then
\[
\iint_{X\times X}L_F(x,y)\,\dd\mu(x)\dd\mu(y)
=
\iint_{F(X)\times F(X)}\log|u-v|\,\dd\nu(u)\dd\nu(v)
\]
as an identity in \([ -\infty,\infty)\).  In particular the left-hand side is greater than \(-\infty\) if and only if the pushforward logarithmic energy is greater than \(-\infty\).
\end{lemma}

\begin{proof}
For \(M>0\), put \(h_M(u,v)=\max\{\log|u-v|,-M\}\).  This is a bounded Borel function on the compact set \(F(X)\times F(X)\), with the value \(-M\) on the diagonal.  Since \((F\times F)_\ast(\mu\times\mu)=\nu\times\nu\),
\[
\iint_{X\times X} h_M(F(x),F(y))\,\dd\mu(x)\dd\mu(y)
=
\iint_{F(X)\times F(X)}h_M(u,v)\,\dd\nu(u)\dd\nu(v).
\]
As \(M\to\infty\), the kernels \(h_M\) decrease pointwise to the extended logarithmic kernel \(\log|u-v|\), with value \(-\infty\) on the diagonal.  The image \(F(X)\) is compact, so these kernels are bounded above uniformly.  Applying monotone convergence to a fixed upper bound minus the two sides gives equality of the lower-truncated limits.  The final equivalence is the same identity read in the extended real line.
\end{proof}

\begin{lemma}[Compact-fiber Borel selection]\label{lem:compact-fiber-borel-selection}
Let \(X\) and \(Y\) be compact metric spaces, and let \(G\subset X\times Y\) be closed with every vertical section
\[
G_x:=\{y\in Y:(x,y)\in G\}
\]
nonempty.  Then there is a Borel map \(s:X\to Y\) such that \((x,s(x))\in G\) for every \(x\in X\).
\end{lemma}

\begin{proof}
This is the compact-valued case of the Kuratowski--Ryll-Nardzewski measurable-selection theorem, equivalently the closed-section form of the Jankov--von Neumann uniformization theorem for standard Borel spaces \citep[Theorem~18.13]{Kechris1995}.  Indeed, compact metric spaces are Polish and carry their standard Borel structures, while the multifunction \(x\mapsto G_x\) has nonempty closed values and Borel graph, because \(G\) is closed in \(X\times Y\).  The theorem therefore gives a Borel selector \(s:X\to Y\) with \(s(x)\in G_x\) for every \(x\in X\), which is precisely \((x,s(x))\in G\).
\end{proof}

\begin{lemma}[Divided-difference kernel for rational pullbacks]\label{lem:rational-divided-difference-kernel}
Let \(R=A/B\) be a rational map with coprime \(A,B\in\CC[z]\), and set
\[
\Phi_R(z,t):=A(z)B(t)-A(t)B(z).
\]
Then \(\Phi_R\) is divisible by \(z-t\) in \(\CC[z,t]\).  The quotient
\[
\Psi_R(z,t):=\frac{\Phi_R(z,t)}{z-t}
\]
is therefore a polynomial in the two variables and has diagonal value
\[
\Psi_R(z,z)=A'(z)B(z)-A(z)B'(z).
\]
If \(K\subset\CC\) is compact and \(B\ne0\) on \(K\), then, as lower-truncated extended kernels on \(K\times K\),
\[
\log|R(z)-R(t)|
=
\log|z-t|+\log|\Psi_R(z,t)|-
\log|B(z)|-\log|B(t)|,
\]
where both sides take the value \(-\infty\) on the collision set \(R(z)=R(t)\).  Consequently, for every \(\mu\in\mathcal P(K)\), the lower-truncated integral of the displayed pullback kernel is finite if and only if \(R_\ast\mu\) has finite logarithmic energy.
\end{lemma}

\begin{proof}
The polynomial \(\Phi_R\) vanishes identically on the diagonal \(z=t\).  Dividing \(\Phi_R\) by \(z-t\), viewed as a monic polynomial in \(z\) with coefficients in \(\CC[t]\), leaves a remainder depending only on \(t\); evaluating at \(z=t\) shows that this remainder is zero.  Hence \(\Psi_R\in\CC[z,t]\).  Taking the limit as \(t\to z\), equivalently differentiating \(\Phi_R(z,t)\) in the second variable and accounting for the denominator \(z-t\), gives
\[
\Psi_R(z,z)=A'(z)B(z)-A(z)B'(z).
\]

For \(z\ne t\) with \(B(z)B(t)\ne0\),
\[
R(z)-R(t)=\frac{A(z)B(t)-A(t)B(z)}{B(z)B(t)}
=\frac{(z-t)\Psi_R(z,t)}{B(z)B(t)}.
\]
Taking logarithmic absolute values gives the asserted identity away from the collision set.  On the collision set the factor \((z-t)\Psi_R(z,t)\) vanishes and the logarithmic kernel is \(-\infty\); the two terms involving \(B\) remain bounded and continuous on \(K\), because \(B\) has no zeros there.  Thus the equality holds for the lower-truncated extended kernels.  Applying Lemma \ref{lem:extended-pullback-energy-equals-pushforward-energy} to the continuous map \(R:K\to R(K)\) proves the final equivalence of finite energies.
\end{proof}


\begin{theorem}[Finite-degree rational-image capacity and equilibrium lifts]\label{thm:finite-degree-rational-image-capacity}
Let \(R=A/B\) be a nonconstant rational map of finite degree, with coprime polynomials \(A,B\in\CC[z]\), and let \(K\subset\CC\) be compact and nonpolar with \(B\ne0\) on \(K\).  Put
\[
\Psi_R(z,t):=\frac{A(z)B(t)-A(t)B(z)}{z-t}\qquad (z\ne t),
\]
and extend this polynomial to the diagonal by
\[
\Psi_R(z,z)=A'(z)B(z)-A(z)B'(z).
\]
For a Borel probability measure \(\mu\) on \(K\), define
\[
\mathcal E_R(\mu):=\iint_{K\times K}
\Bigl(\log|z-t|+\log|\Psi_R(z,t)|-\log|B(z)|-\log|B(t)|\Bigr)
\,\dd\mu(z)\dd\mu(t),
\]
where the integral is understood as the monotone limit of the kernels truncated below.  A measure is called an admissible finite-energy lift if \(R_\ast\mu\) has finite logarithmic energy on \(R(K)\); by Lemma \ref{lem:extended-pullback-energy-equals-pushforward-energy}, this is equivalent to the displayed integral being greater than \(-\infty\).  Then
\[
\boxed{\;
\log\operatorname{cap}(R(K))
=\sup_{\mu}\mathcal E_R(\mu),
\;}
\]
where the supremum ranges over all admissible finite-energy lifts \(\mu\) on \(K\).  Moreover the maximizers are exactly the finite-energy lifts of the equilibrium measure \(\nu_{R(K)}\) of \(R(K)\):
\[
\mathcal E_R(\mu)=\log\operatorname{cap}(R(K))
\quad\Longleftrightarrow\quad
R_\ast\mu=\nu_{R(K)}.
\]
In particular \(R(K)\) is compact and nonpolar, its equilibrium measure exists, and at least one maximizing lift exists.

The singular sets in the formula have the following explicit interpretation.  The factor \(z-t\) records the ordinary diagonal collision.  The diagonal zeros \(\Psi_R(z,z)=0\) are precisely the finite critical points of \(R\) in the source plane, and their images are the corresponding finite branch values.  The off-diagonal equation \(\Psi_R(z,t)=0\), with \(z\ne t\), is the equal-value collision relation \(R(z)=R(t)\) between two distinct source points.  Equivalently, in local coordinates \(u=z-z_0\), \(v=t-z_0\), an ordinary noncritical diagonal has \(R(z)-R(t)\) equal to \(u-v\) times a holomorphic unit, a critical point of local degree \(e\ge2\) has \(R(z)-R(t)=(u-v)H(u,v)\), where \(H\) has leading homogeneous part a nonzero scalar multiple of \(u^{e-1}+u^{e-2}v+\cdots+v^{e-1}\), and an off-diagonal collision at \((z_0,t_0)\), \(z_0\ne t_0\), is exactly a zero of the remaining factor \(\Psi_R\).  Each of these sets contributes the value \(-\infty\) to the pullback kernel, and an admissible finite-energy lift gives them zero \(\mu\times\mu\)-mass.
\end{theorem}

\begin{lemma}[Local collision skeleton for the rational pullback kernel]\label{lem:rational-local-collision-skeleton}
Let \(R=A/B\) be as in Theorem \ref{thm:finite-degree-rational-image-capacity}, and let \(z_0,t_0\in\CC\) with \(B(z_0)B(t_0)\ne0\).  The singular part of the local pullback kernel near \((z_0,t_0)\) is exhausted by the following three cases.
\begin{enumerate}[label=\textup{(\roman*)}]
\item If \(z_0=t_0\) and \(R'(z_0)\ne0\), then in a local coordinate \(u=z-z_0\), \(v=t-z_0\),
\[
R(z)-R(t)=(u-v)U(u,v)
\]
with \(U(0,0)\ne0\).  Hence the only singular term is the ordinary logarithmic collision \(\log|z-t|\).
\item If \(z_0=t_0\) and \(R\) has local degree \(e\ge2\) at \(z_0\), then
\[
R(z)-R(t)=(u-v)H(u,v)
\]
where \(H\) has leading homogeneous part a nonzero scalar multiple of \(u^{e-1}+u^{e-2}v+\cdots+v^{e-1}\).  Thus \(z_0\) is a finite critical point, \(R(z_0)\) is the corresponding finite branch value, and the diagonal value \(\Psi_R(z_0,z_0)=0\) records exactly the nontrivial ramification factor beyond the ordinary collision.
\item If \(z_0\ne t_0\), then \(R(z_0)=R(t_0)\) if and only if \(\Psi_R(z_0,t_0)=0\).  In this case the singularity is an off-diagonal equal-value collision and is not a ramification statement unless one of the two source points is separately critical.
\end{enumerate}
\end{lemma}

\begin{proof}
Because \(B(z_0)\ne0\), the rational map is holomorphic near \(z_0\).  If \(z_0=t_0\), choose a local coordinate \(u=z-z_0\) and write
\[
R(z_0+u)=R(z_0)+u^e U_0(u)
\]
where \(e\ge1\) is the local degree and \(U_0(0)\ne0\).  For two nearby points \(z=z_0+u\) and \(t=z_0+v\),
\[
R(z)-R(t)=u^eU_0(u)-v^eU_0(v).
\]
This holomorphic function vanishes on the diagonal \(u=v\), hence is divisible by \(u-v\).  When \(e=1\), the quotient has value \(U_0(0)\ne0\), giving (i).  When \(e\ge2\), the quotient \(H(u,v):=(R(z_0+u)-R(z_0+v))/(u-v)\) has initial homogeneous part
\[
U_0(0)\bigl(u^{e-1}+u^{e-2}v+\cdots+v^{e-1}\bigr),
\]
which is nonzero as a polynomial and records the ramification beyond the diagonal factor.  The identity
\[
R'(z)=\frac{A'(z)B(z)-A(z)B'(z)}{B(z)^2}=\frac{\Psi_R(z,z)}{B(z)^2}
\]
shows that the diagonal zeros of \(\Psi_R\) are exactly the finite critical points, and their images under \(R\) are the finite branch values.  Finally, if \(z_0\ne t_0\), then \(z-t\) and \(B(z)B(t)\) are nonzero near \((z_0,t_0)\), so
\[
R(z)-R(t)=\frac{(z-t)\Psi_R(z,t)}{B(z)B(t)}
\]
vanishes at \((z_0,t_0)\) exactly when \(\Psi_R(z_0,t_0)=0\).  This proves the off-diagonal alternative and completes the classification.
\end{proof}

\begin{theorem}[Degree-two Joukowsky capacity and collision skeleton]\label{thm:degree-two-joukowsky-capacity-skeleton}
Let \(r>1\), let
\[
J_r(z)=rz+r^{-1}z^{-1}=\frac{rz^2+r^{-1}}{z}
\]
and let \(K\subset\CC\setminus\{0\}\) be compact and nonpolar.  Then \(J_r(K)\) is compact and nonpolar and
\[
\log\operatorname{cap}(J_r(K))
=\sup_{\mu}\iint_{K\times K}\log\left|(z-t)\left(r-r^{-1}z^{-1}t^{-1}\right)\right|\,
\dd\mu(z)\dd\mu(t),
\]
where the supremum is over all finite-energy lifts for which \((J_r)_\ast\mu\) has finite logarithmic energy.  The complete collision skeleton on \(K\times K\) is
\[
\{(z,t):z=t\}\ \cup\ \{(z,t):zt=r^{-2}\}.
\]
Thus every bad collision is either the ordinary logarithmic diagonal or the single involutive partner \(t=r^{-2}/z\).  In particular the off-diagonal branch graph is disjoint from \(\TT\times\TT\); hence the elliptic-boundary equilibrium problem in Theorem \ref{thm:sharp-joukowsky-equilibrium-stability} has only the ordinary diagonal collision singularity on the source circle.  This degree-two theorem is the operative specialization used in the Joukowsky ellipse, Blaschke-cover, and Mahler-transport theorem chain below; Theorem \ref{thm:finite-degree-rational-image-capacity} supplies the finite-degree extremal principle and the lift classification from which the degree-two collision skeleton is read.
\end{theorem}

\begin{proof}
For \(A(z)=rz^2+r^{-1}\) and \(B(z)=z\), the difference quotient in Theorem \ref{thm:finite-degree-rational-image-capacity} is
\[
\Psi_{J_r}(z,t)=\frac{(rz^2+r^{-1})t-(rt^2+r^{-1})z}{z-t}
=rzt-r^{-1}.
\]
Substitution into the pullback kernel gives
\[
\log|z-t|+
\log|rzt-r^{-1}|-
\log|z|-
\log|t|
=
\log\left|(z-t)\left(r-r^{-1}z^{-1}t^{-1}\right)\right|.
\]
The off-diagonal collision relation is \(rzt-r^{-1}=0\), equivalently \(zt=r^{-2}\).  If \(|z|=|t|=1\), this relation cannot hold for \(r>1\).  The final assertion therefore follows by specializing the general singularity description in Theorem \ref{thm:finite-degree-rational-image-capacity} to the unit circle.
\end{proof}

\begin{proposition}[Operative boundary reduction for the degree-two Joukowsky spine]\label{prop:degree-two-boundary-operativity}
Let \(r>1\).  On the source circle \(\TT\), the degree-two Joukowsky map has no branch-graph collision except the ordinary diagonal:
\[
\{(z,t)\in\TT^2:J_r(z)=J_r(t)\}=\{(z,z):z\in\TT\}.
\]
Consequently every capacity and equilibrium calculation in the sequel for \(E_r=J_r(\TT)\) and for its finite Blaschke covers is obtained from the degree-two kernel
\[
\log|J_r(z)-J_r(t)|=
\log\left|(z-t)\left(r-r^{-1}z^{-1}t^{-1}\right)\right|,
\qquad z,t\in\TT,
\]
where the second factor is continuous and nonzero on \(\TT^2\).  More precisely, if \(B\) is a finite Blaschke product and \(\mu\in\mathcal P(\TT)\), then all covered-boundary energies satisfy
\[
I((J_r\circ B)_\ast\mu)=I(J_{r\ast}(B_\ast\mu))
\]
as lower-truncated logarithmic energies, where \(I(\sigma)=\iint\log|x-y|\,\dd\sigma(x)\dd\sigma(y)\).  Thus the only singularities relevant to the paper's main theorem chain are the ordinary logarithmic collision after the boundary variable has been pushed to \(\TT\), while Theorem \ref{thm:finite-degree-rational-image-capacity} supplies the finite-degree variational theorem before this degree-two boundary reduction.
\end{proposition}

\begin{proof}
By Theorem \ref{thm:degree-two-joukowsky-capacity-skeleton}, the full collision skeleton for \(J_r\) is the union of the diagonal \(z=t\) and the involutive partner relation \(zt=r^{-2}\).  On \(\TT^2\) the second relation is impossible, because \(|zt|=1\) while \(r^{-2}<1\).  Hence \(J_r\) is injective on \(\TT\), and the only boundary collision for \(J_r\) is the ordinary diagonal.

The displayed kernel identity is the specialization of the same theorem to \(K=\TT\).  Its second factor never vanishes on \(\TT^2\), so it contributes a continuous bounded term to the lower-truncated logarithmic energy and introduces no additional singular set.

Finally, if \(B\) is a finite Blaschke product, then \(B(\TT)\subset\TT\) and functoriality of pushforward gives
\[
(J_r\circ B)_\ast\mu=J_{r\ast}(B_\ast\mu).
\]
Applying the lower-truncated logarithmic energy functional to both sides proves the covered-boundary identity.  Therefore the finite-degree rational formula is not an additional general direction in the paper: after it supplies the pullback identity, all subsequent capacity arguments pass through the degree-two Joukowsky boundary reduction proved here.
\end{proof}



\begin{proof}[Proof of Theorem \ref{thm:finite-degree-rational-image-capacity}]
Lemma \ref{lem:nonpolar-rational-image} shows that \(E:=R(K)\) is compact and nonpolar, so the equilibrium variational principle for \(E\) is available by the compact-set equilibrium theorem of \citet[Chap.~5, pp.~105--107]{Ransford1995}, equivalently Frostman's theorem in the form of \citet[Chap.~III, Secs.~15--17]{Tsuji1959}.

Lemma \ref{lem:rational-divided-difference-kernel} shows that the displayed extended kernel is exactly \(L_R(z,t)=\log|R(z)-R(t)|\), with value \(-\infty\) on every diagonal, branch-graph, and off-diagonal collision point.  Lemma \ref{lem:extended-pullback-energy-equals-pushforward-energy} gives
\[
\mathcal E_R(\mu)=\iint_{R(K)\times R(K)}\log|x-y|\,\dd(R_\ast\mu)(x)\dd(R_\ast\mu)(y)
\]
for every \(\mu\in\mathcal P(K)\), as an identity of lower-truncated extended integrals.  Consequently \(R_\ast\mu\) has finite logarithmic energy if and only if the displayed pullback integral is greater than \(-\infty\).

By the compact-set equilibrium theorem recalled at the beginning of the section, the logarithmic capacity of a compact nonpolar set is characterized by
\[
\log\operatorname{cap}(E)=\sup_{\nu\in\mathcal P(E)}
\iint_{E\times E}\log|x-y|\,\dd\nu(x)\dd\nu(y),
\]
with the supremum attained by a unique equilibrium measure.  Applying this to the nonpolar set \(E=R(K)\) shows immediately that
\[
\sup_{\mu\ \mathrm{admissible}}\mathcal E_R(\mu)
\le \log\operatorname{cap}(R(K)),
\]
because every admissible lift pushes forward to a probability measure on \(R(K)\).

Conversely, let \(\nu_E\) be the equilibrium measure of \(E=R(K)\).  Since \(K\) is compact and \(R\) is continuous on \(K\), the closed set
\[
\{(x,z)\in E\times K:R(z)=x\}
\]
has nonempty compact sections over the first coordinate.  Lemma \ref{lem:compact-fiber-borel-selection} gives a Borel map \(s:E\to K\) with \(R(s(x))=x\).  Set \(\mu=s_\ast\nu_E\), a Borel probability measure on \(K\).  Because \(s\) is Borel and \(R\circ s=\operatorname{id}_E\), this selected lift has pushforward exactly equal to the equilibrium measure:
\[
R_\ast\mu=(R\circ s)_\ast\nu_E=\nu_E.
\]
This is the point at which admissibility of the selected lift enters: the equilibrium measure \(\nu_E\) has finite logarithmic energy equal to \(\log\operatorname{cap}(E)>-\infty\), and Lemma \ref{lem:extended-pullback-energy-equals-pushforward-energy} identifies the lower-truncated pullback integral of the selected measure \(\mu\) with exactly that pushforward energy.  Therefore the branch and collision locus has zero \(\mu\times\mu\)-mass and the singular pullback kernel is not being assigned a spurious finite value; \(\mu\) is an admissible finite-energy lift in the sense of the theorem.  Consequently
\[
\mathcal E_R(\mu)=\iint\log|x-y|\,\dd\nu_E(x)\dd\nu_E(y)
=\log\operatorname{cap}(E).
\]
This proves the reverse inequality and the capacity identity.

The same argument gives the announced extremal classification.  If \(\mu\) is admissible and maximizes \(\mathcal E_R\), then the pushforward identity above shows that \(R_\ast\mu\) is a probability measure on \(E\) whose logarithmic energy equals \(\log\operatorname{cap}(E)\).  Uniqueness of the equilibrium measure for a compact nonpolar set gives \(R_\ast\mu=\nu_E\).  Conversely, any admissible \(\mu\) with \(R_\ast\mu=\nu_E\) has \(\mathcal E_R(\mu)=I(\nu_E)=\log\operatorname{cap}(E)\).  The selected lift \(s_\ast\nu_E\) constructed in the preceding paragraph proves that such maximizers exist.

It remains only to justify the description of the singularities.  For \(z\ne t\), the equation \(\Psi_R(z,t)=0\) is equivalent to \(A(z)B(t)-A(t)B(z)=0\), hence to \(R(z)=R(t)\).  On the diagonal, the extended value \(\Psi_R(z,z)=A'(z)B(z)-A(z)B'(z)\) is \(B(z)^2R'(z)\), so its zeros are precisely the finite branch points of \(R\) lying in \(K\).  If \(R_\ast\mu\) has finite logarithmic energy then it has no atoms; consequently the pullback collision relation \(R(z)=R(t)\), including its diagonal and branch parts, has zero \(\mu\times\mu\)-mass.
\end{proof}

\begin{lemma}[Lower-truncated Fourier evaluation on the circle]\label{lem:circle-log-kernel-fourier-truncation}
Let \(\mu\) be a probability measure on \(\TT\) with finite logarithmic energy.  Then
\[
\iint_{\TT\times\TT}\log|e^{i\theta}-e^{i\phi}|\,\dd\mu(\theta)\dd\mu(\phi)
=-
\sum_{k=1}^{\infty}\frac{|\widehat\mu(k)|^2}{k},
\]
where the integral is the lower-truncated logarithmic integral.  If \(q\in\RR\) and \(|q|<1\), then the continuous kernel \(\log|1-qe^{-i(\theta+\phi)}|\) satisfies
\[
\iint\log|1-qe^{-i(\theta+\phi)}|\,\dd\mu(\theta)\dd\mu(\phi)
=-
\sum_{k=1}^{\infty}\frac{q^k\operatorname{Re}(\widehat\mu(k)^2)}{k}.
\]
\end{lemma}

\begin{proof}
For \(0<\rho<1\), the Abel-regularized identity
\[
\log|1-\rho e^{i(\theta-\phi)}|
=-\sum_{k\ge1}\frac{\rho^k\cos k(\theta-\phi)}{k}
\]
is uniformly convergent on compact subannuli and may be integrated term by term against \(\mu\times\mu\).  It gives
\[
\iint\log|1-\rho e^{i(\theta-\phi)}|\,\dd\mu(\theta)\dd\mu(\phi)
=-
\sum_{k\ge1}\frac{\rho^k|\widehat\mu(k)|^2}{k}.
\]
We now pass from the Abel kernels to the lower-truncated boundary kernel without using any pointwise monotonicity in \(\rho\).  Put
\[
\kappa(\alpha)=\log|1-e^{i\alpha}|,
\qquad
\kappa_\rho(\alpha)=\log|1-\rho e^{i\alpha}|,
\]
and let \(\lambda\) be the image of \(\mu\times\mu\) under
\((\theta,\phi)\mapsto\theta-\phi\).  The finite-energy hypothesis is exactly
\(\int\kappa\,\dd\lambda>-\infty\), with the integral understood by lower truncation, because
\(|e^{i\theta}-e^{i\phi}|=|1-e^{i(\theta-\phi)}|\).  The classical Abel convergence theorem for logarithmic potentials on the circle says that the Poisson-Abel regularizations of this kernel converge in energy against every finite-energy measure:
\[
\lim_{\rho\uparrow1}\int\kappa_\rho(\alpha)\,\dd\lambda(\alpha)
=\int\kappa(\alpha)\,\dd\lambda(\alpha),
\]
where the right-hand side is the lower-truncated logarithmic integral; this follows, equivalently, by applying the Poisson integral to the lower truncations
\(\max\{\kappa,-M\}\), passing \(\rho\uparrow1\) for the continuous truncated kernels, and then letting \(M\to\infty\) under the finite-energy tail condition \citep[Chap.~III, Sec.~1]{Tsuji1959}.  Hence the left side of the Abel identity tends to
\[
\iint\log|e^{i\theta}-e^{i\phi}|\,\dd\mu(\theta)\dd\mu(\phi).
\]
On the right, \(|\widehat\mu(k)|^2/k\ge0\), so monotone convergence for the nonnegative partial Abel sums gives
\[
\lim_{\rho\uparrow1}
\sum_{k\ge1}\frac{\rho^k|\widehat\mu(k)|^2}{k}
=
\sum_{k\ge1}\frac{|\widehat\mu(k)|^2}{k}
\]
in \([0,\infty]\).  The finite lower-truncated energy just obtained forces this series to be finite, and the first identity follows.  For real \(|q|<1\), the expansion of \(\log(1-qe^{-i(\theta+\phi)})\) is uniformly and absolutely convergent on \(\TT^2\), and termwise integration gives the second identity because \(\int e^{-ik\theta}\,\dd\mu(\theta)=\widehat\mu(k)\).
\end{proof}

\begin{theorem}[Sharp Joukowsky equilibrium stability]\label{thm:sharp-joukowsky-equilibrium-stability}
Let \(r>1\), let \(J_r(z)=rz+r^{-1}z^{-1}\), and let
\[
E_r=J_r(\TT),\qquad \mu_r=J_{r\ast}\!\left(\frac{\dd\theta}{2\pi}\right).
\]
For a probability measure \(\mu\) on \(\TT\), write
\[
\widehat\mu(k)=\int_{0}^{2\pi}e^{-ik\theta}\,\dd\mu(\theta),\qquad k\in\ZZ,
\]
and let
\[
I(\sigma):=\iint\log|x-y|\,\dd\sigma(x)\dd\sigma(y)
\]
denote logarithmic energy as the lower-truncated extended logarithmic integral.  If \(\mu\) has finite logarithmic energy on \(\TT\), then
\[
\boxed{\;
\log r-I(J_{r\ast}\mu)
=\sum_{k=1}^{\infty}
\frac{|\widehat\mu(k)|^2+r^{-2k}\operatorname{Re}(\widehat\mu(k)^2)}{k}.
\;}
\]
Consequently,
\[
\sum_{k=1}^{\infty}\frac{(1-r^{-2k})|\widehat\mu(k)|^2}{k}
\le \log r-I(J_{r\ast}\mu)
\le
\sum_{k=1}^{\infty}\frac{(1+r^{-2k})|\widehat\mu(k)|^2}{k}.
\]
The unique equilibrium measure of \(E_r\) is \(\mu_r\), and \(\operatorname{cap}(E_r)=r\).  For each \(k\ge1\) the constants \(1+r^{-2k}\) and \(1-r^{-2k}\) are sharp on the \(k\)-th Fourier mode: for \(|\varepsilon|\le1/2\),
\[
\dd\mu^c_{k,\varepsilon}=(1+2\varepsilon\cos k\theta)\frac{\dd\theta}{2\pi},
\qquad
\dd\mu^s_{k,\varepsilon}=(1+2\varepsilon\sin k\theta)\frac{\dd\theta}{2\pi}
\]
satisfy
\[
\log r-I(J_{r\ast}\mu^c_{k,\varepsilon})=\frac{(1+r^{-2k})\varepsilon^2}{k},
\qquad
\log r-I(J_{r\ast}\mu^s_{k,\varepsilon})=\frac{(1-r^{-2k})\varepsilon^2}{k}.
\]
\end{theorem}

\begin{proof}
For \(z=e^{i\theta}\) and \(w=e^{i\phi}\),
\[
J_r(z)-J_r(w)=(z-w)\left(r-r^{-1}z^{-1}w^{-1}\right),
\]
and hence
\[
\log|J_r(e^{i\theta})-J_r(e^{i\phi})|
=\log r+\log|e^{i\theta}-e^{i\phi}|
 +\log|1-r^{-2}e^{-i(\theta+\phi)}|.
\]
The standard Fourier expansions, interpreted in the lower-truncated sense justified by Lemma \ref{lem:circle-log-kernel-fourier-truncation},
\[
\log|e^{i\theta}-e^{i\phi}|
=-\sum_{k=1}^{\infty}\frac{\operatorname{Re}(e^{ik(\theta-\phi)})}{k},
\quad
\log|1-qe^{-i(\theta+\phi)}|
=-\sum_{k=1}^{\infty}\frac{q^k\operatorname{Re}(e^{-ik(\theta+\phi)})}{k}
\]
with \(q=r^{-2}\), give
\[
I(J_{r\ast}\mu)
=\log r-
\sum_{k=1}^{\infty}\frac{|\widehat\mu(k)|^2+r^{-2k}\operatorname{Re}(\widehat\mu(k)^2)}{k}.
\]
This proves the exact diagonalization.

Writing \(\widehat\mu(k)=x_k+i y_k\), the \(k\)-th summand numerator is
\[
|\widehat\mu(k)|^2+r^{-2k}\operatorname{Re}(\widehat\mu(k)^2)
=(1+r^{-2k})x_k^2+(1-r^{-2k})y_k^2,
\]
which proves the two modewise bounds.  Equality in the left or right coefficient for a fixed mode is attained by purely imaginary or purely real \(\widehat\mu(k)\), respectively.  The densities \(\mu^c_{k,\varepsilon}\) and \(\mu^s_{k,\varepsilon}\) are nonnegative probability measures for \(|\varepsilon|\le1/2\), have no nonzero Fourier coefficients except in modes \(\pm k\), and have \(\widehat\mu^c_{k,\varepsilon}(k)=\varepsilon\) and \(\widehat\mu^s_{k,\varepsilon}(k)=-i\varepsilon\).  The displayed extremal identities follow.

Since every term in the deficit is nonnegative and \(1-r^{-2k}>0\), one has \(I(J_{r\ast}\mu)\le\log r\) for finite-energy source measures, with equality only when \(\widehat\mu(k)=0\) for all \(k\ne0\), that is, when \(\mu\) is Haar measure on \(\TT\).

It remains to identify the global equilibrium and the capacity of \(E_r\).  Put \(\zeta=rz\).  Then
\[
J_r(z)=\zeta+\zeta^{-1},
\]
and the exterior of \(E_r\) is the conformal image of \(\{|\zeta|>r\}\) under \(\zeta\mapsto\zeta+\zeta^{-1}\).  Its inverse branch satisfies \(\zeta(w)=w+O(1/w)\) at infinity, so the Green function of \(\CC\setminus E_r\) with pole at infinity is
\[
g_{E_r}(w)=\log|\zeta(w)|-\log r.
\]
The expansion at infinity is \(g_{E_r}(w)=\log|w|-\log r+o(1)\), hence \(\operatorname{cap}(E_r)=r\).  Harmonic measure from infinity is the pushforward of normalized arclength on \(|\zeta|=r\), equivalently \(\mu_r=J_{r\ast}(\dd\theta/(2\pi))\).  By uniqueness of the equilibrium measure, \(\mu_r\) is the unique equilibrium measure of \(E_r\).
\end{proof}

\begin{proposition}[Extended-energy Joukowsky equilibrium bound]\label{prop:extended-energy-joukowsky-equilibrium-bound}
Let \(r>1\) and let \(\mu\) be an arbitrary Borel probability measure on \(\TT\).  With logarithmic energies interpreted by lower truncation,
\[
I(J_{r\ast}\mu)\le \log r.
\]
Equality holds if and only if \(\mu\) is Haar measure on \(\TT\).  Equivalently, for every probability measure \(\eta\) on \(\TT\), finite source energy is not needed for the sharp equilibrium bound
\[
I(J_{r\ast}\eta)\le \log r,
\]
and the unique maximizer is \(\eta=\dd\theta/(2\pi)\).
\end{proposition}

\begin{proof}
Theorem \ref{thm:sharp-joukowsky-equilibrium-stability} gives \(\operatorname{cap}(E_r)=r\) and identifies \(\mu_r=J_{r\ast}(\dd\theta/(2\pi))\) as the unique equilibrium measure of \(E_r\).  The logarithmic-capacity variational principle therefore gives, for every probability measure \(\nu\) on \(E_r\),
\[
I(\nu)\le\log\operatorname{cap}(E_r)=\log r,
\]
with equality only for \(\nu=\mu_r\).  Apply this to \(\nu=J_{r\ast}\mu\).  Since \(J_r:\TT\to E_r\) is a homeomorphism for \(r>1\), the equality condition \(J_{r\ast}\mu=\mu_r\) is equivalent to \(\mu=\dd\theta/(2\pi)\).  The final formulation is the same assertion with \(\eta\) in place of \(\mu\).
\end{proof}


\begin{lemma}[Boundary covering formula for finite Blaschke products]\label{lem:finite-blaschke-boundary-covering-formula}
Let \(B\) be a finite Blaschke product of degree \(d\ge1\).  Then \(B\) has no critical point on \(\TT\), the boundary map \(B:\TT\to\TT\) is an orientation-preserving degree-\(d\) covering, and for every continuous function \(f\) on \(\TT\),
\[
\int_{0}^{2\pi} f(B(e^{i\theta}))\,|B'(e^{i\theta})|\,\dd\theta
=d\int_{0}^{2\pi}f(e^{it})\,\dd t.
\]
Consequently the probability measure
\[
\dd\mu_B(e^{i\theta})=\frac{|B'(e^{i\theta})|}{2\pi d}\,\dd\theta
\]
satisfies \(B_\ast\mu_B=\dd t/(2\pi)\).
\end{lemma}

\begin{proof}
Write
\[
B(z)=\zeta\prod_{j=1}^{d}\frac{z-a_j}{1-\overline{a_j}z},
\qquad |\zeta|=1,
\qquad a_j\in\DD.
\]
The following boundary computation is the standard finite-Blaschke boundary argument and Poisson-kernel derivative formula \citep[see, e.g.,][Sec.~1.2]{Duren1970}.  For \(z=e^{i\theta}\), differentiating a continuous boundary argument \(B(e^{i\theta})=e^{i\beta(\theta)}\) on any local arc gives
\[
\beta'(\theta)=\operatorname{Re}\left(\frac{e^{i\theta}B'(e^{i\theta})}{B(e^{i\theta})}\right)
=
\sum_{j=1}^{d}\frac{1-|a_j|^2}{|e^{i\theta}-a_j|^2}>0.
\]
Thus \(B'(e^{i\theta})\ne0\) on \(\TT\), and \(|B'(e^{i\theta})|=\beta'(\theta)\).  The argument increase over one full turn is \(\beta(2\pi)-\beta(0)=2\pi d\), so \(B:\TT\to\TT\) is an orientation-preserving covering of degree \(d\).  Partition \([0,2\pi]\) into the \(d\) inverse branches on which \(\beta\) maps the branch interval bijectively onto an interval of length \(2\pi\).  On each branch the substitution \(t=\beta(\theta)\) gives one copy of \(\int_0^{2\pi}f(e^{it})\,\dd t\).  Summing the \(d\) branches proves the change-of-variables formula, and division by \(2\pi d\) gives the asserted pushforward identity.
\end{proof}

\begin{theorem}[Finite Blaschke--Joukowsky covers]\label{thm:finite-blaschke-joukowsky-cover}
Let \(B\) be a finite Blaschke product of degree \(d\ge1\), let \(r>1\), and set
\[
R_{B,r}(e^{i\theta})=J_r(B(e^{i\theta})).
\]
For every probability measure \(\mu\) on \(\TT\), with \(\eta:=B_\ast\mu\), one has the extended-energy bound
\[
I((R_{B,r})_\ast\mu)
=I(J_{r\ast}\eta)
\le \log r,
\]
and equality holds if and only if \(\eta\) is Haar measure.  If \(\eta\) has finite logarithmic energy, then the sharper Fourier identity holds:
\[
\log r-I((R_{B,r})_\ast\mu)
=\sum_{k=1}^{\infty}
\frac{|\widehat\eta(k)|^2+r^{-2k}\operatorname{Re}(\widehat\eta(k)^2)}{k}.
\]
Thus the energy deficit and the sharp equilibrium bound factor only through \(\eta=B_\ast\mu\).  The equilibrium lifts are exactly the probability measures \(\mu\) satisfying
\[
B_\ast\mu=\frac{\dd\theta}{2\pi}.
\]
Among them is the canonical absolutely continuous lift
\[
\boxed{\;
\dd\mu_B(e^{i\theta})=\frac{|B'(e^{i\theta})|}{2\pi d}\,\dd\theta.
\;}
\]
\end{theorem}

\begin{proof}
The identity \((R_{B,r})_\ast\mu=J_{r\ast}(B_\ast\mu)=J_{r\ast}\eta\) reduces the all-measure maximal-energy statement directly to Proposition \ref{prop:extended-energy-joukowsky-equilibrium-bound}.  When \(\eta\) has finite logarithmic energy, Theorem \ref{thm:sharp-joukowsky-equilibrium-stability} gives the displayed Fourier deficit formula.  The same proposition shows that equality in the maximal-energy problem occurs exactly when \(\eta\) is Haar measure.  This proves the characterization of all equilibrium lifts without any finite source-energy assumption on \(\mu\).

It remains to verify the canonical density.  Lemma \ref{lem:finite-blaschke-boundary-covering-formula} applies to every continuous test function on \(\TT\) and gives \(B_\ast\mu_B\) equal to Haar measure.  Thus \(\mu_B\) is an equilibrium lift.
\end{proof}

\begin{proposition}[Canonical Blaschke lift source energy]\label{prop:canonical-blaschke-lift-source-energy}
Let
\[
B(z)=\zeta\prod_{j=1}^{d}\frac{z-a_j}{1-\overline{a_j}z},
\qquad |\zeta|=1,
\qquad a_j\in\DD,
\]
and let \(\mu_B\) be the canonical lift from Theorem \ref{thm:finite-blaschke-joukowsky-cover}.  Then its source-circle logarithmic energy is
\[
\boxed{\;
I_{\TT}(\mu_B):=\iint_{\TT\times\TT}\log|z-t|\,\dd\mu_B(z)\dd\mu_B(t)
=\frac1{d^2}\sum_{i,j=1}^{d}\log|1-\overline{a_i}a_j|.
\;}
\]
In particular \(I_{\TT}(\mu_B)\le0\), with equality if and only if \(a_1=\cdots=a_d=0\), equivalently \(B(z)=\zeta z^d\).
\end{proposition}

\begin{proof}
The boundary derivative of a finite Blaschke product is the sum of the Poisson kernels of its zeros:
\[
\frac{|B'(e^{i\theta})|}{d}
=\frac1d\sum_{j=1}^{d}\frac{1-|a_j|^2}{|e^{i\theta}-a_j|^2}.
\]
With the convention \(\widehat\sigma(k)=\int e^{-ik\theta}\,\dd\sigma(\theta)\), the Poisson kernel has Fourier expansion
\[
\frac{1-|a|^2}{|e^{i\theta}-a|^2}=
1+2\operatorname{Re}\sum_{k\ge1}\overline a^{\,k}e^{ik\theta},
\]
so its \(k\)-th coefficient is \(\overline a^{\,k}\).  The sign below is the same as in Lemma \ref{lem:circle-log-kernel-fourier-truncation}: the circle logarithmic kernel has no constant term and contributes \(-|\widehat\sigma(k)|^2/k\) in mode \(k\).
Thus, for \(k\ge1\), the Fourier coefficients of \(\mu_B\) are
\[
\widehat\mu_B(k)=\frac1d\sum_{j=1}^{d}\overline{a_j}^{\ k}.
\]
Using the circle-kernel expansion
\[
\log|e^{i\theta}-e^{i\phi}|
=-\sum_{k=1}^{\infty}\frac{\operatorname{Re}(e^{ik(\theta-\phi)})}{k},
\]
we get
\[
I_{\TT}(\mu_B)=-\sum_{k=1}^{\infty}\frac{|\widehat\mu_B(k)|^2}{k}
=-\frac1{d^2}\sum_{i,j=1}^{d}\sum_{k=1}^{\infty}\frac{(\overline{a_i}a_j)^k}{k}.
\]
Since \(|\overline{a_i}a_j|<1\), the inner series is \(-\log(1-\overline{a_i}a_j)\).  Taking real parts gives
\[
I_{\TT}(\mu_B)=\frac1{d^2}\sum_{i,j=1}^{d}\log|1-\overline{a_i}a_j|.
\]
The Fourier form also gives \(I_{\TT}(\mu_B)=-\sum_{k\ge1}|\widehat\mu_B(k)|^2/k\le0\).  Equality holds if and only if all positive Fourier coefficients vanish, equivalently \(\sum_j\overline{a_j}^{\ k}=0\) for every \(k\ge1\).  Newton's identities then force all elementary symmetric functions of \(a_1,\dots,a_d\) to vanish, so \(a_1=\cdots=a_d=0\).  Conversely, if all zeros are zero, then \(B(z)=\zeta z^d\), \(\mu_B\) is Haar measure, and the energy is \(0\).
\end{proof}
