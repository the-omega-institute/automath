\subsection{Multiscale Laurent Coefficient Interpolation}\label{subsec:multiscale-coefficient-vector-inverse}

\begin{lemma}[Parity-restricted Laurent interpolation]\label{lem:parity-restricted-laurent-interpolation}
Fix \(k\ge0\) and distinct positive numbers \(r_0,
\dots,r_k\).  The evaluation map
\[
\sum_{j=0}^{k}\alpha_j r^{k-2j}
\longmapsto
\left(\sum_{j=0}^{k}\alpha_j r_i^{k-2j}\right)_{i=0}^{k}
\]
from the \((k+1)\)-dimensional space spanned by \(r^k,r^{k-2},\dots,r^{-k}\) to \(\CC^{k+1}\) is an isomorphism.  Hence any coefficient family supported on this parity class is determined by its values at \(k+1\) distinct scales; in particular \(n+1\) distinct scales determine it for every \(0\le k\le n\).
\end{lemma}

\begin{proof}
Multiplication by \(r^k\) sends the displayed Laurent polynomial to
\[
\sum_{j=0}^{k}\alpha_j r^{2k-2j}
=\sum_{j=0}^{k}\alpha_j (r^2)^{k-j},
\]
a polynomial of degree at most \(k\) in \(u=r^2\).  The evaluation matrix at \(u_i=r_i^2\) is the Vandermonde matrix \((u_i^{k-j})_{0\le i,j\le k}\), whose determinant is, up to sign, \(\prod_{0\le i<j\le k}(u_j-u_i)\).  This determinant is nonzero because the scales \(r_i\) are distinct and positive.  Therefore the evaluations determine the coefficients \(\alpha_j\) uniquely.  If \(n+1\) scales are available and \(k\le n\), restricting to any \(k+1\) of them already gives the stated interpolation.
\end{proof}

\begin{lemma}[Full and monic multiscale normalization]\label{lem:full-monic-multiscale-normalization}
Let \(P(z)=\prod_{j=1}^{n}(z-z_j)\) be monic with \(z_1\cdots z_n\ne0\), let
\[
\begin{gathered}
Q_r(w)=\sum_{k=0}^{n}A_k(r)w^{n-k},
\qquad
a_0=(-1)^nz_1\cdots z_n,\\
\widetilde Q_r(w):=a_0^{-1}Q_r(w)
=\sum_{k=0}^{n}\widetilde A_k(r)w^{n-k}.
\end{gathered}
\]
and let \(r_0,
\dots,r_n>0\) be distinct.  The source polynomial is reconstructed by the following two algebraic normalization procedures.
\begin{enumerate}[label=\textup{(\roman*)}]
\item \emph{Full-normalized data.}  If the coefficient lists of \(Q_{r_i}\) are known, then the common leading coefficient gives \(a_0\).  For each \(k\), interpolate \(A_k(r)\) in the parity-supported Laurent span \(r^k,r^{k-2},
\dots,r^{-k}\); then
\[
e_k(z_1,
\dots,z_n)=(-1)^k a_0^{-1}[r^k]A_k(r).
\]
\item \emph{Monic-normalized data.}  If only nonzero scalar multiples of the coefficient lists are known, each scale may carry its own unknown nonzero scalar.  Because the underlying exact object is a transported degree-\(n\) zero divisor, the leading entry of every complete list is nonzero: before rescaling it is \(a_0\ne0\), and after rescaling it is multiplied by a nonzero scalar.  Divide each list by its own leading entry, obtaining the monic lists of \(\widetilde Q_{r_i}\).  For each \(k\), interpolate \(\widetilde A_k(r)\) in the same parity-supported Laurent span; then
\[
e_k(z_1,
\dots,z_n)=(-1)^k[r^k]\widetilde A_k(r),
\qquad
a_0=(-1)^ne_n(z_1,
\dots,z_n)=[r^n]\widetilde A_n(r).
\]
\end{enumerate}
In both cases the recovered elementary symmetric functions determine the original monic source polynomial \(P\).
\end{lemma}

\begin{proof}
The leading coefficient of \(Q_r\) is the nonzero scalar \(a_0\), independent of \(r\), so full-normalized data immediately identify it.  If a known list is multiplied by an unknown nonzero scalar, its leading entry is multiplied by the same scalar; division by that leading entry therefore gives the coefficient list of the monic transport \(\widetilde Q_r\).

For fixed \(k\), the coefficient of \(w^{n-k}\) in \(Q_r\) is
\[
A_k(r)=(-1)^k a_0\sum_{|I|=k}\prod_{j\in I}(rz_j+r^{-1}z_j^{-1}),
\]
and hence is supported on the exponents \(k,k-2,
\dots,-k\).  Lemma \ref{lem:parity-restricted-laurent-interpolation} reconstructs this Laurent polynomial from any \(k+1\) of the available scale values, and therefore from the displayed \(n+1\) scales.  Its highest coefficient is \((-1)^ka_0e_k(z_1,
\dots,z_n)\), proving the full-normalized formula.  Dividing the same identity by \(a_0\) gives
\[
\widetilde A_k(r)=(-1)^k\sum_{|I|=k}\prod_{j\in I}(rz_j+r^{-1}z_j^{-1}),
\]
whose highest coefficient is \((-1)^ke_k(z_1,
\dots,z_n)\), proving the monic-normalized formula.  For \(k=n\), the highest coefficient of \(\widetilde A_n\) is \((-1)^ne_n=a_0\), so the leading coefficient of the full transport is also recovered from monic-normalized data.  Finally, the monic polynomial \(P(z)=z^n-e_1z^{n-1}+\cdots+(-1)^ne_n\) is determined by the recovered elementary symmetric functions.
\end{proof}

The data model for the following theorem is the one fixed in
\nameref{par:scope-of-inverse-theorems}; the normalization step is Lemma
\ref{lem:full-monic-multiscale-normalization}.

\begin{theorem}[Exact complete-coefficient-vector inverse theorem]\label{thm:algebraic-mahler-transport-inverse}
Let \(n\ge1\), and let \(P(z)=\prod_{j=1}^{n}(z-z_j)\) be monic with \(z_1\cdots z_n\ne0\).  For each \(r>1\), let \(Q_r\) be the Joukowsky transport polynomial normalized as in Table~\ref{tab:normalization-ledger}, so
\[
Q_r(w)=\sum_{k=0}^{n}A_k(r)w^{n-k},
\qquad
A_0(r)=a_0=(-1)^n z_1\cdots z_n\ne0.
\]
Assume the exact data are complete coefficient-vector data in one of the
two models of Definition \ref{def:transport-coefficient-observations}.

The interpolation-budget assertion is the Laurent-coefficient sharpness
statement of Remark~\ref{rem:multiscale-sharpness-scope}.  Then the
following inverse statements hold.
\begin{enumerate}[label=\textup{(\roman*)}]
  \item \leavevmode\par\noindent\textbf{Sharpness for Laurent coefficient interpolation.}

  If \(r_0,\dots,r_n>1\) are distinct and exact complete coefficient-vector data are observed at these scales, then \(P\) is recovered uniquely by the full or monic algebraic normalization procedure of Lemma \ref{lem:full-monic-multiscale-normalization}.  Equivalently, the coefficient \(A_k(r)\) of \(w^{n-k}\) lies in the parity-supported Laurent span \(r^k,r^{k-2},\dots,r^{-k}\), and its top Laurent coefficient recovers \((-1)^ka_0e_k(z_1,\dots,z_n)\) in the full normalization and \((-1)^ke_k(z_1,\dots,z_n)\) in the monic normalization.  The uniform budget \(n+1\) is sufficient for exact source recovery by this coefficient-interpolation method, and its sharpness is only the interpolation-budget sharpness of Remark~\ref{rem:multiscale-sharpness-scope}.

  \item \textbf{One-scale resonance fibers.}  Fix one scale \(r>1\), and let \(\tau_r(z)=r^{-2}/z\).  The full one-scale fiber, including repeated roots, overlapping reflected pairs, and the fixed critical points \(z=\pm r^{-1}\), is the \(\tau_r\)-orbit matching set described in Lemma \ref{lem:one-scale-orbit-fiber-matching}: a competitor must preserve the total multiplicity on every \(\tau_r\)-orbit and must preserve the product of all source roots.  Equivalently, on each two-point orbit \(O=\{x,\tau_r(x)\}\) with total multiplicity \(N_O\), a competitor chooses an integer \(c_O\in\{0,\dots,N_O\}\) as the multiplicity at \(x\), while a one-point fixed orbit \(x=\tau_r(x)\) has no choice.  These choices are allowed exactly when
  \[
  \prod_{O=\{x,\tau_r(x)\}}
  \left(\frac{x}{\tau_r(x)}\right)^{c_O-m_x}=1,
  \]
  the product being over the two-point orbits and with the original divisor corresponding to \(c_O=m_x\) for every \(O\).  In the separated simple case where the \(2n\) points \(z_j\) and \(r^{-2}/z_j\) are pairwise distinct, this reduces to choosing one element from each indexed pair \(\{z_j,r^{-2}/z_j\}\).  If \(S\subset\{1,\dots,n\}\) is the set of indices for which the reflected point \(r^{-2}/z_j\) is chosen, then the choice is in the fiber exactly when
  \[
  \prod_{j\in S}z_j^2=r^{-2|S|}.
  \]

  \item \textbf{Unit-circle zero and separation injectivity.}  If \(|z_j|=1\) for all \(j\), then the one-scale map is injective on the monic unit-circle zero class, including repeated roots.  More generally, one-scale injectivity holds for every source divisor \(D\) for which the orbitwise integer choice \((c_O)_O=(m_x)_O\) is the only choice satisfying the preceding resonance product.  In the separated simple case this criterion is exactly the absence of an alternative subset choice \(S\ne\varnothing\) satisfying the displayed resonance equation; the empty subset is the original divisor and is not a competitor.
\end{enumerate}
\end{theorem}

\begin{proof}
For (i), the data restriction in Definition \ref{def:transport-coefficient-observations} is used at the first step: at each scale the proof knows the entire ordered coefficient vector of the degree-\(n\) polynomial \(Q_{r_i}\), or a projective multiple of that entire vector with its nonzero leading coordinate visible.  Thus coefficient values may be interpolated coefficient-by-coefficient.  Write
\[
\widetilde Q_r(w)=\prod_{j=1}^{n}(w-rz_j-r^{-1}z_j^{-1}).
\]
for the monic transport.  If the exact data are scalar multiples of the coefficient lists, we first divide each list by its leading coefficient; this gives \(\widetilde Q_{r_i}\).  The constant coefficient of \(\widetilde Q_r\), as a parity-restricted Laurent polynomial in \(r\), has bottom coefficient \((-1)^n(z_1\cdots z_n)^{-1}\), so interpolation from the distinct scales recovers this nonzero number and hence recovers
\[
a_0=(-1)^n z_1\cdots z_n.
\]
For fully normalized transport data this same scalar is read off directly as the leading coefficient of any known polynomial \(Q_{r_i}\).  The actual transport polynomial is multiplied by
\[
a_0=(-1)^n z_1\cdots z_n,
\]
and the coefficient \(A_k(r)\) of \(w^{n-k}\) in the actual polynomial
\[
Q_r(w)=a_0\prod_{j=1}^{n}(w-rz_j-r^{-1}z_j^{-1})
\]
is
\[
(-1)^k a_0\sum_{|I|=k}\prod_{j\in I}(rz_j+r^{-1}z_j^{-1}).
\]
It is a Laurent polynomial in \(r\) with exponents \(k,k-2,\dots,-k\).  Lemma \ref{lem:parity-restricted-laurent-interpolation} applies to this parity-restricted family: after multiplication by \(r^k\), its coefficient values form a degree-at-most-\(k\) polynomial in \(r^2\), and the Vandermonde matrix on any \(k+1\) of the available \(n+1\) distinct scales has full rank.  Hence the scale values determine this Laurent polynomial uniquely, for every \(k\le n\).  Its highest \(r\)-power coefficient is \((-1)^k a_0 e_k(z_1,\dots,z_n)\), so division by the already known nonzero coefficient \(a_0\) recovers \(e_k(z_1,
\dots,z_n)\).  In the monic-normalized observation convention the same conclusion follows from the highest coefficient of \(\widetilde A_k(r)=a_0^{-1}A_k(r)\), namely \((-1)^ke_k(z_1,\dots,z_n)\); for \(k=n\) this highest coefficient is \((-1)^ne_n(z_1,\dots,z_n)=a_0\), so the full transport scalar is recovered after the monic interpolation.  For \(k=n\), the full-normalized constant term has highest coefficient \((-1)^n a_0(z_1\cdots z_n)=(z_1\cdots z_n)^2\), and its lowest coefficient is \((-1)^n a_0(z_1\cdots z_n)^{-1}=1\).  Dividing by \(a_0\) gives the displayed top and bottom coefficients for the monic transport.  Thus all elementary symmetric functions of \(z_1,
\dots,z_n\) are recovered, and these are precisely the coefficients of the monic polynomial \(P\).

The sufficiency of the uniform scale count is Lemma \ref{lem:parity-restricted-laurent-interpolation} applied coefficient by coefficient.  The largest coefficient level has parity-restricted Laurent support
\[
r^n,r^{n-2},\dots,r^{-n},
\]
a vector space of dimension \(n+1\).  Values at \(n+1\) distinct positive scales give a nonsingular Vandermonde system in \(r^2\), while values at only \(n\) prescribed scales leave a nonzero kernel on this unrestricted coefficient space.  Hence \(n+1\) is the exact uniform budget for the unrestricted Laurent interpolation used here.  This proves the advertised sharpness for Laurent coefficient interpolation itself; it does not assert that every algebraically constrained subfamily of source polynomials realizes every vector in the interpolation kernel.

For (ii), apply Lemma \ref{lem:one-scale-orbit-fiber-matching}.  It gives the necessary and sufficient orbit-multiplicity matching condition for arbitrary repeated and overlapping reflected roots.  On a two-point orbit \(O=\{x,\tau_r(x)\}\), condition \textup{(a)} says only that the total multiplicity \(N_O\) is fixed, so every competitor is obtained by choosing an integer \(c_O\in\{0,\dots,N_O\}\) at \(x\) and \(N_O-c_O\) at \(\tau_r(x)\).  The quotient of the new root product by the old one on this orbit is \((x/\tau_r(x))^{c_O-m_x}\).  Multiplying over the two-point orbits gives the displayed orbitwise resonance product.  If \(x=\tau_r(x)=\pm r^{-1}\), then the orbit has one point and condition \textup{(a)} fixes its multiplicity, so fixed critical orbits contribute no choice and no resonance factor.  The final separated-case clause of Lemma \ref{lem:one-scale-orbit-fiber-matching} is the special case \(N_O=1\) for every indexed pair, where choosing the reflected point for precisely the indices in \(S\) gives the stated subset equation.

For (iii), assume first that all source roots lie on \(\TT\).  Let \(\widetilde D\) satisfy the two orbit-matching conditions of Lemma \ref{lem:one-scale-orbit-fiber-matching}.  Moving one unit of multiplicity from a unit-circle point \(x\) to its reflected point \(\tau_r(x)=r^{-2}/x\) changes the modulus of the source-root product by the factor \(r^{-2}\); moving in the reverse direction changes it by \(r^2\).  Since the original divisor has no mass on the inner shell \(|z|=r^{-2}\), every nontrivial orbit redistribution away from the original unit-circle divisor decreases the product modulus by a positive power of \(r^2\).  It therefore cannot satisfy the product-preservation condition.  Hence \(\widetilde D=D\), proving unit-circle injectivity even with repeated roots.  The stated general criterion is precisely the uniqueness of the orbitwise integer choice satisfying the resonance product in (ii).  In the separated simple case the original choice is \(S=\varnothing\); an actual competitor must have \(S\ne\varnothing\), so injectivity is equivalent to the nonexistence of such a nonempty resonant subset.  Repeated roots and overlapping reflected pairs are not indexed by subsets: they are governed by the orbit totals \(N_O\), while fixed orbits remain forced.
\end{proof}

\begin{remark}[Sharpness for Laurent coefficient interpolation]\label{rem:multiscale-sharpness-scope}
For the coefficient \(A_k(r)\) of \(w^{n-k}\), Lemma \ref{lem:parity-restricted-laurent-interpolation} shows that any \(k+1\) distinct positive scales determine the parity-supported Laurent polynomial in
\[
\operatorname{span}\{r^k,r^{k-2},\dots,r^{-k}\}.
\]
This number is sharp for that unrestricted vector space, because evaluation at only \(k\) scales has a nontrivial kernel.  The theorem therefore uses the uniform budget \(n+1\) to cover all coefficient levels \(0\le k\le n\) at once.  This sharpness claim is only about unrestricted parity-supported Laurent interpolation.  It is not a lower-bound theorem for every algebraically constrained subfamily of source polynomials.
\end{remark}

\begin{theorem}[Exact one-scale unit-circle zero inverse theorem from an exact complete coefficient vector]\label{thm:leyang-single-scale-inverse}
Fix an exact real number \(r>1\).  Let \(\mathcal L\) be the class of all monic polynomials \(P\in\CC[z]\) of positive degree with every zero on \(\TT\), counted with multiplicity.  For \(P\in\mathcal L\), let \(Q_r(P)\), known as an exact complete coefficient vector, be its transport polynomial from Table~\ref{tab:normalization-ledger}.  Then the one-scale transport map
\[
\mathcal T_r:\ P\longmapsto Q_r(P)
\]
is injective on \(\mathcal L\), and the original source is recovered from the exact pair \((r,Q_r(P))\), where \(Q_r(P)\) is an exact complete coefficient vector, by the shell selection of Proposition \ref{prop:exact-shell-recovery}.  The unit-circle hypothesis \(P\in\mathcal L\) is essential to this one-scale collapse; outside it, the complete one-scale fiber is the finite orbit-redistribution fiber described in Theorem \ref{thm:algebraic-mahler-transport-inverse}\textup{(ii)}.
\end{theorem}

\begin{proof}
This is the unit-circle zero case of Theorem \ref{thm:algebraic-mahler-transport-inverse}(iii), with the constructive shell recovery supplied by Proposition \ref{prop:exact-shell-recovery}.
\end{proof}

\begin{corollary}[No-resonance unit-circle fiber collapse]\label{cor:no-resonance-leyang-fiber-collapse}
Fix \(r>1\), and let \(P\in\mathcal L\) have degree \(n\) and zero divisor \(D=\sum_xm_x[x]\).  Among all monic degree-\(n\) polynomials with nonzero constant term, the complete one-scale fiber over \(Q_r(P)\) consists of the orbitwise redistributions described in Theorem \ref{thm:algebraic-mahler-transport-inverse}\textup{(ii)}.  Its intersection with the unit-circle class \(\mathcal L\) is the singleton \(\{P\}\).  In the separated simple-root case this says that every nonempty reflected-choice subset \(S\) fails the resonance equation
\[
\prod_{j\in S}z_j^2=r^{-2|S|}
\]
for the elementary reason that the two sides have moduli \(1\) and \(r^{-2|S|}<1\), respectively.

Consequently the annular shell selection below is not an additional hypothesis: it is the geometric form of the same fiber collapse after the two shells \(|z|=1\) and \(|z|=r^{-2}\) have been separated.
\end{corollary}

\begin{proof}
The first sentence is exactly the fiber classification in Theorem \ref{thm:algebraic-mahler-transport-inverse}\textup{(ii)}.  The same theorem, part \textup{(iii)}, proves that if the original divisor lies on \(\TT\), then no nontrivial redistribution inside the reflected orbits can both preserve the transport polynomial and keep the source-root product required by the normalization; hence the only unit-circle divisor in the fiber is \(D\) itself.  In the separated simple-root case the reflected choices are indexed by subsets \(S\).  If \(S\ne\varnothing\), then \(|\prod_{j\in S}z_j^2|=1\), whereas \(|r^{-2|S|}|=r^{-2|S|}<1\), so the resonance equation cannot hold.  The shell-selection statement is Corollary \ref{cor:leyang-annular-shell-projection}: any radius \(\rho\) with \(r^{-2}<\rho<1\) separates the forced outer unit-circle roots from their reflected inner partners, so the unique outer monic factor is precisely \(P\).
\end{proof}

\begin{corollary}[Annular shell projection for exact unit-circle data]\label{cor:leyang-annular-shell-projection}
Under the hypotheses of Theorem \ref{thm:leyang-single-scale-inverse}, let \(\rho\) be any real number satisfying
\[
r^{-2}<\rho<1.
\]
Then \(P\) is equivalently recovered as the unique monic degree-\(n\) factor of \(S_{r,Q_r}\) whose zeros lie in the exterior annulus \(\{|z|>\rho\}\).  The complementary monic factor has all its zeros in \(\{|z|<\rho\}\).
\end{corollary}

\begin{proof}
The zeros of \(S_{r,Q_r}\) are the disjoint union
\[
\{z_1,
\dots,z_n\}\cup\{r^{-2}z_1^{-1},
\dots,r^{-2}z_n^{-1}\}.
\]
The first set lies on \(|z|=1\), and the second lies on \(|z|=r^{-2}\).  The chosen \(\rho\) separates the two shells, so the exterior factor is exactly \(P\).
\end{proof}

\subsection{Local Contour-Selected Annular Divisor Stability after Exact Model Identification}\label{subsec:annular-rouche-stability}

\begin{lemma}[Outer Rouch\'e count prevents degree loss]\label{lem:outer-rouche-prevents-degree-loss}
Let \(S\) be a polynomial of degree \(2n\), let \(R>0\), and assume that every zero of \(S\) lies in \(|z|<R\).  Let
\[
\widetilde Q(w)=\sum_{\ell=0}^{n}\widetilde b_\ell w^\ell
\]
with coefficients allowed to satisfy \(\widetilde b_n=0\), and put
\[
\widetilde S(z):=z^n\widetilde Q(J_r(z)),
\qquad J_r(z)=rz+r^{-1}z^{-1}.
\]
Then \(\widetilde S\) is a polynomial with expansion
\[
\widetilde S(z)=
\sum_{\ell=0}^{n}\widetilde b_\ell
\sum_{j=0}^{\ell}\binom{\ell}{j}r^{2j-\ell}z^{n-\ell+2j}.
\]
Its trailing coefficient is \(r^{-n}\widetilde b_n\), and its leading coefficient is \(r^n\widetilde b_n\).  If
\[
|\widetilde S(z)-S(z)|<|S(z)|\qquad(|z|=R),
\]
then \(\widetilde b_n\ne0\), \(\deg\widetilde S=2n\), and \(\widetilde S\) has exactly \(2n\) finite zeros in \(|z|<R\), counted with multiplicity.  Thus no part of the Rouch\'e count is lost through a vanished leading coefficient or a zero at infinity.
\end{lemma}

\begin{proof}
For \(0\le\ell\le n\), the binomial expansion gives
\[
z^nJ_r(z)^\ell
=z^n\sum_{j=0}^{\ell}\binom{\ell}{j}(rz)^j(r^{-1}z^{-1})^{\ell-j}
=\sum_{j=0}^{\ell}\binom{\ell}{j}r^{2j-\ell}z^{n-\ell+2j}.
\]
This proves the displayed expansion and shows that \(\widetilde S\) is a polynomial of degree at most \(2n\).  The only term capable of contributing to degree \(0\) is \(\ell=n,j=0\), so the trailing coefficient is \(r^{-n}\widetilde b_n\).  The only term capable of contributing to degree \(2n\) is \(\ell=n,j=n\), so the leading coefficient is \(r^n\widetilde b_n\).

Rouch\'e's theorem on \(|z|=R\) gives the same number of zeros for \(\widetilde S\) and \(S\) in the disk \(|z|<R\).  Since all \(2n\) zeros of the degree-\(2n\) polynomial \(S\) lie in this disk, \(\widetilde S\) has \(2n\) finite zeros there, counted with multiplicity.  A nonzero polynomial of degree at most \(2n\) with \(2n\) finite zeros must have degree at least \(2n\), hence exactly \(2n\).  The leading coefficient formula then forces \(\widetilde b_n\ne0\), and the trailing coefficient is nonzero as well.  This rules out both the possible degree loss from \(\deg\widetilde Q<n\) and any missing zero at infinity.
\end{proof}

\begin{lemma}[Reciprocal shell scalar for monic unit-circle sources]\label{lem:unit-circle-reciprocal-shell-scalar}
Let
\[
P(z)=\prod_{j=1}^{n}(z-z_j)
\]
be monic, assume \(|z_j|=1\) for all \(j\), let \(r>0\), and put
\[
a_0=P(0)=(-1)^n\prod_{j=1}^{n}z_j .
\]
Then, for \(z\ne0\),
\[
z^nP(r^{-2}z^{-1})
=a_0\prod_{j=1}^{n}(z-r^{-2}z_j^{-1}).
\]
Consequently \(|a_0|=1\), and the modulus of this reciprocal-shell
factor is
\[
\left|z^nP(r^{-2}z^{-1})\right|
=\prod_{j=1}^{n}|z-r^{-2}z_j^{-1}|.
\]
Moreover, for every \(\rho>0\) and every \(z\) with \(|z|=\rho\),
\[
\prod_{j=1}^{n}|z-z_j|
\prod_{j=1}^{n}|z-r^{-2}z_j^{-1}|
\ge |1-\rho|^n|\rho-r^{-2}|^n .
\]
Thus the reciprocal-shell scalar \(a_0\) never changes any
absolute-value shell estimate for monic unit-circle sources.
\end{lemma}

\begin{proof}
For each root,
\[
r^{-2}-zz_j=-z_j(z-r^{-2}z_j^{-1}).
\]
Multiplying these identities gives
\[
z^nP(r^{-2}z^{-1})
=\prod_{j=1}^{n}(r^{-2}-zz_j)
=(-1)^n\left(\prod_{j=1}^{n}z_j\right)
\prod_{j=1}^{n}(z-r^{-2}z_j^{-1})
=a_0\prod_{j=1}^{n}(z-r^{-2}z_j^{-1}).
\]
Since the roots lie on \(\TT\), the displayed formula for \(a_0\) gives
\(|a_0|=1\), and taking absolute values proves the final identity.
If \(|z|=\rho\), then
\[
|z-z_j|\ge \bigl||z|-|z_j|\bigr|=|1-\rho|
\]
and
\[
|z-r^{-2}z_j^{-1}|
\ge \bigl||z|-r^{-2}|z_j|^{-1}\bigr|
=|\rho-r^{-2}|.
\]
Multiplying these inequalities over \(j\) proves the displayed shell
bound.  Since \(|a_0|=1\), inserting the reciprocal-shell scalar in
\(S_r\) leaves these absolute-value estimates unchanged.
\end{proof}

The next theorem is the local Rouch\'e selector allowed by
\nameref{par:scope-of-inverse-theorems}: exact recognition first supplies
the model pair \((r,P)\), and only then do fixed separating contours and
\(\Delta(r,P;\rho_-,\rho_+)\) give a perturbative annular-divisor
statement.  The analytic input is Rouch\'e's theorem in its usual
zero-counting form for a contour and its interior
\citep[pp.~152--153]{Ahlfors1979}.  The unperturbed polynomial
\(S_r(z)=z^nQ_r(J_r(z))\) has one zero shell on \(|z|=r^{-2}\) and one on \(|z|=1\); the reciprocal-shell scalar is the \(a_0\) of Lemma \ref{lem:unit-circle-reciprocal-shell-scalar}, whose modulus is one.  The inner contour \(\Gamma_-\) surrounds the reflected shell, the middle contour \(\Gamma_+\) separates it from the unit-circle shell, and the outer contour \(\Gamma_0\) encloses both shells while preventing degree loss at infinity.  A perturbation whose pullback is smaller than the displayed Rouch\'e envelope on these three contours therefore preserves only the zero count of the annular divisor in \(\rho_+<|z|<\rho_+^{-1}\).  This divisor is the contour-selected annular factor associated with the perturbed transported polynomial and is not intrinsic before \((r,P,\rho_-,\rho_+)\) are fixed.  Whether that selected factor is an exact unit-circle source realization is a separate algebraic question, tested by the unit-circle and reciprocal-shell criteria below and in Theorem \ref{thm:intrinsic-leyang-shell-recognition}.

\begin{theorem}[Local contour-selected annular divisor stability after exact model identification]\label{thm:quantitative-annular-rouche-stability}
Let \(P\in\CC[z]\) be a monic polynomial of degree \(n\ge1\), all of whose zeros lie on \(\TT\), let \(r>1\), let \(Q_r\) be its transport polynomial from Table~\ref{tab:normalization-ledger}, and define
\[
S_r(z):=S_{r,Q_r}(z)=z^nQ_r(J_r(z)).
\]
Choose radii
\[
r^{-2}<\rho_-<\rho_+<1
\]
and set
\[
\Gamma_-:=\{|z|=\rho_-\},\qquad
\Gamma_+:=\{|z|=\rho_+\},\qquad
\Gamma_0:=\{|z|=\rho_+^{-1}\},
\]
so that the bounded annulus \(\rho_+<|z|<\rho_+^{-1}\) contains precisely the \(n\) unit-circle zeros of \(S_r\), while the disk \(|z|<\rho_-\) contains precisely the \(n\) inner zeros.  Define the Rouch\'e envelope
\[
\Delta(r,P;\rho_-,\rho_+):=\frac12\min\left\{
\inf_{z\in\Gamma_-}|S_r(z)|,
\inf_{z\in\Gamma_+}|S_r(z)|,
\inf_{z\in\Gamma_0}|S_r(z)|
\right\}>0.
\]
Equivalently, writing \(P(z)=\prod_{j=1}^{n}(z-z_j)\), one has the
exact factorization
\[
S_r(z)
=a_0r^n\prod_{j=1}^{n}(z-z_j)
\prod_{j=1}^{n}(z-r^{-2}z_j^{-1})
=r^nP(z)\,z^nP(r^{-2}z^{-1}),
\]
where
\[
a_0=P(0)=(-1)^n\prod_{j=1}^{n}z_j .
\]
Thus the scalar \(a_0\) is not being discarded in the exact identity; it
is exactly the reciprocal-shell scalar in \(z^nP(r^{-2}z^{-1})\).
Since \(|a_0|=1\) for unit-circle roots, taking absolute values gives
\[
|S_r(z)|=r^n\prod_{j=1}^{n}|z-z_j|
\prod_{j=1}^{n}|z-r^{-2}z_j^{-1}|,
\qquad |z_j|=1,
\]
which is the absolute-value equality used for the Rouch\'e lower bounds below.  The
displayed algebraic factorization above still includes the scalar
\(a_0\); only the absolute-value estimate removes it.  Lemma
\ref{lem:unit-circle-reciprocal-shell-scalar} gives the
equivalent reciprocal-shell modulus identity
\[
\left|z^nP(r^{-2}z^{-1})\right|
=\prod_{j=1}^{n}|z-r^{-2}z_j^{-1}|
\]
for the last factor.  Hence the known two-shell geometry gives the
explicit product lower bound
\[
\begin{aligned}
\Delta(r,P;\rho_-,\rho_+)
&\ge
\frac{r^n}{2}\min\Bigl\{
(1-\rho_-)^n(\rho_- - r^{-2})^n,\\
&\hspace{5.7em}
(1-\rho_+)^n(\rho_+ - r^{-2})^n,\,
(\rho_+^{-1}-1)^n(\rho_+^{-1}-r^{-2})^n
\Bigr\}.
\end{aligned}
\]
The geometry and degree preservation are fixed before perturbing: the contours \(\Gamma_-\), \(\Gamma_+\), and \(\Gamma_0\) are pairwise disjoint circles, \(\Gamma_-\) encloses exactly the reflected shell, \(\Gamma_+\) separates that shell from the unit-circle shell, and \(\Gamma_0\) encloses both shells.  The outer-contour count is used to prevent loss of degree at infinity.  If \(\widetilde Q\in\CC[w]\) has \(\deg \widetilde Q\le n\) and
\[
\sup_{z\in\Gamma_-\cup\Gamma_+\cup\Gamma_0}
\left|z^n\widetilde Q(J_r(z))-S_r(z)\right|<\Delta(r,P;\rho_-,\rho_+),
\]
then \(\widetilde S(z):=z^n\widetilde Q(J_r(z))\) is in fact a degree-\(2n\) polynomial, even if the input polynomial was initially allowed to have \(\deg\widetilde Q<n\), and it has exactly \(n\) zeros, counted with multiplicity, in the annulus
\[
\rho_+<|z|<\rho_+^{-1}
\]
and no zeros on any of \(\Gamma_-\), \(\Gamma_+\), or \(\Gamma_0\).  The monic degree-\(n\) polynomial formed from the zeros in \(\rho_+<|z|<\rho_+^{-1}\) is denoted
\[
P_{\mathrm{ann}}^{\rho_-,\rho_+}(\widetilde Q;r,P).
\]
This is a Rouch\'e zero-count selection statement: it identifies the unique monic factor supported on the selected annular divisor after the model \((r,P)\) and contours \((\rho_-,\rho_+)\) have been fixed.  It does not assert that the zeros of this factor lie on \(\TT\), nor that \(\widetilde Q\) is of the form \(Q_r(\widetilde P)\).  For \(\widetilde Q=Q_r\) this factor is exactly \(P\).

The contour-selected annular factor is quantitatively stable in the following local sense.  Write the zero divisor of \(P\) as \(\sum_{a\in Z(P)}m_a[a]\), where \(Z(P)\) is the finite set of distinct zeros.  Let \(U_a\) \((a\in Z(P))\) be pairwise disjoint Jordan domains whose closures lie in \(\rho_+<|z|<\rho_+^{-1}\), with \(a\in U_a\) and with no zero of \(P\) other than \(a\) in \(U_a\).  Put
\[
\Delta_{\mathcal U}(r,P)
:=\frac12\min_{a\in Z(P)}\inf_{z\in\partial U_a}|S_r(z)|>0.
\]
If, in addition,
\[
\sup_{z\in\bigcup_{a\in Z(P)}\partial U_a}
\left|z^n\widetilde Q(J_r(z))-S_r(z)\right|<\Delta_{\mathcal U}(r,P),
\]
then \(P_{\mathrm{ann}}^{\rho_-,\rho_+}(\widetilde Q;r,P)(z)=\prod_{j=1}^{n}(z-\widetilde z_j)\) has exactly \(m_a\) zeros in each \(U_a\).  Consequently, if the domains are chosen so that \(U_a\subset\{|z-a|<\eta\}\) for all \(a\in Z(P)\), then the zeros can be labelled against the zeros \(z_1,\dots,z_n\) of \(P\) so that \(|\widetilde z_j-z_j|<\eta\) for every \(j\).  Writing
\[
P(z)=z^n+\sum_{k=1}^{n}p_k z^{n-k},
\qquad
P_{\mathrm{ann}}^{\rho_-,\rho_+}(\widetilde Q;r,P)(z)=z^n+\sum_{k=1}^{n}\widetilde p_k z^{n-k},
\]
one has the explicit coefficient estimate
\[
|\widetilde p_k-p_k|
\le \binom{n}{k}k\eta(1+\eta)^{k-1},
\qquad 1\le k\le n.
\]
After the model \((r,P)\) and the three contours have been selected, the theorem produces and stabilizes the zero count of the annular divisor determined by that contour envelope.  Exact unit-circle zeros and reciprocal transport factorization remain the separate algebraic realizability conditions stated in Theorem \ref{thm:intrinsic-leyang-shell-recognition} and in the fixed-scale criteria.
\end{theorem}

\begin{proof}
The zeros of \(S_r\) are on \(|z|=r^{-2}\) and \(|z|=1\), so none of \(\Gamma_-\), \(\Gamma_+\), or \(\Gamma_0\) meets the zero set.  More explicitly, if the zeros of \(P\) are \(z_1,\dots,z_n\), then the shell factorization gives the exact identity
\[
S_r(z)
=a_0r^n\prod_{j=1}^n(z-z_j)
\prod_{j=1}^n(z-r^{-2}z_j^{-1})
=r^nP(z)\,z^nP(r^{-2}z^{-1}),
\]
where Lemma \ref{lem:unit-circle-reciprocal-shell-scalar} identifies
\(a_0=P(0)=(-1)^n\prod_jz_j\) as the reciprocal-shell scalar.  Therefore
\[
|S_r(z)|
=|a_0|r^n\prod_{j=1}^n |z-z_j|\prod_{j=1}^n |z-r^{-2}z_j^{-1}|.
\]
The scalar \(a_0\), the last product, and the following circle estimate
are exactly those in Lemma \ref{lem:unit-circle-reciprocal-shell-scalar};
in particular \(|a_0|=1\), because \(P\) is monic and all \(z_j\) lie on
\(\TT\).  Thus, on any circle \(|z|=\rho\),
\[
|S_r(z)|\ge r^n |1-\rho|^n|\rho-r^{-2}|^n.
\]
Applying this with \(\rho=\rho_-\), \(\rho=\rho_+\), and
\(\rho=\rho_+^{-1}\) gives the displayed product lower bound for
\(\Delta(r,P;\rho_-,\rho_+)\), in particular the three infima in the
definition of \(\Delta\) are positive.  On each contour the perturbation is strictly smaller than \(|S_r|\), so Rouch\'e's theorem implies that \(\widetilde S\) and \(S_r\) have the same number of zeros in the disks bounded by these contours.  The outer contour case is exactly Lemma \ref{lem:outer-rouche-prevents-degree-loss} with \(R=\rho_+^{-1}\), since all zeros of \(S_r\) lie in \(|z|<\rho_+^{-1}\); therefore \(\widetilde S\) is degree \(2n\) and the count on \(\Gamma_0\) is a finite zero count rather than a count with missing zeros at infinity.  Thus \(\widetilde S\) has \(n\) zeros in \(|z|<\rho_-\), \(n\) zeros in \(|z|<\rho_+\), and \(2n\) zeros in \(|z|<\rho_+^{-1}\).  Therefore no zeros lie in \(\rho_-<|z|<\rho_+\), and exactly \(n\) zeros lie in \(\rho_+<|z|<\rho_+^{-1}\).  This is the full Rouch\'e conclusion: a preserved contour zero count, plus degree preservation from the outer contour.  A finite zero divisor determines a unique monic polynomial, so those \(n\) annular zeros define the unique contour-selected annular factor \(P_{\mathrm{ann}}^{\rho_-,\rho_+}(\widetilde Q;r,P)\).  In the unperturbed case the annular zeros are precisely the zeros of \(P\).

For the local stability statement, the boundaries \(\partial U_a\) are compact and avoid the zero set of \(S_r\), while each \(U_a\) contains exactly the multiplicity \(m_a\) zero of \(S_r\) at \(a\).  Hence \(\Delta_{\mathcal U}(r,P)>0\).  The additional perturbation bound and Rouch\'e's theorem on every \(\partial U_a\) show that \(\widetilde S\) has exactly \(m_a\) zeros in \(U_a\), counted with multiplicity.  These zeros are among the annular zeros already selected, because every \(\overline{U_a}\) lies inside \(\rho_+<|z|<\rho_+^{-1}\), and their total number is \(\sum_a m_a=n\); therefore they are precisely the zeros of \(P_{\mathrm{ann}}^{\rho_-,\rho_+}(\widetilde Q;r,P)\).  If \(U_a\subset\{|z-a|<\eta\}\), label the \(m_a\) zeros in \(U_a\) against the \(m_a\) copies of the zero \(a\) of \(P\).  This gives \(|\widetilde z_j-z_j|<\eta\).  Since \(|z_j|=1\), the labelled zeros satisfy \(|\widetilde z_j|\le1+\eta\).  For each subset \(I\) of cardinality \(k\), the telescoping identity gives
\[
\left|\prod_{j\in I}\widetilde z_j-
\prod_{j\in I}z_j\right|
\le k\eta(1+\eta)^{k-1}.
\]
Summing over the \(\binom nk\) subsets yields the coefficient estimate, because
\[
p_k=(-1)^ke_k(z_1,
\dots,z_n),\qquad
\widetilde p_k=(-1)^ke_k(\widetilde z_1,
\dots,
\widetilde z_n).
\]
\end{proof}

\begin{corollary}[Coefficient-level local annular zero-count selector]\label{cor:coefficient-annular-stability}
Retain the hypotheses and notation of Theorem \ref{thm:quantitative-annular-rouche-stability}, and write
\[
Q_r(w)=\sum_{\ell=0}^{n}b_\ell w^\ell,
\qquad
\widetilde Q(w)=\sum_{\ell=0}^{n}\widetilde b_\ell w^\ell,
\]
allowing \(\widetilde b_n=0\) if \(\deg\widetilde Q<n\).  Put
\[
B_n(r;\rho_-,\rho_+)
:=
\max_{z\in\Gamma_-\cup\Gamma_+\cup\Gamma_0}
|z|^n\sum_{\ell=0}^{n}|J_r(z)|^\ell.
\]
The denominator \(\Delta(r,P;\rho_-,\rho_+)\) is the Rouch\'e envelope
from Theorem \ref{thm:quantitative-annular-rouche-stability}; its
explicit product lower bound uses Lemma
\ref{lem:unit-circle-reciprocal-shell-scalar} to remove the
reciprocal-shell scalar in absolute value.
If the coefficient errors satisfy
\[
\max_{0\le \ell\le n}|\widetilde b_\ell-b_\ell|
<
\frac{\Delta(r,P;\rho_-,\rho_+)}{B_n(r;\rho_-,\rho_+)},
\]
then, with \(\widetilde S(z):=z^n\widetilde Q(J_r(z))\),
\[
\sup_{\Gamma_-\cup\Gamma_+\cup\Gamma_0}
\left|z^n\widetilde Q(J_r(z))-S_r(z)\right|
<\Delta(r,P;\rho_-,\rho_+).
\]
Consequently \(\widetilde S\) satisfies the Rouch\'e envelope in Theorem
\ref{thm:quantitative-annular-rouche-stability}.  The associated
factor is the contour-selected annular factor associated with
\((\widetilde Q;r,P,\rho_-,\rho_+)\), namely
\(P_{\mathrm{ann}}^{\rho_-,\rho_+}(\widetilde Q;r,P)\), and it has
degree \(n\).  This coefficient condition is only a sufficient way to
obtain the contour zero-count hypothesis; it does not by itself certify
unit-circle source realizability.  For \(\widetilde Q=Q_r\) this factor is
exactly \(P\).
More generally, the same conclusion holds under the weighted \(\ell^1\)
coefficient condition
\[
\max_{z\in\Gamma_-\cup\Gamma_+\cup\Gamma_0}
|z|^n\sum_{\ell=0}^{n}|\widetilde b_\ell-b_\ell|\,|J_r(z)|^\ell
<\Delta(r,P;\rho_-,\rho_+).
\]
\end{corollary}

\begin{proof}
For each point of the three contours,
\[
\left|z^n\widetilde Q(J_r(z))-S_r(z)\right|
=
\left|z^n\sum_{\ell=0}^{n}(\widetilde b_\ell-b_\ell)J_r(z)^\ell\right|
\le
|z|^n\sum_{\ell=0}^{n}|\widetilde b_\ell-b_\ell|\,|J_r(z)|^\ell.
\]
The weighted \(\ell^1\) hypothesis is exactly the assertion that the right side is uniformly smaller than \(\Delta(r,P;\rho_-,\rho_+)\).  The simpler maximum-coefficient hypothesis implies it because the last display is at most
\[
\max_{0\le \ell\le n}|\widetilde b_\ell-b_\ell|\,
B_n(r;\rho_-,\rho_+).
\]
Taking the supremum over the three contours gives the displayed usable
perturbation bound.  Thus the contour hypothesis of Theorem
\ref{thm:quantitative-annular-rouche-stability} holds, and the announced
contour-selected annular factor statement follows from that theorem.
\end{proof}
