\section{The collapsed equilibrium fiber}\label{sec:collapsed-fiber}

At \(r=1\), the partner relation \(zt=r^{-2}\) from
\eqref{eq:collision-relation} becomes \(zt=1\), or
\(t=\overline z\) on \(\TT\).  We write
\[
 \iota(z)=\overline z,
 \qquad
 \mathsf S\eta=\frac12(\eta+\iota_\ast\eta).
\]
The operator \(\mathsf S\) is the projection of circle measures onto
their reflection-invariant component.  The next theorem describes both
its full equilibrium fiber and the penalty that appears when the
ellipse reopens.

\begin{theorem}[Collapsed-ellipse equilibrium fiber and opening
selection]\label{thm:collapsed-selection}
Let \(J_1(z)=z+z^{-1}\), and let
\(\eta\in\mathcal P(\TT)\).  The following assertions hold.

\begin{enumerate}[label=\textup{(\alph*)},leftmargin=*]
\item \emph{Collapsed identity.}  In the lower-truncated extended
sense,
\begin{equation}\label{eq:collapsed-identity}
 \boxed{I(J_{1\ast}\eta)=2I_{\TT}(\mathsf S\eta).}
\end{equation}
Consequently \(I(J_{1\ast}\eta)\leq0\), with equality if and only if
\(\mathsf S\eta=m_{\TT}\).

\item \emph{Equality classification.}  The complete equality class is
\begin{equation}\label{eq:complete-fiber}
 \boxed{
 \dd\eta=(1+h)\dd m_{\TT},\quad
 h\in L^\infty(m_{\TT};\RR),\quad
 h(\overline z)=-h(z),\quad |h|\leq1\ \text{a.e.}}
\end{equation}
Every measure in this affine fiber has the same image, the arcsine
equilibrium measure
\begin{equation}\label{eq:arcsine-measure}
 J_{1\ast}m_{\TT}
 =\frac{\mathbf1_{(-2,2)}(x)}{\pi\sqrt{4-x^2}}\dd x
\end{equation}
on \([-2,2]\).

\item \emph{Fourier description.}  If
\(I_{\TT}(\eta)>-\infty\), then
\begin{equation}\label{eq:endpoint-fourier}
 -I(J_{1\ast}\eta)
 =2\sum_{k=1}^{\infty}
 \frac{(\operatorname{Re}\widehat\eta(k))^2}{k}.
\end{equation}
Thus the collapsed energy detects exactly the reflection-symmetric
Fourier component.  The reflection-antisymmetric component is precisely
the freedom in \eqref{eq:complete-fiber}.

\item \emph{Opening-selection law.}  Suppose
\(\mathsf S\eta=m_{\TT}\), and write
\(\dd\eta=(1+h)\dd m_{\TT}\) as in
\eqref{eq:complete-fiber}.  For \(s>0\), let \(r=e^s\) and put
\[
 \varphi(x)=\frac{1-e^{-x}}{x},\qquad x>0.
\]
Then the exact selection identity is
\begin{equation}\label{eq:exact-selection}
 \frac{s-I(J_{e^s\ast}\eta)}{2s}
 =\sum_{k=1}^{\infty}\varphi(2ks)
 |\widehat\eta(k)|^2.
\end{equation}
Moreover,
\begin{equation}\label{eq:selection-limit}
 \boxed{
 \lim_{s\downarrow0}
 \frac{s-I(J_{e^s\ast}\eta)}{2s}
 =\sum_{k=1}^{\infty}|\widehat\eta(k)|^2
 =\frac12\int_{\TT}|h|^2\dd m_{\TT}.}
\end{equation}
The limiting functional has sharp range
\begin{equation}\label{eq:selection-range}
 0\leq\frac12\|h\|_{L^2(m_{\TT})}^2\leq\frac12.
\end{equation}
Its minimum is attained only at \(\eta=m_{\TT}\).
\end{enumerate}
\end{theorem}

\begin{proof}
For \(z,t\in\TT\), the collapsed factorization
\eqref{eq:collapsed-kernel-split} and the identity
\(|1-z^{-1}t^{-1}|=|z-\overline t|\) give
\[
 \log|J_1(z)-J_1(t)|
 =\log|z-t|+\log|z-\iota(t)|.
\]
Using \eqref{eq:pushforward-energy}, we obtain
\begin{equation}\label{eq:collapsed-energy-expanded}
 I(J_{1\ast}\eta)
 =I_{\TT}(\eta)+I_{\TT}(\eta,\iota_\ast\eta).
\end{equation}
This equality is valid in the extended sense because both complete
logarithmic kernels are bounded above; equivalently, one may subtract a
common upper bound and apply monotone convergence to nonnegative
kernels.

The circle kernel is invariant under \(\iota\).  Expanding the energy of
the symmetrization gives
\begin{align*}
 I_{\TT}(\mathsf S\eta)
 &=\frac14\bigl(
 I_{\TT}(\eta)
 +2I_{\TT}(\eta,\iota_\ast\eta)
 +I_{\TT}(\iota_\ast\eta)\bigr)\\
 &=\frac12\bigl(
 I_{\TT}(\eta)+I_{\TT}(\eta,\iota_\ast\eta)
 \bigr).
\end{align*}
Together with \eqref{eq:collapsed-energy-expanded}, this proves
\eqref{eq:collapsed-identity}.

The unit circle has capacity one and unique equilibrium measure
\(m_{\TT}\).  Frostman's theorem therefore gives
\[
 I_{\TT}(\mathsf S\eta)\leq0,
\]
with equality exactly when \(\mathsf S\eta=m_{\TT}\).  This proves all
claims in part~\textup{(a)}.  Notice also that
\(J_1\circ\iota=J_1\), so
\begin{equation}\label{eq:image-symmetrization}
 J_{1\ast}\eta=J_{1\ast}(\mathsf S\eta).
\end{equation}

We next classify the equality case.  If
\(\mathsf S\eta=m_{\TT}\), then
\[
 \eta+\iota_\ast\eta=2m_{\TT}.
\]
For every Borel set \(A\subset\TT\), this implies
\(\eta(A)\leq2m_{\TT}(A)\).  Hence
\(\eta=fm_{\TT}\) for a measurable function \(0\leq f\leq2\).  Since
\(m_{\TT}\) is invariant under reflection, the symmetrization condition
is equivalent to
\[
 f(z)+f(\overline z)=2
 \quad\text{for }m_{\TT}\text{-almost every }z.
\]
Setting \(h=f-1\) gives a real function with
\(|h|\leq1\) and \(h(\overline z)=-h(z)\).  Conversely, every such
function defines a probability density (1+h) whose symmetrization is
one.  This proves \eqref{eq:complete-fiber}.

Formula \eqref{eq:image-symmetrization} shows that every member of the
fiber has image \(J_{1\ast}m_{\TT}\).  Finally,
\(J_1(e^{i\theta})=2\cos\theta\).  Splitting the circle into two
semicircles and changing variables \(x=2\cos\theta\) gives
\[
 \dd(J_{1\ast}m_{\TT})(x)
 =\frac{\dd x}{\pi\sqrt{4-x^2}},
 \qquad -2<x<2,
\]
which is \eqref{eq:arcsine-measure}.  This completes part~\textup{(b)}.

Suppose now that \(\eta\) has finite circle energy.  Reflection acts on
Fourier coefficients by conjugation:
\[
 \widehat{\iota_\ast\eta}(k)
 =\overline{\widehat\eta(k)}.
\]
Consequently
\begin{equation}\label{eq:symmetrized-fourier}
 \widehat{\mathsf S\eta}(k)
 =\operatorname{Re}\widehat\eta(k).
\end{equation}
Lemma~\ref{lem:circle-fourier}, applied to
\(\mathsf S\eta\), gives
\[
 I_{\TT}(\mathsf S\eta)
 =-\sum_{k\geq1}
 \frac{(\operatorname{Re}\widehat\eta(k))^2}{k}.
\]
The collapsed identity then proves \eqref{eq:endpoint-fourier} and
part~\textup{(c)}.

Finally assume \(\mathsf S\eta=m_{\TT}\).  Part~\textup{(b)} gives a
bounded density (1+h), so \(\eta\) has finite circle energy.  Equation
\eqref{eq:symmetrized-fourier} also gives
\begin{equation}\label{eq:purely-imaginary-modes}
 \operatorname{Re}\widehat\eta(k)=0,
 \qquad k\geq1.
\end{equation}
Apply Proposition~\ref{prop:elliptic-deficit} with \(r=e^s\).  Only the
imaginary components survive, and
\begin{equation}\label{eq:selection-before-division}
 s-I(J_{e^s\ast}\eta)
 =\sum_{k=1}^{\infty}
 \frac{1-e^{-2ks}}{k}|\widehat\eta(k)|^2.
\end{equation}
Division by (2s) proves \eqref{eq:exact-selection}.

For every fixed \(k\), \(\varphi(2ks)\) increases to one as
\(s\downarrow0\).  Monotone convergence for counting measure on
\(\NN\) therefore yields
\[
 \lim_{s\downarrow0}
 \sum_{k\geq1}\varphi(2ks)|\widehat\eta(k)|^2
 =\sum_{k\geq1}|\widehat\eta(k)|^2.
\]
The nonzero Fourier coefficients of \(\eta\) are those of \(h\).
Because \(h\) is real, Parseval's identity gives
\[
 \int_{\TT}|h|^2\dd m_{\TT}
 =\sum_{k\in\ZZ}|\widehat h(k)|^2
 =2\sum_{k=1}^{\infty}|\widehat\eta(k)|^2.
\]
This proves \eqref{eq:selection-limit}.  The pointwise bound
\(|h|\leq1\) proves the upper bound in
\eqref{eq:selection-range}.  The limit is zero only when every nonzero
Fourier coefficient of \(h\) vanishes, hence only when \(h=0\) almost
everywhere.  This is exactly \(\eta=m_{\TT}\), completing the proof.
\end{proof}

The theorem separates two kinds of uniqueness.  The equilibrium measure
on the image remains unique at \(r=1\); what fails is uniqueness of its
source under the collapsed parametrization.  The opening functional is
defined on that entire source fiber and recovers source-level uniqueness
to first order.
