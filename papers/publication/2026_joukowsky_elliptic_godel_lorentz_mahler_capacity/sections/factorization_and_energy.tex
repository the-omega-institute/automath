\section{Boundary factorization and energy}\label{sec:factorization-energy}

\subsection{Lower-truncated logarithmic energy}

Let \(X\subset\CC\) be compact and let \(\sigma\) be a Borel
probability measure on \(X\).  For \(M>0\), set
\[
 I_X^{(M)}(\sigma)
 =\iint_{X\times X}
 \max\{\log|x-y|,-M\}\dd\sigma(x)\dd\sigma(y).
\]
The kernels decrease as \(M\) increases and are uniformly bounded
above on \(X\times X\).  We use the lower-truncated extended value
\begin{equation}\label{eq:lower-truncated-energy}
 I_X(\sigma)=\lim_{M\to\infty}I_X^{(M)}(\sigma)\in[-\infty,\infty).
\end{equation}
When the ambient set is clear, we write \(I(\sigma)\).  On \(\TT\) we
write \(I_{\TT}\).  A measure has finite logarithmic energy if this
value is greater than \(-\infty\).

The same convention applies to a mutual energy.  For probability
measures \(\alpha,\beta\) on \(X\), define
\[
 I_X(\alpha,\beta)
 =\lim_{M\to\infty}
 \iint\max\{\log|x-y|,-M\}\dd\alpha(x)\dd\beta(y).
\]
All kernels used below are bounded above, so no \(+\infty\) term occurs.
In particular, identities between complete logarithmic kernels can be
integrated in the extended sense without forming an indeterminate
difference of infinities.

We will separate a bounded factor from a singular logarithmic kernel.
The following elementary observation specifies when that operation is
legitimate.

\begin{lemma}[Bounded-term separation]\label{lem:bounded-separation}
Let \(L\) be an upper-bounded Borel kernel on a compact probability
space and let \(b\) be bounded and Borel.  If integrals of \(L\) and
(L+b) are defined by lower truncation, then
\[
 \iint(L+b)=\iint L+\iint b
\]
in the extended real line.
\end{lemma}

\begin{proof}
If the lower-truncated integral of \(L\) is finite, then the negative
part of \(L\) is integrable.  The truncated kernels converge pointwise
and are dominated after subtracting a common upper bound, so dominated
convergence permits addition of \(b\).  If the integral of \(L\) is
\(-\infty\), the inequalities
\[
 \max\{L,-M\}-\|b\|_\infty
 \leq \max\{L+b,-M-\|b\|_\infty\}
 \leq \max\{L,-M-2\|b\|_\infty\}+\|b\|_\infty
\]
show that the integral of (L+b) also tends to \(-\infty\).  The
claimed identity follows in both cases.
\end{proof}

For a continuous map \(F:X\to Y\), functoriality of pushforward gives
the exact lower-truncated identity
\begin{equation}\label{eq:pushforward-energy}
 I_Y(F_\ast\sigma)
 =\iint_{X\times X}\log|F(x)-F(y)|
   \dd\sigma(x)\dd\sigma(y).
\end{equation}
Indeed the equality holds for every bounded truncated kernel before the
limit is taken.  We will use \eqref{eq:pushforward-energy} for
\(F=J_r\) and \(F=J_r\circ B\).

\subsection{The minimum Joukowsky factorization}

Put
\[
 J_r(z)=rz+r^{-1}z^{-1},\qquad r\geq1,
\]
on \(\CC^\times\).  Direct subtraction gives
\begin{equation}\label{eq:boundary-factorization}
 J_r(z)-J_r(t)
 =(z-t)\left(r-r^{-1}z^{-1}t^{-1}\right).
\end{equation}
Equivalently,
\begin{equation}\label{eq:collision-relation}
 J_r(z)=J_r(t)
 \quad\Longleftrightarrow\quad
 z=t\quad\text{or}\quad zt=r^{-2}.
\end{equation}
This is the only rational-map factorization needed in the paper.

If \(z,t\in\TT\) and \(r>1\), the second relation in
\eqref{eq:collision-relation} is impossible.  Hence
\(J_r:\TT\to E_r:=J_r(\TT)\) is a continuous bijection from a compact
space to a Hausdorff space and therefore a homeomorphism.  Moreover,
\begin{equation}\label{eq:elliptic-kernel-split}
 \log|J_r(z)-J_r(t)|
 =\log r+\log|z-t|
  +\log|1-r^{-2}z^{-1}t^{-1}|.
\end{equation}
The last term is continuous on \(\TT^2\).  Lemma
\ref{lem:bounded-separation} therefore permits the singular circle
kernel and the continuous factor to be integrated separately.

At \(r=1\), the second collision graph enters \(\TT^2\).  The same
factorization becomes
\begin{equation}\label{eq:collapsed-kernel-split}
 J_1(z)-J_1(t)=(z-t)(1-z^{-1}t^{-1}).
\end{equation}
For \(z,t\in\TT\),
\[
 |1-z^{-1}t^{-1}|=|z-\overline t|,
\]
and hence
\begin{equation}\label{eq:collapsed-two-kernels}
 \log|J_1(z)-J_1(t)|
 =\log|z-t|+\log|z-\overline t|.
\end{equation}
Both logarithms are upper bounded.  Formula
\eqref{eq:collapsed-two-kernels}, interpreted as an identity of complete
extended kernels, is the source of the collapsed fiber.

\subsection{Circle Fourier energy}

For a finite Borel measure \(\sigma\) on \(\TT\), use the convention
\[
 \widehat\sigma(k)=\int_{\TT}z^{-k}\dd\sigma(z),
 \qquad k\in\ZZ.
\]
The classical Fourier expansion of the circle logarithmic kernel is
needed only under a finite-energy hypothesis.

\begin{lemma}[Circle logarithmic energy]\label{lem:circle-fourier}
If \(\sigma\in\mathcal P(\TT)\) has finite logarithmic energy, then
\begin{equation}\label{eq:circle-fourier-energy}
 I_{\TT}(\sigma)
 =-\sum_{k=1}^{\infty}\frac{|\widehat\sigma(k)|^2}{k}.
\end{equation}
If \(|q|<1\), then the continuous kernel satisfies
\begin{equation}\label{eq:continuous-fourier-factor}
 \iint_{\TT^2}\log|1-qz^{-1}t^{-1}|
 \dd\sigma(z)\dd\sigma(t)
 =-\sum_{k=1}^{\infty}
 \frac{\operatorname{Re}(q^k\widehat\sigma(k)^2)}{k}.
\end{equation}
The series in \eqref{eq:continuous-fourier-factor} converges absolutely.
\end{lemma}

\begin{proof}
For \(0<\rho<1\), the Abel-regularized kernel has the uniformly
convergent Fourier series
\[
 \log|1-\rho e^{i(\theta-\phi)}|
 =-\sum_{k\geq1}\frac{\rho^k\cos k(\theta-\phi)}{k}.
\]
Termwise integration gives
\begin{equation}\label{eq:abel-integrated}
 \iint\log|1-\rho e^{i(\theta-\phi)}|
 \dd\sigma(\theta)\dd\sigma(\phi)
 =-\sum_{k\geq1}\frac{\rho^k|\widehat\sigma(k)|^2}{k}.
\end{equation}
It remains to identify the boundary limit on the left.  Write
\[
 \kappa(\alpha)=\log|1-e^{i\alpha}|,
 \qquad
 \kappa_\rho(\alpha)=\log|1-\rho e^{i\alpha}|,
\]
and let \(\lambda\) be the image of \(\sigma\times\sigma\) under
\((\theta,\phi)\mapsto\theta-\phi\).  Finite circle energy says that
the negative part of \(\kappa\) is \(\lambda\)-integrable.  For
\(\rho\geq1/2\),
\[
 |1-\rho e^{i\alpha}|^2
 =(1-\rho)^2+\rho|1-e^{i\alpha}|^2,
\]
so
\[
 \kappa(\alpha)+\frac12\log\rho
 \leq\kappa_\rho(\alpha)\leq\log2.
\]
Dominated convergence therefore identifies the left side of
\eqref{eq:abel-integrated} with \(I_{\TT}(\sigma)\) as
\(\rho\uparrow1\).  On the right, monotone convergence for the
nonnegative series gives \eqref{eq:circle-fourier-energy}.

For \(|q|<1\), the power series
\[
 \log|1-qe^{-i(\theta+\phi)}|
 =-\sum_{k\geq1}
 \frac{\operatorname{Re}(q^ke^{-ik(\theta+\phi)})}{k}
\]
converges uniformly and absolutely.  Its integral is
\eqref{eq:continuous-fourier-factor} because
\[
 \iint e^{-ik(\theta+\phi)}
 \dd\sigma(\theta)\dd\sigma(\phi)
 =\widehat\sigma(k)^2.
\]
\end{proof}

\subsection{The noncollapsed deficit}

Combining the preceding two formulas gives exactly the Fourier statement
needed later.  We retain its real and imaginary components because their
different endpoint behavior is the selection mechanism.

\begin{proposition}[Elliptic Fourier deficit]\label{prop:elliptic-deficit}
Let \(r>1\) and let \(\eta\in\mathcal P(\TT)\) have finite circle
logarithmic energy.  Write
\(\widehat\eta(k)=x_k+iy_k\).  Then
\begin{equation}\label{eq:elliptic-deficit}
 \log r-I(J_{r\ast}\eta)
 =\sum_{k=1}^{\infty}
 \frac{(1+r^{-2k})x_k^2+(1-r^{-2k})y_k^2}{k}.
\end{equation}
In particular, every summand is nonnegative, and the imaginary weights
\(1-r^{-2k}\) vanish as \(r\downarrow1\).
\end{proposition}

\begin{proof}
Apply Lemma~\ref{lem:bounded-separation} to
\eqref{eq:elliptic-kernel-split}.  Lemma~\ref{lem:circle-fourier} with
\(q=r^{-2}\) gives
\[
 I(J_{r\ast}\eta)
 =\log r-\sum_{k\geq1}
 \frac{|\widehat\eta(k)|^2
 +r^{-2k}\operatorname{Re}(\widehat\eta(k)^2)}{k}.
\]
Since
\[
 |x_k+iy_k|^2+r^{-2k}\operatorname{Re}(x_k+iy_k)^2
 =(1+r^{-2k})x_k^2+(1-r^{-2k})y_k^2,
\]
the claimed form follows.
\end{proof}

The Fourier identity is not asserted for arbitrary measures.  The
all-measure extremal statement instead comes directly from classical
potential theory.

\begin{proposition}[Unique maximizer for \(r>1\)]
\label{prop:unique-elliptic-maximizer}
For \(r>1\),
\[
 \operatorname{cap}(E_r)=r,
 \qquad
 \nu_{E_r}=J_{r\ast}m_{\TT}.
\]
For every \(\eta\in\mathcal P(\TT)\), with no finite-energy assumption,
\begin{equation}\label{eq:all-measure-bound}
 I(J_{r\ast}\eta)\leq\log r,
\end{equation}
and equality holds if and only if \(\eta=m_{\TT}\).
\end{proposition}

\begin{proof}
The inverse relation \(w=J_r(z)\) is
\[
 rz^2-wz+r^{-1}=0.
\]
The root satisfying \(|z|>1\) near infinity is
\[
 z_+(w)=\frac{w+\sqrt{w^2-4}}{2r}
 =\frac{w}{r}+O(w^{-1}),\qquad w\to\infty,
\]
where the square root is normalized by
\(\sqrt{w^2-4}=w+O(w^{-1})\).  Relation
\eqref{eq:collision-relation} shows that \(J_r\) is one-to-one on the
exterior disk, and its critical points \(\pm r^{-1}\) lie inside
\(\DD\).  Hence the Green function of \(\CC\setminus E_r\) is
\[
 g_{E_r}(w)=\log|z_+(w)|
 =\log|w|-\log r+o(1).
\]
Comparison with the standard normalization
\(g_K(w)=\log|w|-\log\operatorname{cap}(K)+o(1)\) gives
\(\operatorname{cap}(E_r)=r\).  Harmonic measure from infinity is the
pushforward \(J_{r\ast}m_{\TT}\), so this is the equilibrium measure.

Frostman's variational principle now gives
\(I(\nu)\leq\log r\) for every probability measure \(\nu\) on
\(E_r\), with equality only at \(\nu_{E_r}\).  Apply this to
\(\nu=J_{r\ast}\eta\).  Since \(J_r:\TT\to E_r\) is a
homeomorphism, equality of the pushforwards is equivalent to
\(\eta=m_{\TT}\).
\end{proof}

Propositions~\ref{prop:elliptic-deficit} and
\ref{prop:unique-elliptic-maximizer} have distinct roles.  The second
gives the unique-maximizer regime for all probability measures.  The
first exposes the mode weights whose singular limit will distinguish
the collapsed fiber.  Conflating the two would either impose an
unnecessary finite-energy hypothesis on Frostman extremality or apply a
Fourier identity outside its valid class.
