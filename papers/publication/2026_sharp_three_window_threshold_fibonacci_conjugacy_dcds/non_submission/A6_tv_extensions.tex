\section{Further consequences of the total-variation bound}\label{app:tv-extensions}

\begin{theorem}[Bound for a coarse free-energy functional]\label{thm:free-energy-certificate}
See \cite[Theorem~A.1]{MaZhang2026Nonlinearity}.
\end{theorem}

\begin{corollary}[Golden-branch optimality among metallic means]\label{cor:golden-optimal}
For constant-type continued fractions $\alpha_A=[0;A,A,\dots]$, the golden
ratio case $A=1$ uniquely minimizes the leading discrepancy coefficient
$\kappa(A)=A/\log((A+\sqrt{A^2+4})/2)$ among metallic means in the
total-variation, bounded-observable, and observational free-energy
certificates.
\end{corollary}

\begin{remark}
The free-energy certificate is \cite[Theorem~A.1]{MaZhang2026Nonlinearity}.
The local increment in Corollary~\ref{cor:golden-optimal} is the metallic-mean
optimization obtained by inserting the standard constant-type discrepancy
coefficient into that certificate.
\end{remark}
