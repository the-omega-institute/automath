\section{Cyclic rank recoding for quadratic Pisot languages}
\label{sec:quadratic-pisot-extension}

This section closes the interface left by Remark~\ref{rem:metallic-beta-scope}.
The point requiring care is that a quadratic Pisot number need not be a unit
and its conjugate need not be negative.  Consequently the boundary-modified
constant-type language is not the appropriate common object.  We instead use
the greedy $\beta$-language and its colexicographic ordered-language rank, and
we introduce a cyclic reduction of that rank.  This recoding is not numerical
$\beta$-value normalization: it asserts congruence only between integer
language ranks and does not assert equality in $\mathbb Z[\beta]$.  The
classical $\beta$-shift and automata literature supplies context
\cite{Renyi1957,ItoTakahashi1974,Parry1960,Blanchard1989,Loraud1995,
BruyereHansel1997,LecomteRigo2001}; every finite-rank and
sliding-congruence statement used below is proved here.

\subsection{Quadratic classification and canonical finite windows}

\begin{lemma}[The two quadratic Pisot parameter chambers]
\label{lem:quadratic-pisot-chambers}
Let $\beta>1$ be a quadratic algebraic integer and let $\beta'$ be its
conjugate.  Then $|\beta'|<1$ if and only if the minimal polynomial of
$\beta$ belongs to exactly one of the following two families:
\begin{align*}
P^-_{a,b}(x)&=x^2-ax-b, & a\ge b\ge1,\\
P^+_{a,b}(x)&=x^2-ax+b, & a\ge3,\quad 1\le b\le a-2.
\end{align*}
In the first chamber $\beta'<0$, $\lfloor\beta\rfloor=a$, and
\[
d_\beta(1)=ab0^\infty,\qquad
d_\beta^*(1)=\bigl(a(b-1)\bigr)^\infty.
\]
In the second chamber $\beta'>0$, $\lfloor\beta\rfloor=a-1$, and
\[
d_\beta(1)=(a-1)(a-b-1)^\infty=d_\beta^*(1).
\]
Here $d_\beta$ is the greedy expansion and $d_\beta^*$ is Parry's
quasi-greedy boundary word.
\end{lemma}

\begin{proof}
Write the monic minimal polynomial as $x^2-ax+c$.  Since
$c=\beta\beta'\ne0$, the sign of $c$ is the sign of $\beta'$.  If
$c=-b<0$, then $b/\beta=|\beta'|<1$.  The positive root exceeds $a$;
the last inequality is equivalent, for integral $a,b$, to $1\le b\le a$.
Conversely these inequalities place the negative root in $(-1,0)$.
If $c=b>0$, then $0<b/\beta=\beta'<1$ and $a-1<\beta<a$; evaluating
$P^+_{a,b}$ at $a-1$ and $a$ gives exactly $a\ge3$ and
$1\le b\le a-2$.  The converses follow from the same sign test.

For $P^-_{a,b}$ one has $\beta^2=a\beta+b$.  The greedy algorithm applied
to $1$ therefore gives the two digits $a,b$ and then terminates; Parry's
standard replacement of a finite expansion gives
$d_\beta^*(1)=(a(b-1))^\infty$.  For $P^+_{a,b}$, put
$r=\beta-a+1=1-b/\beta$.  Then
$\lfloor\beta\rfloor=a-1$ and
\[
\beta r=\beta-b=(a-b-1)+r,
\]
so every digit after the first is $a-b-1$.  The expansion is infinite and
hence already quasi-greedy.  This is precisely Parry's calculation
\cite{Parry1960}; it is included to fix the parameters, not as a new proof of
his general admissibility theorem.
\end{proof}

For either chamber put $d=\lfloor\beta\rfloor$ and
$\mathcal A_\beta=\{0,\ldots,d\}$.  Let $\mathcal L_m(X_\beta)$ be the
length-$m$ language of the two-sided greedy $\beta$-shift, characterized by
Parry's suffix inequalities against $d_\beta^*(1)$.

For words $x=(x_1,\ldots,x_m)$ and $y=(y_1,\ldots,y_m)$ in low-to-high
orientation, write $x<_{\rm colex}y$ when
$(x_m,\ldots,x_1)<_{\rm lex}(y_m,\ldots,y_1)$.  Lexicographic comparison of a
finite word with an infinite boundary uses the available finite prefix: the
first unequal digit decides the comparison, and equality through the end of
the finite word is allowed.  A comparison with $c_0^m$ is the literal
finite-word comparison.

\begin{definition}[Cyclic rank recoding attached to the Parry language]
\label{def:quadratic-parry-fold}
For $m\ge0$ define $Q_m(\beta)=|\mathcal L_m(X_\beta)|$; the recurrences for
these counts are proved in Proposition~\ref{prop:parry-rank-bijection}.
Use the low-to-high orientation
\[
X_m^{(\beta)}:=\{x\in\mathcal A_\beta^m:
(x_m,\ldots,x_1)\in\mathcal L_m(X_\beta)\}
\]
and set
\[
\operatorname{Rank}_{\beta,m}(x)=\sum_{j=0}^{m-1}x_{j+1}Q_j(\beta).
\]
The cyclic language-rank recoding is
\[
\Fold_{\beta,m}(w)=
\bigl(\operatorname{Rank}_{\beta,m}|_{X_m^{(\beta)}}\bigr)^{-1}
\left(\operatorname{Rank}_{\beta,m}(w)\bmod Q_m(\beta)\right).
\]
Finally,
\[
(\Phi_{\beta,m}(z))_t
=\Fold_{\beta,m}(z_t,\ldots,z_{t+m-1}),\qquad
Y_{\beta,m}=\Phi_{\beta,m}(\mathcal A_\beta^{\mathbb Z}).
\]
The nontrivial overlap threshold is
\[
m_*(\beta)=\min\{m\ge2:\Phi_{\beta,m}\text{ is a conjugacy onto the
SFT }Y_{\beta,m}\}.
\]
\end{definition}

\begin{proposition}[Parry rank is consecutive and canonical]
\label{prop:parry-rank-bijection}
The language counts satisfy
\begin{align}
Q_0&=1,\quad Q_1=a+1,\quad Q_m=aQ_{m-1}+bQ_{m-2}
&& (P^-_{a,b},\ m\ge2),\label{eq:qminus}\\
Q_0&=1,\quad Q_1=a,\quad Q_m=aQ_{m-1}-bQ_{m-2}
&& (P^+_{a,b},\ m\ge2).\label{eq:qplus}
\end{align}
For every quadratic Pisot $\beta$ and $m\ge1$,
\[
\operatorname{Rank}_{\beta,m}|_{X_m^{(\beta)}}:
X_m^{(\beta)}\longrightarrow\{0,\ldots,Q_m(\beta)-1\}
\]
is an order-preserving bijection when $X_m^{(\beta)}$ is read in the
colexicographic order.  Thus Definition~\ref{def:quadratic-parry-fold} is
determined by the greedy language and its Parry word; it is not a choice of
an auxiliary enumeration.  For $P^-_{a,1}$ it is exactly
Definition~\ref{def:metallic-window-data}.
\end{proposition}

\begin{proof}
In the negative chamber, Parry admissibility reduces in the low-to-high
orientation to
\[
x_k=a\Longrightarrow x_{k-1}<b.
\]
We use induction on $m$, with the invariant that the ranks of $X_m^{(\beta)}$
form the initial interval $[0,Q_m-1]$.  It is immediate for $m=0,1$.  For
$m\ge2$, a high digit $h\in\{0,\ldots,a-1\}$ leaves an arbitrary word of
$X_{m-1}^{(\beta)}$ and hence contributes
\[
[hQ_{m-1},(h+1)Q_{m-1}-1].
\]
For high digit $a$, the next digit is
$g\in\{0,\ldots,b-1\}$ and the remaining word is arbitrary in
$X_{m-2}^{(\beta)}$; its ranks are
\[
aQ_{m-1}+[gQ_{m-2},(g+1)Q_{m-2}-1].
\]
The intervals are consecutive in colexicographic order and their union is
$[0,aQ_{m-1}+bQ_{m-2}-1]$.  This proves simultaneously the negative
recurrence, the induction invariant, and order preservation.

For the positive chamber put
\[
d=a-1,\qquad c_0=a-b-1,
\]
and define the boundary-bounded sublanguage
\[
B_m:=\{x\in X_m^{(\beta)}:
(x_m,\ldots,x_1)\le_{\rm lex}c_0^m\},
\qquad R_m:=|B_m|.
\]
We prove simultaneously, for every $m\ge0$, the coupled induction invariant
\begin{equation}
\Rank_{\beta,m}(X_m^{(\beta)})=[0,Q_m-1],
\qquad
\Rank_{\beta,m}(B_m)=[0,R_m-1].
\label{eq:coupled-rank-invariant}
\end{equation}
Here $X_0^{(\beta)}=B_0=\{\varnothing\}$, so $Q_0=R_0=1$ and the assertion
holds at $m=0$.  At $m=1$, one has $Q_1=d+1=a$ and
$R_1=c_0+1=a-b$, and the ranks are the stated intervals.

Assume \eqref{eq:coupled-rank-invariant} through length $m-1$.  Parry
admissibility says that a high digit $h<d$ may be followed by any word in
$X_{m-1}^{(\beta)}$, whereas a high digit $d$ must be followed by a word in
$B_{m-1}$.  The classes $h=0,\ldots,d-1$ therefore give the consecutive
intervals
\[
[hQ_{m-1},(h+1)Q_{m-1}-1],
\]
and the high class $h=d$ gives
\[
[dQ_{m-1},dQ_{m-1}+R_{m-1}-1].
\]
Consequently
\begin{equation}
Q_m=dQ_{m-1}+R_{m-1},
\qquad
\Rank_{\beta,m}(X_m^{(\beta)})=[0,Q_m-1].
\label{eq:positive-X-coupling}
\end{equation}

The same decomposition inside $B_m$ has high digits
$0,\ldots,c_0-1$ followed by arbitrary words of $X_{m-1}^{(\beta)}$, and
high digit $c_0$ followed by a word of $B_{m-1}$.  Hence
\begin{equation}
R_m=c_0Q_{m-1}+R_{m-1},
\qquad
\Rank_{\beta,m}(B_m)=[0,R_m-1].
\label{eq:positive-B-coupling}
\end{equation}
This closes the simultaneous induction and proves consecutiveness rather than
assuming it.  Subtracting \eqref{eq:positive-B-coupling} from
\eqref{eq:positive-X-coupling} gives
\[
Q_m-R_m=(d-c_0)Q_{m-1}=bQ_{m-1}.
\]
Thus, for $m\ge2$,
\[
R_{m-1}=Q_{m-1}-bQ_{m-2}
\]
and
\[
Q_m=dQ_{m-1}+R_{m-1}
=(d+1)Q_{m-1}-bQ_{m-2}
=aQ_{m-1}-bQ_{m-2},
\]
which proves \eqref{eq:qplus}.  The order of the displayed digit classes is
the colexicographic order, so the same induction proves order preservation.
When $b=1$ in the negative chamber, the rule and recurrence are precisely
those of Definition~\ref{def:metallic-window-data}.
\end{proof}

\subsection{The bounded sliding-congruence lemma}

The separation estimate used in the sliding argument is isolated first.  In
particular, the integer multiple and the error term are both retained.

\begin{lemma}[Nearest-multiple separation]
\label{lem:nearest-multiple-separation}
Let $Q_0,Q_1,\ldots$ satisfy either \eqref{eq:qminus} or \eqref{eq:qplus},
write $s=1$ or $s=-1$ accordingly, and put
\[
d=\begin{cases}a,&s=1,\\a-1,&s=-1.\end{cases}
\]
Then, for $r\ge4$ and $1\le |e|\le d$,
\begin{equation}
\operatorname{dist}(beQ_{r-1},Q_r\mathbb Z)\ge Q_{r-2}.
\label{eq:quadratic-separation}
\end{equation}
\end{lemma}

\begin{proof}
Distance is unchanged by replacing $e$ with $-e$, so assume $1\le e\le d$.
Set
\[
R_r:=\frac{Q_{r-1}}{Q_{r-2}}.
\]
The recurrence gives $R_{r+1}=a+sb/R_r$.  Its initial values and the parameter
ranges imply, by induction,
\begin{align}
a+\frac{b}{a+1}&\le R_r<a+\frac ba &&(s=1,\ r\ge3),
\label{eq:ratio-plus}\\
a-\frac{b}{a-1}&<R_r\le a-\frac ba &&(s=-1,\ r\ge3).
\label{eq:ratio-minus}
\end{align}
For completeness, in the positive-recurrence case the map
$x\mapsto a+b/x$ reverses order, sends
$[a+b/(a+1),a+b/a)$ into itself, and $R_3=a+b/(a+1)$.
In the negative-recurrence case $x\mapsto a-b/x$ preserves order, the interval
in \eqref{eq:ratio-minus} lies in $(a-1,a)$, and $R_3=a-b/a$.
These observations prove the two inductions, including the stated strict
endpoints.

We shall also use the following sharpened bounds, obtained by applying the
same recurrence once more:
\begin{align*}
L_+&:=a+\frac{b}{a+b/(a+b/(a+1))},&
U_+&:=a+\frac{b}{a+b/(a+1)},\\
L_-&:=a-\frac{b}{a-1},&
U_-&:=a-\frac{b}{a-b/a}.
\end{align*}
Thus
\begin{equation}
L_+\le R_r\le U_+\quad(s=1,r\ge4),
\qquad
L_-<R_r\le U_-\quad(s=-1,r\ge4).
\label{eq:sharpened-ratio-bounds}
\end{equation}
The first assertion follows because the order-reversing map interchanges the
endpoints $R_4=U_+$ and $R_5=L_+$ and maps $[L_+,U_+]$ into itself.  The
second follows because $R_3=a-b/a$, $R_4=U_-$, and the increasing map sends
$(L_-,U_-]$ into itself.

Choose a nearest integer $k$ to
\[
z:=\frac{beQ_{r-1}}{Q_r}=\frac{be}{R_{r+1}}.
\]
The lower bounds $R_{r+1}>a$ for $s=1$ and $R_{r+1}>a-1$ for $s=-1$ give
\[
0<z<b.
\]
Indeed $e\le a$ in the first case and $e\le a-1$ in the second.  Hence every
nearest integer satisfies
\begin{equation}
0\le k\le b.
\label{eq:nearest-k-bound}
\end{equation}
Put $h:=be-ak$.  The exact error, with no omitted remainder, is
\begin{align}
E&:=beQ_{r-1}-kQ_r\notag\\
 &=(be-ak)Q_{r-1}-skbQ_{r-2}
   =(hR_r-skb)Q_{r-2}.
\label{eq:nearest-error}
\end{align}

We record the bounded integer exclusion used for the only cancelling sign
cases.  This is separated from the recurrence argument so that all of its
integer hypotheses are visible.  For the parameter ranges above and integers
$1\le e\le d$, $0\le k\le b$, $h=be-ak$, one has
\begin{align}
h>0, s=1&\Longrightarrow
  hL_+-kb\ge1\quad\hbox{or}\quad kb-hU_+\ge1,
\label{eq:endpoint-plus}\\
h<0, s=-1&\Longrightarrow
  (-h)L_--kb\ge1\quad\hbox{or}\quad kb-(-h)U_-\ge1.
\label{eq:endpoint-minus}
\end{align}
Here is the integer calculation.  Write $j=kb-ah$ in the first line and
$j=a(-h)-kb$ in the second.  Since $h=be-ak$, solving for $(e,k)$ gives
\[
k=\frac{ah+j}{b},\qquad
e=\frac{(a^2+b)h+aj}{b^2}\quad(s=1),
\]
and, with $g=-h$,
\[
k=\frac{ag-j}{b},\qquad
e=\frac{(a^2-b)g-aj}{b^2}\quad(s=-1).
\]
Thus the two displayed numerators are divisible by $b$ and $b^2$,
respectively.  We give the remaining Euclidean divisions explicitly.  In the
$s=1$ case write
\[
be=aq+r,\qquad qb=ap+t,\qquad 0\le r,t<a.
\]
If $k\le q-1$, then $h\ge a+r$ and
\[
hL_+-kb>(a+r)a-(b-1)b\ge1.
\]
If $k\ge q+1$, then $h<0$, so this case is not in the hypothesis of
\eqref{eq:endpoint-plus}.  Hence only $k=q$ remains, and then $h=r$.
Now $j=a(p-r)+t$.  The cases $p<r$ and $p>r$ give, respectively,
$hL_+-kb\ge1$ and $kb-hU_+\ge1$ directly from $0\le t<a$ and
$0<U_+-a<b/a$.  When $p=r$, put $w=rb-at$.  The two division identities give
\begin{equation}
b^2e=r(a^2+b)+at,\qquad w=r(a^2+2b)-b^2e.
\label{eq:plus-terminal-divisions}
\end{equation}
The integer $w$ is nonzero: if $w=0$, the three divisions imply successively
$a\mid rb$, $a\mid qb$ and $a\mid eb$, contrary to $0<r<a$.  Substitution of
\eqref{eq:plus-terminal-divisions}, followed by multiplication by the positive
denominators, gives the exhaustive terminal alternatives
\[
\begin{array}{c|c}
w>0& rb(a^2+a+b)\ge(t+1)(a^3+a^2+2ab+b),\\
w<0& t(a^2+a+b)\ge rb(a+1)+(a^2+a+b).
\end{array}
\]
Indeed, after moving the right side to the left, division first by $a$ and
then by $b$ has remainders $w$ and $r$, respectively; in the first row both
remainders are positive, while in the second their negatives are positive.
These are precisely $rL_+-t\ge1$ and $t-r(U_+-a)\ge1$.  This proves
\eqref{eq:endpoint-plus}.

For $s=-1$, put $g=-h>0$ and perform the same divisions with subtraction:
\[
be=aq+r,\qquad kb=ap+t,\qquad 0\le r,t<a.
\]
The inequality $g>0$ forces $k\ge q+1$.  If $k\ge q+2$, then
$g=ak-be\ge a+1$, and, since $k\le b$ and $b\le a-2$,
\[
gL_--kb
\ge (a+1)\left(a-\frac b{a-1}\right)-b^2\ge1.
\]
It remains to take $k=q+1$, for which $g=a-r$.  If $p<g$, then
$j=ag-kb=a(g-p)-t$, and clearing the denominator in $gL_--kb\ge1$ gives
\[
(a-1)\{a(g-p)-t-1\}-gb\ge0.                 \tag{*}
\]
Substitute $r=a-g$, $be=aq+r$, and $kb=ap+t$.  Successive division by
$a$ and $a-1$ leaves the nonnegative remainders
$g-p-1$, $a-1-e$, and $a-2-b$; hence (*) holds.  If $p>g$, then
$kb-ag=a(p-g)+t\ge1$, so $kb-gU_-\ge1$.  Finally, if $p=g$, then
$kb-ag=t$.  This again gives $kb-gU_-\ge1$ when $t>0$; when $t=0$, the
three exact divisions, together with $e\le a-1$, give
$gab\ge a^2-b$, which is the same inequality after multiplication by
$a^2-b$.  These cases are exhaustive and prove
\eqref{eq:endpoint-minus}.

We now cover all signs explicitly.  If $s=1$ and $h<0$, then
\[
|hR_r-kb|=(-h)R_r+kb>1.
\]
If $h=0$, then $k\ge1$ (because $e\ge1$), and
$|hR_r-kb|=kb\ge1$.  If $h>0$, the two alternatives in
\eqref{eq:endpoint-plus}, together with
$L_+\le R_r\le U_+$, give $|hR_r-kb|\ge1$.

If $s=-1$ and $h>0$, then $hR_r+kb>1$.  If $h=0$, again $k\ge1$ and the
modulus is $kb\ge1$.  If $h<0$, put $g=-h$; then
$|hR_r+kb|=|kb-gR_r|$, and
\eqref{eq:endpoint-minus} with $L_-<R_r\le U_-$ gives the same lower bound.
Thus in both chambers
\[
|hR_r-skb|\ge1.
\]
Equation~\eqref{eq:nearest-error} now yields $|E|\ge Q_{r-2}$, which is
\eqref{eq:quadratic-separation}.
\end{proof}

The following sliding lemma is the arithmetic core of the block decoder.

\begin{lemma}[Quadratic bounded sliding congruences]
\label{lem:quadratic-sliding-congruences}
Take either parameter chamber in
Lemma~\ref{lem:quadratic-pisot-chambers}, let $d=\lfloor\beta\rfloor$, and
let $(Q_j)$ be \eqref{eq:qminus} or \eqref{eq:qplus}.  If $m\ge3$ and
$e_0,\ldots,e_{2m-2}\in[-d,d]\cap\mathbb Z$ satisfy
\begin{equation}
\sum_{j=0}^{m-1}Q_j e_{t+j}\equiv0\pmod{Q_m}
\qquad(0\le t<m),\label{eq:sliding-kernel}
\end{equation}
then $e_0=\cdots=e_{2m-2}=0$.
\end{lemma}

\begin{proof}
Write $s=1$ in the negative chamber and $s=-1$ in the positive chamber, so
$Q_{j+2}=aQ_{j+1}+sbQ_j$, and put
$\delta=Q_1-a$; hence $(d,\delta)=(a,1)$ for $s=1$ and
$(d,\delta)=(a-1,0)$ for $s=-1$.  Subtracting $a$ times the congruence at
$t+1$ and $sb$ times that at $t+2$ from the congruence at $t$ cancels every
interior coefficient and gives
\begin{equation}
e_t+\delta e_{t+1}-sbQ_{m-1}e_{t+m+1}\equiv0\pmod{Q_m}.
\label{eq:quadratic-elimination}
\end{equation}

Suppose first that $m\ge4$.  The small part of
\eqref{eq:quadratic-elimination} has modulus at most
$d(1+\delta)$.  On the other hand,
\[
Q_{m-2}\ge Q_2=
\begin{cases}
a^2+a+b>2a=d(1+\delta),&s=1,\\
a^2-b>a-1=d(1+\delta),&s=-1.
\end{cases}
\]
Lemma~\ref{lem:nearest-multiple-separation} therefore gives a strictly larger
distance whenever $e_{t+m+1}\ne0$.  Hence
$e_{t+m+1}=0$ for $0\le t\le m-3$.  Thus
$e_{m+1}=\cdots=e_{2m-2}=0$.  The final original congruence reduces to
\[
e_{m-1}+Q_1e_m\equiv0\pmod{Q_m}.
\]
Its left side has modulus at most $d(1+Q_1)<Q_m$ and $Q_1>d$, so both
digits vanish.  Back substitution in the preceding congruences then gives
$e_{m-2},\ldots,e_0=0$ successively.

It remains to check $m=3$, where endpoint equality in the separation bound
can occur.  Here each integer sum in \eqref{eq:sliding-kernel} has modulus
less than $2Q_3$, so it is $k_tQ_3$ with $k_t\in\{-1,0,1\}$.  Substituting
the three values of $t$ gives
\begin{equation}
e_t+Q_1e_{t+1}+Q_2e_{t+2}=k_tQ_3,
\qquad t=0,1,2.
\label{eq:quadratic-m3-equalities}
\end{equation}
We now eliminate the two conjugate signs separately.

Suppose first that $s=1$.  Thus $d=a$ and
\[
(Q_1,Q_2,Q_3)
=(a+1,a^2+a+b,a^3+a^2+2ab+b).
\]
Consider one equation $x+Q_1y+Q_2z=kQ_3$ occurring in
\eqref{eq:quadratic-m3-equalities}.  By symmetry it suffices to treat
$k=0,1$.  If $k=0$, then $|z|\ge2$ is impossible because
$2Q_2>a(1+Q_1)$.  The choice $z=0$ gives $x=y=0$, while $z=1$ gives
\[
x+b+(a+1)(y+a)=0.
\]
Since $y+a\ge0$ and $-(x+b)\le a-b<a+1$, this forces
$(x,y,z)=(-b,-a,1)$; the case $z=-1$ is its negative.  If $k=1$ and
$z\le a-1$, then
\[
x+Q_1y+Q_2z
\le a+aQ_1+(a-1)Q_2
=Q_3-\bigl(b(a+2)-a\bigr)<Q_3.
\]
Hence $z=a$, after which $x+(a+1)y=b(a+1)$ forces $(x,y)=(0,b)$.
Consequently every bounded solution of one equation is
\[
(x,y,z;k)\in
\{(0,0,0;0),\ \pm(b,a,-1;0),\ \pm(0,b,a;1)\}.
\]
The terminal pairs of these triples are
$(0,0)$, $\pm(a,-1)$, and $\pm(b,a)$.  If $e_4\ne0$ and the $t=2$
triple is $\pm(0,b,a)$, its initial pair $\pm(0,b)$ cannot be a terminal
pair of the $t=1$ triple.  If instead the $t=2$ triple is
$\pm(b,a,-1)$, the overlap forces the $t=1$ triple to be
$\pm(0,b,a)$, whose initial pair $\pm(0,b)$ cannot be a terminal pair of
the $t=0$ triple.  Both alternatives are impossible, and therefore
$e_4=0$.

Suppose next that $s=-1$.  Put $C=a-b$ and $d=a-1$; then
$2\le C\le d$ and
\[
(Q_1,Q_2,Q_3)=(a,a^2-b,a^3-2ab).
\]
For one equation $x+ay+Q_2z=kQ_3$, again it suffices to take $k=0,1$.
When $k=0$, the inequality $2Q_2>d(1+a)=a^2-1$ gives $|z|\le1$.
The choice $z=0$ gives $x=y=0$, whereas $z=1$ gives
\[
x-b+a(y+a)=0.
\]
Here $1\le y+a\le2a-1$ and $b-x\le b+a-1\le2a-3$, so necessarily
$y+a=1$ and $x=b-a=-C$; the case $z=-1$ is the negative solution.
For $k=1$, if $z\le a-2$, then
\[
x+ay+Q_2z
\le d(1+a)+(a-2)Q_2
=Q_3-\bigl(a(a-b)-2b+1\bigr)<Q_3,
\]
because $a-b\ge2$.  Thus $z=d$, and
\[
x+ay=a(a-b)-b=aC-b.
\]
Reduction modulo $a$, together with $|x|\le a-1$, leaves precisely
$(x,y)=(-b,C)$ and $(x,y)=(C,C-1)$.  Hence the complete list is
\[
(x,y,z;k)\in
\{(0,0,0;0),\ \pm(C,d,-1;0),\
  \ \pm(-b,C,d;1),\ \pm(C,C-1,d;1)\}.
\]
Its terminal pairs are $(0,0)$, $\pm(d,-1)$, $\pm(C,d)$, and
$\pm(C-1,d)$.  Since $2\le C\le d$, neither $\pm(-b,C)$ nor
$\pm(C,C-1)$ occurs in this list of terminal pairs.  Thus a $t=2$ triple
of either of the last two types cannot overlap a $t=1$ triple.  The only
remaining nonzero possibility is $\pm(C,d,-1)$; it forces the $t=1$
triple to be $\pm(-b,C,d)$, whose initial pair $\pm(-b,C)$ cannot overlap
a $t=0$ triple.  Therefore $e_4=0$ also in this chamber.

In either sign, every nonzero triple in the corresponding list has nonzero
last coordinate.  It follows successively from the equations at
$t=2,1,0$ that $e_2=e_3=e_4=0$, then $e_1=0$, and finally $e_0=0$.
Thus the kernel is trivial also at $m=3$.  The finite computation described in
the data-availability statement includes this case as regression testing; it
is not used as proof of the parameter-uniform statement.
\end{proof}

\subsection{Sharp threshold and extremal bases}

\begin{theorem}[Full quadratic-Pisot cyclic-rank threshold]
\label{thm:full-quadratic-pisot-threshold}
Let $\beta$ be an arbitrary quadratic Pisot number and use exactly the
greedy language, Parry data, rank, cyclic modulus and fold of
Definition~\ref{def:quadratic-parry-fold}.  Then
\[
m_*(\beta)=
\begin{cases}
3,&\beta\text{ is the root of }x^2-ax-a,\quad a\ge1,\\
3,&\beta\text{ is the root of }x^2-ax+1,\quad a\ge3,\\
2,&\text{otherwise}.
\end{cases}
\]
More strongly, for every $m\ge m_*(\beta)$ and every $n\ge m$, the
finite-block sliding fold is injective and
\begin{equation}
|\mathcal L_n(Y_{\beta,m})|
=(\lfloor\beta\rfloor+1)^{n+m-1}.
\label{eq:quadratic-pisot-block-count}
\end{equation}
At these apertures $\Phi_{\beta,m}$ is a topological conjugacy from the full
$(\lfloor\beta\rfloor+1)$-shift onto the SFT $Y_{\beta,m}$, with inverse of
memory zero and anticipation $m-1$.

Consequently the maximum of the threshold function on the full quadratic
Pisot family is three.  Its extremizers are exactly the negative-conjugate
diagonal $P^-_{a,a}$ and the positive-conjugate norm-one family
$P^+_{a,1}$.  The golden mean is the least extremizer, but it is not the
unique extremizer after passage from metallic units to all quadratic Pisot
bases.
\end{theorem}

\begin{proof}
Equality of two folded output blocks is, by
Proposition~\ref{prop:parry-rank-bijection}, equivalent to the sliding
congruences \eqref{eq:sliding-kernel} for their digit differences.  For
$m\ge3$, Lemma~\ref{lem:quadratic-sliding-congruences} makes every
length-$m$ finite-block map injective.  Consecutive decoded blocks cover a
longer raw block, proving injectivity for every $n\ge m$ and the count
\eqref{eq:quadratic-pisot-block-count}.  The standard gluing argument used in
Theorem~\ref{thm:metallic-threshold-classification} then gives a local inverse
and an SFT description by forbidden output words of length at most $m+1$.

It remains to classify $m=2$.  Put
\[
\kappa=Q_2-dQ_1=
\begin{cases}b,&P^-_{a,b},\\ a-b,&P^+_{a,b}.
\end{cases}
\]
Proposition~\ref{prop:parry-rank-bijection} gives the same explicit rule in
both chambers:
\[
\Fold_{\beta,2}(u,v)=
\begin{cases}
(u,v),&v<d\text{ or }u<\kappa,\\
(u-\kappa,0),&v=d\text{ and }u\ge\kappa.
\end{cases}
\]
If $d=\kappa$, the two constant raw points
$0^{\mathbb Z}$ and $d^{\mathbb Z}$ both map to the constant zero output,
so conjugacy fails.  This occurs exactly for $P^-_{a,a}$ and $P^+_{a,1}$.

Now assume $d>\kappa$, which is equivalent to $a>b$ in the negative chamber
and to $b>1$ in the positive chamber.  We prove finite-block injectivity,
including the boundary digits, rather than inferring it from a two-sided
inverse.  Let a raw triple $(a_0,a_1,a_2)$ produce
\[
y_i=(p_i,q_i)=\Fold_{\beta,2}(a_i,a_{i+1}),\qquad i=0,1.
\]
First recover the central digit from the two output symbols:
\begin{equation}
a_1=
\begin{cases}
q_0,&q_0>0,\\
0,&q_0=0\text{ and }p_1=0,\\
d,&q_0=0\text{ and }p_1\in\{d-\kappa,d\}.
\end{cases}
\label{eq:m2-central-decoder}
\end{equation}
Indeed, $q_0=0$ implies $a_1\in\{0,d\}$.  If $a_1=0$, the next pair has
first output coordinate $p_1=0$.  If $a_1=d$, then $p_1=d$ when
$a_2<d$ and $p_1=d-\kappa$ when $a_2=d$.  The alternatives are disjoint
because $d>\kappa>0$.

Having found $a_1$, recover the left boundary digit by
\begin{equation}
a_0=
\begin{cases}
p_0,&q_0>0,\\
p_0,&q_0=0\text{ and }a_1=0,\\
p_0+\kappa,&q_0=0\text{ and }a_1=d.
\end{cases}
\label{eq:m2-left-decoder}
\end{equation}
The last line is exactly the wrapped case in the explicit fold rule.  Finally,
recover the right boundary digit by
\begin{equation}
a_2=
\begin{cases}
q_1,&q_1>0,\\
0,&q_1=0\text{ and }p_1=a_1,\\
d,&q_1=0\text{ and }p_1=a_1-\kappa.
\end{cases}
\label{eq:m2-right-decoder}
\end{equation}
When $q_1=0$, the raw digit $a_2$ is $0$ or $d$; these give the last two
lines, which are disjoint because $\kappa>0$.  Equations
\eqref{eq:m2-central-decoder}--\eqref{eq:m2-right-decoder} prove directly that
\[
\Phi_{\beta,2,2}:\mathcal A_\beta^3
\longrightarrow (X_2^{(\beta)})^2
\]
is injective.

For any $n\ge2$, every consecutive pair $(y_t,y_{t+1})$ decodes the raw
triple $(a_t,a_{t+1},a_{t+2})$.  These triples cover the entire raw block
$a_0\ldots a_{n+1}$, so $\Phi_{\beta,2,n}$ is injective and
$|\mathcal L_n(Y_{\beta,2})|=(d+1)^{n+1}$.  On trajectories, using the first
coordinate of the triple decoded from $(y_t,y_{t+1})$ gives a genuine
memory-zero, anticipation-one inverse.

For the two extremal loci, Lemma~\ref{lem:quadratic-sliding-congruences} shows
that aperture three succeeds.  The chamber classification is exhaustive by
Lemma~\ref{lem:quadratic-pisot-chambers}, completing the proof.
\end{proof}

\subsection{Arithmetic identifiability and optimal local structure}

The threshold classification leaves three finer questions: whether a fixed
aperture identifies the parameter chamber, how many consecutive output
symbols are actually needed by a causal inverse, and whether the finite-block
and finite-type bounds in Theorem~\ref{thm:full-quadratic-pisot-threshold}
are sharp.  We answer all three for the recoding of
Definition~\ref{def:quadratic-parry-fold}.

\begin{theorem}[Aperture-two chamber duality and first separation]
\label{thm:quadratic-aperture-two-duality}
Let $q\ge3$ and $2\le\kappa\le q-1$, and let
\begin{align*}
\beta^-_{q,\kappa}&>1,
& (\beta^-_{q,\kappa})^2-(q-1)\beta^-_{q,\kappa}-\kappa&=0,\\
\beta^+_{q,\kappa}&>1,
& (\beta^+_{q,\kappa})^2-q\beta^+_{q,\kappa}+(q-\kappa)&=0.
\end{align*}
After identifying both digit alphabets with $\{0,\ldots,q-1\}$, their
aperture-two legal languages, ranks, folds, and sliding recodings coincide
pointwise.  Explicitly,
\[
Q_0=1,\qquad Q_1=q,\qquad Q_2=q(q-1)+\kappa,
\]
and
\[
X_2=\{(u,v):v<q-1\ \hbox{or}\ u<\kappa\},
\qquad
\Fold_2(u,v)=
\begin{cases}
(u,v),&v<q-1\ \hbox{or}\ u<\kappa,\\
(u-\kappa,0),&v=q-1\ \hbox{and}\ u\ge\kappa.
\end{cases}
\]
Moreover, these are all cross-chamber coincidences at aperture two: the pair
$(q,\kappa)$ is recovered from the digit alphabet and the fold, and its
equivalence class contains the displayed negative- and positive-conjugate
bases when $\kappa\ge2$.  For $\kappa=1$ only the negative-conjugate member
is a quadratic Pisot base in the parameter ranges of
Lemma~\ref{lem:quadratic-pisot-chambers}.

The ambiguity is first resolved at aperture three.  In low-to-high
orientation,
\[
X_3^{(\beta^-_{q,\kappa})}
 \setminus X_3^{(\beta^+_{q,\kappa})}
=\{(h,\kappa-1,q-1):\kappa\le h\le q-1\},
\]
the reverse difference is empty, and
\[
Q_3(\beta^-_{q,\kappa})-Q_3(\beta^+_{q,\kappa})=q-\kappa.
\]
Thus the complete aperture-two input--output map does not determine the sign
of the conjugate, whereas the aperture-three legal language does.
\end{theorem}

\begin{proof}
For the negative chamber take $(a,b)=(q-1,\kappa)$; for the positive chamber
take $(a,b)=(q,q-\kappa)$.  The stated parameter bounds are precisely those
which place both choices in Lemma~\ref{lem:quadratic-pisot-chambers}.  In both
cases $d=q-1$, while Proposition~\ref{prop:parry-rank-bijection} gives
\[
Q_2=(q-1)q+\kappa=q^2-(q-\kappa).
\]
The two length-two Parry conditions both say that a high digit $q-1$ must be
followed, in the low-to-high convention, by a digit smaller than $\kappa$.
The displayed fold now follows directly from cyclic reduction of
$u+qv$ modulo $Q_2$.  This proves pointwise equality of the two sliding maps.

Conversely, the alphabet gives $q$, and the least $u$ for which
$(u,q-1)$ is sent to $(0,0)$ gives $\kappa$.  The chamber parametrization in
Lemma~\ref{lem:quadratic-pisot-chambers} then leaves exactly the two displayed
members when $\kappa\ge2$ and only the negative member when $\kappa=1$.

Finally, the two quasi-greedy boundary words are
\[
d_{\beta^-}^*(1)=((q-1)(\kappa-1))^\infty,
\qquad
d_{\beta^+}^*(1)=(q-1)(\kappa-1)^\infty.
\]
Their first two digits agree.  At length three, the negative boundary permits
a suffix $(q-1,\kappa-1,h)$ for every $h\le q-1$, whereas the positive
boundary requires $h\le\kappa-1$.  All other suffix tests agree.  Reversing
to the low-to-high convention gives exactly the stated set difference.
There are $q-\kappa$ such words.  Equivalently, the two recurrences give
\[
Q_3^-=(q-1)Q_2+\kappa q,\qquad
Q_3^+=qQ_2-(q-\kappa)q,
\]
whose difference is $q^2-Q_2=q-\kappa$.
\end{proof}

We next isolate the only bounded-carry fact not already contained in
Lemma~\ref{lem:quadratic-sliding-congruences}.

\begin{lemma}[Negative-chamber two-window bounded carry]
\label{lem:negative-two-window-bounded-carry}
Let $a\ge b\ge1$, let
\[
Q_0=1,\quad Q_1=a+1,\quad Q_{j+2}=aQ_{j+1}+bQ_j,
\]
and let $m\ge3$.  If $e_0,\ldots,e_m\in[-a,a]\cap\mathbb Z$ satisfy
\[
\sum_{j=0}^{m-1}Q_je_j\equiv0\pmod{Q_m},
\qquad
\sum_{j=0}^{m-1}Q_je_{j+1}\equiv0\pmod{Q_m},
\]
then $e_0=0$.
\end{lemma}

\begin{proof}
We give the bounded coefficient descent because the digit bound $a$ is
essential.  Put $P(z)=z^2-az-b$ and
$E(z)=\sum_{j=0}^m e_jz^j$.  Divide in $\mathbb Z[z]$:
\[
E(z)=P(z)\sum_{i=0}^{m-2}c_iz^i+r_0+r_1z.
\]
Coefficient comparison gives
\begin{align*}
e_0&=-bc_0+r_0,\\
e_1&=-ac_0-bc_1+r_1,\\
e_j&=c_{j-2}-ac_{j-1}-bc_j &&(2\le j\le m-2),\\
e_{m-1}&=c_{m-3}-ac_{m-2},\qquad e_m=c_{m-2}.
\end{align*}
The linear functional $L(\sum f_jz^j)=\sum f_jQ_j$ annihilates every
$z^iP(z)$.  Removing the top term before applying $L$, and dividing
$E-e_0$ by $z$, therefore give respectively
\begin{equation}
r_0+(a+1)r_1\equiv0\pmod{Q_m},
\qquad r_1+c_0\equiv0\pmod{Q_m}.
\label{eq:negative-carry-remainders}
\end{equation}

For completeness, write the two original sums as $k_0Q_m$ and $k_1Q_m$.
The bound
\[
a\sum_{j=0}^{r-1}Q_j<2Q_r\qquad(r\ge3)
\]
follows by induction from the recurrence (the case $a=b=1$ is the only
$a=1$ case and is immediate).  Hence $k_0,k_1\in\{-1,0,1\}$.

We now perform coefficient descent.  Starting with the two highest
coefficients and moving toward $c_0$, the only update is
\begin{equation}
c_{j-2}=e_j+ac_{j-1}+bc_j.
\label{eq:negative-carry-descent}
\end{equation}
At the low end one uses
\eqref{eq:negative-carry-remainders} and the first two coefficient equations.
Induction in the number of applications of
\eqref{eq:negative-carry-descent} gives the following exhaustive table:
\[
\begin{array}{c|c|c}
k_1&k_0&\text{consequence}\\ \hline
-1&-1,0,1&\max_{0\le j\le m}|e_j|\ge a+1,\\
0&-1,0,1&e_0=0,\\
1&-1,0,1&\max_{0\le j\le m}|e_j|\ge a+1.
\end{array}
\]
To check the induction step, substitute
\eqref{eq:negative-carry-descent} in the next lower coefficient.  If the two
successive carries have the same sign, the term $ac_{j-1}+bc_j$ has modulus
at least $a+1$ unless the next carry is zero.  If they have opposite signs,
the substitution changes the sign and reduces to the preceding row.  Thus a
nonzero $k_1$ reaches an endpoint coefficient of modulus at least $a+1$.
When $k_1=0$, the descent reaches
$r_1=-c_0$; substituting this in the other congruence and then in
$e_0=-bc_0+r_0$ gives
\[
e_0=(a+1-b)c_0+\nu Q_m
\]
for some $\nu\in\mathbb Z$.  The same-sign alternative gives modulus at
least $a+1$, while the opposite-sign alternative descends once more; at the
endpoint it leaves $c_0=\nu=0$.  Hence $e_0=0$.  These are exactly the three
rows of the table, so the descent is exhaustive.  The assumed bound excludes
the first and third rows and proves the lemma.
\end{proof}

\begin{theorem}[Optimal causal inverse length]
\label{thm:quadratic-optimal-causal-inverse}
For $m\ge3$ define
\[
\ell_{\rm cau}(\beta,m)
=\min\{\ell\ge1:\ a_t=\psi(y_t,\ldots,y_{t+\ell-1})
\text{ for every }y=\Phi_{\beta,m}(a)\}.
\]
Then
\[
\ell_{\rm cau}(\beta,m)=
\begin{cases}
2,&\beta'<0,\\
3,&\beta'>0.
\end{cases}
\]
In particular, this optimal length is independent of $m$; the anticipation
is respectively one or two.

If, explicitly, $(P)\subset\mathbb Z[z]$ denotes the principal ideal
generated by $P$ and
\[
\mathfrak a(P,d)=\min\{\deg H:0\ne H\in(P),\ H(0)\ne0,
\ \max_j|[z^j]H|\le d\},
\]
then
\[
\ell_{\rm cau}(\beta,m)=\mathfrak a(P_\beta,d),
\qquad d=\lfloor\beta\rfloor,\quad m\ge3.
\]
\end{theorem}

\begin{proof}
Suppose first that $P=P^-_{a,b}$, so $d=a$.  The coefficient vector
$(b,a,-1)$ of $-P$ is the difference of two digit blocks and has nonzero
first coordinate.  Padding it by zeros gives two raw length-$m$ words with
the same single output, so one output never suffices.  Conversely, equality
of two consecutive outputs gives the two congruences in
Lemma~\ref{lem:negative-two-window-bounded-carry} for the digit differences;
that lemma gives $e_0=0$.  Hence two outputs determine the first digit.

Now let $P=P^+_{a,b}$, so $d=a-1$.  The polynomial
\[
(1+z)P(z)=b+(b-a)z+(1-a)z^2+z^3
\]
has all coefficients in $[-d,d]$ and nonzero constant term.  Its positive
and negative parts, padded by zeros, give two raw blocks with different first
digits and the same two consecutive outputs.  Indeed, the first weighted sum
is zero when $m\ge4$ and is $-Q_3$ when $m=3$, while the shifted sum is zero.
Thus two outputs do not suffice.

Three outputs do suffice.  For $m=3$ this is
Lemma~\ref{lem:quadratic-sliding-congruences}.  For $m\ge4$, apply the
elimination leading to \eqref{eq:quadratic-elimination} to the first three
equal-output congruences.  Since here $s=-1$ and $\delta=0$, it gives
\[
e_0+bQ_{m-1}e_{m+1}\equiv0\pmod{Q_m}.
\]
If $e_{m+1}\ne0$, Lemma~\ref{lem:nearest-multiple-separation} makes the
distance of the second term from $Q_m\mathbb Z$ at least
$Q_{m-2}>a-1\ge|e_0|$, a contradiction.  Hence $e_{m+1}=0$, and then
$|e_0|<Q_m$ forces $e_0=0$.

Finally, no nonzero polynomial of degree below two belongs to $(P_\beta)$.
For $P^-_{a,b}$ the polynomial $-P^-_{a,b}$ has coefficient norm $a$, so
$\mathfrak a(P^-_{a,b},a)=2$.  For $P^+_{a,b}$ every degree-two member of
the integral ideal is a nonzero integral multiple of $P^+_{a,b}$ and has
coefficient norm at least $a>d$, while $(1+z)P^+_{a,b}$ has coefficient norm
at most $a-1$.  Hence $\mathfrak a(P^+_{a,b},a-1)=3$, proving the last
assertion.
\end{proof}

\begin{theorem}[Critical two-fixed-point normal form]
\label{thm:quadratic-critical-normal-form}
Let $\beta$ be an extremizer in
Theorem~\ref{thm:full-quadratic-pisot-threshold}, put
$q=\lfloor\beta\rfloor+1$ and $d=q-1$, and consider its critical map
$\Phi_{\beta,2}$.  Then
\[
\Fold_{\beta,2}(u,v)=
\begin{cases}
(u,v),&(u,v)\ne(d,d),\\
(0,0),&(u,v)=(d,d).
\end{cases}
\]
Its only nonsingleton two-sided fiber is
\[
\Phi_{\beta,2}^{-1}((00)^{\mathbb Z})
=\{0^{\mathbb Z},d^{\mathbb Z}\}.
\]
Consequently
\[
h_{\rm top}(Y_{\beta,2})=\log q,
\qquad
\#\operatorname{Fix}(\sigma^n|_{Y_{\beta,2}})=q^n-1,
\qquad
\zeta_{Y_{\beta,2}}(z)=\frac{1-z}{1-qz}.
\]
The critical image is strictly sofic.  Its right Fischer cover has vertices
$0,\ldots,d$ and one edge $u\to v$ for every ordered pair; the edge label is
$(u,v)$ except that $d\to d$ is labeled $(0,0)$.  The cover is bi-resolving,
and the factor is one-to-one away from the two fixed points above.  Hence, in
the standard terminology, the critical image is a near-Markov AFT shift and
its multiplicity graph is $K_{2,1}$: the two exceptional domain orbits lie
over the single exceptional image orbit.
\end{theorem}

\begin{proof}
At either extremal locus the proof of
Theorem~\ref{thm:full-quadratic-pisot-threshold} has $\kappa=d$, which gives
the displayed rule.  Every nonzero output symbol records its raw edge
uniquely and hence propagates the raw digits uniquely in both directions.
If every output symbol is $(0,0)$, each raw edge is either $0\to0$ or
$d\to d$; overlap forces one of the two constant paths.  This proves the
fiber statement.

The map is finite-to-one from the full $q$-shift, so entropy is $\log q$.
A point fixed by $\sigma^n$ outside the exceptional image has a unique
preimage and that preimage is fixed by $\sigma^n$.  The $q^n$ fixed points
of the full shift therefore map to $q^n-1$ fixed image points, since exactly
the two constant preimages are identified.  The zeta identity follows from
\[
\exp\left(\sum_{n\ge1}\frac{(q^n-1)z^n}{n}\right)
=\frac{1-z}{1-qz}.
\]

The displayed graph is right- and left-resolving and follower-separated.
Every vertex is reached by a one-symbol label different from $(0,0)$, which
is intrinsically synchronizing.  Lemma~\ref{lem:fischer-cover-criterion}
therefore identifies it as the right Fischer cover.

To see that the image is not an SFT, fix $N$ and choose $L>N$.  The word
\[
(d,0)(0,0)^L(d,0)
\]
is not admissible: the first anchor forces the intervening hidden state to be
$0$, while the second requires it to be $d$.  Every subword of length at
most $N+1$ meets at most one anchor and is admissible, using the constant
$0$ or constant $d$ hidden path as appropriate.  Thus no finite forbidden
list defines the image.
\end{proof}

\begin{theorem}[Exact finite-block onset, Fischer cover, and Markov order]
\label{thm:quadratic-exact-local-order}
Let $m\ge m_*(\beta)$ and define
\[
\delta(\beta,m)=\min\{n\ge1:\Phi_{\beta,m,n}\text{ is injective}\}.
\]
Then
\[
\delta(\beta,m)=m.
\]
The natural overlap graph with vertex set
$\mathcal A_\beta^{m-1}$ and edge labels $\Fold_{\beta,m}$ is the right
Fischer cover of $Y_{\beta,m}$ and has exactly
$(\lfloor\beta\rfloor+1)^{m-1}$ vertices.  In the output alphabet fixed in
Definition~\ref{def:quadratic-parry-fold}, the image is an $m$-step SFT but
not an $(m-1)$-step SFT.
\end{theorem}

\begin{proof}
The upper bound $\delta(\beta,m)\le m$ is
Theorem~\ref{thm:full-quadratic-pisot-threshold}.  For $m\ge3$ the reverse
inequality has a uniform certificate.  In the negative chamber, place the
difference vector $(b,a,-1)$ in raw positions $m-2,m-1,m$ and zeros
elsewhere.  In the positive chamber place
$(b,b-a,1-a,1)$ in positions $m-3,\ldots,m$.  Each vector is the difference
of two digit blocks.  The recurrence shows that all $m-1$ consecutive output
ranks agree modulo $Q_m$, while the raw blocks differ.  For an invertible
aperture $m=2$, the two pairs $(0,0)$ and $(\kappa,d)$ have the same single
output, so again $\delta\ge m$.

The overlap graph is essential and irreducible because its underlying graph
is the full de Bruijn graph.  It is right-resolving: two edges with the same
initial vertex and different appended digits have rank difference
$hQ_{m-1}$ with $1\le|h|\le d$, which is nonzero modulo $Q_m$.  It is
left-resolving because two edges with the same terminal vertex and different
initial digits have nonzero rank difference of modulus at most $d<Q_m$.
By $\delta=m$, every output word of length $m$ labels a unique path and is
intrinsically synchronizing.  Distinct vertices have distinct follower
languages: a common right-infinite label would be decoded by
Theorem~\ref{thm:quadratic-optimal-causal-inverse} when $m\ge3$, or by
\eqref{eq:m2-central-decoder}--\eqref{eq:m2-right-decoder} when $m=2$, and
would force the two raw vertices to coincide.  The Fischer-cover criterion
now proves the cover assertion and its state count.

The gluing proof in Theorem~\ref{thm:full-quadratic-pisot-threshold} defines
$Y_{\beta,m}$ by forbidden output words of length at most $m+1$, so it is an
$m$-step SFT.  The difference certificates above give a length-$(m-1)$
output word which labels two paths ending at distinct Fischer states; for
$m=2$ the same role is played by the common label $(0,0)$.  Since the states
have distinct follower languages, this word is not intrinsically
synchronizing.  Every word of length $r$ in an $r$-step SFT is intrinsically
synchronizing.  Hence $Y_{\beta,m}$ is not an $(m-1)$-step SFT.
\end{proof}

\subsection{Worked cases in both chambers}

The following computations use the low-to-high convention $(u,v)$ and show
the Parry boundary, the legal rank interval, the cyclic reduction, and the
threshold classification in four genuinely nonmetallic cases.

\begin{example}[Negative conjugate, extremal]
Let $\beta$ be the positive root of $x^2-2x-2$.  Here
$(a,b,d)=(2,2,2)$,
\[
d_\beta^*(1)=(21)^\infty,\qquad (Q_0,Q_1,Q_2)=(1,3,8),
\qquad \kappa=b=2=d.
\]
At length two, legality is $v=2\Rightarrow u<2$.  Thus the legal pairs are
\[
(0,0),(1,0),(2,0),(0,1),(1,1),(2,1),(0,2),(1,2),
\]
and their ranks $u+3v$ are exactly $0,\ldots,7$.  The raw pair $(2,2)$ has
rank $8$, so cyclic reduction modulo $Q_2=8$ gives
$\Fold_{\beta,2}(2,2)=(0,0)$.  Consequently the constant raw points
$0^{\mathbb Z}$ and $2^{\mathbb Z}$ collide at aperture two.  Theorem
\ref{thm:full-quadratic-pisot-threshold} gives $m_*(\beta)=3$, as predicted by
the negative extremal condition $a=b$.
\end{example}

\begin{example}[Negative conjugate, nonextremal]
Let $\beta$ be the positive root of $x^2-3x-2$.  Then
$(a,b,d)=(3,2,3)$,
\[
d_\beta^*(1)=(31)^\infty,\qquad (Q_0,Q_1,Q_2)=(1,4,14),
\qquad \kappa=b=2<d.
\]
The legal pairs obey $v=3\Rightarrow u<2$; explicitly,
\[
X_2^{(\beta)}=\{0,1,2,3\}\times\{0,1,2\}
\ \cup\ \{(0,3),(1,3)\},
\]
and their ranks $u+4v$ fill $0,\ldots,13$.  The only wrapped pairs are
\[
\Fold_{\beta,2}(2,3)=(0,0),\qquad
\Fold_{\beta,2}(3,3)=(1,0).
\]
For the raw triple $(3,3,0)$ the two outputs are
$y_0=(1,0)$ and $y_1=(3,0)$.  Formula
\eqref{eq:m2-central-decoder} reads $p_1=3\in\{d-\kappa,d\}=\{1,3\}$ and
recovers $a_1=3$; then \eqref{eq:m2-left-decoder} gives
$a_0=1+2=3$, and \eqref{eq:m2-right-decoder} gives $a_2=0$ because
$p_1=a_1$.  Hence the boundary triple is recovered from two outputs and
$m_*(\beta)=2$.
\end{example}

\begin{example}[Positive conjugate, extremal]
Let $\beta$ be the positive root of $x^2-3x+1$.  Here
$(a,b,d,c_0)=(3,1,2,1)$,
\[
d_\beta^*(1)=21^\infty,\qquad (Q_0,Q_1,Q_2)=(1,3,8),
\qquad \kappa=a-b=2=d.
\]
The boundary condition at length two is $v=2\Rightarrow u\le1$.  Hence the
legal pairs are
\[
(0,0),(1,0),(2,0),(0,1),(1,1),(2,1),(0,2),(1,2),
\]
with ranks $0,\ldots,7$, while $(2,2)$ has rank $8$ and is recoded to
$(0,0)$.  The constant raw points $0^{\mathbb Z}$ and
$2^{\mathbb Z}$ therefore collide at aperture two.  This is the positive
norm-one extremal family $b=1$, and $m_*(\beta)=3$.
\end{example}

\begin{example}[Positive conjugate, nonextremal]
Let $\beta$ be the positive root of $x^2-4x+2$.  Then
$(a,b,d,c_0)=(4,2,3,1)$,
\[
d_\beta^*(1)=31^\infty,\qquad (Q_0,Q_1,Q_2)=(1,4,14),
\qquad \kappa=a-b=2<d.
\]
The length-two boundary again gives $v=3\Rightarrow u\le1$, so the legal
ranks on
\[
X_2^{(\beta)}=\{0,1,2,3\}\times\{0,1,2\}
\ \cup\ \{(0,3),(1,3)\}
\]
are $0,\ldots,13$, and the wrapped pairs are
$(2,3)\mapsto(0,0)$ and $(3,3)\mapsto(1,0)$.  For the raw triple $(3,3,0)$,
the output pair $((1,0),(3,0))$ is decoded exactly as in the preceding
nonextremal example: $a_1=3$, $a_0=1+\kappa=3$, and $a_2=0$.  Thus the
positive-chamber nonextremal case has $m_*(\beta)=2$.
\end{example}

\begin{remark}[What is and is not claimed]
\label{rem:quadratic-open-interface}
The theorem concerns the cyclic language-rank recoding of
Definition~\ref{def:quadratic-parry-fold}.  Published finite-transducer
normalization results for Pisot bases
\cite{Frougny1992,FrougnySteiner2008,BertheFrougnyRigoSakarovitch2019}
establish recognizability for numerical representations on other redundant
digit sets and boundary conventions, but they do not identify those
normalizations with this cyclic rank map.
No threshold claim is made for signed digits, non-greedy expansions,
non-cyclic truncation, degree greater than two, or general Pisot substitution
codings.  These choices are genuine residual interfaces rather than
consequences of Theorem~\ref{thm:full-quadratic-pisot-threshold}.
\end{remark}
