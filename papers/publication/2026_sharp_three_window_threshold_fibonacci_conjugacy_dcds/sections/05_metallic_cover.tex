\section{Metallic comparison and the Fischer cover}\label{sec:metallic-cover}

The sharp Fibonacci threshold can be compared with the constant-type metallic
finite-window systems of Definition~\ref{def:metallic-window-data}. The first
result shows that the golden case is exceptional inside that model: once the
constant partial quotient is at least two, the two-window branch disappears.
The second result identifies the intrinsic right-resolving presentation of the
Fibonacci image shift above threshold.

\subsection{The metallic two-window boundary}

\begin{definition}[Constant-type metallic window data]\label{def:metallic-window-data}
Fix an integer $A\ge 2$ and define
\[
Q_0^{(A)}:=1,\qquad Q_1^{(A)}:=A+1,\qquad
Q_{j+2}^{(A)}:=A Q_{j+1}^{(A)}+Q_j^{(A)}.
\]
This is the metallic recurrence associated with the continued-fraction tail
$[A,A,\ldots]$, shifted so that the golden case $A=1$ gives
$Q_j^{(1)}=F_{j+2}$.
Let
\[
\mathcal A_A:=\{0,1,\ldots,A\}.
\]
For $m\ge 1$, set
\[
X_m^{(A)}
:=
\{x\in \mathcal A_A^m:\ x_k=A\Rightarrow x_{k-1}=0
\text{ for }2\le k\le m\},
\]
and define
\[
\Val_{A,m}(x):=\sum_{k=1}^m x_k Q_{k-1}^{(A)}.
\]
The corresponding finite-window normalization is
\[
\Fold_{A,m}(\omega)
:=
\bigl(\Val_{A,m}|_{X_m^{(A)}}\bigr)^{-1}
\bigl(\Val_{A,m}(\omega)\bmod Q_m^{(A)}\bigr),
\]
whenever the displayed restriction is bijective.  The sliding two-window map is
\[
(\Phi_{A,2}(a))_t:=\Fold_{A,2}(a_t,a_{t+1}),
\qquad a\in \mathcal A_A^{\mathbb Z}.
\]
\end{definition}

\begin{lemma}[Golden member of the metallic notation]\label{lem:metallic-golden-member}
If the formulas of Definition~\ref{def:metallic-window-data} are evaluated at
$A=1$, then
\[
Q_j^{(1)}=F_{j+2},\qquad
X_m^{(1)}=X_m,\qquad
\Val_{1,m}=\Val_m,\qquad
\Fold_{1,m}=\Fold_m.
\]
In particular $\Phi_{1,2}$ is exactly the Fibonacci two-window map $\Phi_2$.
\end{lemma}

\begin{proof}
The sequence $(Q_j^{(1)})$ has initial values $Q_0^{(1)}=1=F_2$ and
$Q_1^{(1)}=2=F_3$, and satisfies
$Q_{j+2}^{(1)}=Q_{j+1}^{(1)}+Q_j^{(1)}$.  Hence
$Q_j^{(1)}=F_{j+2}$ for all $j$.

When $A=1$, the alphabet is $\{0,1\}$ and the legality condition
$x_k=1\Rightarrow x_{k-1}=0$ is exactly the no-adjacent-ones condition
defining $X_m$.  The valuation becomes
\[
\Val_{1,m}(x)=\sum_{k=1}^m x_kQ_{k-1}^{(1)}
=\sum_{k=1}^m x_kF_{k+1}=\Val_m(x).
\]
The modulus $Q_m^{(1)}$ is $F_{m+2}$, so the two definitions of finite-window
normalization agree.  Applying this at every coordinate for $m=2$ gives
$\Phi_{1,2}=\Phi_2$.
\end{proof}

\begin{theorem}[Metallic two-window rigidity within the constant-type model]
\label{thm:metallic-two-window-rigidity}
\leanpending{paper_fibonacci_stabilization_metallic_two_window_rigidity}
Within the constant-type metallic finite-window normalization family of
Definition~\ref{def:metallic-window-data}, for every integer $A\ge 2$ the
following statements hold.
\begin{enumerate}[label=(\roman*),leftmargin=*,itemsep=2pt]
  \item For every $m\ge 1$, the restriction
  \[
  \Val_{A,m}|_{X_m^{(A)}}:X_m^{(A)}\to
  \{0,1,\ldots,Q_m^{(A)}-1\}
  \]
  is a bijection.
  \item At $m=2$ the normalization is explicit:
  \[
  \Fold_{A,2}(u,v)=
  \begin{cases}
  (u,v), & v<A\text{ or }u=0,\\
  (u-1,0), & v=A\text{ and }u\ge 1.
  \end{cases}
  \]
  \item The two-window sliding map
  \[
  \Phi_{A,2}:(\mathcal A_A^{\mathbb Z},\sigma)\to
  (Y_{A,2},\sigma),
  \qquad
  Y_{A,2}:=\Phi_{A,2}(\mathcal A_A^{\mathbb Z}),
  \]
  is a topological conjugacy onto its image.  Its inverse is the
  two-symbol sliding-block code
  \[
  a_t=\psi_A(y_{t-1},y_t),\qquad y=\Phi_{A,2}(a),
  \]
  where, for $y_s=(p_s,q_s)\in X_2^{(A)}$,
  \[
  \psi_A(y_{t-1},y_t)=
  \begin{cases}
  q_{t-1}, & q_{t-1}>0,\\
  0, & q_{t-1}=0\text{ and }p_t=0,\\
  A, & q_{t-1}=0\text{ and }p_t\in\{A-1,A\}.
  \end{cases}
  \]
  The three alternatives are exhaustive for pairs in $\mathcal L_2(Y_{A,2})$.
\end{enumerate}
Consequently, within the constant-type metallic finite-window normalization
family of Definition~\ref{def:metallic-window-data}, the golden case $A=1$ is
the only member of this model in which a nontrivial two-window branch locus
remains.
\end{theorem}

\begin{proof}
First prove the finite-window range statement.  Let $N_m:=|X_m^{(A)}|$.
Clearly $N_0=1$ and $N_1=A+1$.  For $m\ge2$, a legal word ending in a digit
$0,\ldots,A-1$ has an arbitrary legal prefix of length $m-1$, while a legal
word ending in $A$ must have penultimate digit $0$ and therefore has an
arbitrary legal prefix of length $m-2$.  Hence
\[
N_m=A N_{m-1}+N_{m-2}.
\]
Thus $N_m=Q_m^{(A)}$ for all $m$.

The same last-digit decomposition proves that the valuation gives disjoint
consecutive intervals. For $m=1$ this is immediate. Assume the statement
known up to length $m-1$. If the last digit is $d<A$, then
\[
\Val_{A,m}(x)=\Val_{A,m-1}(x_1,\ldots,x_{m-1})+dQ_{m-1}^{(A)}
\]
runs through the interval
\[
[dQ_{m-1}^{(A)},(d+1)Q_{m-1}^{(A)}-1].
\]
As $d=0,\ldots,A-1$, these intervals cover
$[0,AQ_{m-1}^{(A)}-1]$. If the last digit is $A$, legality forces
$x_{m-1}=0$, and the remaining prefix of length $m-2$ gives precisely
\[
[AQ_{m-1}^{(A)},AQ_{m-1}^{(A)}+Q_{m-2}^{(A)}-1]
=[AQ_{m-1}^{(A)},Q_m^{(A)}-1].
\]
The intervals are disjoint and cover exactly
$\{0,\ldots,Q_m^{(A)}-1\}$.  This proves (i).

For $m=2$ one has $Q_2^{(A)}=A^2+A+1$ and
\[
\Val_{A,2}(u,v)=u+(A+1)v.
\]
If $v<A$ or $u=0$, the pair $(u,v)$ is legal and its valuation lies in
$[0,Q_2^{(A)}-1]$, so it is fixed by normalization.  If $v=A$ and $u\ge1$,
then
\[
u+(A+1)A=Q_2^{(A)}+(u-1),
\]
whose residue is $u-1$.  The unique legal pair of valuation $u-1$ is
$(u-1,0)$.  This proves (ii).

It remains to prove the path-space assertion. Let
$a\in\mathcal A_A^{\mathbb Z}$ and $y=\Phi_{A,2}(a)$, with
$y_t=(p_t,q_t)$.  By the explicit formula in (ii), if $q_{t-1}>0$ then the
raw second digit of the pair $(a_{t-1},a_t)$ is exactly $q_{t-1}$, hence
$a_t=q_{t-1}$.

Suppose next that $q_{t-1}=0$.  The explicit formula in (ii) shows that the
only possible values of $a_t$ are $0$ and $A$.  If $a_t=0$, then the next raw
pair starts with $0$, is already legal for every value of $a_{t+1}$, and its
normalized output has first coordinate $p_t=0$.  If $a_t=A$, then the next raw
pair starts with $A$.  When $a_{t+1}<A$, the pair is legal and the next output
has first coordinate $p_t=A$; when $a_{t+1}=A$, the pair $(A,A)$ normalizes to
$(A-1,0)$ and the next output has first coordinate $p_t=A-1$.  Since
$A\ge2$, the sets $\{0\}$ and $\{A-1,A\}$ are disjoint.  Therefore the displayed
formula for $\psi_A$ recovers $a_t$ uniquely from $(y_{t-1},y_t)$.

The map $\Phi_{A,2}$ is continuous, shift-equivariant and surjective onto
$Y_{A,2}$ by definition. The preceding local formula gives a continuous
shift-equivariant inverse on $Y_{A,2}$, proving that $\Phi_{A,2}$ is a
topological conjugacy onto its image.

By Lemma~\ref{lem:metallic-golden-member}, the formal $A=1$ member of this
notation is the Fibonacci map $\Phi_2$.  For $A=1$, the separating sets become
$\{0\}$ and $\{0,1\}$, so the preceding local distinction collapses. The
remaining ambiguity is exactly the golden two-window branch locus classified
in Theorem~\ref{thm:m2-branch-locus}.
\end{proof}

\subsection{The Fibonacci image as a Fischer cover}

The overlap graph $\mathcal G_m$ gives a concrete presentation of $Y_m$. Above
threshold this presentation is not merely convenient; it is the intrinsic
minimal right-resolving presentation.

We use the following standard characterization of the right Fischer cover of
an irreducible synchronized sofic shift: it is the unique essential
irreducible right-resolving follower-separated presentation whose vertices are
the follower sets of intrinsically synchronizing words. This is the
Fischer--Krieger cover construction in its right-resolving form; see
\cite{Fischer1975,Krieger1984,Krieger1987,LindMarcus1995,BoyleKitchensMarcus1985,Johansen2011}.

For a vertex $u\in V_m=\{0,1\}^{m-1}$, let
\[
\mathcal F^+(u):=
\{w\in X_m^*:\text{ there exists a path in }\mathcal G_m
\text{ beginning at }u\text{ with label }w\}
\]
be its right follower language.

\begin{lemma}[Fischer-cover criterion]\label{lem:fischer-cover-criterion}
Let $X$ be an irreducible sofic shift and let $G$ be a finite labeled graph
presenting $X$. Assume that:
\begin{enumerate}[label=(\roman*),leftmargin=*,itemsep=2pt]
  \item $G$ is essential and irreducible;
  \item $G$ is right-resolving;
  \item $G$ is follower-separated, meaning distinct vertices have distinct
  right follower languages in $X$;
  \item every vertex $v$ of $G$ is the terminal vertex of a path whose label
  $w_v$ is intrinsically synchronizing for $X$ and labels a unique path in
  $G$.
\end{enumerate}
Then $G$ is the right Fischer cover of $X$. In particular, every
right-resolving presentation of $X$ has at least as many vertices as $G$.
\end{lemma}

\begin{proof}
For each vertex $v$, fix a word $w_v$ as in hypothesis (iv).  We first show
that the right follower set of the word $w_v$ in the shift is exactly the
right follower language of the terminal vertex:
\[
F_X(w_v)=\mathcal F_G^+(v).
\]
If $z\in F_X(w_v)$, then $w_vz$ occurs as the label of a path in the
presentation $G$.  The prefix labeled $w_v$ must be the unique path labeled
$w_v$, and hence it ends at $v$; the suffix path labeled $z$ starts at $v$.
Thus $z\in\mathcal F_G^+(v)$.  The reverse inclusion is immediate because any
path starting at $v$ may be appended to the fixed path labeled $w_v$ ending at
$v$.  Therefore every vertex of $G$ is precisely the follower set of an
intrinsically synchronizing word.

By the standard Fischer-cover characterization recalled above, an essential
irreducible right-resolving follower-separated presentation whose vertices are
precisely follower sets of intrinsically synchronizing words is the right
Fischer cover.  The displayed identity verifies that hypothesis for $G$:
under the map $v\mapsto F_X(w_v)$, each vertex is one of the canonical
synchronizing follower-set vertices, and follower-separatedness makes the
identification injective.

For completeness, we spell out why the labeled edges agree under this vertex
identification.  If $v\xrightarrow{a}u$ is an edge of $G$, then right
resolvingness and the equality
$F_X(w_v)=\mathcal F_G^+(v)$ give
\[
\{z:az\in F_X(w_v)\}=\mathcal F_G^+(u).
\]
Indeed, any path labeled $az$ from $v$ begins with the unique outgoing
$a$-edge from $v$ and then reads $z$ from $u$, and the converse is immediate
by prepending that edge.  This is exactly the canonical follower-set
transition from $F_X(w_v)$ under the symbol $a$. Conversely, every canonical
transition arises from an admissible symbol after the corresponding vertex,
hence from an edge of $G$ because $G$ presents $X$ and is right-resolving.
Thus the identification preserves and reflects all labeled edges, so $G$ is
isomorphic to the right Fischer cover of $X$.  The lower bound for
right-resolving presentations is the usual minimality of the right Fischer
cover.
\end{proof}

\begin{theorem}[Fischer cover of the stabilized image]
\label{thm:fischer-cover-stabilized-image}
\leanpending{paper_fibonacci_stabilization_fischer_cover_stabilized_image}
Assume $m\ge3$.  Then:
\begin{enumerate}[label=(\roman*),leftmargin=*,itemsep=2pt]
  \item Every word $w\in\mathcal L_m(Y_m)$ labels exactly one path of length
  $m$ in the overlap graph $\mathcal G_m$. Consequently every word in
  $\mathcal L_m(Y_m)$ is intrinsically synchronizing for $Y_m$.
  \item The follower languages $\mathcal F^+(u)$, $u\in V_m$, are pairwise
  distinct.  Hence $\mathcal G_m$ is follower-separated.
  \item The labeled graph $\mathcal G_m$ is the right Fischer cover of $Y_m$.
  In particular, every right-resolving presentation of $Y_m$ has at least
  $2^{m-1}$ vertices.
\end{enumerate}
\end{theorem}

\begin{proof}
A path of length $m$ in $\mathcal G_m$ is exactly a raw binary block
\[
b_0b_1\cdots b_{2m-2}\in\{0,1\}^{2m-1},
\]
and its label is
\[
\Phi_{m,m}(b_0b_1\cdots b_{2m-2}).
\]
By Theorem~\ref{thm:block-invertibility}, the map $\Phi_{m,m}$ is injective.
Thus each legal word $w\in\mathcal L_m(Y_m)$ has a unique preimage raw
block and hence labels exactly one path in $\mathcal G_m$.

The same uniqueness gives intrinsic synchronization. If $uw$ and $wv$ are
admissible words in $Y_m$, then in any path labeled by $uw$ the subpath labeled
by $w$ is the unique path labeled by $w$, and similarly for any path labeled by
$wv$.  The first path therefore reaches the terminal vertex of this unique
$w$-path, and the second path leaves that same terminal vertex after reading
$w$.  Concatenation gives an admissible path labeled by $uwv$.  Hence $w$ is
intrinsically synchronizing.  This proves (i).

Let $u,v\in V_m$ with $u\ne v$.  Choose any path of length $m$ beginning at
$u$, and let $w$ be its label.  If $w$ also belonged to $\mathcal F^+(v)$, then
$w$ would label a second length-$m$ path, now beginning at $v$.  This contradicts
(i), since the initial vertices are different.  Thus
$\mathcal F^+(u)\ne\mathcal F^+(v)$, proving (ii).

It remains to verify the four hypotheses of
Lemma~\ref{lem:fischer-cover-criterion}. The graph $\mathcal G_m$ is finite by
definition. The underlying de Bruijn graph on
$\{0,1\}^{m-1}$ is essential: every vertex
$u=(u_1,\ldots,u_{m-1})$ lies on a bi-infinite path, obtained for example by
using the periodic raw sequence with period $u$ when $u$ is nonempty. It is
irreducible because, from any vertex $u$ to any vertex $v$, one follows the
raw word obtained by appending the symbols of $v$ after $u$, which gives a
path whose terminal vertex is $v$. Hence the presented shift $Y_m$ is
irreducible and $\mathcal G_m$ is an essential irreducible presentation.
This verifies hypothesis (i). Theorem~\ref{thm:finite-state-lift} proves that
the labeling is right-resolving, which is hypothesis (ii), and statement (ii)
above proves follower separation, which is hypothesis (iii).

Finally, every vertex is the terminal vertex of some length-$m$ path: append
the symbols of that vertex after any initial bit. By (i), the label of every
length-$m$ path labels a unique path and is intrinsically synchronizing. Thus
every vertex is reached by a unique-path intrinsically synchronizing word,
verifying hypothesis (iv).
Lemma~\ref{lem:fischer-cover-criterion} therefore identifies $\mathcal G_m$
with the right Fischer cover of $Y_m$ and gives the lower bound for all
right-resolving presentations. The vertex set has cardinality
$|V_m|=2^{m-1}$.
\end{proof}

\begin{theorem}[Threshold--conjugacy--cover rigidity package]
\label{thm:threshold-conjugacy-cover-package}
The Fibonacci stabilized-window system has the following complete structural
classification.
\begin{enumerate}[label=(\roman*),leftmargin=*,itemsep=2pt]
  \item The only failure of path-space injectivity in the Fibonacci family
  $m\ge2$ occurs at $m=2$, where
  $\Phi_2^{-1}((\ldots,00,00,\ldots))=\{0^{\ZZ},1^{\ZZ}\}$ and every other
  fiber is a singleton.
  \item For every $m\ge3$, $\Phi_m$ is a finite-memory topological conjugacy
  from the full two-shift onto the subshift of finite type $Y_m$, with inverse
  given by the local decoder of Corollary~\ref{cor:explicit-decoder}.  Under
  this conjugacy invariant measures, entropy and periodic-point data are
  transported exactly.
  \item For every $m\ge3$, the overlap graph $\mathcal G_m$ is the right
  Fischer cover of $Y_m$, and every right-resolving presentation of $Y_m$ has
  at least $2^{m-1}$ states.
  \item In the constant-type metallic family, the golden parameter $A=1$ is
  the unique value for which two-window normalization has a nontrivial
  path-space branch; for every integer $A\ge2$ the metallic two-window map is
  already a sliding-block conjugacy onto its image.
\end{enumerate}
\end{theorem}

\begin{proof}
The branch statement in (i) is Theorem~\ref{thm:m2-branch-locus}; the final
sentence of Theorem~\ref{thm:phase-transition} records that no other
Fibonacci window length produces an intermediate degeneration.  Statement
(ii) combines Theorem~\ref{thm:two-sided-conjugacy}, which gives the
topological conjugacy, SFT closure and local inverse, with
Corollary~\ref{cor:explicit-decoder} for the explicit decoder.  The exact
transport of invariant measures and periodic data is
Proposition~\ref{prop:conjugacy-consequences}, and the entropy identity is
Proposition~\ref{prop:measure-entropy-preservation} together with
Corollary~\ref{cor:entropy-preservation}.  Statement (iii) is precisely
Theorem~\ref{thm:fischer-cover-stabilized-image}.  Finally, (iv) is
Theorem~\ref{thm:metallic-two-window-rigidity}: its proof separates the
alternatives $\{0\}$ and $\{A-1,A\}$ for $A\ge2$, while for $A=1$ these
alternatives collapse and the remaining ambiguity is the branch locus already
classified in (i).  These results exhaust the stated Fibonacci and
constant-type metallic cases.
\end{proof}
