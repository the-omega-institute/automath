\section{The primitive-divisor alternative}
\label{sec:primitive-alternative}

For $n\geq13$, define the standard primitive part
\[
 U_n^{\rm prim}:=
 \prod_{\substack{p\mid F_n\\p\nmid5F_1\cdots F_{n-1}}}
 p^{\nu_p(F_n)}.
\]

\begin{lemma}[Primitive part of the fibotomic factor]
\label{lem:classical-primitive-part}
For every $n\geq13$, there is an integer $c_n$ such that
\begin{equation}
\label{eq:primitive-part-factorization}
 \Psi_n(1)=c_nU_n^{\rm prim},
 \qquad
 c_n=1\quad\text{or}\quad c_n\text{ is a prime divisor of }n.
\end{equation}
In particular,
\begin{equation}
\label{eq:primitive-part-ratio}
 1\leq\frac{\Psi_n(1)}{U_n^{\rm prim}}\leq n
\end{equation}
and
\begin{equation}
\label{eq:primitive-part-log-mass}
 \log U_n^{\rm prim}=\phi(n)\log\varphi+O(\log n).
\end{equation}
\end{lemma}

\begin{proof}
The Lucas cyclotomic factor denoted $\phi_n$ by Granville is
\[
 \prod_{d\mid n}F_d^{\mu(n/d)}=\Psi_n(1)
\]
in the Fibonacci specialization.  Granville's cyclotomic-factor theorem
states that a characteristic prime of $F_n$ occurs in this factor with the
same exponent as in $F_n$.  Every noncharacteristic prime $p$ in the factor
occurs only to the first power and satisfies
$n/\alpha(p)=p^a$ for some $a\geq1$; apart from the indices $6$ and $12$,
there is at most one such prime
\cite[Corollaries~3 and~4]{Granville2012PrimitivePrimeFactors}.

At $n\geq13$, the characteristic primes are exactly the primes dividing
$F_n$ but no earlier Fibonacci number.  None is $5$, since
$\alpha(5)=5$, so these are precisely the primitive primes in the
convention above.  Any remaining prime $p$ in $\Psi_n(1)$ divides $n$
because $n/\alpha(p)=p^a$; this includes the possible discriminant prime
$5$ at an index divisible by a higher power of $5$.  Granville's theorem
therefore gives \eqref{eq:primitive-part-factorization}, including all
primitive multiplicities, and then \eqref{eq:primitive-part-ratio}.
Finally, \eqref{eq:fibotomic-binet-error} and $1\leq c_n\leq n$ give
\eqref{eq:primitive-part-log-mass}.
\end{proof}

\begin{theorem}[Large primitive divisor--Fibonacci--Wieferich alternative]
\label{thm:large-primitive-wieferich-alternative}
For every real $\varepsilon>0$, there is an integer
$N_\varepsilon\geq13$ such that, for every $n\geq N_\varepsilon$, at least
one of the following holds:
\begin{enumerate}[label=\textup{(\roman*)}]
\item $F_n$ has a primitive prime divisor $p\geq n^{2-\varepsilon}$;
\item $F_n$ has a primitive prime divisor $q$ such that $q^2\mid F_n$.
\end{enumerate}
In the second alternative, $\alpha(q)=n$ and $q$ is a Wall--Sun--Sun,
equivalently Fibonacci--Wieferich, prime.
One admissible explicit choice is
\begin{equation}
\label{eq:explicit-alternative-threshold}
 N_\varepsilon=
 \left\lceil
  \exp\!\left(\frac{100}{\eta}\log\frac{100}{\eta}\right)
 \right\rceil,
 \qquad \eta:=\min\{\varepsilon,1\}.
\end{equation}
\end{theorem}

\begin{proof}
Fix $n\geq13$ and suppose alternative \textup{(ii)} fails.  Every
primitive prime then occurs in $F_n$ to exponent one.  By
Theorem~\ref{thm:exact-rank-existence}, at least one such prime exists, and
therefore
\begin{equation}
\label{eq:primitive-part-max-bound}
 \log U_n^{\rm prim}
 \leq a(n)\log P_{\rm prim}(F_n).
\end{equation}
For $n\geq13$, the exact-rank primes counted by $a(n)$ are precisely the
primitive prime divisors in the convention of the theorem.  By
\eqref{eq:pointwise-exact-rank-bound} and
\eqref{eq:primitive-part-log-mass},
\[
 a(n)\leq
 \left(\frac{\log\varphi}{2}+o(1)\right)
 \frac{\phi(n)}{\log n},
 \qquad
 \log U_n^{\rm prim}
 =(\log\varphi+o(1))\phi(n).
\]
Here $O(\log n)=o(\phi(n))$ follows from the standard lower bound
$\phi(n)\gg n/\log\log(3n)$.  Substitution in
\eqref{eq:primitive-part-max-bound} gives, uniformly as $n\to\infty$,
\[
 \log P_{\rm prim}(F_n)\geq(2-o(1))\log n.
\]
Thus, whenever alternative \textup{(ii)} fails, alternative \textup{(i)}
holds for all sufficiently large $n$, with the threshold depending only on
$\varepsilon$.

We next verify the explicit choice
\eqref{eq:explicit-alternative-threshold}.  Write
$\lambda=\log\varphi$, $s=\log n$, and $k=a(n)$.  If
$r=\varphi^{-2}$, then
\[
 \left|\log\bigl(1-(-r)^j\bigr)\right|
 \leq\frac{r^j}{1-r}
 \quad(j\geq1).
\]
Since $1-r=\varphi^{-1}$, summation gives
\begin{equation}
\label{eq:explicit-binet-constant}
 C_F\leq\frac{r}{(1-r)^2}=1.
\end{equation}
Consequently Theorem~\ref{thm:fibotomic-rank-entropy} and
Lemma~\ref{lem:classical-primitive-part} give
\begin{equation}
\label{eq:explicit-count-and-mass}
 \mathcal H_n(k)\leq M:=\lambda\phi(n)+1,
 \qquad
 \log U_n^{\rm prim}\geq A:=\lambda\phi(n)-1-s.
\end{equation}

For $k\geq2$, put $m=\lfloor k/2\rfloor$.  The elementary bound
$\log(m!)\geq m\log m-m$ and the inequality $m\geq k/3$ imply
\begin{equation}
\label{eq:explicit-entropy-lower-bound}
 \mathcal H_n(k)
 \geq k\left(\log\frac{2nk}{9}-1\right).
\end{equation}
Indeed, the terms following $k\log(2n/3)$ in the definition of
$\mathcal H_n(k)$ are at least $k\log m-k$; this also covers odd $k$
because its unpaired term is $\log(m+1)\geq\log m$.

We use the explicit totient estimate
\begin{equation}
\label{eq:explicit-totient-bound}
 \phi(n)>\frac{n}{B(n)},
 \qquad
 B(n):=e^\gamma\log\log n+\frac3{\log\log n},
\end{equation}
valid for $n\geq3$ by the sharper bound of Rosser and Schoenfeld
\cite[Theorem~15]{RosserSchoenfeld1962ApproximateFormulas}.
Let $0<\eta\leq1$, put $t=100/\eta$, and suppose
\[
 s\geq S_\eta:=t\log t.
\]
Then
\begin{equation}
\label{eq:explicit-elementary-inequalities}
 \begin{aligned}
 B(n)&\leq s,
 &4\log s+3&\leq\frac{\eta s}{2},\\
 \frac{s(s+2)}{\lambda e^s}&\leq\frac\eta8,
 &s(s+1)e^{-s}&\leq\frac\lambda2,
 &\frac{\lambda s}{2}&\geq2-\eta.
 \end{aligned}
\end{equation}
Here are the details.  Since $t\geq100$, at $s=t\log t$ one has
\[
 4\log s+3
 =4(\log t+\log\log t)+3<50\log t=\frac{\eta s}{2}.
\]
The difference between the right and left sides is increasing thereafter,
because $\eta/2-4/s>0$.  Also $s>460$, so
$B(n)<2\log s<s$, using $e^\gamma<1.782$.  The inequality
$s>4\log s$ gives $e^s>s^4$, and hence
\[
 s(s+2)e^{-s}<\frac2{s^2}\leq\frac{\eta}{5000}
 <\frac{\lambda\eta}{8},
 \qquad
 s(s+1)e^{-s}<\frac2{s^2}<\frac\lambda2,
\]
where $\lambda>0.48$.  Finally, $\lambda s/2>110>2-\eta$.
This proves \eqref{eq:explicit-elementary-inequalities}.  The constant
$100$ was chosen only to make these estimates immediate; no optimization
is intended.

Suppose again that alternative \textup{(ii)} fails, so
$\log P_{\rm prim}(F_n)\geq A/k$.  If
$k\leq\phi(n)/s^2$, then
\eqref{eq:explicit-totient-bound} and
\eqref{eq:explicit-elementary-inequalities} give
\[
 \log P_{\rm prim}(F_n)
 \geq\left(\lambda-\frac{s+1}{\phi(n)}\right)s^2
 \geq\frac{\lambda s^2}{2}
 \geq(2-\eta)s.
\]
If $k>\phi(n)/s^2$, then
\[
 \log k\geq s-3\log s.
\]
Equations~\eqref{eq:explicit-count-and-mass} and
\eqref{eq:explicit-entropy-lower-bound} therefore give
\[
 k\leq\frac{M}{D},
 \qquad
 D:=2s-4\log s-3\geq\left(2-\frac\eta2\right)s.
\]
Moreover,
\[
 \frac AM
 =1-\frac{s+2}{\lambda\phi(n)+1}
 \geq1-\frac{s(s+2)}{\lambda e^s}
 \geq1-\frac\eta8.
\]
It follows that
\[
 \log P_{\rm prim}(F_n)
 \geq\frac{AD}{M}
 \geq\left(1-\frac\eta8\right)
       \left(2-\frac\eta2\right)s
 \geq(2-\eta)s
 \geq(2-\varepsilon)\log n.
\]
Thus \eqref{eq:explicit-alternative-threshold} is valid.

It remains to identify the exceptional alternative.  A primitive $q$ has
$q>5$ and $\alpha(q)=n$.  Proposition~\ref{prop:rank-lifting} shows that
\[
 q^2\mid F_n=F_{\alpha(q)}
 \quad\Longleftrightarrow\quad
 q^2\mid F_{q-(5/q)}.
\]
This is exactly the Wall--Sun--Sun, or Fibonacci--Wieferich, condition.
This proves the alternative.
\end{proof}

The exceptional branch still forces a large primitive factor when the full
primary component, rather than only its prime base, is measured.  This
removes the lifting hypothesis from the resulting prime-power bound.

\begin{theorem}[Unconditional primitive primary component]
\label{thm:primitive-primary-component}
For $n\geq13$, define
\[
 Q_{\rm prim}(F_n):=
 \max_{\substack{p\mid F_n\\p\nmid5F_1\cdots F_{n-1}}}
 p^{\nu_p(F_n)}.
\]
Then, for every real $\varepsilon>0$, one has
\[
 Q_{\rm prim}(F_n)\geq n^{2-\varepsilon}
\]
for all sufficiently large $n$.  Equivalently,
\[
 \liminf_{n\to\infty}
 \frac{\log Q_{\rm prim}(F_n)}{\log n}\geq2.
\]

If there are only finitely many Wall--Sun--Sun primes and
$P_{\rm prim}(F_n)$ denotes the largest primitive prime divisor of $F_n$,
then
\[
 \liminf_{n\to\infty}
 \frac{\log P_{\rm prim}(F_n)}{\log n}\geq2.
\]
\end{theorem}

\begin{proof}
Fix $\varepsilon>0$ and take $n$ sufficiently large for
Theorem~\ref{thm:large-primitive-wieferich-alternative} and for
$(n-1)^2\geq n^{2-\varepsilon}$.  In the first branch of that theorem, its
primitive prime $p$ gives
\[
 Q_{\rm prim}(F_n)\geq p^{\nu_p(F_n)}\geq p
 \geq n^{2-\varepsilon}.
\]
In the second branch, Proposition~\ref{prop:rank-lifting}\textup{(ii)} gives
\[
 n=\alpha(q)\mid q-(5/q),
\]
and therefore $q\geq n-1$.  Since $q^2\mid F_n$, this yields
\[
 Q_{\rm prim}(F_n)\geq q^{\nu_q(F_n)}
 \geq q^2\geq(n-1)^2\geq n^{2-\varepsilon}.
\]
The first liminf follows because $\varepsilon$ is arbitrary.

Now suppose that the set $\mathcal W$ of Wall--Sun--Sun primes is finite.
Every prime in the second branch at index $n$ belongs to $\mathcal W$ and
has rank exactly $n$.  Hence that branch is impossible once
\[
 n>\max\{\alpha(q):q\in\mathcal W\},
\]
where the maximum is interpreted as $0$ if $\mathcal W$ is empty.  The
first branch consequently holds for every sufficiently large $n$, for each
fixed $\varepsilon>0$, and the asserted liminf for
$P_{\rm prim}(F_n)$ follows.
\end{proof}
