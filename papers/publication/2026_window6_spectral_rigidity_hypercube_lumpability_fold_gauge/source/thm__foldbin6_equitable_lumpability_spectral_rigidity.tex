\begin{proof}

\begin{lemma}[Characteristic polynomial and numerical root certificate]\label{lem:foldbin6-characteristic-certificate}
For the exact matrix of Proposition \ref{prop:foldbin6-exact-pushforward-matrix},

\]
Consequently \(P_6\) has eigenvalues whose ten-decimal certificates are
\[
0.4841207858,
\qquad

\]
and the second has absolute value \(0.6030939754\) to ten decimals.
\end{lemma}
\begin{proof}
The characteristic polynomial is obtained from the displayed rational matrix by exact fraction-free elimination over \(\QQ[t]\).  Equivalently, after multiplying \(q_6\) by
\[
L_6=3234734993627557134336
\]
the displayed identity becomes a polynomial identity in \(\ZZ[t]\) for \(L_6\det(tI-P_6)/(t-1)\), and so it is verified by exact integer arithmetic.  The coefficient stream is the source-level stream \(\texttt{Q6=<coefficients>}\) recorded in Proposition \ref{prop:audited-computation-reproducibility-certificate}\textup{(iv)}.

\[
\begin{aligned}
&Q_6(4841207858/10^{10})<0,
&&Q_6(4841207859/10^{10})>0,\\
&Q_6(-6030939755/10^{10})>0,
&&Q_6(-6030939754/10^{10})<0.
\end{aligned}

\[
1,
\quad 0,
\quad 20,
\quad 19.
\]
Sturm's theorem therefore gives one real zero in each of the two rational intervals.  The signs, variation counts, and Sturm-stream hash are exactly those recorded in Proposition \ref{prop:audited-computation-reproducibility-certificate}\textup{(v)}.  Since \(Q_6\ne0\) is a scalar multiple of the nontrivial factor of \(\chi_{P_6}\), these two roots are eigenvalues of \(P_6\).  The intervals have length \(10^{-10}\), which is precisely the advertised ten-decimal certificate. \qedhere
\end{proof}

\begin{lemma}[Exact interval certificate for the audited window-\texorpdfstring{$6$}{6} eigenvalues]\label{lem:foldbin6-eigenvalue-interval-certificate}
For the audited window-\(6\) pushforward kernel \(P_6\), there are spectral values \(\lambda_+,\lambda_-\in\Spec(P_6)\) satisfying
\[
0.4841207858<\lambda_+<0.4841207859,
\qquad
-0.6030939755<\lambda_-<-0.6030939754.
\]
In particular, \(\lambda_+\), \(\lambda_-\), and \(|\lambda_-|\) are not elements of
\[
\left\{\,1,\ \frac23,\ \frac13,\ 0,\ -\frac13,\ -\frac23,\ -1\,\right\}.
\]
\end{lemma}
\begin{proof}
The existence and the displayed intervals are exactly Lemma \ref{lem:foldbin6-characteristic-certificate}.  The positive interval lies strictly between \(1/3\) and \(2/3\), while the negative interval lies strictly between \(-2/3\) and \(-1/3\).  Neither interval contains an endpoint of the window-\(6\) hypercube grid.  Hence the certified eigenvalues are off the window-\(6\) grid, and so is the absolute value \(|\lambda_-|\). \qedhere
\end{proof}

\begin{corollary}[Window-\(6\) is spectrally nonlumpable]\label{cor:terminal-window6-nonlumpable-by-spectrum}
The audited window-\(6\) pushforward kernel \(P_6\) associated with \(\Fold_6^{\mathrm{bin}}\) is not the transition kernel of an equitable quotient chain on \(Q_6\).  Equivalently, the window-\(6\) certificate is not hidden-state-free strongly lumpable.
\end{corollary}

\begin{proof}
By Theorem \ref{thm:window6-strong-lumpability-spectral-rigidity}, every equitable quotient of \(Q_6\) has spectrum contained in
\(
\{1-2k/6:0\le k\le 6\}
\).
Lemma \ref{lem:foldbin6-eigenvalue-interval-certificate} supplies two spectral values of the exact audited pushforward matrix \(P_6\) outside this set.  Therefore \(P_6\) cannot be the transition matrix of an equitable quotient of the hypercube walk. \qedhere
\end{proof}
