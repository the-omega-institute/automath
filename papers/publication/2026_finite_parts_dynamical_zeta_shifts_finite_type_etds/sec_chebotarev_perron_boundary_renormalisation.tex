\subsubsection{Perron-boundary Adams--M\"obius renormalisation}
\label{subsubsec:perron-boundary-adams-mobius-renormalisation}

This subsection contains the Perron-boundary theorem used in the rest of the
paper.  It does not assume the strict twisted gap: it keeps all peripheral
phases \(\lambda\zeta\), subtracts only the logarithmic part coming from
\(\zeta=1\), and evaluates the constants by reduced determinant finite parts.
The proof is split into four labelled components, but these components are
not separate theorem spines; they are the primitive-channel, class-Mertens,
determinant, and product steps of the single boundary theorem.

\begin{definition}[Peripheral Peter--Weyl data]
  \label{def:peripheral-peter-weyl-data}
  Assume \(A\) is primitive with Perron root \(\lambda>1\).  For
  \(\pi\in\Irr(G)\), define
  \[
    \Gamma_\pi:=
    \{\zeta\in\TT:\lambda\zeta\in\spec(A_\pi)\}.
  \]
  For \(\zeta\in\Gamma_\pi\), let
  \(m_{\pi,\zeta}\) be the algebraic multiplicity of
  \(\lambda\zeta\) in \(A_\pi\); equivalently,
  \(m_{\pi,\zeta}\) is the rank, or trace, of the Riesz spectral
  projection onto the generalized \(\lambda\zeta\)-eigenspace.  The
  optional semisimplicity certificate below further identifies this
  generalized eigenspace with the ordinary eigenspace.  Put
  \[
    s_\pi:=m_{\pi,1},
  \]
  with \(s_\pi=0\) if \(1\notin\Gamma_\pi\).  Let
  \[
    \Lambda_\perp:=
    \max\{|\alpha|:\alpha\in\spec(A_\pi),\ \pi\in\Irr(G),\
      |\alpha|<\lambda\},
  \]
  after deleting all eigenvalues \(\lambda\zeta\) with
  \(\zeta\in\Gamma_\pi\).  If the remaining set is empty, set
  \(\Lambda_\perp=0\).  Finally set
  \[
    \Lambda_\partial:=\max\{\Lambda_\perp,\lambda^{1/2}\}.
  \]
  The term \(\lambda^{1/2}\) is the same Adams--M\"obius divisor-loss scale
  that appears when the \(m>1\) divisors are bounded in
  \Cref{thm:primitive-peter-weyl-trace}.

  The spectral entries \(\Gamma_\pi\) and \(m_{\pi,\zeta}\) are not by
  themselves a branch convention for boundary logarithms.  Write
  \(P_\pi(z)=\det(I-zA_\pi)\), let \(\Log_0\) denote the determinant-logarithm
  germ normalized to be \(0\) at \(z=0\), and put
  \(s_\pi=m_{\pi,1}\), as above.  The open-disc primitive coefficients are
  determined by the smaller datum
  \[
    (G,\operatorname{Pow}_m,\{P_\pi(z)=\det(I-zA_\pi)\}_\pi).
  \]
  This means the coefficients of the primitive power series inside
  \(|z|<\lambda^{-1}\); it is not a branch-normalized finite-part datum at
  \(z=\lambda^{-1}\).
  The Perron-boundary constants require the following larger but still finite
  branch-normalized recipe:
  \begin{equation}\label{eq:perron-boundary-renormalized-data-local}
  \begin{aligned}
    D_{\partial}^{\mathrm{ren}}
    :=
    \bigl(&G,\operatorname{Pow}_m,\{P_\pi\}_{\pi\in\Irr(G)},\lambda,\\
      &\{\,P_\pi(t/\lambda)=
        \prod_{\zeta\in\Gamma_\pi}(1-t\zeta)^{m_{\pi,\zeta}}
        Q_{\partial,\pi}(t)\,\}_{\pi\in\Irr(G)},\\
      &\{\Gamma_\pi\}_{\pi\in\Irr(G)},
        \{m_{\pi,\zeta}\}_{\zeta\in\Gamma_\pi,\ \pi\in\Irr(G)},
        \{s_\pi\}_{\pi\in\Irr(G)},\\
      &\Log_0,n_\partial\bigr).
  \end{aligned}
  \end{equation}
  Here \(\operatorname{Pow}_m\) denotes the finite conjugacy-class power table
  used to compute the Adams coefficients \(a_{\rho,\pi}^{(m)}\).  The displayed
  peripheral factorisations record the finite factors \(Q_{\partial,\pi}\)
  with \(Q_{\partial,\pi}(1)\ne0\); they are determined by \(P_\pi\),
  \(\lambda\), and the listed peripheral roots and multiplicities, but are
  named to make the boundary evaluation unambiguous.  The branch rule is:
  continue \(\Log_0\) from \(0\) along the Abel path \(0\le t<1\) for \(k=1\),
  use the ordinary interior value at \(\lambda^{-k}\) for \(k\ge2\), and use
  \(n_\partial\) for the positive scalar/product logarithm normalization.

  The tuple \(D_{\partial}^{\mathrm{ren}}\) is input data; the determinant
  finite parts and product constants are outputs computed from it.  Once this
  finite datum is fixed, define for every \(\pi\in\Irr(G)\) and \(k\ge1\) the
  derived branch-normalized determinant layer
  \begin{equation}\label{eq:perron-boundary-renormalized-b-data}
    B_{k,\pi}^{\mathrm{ren}}
    =\mathcal B_{k,\pi}^{\mathrm{ren}}:=
    \begin{cases}
      \displaystyle
      \lim_{t\uparrow1}
      \left(
        \Log_0\det(I-t\lambda^{-1}A_\pi)^{-1}
        +s_\pi\log(1-t)
      \right), & k=1,\\[1.2em]
      \displaystyle
      \Log_0\det(I-\lambda^{-k}A_\pi)^{-1}, & k\ge2.
    \end{cases}
  \end{equation}
  Equivalently, from the factorisation component
  \[
    P_\pi(t/\lambda)
    =
    \prod_{\zeta\in\Gamma_\pi}(1-t\zeta)^{m_{\pi,\zeta}}
      Q_{\partial,\pi}(t),
    \qquad Q_{\partial,\pi}(1)\ne0,
  \]
  the \(k=1\) entry is
  \begin{equation}\label{eq:perron-boundary-renormalized-b1-factor}
    B_{1,\pi}^{\mathrm{ren}}
    =
    -\sum_{\substack{\zeta\in\Gamma_\pi\\\zeta\ne1}}
      m_{\pi,\zeta}\Log_0(1-\zeta)
    -\Log_0 Q_{\partial,\pi}(1).
  \end{equation}
  Consequently the reduced fixed-label boundary coordinate is the derived
  quantity
  \begin{equation}\label{eq:perron-boundary-fren-derived-data}
    F_\varrho^{\mathrm{ren}}
    =
    \sum_{r,m\ge1}\frac{\mu(m)}{rm}
      \sum_{\pi\in\Irr(G)}
        a_{\varrho,\pi}^{(m)}B_{rm,\pi}^{\mathrm{ren}}.
  \end{equation}
  For \(k=1\), \(\Log_0\) is continued only along the Abel path
  \(0\le t<1\); for \(k\ge2\), the evaluation is an ordinary interior-disc
  value.  Thus later ``finite determinant'' statements refer to the finite
  determinant polynomials together with this finite branch-normalized recipe;
  the values \(B_{k,\pi}^{\mathrm{ren}}\) are outputs of that recipe, not
  independent inputs.
\end{definition}

\begin{definition}[Boundary logarithm convention]
  \label{def:perron-boundary-logarithm-convention}
  Fix the peripheral data of
  \Cref{def:peripheral-peter-weyl-data} and write
  \(P_\pi(z)=\det(I-zA_\pi)\).  All determinant logarithms below are the
  normalized germs \(\Log_0\) at \(z=0\), continued only along the indicated
  Abel path or evaluated at an interior point.  For \(k\ge1\) and
  \(\pi\in\Irr(G)\), the boundary logarithm is the value
  \(\mathcal B_{k,\pi}^{\mathrm{ren}}\) defined in
  \eqref{eq:perron-boundary-renormalized-b-data}.
  Thus every \(k=1\) Perron-boundary block is read as a reduced Abel finite
  part: exactly the \(s_\pi=m_{\pi,1}\) copies of the non-oscillatory factor
  \(1-t\) are cancelled, while the remaining peripheral factors
  \(1-t\zeta\), \(\zeta\ne1\), are evaluated on the Abel branch reached from
  the origin.  Every \(k\ge2\) term is an ordinary interior-disc logarithm.
  The positive scalar/product logarithm normalization used when exponentiating
  product constants is the separate datum \(n_\partial\) in
  \(D_{\partial}^{\mathrm{ren}}\).

  In the strict-gap specialization, where
  \(s_{\mathbf1}=1\), \(s_\pi=0\) for \(\pi\ne\mathbf1\), and no non-trivial
  block has a Perron-boundary eigenvalue, the singular scalar boundary value
  \(\mathcal B_{1,\mathbf1}^{\mathrm{ren}}\) is not used in non-trivial
  fixed-label coordinates.  For those coordinates we write
  \(\mathcal L_{k,\pi}:=\mathcal B_{k,\pi}^{\mathrm{ren}}\) except that the
  triple \((k,\pi)=(1,\mathbf1)\), equivalently
  \((r,m,\pi)=(1,1,\mathbf1)\), is deleted before any endpoint value is
  chosen.  Non-trivial \(k=1\) terms are Abel boundary values and all
  \(k\ge2\) terms, including scalar terms created by Adams powers, remain
  ordinary interior determinant logarithms.
\end{definition}

\begin{lemma}[Finite-cover peripheral expansion and optional certificate]
  \label{lem:finite-cover-peripheral-expansion}
  Let \(\widetilde A\) be the finite skew-product adjacency matrix attached
  to \(\tau\).  Suppose the peripheral algebraic data of
  \Cref{def:peripheral-peter-weyl-data} have been computed: for each
  \(\pi\), the list \(\Gamma_\pi\) consists of the roots
  \(\lambda\zeta\) of \(\det(xI-A_\pi)\) on the circle \(|x|=\lambda\),
  \(m_{\pi,\zeta}\) is the algebraic multiplicity of that root, and
  \(\Lambda_\perp<\lambda\) is the maximum modulus of the remaining roots
  (or \(0\) if there are none).  These algebraic data are obtained by the
  finite tests
  \begin{enumerate}[label=\textup{(\roman*)},leftmargin=*]
    \item compute the roots \(\alpha\) of
      \(\det(xI-A_\pi)\) with \(|\alpha|=\lambda\), write
      \(\alpha=\lambda\zeta\), and record
      \(\Gamma_\pi\) and the algebraic multiplicity \(m_{\pi,\zeta}\);
    \item set \(\Lambda_\perp\) to the maximum modulus of all remaining
    roots, or to \(0\) if there are none, and check
    \(\Lambda_\perp<\lambda\).
  \end{enumerate}
  These tests use only determinants and finite algebraic comparison of
  moduli; no limiting orbit counts enter.
  If one also wants the stronger semisimplicity certificate
  \(\mathfrak s_\partial\), one adds the finite rank test
  \[
    \dim\ker(A_\pi-\lambda\zeta I)=m_{\pi,\zeta}
  \]
  for every recorded peripheral root, equivalently that the local factor of
  the minimal polynomial at \(x=\lambda\zeta\) is \(x-\lambda\zeta\).  This
  additional certificate may be checked from the finite skew-product matrix
  \(\widetilde A\) after its Frobenius normal form has no nilpotent coupling
  between same-radius peripheral components.  The determinant and trace
  identities below use the algebraic multiplicities recorded above.

  Under the peripheral algebraic multiplicity data above, for every
  \(\pi\in\Irr(G)\),
  \begin{equation}\label{eq:finite-cover-peripheral-trace-expansion}
    \Tr(A_\pi^n)=
    \lambda^n\sum_{\zeta\in\Gamma_\pi}m_{\pi,\zeta}\zeta^n
    +O(\Lambda_\perp^n),
    \qquad n\ge1,
  \end{equation}
  with a constant depending only on the finite determinant model.  In
  particular all peripheral contributions are finite root-of-unity phases
  inherited from the finite cover, and the trace remainder is strictly below
  modulus \(\lambda\).
\end{lemma}

\begin{proof}
  By \Cref{prop:kernel-peter-weyl-block-diagonalisation},
  \(\widetilde A\) is similar to
  \(\bigoplus_{\pi\in\Irr(G)}A_\pi^{\oplus\dim\pi}\).  Thus the peripheral
  eigenvalues of the twisted blocks are exactly the Peter--Weyl pieces of
  the peripheral spectrum of the finite non-negative matrix
  \(\widetilde A\), counted with multiplicity.  On each irreducible
  component of \(\widetilde A\) with spectral radius \(\lambda\), the
  Perron--Frobenius cyclic theorem gives a finite cyclic peripheral set
  \(\lambda\zeta\), with \(\zeta\) a root of unity.  Components with smaller
  spectral radius contribute only eigenvalues of modulus \(<\lambda\).  The
  optional semisimplicity certificate in the statement would rule out higher
  Jordan blocks for the twisted Peter--Weyl pieces at these peripheral
  values, but the trace identity itself does not require this.

  Fix \(\pi\).  Put \(A_\pi\) in Jordan normal form.  If the optional
  semisimplicity certificate is present, every peripheral Jordan block is
  \(1\times1\).
  Without it, a Jordan block \(J=\lambda\zeta I+N\) of size \(d\) still
  satisfies
  \[
    \Tr(J^n)
    =
    \Tr\left(\sum_{a=0}^{d-1}\binom na(\lambda\zeta)^{n-a}N^a\right)
    =
    d(\lambda\zeta)^n,
  \]
  because \(N^a\) is nilpotent, and hence has trace \(0\), for \(a\ge1\).
  Thus the total contribution of all Jordan blocks at \(\lambda\zeta\) is
  \(m_{\pi,\zeta}(\lambda\zeta)^n\), with \(m_{\pi,\zeta}\) the algebraic
  multiplicity.  Every remaining eigenvalue has modulus at most
  \(\Lambda_\perp\).  Applying the same trace calculation to the finitely
  many non-peripheral Jordan blocks, the sum of the remaining powers is
  \(O(\Lambda_\perp^n)\), with the convention that the term is zero if
  \(\Lambda_\perp=0\).  This proves
  \eqref{eq:finite-cover-peripheral-trace-expansion}.
\end{proof}

\begin{proposition}[Primitive-channel estimate at the Perron boundary]
  \label{prop:peripheral-primitive-channel-asymptotic-component}
  Under the peripheral algebraic multiplicity hypotheses of
  \Cref{def:peripheral-peter-weyl-data,lem:finite-cover-peripheral-expansion},
  every irreducible \(\varrho\in\Irr(G)\) satisfies
  \begin{equation}\label{eq:peripheral-primitive-channel-asymptotic-component}
    \pi_{n,\varrho}(A,\tau)
    =
    \frac{\lambda^n}{n}
      \sum_{\zeta\in\Gamma_\varrho}m_{\varrho,\zeta}\zeta^n
    +
    O\!\left(\frac{\Lambda_\partial^n}{n}\right).
  \end{equation}
  More explicitly, the \(m=1\) term in the Adams--M\"obius trace formula
  gives the displayed peripheral sum and the interior
  \(O(\Lambda_\perp^n/n)\) remainder, while the full \(m>1\) divisor
  correction is
  \[
    \frac{1}{n}\sum_{\substack{m\mid n\\m>1}}\mu(m)
      \sum_{\pi\in\Irr(G)}
        a_{\varrho,\pi}^{(m)}\Tr(A_\pi^{n/m})
    =
    O\!\left(\frac{\lambda^{n/2}}{n}\right).
  \]
\end{proposition}

\begin{proof}
  Start from the coefficient formula
  \eqref{eq:primitive-peter-weyl-trace}:
  \[
    \pi_{n,\varrho}(A,\tau)
    =
    \frac{1}{n}\Tr(A_\varrho^n)
    +
    \frac{1}{n}\sum_{\substack{m\mid n\\m>1}}\mu(m)
      \sum_{\pi\in\Irr(G)}
        a_{\varrho,\pi}^{(m)}\Tr(A_\pi^{n/m}).
  \]
  Apply \Cref{lem:finite-cover-peripheral-expansion} to the first trace.
  This gives
  \[
    \frac{1}{n}\Tr(A_\varrho^n)
    =
    \frac{\lambda^n}{n}
      \sum_{\zeta\in\Gamma_\varrho}m_{\varrho,\zeta}\zeta^n
    +O(\Lambda_\perp^n/n).
  \]
  For the remaining divisors, the Adams coefficients are uniformly bounded
  because \(G\) has finitely many irreducible characters, and
  \(\rho(A_\pi)\le\lambda\) by
  \Cref{prop:twisted-spectral-radius-bound}.  Hence
  \(|\Tr(A_\pi^\ell)|\le C\lambda^\ell\), uniformly in \(\pi\) and
  \(\ell\).  If \(m>1\) divides \(n\), then \(n/m\le n/2\), and therefore
  \[
    \sum_{\substack{m\mid n\\m>1}}\lambda^{n/m}
    \le \sum_{k\le n/2}\lambda^k
    =O(\lambda^{n/2}).
  \]
  This proves the divisor-loss estimate and hence
  \eqref{eq:peripheral-primitive-channel-asymptotic-component}.
\end{proof}

\begin{proposition}[Class-Mertens finite part at the Perron boundary]
  \label{prop:renormalized-class-mertens-component}
  Assume the Perron-boundary algebraic multiplicity data above.  For every
  \(\varrho\in\Irr(G)\), define
  \begin{equation}\label{eq:renormalized-primitive-channel-finite-part}
    R_\varrho:=
    \lim_{t\uparrow1}
      \left(\Pi_\varrho(t\lambda^{-1})+s_\varrho\log(1-t)\right).
  \end{equation}
  Then the limit exists and
  \begin{equation}\label{eq:renormalized-primitive-channel-constant}
    R_\varrho=
    -\sum_{\substack{\zeta\in\Gamma_\varrho\\\zeta\ne1}}
      m_{\varrho,\zeta}\Log(1-\zeta)
    +R_{\varrho,\perp},
  \end{equation}
  where \(R_{\varrho,\perp}\) is the absolutely convergent interior and
  Adams-divisor correction obtained after subtracting all peripheral
  terms.  Moreover, for every \(N\ge1\),
  \begin{equation}\label{eq:renormalized-channel-partial-sum}
    \sum_{n\le N}\frac{\pi_{n,\varrho}(A,\tau)}{\lambda^n}
    =
    s_\varrho H_N
    +R_\varrho
    -
    \sum_{\substack{\zeta\in\Gamma_\varrho\\\zeta\ne1}}
      m_{\varrho,\zeta}\sum_{n>N}\frac{\zeta^n}{n}
    +O(\Lambda_\partial^N/(N\lambda^N)).
  \end{equation}
  In particular the oscillatory tails with \(\zeta\ne1\) are
  \(O(N^{-1})\).

  For a conjugacy class \(C\subset G\), put
  \begin{equation}\label{eq:renormalized-class-density}
    \delta_C:=
    \frac{|C|}{|G|}
      \sum_{\varrho\in\Irr(G)}
        \chi_\varrho^\ast(C)s_\varrho .
  \end{equation}
  Then the pointwise class-count expansion is
  \begin{equation}\label{eq:renormalized-class-pointwise-expansion}
    p_{n,C}
    =
    \frac{|C|}{|G|}
    \frac{\lambda^n}{n}
      \sum_{\varrho\in\Irr(G)}
        \chi_\varrho^\ast(C)
        \sum_{\zeta\in\Gamma_\varrho}
          m_{\varrho,\zeta}\zeta^n
    +O(\Lambda_\partial^n/n),
  \end{equation}
  and the summatory Mertens formula is
  \begin{equation}\label{eq:renormalized-class-mertens-harmonic}
    \sum_{n\le N}\frac{p_{n,C}}{\lambda^n}
    =
    \delta_C H_N+
    \frac{|C|}{|G|}
      \sum_{\varrho\in\Irr(G)}
        \chi_\varrho^\ast(C)R_\varrho
    +O(N^{-1}).
  \end{equation}
  Equivalently,
  \begin{equation}\label{eq:renormalized-class-mertens-log}
    \sum_{n\le N}\frac{p_{n,C}}{\lambda^n}
    =
    \delta_C\log N+M_C^{\mathrm{ren}}+O(N^{-1}),
    \qquad
    M_C^{\mathrm{ren}}:=
    \frac{|C|}{|G|}
      \sum_{\varrho\in\Irr(G)}
        \chi_\varrho^\ast(C)(s_\varrho\gamma+R_\varrho).
  \end{equation}
  The exponent \(\delta_C\) is non-negative, and the displayed
  \(M_C^{\mathrm{ren}}\) is real, by
  \Cref{lem:renormalized-reality-positivity}.
\end{proposition}

\begin{proof}
  This proof uses only the primitive-channel estimate of
  \Cref{prop:peripheral-primitive-channel-asymptotic-component}, not the
  consolidated theorem stated below.  Define
  \begin{equation}\label{eq:renormalized-channel-error-b}
    b_{\varrho,n}:=
    \lambda^{-n}\pi_{n,\varrho}(A,\tau)
    -\frac1n
      \sum_{\zeta\in\Gamma_\varrho}
        m_{\varrho,\zeta}\zeta^n .
  \end{equation}
  By \Cref{prop:peripheral-primitive-channel-asymptotic-component},
  \[
    |b_{\varrho,n}|
    \le
    C\frac{(\Lambda_\partial/\lambda)^n}{n}.
  \]
  Hence \(\sum_{n\ge1}|b_{\varrho,n}|<\infty\), because
  \(\Lambda_\partial/\lambda<1\).  Therefore, for \(0\le t<1\),
  \[
    \Pi_\varrho(t\lambda^{-1})
    =
    \sum_{\zeta\in\Gamma_\varrho}
      m_{\varrho,\zeta}\sum_{n\ge1}\frac{(\zeta t)^n}{n}
    +\sum_{n\ge1}b_{\varrho,n}t^n,
    \qquad
    \sum_n|b_{\varrho,n}|<\infty .
  \]
  The term \(\zeta=1\) is \(-s_\varrho\log(1-t)\).  For
  \(\zeta\ne1\), the Abel limit
  \(\sum_{n\ge1}\zeta^n t^n/n\to-\Log(1-\zeta)\) exists in the branch
  reached from the origin along \(0\le t<1\).  This proves existence of
  \(R_\varrho\) and the expression
  \eqref{eq:renormalized-primitive-channel-constant}, with the interior
  correction explicitly given by
  \[
    R_{\varrho,\perp}=\sum_{n\ge1}b_{\varrho,n}.
  \]
  The same decomposition of partial sums gives
  \eqref{eq:renormalized-channel-partial-sum}; Dirichlet summation gives
  \(\sum_{n>N}\zeta^n/n=O(N^{-1})\) for \(\zeta\ne1\), and the interior tail
  is geometric with size \(O(\Lambda_\partial^N/(N\lambda^N))\).

  Apply the class decomposition
  \eqref{eq:class-primitive-coefficients} and substitute
  \Cref{prop:peripheral-primitive-channel-asymptotic-component}.  The finite character
  sum gives the pointwise expansion
  \eqref{eq:renormalized-class-pointwise-expansion}; the coefficient of
  the non-oscillatory harmonic term is exactly
  \(\delta_C\).  Summing the channel identities
  \eqref{eq:renormalized-channel-partial-sum} over
  \(\varrho\) with the Hermitian class coefficients gives
  \eqref{eq:renormalized-class-mertens-harmonic}.  Finally
  \(H_N=\log N+\gamma+O(N^{-1})\), which changes the displayed constant to
  \(M_C^{\mathrm{ren}}\) and proves
  \eqref{eq:renormalized-class-mertens-log}.
\end{proof}

\begin{proposition}[Fixed-label determinant component of the boundary theorem]
  \label{prop:renormalized-fixed-label-determinant-component}
  Under the peripheral algebraic multiplicity hypotheses of
  \Cref{def:peripheral-peter-weyl-data,lem:finite-cover-peripheral-expansion},
  fix the branch-normalized boundary datum
  \(D_{\partial}^{\mathrm{ren}}\) from
  \eqref{eq:perron-boundary-renormalized-data-local}.  For
  \(\pi\in\Irr(G)\), define the reduced boundary determinant logarithm
  \begin{equation}\label{eq:renormalized-reduced-boundary-log}
    \mathcal B_{1,\pi}^{\mathrm{ren}}
    :=
    \lim_{t\uparrow1}
      \left(
        \Log_0\det(I-t\lambda^{-1}A_\pi)^{-1}
        +s_\pi\log(1-t)
      \right).
  \end{equation}
  Equivalently,
  \[
    \mathcal B_{1,\pi}^{\mathrm{ren}}
    =
    -\sum_{\substack{\zeta\in\Gamma_\pi\\\zeta\ne1}}
      m_{\pi,\zeta}\Log(1-\zeta)
    +
    \Log C_{\partial,\pi},
  \]
  where \(C_{\partial,\pi}\) is the product of
  \((1-\alpha/\lambda)^{-1}\) over the non-peripheral eigenvalues
  \(\alpha\) of \(A_\pi\), counted with algebraic multiplicity.  If
  \(1\in\Gamma_\pi\), this is the reduced determinant finite part
  \(\Log C_\lambda(A_\pi)\) after deleting the factor
  \((1-t)^{s_\pi}\); if \(1\notin\Gamma_\pi\), it is the ordinary Abel
  boundary value with the non-real peripheral factors retained.

  This is the \(k=1\) instance of
  \Cref{def:perron-boundary-logarithm-convention}.  For every
  \(\varrho\in\Irr(G)\), define
  \begin{equation}\label{eq:renormalized-fixed-label-def}
    F_\varrho^{\mathrm{ren}}
    :=
    \lim_{t\uparrow1}
      \left(F_\varrho(t\lambda^{-1})+s_\varrho\log(1-t)\right).
  \end{equation}
  Then the limit exists and is given by the branch-compatible
  Adams--M\"obius determinant formula
  \begin{equation}\label{eq:renormalized-fixed-label-determinant}
    F_\varrho^{\mathrm{ren}}
    =
    \sum_{r,m\ge1}\frac{\mu(m)}{rm}
      \sum_{\pi\in\Irr(G)}
        a_{\varrho,\pi}^{(m)}\,
        B_{rm,\pi}^{\mathrm{ren}},
  \end{equation}
  where the values \(\mathcal B_{rm,\pi}^{\mathrm{ren}}\) are those of
  \Cref{def:perron-boundary-logarithm-convention}.  Thus
  \(\mathcal B_{1,\pi}^{\mathrm{ren}}\) and
  \(\mathcal B_{k,\pi}^{\mathrm{ren}}\) are computed from the finite
  branch-normalized input \(D_{\partial}^{\mathrm{ren}}\): the determinant
  polynomial and recorded peripheral factorisation supply the factors, while
  the branch rule and scalar normalization specify the logarithmic values used
  below.
  Equivalently, the branch-normalized algorithm is:
  factor \(P_\pi(t/\lambda)\), isolate the factors
  \((1-t\zeta)^{m_{\pi,\zeta}}\) with \(\zeta\in\Gamma_\pi\), cancel the
  \(s_\pi=m_{\pi,1}\) copies of \(1-t\), take the Abel limit along
  \(0\le t<1\) using the recorded \(\Log_0\) branch, and use the ordinary
  germ value of \(\Log_0 P_\pi(\lambda^{-k})^{-1}\) for \(k\ge2\).  Thus every \(k=1\)
  Perron-boundary block is evaluated by its reduced finite part, every
  \(k\ge2\) term remains an ordinary interior determinant logarithm, and the
  scalar/product normalization is supplied by
  \(n_\partial\).
\end{proposition}

\begin{proof}
  The determinant logarithm decomposes spectrally as
  \[
    \Log\det(I-t\lambda^{-1}A_\pi)^{-1}
    =
    \sum_{\zeta\in\Gamma_\pi}
      -m_{\pi,\zeta}\Log(1-t\zeta)
    +H_\pi(t),
  \]
  where \(H_\pi(t)\) is analytic at \(t=1\), because all non-peripheral
  eigenvalues satisfy \(|\alpha|<\lambda\).  This factorisation uses only
  the characteristic polynomial, so the exponents are algebraic
  multiplicities; nilpotent parts of peripheral Jordan blocks do not change
  the determinant factors.  Adding
  \(s_\pi\log(1-t)\) cancels precisely the \(\zeta=1\) logarithm, while
  the \(\zeta\ne1\) terms have finite Abel limits.  This proves
  \eqref{eq:renormalized-reduced-boundary-log}.

  For \(0<t<1\), the open-disc fixed-label identity
  \eqref{eq:fixed-label-euler-transform-adams-mobius}, followed by
  determinant linearisation, gives
  \[
    F_\varrho(t\lambda^{-1})
    =
    \sum_{r,m\ge1}\frac{\mu(m)}{rm}
      \sum_{\pi\in\Irr(G)}
        a_{\varrho,\pi}^{(m)}
        \Log\det(I-t^{rm}\lambda^{-rm}A_\pi)^{-1}.
  \]
  The only terms that can fail to be ordinary interior values at
  \(t=1\) have \(rm=1\), hence \(r=m=1\).  For those terms the Adams
  coefficient is \(a_{\varrho,\pi}^{(1)}=\delta_{\varrho\pi}\), so the
  logarithmic singular coefficient is exactly \(s_\varrho\log(1-t)\).
  Adding this term converts the whole \(rm=1\) boundary contribution into
  \(\mathcal B_{1,\varrho}^{\mathrm{ren}}\).  Every \(rm\ge2\) term is inside
  the Perron disc because
  \(\rho(\lambda^{-rm}A_\pi)\le\lambda^{1-rm}<1\).  The \(rm\ge2\) tail is
  dominated directly from the finite-dimensional spectral-radius bound
  \(\rho(A_\pi)\le\lambda\).  Since there are only finitely many
  \(\pi\), there is a constant \(C\) such that
  \[
    \abs{\Tr(A_\pi^\ell)}\le C\lambda^\ell
    \qquad(\pi\in\Irr(G),\ \ell\ge1).
  \]
  Hence, uniformly for \(0\le t\le1\),
  \[
    \abs{\Log\det(I-t^{rm}\lambda^{-rm}A_\pi)^{-1}}
    \le
    C'\sum_{\ell\ge1}\frac{\lambda^\ell}{\ell}\lambda^{-rm\ell}
    \le C''\lambda^{1-rm}
    \qquad(rm\ge2).
  \]
  Therefore
  \(\sum_{rm\ge2}\lambda^{1-rm}/(rm)<\infty\) gives an \(L^1\) majorant
  independent of \(t\).
  Dominated convergence gives
  \eqref{eq:renormalized-fixed-label-determinant}.
\end{proof}

\begin{lemma}[Contragredient reality and product positivity]
  \label{lem:renormalized-reality-positivity}
  Under the Perron-boundary algebraic multiplicity data above, the renormalized
  coordinates satisfy
  \[
    s_{\varrho^\vee}=s_\varrho,\qquad
    R_{\varrho^\vee}=\overline{R_\varrho},\qquad
    F_{\varrho^\vee}^{\mathrm{ren}}
      =\overline{F_\varrho^{\mathrm{ren}}}
  \]
  for every \(\varrho\in\Irr(G)\).  Consequently, for every conjugacy class
  \(C\subset G\), the sums defining
  \(\delta_C\), \(M_C^{\mathrm{ren}}\), and \(K_C^{\mathrm{ren}}\) are real,
  \(\delta_C\ge0\), and \(K_C^{\mathrm{ren}}>0\).
  The conjugations are taken with the boundary logarithm convention of
  \Cref{def:perron-boundary-logarithm-convention}.
\end{lemma}

\begin{proof}
  Choose the unitary model used in \eqref{eq:twisted-matrix-def}.  The
  contragredient block is obtained by conjugating each matrix coefficient:
  \(A_{\varrho^\vee}\) is entrywise conjugate to \(A_\varrho\), after the
  canonical identification \(V_{\varrho^\vee}=\overline{V_\varrho}\).
  Hence the peripheral spectrum and algebraic multiplicities are conjugate:
  if \(\lambda\zeta\) occurs in \(A_\varrho\) with multiplicity
  \(m_{\varrho,\zeta}\), then
  \(\lambda\overline\zeta\) occurs in \(A_{\varrho^\vee}\) with the same
  multiplicity.  In particular
  \(s_{\varrho^\vee}=m_{\varrho^\vee,1}=m_{\varrho,1}=s_\varrho\).

  For \(0<t<1\), the primitive trace series has real coefficients paired by
  contragredients:
  \[
    \Pi_{\varrho^\vee}(t\lambda^{-1})
    =
    \overline{\Pi_\varrho(t\lambda^{-1})}.
  \]
  Since \(s_{\varrho^\vee}=s_\varrho\) and \(\log(1-t)\) is real on the Abel
  path, taking the reduced finite-part limit in
  \eqref{eq:renormalized-primitive-channel-finite-part} gives
  \(R_{\varrho^\vee}=\overline{R_\varrho}\).  The same argument applies to
  the fixed-label Euler transform:
  \(F_{\varrho^\vee}(t\lambda^{-1})
  =\overline{F_\varrho(t\lambda^{-1})}\) for \(0<t<1\), and the defining
  finite-part limit \eqref{eq:renormalized-fixed-label-def} gives
  \(F_{\varrho^\vee}^{\mathrm{ren}}
  =\overline{F_\varrho^{\mathrm{ren}}}\).

  The Hermitian class coefficient also conjugates under duality:
  \[
    \chi_{\varrho^\vee}^\ast(C)
    =
    \overline{\chi_\varrho^\ast(C)}.
  \]
  Thus each non-self-dual pair
  \(\{\varrho,\varrho^\vee\}\) contributes twice the real part to the sums
  defining \(\delta_C\), \(M_C^{\mathrm{ren}}\), and
  \(\log K_C^{\mathrm{ren}}\), while a self-dual summand is real by the same
  identities.  This proves the asserted reality.

  It remains to record the signs.  For every \(N\),
  \(\sum_{n\le N}p_{n,C}\lambda^{-n}\ge0\).  Divide
  \eqref{eq:renormalized-class-mertens-log} by \(\log N\) and let
  \(N\to\infty\); the bounded constant and \(O(N^{-1})\) terms vanish, so
  \(\delta_C\ge0\).  Finally,
  \[
    \log K_C^{\mathrm{ren}}
    =
    -\frac{|C|}{|G|}
      \sum_{\varrho\in\Irr(G)}
        \chi_\varrho^\ast(C)
        \bigl(s_\varrho\gamma+F_\varrho^{\mathrm{ren}}\bigr)
  \]
  is real by the preceding paragraph.  Therefore
  \(K_C^{\mathrm{ren}}=\exp(\log K_C^{\mathrm{ren}})>0\).
\end{proof}

\begin{lemma}[Perron-boundary product-tail reduction]
  \label{lem:perron-boundary-product-tail-reduction}
  Under the peripheral algebraic multiplicity hypotheses of
  \Cref{def:peripheral-peter-weyl-data,lem:finite-cover-peripheral-expansion},
  every conjugacy class \(C\subset G\) satisfies
  \begin{equation}\label{eq:perron-boundary-tail-class-expansion}
    p_{n,C}
    =
    \frac{|C|}{|G|}
    \frac{\lambda^n}{n}
      \sum_{\varrho\in\Irr(G)}
        \chi_\varrho^\ast(C)
        \sum_{\zeta\in\Gamma_\varrho}
          m_{\varrho,\zeta}\zeta^n
    +O(\Lambda_\partial^n/n).
  \end{equation}
  Consequently the primitive class counts satisfy a class-uniform bound
  \begin{equation}\label{eq:perron-boundary-primitive-class-bound}
    0\le p_{n,C}\le B_{\partial}\frac{\lambda^n}{n}
    \qquad(n\ge1,\ C\subset G),
  \end{equation}
  for a constant \(B_{\partial}\) depending only on the finite determinant
  model.  For every conjugacy class \(C\),
  \begin{equation}\label{eq:perron-boundary-product-tail-omitted}
    \sum_{r\ge2}\frac1r\sum_{n>N}p_{n,C}\lambda^{-rn}
    =
    O(\lambda^{-N}/N),
  \end{equation}
  uniformly in \(C\).  Moreover
  \begin{equation}\label{eq:perron-boundary-tail-class-mertens}
    \sum_{n\le N}\frac{p_{n,C}}{\lambda^n}
    =
    \delta_C\log N+M_C^{\mathrm{ren}}+O(N^{-1}),
  \end{equation}
  with \(\delta_C\) and \(M_C^{\mathrm{ren}}\) as in
  \eqref{eq:renormalized-class-density} and
  \eqref{eq:renormalized-class-mertens-log}.
  Finally, the repetition tail
  \[
    T_C(N):=
      \sum_{r\ge2}\frac1r\sum_{n\le N}p_{n,C}\lambda^{-rn}
  \]
  has an absolutely convergent endpoint
  \[
    T_C(\infty):=
      \sum_{r\ge2}\frac1r\sum_{n\ge1}p_{n,C}\lambda^{-rn}
  \]
  and
  \begin{equation}\label{eq:perron-boundary-product-tail-truncation}
    T_C(N)=T_C(\infty)+O(\lambda^{-N}/N),
  \end{equation}
  uniformly in \(C\).  Moreover,
  \begin{equation}\label{eq:perron-boundary-product-tail-fourier}
    T_C(\infty)
    =
    \frac{|C|}{|G|}
      \sum_{\varrho\in\Irr(G)}
        \chi_\varrho^\ast(C)
        \sum_{r\ge2}\frac{\Pi_\varrho(\lambda^{-r})}{r},
  \end{equation}
  where the \(r\ge2\) series is absolutely convergent.
\end{lemma}

\begin{proof}
  Apply the class-indicator expansion
  \eqref{eq:class-primitive-coefficients} to the primitive-channel estimate
  of \Cref{prop:peripheral-primitive-channel-asymptotic-component}.  The
  irreducible character set is finite, so the finitely many
  \(O(\Lambda_\partial^n/n)\) errors remain
  \(O(\Lambda_\partial^n/n)\) after multiplication by
  \((|C|/|G|)\chi_\varrho^\ast(C)\) and summation over \(\varrho\).  This
  proves \eqref{eq:perron-boundary-tail-class-expansion}.  Since
  \(|\zeta|=1\), \(\Lambda_\partial<\lambda\), and there are only finitely
  many peripheral phases and irreducible characters,
  \[
    \abs{p_{n,C}}\le B_{\partial}\frac{\lambda^n}{n}
  \]
  after enlarging \(B_{\partial}\) over the finite set of classes.  The lower
  bound is the definition of \(p_{n,C}\) as a primitive-orbit count.  This
  proves \eqref{eq:perron-boundary-primitive-class-bound}.

  Hence the omitted part of the repetition tail is bounded by
  \(B_{\partial}\) times
  \[
    \sum_{r\ge2}\frac1r\sum_{n>N}\frac{\lambda^{(1-r)n}}{n}.
  \]
  Since \(n\ge N+1\) in the inner sum,
  \[
    \sum_{r\ge2}\frac1r\sum_{n>N}\frac{\lambda^{(1-r)n}}{n}
    \le
    \frac1{N+1}
    \sum_{r\ge2}
      \frac{\lambda^{(1-r)(N+1)}}{r(1-\lambda^{1-r})}.
  \]
  With \(q=\lambda^{-1}<1\), the right-hand side is at most
  \[
    \frac1{N+1}
    \frac{1}{2(1-q)}
      \sum_{s\ge1}q^{s(N+1)}
    =
    O(\lambda^{-N}/N),
  \]
  uniformly in \(C\).  This proves
  \eqref{eq:perron-boundary-product-tail-omitted}.  It also proves the
  absolute convergence of \(T_C(\infty)\) and the truncation estimate
  \eqref{eq:perron-boundary-product-tail-truncation}.

  The summatory formula
  \eqref{eq:perron-boundary-tail-class-mertens} is exactly
  \eqref{eq:renormalized-class-mertens-log}, already obtained from the same
  primitive-channel estimate in
  \Cref{prop:renormalized-class-mertens-component}.  Explicitly, summing the
  absolutely summable channel remainders and the finite oscillatory
  \(\zeta\ne1\) tails gives \(O(N^{-1})\), while the \(\zeta=1\) terms
  contribute \(\delta_C H_N\); substituting
  \(H_N=\log N+\gamma+O(N^{-1})\) gives the displayed
  \(M_C^{\mathrm{ren}}\).

  For each \(r\ge2\), the point \(\lambda^{-r}\) lies strictly inside the
  primitive-series disc.  Thus
  \[
    \sum_{n\ge1}p_{n,C}\lambda^{-rn}
    =
    \frac{|C|}{|G|}
      \sum_{\varrho\in\Irr(G)}
        \chi_\varrho^\ast(C)\Pi_\varrho(\lambda^{-r})
  \]
  by \eqref{eq:class-primitive-generating-function}.  The bound just proved,
  together with the corresponding channel bound
  \(\abs{\pi_{n,\varrho}(A,\tau)}\le C\lambda^n/n\), gives an absolute
  majorant
  \[
    \sum_{r\ge2}\frac1r\sum_{n\ge1}\frac{\lambda^{(1-r)n}}{n}<\infty .
  \]
  Therefore the sums over \(r\), \(n\), and the finite set of irreducible
  characters may be interchanged, proving
  \eqref{eq:perron-boundary-product-tail-fourier}.
\end{proof}

\begin{proposition}[Product component of the boundary theorem]
  \label{prop:renormalized-frobenius-product-component}
  Under the peripheral algebraic multiplicity hypotheses of
  \Cref{def:peripheral-peter-weyl-data,lem:finite-cover-peripheral-expansion},
  and for the fixed branch-normalized boundary datum
  \(D_{\partial}^{\mathrm{ren}}\) from
  \eqref{eq:perron-boundary-renormalized-data-local},
  the Frobenius-class Euler product
  \[
    P_C(N):=
    \prod_{\substack{\gamma\in\cP:\ |\gamma|\le N\\
      \tau(\gamma)\in C}}
      (1-\lambda^{-|\gamma|})
  \]
  satisfies
  \begin{equation}\label{eq:renormalized-frobenius-product-component}
    P_C(N)=K_C^{\mathrm{ren}}N^{-\delta_C}\bigl(1+O(N^{-1})\bigr),
  \end{equation}
  where \(\delta_C\) is defined in
  \eqref{eq:renormalized-class-density} and
  \begin{equation}\label{eq:renormalized-frobenius-product-constant-component}
    K_C^{\mathrm{ren}}
    :=
    \exp\!\left(
      -\frac{|C|}{|G|}
        \sum_{\varrho\in\Irr(G)}
          \chi_\varrho^\ast(C)
          \bigl(s_\varrho\gamma+F_\varrho^{\mathrm{ren}}\bigr)
    \right).
  \end{equation}
  The constant is positive by
  \Cref{lem:renormalized-reality-positivity}.
  The exponent \(\delta_C\) is therefore the actual logarithmic density of
  the non-oscillatory Perron boundary, not necessarily \(|C|/|G|\).
\end{proposition}

\begin{proof}
  Taking logarithms gives
  \[
    \log P_C(N)
    =
    -\sum_{n\le N}\frac{p_{n,C}}{\lambda^n}
    -
    \sum_{r\ge2}\frac1r\sum_{n\le N}p_{n,C}\lambda^{-rn}.
  \]
  The first sum is evaluated by
  \eqref{eq:perron-boundary-tail-class-mertens}, the Mertens part of
  \Cref{lem:perron-boundary-product-tail-reduction}.  The second sum is the
  repetition tail \(T_C(N)\).  The same lemma gives
  \[
    T_C(N)
    =
    \frac{|C|}{|G|}
      \sum_{\varrho\in\Irr(G)}
        \chi_\varrho^\ast(C)
        \sum_{r\ge2}\frac{\Pi_\varrho(\lambda^{-r})}{r}
    +O(\lambda^{-N}/N).
  \]
  On the open disc, the fixed-label Euler transform separates as
  \[
    F_\varrho(t\lambda^{-1})
    =
    \Pi_\varrho(t\lambda^{-1})
    +
    \sum_{r\ge2}\frac{\Pi_\varrho(t^r\lambda^{-r})}{r}.
  \]
  The same majorant used in
  \Cref{lem:perron-boundary-product-tail-reduction} gives dominated
  convergence for the \(r\ge2\) series as \(t\uparrow1\).  Combining this
  with the renormalized \(r=1\) limit
  \(R_\varrho\) from
  \Cref{prop:renormalized-class-mertens-component} gives
  \[
    F_\varrho^{\mathrm{ren}}
    =
    R_\varrho+
    \sum_{r\ge2}\frac{\Pi_\varrho(\lambda^{-r})}{r}.
  \]
  Hence
  \[
    \log P_C(N)
    =
    -\delta_C\log N
    -\frac{|C|}{|G|}
      \sum_{\varrho\in\Irr(G)}
        \chi_\varrho^\ast(C)
        (s_\varrho\gamma+F_\varrho^{\mathrm{ren}})
    +O(N^{-1}).
  \]
  By \Cref{lem:renormalized-reality-positivity}, the exponent and constant
  term are real and \(K_C^{\mathrm{ren}}>0\).  Exponentiating with
  \Cref{lem:class-product-log-to-product-error} proves
  \eqref{eq:renormalized-frobenius-product-component}.
\end{proof}

\begin{theorem}[Main Theorem B: Perron-boundary determinant and product theorem]
  \label{thm:perron-boundary-determinant-product}
  \label{thm:peripheral-primitive-channel-asymptotic}
  \label{thm:renormalized-class-mertens}
  \label{thm:renormalized-fixed-label-determinant}
  \label{thm:renormalized-frobenius-product}
  Under the peripheral algebraic multiplicity hypotheses of
  \Cref{def:peripheral-peter-weyl-data,lem:finite-cover-peripheral-expansion},
  and for the fixed branch-normalized boundary datum
  \(D_{\partial}^{\mathrm{ren}}\) of
  \eqref{eq:perron-boundary-renormalized-data-local},
  the following assertions hold.
  \begin{enumerate}[label=\textup{(\roman*)},leftmargin=*]
  \item For every irreducible \(\varrho\in\Irr(G)\),
  \begin{equation}\label{eq:peripheral-primitive-channel-asymptotic}
    \pi_{n,\varrho}(A,\tau)
    =
    \frac{\lambda^n}{n}
      \sum_{\zeta\in\Gamma_\varrho}m_{\varrho,\zeta}\zeta^n
    +
    O\!\left(\frac{\Lambda_\partial^n}{n}\right).
  \end{equation}
  \item The renormalized primitive-channel finite part
  \[
    R_\varrho:=
    \lim_{t\uparrow1}
      \left(\Pi_\varrho(t\lambda^{-1})+s_\varrho\log(1-t)\right)
  \]
  exists.  For every conjugacy class \(C\subset G\),
  \[
    \delta_C=
    \frac{|C|}{|G|}
      \sum_{\varrho\in\Irr(G)}\chi_\varrho^\ast(C)s_\varrho
  \]
  and
  \begin{equation}\label{eq:perron-boundary-main-class-mertens}
    \sum_{n\le N}\frac{p_{n,C}}{\lambda^n}
    =
    \delta_C\log N+M_C^{\mathrm{ren}}+O(N^{-1}),
    \qquad
    M_C^{\mathrm{ren}}=
    \frac{|C|}{|G|}
      \sum_{\varrho\in\Irr(G)}
        \chi_\varrho^\ast(C)(s_\varrho\gamma+R_\varrho).
  \end{equation}
  These constants are real in the class sums, with
  \(\delta_C\ge0\), by
  \Cref{lem:renormalized-reality-positivity}.
  \item The reduced fixed-label Euler transform
  \[
    F_\varrho^{\mathrm{ren}}
    =
    \lim_{t\uparrow1}
      \left(F_\varrho(t\lambda^{-1})+s_\varrho\log(1-t)\right)
  \]
  exists and is evaluated by
  \begin{equation}\label{eq:perron-boundary-main-fixed-label}
    F_\varrho^{\mathrm{ren}}
    =
    \sum_{r,m\ge1}\frac{\mu(m)}{rm}
      \sum_{\pi\in\Irr(G)}
        a_{\varrho,\pi}^{(m)}
        B_{rm,\pi}^{\mathrm{ren}},
  \end{equation}
  where \(k=1\) terms are the reduced Perron-boundary finite parts computed
  from the factorisation and Abel branch rule, and \(k\ge2\) terms are ordinary
  interior determinant logarithms, as in
  \eqref{eq:renormalized-fixed-label-determinant}.  Thus the determinant finite
  parts \(B_{k,\pi}^{\mathrm{ren}}=\mathcal B_{k,\pi}^{\mathrm{ren}}\), and
  hence the output \(F_\varrho^{\mathrm{ren}}\), are derived from the finite
  datum \(D_{\partial}^{\mathrm{ren}}\), not supplied as extra inputs and not
  determined by raw determinant polynomials without the branch rule.
  \item The Frobenius-class Euler product
  \[
    P_C(N):=
    \prod_{\substack{\gamma\in\cP:\ |\gamma|\le N\\
      \tau(\gamma)\in C}}
      (1-\lambda^{-|\gamma|})
  \]
  satisfies
  \begin{equation}\label{eq:renormalized-frobenius-product}
    P_C(N)=K_C^{\mathrm{ren}}N^{-\delta_C}\bigl(1+O(N^{-1})\bigr),
  \end{equation}
  with
  \begin{equation}\label{eq:renormalized-frobenius-product-constant}
    K_C^{\mathrm{ren}}
    :=
    \exp\!\left(
      -\frac{|C|}{|G|}
        \sum_{\varrho\in\Irr(G)}
          \chi_\varrho^\ast(C)
          \bigl(s_\varrho\gamma+F_\varrho^{\mathrm{ren}}\bigr)
    \right).
  \end{equation}
  The exponential is taken of a real number and
  \(K_C^{\mathrm{ren}}>0\), by
  \Cref{lem:renormalized-reality-positivity}.
  Thus \(\delta_C\) is the non-oscillatory Perron-boundary density, not
  necessarily \(|C|/|G|\).
  \end{enumerate}
\end{theorem}

\begin{proof}
  We prove the four assertions in order, using only the peripheral algebraic
  multiplicity hypotheses of
  \Cref{def:peripheral-peter-weyl-data,lem:finite-cover-peripheral-expansion}
  and the fixed branch-normalized datum \(D_{\partial}^{\mathrm{ren}}\).

  The primitive-channel assertion is
  \Cref{prop:peripheral-primitive-channel-asymptotic-component}.  Its proof
  begins with the coefficient formula
  \[
    \pi_{n,\varrho}(A,\tau)
    =
    \frac1n\sum_{m\mid n}\mu(m)
      \sum_{\pi\in\Irr(G)}
        a_{\varrho,\pi}^{(m)}\Tr(A_\pi^{n/m}).
  \]
  The \(m=1\) summand is \(\Tr(A_\varrho^n)/n\), because
  \(a_{\varrho,\pi}^{(1)}=\delta_{\varrho\pi}\).  Applying the peripheral
  trace expansion to this summand gives
  \[
    \frac{\Tr(A_\varrho^n)}n
    =
    \frac{\lambda^n}{n}
      \sum_{\zeta\in\Gamma_\varrho}m_{\varrho,\zeta}\zeta^n
    +O(\Lambda_\perp^n/n).
  \]
  For \(m>1\), one has \(n/m\le n/2\), and the uniform finite-dimensional
  bound \(\Tr(A_\pi^\ell)=O(\lambda^\ell)\) gives a total divisor correction
  \(O(\lambda^{n/2}/n)\).  Since
  \(\Lambda_\partial=\max\{\Lambda_\perp,\lambda^{1/2}\}\), this proves
  \eqref{eq:peripheral-primitive-channel-asymptotic}.

  Next define
  \[
    b_{\varrho,n}
    =
    \lambda^{-n}\pi_{n,\varrho}(A,\tau)
    -\frac1n\sum_{\zeta\in\Gamma_\varrho}m_{\varrho,\zeta}\zeta^n .
  \]
  The estimate just proved gives
  \(|b_{\varrho,n}|\le C(\Lambda_\partial/\lambda)^n/n\), hence
  \(\sum_n|b_{\varrho,n}|<\infty\).  Therefore, for \(0\le t<1\),
  \[
    \Pi_\varrho(t\lambda^{-1})
    =
    \sum_{\zeta\in\Gamma_\varrho}
      m_{\varrho,\zeta}\sum_{n\ge1}\frac{(\zeta t)^n}{n}
    +\sum_{n\ge1}b_{\varrho,n}t^n .
  \]
  The \(\zeta=1\) contribution is \(-s_\varrho\log(1-t)\); the
  \(\zeta\ne1\) contributions have Abel limits
  \(-\Log_0(1-\zeta)\), and the \(b_{\varrho,n}\)-series converges
  absolutely at \(t=1\).  This proves existence of \(R_\varrho\).  Summing
  the same decomposition up to \(N\), using Dirichlet summation for the
  \(\zeta\ne1\) tails and the geometric bound for the \(b_{\varrho,n}\)-tail,
  gives
  \[
    \sum_{n\le N}\lambda^{-n}\pi_{n,\varrho}(A,\tau)
    =
    s_\varrho H_N+R_\varrho+O(N^{-1}).
  \]
  Multiplying by the class-indicator coefficients
  \((|C|/|G|)\chi_\varrho^\ast(C)\), summing over the finite set
  \(\Irr(G)\), and using \(H_N=\log N+\gamma+O(N^{-1})\) gives
  \eqref{eq:perron-boundary-main-class-mertens} with the displayed
  \(\delta_C\) and \(M_C^{\mathrm{ren}}\).

  The determinant evaluation of \(F_\varrho^{\mathrm{ren}}\) is the
  Adams--M\"obius fixed-label identity with the boundary logarithm convention
  inserted.  For \(k=1\), the definition of
  \(B_{1,\pi}^{\mathrm{ren}}\) subtracts exactly \(s_\pi\log(1-t)\) from the
  normalized determinant germ; equivalently it is computed from the
  factorization by
  \[
    B_{1,\pi}^{\mathrm{ren}}
    =
    -\sum_{\substack{\zeta\in\Gamma_\pi\\\zeta\ne1}}
      m_{\pi,\zeta}\Log_0(1-\zeta)
    -\Log_0 Q_{\partial,\pi}(1).
  \]
  For \(k\ge2\), the point \(\lambda^{-k}\) lies strictly inside the Perron
  disc and the value is the ordinary interior value
  \(B_{k,\pi}^{\mathrm{ren}}=\Log_0\det(I-\lambda^{-k}A_\pi)^{-1}\).  Applying
  the open-disc fixed-label identity and passing to the Abel limit with these
  \(B_{k,\pi}^{\mathrm{ren}}\) values gives
  \eqref{eq:perron-boundary-main-fixed-label}.  This is precisely the
  derived-coordinate formula
  \eqref{eq:perron-boundary-fren-derived-data}; the raw polynomials
  \(\{P_\pi\}\) are not being used without the branch recipe.

  It remains to prove the product asymptotic.  Taking logarithms of the
  finite Euler product gives the exact identity
  \[
    \log P_C(N)
    =
    \sum_{n\le N}p_{n,C}\log(1-\lambda^{-n})
    =
    -\sum_{n\le N}\frac{p_{n,C}}{\lambda^n}
    -\sum_{r\ge2}\frac1r\sum_{n\le N}p_{n,C}\lambda^{-rn}.
  \]
  The class expansion derived above also gives, as recorded in
  \Cref{lem:perron-boundary-product-tail-reduction},
  \[
    p_{n,C}
    =
    \frac{|C|}{|G|}\frac{\lambda^n}{n}
      \sum_{\varrho\in\Irr(G)}
        \chi_\varrho^\ast(C)
        \sum_{\zeta\in\Gamma_\varrho}
          m_{\varrho,\zeta}\zeta^n
    +O(\Lambda_\partial^n/n).
  \]
  Since \(|\zeta|=1\) and \(\Lambda_\partial<\lambda\), this implies
  \(0\le p_{n,C}\le B_\partial\lambda^n/n\), uniformly in \(C\).  Hence
  the omitted \(r\ge2\) tail is bounded by
  \[
    B_\partial
    \sum_{r\ge2}\frac1r\sum_{n>N}\frac{\lambda^{(1-r)n}}n
    \le
    \frac{B_\partial}{N+1}
    \sum_{r\ge2}
      \frac{\lambda^{(1-r)(N+1)}}{r(1-\lambda^{1-r})}
    =
    O(\lambda^{-N}/N).
  \]
  Therefore
  \[
    \sum_{r\ge2}\frac1r\sum_{n\le N}p_{n,C}\lambda^{-rn}
    =
    \sum_{r\ge2}\frac1r\sum_{n\ge1}p_{n,C}\lambda^{-rn}
    +O(\lambda^{-N}/N),
  \]
  and the endpoint is absolutely convergent.  By class Fourier expansion
  inside the open disc,
  \[
    \sum_{r\ge2}\frac1r\sum_{n\ge1}p_{n,C}\lambda^{-rn}
    =
    \frac{|C|}{|G|}
      \sum_{\varrho\in\Irr(G)}
        \chi_\varrho^\ast(C)
        \sum_{r\ge2}\frac{\Pi_\varrho(\lambda^{-r})}{r}.
  \]
  Combining this identity with
  \eqref{eq:perron-boundary-main-class-mertens} and with
  \[
    F_\varrho^{\mathrm{ren}}
    =
    R_\varrho+\sum_{r\ge2}\frac{\Pi_\varrho(\lambda^{-r})}{r}
  \]
  gives the real logarithmic asymptotic
  \[
    \log P_C(N)
    =
    -\delta_C\log N
    -\frac{|C|}{|G|}
      \sum_{\varrho\in\Irr(G)}
        \chi_\varrho^\ast(C)
        \bigl(s_\varrho\gamma+F_\varrho^{\mathrm{ren}}\bigr)
    +O(N^{-1}).
  \]
  \Cref{lem:renormalized-reality-positivity} shows that the exponent and
  constant term are real, that \(\delta_C\ge0\), and that the exponential of
  the displayed constant term is positive.  Exponentiating the logarithmic
  asymptotic with \Cref{lem:class-product-log-to-product-error} proves
  \eqref{eq:renormalized-frobenius-product} and
  \eqref{eq:renormalized-frobenius-product-constant}.  This completes the
  theorem.
\end{proof}

\begin{corollary}[Corollary D: product-coordinate inversion]
  \label{thm:renormalized-coordinate-readout}
  \label{cor:renormalized-coordinate-readout}
  \label{thm:class-product-determinant-rigidity}
  \label{prop:class-product-downstream-coordinate-status}
  \label{cor:class-product-hyperplane-stability}
  \label{cor:class-product-stability-scalar-independence}
  \label{prop:class-product-fixed-coordinate-equivalence}
  \label{thm:class-product-minimal-obstruction-skeleton}
  \label{cor:class-product-minimal-obstruction-skeleton}
  \label{cor:class-product-artin-obstruction-recovery}
  \label{cor:class-product-artin-obstruction-readout}
  \label{thm:main-iii-certificate-chain}
  \label{thm:adams-obstruction-class-product}
  \label{cor:adams-obstruction-bound}
  \label{cor:adams-obstruction-budget}
  \label{cor:class-product-constant-rigidity}
  \label{cor:class-euler-product-determinant-evaluation}
  \label{prop:classical-layer-adams-corrected-constant-separation}
  Under the hypotheses of
  \Cref{thm:perron-boundary-determinant-product}, define the positive-branch
  logarithmic class profile
  \begin{equation}\label{eq:renormalized-coordinate-profile}
    H_C^{\mathrm{ren}}
    :=
    -\frac{|G|}{|C|}\log K_C^{\mathrm{ren}}
    \qquad(C\subset G).
  \end{equation}
  Then the product constants determine the shifted fixed-label coordinates:
  \begin{equation}\label{eq:renormalized-coordinate-forward}
    H_C^{\mathrm{ren}}
    =
    \sum_{\pi\in\Irr(G)}
      \chi_\pi^\ast(C)\bigl(s_\pi\gamma+F_\pi^{\mathrm{ren}}\bigr)
    \qquad(C\subset G).
  \end{equation}
  The exponent profile determines the boundary multiplicities by the
  finite Fourier inversion
  \begin{equation}\label{eq:renormalized-coordinate-s-inverse}
    s_\pi
    =
    \sum_{C\subset G}\delta_C\chi_\pi(C)
    \qquad(\pi\in\Irr(G)).
  \end{equation}
  Conversely, for every \(\pi\in\Irr(G)\),
  \begin{equation}\label{eq:renormalized-coordinate-inverse}
    F_\pi^{\mathrm{ren}}
    =
    \sum_{C\subset G}\frac{|C|}{|G|}
      H_C^{\mathrm{ren}}\chi_\pi(C)
    -s_\pi\gamma .
  \end{equation}
  The coordinate \(F_\pi^{\mathrm{ren}}\) appearing here is exactly the
  reduced Adams--M\"obius determinant finite part of
  \Cref{prop:renormalized-fixed-label-determinant-component}, namely
  \[
    F_\pi^{\mathrm{ren}}
    =
    \sum_{r,m\ge1}\frac{\mu(m)}{rm}
      \sum_{\sigma\in\Irr(G)}
        a_{\pi,\sigma}^{(m)}\,
        B_{rm,\sigma}^{\mathrm{ren}}.
  \]
  Thus \((K_C^{\mathrm{ren}})_C\), after the positive logarithm is fixed,
  together with \((\delta_C)_C\), equivalently with \((s_\pi)_\pi\), recovers
  the full renormalized Peter--Weyl fixed-label determinant vector.  The
  positive product constants alone recover only the shifted coordinates
  \(s_\pi\gamma+F_\pi^{\mathrm{ren}}\).

  In the strict-gap specialization of
  \Cref{cor:strict-gap-specialization}, this same finite character transform
  gives the scalar-normalized product chart used later.  Set
  \[
    \alpha_A:=\gamma+\log C(A),
  \]
  and, for every non-trivial \(\varrho\in\Irr(G)\), define
  \begin{equation}\label{eq:class-product-rigidity-determinant-coordinate}
    \mathfrak D_\varrho:=
    \sum_{r,m\ge1}\frac{\mu(m)}{rm}
      \sum_{\substack{\pi\in\Irr(G)\\(r,m,\pi)\neq(1,1,\mathbf1)}}
        a_{\varrho,\pi}^{(m)}
        \mathcal L_{rm,\pi}.
  \end{equation}
  The logarithms are interpreted by the strict-gap specialization of
  \Cref{def:perron-boundary-logarithm-convention}; no peripheral
  semisimplicity certificate is used in this strict-gap chart.  Then
  \begin{equation}\label{eq:class-product-rigidity-d-equals-f}
    \mathfrak D_\varrho=F_\varrho(\lambda^{-1})
    \qquad(\varrho\ne\mathbf1),
  \end{equation}
  and the series in
  \eqref{eq:class-product-rigidity-determinant-coordinate} is absolutely
  convergent after the finitely many regular \(rm=1\) non-trivial boundary
  terms have been separated.  If \((K_C)_{C\subset G}\) are the strict-gap
  Frobenius-class product constants of
  \Cref{thm:class-euler-product-mertens}, put
  \begin{equation}\label{eq:class-product-log-vector}
    H_K(C):=-\frac{|G|}{|C|}\log K_C-\alpha_A .
  \end{equation}
  Then
  \begin{equation}\label{eq:class-product-normalisation-constraint}
    \prod_{C\subset G}K_C=\exp(-\alpha_A),
    \qquad
    \sum_{C\subset G}\frac{|C|}{|G|}H_K(C)=0,
  \end{equation}
  \label{eq:class-product-rigidity-hyperplane}
  and
  \begin{equation}\label{eq:class-product-fourier-coordinates}
    H_K(C)=
    \sum_{\varrho\ne\mathbf1}
      \chi_\varrho^\ast(C)\mathfrak D_\varrho,
    \qquad
    \sum_{C\subset G}\frac{|C|}{|G|}H_K(C)\chi_\pi(C)
    =\mathfrak D_\pi.
  \end{equation}
  \label{eq:class-product-rigidity-forward}
  Equivalently,
  \begin{equation}\label{eq:class-product-rigidity-product-formula}
    K_C=
    \exp\!\left(
      -\frac{|C|}{|G|}
      \left(
        \alpha_A+
        \sum_{\varrho\ne\mathbf1}
          \chi_\varrho^\ast(C)\mathfrak D_\varrho
      \right)
    \right).
  \end{equation}
  The Parseval identity in this zero-mean chart is
  \begin{equation}\label{eq:class-product-parseval}
    \sum_{C\subset G}\frac{|C|}{|G|}|H_K(C)|^2
    =
    \sum_{\varrho\ne\mathbf1}|\mathfrak D_\varrho|^2.
  \end{equation}
  \label{eq:class-product-rigidity-parseval}

  Let
  \[
    \mathcal H_G:=
    \left\{H\text{ a real class function}:
      \sum_C\frac{|C|}{|G|}H(C)=0\right\},
  \]
  and let \(\mathcal D_G^{\mathbb R}\) be the corresponding non-trivial
  Peter--Weyl coordinate space, namely the vectors
  \((d_\varrho)_{\varrho\ne\mathbf1}\) for which
  \(C\mapsto\sum_{\varrho\ne\mathbf1}\chi_\varrho^\ast(C)d_\varrho\)
  belongs to \(\mathcal H_G\).  For the fixed scalar \(\alpha_A\), set
  \[
    \mathscr K_{\alpha_A}:=
    \left\{(L_C)_C:
      L_C>0,\quad
      C\longmapsto -\frac{|G|}{|C|}\log L_C-\alpha_A\in\mathcal H_G
    \right\}.
  \]
  The finite Fourier map
  \begin{equation}\label{eq:class-product-rigidity-chart-map}
    (d_\varrho)_{\varrho\ne\mathbf1}
    \longmapsto
    \left(
      \exp\!\left(
        -\frac{|C|}{|G|}
        \left(
          \alpha_A+
          \sum_{\varrho\ne\mathbf1}\chi_\varrho^\ast(C)d_\varrho
        \right)
      \right)
    \right)_{C\subset G}
  \end{equation}
  \label{eq:fixed-coordinate-support-product-map}
  is a bijection from \(\mathcal D_G^{\mathbb R}\) onto
  \(\mathscr K_{\alpha_A}\).  Its inverse is
  \begin{equation}\label{eq:class-product-rigidity-chart-inverse}
    (L_C)_C\longmapsto
    \left(
      \sum_C\frac{|C|}{|G|}
      \left(-\frac{|G|}{|C|}\log L_C-\alpha_A\right)
      \chi_\pi(C)
    \right)_{\pi\ne\mathbf1}.
  \end{equation}
  \label{eq:class-product-rigidity-inverse}
  The Peter--Weyl coordinate record on \(\mathcal H_G\),
  \begin{equation}\label{eq:fixed-coordinate-support-peter-weyl-record}
    H\longmapsto
    \left(
      \sum_C\frac{|C|}{|G|}H(C)\chi_\varrho(C)
    \right)_{\varrho\ne\mathbf1},
  \end{equation}
  is isometric.  Hence, if \(q\) is the number of conjugacy classes and
  \(R:\mathcal H_G\to W\) is any real linear record in this fixed chart with
  \(R(H)=R(H')\Rightarrow \mathcal K_{\alpha_A}(H)=\mathcal K_{\alpha_A}(H')\),
  then
  \begin{equation}\label{eq:fixed-coordinate-support-rank-bound}
    \operatorname{rank}R\ge\dim\mathcal H_G=q-1.
  \end{equation}

  Finally, the same chart gives the replacement-vector and Artin-obstruction
  tests.  If
  \(b=(b_\varrho)_{\varrho\ne\mathbf1}\in\mathcal D_G^{\mathbb R}\) is any
  branch-compatible replacement vector and \(K_C(b)\) is obtained from
  \eqref{eq:class-product-rigidity-product-formula} by replacing
  \(\mathfrak D_\varrho\) with \(b_\varrho\), then
  \begin{equation}\label{eq:classical-layer-replacement-error}
    E_b(C):=-\frac{|G|}{|C|}\log\frac{K_C}{K_C(b)}
    =
    \sum_{\varrho\ne\mathbf1}
      \chi_\varrho^\ast(C)(\mathfrak D_\varrho-b_\varrho)
  \end{equation}
  and
  \begin{equation}\label{eq:class-product-rigidity-replacement-parseval}
    \sum_{C\subset G}\frac{|C|}{|G|}|E_b(C)|^2
    =
    \sum_{\varrho\ne\mathbf1}|\mathfrak D_\varrho-b_\varrho|^2 .
  \end{equation}
  \label{eq:adams-obstruction-bound-parseval}
  Consequently a replacement vector gives all strict-gap product constants
  only when it equals the determinant vector.  If
  \(\delta^2:=\sum_{\varrho\ne\mathbf1}|\mathfrak D_\varrho-b_\varrho|^2>0\),
  then some class satisfies
  \begin{equation}\label{eq:class-product-rigidity-class-lower-bound}
    \left|\frac{|G|}{|C|}\log\frac{K_C}{K_C(b)}\right|
    \ge \delta .
  \end{equation}
  For the regular periodic Artin replacement
  \(b_\varrho=\mathcal L_\varrho(\lambda^{-1})\), the normalized error
  \(\mathcal E(C):=E_b(C)\) satisfies
  \begin{equation}\label{eq:adams-obstruction-bound-zero-mean}
    \sum_C\frac{|C|}{|G|}\mathcal E(C)=0,\qquad
    F_\pi(\lambda^{-1})-\mathcal L_\pi(\lambda^{-1})
    =
    \sum_C\frac{|C|}{|G|}\mathcal E(C)\chi_\pi(C),
  \end{equation}
  \begin{equation}\label{eq:adams-obstruction-bound-forward}
    \mathcal E(C)=
    \sum_{\varrho\ne\mathbf1}\chi_\varrho^\ast(C)
    \bigl(F_\varrho(\lambda^{-1})
      -\mathcal L_\varrho(\lambda^{-1})\bigr),
  \end{equation}
  \label{eq:fixed-coordinate-support-error-bound}
  and
  \begin{equation}\label{eq:adams-obstruction-ratio}
    \log\frac{K_C}{K_C^{\mathrm{Art}}}
    =
    -\frac{|C|}{|G|}
    \sum_{\varrho\ne\mathbf1}\chi_\varrho^\ast(C)
    \bigl(F_\varrho(\lambda^{-1})
      -\mathcal L_\varrho(\lambda^{-1})\bigr).
  \end{equation}
  The ordinary Artin substitution gives all class-product constants if and
  only if
  \begin{equation}\label{eq:adams-obstruction-class-function}
    C\longmapsto
    \frac{|C|}{|G|}
    \sum_{\varrho\ne\mathbf1}\chi_\varrho^\ast(C)
    \bigl(F_\varrho(\lambda^{-1})
      -\mathcal L_\varrho(\lambda^{-1})\bigr)
  \end{equation}
  vanishes identically.  If
  \[
    \Delta^2=\sum_{\varrho\ne\mathbf1}
      |F_\varrho(\lambda^{-1})-\mathcal L_\varrho(\lambda^{-1})|^2,
  \]
  then some conjugacy class satisfies
  \begin{equation}\label{eq:adams-obstruction-bound-sup-lower-bound}
    \left|
      \frac{|G|}{|C|}
      \log\frac{K_C}{K_C^{\mathrm{Art}}}
    \right|\ge\Delta .
  \end{equation}
  Consequently any certified lower bound \(\Delta\ge\delta\) gives
  \begin{equation}\label{eq:adams-obstruction-bound-lower-bound}
    \left|
      \frac{|G|}{|C|}
      \log\frac{K_C}{K_C^{\mathrm{Art}}}
    \right|\ge\delta
  \end{equation}
  for at least one conjugacy class.
\end{corollary}

\begin{proof}
  Formula \eqref{eq:renormalized-frobenius-product-constant} gives
  \[
    \log K_C^{\mathrm{ren}}
    =
    -\frac{|C|}{|G|}
      \sum_{\pi\in\Irr(G)}
        \chi_\pi^\ast(C)
        \bigl(s_\pi\gamma+F_\pi^{\mathrm{ren}}\bigr).
  \]
  Multiplying by \(-|G|/|C|\) proves
  \eqref{eq:renormalized-coordinate-forward}.  Similarly, the definition of
  \(\delta_C\) in
  \Cref{thm:perron-boundary-determinant-product} gives
  \[
    \frac{|G|}{|C|}\delta_C
    =
    \sum_{\sigma\in\Irr(G)}\chi_\sigma^\ast(C)s_\sigma .
  \]
  The irreducible characters are
  an orthonormal basis of the class functions for
  \[
    \langle U,V\rangle_G
    =
    \sum_{C\subset G}\frac{|C|}{|G|}U(C)\overline{V(C)}.
  \]
  We use the convention fixed after \eqref{eq:class-function-fourier}:
  the forward basis in the displayed class profiles is
  \(C\mapsto\chi_\sigma^\ast(C)\), and its Hermitian coefficient is
  recovered by pairing with the same basis vector.  Written out, the
  conjugate in the second slot is
  \(\overline{\chi_\pi^\ast(C)}=\chi_\pi(C)\).  Hence the
  \(\pi\)-coordinate of the last display is
  \[
    \sum_C\frac{|C|}{|G|}
      \frac{|G|}{|C|}\delta_C\,\chi_\pi(C)
    =
    s_\pi,
  \]
  which is \eqref{eq:renormalized-coordinate-s-inverse}.
  The same Hermitian recovery applied to
  \(C\mapsto H_C^{\mathrm{ren}}\) in
  \eqref{eq:renormalized-coordinate-forward} gives, again with
  \(\chi_\pi(C)=\overline{\chi_\pi^\ast(C)}\) coming from the second
  slot,
  \[
    \sum_C\frac{|C|}{|G|}
      H_C^{\mathrm{ren}}\chi_\pi(C)
    =
    s_\pi\gamma+F_\pi^{\mathrm{ren}},
  \]
  which is \eqref{eq:renormalized-coordinate-inverse}.  Finally,
  \Cref{prop:renormalized-fixed-label-determinant-component} is precisely the reduced
  boundary determinant evaluation of \(F_\pi^{\mathrm{ren}}\); substituting
  \eqref{eq:renormalized-fixed-label-determinant} gives the displayed
  Adams--M\"obius reduced determinant expression.  No input beyond finite
  character orthogonality and the already proved determinant formula is used.

  In the strict-gap specialization,
  \Cref{cor:strict-gap-specialization} identifies
  \(F_\varrho^{\mathrm{ren}}\) with \(F_\varrho(\lambda^{-1})\) for
  non-trivial \(\varrho\), and
  \Cref{prop:fixed-label-determinant-perron} gives the determinant evaluation
  \eqref{eq:class-product-rigidity-d-equals-f} with the local deletion of
  \((1,1,\mathbf1)\).  Substituting these identities into the product
  formula of \Cref{thm:class-euler-product-mertens} gives
  \eqref{eq:class-product-fourier-coordinates} and
  \eqref{eq:class-product-rigidity-product-formula}.  Averaging
  \eqref{eq:class-product-fourier-coordinates} over classes with weights
  \(|C|/|G|\) gives the zero-mean identity in
  \eqref{eq:class-product-normalisation-constraint}, since every non-trivial
  irreducible character is orthogonal to the trivial character; summing the
  logarithmic definition of \(H_K\) then gives the product constraint.

  We record the only finite Fourier input used in these downstream clauses.
  For the Hermitian class-function inner product
  \[
    \langle U,V\rangle_G
    =
    \sum_{C\subset G}\frac{|C|}{|G|}U(C)\overline{V(C)},
  \]
  the irreducible characters are an orthonormal basis, equivalently
  \[
    \sum_{C\subset G}\frac{|C|}{|G|}
      \chi_\varrho^\ast(C)\chi_\pi(C)=\delta_{\varrho,\pi}.
  \]
  Thus the inverse coordinate formula in
  \eqref{eq:class-product-fourier-coordinates} and
  \eqref{eq:class-product-rigidity-chart-inverse} is obtained by pairing
  with \(C\mapsto\chi_\pi^\ast(C)\), written out as multiplication by
  \(\chi_\pi(C)=\overline{\chi_\pi^\ast(C)}\).  Expanding the square norm of
  \(\sum_{\varrho\ne\mathbf1}\chi_\varrho^\ast(C)\mathfrak D_\varrho\) and
  applying the same orthogonality relation gives
  \eqref{eq:class-product-parseval}.  This proves the chart map and inverse:
  a non-trivial Fourier vector gives a zero-mean real class function, the
  displayed exponential gives positive constants with weighted product
  \(\exp(-\alpha_A)\), and the inverse transform recovers the original
  Fourier vector.  The Peter--Weyl coordinate record is isometric by this
  Parseval identity.  Hence any real record determining the positive product
  vector on the fixed scalar hyperplane must be injective on the
  \(q-1\)-dimensional space \(\mathcal H_G\), proving the rank bound.

  The replacement-vector assertions are the same calculation after
  subtraction.  The logarithmic ratio between the true constants and the
  constants obtained from \(b\) is exactly
  \eqref{eq:classical-layer-replacement-error}; applying Parseval to this
  error function gives
  \eqref{eq:class-product-rigidity-replacement-parseval}.  Equality of all
  positive constants is therefore equivalent to the vanishing of all
  non-trivial Fourier coordinates.  Since the weights \(|C|/|G|\) are
  positive and sum to one, a non-zero error vector of squared norm
  \(\delta^2\) has a class with normalized absolute error at least
  \(\delta\), proving
  \eqref{eq:class-product-rigidity-class-lower-bound}.  The Artin
  obstruction formulas are obtained by the specialization
  \(b_\varrho=\mathcal L_\varrho(\lambda^{-1})\), and the lower bounds follow
  from the same Parseval identity.
\end{proof}

\begin{corollary}[Corollary C: strict-gap specialization]
  \label{cor:strict-gap-specialization}
  Suppose the only Perron collision is the scalar trivial block:
  \(s_{\mathbf1}=1\), \(s_\varrho=0\) for every
  \(\varrho\ne\mathbf1\), and
  \(\Gamma_\varrho=\varnothing\) for every non-trivial \(\varrho\).  Then
  \(\delta_C=|C|/|G|\), the renormalized non-trivial values
  \(F_\varrho^{\mathrm{ren}}\) are the ordinary strict-gap values
  \(F_\varrho(\lambda^{-1})\), the scalar reduced finite part is
  \(\gamma+\log C(A)\), and
  \Cref{thm:perron-boundary-determinant-product}
  reduces exactly to the strict-gap class Mertens theorem
  \Cref{thm:class-mertens-explicit}, the determinant readout
  \Cref{prop:fixed-label-determinant-perron}, and the product formula
  \Cref{thm:class-euler-product-mertens}.
\end{corollary}

\begin{proof}
  Under the stated hypotheses the only \(s_\varrho\) contribution is the
  trivial channel, so
  \(\delta_C=(|C|/|G|)\chi_{\mathbf1}^\ast(C)=|C|/|G|\).  No non-trivial
  boundary channel has a peripheral eigenvalue, hence all non-trivial
  \(r=1\) Abel values are ordinary endpoint values and
  \(F_\varrho^{\mathrm{ren}}=F_\varrho(\lambda^{-1})\) for
  \(\varrho\ne\mathbf1\).  In the trivial channel, the reduced determinant
  finite part is the usual scalar normalization \(C(A)\), so the product
  scalar term is \(\gamma+\log C(A)\).  Substituting these identities into
  the three renormalized formulas gives exactly the displayed constants,
  exponent, and branch convention of the strict-gap theorems cited in the
  statement.
\end{proof}

\subsubsection{Examples and obstruction tests}
\label{subsubsec:perron-boundary-examples}

The following examples and tests illustrate the boundary theorem and the
coordinate readout.  The cyclic lift is a finite boundary instance of
\Cref{thm:perron-boundary-determinant-product}, and the Artin-substitution
test is obtained by comparing two coordinate vectors in
\Cref{cor:renormalized-coordinate-readout}.

\begin{remark}[Canonical cyclic-lift boundary]
  \label{cor:canonical-cyclic-lift-boundary}
  Let \(G=\ZZ/q\ZZ\), let every edge have cocycle label \(1\), and let
  \(\chi_j(1)=\omega_q^j\).  Then
  \(A_{\chi_j}=\omega_q^jA\),
  \[
    \Gamma_{\chi_j}=\{\omega_q^j\},\qquad
    s_{\chi_0}=1,\qquad s_{\chi_j}=0\quad(j\ne0),
  \]
  and every congruence class \(C\subset\ZZ/q\ZZ\) has
  \[
    \delta_C=\frac1q .
  \]
  Consequently the canonical cyclic lift is a genuine Perron-boundary case
  of \Cref{thm:perron-boundary-determinant-product}: its constants are obtained
  from the \(q\) root-of-unity reduced determinant layers
  \[
    \mathcal B_{\chi_j}^{\mathrm{ren}}
    =
    \operatorname*{Abel\,FP}_{t\uparrow1}
      \Log\det(I-t\omega_q^j\lambda^{-1}A)^{-1},
    \qquad 0\le j<q,
  \]
  together with the ordinary \(k\ge2\) interior determinant logarithms in
  \eqref{eq:renormalized-fixed-label-determinant}.
\end{remark}

\begin{proof}
  Since every edge label is \(1\in\ZZ/q\ZZ\), twisting by \(\chi_j\)
  multiplies every edge weight by \(\omega_q^j\), hence
  \(A_{\chi_j}=\omega_q^jA\).  The primitive base matrix \(A\) has the
  single peripheral eigenvalue \(\lambda\), so \(A_{\chi_j}\) has the
  single peripheral eigenvalue \(\lambda\omega_q^j\).  This proves the
  displayed \(\Gamma\)-sets and the values of \(s_{\chi_j}\).  The
  character expansion for a singleton class in the abelian group gives
  \(|C|/|G|=1/q\) and only \(\chi_0\) contributes to the sum defining
  \(\delta_C\); hence \(\delta_C=1/q\).  The determinant-layer statement is
  now exactly the \(k=1\) part of
  \Cref{prop:renormalized-fixed-label-determinant-component}; all \(k\ge2\) terms lie
  inside the Perron disc and are unchanged.
\end{proof}

\begin{remark}[Corollary D, Artin obstruction component]
  \label{thm:renormalized-artin-obstruction}
  Under the Perron-boundary algebraic multiplicity data above, define
  \[
    \mathcal L_\varrho^{\mathrm{ren}}
    :=
    \lim_{t\uparrow1}
      \left(\mathcal L_\varrho(t\lambda^{-1})
      +s_\varrho\log(1-t)\right),
  \]
  using the same reduced finite-part convention as in
  \Cref{prop:renormalized-fixed-label-determinant-component}.  Let
  \(K_C^{\mathrm{Art,ren}}\) be the constants obtained from
  \eqref{eq:renormalized-frobenius-product-constant} after replacing
  every \(F_\varrho^{\mathrm{ren}}\) by
  \(\mathcal L_\varrho^{\mathrm{ren}}\).  Then
  \[
    \log\frac{K_C^{\mathrm{ren}}}{K_C^{\mathrm{Art,ren}}}
    =
    -\frac{|C|}{|G|}
      \sum_{\varrho\in\Irr(G)}
        \chi_\varrho^\ast(C)
        \left(F_\varrho^{\mathrm{ren}}
        -\mathcal L_\varrho^{\mathrm{ren}}\right).
  \]
  Hence the Artin-substituted constants agree with the fixed-label
  constants for every conjugacy class if and only if all non-trivial
  renormalized Peter--Weyl discrepancies vanish:
  \[
    F_\varrho^{\mathrm{ren}}
    =
    \mathcal L_\varrho^{\mathrm{ren}}
    \qquad(\varrho\ne\mathbf1).
  \]
\end{remark}

\begin{proof}
  The logarithmic ratio is obtained by subtracting the two defining
  logarithms.  The scalar \(s_\varrho\gamma\) terms are identical in the two
  constants, so only the displayed discrepancy vector remains.  If all
  non-trivial discrepancies vanish, the two constants are plainly equal for
  all classes.  Conversely, equality for all classes says that the class
  function
  \[
    C\longmapsto
    \sum_{\varrho\in\Irr(G)}
      \chi_\varrho^\ast(C)
      (F_\varrho^{\mathrm{ren}}
      -\mathcal L_\varrho^{\mathrm{ren}})
  \]
  vanishes.  The trivial coordinate is fixed by the common scalar
  normalization, and the irreducible characters are an orthonormal basis of
  class functions.  Therefore every non-trivial Fourier coefficient of this
  class function is zero, which is exactly the stated vanishing of the
  renormalized discrepancies.
\end{proof}
