\subsection{Odd-Adams-invariant abelian two-group extensions}
\label{sec:odd-adams-sampling}

The dyadic reduction extends beyond elementary two-groups precisely when the
odd Adams directions are constant.  We first record the two algebraic facts
needed for that extension.

\begin{lemma}[Odd-Adams integrality and M\"obius collapse]
\label{lem:odd-adams-mobius-collapse}
Let \(G\) be a finite abelian \(2\)-group and let
\(\tau:E(\Gamma)\to G\) and \(\tau':E(\Gamma')\to G\) be one-step
cocycles.  For \(\chi\in\widehat G\), write
\[
 P_{\chi,\tau}(z)=\det(I-zA_{\chi,\tau}),\qquad
 P_{\chi,\tau'}(z)=\det(I-zA_{\chi,\tau'}),
\]
and assume, separately for the two cocycles, the condition denoted (OA):
\begin{equation}\label{eq:odd-adams-determinant-invariance}
 \begin{aligned}
 P_{\chi^u,\tau}(z)&=P_{\chi,\tau}(z),\\
 P_{\chi^u,\tau'}(z)&=P_{\chi,\tau'}(z)
 \end{aligned}
 \qquad(\chi\in\widehat G,\ u\ \text{odd}).
\end{equation}
Then \(P_{\chi,\tau},P_{\chi,\tau'}\in\ZZ[z]\) for every \(\chi\).
If
\[
 H_\chi(z):=\frac{P_{\chi,\tau}(z)}{P_{\chi,\tau'}(z)},\qquad
 \Delta_\chi(z):=F_\chi^\tau(z)-F_\chi^{\tau'}(z),
\]
then, on the common open Perron disk,
\begin{equation}\label{eq:odd-adams-mobius-collapse}
 \Delta_\chi(z)
 =-\sum_{j\ge0}2^{-j}\Log_0H_\chi(z^{2^j})
  +\sum_{j\ge1}2^{-j}\Log_0H_{\chi^2}(z^{2^j}).
\end{equation}
\end{lemma}

\begin{proof}
Suppose that \(\chi\) has order \(2^a\), and put
\(K=\QQ(\chi(G))\).  The entries of \(A_{\chi,\tau}\) lie in the ring of
integers of \(K\), so the coefficients of \(P_{\chi,\tau}\) are algebraic
integers.  Every element of \(\operatorname{Gal}(K/\QQ)\) acts on the values
of \(\chi\) by \(\chi(g)\mapsto\chi(g)^u\) for some odd \(u\), and therefore
sends \(P_{\chi,\tau}\) to \(P_{\chi^u,\tau}\).  Condition
\eqref{eq:odd-adams-determinant-invariance} fixes this polynomial under the
whole Galois group.  Its coefficients are rational algebraic integers, hence
integers.  The same argument applies to \(\tau'\).

For the collapse, the abelian specialization of
\eqref{eq:fixed-determinant-open-disc} gives the absolutely convergent identity
\begin{align}
 \Delta_\chi(z)
 &=-\sum_{r,m\ge1}\frac{\mu(m)}{rm}
      \Log_0H_{\chi^m}(z^{rm})\notag\\
 &=-\sum_{k\ge1}\frac1k\sum_{m\mid k}\mu(m)
      \Log_0H_{\chi^m}(z^k).
 \label{eq:odd-adams-collapse-before-divisors}
\end{align}
Write \(k=2^jq\), where \(q\) is odd.  The odd divisors contributing to the
inner sum have the form \(m=d\), \(d\mid q\), and
\[
 \sum_{\substack{m\mid k\\m\ {\rm odd}}}\mu(m)
 =\sum_{d\mid q}\mu(d)
 =\begin{cases}1,&q=1,\\0,&q>1.\end{cases}
\]
If \(j\ge1\), the square-free even divisors are exactly \(m=2d\), with
\(d\mid q\) odd, and
\[
 \sum_{\substack{m\mid k\\m\ {\rm even}}}\mu(m)
 =\sum_{d\mid q}\mu(2d)
 =-\sum_{d\mid q}\mu(d).
\]
For odd \(d\), condition (OA) gives
\[
 H_{\chi^d}=H_\chi,\qquad
 H_{\chi^{2d}}=H_{(\chi^2)^d}=H_{\chi^2}.
\]
Thus the inner divisor sum in
\eqref{eq:odd-adams-collapse-before-divisors} vanishes unless \(q=1\);
for \(k=1\) it is \(\Log_0H_\chi(z)\), and for \(k=2^j\), \(j\ge1\), it is
\[
 \Log_0H_\chi(z^{2^j})-\Log_0H_{\chi^2}(z^{2^j}).
\]
Substitution proves \eqref{eq:odd-adams-mobius-collapse}.  This is an
identity of analytic germs, and hence throughout the common Perron disk.
\end{proof}

\begin{theorem}[Odd-Adams-invariant finite radial determination]
\label{thm:odd-adams-abelian-two-group-finite-sampling}
Let \(G\) be a finite abelian \(2\)-group.  Let
\(\tau:E(\Gamma)\to G\) and \(\tau':E(\Gamma')\to G\) be one-step
cocycles over primitive non-negative integral base matrices \(A,A'\), whose
sizes are at most \(V\) and whose Perron roots are
\(\lambda,\lambda'>1\).  Assume that both cocycles satisfy (OA) in
\eqref{eq:odd-adams-determinant-invariance}.  Put
\[
 x=\min\{\lambda^{-1},(\lambda')^{-1}\},\qquad
 M(V)=2V\lceil\log_2(4V)\rceil.
\]
For \(g\in G\), define
\[
 \mathcal E_g^\tau(y)
 =\sum_{n\ge1}p_{n,g}(\tau)\log(1-y^n),\qquad 0<y<x,
\]
and define \(\mathcal E_g^{\tau'}\) analogously.  Suppose that
\begin{equation}\label{eq:odd-adams-sampled-profile-equality}
 \mathcal E_g^\tau(y_i)=\mathcal E_g^{\tau'}(y_i)
 \qquad(g\in G,\ 1\le i\le M(V))
\end{equation}
at \(M(V)\) distinct radii in \((0,x)\), at least one of which is algebraic.
Then
\[
 p_{n,g}(\tau)=p_{n,g}(\tau')
 \qquad(n\ge1,\ g\in G).
\]
The number \(M(V)\) of radial locations is independent of the rank and
exponent of \(G\), while the total number of scalar equalities in
\eqref{eq:odd-adams-sampled-profile-equality} is \(\abs G M(V)\).
Neither a common base nor a strict twisted spectral gap is required.
\end{theorem}

\begin{proof}
All profile series converge absolutely on the stated interval by the
periodic-point estimate in \Cref{thm:finite-radial-sampling}.  Summing
\eqref{eq:odd-adams-sampled-profile-equality} over \(g\) gives
\[
 \log\det(I-y_iA)=\log\det(I-y_iA')
 \qquad(1\le i\le M(V)).
\]
The determinants are positive on \((0,x)\).  Their difference has degree at
most \(V\), vanishes at the sampled radii and at zero, and hence is zero:
\begin{equation}\label{eq:odd-adams-base-determinant-equality}
 P_{\mathbf1,\tau}(z)=\det(I-zA)=\det(I-zA')
 =P_{\mathbf1,\tau'}(z).
\end{equation}
Thus \(H_{\mathbf1}=1\).

Fourier transformation of the sampled element-profile equalities gives
\begin{equation}\label{eq:odd-adams-sampled-fourier-equality}
 \Delta_\chi(y_i)
 =-\sum_{g\in G}\chi(g)
   \bigl(\mathcal E_g^\tau(y_i)-\mathcal E_g^{\tau'}(y_i)\bigr)
 =0
\end{equation}
for every \(\chi\in\widehat G\) and every \(i\).  We prove
\(H_\chi=1\) by induction on the order of \(\chi\).  The order-one case is
\eqref{eq:odd-adams-base-determinant-equality}.  Assume the assertion for
all characters of order less than \(2^a\), and let \(\chi\) have order
\(2^a\).  Then \(H_{\chi^2}=1\), so
\Cref{lem:odd-adams-mobius-collapse} reduces to
\begin{equation}\label{eq:odd-adams-inductive-dyadic-orbit}
 \Delta_\chi(y)
 =-\sum_{j\ge0}2^{-j}\log H_\chi(y^{2^j})
 \qquad(0<y<x).
\end{equation}

By \Cref{lem:odd-adams-mobius-collapse}, \(H_\chi\in\QQ(z)^\times\).
Entrywise,
\[
 \abs{(A_{\chi,\tau})_{uv}}\le A_{uv},\qquad
 \abs{(A_{\chi,\tau'})_{uv}}\le A'_{uv}.
\]
Perron-weighted domination therefore makes both character determinants
zero-free on their open Perron disks.  Since they are real polynomials with
value one at zero, they and \(H_\chi\) are positive on \((0,x)\).

Choose an algebraic sampled radius \(y_0\).  Equations
\eqref{eq:odd-adams-sampled-fourier-equality} and
\eqref{eq:odd-adams-inductive-dyadic-orbit} give
\[
 \prod_{j\ge0}H_\chi(y_0^{2^j})^{2^{-j}}=1.
\]
If \(H_\chi\ne1\), \Cref{thm:general-rational-p-mahler-lifting} with \(p=2\)
gives the unique normalized \(R_\chi\in\QQ(z)^\times\) such that
\[
 H_\chi(z)=\frac{R_\chi(z^2)}{R_\chi(z)^2},\qquad R_\chi(y_0)=1.
\]
Write \(H_\chi=P_{0,\chi}/P_{1,\chi}\) in reduced form and put
\(D_\chi=\deg P_{0,\chi}+\deg P_{1,\chi}\).  Both determinants have degree
at most \(V\), so \(D_\chi\le2V\).  The rational telescope and
\Cref{prop:logarithmic-mahler-divisor-bound} show that every real zero of
\(\Delta_\chi\) is a non-zero root of \(R_\chi-1\), and that there are at
most
\[
 b_2(D_\chi)-1
 =D_\chi\lceil\log_2(2D_\chi)\rceil-1
 \le M(V)-1
\]
such roots.  This contradicts the \(M(V)\) distinct zeros in
\eqref{eq:odd-adams-sampled-fourier-equality}.  Hence \(H_\chi=1\), closing
the induction.

All character determinants now agree between the two systems.  The
cross-base periodic-data dictionary in \Cref{prop:comparison-dictionary},
followed by Fourier inversion on \(G\), recovers every
\(p_{n,g}\), as asserted.
\end{proof}

\begin{remark}[Concrete scope of (OA)]
\label{rem:odd-adams-concrete-scope}
Condition (OA) is automatic for \(G=(C_2)^r\).  A stronger entrywise
sufficient condition for a general abelian \(2\)-group is that, for every
ordered vertex pair, every \(g\in G\), and every odd \(u\), the number of
\(g\)-labelled edges equals the number of \(g^u\)-labelled edges.  It then
gives \(A_{\chi^u}=A_\chi\), not merely equality of their determinants.

For \(C_4=\langle a\rangle\), this condition says exactly that the
\(a\)- and \(a^{-1}\)-labelled edge counts agree entrywise; the \(e\)- and
\(a^2\)-counts are unrestricted.  It does not force factorization through
\(C_2\): a one-vertex graph with equally many \(a\)- and \(a^{-1}\)-loops
has primitive base and holonomy subgroup \(C_4\).  Likewise, assigning equal
counts on each odd-power orbit in \(C_{2^s}\) gives examples with genuine
holonomy of order \(2^s\).
\end{remark}
