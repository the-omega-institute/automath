\section{Appendix: reproducibility record for the \texorpdfstring{\(S_3\)}{S3} fixed-label witness}
\label{app:fixed-label-local-certificates}

This appendix is a reproducibility appendix.  It records how the fixed-label
finite-part algorithm can be checked by the finite rational enclosures in the
exact \(S_3\) certificate of \Cref{app:s3-log-certificates}.  The main text
imports only the determinant model and the four rational inequalities collected
in \Cref{cor:s3-displayed-rational-certificate-windows}; the finite enclosure
format below is not a separate theorem-level input to the Perron-boundary
product theorem.

\begin{lemma}[Reproducibility enclosure data for fixed labels]\label{prop:fixed-label-rational-certificate-record}
\label{prop:fixed-label-finite-io-certificate-summary}
  Assume the hypotheses of \Cref{thm:class-mertens-explicit}.  Fix a
  non-trivial \(\varrho\in\Irr(G)\), a tolerance \(\epsilon>0\), and a
  truncation level \(M\) satisfying
  \eqref{eq:fixed-label-perron-finite-algorithm-tail-choice}.  The enclosure
  produced by \Cref{prop:fixed-label-perron-finite-algorithm} is reproduced by
  the following finite data:
  \begin{enumerate}[label=\textup{(\roman*)},leftmargin=*]
    \item the finite list of triples \((r,m,\pi)\) with \(rm\le M\), with the
      scalar Perron triple \((1,1,\mathbf1)\) deleted as in
      \Cref{lem:fixed-label-singular-triple-deletion};
    \item the Adams coefficients
      \(a_{\varrho,\pi}^{(m)}
      =\langle\psi^m\chi_\varrho,\chi_\pi\rangle_G\), computed from the
      character table and class-power table;
    \item for every remaining triple, the branch tag specifying either the
      Abel non-trivial Perron branch \((rm=1,\pi\ne\mathbf1)\) or the ordinary
      germ value inside the Perron disc \((rm\ge2)\), together with a rational
      rectangular enclosure for
      \(\Log\det(I-\lambda^{-rm}A_\pi)^{-1}\);
    \item rational bounds \(R_M\) and \(R_{\mathrm{num}}\) for the discarded
      analytic tail and finite interval-propagation error.
  \end{enumerate}
  These data verify the rational inclusion
  \begin{equation}\label{eq:fixed-label-certificate-interface-inclusion}
    F_\varrho(\lambda^{-1})
    \in
    I_M+\{w\in\CC:\abs{w}\le R_M+R_{\mathrm{num}}\},
  \end{equation}
  where \(I_M\) is the rational rectangular interval obtained by signed
  interval summation over the listed triples.
\end{lemma}

\begin{proof}
  Finiteness of the triple list and of the Adams coefficients is immediate
  from \(rm\le M\), the finite group \(G\), and the finite character table.  The
  deleted scalar triple is exactly the singular term omitted in
  \Cref{lem:fixed-label-singular-triple-deletion,prop:fixed-label-perron-finite-algorithm}.
  Each non-deleted branch tag is one of the two branches already specified in
  \Cref{prop:fixed-label-perron-finite-algorithm}: the strict twisted gap makes
  the non-trivial \(rm=1\) Abel branch regular, and \(rm\ge2\) lies strictly
  inside the Perron disc.  Signed interval summation gives \(I_M\), and the two
  rational radii are exactly the bounds in
  \eqref{eq:fixed-label-perron-finite-algorithm-error} and
  \eqref{eq:fixed-label-perron-finite-algorithm-roundoff}.  This proves
  \eqref{eq:fixed-label-certificate-interface-inclusion}.
\end{proof}

\begin{lemma}[Reproducibility branch--tail normal form for the fixed-label example]\label{prop:fixed-label-certificate-normal-form}
\label{prop:fixed-label-product-certificate-record}
\label{prop:fixed-label-product-enclosure-check}
\label{prop:fixed-label-five-field-rational-record}
  For the strict-gap fixed-label enclosures used in the \(S_3\) witness, the
  finite record has five fields:
  \begin{enumerate}[label=\textup{(\roman*)},leftmargin=*]
    \item the determinant model \((D_\pi(z))_{\pi\in\Irr(G)}\), the Perron root
      \(\lambda\), the verified gap \(\eta<\lambda\), the character table, and
      the class-power maps;
    \item rational majorant data \(C_\varrho,Q_\varrho,\theta,M\) satisfying
      \(\lambda^{-1}<\theta<1\) and the tail condition
      \eqref{eq:fixed-label-perron-finite-algorithm-rational-tail-choice};
    \item the finite list of triples \((r,m,\pi)\) with \(rm\le M\), the
      scalar Perron triple deleted, and the Adams coefficients on that list;
    \item the branch tag and rational determinant-logarithm enclosure for each
      non-deleted triple;
    \item the rational finite-propagation radius and the rational tail radius
      \(R_{M,\varrho}=Q_\varrho\theta^{M+1}/(1-\theta)\).
  \end{enumerate}
  These fields give the inclusion \eqref{eq:fixed-label-certificate-interface-inclusion}
  for each non-trivial \(\varrho\).  Applying the finite character-table
  Fourier formula \eqref{eq:class-product-fourier-coordinates} then gives the
  corresponding rational enclosures for the Frobenius-class product constants
  \((K_C)_C\).
\end{lemma}

\begin{proof}
  This is \Cref{prop:fixed-label-rational-certificate-record} specialized to
  the determinant model and branch choices of the \(S_3\) example.  The first
  four fields determine the finite interval sum \(I_\varrho\); the fifth field
  supplies the two rational radii.  The inclusion follows from
  \eqref{eq:fixed-label-certificate-interface-inclusion}.  The passage from
  the fixed-label vector to the class-product constants is the finite
  Peter--Weyl readout
  \eqref{eq:class-product-fourier-coordinates}; no additional analytic
  estimate is introduced at that stage.
\end{proof}
