\begin{proposition}[Fixed-label determinant component of the boundary theorem]
  \label{prop:renormalized-fixed-label-determinant-component}
  Under the peripheral algebraic multiplicity hypotheses of
  \Cref{def:peripheral-peter-weyl-data,lem:finite-cover-peripheral-expansion},
  fix the branch-normalized boundary datum
  \(D_{\partial}^{\mathrm{ren}}\) from
  \eqref{eq:perron-boundary-renormalized-data-local}.  For
  \(\pi\in\Irr(G)\), define the reduced boundary determinant logarithm
  \begin{equation}\label{eq:renormalized-reduced-boundary-log}
    \mathcal B_{1,\pi}^{\mathrm{ren}}
    :=
    \lim_{t\uparrow1}
      \left(
        \Log_0\det(I-t\lambda^{-1}A_\pi)^{-1}
        +s_\pi\log(1-t)
      \right).
  \end{equation}
  Equivalently,
  \[
    \mathcal B_{1,\pi}^{\mathrm{ren}}
    =
    -\sum_{\substack{\zeta\in\Gamma_\pi\\\zeta\ne1}}
      m_{\pi,\zeta}\Log(1-\zeta)
    +
    \Log C_{\partial,\pi},
  \]
  where \(C_{\partial,\pi}\) is the product of
  \((1-\alpha/\lambda)^{-1}\) over the non-peripheral eigenvalues
  \(\alpha\) of \(A_\pi\), counted with algebraic multiplicity.  If
  \(1\in\Gamma_\pi\), this is the reduced determinant finite part
  \(\Log C_\lambda(A_\pi)\) after deleting the factor
  \((1-t)^{s_\pi}\); if \(1\notin\Gamma_\pi\), it is the ordinary Abel
  boundary value with the non-real peripheral factors retained.

  This is the \(k=1\) instance of
  \Cref{def:perron-boundary-logarithm-convention}.  For every
  \(\varrho\in\Irr(G)\), define
  \begin{equation}\label{eq:renormalized-fixed-label-def}
    F_\varrho^{\mathrm{ren}}
    :=
    \lim_{t\uparrow1}
      \left(F_\varrho(t\lambda^{-1})+s_\varrho\log(1-t)\right).
  \end{equation}
  Then the limit exists and is given by the branch-compatible
  Adams--M\"obius determinant formula
  \begin{equation}\label{eq:renormalized-fixed-label-determinant}
    F_\varrho^{\mathrm{ren}}
    =
    \sum_{r,m\ge1}\frac{\mu(m)}{rm}
      \sum_{\pi\in\Irr(G)}
        a_{\varrho,\pi}^{(m)}\,
        B_{rm,\pi}^{\mathrm{ren}},
  \end{equation}
  where the values \(\mathcal B_{rm,\pi}^{\mathrm{ren}}\) are those of
  \Cref{def:perron-boundary-logarithm-convention}.  Thus
  \(\mathcal B_{1,\pi}^{\mathrm{ren}}\) and
  \(\mathcal B_{k,\pi}^{\mathrm{ren}}\) are computed from the finite
  branch-normalized input \(D_{\partial}^{\mathrm{ren}}\): the determinant
  polynomial and recorded peripheral factorisation supply the factors, while
  the branch families \(\{\Log_{\pi,\zeta}\}\) and
  \(\{\Log_{\partial,\pi}\}\) specify the Abel boundary values used below.
  Equivalently, the branch-normalized algorithm is:
  factor \(P_\pi(t/\lambda)\), isolate the factors
  \((1-t\zeta)^{m_{\pi,\zeta}}\) with \(\zeta\in\Gamma_\pi\), cancel the
  \(s_\pi=m_{\pi,1}\) copies of \(1-t\), take the Abel limit along
  \(0\le t<1\) using those branch families, and use the ordinary power-series
  germ value of \(\Log_0 P_\pi(\lambda^{-k})^{-1}\) for \(k\ge2\).  Thus every \(k=1\)
  Perron-boundary block is evaluated by its reduced finite part, every
  \(k\ge2\) term remains an ordinary interior determinant logarithm, and no
  infinite sequence \(\{B_{k,\pi}^{\mathrm{ren}}\}\) is supplied as input.
\end{proposition}

\begin{proof}
  The determinant logarithm decomposes spectrally as
  \[
    \Log\det(I-t\lambda^{-1}A_\pi)^{-1}
    =
    \sum_{\zeta\in\Gamma_\pi}
      -m_{\pi,\zeta}\Log(1-t\zeta)
    +H_\pi(t),
  \]
  where \(H_\pi(t)\) is analytic at \(t=1\), because all non-peripheral
  eigenvalues satisfy \(|\alpha|<\lambda\).  This factorisation uses only
  the characteristic polynomial, so the exponents are algebraic
  multiplicities; nilpotent parts of peripheral Jordan blocks do not change
  the determinant factors.  Adding
  \(s_\pi\log(1-t)\) cancels precisely the \(\zeta=1\) logarithm, while
  the \(\zeta\ne1\) terms have finite Abel limits.  This proves
  \eqref{eq:renormalized-reduced-boundary-log}.

  For \(0<t<1\), the open-disc fixed-label identity
  \eqref{eq:fixed-label-euler-transform-adams-mobius}, followed by
  determinant linearisation, gives
  \[
    F_\varrho(t\lambda^{-1})
    =
    \sum_{r,m\ge1}\frac{\mu(m)}{rm}
      \sum_{\pi\in\Irr(G)}
        a_{\varrho,\pi}^{(m)}
        \Log\det(I-t^{rm}\lambda^{-rm}A_\pi)^{-1}.
  \]
  The only terms that can fail to be ordinary interior values at
  \(t=1\) have \(rm=1\), hence \(r=m=1\).  For those terms the Adams
  coefficient is \(a_{\varrho,\pi}^{(1)}=\delta_{\varrho\pi}\), so the
  logarithmic singular coefficient is exactly \(s_\varrho\log(1-t)\).
  Adding this term converts the whole \(rm=1\) boundary contribution into
  \(\mathcal B_{1,\varrho}^{\mathrm{ren}}\).  Every \(rm\ge2\) term is inside
  the Perron disc.  The coefficient family
  \(c_{r,m,\pi}=a_{\varrho,\pi}^{(m)}\) is bounded, because the class-power
  table is finite and \(G\) has finitely many irreducibles.  Applying
  \Cref{lem:perron-boundary-fixed-label-majorant} gives the required
  \(t\)-independent absolute majorant for the \(rm\ge2\) tail and the
  reduced Abel limits for the finitely many \(rm=1\) terms.  Dominated
  convergence therefore gives
  \eqref{eq:renormalized-fixed-label-determinant}.
\end{proof}

\begin{lemma}[Contragredient reality and product positivity]
  \label{lem:renormalized-reality-positivity}
  Under the Perron-boundary algebraic multiplicity data above, the renormalized
  coordinates satisfy
  \[
    s_{\varrho^\vee}=s_\varrho,\qquad
    R_{\varrho^\vee}=\overline{R_\varrho},\qquad
    F_{\varrho^\vee}^{\mathrm{ren}}
      =\overline{F_\varrho^{\mathrm{ren}}}
  \]
  for every \(\varrho\in\Irr(G)\).  Consequently, for every conjugacy class
  \(C\subset G\), the sums defining
  \(\delta_C\), \(M_C^{\mathrm{ren}}\), and \(K_C^{\mathrm{ren}}\) are real,
  \(\delta_C\ge0\), and \(K_C^{\mathrm{ren}}>0\).
  The conjugations are taken with the boundary logarithm convention of
  \Cref{def:perron-boundary-logarithm-convention}.
\end{lemma}

\begin{proof}
  Choose the unitary model used in \eqref{eq:twisted-matrix-def}.  The
  contragredient block is obtained by conjugating each matrix coefficient:
  \(A_{\varrho^\vee}\) is entrywise conjugate to \(A_\varrho\), after the
  canonical identification \(V_{\varrho^\vee}=\overline{V_\varrho}\).
  Hence the peripheral spectrum and algebraic multiplicities are conjugate:
  if \(\lambda\zeta\) occurs in \(A_\varrho\) with multiplicity
  \(m_{\varrho,\zeta}\), then
  \(\lambda\overline\zeta\) occurs in \(A_{\varrho^\vee}\) with the same
  multiplicity.  In particular
  \(s_{\varrho^\vee}=m_{\varrho^\vee,1}=m_{\varrho,1}=s_\varrho\).

  For \(0<t<1\), the primitive trace series has real coefficients paired by
  contragredients:
  \[
    \Pi_{\varrho^\vee}(t\lambda^{-1})
    =
    \overline{\Pi_\varrho(t\lambda^{-1})}.
  \]
  Since \(s_{\varrho^\vee}=s_\varrho\) and \(\log(1-t)\) is real on the Abel
  path, taking the reduced finite-part limit in
  \eqref{eq:renormalized-primitive-channel-finite-part} gives
  \(R_{\varrho^\vee}=\overline{R_\varrho}\).  The same argument applies to
  the fixed-label Euler transform:
  \(F_{\varrho^\vee}(t\lambda^{-1})
  =\overline{F_\varrho(t\lambda^{-1})}\) for \(0<t<1\), and the defining
  finite-part limit \eqref{eq:renormalized-fixed-label-def} gives
  \(F_{\varrho^\vee}^{\mathrm{ren}}
  =\overline{F_\varrho^{\mathrm{ren}}}\).

  The Hermitian class coefficient also conjugates under duality:
  \[
    \chi_{\varrho^\vee}^\ast(C)
    =
    \overline{\chi_\varrho^\ast(C)}.
  \]
  Thus each non-self-dual pair
  \(\{\varrho,\varrho^\vee\}\) contributes twice the real part to the sums
  defining \(\delta_C\), \(M_C^{\mathrm{ren}}\), and
  \(\log K_C^{\mathrm{ren}}\), while a self-dual summand is real by the same
  identities.  This proves the asserted reality.

  It remains to record the signs.  For every \(N\),
  \(\sum_{n\le N}p_{n,C}\lambda^{-n}\ge0\).  Divide
  \eqref{eq:renormalized-class-mertens-log} by \(\log N\) and let
  \(N\to\infty\); the bounded constant and \(O(N^{-1})\) terms vanish, so
  \(\delta_C\ge0\).  Finally,
  \[
    \log K_C^{\mathrm{ren}}
    =
    -\frac{|C|}{|G|}
      \sum_{\varrho\in\Irr(G)}
        \chi_\varrho^\ast(C)
        \bigl(s_\varrho\gamma+F_\varrho^{\mathrm{ren}}\bigr)
  \]
  is real by the preceding paragraph.  Therefore
  \(K_C^{\mathrm{ren}}=\exp(\log K_C^{\mathrm{ren}})>0\).
\end{proof}

\begin{lemma}[Perron-boundary product-tail reduction]
  \label{lem:perron-boundary-product-tail-reduction}
  Under the peripheral algebraic multiplicity hypotheses of
  \Cref{def:peripheral-peter-weyl-data,lem:finite-cover-peripheral-expansion},
  every conjugacy class \(C\subset G\) satisfies
  \begin{equation}\label{eq:perron-boundary-tail-class-expansion}
    p_{n,C}
    =
    \frac{|C|}{|G|}
    \frac{\lambda^n}{n}
      \sum_{\varrho\in\Irr(G)}
        \chi_\varrho^\ast(C)
        \sum_{\zeta\in\Gamma_\varrho}
          m_{\varrho,\zeta}\zeta^n
    +O(\Lambda_\partial^n/n).
  \end{equation}
  Consequently the primitive class counts satisfy a class-uniform bound
  \begin{equation}\label{eq:perron-boundary-primitive-class-bound}
    0\le p_{n,C}\le B_{\partial}\frac{\lambda^n}{n}
    \qquad(n\ge1,\ C\subset G),
  \end{equation}
  for a constant \(B_{\partial}\) depending only on the finite determinant
  model.  For every conjugacy class \(C\),
  \begin{equation}\label{eq:perron-boundary-product-tail-omitted}
    \sum_{r\ge2}\frac1r\sum_{n>N}p_{n,C}\lambda^{-rn}
    =
    O(\lambda^{-N}/N),
  \end{equation}
  uniformly in \(C\).  Moreover
  \begin{equation}\label{eq:perron-boundary-tail-class-mertens}
    \sum_{n\le N}\frac{p_{n,C}}{\lambda^n}
    =
    \delta_C\log N+M_C^{\mathrm{ren}}+O(N^{-1}),
  \end{equation}
  with \(\delta_C\) and \(M_C^{\mathrm{ren}}\) as in
  \eqref{eq:renormalized-class-density} and
  \eqref{eq:renormalized-class-mertens-log}.
  Finally, the repetition tail
  \[
    T_C(N):=
      \sum_{r\ge2}\frac1r\sum_{n\le N}p_{n,C}\lambda^{-rn}
  \]
  has an absolutely convergent endpoint
  \[
    T_C(\infty):=
      \sum_{r\ge2}\frac1r\sum_{n\ge1}p_{n,C}\lambda^{-rn}
  \]
  and
  \begin{equation}\label{eq:perron-boundary-product-tail-truncation}
    T_C(N)=T_C(\infty)+O(\lambda^{-N}/N),
  \end{equation}
  uniformly in \(C\).  Moreover,
  \begin{equation}\label{eq:perron-boundary-product-tail-fourier}
    T_C(\infty)
    =
    \frac{|C|}{|G|}
      \sum_{\varrho\in\Irr(G)}
        \chi_\varrho^\ast(C)
        \sum_{r\ge2}\frac{\Pi_\varrho(\lambda^{-r})}{r},
  \end{equation}
  where the \(r\ge2\) series is absolutely convergent.
\end{lemma}

\begin{proof}
  Apply the class-indicator expansion
  \eqref{eq:class-primitive-coefficients} to the primitive-channel estimate
  of \Cref{prop:peripheral-primitive-channel-asymptotic-component}.  The
  irreducible character set is finite, so the finitely many
  \(O(\Lambda_\partial^n/n)\) errors remain
  \(O(\Lambda_\partial^n/n)\) after multiplication by
  \((|C|/|G|)\chi_\varrho^\ast(C)\) and summation over \(\varrho\).  This
  proves \eqref{eq:perron-boundary-tail-class-expansion}.  Since
  \(|\zeta|=1\), \(\Lambda_\partial<\lambda\), and there are only finitely
  many peripheral phases and irreducible characters,
  \[
    \abs{p_{n,C}}\le B_{\partial}\frac{\lambda^n}{n}
  \]
  after enlarging \(B_{\partial}\) over the finite set of classes.  The lower
  bound is the definition of \(p_{n,C}\) as a primitive-orbit count.  This
  proves \eqref{eq:perron-boundary-primitive-class-bound}.

  Hence the omitted part of the repetition tail is bounded by
  \(B_{\partial}\) times
  \[
    \sum_{r\ge2}\frac1r\sum_{n>N}\frac{\lambda^{(1-r)n}}{n}.
  \]
  Since \(n\ge N+1\) in the inner sum,
  \[
    \sum_{r\ge2}\frac1r\sum_{n>N}\frac{\lambda^{(1-r)n}}{n}
    \le
    \frac1{N+1}
    \sum_{r\ge2}
      \frac{\lambda^{(1-r)(N+1)}}{r(1-\lambda^{1-r})}.
  \]
  With \(q=\lambda^{-1}<1\), the right-hand side is at most
  \[
    \frac1{N+1}
    \frac{1}{2(1-q)}
      \sum_{s\ge1}q^{s(N+1)}
    =
    O(\lambda^{-N}/N),
  \]
  uniformly in \(C\).  This proves
  \eqref{eq:perron-boundary-product-tail-omitted}.  It also proves the
  absolute convergence of \(T_C(\infty)\) and the truncation estimate
  \eqref{eq:perron-boundary-product-tail-truncation}.

  The summatory formula
  \eqref{eq:perron-boundary-tail-class-mertens} is exactly
  \eqref{eq:renormalized-class-mertens-log}, already obtained from the same
  primitive-channel estimate in
  \Cref{prop:renormalized-class-mertens-component}.  Explicitly, summing the
  absolutely summable channel remainders and the finite oscillatory
  \(\zeta\ne1\) tails gives \(O(N^{-1})\), while the \(\zeta=1\) terms
  contribute \(\delta_C H_N\); substituting
  \(H_N=\log N+\gamma+O(N^{-1})\) gives the displayed
  \(M_C^{\mathrm{ren}}\).

  For each \(r\ge2\), the point \(\lambda^{-r}\) lies strictly inside the
  primitive-series disc.  Thus
  \[
    \sum_{n\ge1}p_{n,C}\lambda^{-rn}
    =
    \frac{|C|}{|G|}
      \sum_{\varrho\in\Irr(G)}
        \chi_\varrho^\ast(C)\Pi_\varrho(\lambda^{-r})
  \]
  by \eqref{eq:class-primitive-generating-function}.  The bound just proved,
  together with the corresponding channel bound
  \(\abs{c_{n,\varrho}(A,\tau)}\le C\lambda^n/n\), gives an absolute
  majorant
  \[
    \sum_{r\ge2}\frac1r\sum_{n\ge1}\frac{\lambda^{(1-r)n}}{n}<\infty .
  \]
  Therefore the sums over \(r\), \(n\), and the finite set of irreducible
  characters may be interchanged, proving
  \eqref{eq:perron-boundary-product-tail-fourier}.
\end{proof}

\begin{proposition}[Product component of the boundary theorem]
  \label{prop:renormalized-frobenius-product-component}
  Under the peripheral algebraic multiplicity hypotheses of
  \Cref{def:peripheral-peter-weyl-data,lem:finite-cover-peripheral-expansion},
  and for the fixed branch-normalized boundary datum
  \(D_{\partial}^{\mathrm{ren}}\) from
  \eqref{eq:perron-boundary-renormalized-data-local},
  the Frobenius-class Euler product
  \[
    P_C(N):=
    \prod_{\substack{\gamma\in\cP:\ |\gamma|\le N\\
      \tau(\gamma)\in C}}
      (1-\lambda^{-|\gamma|})
  \]
  satisfies
  \begin{equation}\label{eq:renormalized-frobenius-product-component}
    P_C(N)=K_C^{\mathrm{ren}}N^{-\delta_C}\bigl(1+O(N^{-1})\bigr),
  \end{equation}
  where \(\delta_C\) is defined in
  \eqref{eq:renormalized-class-density} and
  \begin{equation}\label{eq:renormalized-frobenius-product-constant-component}
    K_C^{\mathrm{ren}}
    :=
    \exp\!\left(
      -\frac{|C|}{|G|}
        \sum_{\varrho\in\Irr(G)}
          \chi_\varrho^\ast(C)
          \bigl(s_\varrho\gamma+F_\varrho^{\mathrm{ren}}\bigr)
    \right).
  \end{equation}
  The constant is positive by
  \Cref{lem:renormalized-reality-positivity}.
  The exponent \(\delta_C\) is therefore the actual logarithmic density of
  the non-oscillatory Perron boundary, not necessarily \(|C|/|G|\).
\end{proposition}

\begin{proof}
  Taking logarithms gives
  \[
    \log P_C(N)
    =
    -\sum_{n\le N}\frac{p_{n,C}}{\lambda^n}
    -
    \sum_{r\ge2}\frac1r\sum_{n\le N}p_{n,C}\lambda^{-rn}.
  \]
  The first sum is evaluated by
  \eqref{eq:perron-boundary-tail-class-mertens}, the Mertens part of
  \Cref{lem:perron-boundary-product-tail-reduction}.  The second sum is the
  repetition tail \(T_C(N)\).  The same lemma gives
  \[
    T_C(N)
    =
    \frac{|C|}{|G|}
      \sum_{\varrho\in\Irr(G)}
        \chi_\varrho^\ast(C)
        \sum_{r\ge2}\frac{\Pi_\varrho(\lambda^{-r})}{r}
    +O(\lambda^{-N}/N).
  \]
  On the open disc, the fixed-label Euler transform separates as
  \[
    F_\varrho(t\lambda^{-1})
    =
    \Pi_\varrho(t\lambda^{-1})
    +
    \sum_{r\ge2}\frac{\Pi_\varrho(t^r\lambda^{-r})}{r}.
  \]
  The same majorant used in
  \Cref{lem:perron-boundary-product-tail-reduction} gives dominated
  convergence for the \(r\ge2\) series as \(t\uparrow1\).  Combining this
  with the renormalized \(r=1\) limit
  \(R_\varrho\) from
  \Cref{prop:renormalized-class-mertens-component} gives
  \[
    F_\varrho^{\mathrm{ren}}
    =
    R_\varrho+
    \sum_{r\ge2}\frac{\Pi_\varrho(\lambda^{-r})}{r}.
  \]
  Hence
  \[
    \log P_C(N)
    =
    -\delta_C\log N
    -\frac{|C|}{|G|}
      \sum_{\varrho\in\Irr(G)}
        \chi_\varrho^\ast(C)
        (s_\varrho\gamma+F_\varrho^{\mathrm{ren}})
    +O(N^{-1}).
  \]
  By \Cref{lem:renormalized-reality-positivity}, the exponent and constant
  term are real and \(K_C^{\mathrm{ren}}>0\).  Exponentiating with
  \Cref{lem:class-product-log-to-product-error} proves
  \eqref{eq:renormalized-frobenius-product-component}.
\end{proof}

\noindent\textbf{Hypotheses B.}
For the theorem below the standing boundary data are the following finite
objects and conventions.  Let \(e_G\) be the exponent of \(G\).  The input is
the primitive adjacency matrix \(A\) with Perron root \(\lambda>1\), the
finite group \(G\), the edge cocycle \(\tau:E_A\to G\), the twisted matrices
\(A_\pi\), the finite character table and the finite class-power table
\(\operatorname{Pow}_{G}=\{C\mapsto C^a:0\le a<e_G\}\) used to form all
Adams coefficients \(a_{\rho,\pi}^{(m)}\), the determinant polynomials
\(P_\pi(z)=\det(I-zA_\pi)\), the peripheral sets
\(\Gamma_\pi\), algebraic multiplicities \(m_{\pi,\zeta}\), values
\(s_\pi=m_{\pi,1}\), and the bounds
\[
  \Lambda_\perp
  =\max\{|\alpha|:\alpha\in\spec(A_\pi),\ \pi\in\Irr(G),\
    |\alpha|<\lambda\},
  \qquad
  \Lambda_\partial=\max\{\Lambda_\perp,\lambda^{1/2}\}<\lambda .
\]
The boundary datum also includes the factorizations
\[
  P_\pi(t/\lambda)=
  \prod_{\zeta\in\Gamma_\pi}(1-t\zeta)^{m_{\pi,\zeta}}Q_{\partial,\pi}(t),
  \qquad Q_{\partial,\pi}(1)\ne0,
\]
and the branch families \(\{\Log_{\pi,\zeta}\}\) and
\(\{\Log_{\partial,\pi}\}\): for \(0\le t<1\),
\(\Log_{\pi,\zeta}(1-t\zeta)\) is the unique continuous branch with value
\(0\) at \(t=0\), while
\(\Log_{\partial,\pi}Q_{\partial,\pi}(1)\) is obtained by continuing
\(\Log Q_{\partial,\pi}(t)\), normalized at \(0\), along the same real
segment.  For \(k\ge2\) one uses the ordinary interior value
\(\Log_0P_\pi(\lambda^{-k})^{-1}\) from the power-series branch at \(0\).
The positive scalar/product convention is canonical, namely the real
logarithm on \(\mathbb R_{>0}\) after
\Cref{lem:renormalized-reality-positivity}; it is not a separate input.
Consequently \(B_{k,\pi}^{\mathrm{ren}}\), \(R_\rho\),
\(F_\rho^{\mathrm{ren}}\), \(M_C^{\mathrm{ren}}\), and
\(K_C^{\mathrm{ren}}\) are theorem outputs computed from this finite input and
from the stated Abel/interior rule.  Effective finite truncations are the
post-theorem approximation records of
\Cref{cor:assembled-adams-frobenius-product-formula}, not extra hypotheses.
Peripheral semisimplicity is not assumed in the theorem; the optional
semisimplicity certificate in
\Cref{lem:finite-cover-peripheral-expansion} is only an additional finite
rank check identifying algebraic and geometric multiplicities.
