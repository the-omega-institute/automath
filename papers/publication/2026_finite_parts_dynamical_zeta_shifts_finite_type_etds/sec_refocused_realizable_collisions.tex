\subsection{Realizable collision and certificate lower bounds}
\label{sec:realizable-collision-lower-bounds}

We begin with a relative realization fact for sign determinants.  It is
sufficient for both the explicit collision family and the abstract
certificate-degree family.

\begin{lemma}[Relative sign-determinant realization]
\label{lem:relative-sign-determinant-realization}
Let \(P_0,P_1\in\ZZ[z]\) satisfy
\[
 P_0(0)=P_1(0)=1,\qquad P_0\equiv P_1\pmod2.
\]
Put \(d=\max\{1,\deg P_0,\deg P_1\}\).  There are a positive integral
\(d\times d\) matrix \(A\) and two one-step \(C_2\)-cocycles on the same
marked base graph with sign blocks \(C_0,C_1\) such that
\[
 \det(I-zC_i)=P_i(z)\qquad(i=0,1),
\]
and both sign blocks have strict twisted spectral gap:
\[
 \rad(C_i)<\rad(A)\qquad(i=0,1).
\]
\end{lemma}

\begin{proof}
Pad both polynomials to degree \(d\), and let \(C_i\) be the integral
companion matrix of the monic polynomial
\[
 t^dP_i(t^{-1}).
\]
Then
\[
 \det(I-zC_i)=z^d\det(z^{-1}I-C_i)=P_i(z).
\]
The congruence of the polynomials makes the two standard companion matrices
entrywise congruent modulo two.  For each entry choose a positive integer
\(A_{jk}\), in this common parity class, with
\[
 A_{jk}>\max\{\abs{(C_0)_{jk}},\abs{(C_1)_{jk}}\}.
\]
Thus \(A\) is positive and hence primitive.  For each \(i\), the non-negative
integral matrices
\[
 \frac{A+C_i}{2},\qquad \frac{A-C_i}{2}
\]
count respectively the \(e\)- and \(s\)-labelled edges of a \(C_2\)-cocycle
on this same marked base.  Finally,
\[
 \rad(C_i)\le\rad(\abs{C_i})<\rad(A)
\]
by strict Perron--Frobenius monotonicity, proving the gap.
\end{proof}

\begin{theorem}[Realizable linear multi-collisions]
\label{thm:realizable-linear-multicollisions}
For every integer \(m\ge1\), there is a positive primitive directed
multigraph with
\[
 v_m=4m+2
\]
vertices and two one-step \(C_2\)-cocycles \(\tau_m,\tau_m'\) on the same
marked base, both with strict twisted spectral gap, for which the primitive
length--element data differ but
\begin{equation}\label{eq:realizable-linear-profile-collisions}
 \mathcal E_e^{\tau_m}(y_{m,i})
 =\mathcal E_e^{\tau_m'}(y_{m,i}),\qquad
 \mathcal E_s^{\tau_m}(y_{m,i})
 =\mathcal E_s^{\tau_m'}(y_{m,i})
 \quad(1\le i\le m)
\end{equation}
at \(m\) distinct rational radii in the open Perron interval.

More explicitly, put
\[
 s_i=64(m+1)+4i\qquad(1\le i\le m)
\]
and let \(C_m\) be the \((2m+1)\times(2m+1)\) integral matrix indexed by
\[
 u_0,\ldots,u_m,w_1,\ldots,w_m
\]
whose only non-zero entries are
\begin{equation}\label{eq:realizable-collision-Cm-entries}
 \begin{aligned}
  (C_m)_{u_{i-1},u_i}&=-4,&
  (C_m)_{u_{i-1},w_i}&=s_i &&(1\le i\le m),\\
  (C_m)_{w_i,u_i}&=s_i,&
  (C_m)_{u_m,u_0}&=-4.
 \end{aligned}
\end{equation}
Then
\begin{equation}\label{eq:realizable-collision-Qm}
 Q_m(z):=\det(I-zC_m)
 =1+4z\prod_{i=1}^m(-4z+s_i^2z^2).
\end{equation}
Writing \(n=2m+1\), take the two \(2n\times2n\) sign blocks
\begin{equation}\label{eq:realizable-collision-sign-blocks}
 B_m=\begin{pmatrix}0&C_m/2\\2I_n&0\end{pmatrix},
 \qquad B_m'=C_m\oplus C_m.
\end{equation}
They satisfy
\begin{equation}\label{eq:realizable-collision-determinants}
 \det(I-zB_m)=Q_m(z^2),\qquad
 \det(I-zB_m')=Q_m(z)^2.
\end{equation}
The common base may be chosen as
\[
 A_m=s_mJ_{4m+2},\qquad
 \lambda_m=s_m(4m+2),
\]
and the collision radii are
\begin{equation}\label{eq:realizable-collision-radii}
 y_{m,i}=\frac4{s_i^2}\qquad(1\le i\le m).
\end{equation}
\end{theorem}

\begin{proof}
In the directed support of \(C_m\), every non-trivial cycle must traverse
from \(u_0\) through the stages \(1,\ldots,m\) and then use the closing edge
\(u_m\to u_0\).  At stage \(i\), it uses either the direct edge of weight
\(-4\), contributing \(-4z\), or the two-edge detour through \(w_i\),
contributing \(s_i^2z^2\).  A single directed cycle contributes the negative
of the product of its weighted \(z\)-edges to \(\det(I-zC_m)\).  The closing
edge contributes \(-4z\), so summing the \(2^m\) choices gives
\[
 1-(-4z)\prod_{i=1}^m(-4z+s_i^2z^2),
\]
which is \eqref{eq:realizable-collision-Qm}.

All entries of \(C_m\) are divisible by four.  Hence the blocks in
\eqref{eq:realizable-collision-sign-blocks} are integral and even.  The
standard block determinant identity gives
\[
 \det(I-zB_m)=\det(I-z^2C_m)=Q_m(z^2),
\]
while the direct sum gives the second identity in
\eqref{eq:realizable-collision-determinants}.  Moreover
\(\abs{B_m},\abs{B_m'}\le A_m\) entrywise and both inequalities are strict
somewhere.  Since \(A_m\) is positive, the matrices
\((A_m\pm B_m)/2\) and \((A_m\pm B_m')/2\) realize two cocycles on the same
marked base, and strict Perron--Frobenius monotonicity gives
\[
 \rad(B_m),\rad(B_m')<\rad(A_m)=\lambda_m.
\]

The \(s_i\) are strictly increasing, so the radii in
\eqref{eq:realizable-collision-radii} are positive, rational, and distinct.
The largest is \(4/s_1^2\), and direct expansion gives
\[
 s_1^2-4s_m(4m+2)
 =3008m^2+7136m+4112>0.
\]
Thus every \(y_{m,i}<\lambda_m^{-1}\).  The \(i\)-th factor in
\eqref{eq:realizable-collision-Qm} vanishes at \(y_{m,i}\), so
\[
 Q_m(y_{m,i})=1.
\]

For the sign-determinant ratio in the orientation of
\Cref{prop:c2-standard-zeta-ratio}, put
\[
 H_m(z)=\frac{\det(I-zB_m)}{\det(I-zB_m')}
 =\frac{Q_m(z^2)}{Q_m(z)^2}.
\]
The dyadic identity \eqref{eq:finite-radial-boundary-identity} and the
rational telescope give
\[
 F_\varepsilon^{\tau_m}(y)-F_\varepsilon^{\tau_m'}(y)
 =2\log Q_m(y).
\]
It vanishes at every \(y_{m,i}\).  The trivial blocks are both \(A_m\), so
Fourier inversion for \(C_2\) proves both equalities in
\eqref{eq:realizable-linear-profile-collisions}.

The polynomial \(Q_m\) is non-constant.  If
\(Q_m(z^2)=Q_m(z)^2\), comparison of the least non-constant term would give
two different degrees; hence \(H_m\ne1\).  The comparison dictionary therefore
shows that the two primitive length--element data sets differ.  Finally,
\[
 H_m(z)=
 \frac{\zeta_{\widetilde X_{\tau_m'}}(z)}
      {\zeta_{\widetilde X_{\tau_m}}(z)}
\]
by \Cref{prop:c2-standard-zeta-ratio}, so these are collisions of standard
Artin--Mazur zeta ratios.
\end{proof}

\begin{corollary}[Linear lower bound for radial determination]
\label{cor:linear-radial-sampling-lower-bound}
Let \(\mathsf N_{C_2}(V)\) be the least integer \(N\) such that, for every
pair of one-step \(C_2\)-cocycles over primitive bases of at most \(V\)
vertices, equality of both element profiles at \(N\) distinct radii in the
common open Perron interval, with at least one algebraic radius, forces
equality of all primitive length--element data.  Then, for \(V\ge6\),
\begin{equation}\label{eq:linear-radial-sampling-lower-bound}
 \mathsf N_{C_2}(V)\ge
 \left\lfloor\frac{V-2}{4}\right\rfloor+1.
\end{equation}
In particular, the worst-case radial sampling complexity is
\(\Omega(V)\), so the \(O(V\log V)\) upper bound cannot be replaced by
\(o(V)\).
\end{corollary}

\begin{proof}
Take \(m=\lfloor(V-2)/4\rfloor\), which is positive for \(V\ge6\).
Then \(4m+2\le V\), and
\Cref{thm:realizable-linear-multicollisions} gives two distinct periodic-data
sets agreeing at \(m\) rational radii.  A universal determination number must
therefore be at least \(m+1\).
\end{proof}

\begin{corollary}[Realizable logarithmic certificate-degree sharpness]
\label{cor:realizable-logarithmic-certificate-sharpness}
There is a sequence of pairs of strict-gap \(C_2\)-cocycles on common
primitive \(V_J\)-vertex bases whose standard Artin--Mazur zeta ratios have
unique normalized rational Mahler certificates of degree
\[
 \frac{V_J}{2}\log_2V_J=\Omega(V_J\log V_J).
\]
Thus the \(O(D\log D)\) certificate-degree bound has the correct order even
on realizable standard \(C_2\)-cover zeta ratios.
\end{corollary}

\begin{proof}
Let \(Q(z)=1-2z\), \(J\ge0\), and
\[
 R_J(z)=\prod_{j=0}^JQ(z^{2^j})^{2^{J-j}}.
\]
As in \Cref{rem:logarithmic-mahler-degree-sharpness},
\[
 H_J(z):=\frac{R_J(z^2)}{R_J(z)^2}
 =\frac{Q(z^{2^{J+1}})}{Q(z)^{2^{J+1}}},
 \qquad
 \deg R_J=(J+1)2^J.
\]
Put \(V_J=2^{J+1}\).  The numerator and denominator of \(H_J\) have degree
\(V_J\), constant term one, and are both congruent to one modulo two.
\Cref{lem:relative-sign-determinant-realization} therefore realizes them as
the two sign determinants of strict-gap cocycles on one positive
\(V_J\)-vertex base.  Their ratio is the corresponding standard zeta ratio.
Normalized injectivity of the Mahler coboundary map makes \(R_J\) its unique
certificate, and
\[
 \deg R_J=(J+1)2^J=\frac{V_J}{2}\log_2V_J.
\]
\end{proof}

\begin{remark}[Three distinct notions of sharpness]
\label{rem:three-sharpness-levels}
\Cref{cor:realizable-logarithmic-certificate-sharpness} concerns certificate
degree and upgrades the abstract divisor family to standard realizable zeta
ratios.  It does not produce \(\Omega(V\log V)\) positive real collisions.
The genuine sampling statement is
\Cref{cor:linear-radial-sampling-lower-bound}, which gives \(\Omega(V)\)
distinct rational collisions.  No matching \(\Omega(V\log V)\) positive-real
sampling lower bound is claimed.
\end{remark}
