\section[Exact rational verification for the S3 example]{Exact
  rational verification for the \texorpdfstring{$S_3$}{S3}
  example}
\label{app:s3-log-certificates}

This appendix is the exact-arithmetic reproducibility record for the four
rational windows used by the \(S_3\) obstruction in \Cref{sec:shadows}.  The
finite matrices and determinant polynomials are displayed in the main text and
in \Cref{prop:s3-example-channels}.  The tables below list every rational
endpoint used in the verification and every positive integer obtained after
clearing denominators in the final comparisons.

\begin{remark}[Exact rational verification record for the \(S_3\) example]
\label{prop:s3-compact-rational-verification-record}
  For every rational logarithm argument \(q>1\), put
  \[
    u_q:=\frac{q-1}{q+1},\qquad
    L_3(q):=2\sum_{j=0}^{3}\frac{u_q^{2j+1}}{2j+1},
  \]

  \[
    U_3(q):=L_3(q)+\frac{2u_q^9}{9(1-u_q^2)}.
  \]

  \begin{equation}\label{eq:s3-log-lower-upper-bounds}
    L_3(q)<\log q<U_3(q).
  \end{equation}

\begin{lemma}[Exact rational bounds for the \(S_3\)-channels]
  \label{lem:appendix-s3-sharp-certificates}
  Define
  \begin{equation}\label{eq:s3-example-epsilon-certificate-sum}
    \Pi_\varepsilon(1/2)=E_7+\Theta_\varepsilon,
  \end{equation}
  where
  \[
    E_7
    =
    -\frac12\log 2
    +\frac16\log\frac{32}{31}
    +\frac1{10}\log\frac{512}{511}
    +\frac1{14}\log\frac{8192}{8191},
  \]
  and
  \begin{equation}\label{eq:s3-example-epsilon-tail-certificate}
    \abs{\Theta_\varepsilon}\le \frac{1}{1105920}.
  \end{equation}
  Also define
  \begin{equation}\label{eq:s3-example-rho-certificate-sum}
    \Pi_{\varrho_{\mathrm{std}}}(1/2)=R_{15}+\Theta_{\mathrm{std}},
  \end{equation}
  where
  \[
  \begin{aligned}
    R_{15}
    ={}&
    -\log\frac32
    -\frac12\log\frac85
    -\frac13\log\frac43
    +\frac15\log\frac{33}{32}
    +\frac13\log\frac{32}{31}
    +\frac17\log\frac{129}{128} \\
    &+\frac1{10}\log\frac{512\cdot 1024}{511\cdot 1025}
    +\frac1{11}\log\frac{2049}{2048}
    +\frac1{13}\log\frac{8193}{8192}
    +\frac1{14}\log\frac{8192\cdot 16384}{8191\cdot 16385}
    +\frac1{15}\log\frac{16384}{16383},
  \end{aligned}
  \]
  and
  \begin{equation}\label{eq:s3-example-rho-tail-certificate}
    \abs{\Theta_{\mathrm{std}}}\le \frac{9}{524288}.
  \end{equation}
  With \(L_3,U_3\) from \eqref{eq:s3-log-lower-upper-bounds}, set
  \begin{equation}\label{eq:s3-example-e7-r15-endpoints}
  \begin{gathered}
  \begin{aligned}
    E_7^-
    &:=
    -\frac12\,U_3(2)
    +\frac16\,L_3(32/31)
    +\frac1{10}\,L_3(512/511)
    +\frac1{14}\,L_3(8192/8191),\\
    E_7^+
    &:=
    -\frac12\,L_3(2)
    +\frac16\,U_3(32/31)
    +\frac1{10}\,U_3(512/511)
    +\frac1{14}\,U_3(8192/8191),
  \end{aligned}\\[0.5ex]
  \begin{aligned}
    R_{15}^-
    &:=
    -U_3(3/2)-\frac12\,U_3(8/5)-\frac13\,U_3(4/3)
    +\frac15\,L_3(33/32)+\frac13\,L_3(32/31)\\
    &\quad+\frac17\,L_3(129/128)
    +\frac1{10}\,L_3(524288/523775)
    +\frac1{11}\,L_3(2049/2048)\\
    &\quad+\frac1{13}\,L_3(8193/8192)
    +\frac1{14}\,L_3(134217728/134209535)
    +\frac1{15}\,L_3(16384/16383),\\
    R_{15}^+
    &:=
    -L_3(3/2)-\frac12\,L_3(8/5)-\frac13\,L_3(4/3)
    +\frac15\,U_3(33/32)+\frac13\,U_3(32/31)\\
    &\quad+\frac17\,U_3(129/128)
    +\frac1{10}\,U_3(524288/523775)
    +\frac1{11}\,U_3(2049/2048)\\
    &\quad+\frac1{13}\,U_3(8193/8192)
    +\frac1{14}\,U_3(134217728/134209535)
    +\frac1{15}\,U_3(16384/16383).
  \end{aligned}
  \end{gathered}
  \end{equation}
  Then
  \begin{equation}\label{eq:s3-example-certified-intervals-sharp}
    -\frac{341079}{10^6}-\frac{1}{1105920}
    <\Pi_\varepsilon(1/2)<
    -\frac{341071}{10^6}+\frac{1}{1105920},
  \end{equation}
  and
  \begin{equation}\label{eq:s3-example-certified-intervals-sharp-rho}
    -\frac{718352}{10^6}-\frac{9}{524288}
    <\Pi_{\varrho_{\mathrm{std}}}(1/2)<
    -\frac{718351}{10^6}+\frac{9}{524288}.
  \end{equation}
  Hence
  \[
    -\frac{342}{1000}<\Pi_\varepsilon(1/2)<-\frac{341}{1000},
    \qquad
    -\frac{719}{1000}<\Pi_{\varrho_{\mathrm{std}}}(1/2)<-\frac{718}{1000}.
  \]
\end{lemma}

\begin{proof}
  The formulas for \(E_7\) and \(R_{15}\) are the truncations of
  \eqref{eq:s3-example-pi-eps} and \eqref{eq:s3-example-pi-rho}.  For the sign
  tail, with \(x_r=2^{1-2r}\) and odd \(r\ge9\),
  \(-\log(1-x_r)\le 2x_r\), giving
  \[
    \abs{\Theta_\varepsilon}
    \le \frac19\sum_{j\ge0}2^{1-2(9+2j)}
    =\frac1{1105920}.
  \]
  For the standard tail, with \(m\ge16\),
  \(\log(1-2^{1-m})^{-1}\le 2^{2-m}\),
  \(\log(1+2^{-m})\le 2^{-m}\), \(0\le\alpha(m)\le2\), and
  \(0\le\beta(m)\le1\), giving
  \[
    \abs{\Theta_{\mathrm{std}}}
    \le\sum_{m\ge16}\frac{9\,2^{-m}}{m}
    \le \frac9{524288}.
  \]
  The endpoint definitions in \eqref{eq:s3-example-e7-r15-endpoints} and
  \eqref{eq:s3-log-lower-upper-bounds} give
  \(E_7^-<E_7<E_7^+\) and \(R_{15}^-<R_{15}<R_{15}^+\).  The exact rational
  endpoint comparisons are
  \begin{equation}\label{eq:s3-example-finite-sum-windows}
    \begin{aligned}
      -\frac{341079}{10^6}
      &<E_7^-<E_7<E_7^+<-\frac{341071}{10^6},\\
      -\frac{718352}{10^6}
      &<R_{15}^-<R_{15}<R_{15}^+<-\frac{718351}{10^6}.
    \end{aligned}
  \end{equation}
  Together with the two tail radii, the four primitive-channel numerators
  printed in \Cref{prop:s3-compact-rational-verification-record} prove the
  sharp and coarse intervals.
\end{proof}

\begin{lemma}[Exact rational bounds for the \(S_3\) product-level fixed-label values]\label{lem:appendix-s3-f-level-certificates}
  In the \(S_3\)-example of \Cref{sec:shadows}, the Adams-corrected
  fixed-label Euler-transform coordinates entering
  \Cref{thm:class-euler-product-mertens} satisfy
  \begin{equation}\label{eq:s3-f-epsilon-appendix-window}
    -\frac{381}{1000}<F_\varepsilon(1/2)<-\frac{380}{1000},
  \end{equation}
  and
  \begin{equation}\label{eq:s3-f-rho-appendix-window}
    -\frac{923}{1000}<F_{\varrho_{\mathrm{std}}}(1/2)<-\frac{922}{1000}.
  \end{equation}
\end{lemma}

\begin{proof}
  From \eqref{eq:s3-example-pi-eps},
  \[
    F_\varepsilon(1/2)
    =-\frac12\sum_{j\ge0}\frac{1}{2^j}
      \log\left(1-2^{1-2^{j+1}}\right)^{-1}.
  \]
  Let
  \[
    Q_8:=-\frac12\log2-\frac14\log\frac87
      -\frac18\log\frac{128}{127}
      -\frac1{16}\log\frac{32768}{32767},
    \qquad
    B_\varepsilon:=\frac1{2^{35}-8}.
  \]
  The remaining terms have absolute value at most \(B_\varepsilon\).  Define
  \begin{equation}\label{eq:s3-q8-endpoint-values}
  \begin{aligned}
    Q_8^- &:=-\frac12U_3(2)-\frac14U_3(8/7)
      -\frac18U_3(128/127)-\frac1{16}U_3(32768/32767),\\
    Q_8^+ &:=-\frac12L_3(2)-\frac14L_3(8/7)
      -\frac18L_3(128/127)-\frac1{16}L_3(32768/32767).
  \end{aligned}
  \end{equation}
  Hence \(Q_8^- -B_\varepsilon<F_\varepsilon(1/2)<Q_8^+\).
  The printed denominator-cleared checks in
  \Cref{prop:s3-compact-rational-verification-record} give
  \eqref{eq:s3-f-epsilon-appendix-window}.

  For the standard representation, grouping the fixed-label transform by
  \(k=rm\) gives
  \[
    F_{\varrho_{\mathrm{std}}}(1/2)=
    \sum_{k\ge1}\frac{1}{k}
      \left(
        A_k\log(1-2^{1-k})^{-1}
        -B_k\log(1+2^{-k})
      \right),
  \]
  where the first logarithm is absent for \(k=1\),
  \(A_k=\sum_{m\mid k}\mu(m)\alpha(m)\), and
  \(B_k=\sum_{m\mid k}\mu(m)\beta(m)\).  The only non-zero pairs with
  \(k\le32\) are
  \[
  \begin{array}{c|ccccccccc}
    k&1&2&3&4&8&9&16&27&32\\ \hline
    A_k&0&-1&-1&-1&-1&-1&-1&-1&-1\\
    B_k&1&0&1&0&0&1&0&1&0.
  \end{array}
  \]
  For \(k\ge33\), the bounds \(\abs{A_k}\le2\), \(\abs{B_k}\le1\),
  \(\log(1-2^{1-k})^{-1}\le2^{2-k}\), and
  \(\log(1+2^{-k})\le2^{-k}\) give
  \(B_{\mathrm{std}}=(5/33)2^{-32}\).  If \(S_{32}\) is the finite sum over
  the displayed \(k\)'s, define rational endpoints by
  \begin{equation}\label{eq:s3-s32-endpoint-values}
  \begin{aligned}
    S_{32}^-&:=\sum_{\nu=1}^{12}c_\nu U_3(q_\nu),
    &
    S_{32}^+&:=\sum_{\nu=1}^{12}c_\nu L_3(q_\nu),
    &
    B_{\mathrm{std}}&:=\frac{5}{141733920768},
  \end{aligned}
  \end{equation}
  where
  \[
  \begin{array}{cccc}
    q_\nu:\;
    3/2,&2,&4/3,&9/8,\\
    8/7,&128/127,&256/255,&513/512,\\
    32768/32767,&67108864/67108863,&134217729/134217728,&
    2147483648/2147483647,
  \end{array}
  \]
  and
  \[
    c_\nu:\;
    -1,\ -\frac12,\ -\frac13,\ -\frac13,\ -\frac14,\ -\frac18,\
    -\frac19,\ -\frac19,\ -\frac1{16},\ -\frac1{27},\
    -\frac1{27},\ -\frac1{32}.
  \]
  All coefficients are negative, so \(U_3\) gives the lower endpoint and
  \(L_3\) the upper endpoint.  Therefore
  \(S_{32}^- -B_{\mathrm{std}}<F_{\varrho_{\mathrm{std}}}(1/2)
  <S_{32}^+ +B_{\mathrm{std}}\).  The two printed denominator-cleared comparisons in
  \Cref{prop:s3-compact-rational-verification-record} prove
  \eqref{eq:s3-f-rho-appendix-window}.
\end{proof}

\begin{lemma}[Displayed rational interval windows]
  \label{cor:s3-displayed-rational-certificate-windows}
  The rational logarithm brackets, finite endpoints, and tail bounds printed in
  \Cref{prop:s3-compact-rational-verification-record} imply exactly the four
  coarse windows used in the main text:
  \[
    -\frac{342}{1000}<\Pi_\varepsilon(1/2)<-\frac{341}{1000},
    \qquad
    -\frac{719}{1000}<\Pi_{\varrho_{\mathrm{std}}}(1/2)<-\frac{718}{1000},
  \]
  and
  \[
    -\frac{381}{1000}<F_\varepsilon(1/2)<-\frac{380}{1000},
    \qquad
    -\frac{923}{1000}<F_{\varrho_{\mathrm{std}}}(1/2)<-\frac{922}{1000}.
  \]
  Thus the \(S_3\) example gives an exact rational verification of both the
  primitive-channel windows and the Adams-corrected fixed-label product
  windows.
\end{lemma}

\begin{proof}
  This is the collection of the two primitive windows from
  \Cref{lem:appendix-s3-sharp-certificates} and the two fixed-label windows
  from \Cref{lem:appendix-s3-f-level-certificates}.  Each strict inclusion is
  one of the rational comparisons in the exact certificate table.
\end{proof}

\begin{remark}[Finite exact-arithmetic reproducibility]
\label{prop:s3-finite-exact-arithmetic-reproducibility}
  The article-level certificate consists exactly of the formulas for
  \(L_3(q)\) and \(U_3(q)\), the finite signed endpoint sums
  \(E_7^\pm,R_{15}^\pm,Q_8^\pm,S_{32}^\pm\), the four rational tail radii, and
  the denominator-cleared positivity checks in
  \Cref{prop:s3-compact-rational-verification-record}.  This is sufficient to
  verify the four windows by hand using only rational arithmetic: substitute
  \(L_3,U_3\), choose endpoints according to the sign of each rational
  coefficient, add the printed rational tail radius, subtract the printed
  safety margin, and compare the corresponding denominator-cleared numerator
  in the exact certificate table.  Thus the article gives a complete
  publication-level certificate without requiring any external certificate
  file or suppressed arithmetic check.
\end{remark}
