\subsubsection{Product-tail reduction and fixed-label Euler transform}
\label{subsubsec:class-product-tail-reduction}

This block converts the additive constants of Main Theorem~II into the
positive Frobenius-class Euler-product constants of Main Theorem~III.  The
tail reduction separates the scalar factor \(C(A)\) from the non-trivial class
coordinates, and the fixed-label Euler transform is the non-scalar input that
remains.  The determinant evaluation in the next block computes exactly this
fixed-label vector.

\begin{lemma}[Explicit product-tail bound]
  \label{lem:class-euler-product-tail-bound}
  Assume the hypotheses of \Cref{thm:class-mertens-explicit}.  For every
  conjugacy class \(C\subset G\), put
  \[
    T_C(N):=
      \sum_{r\ge2}\frac1r\sum_{n\le N}p_{n,C}\lambda^{-rn}.
  \]
  Then
  \begin{equation}\label{eq:class-product-tail-bound}
    T_C(N)=
      \sum_{r\ge2}\frac{\Pi_{\mathbf1_C}(\lambda^{-r})}{r}
      +O(\lambda^{-N}/N),
  \end{equation}
  uniformly in \(C\). Consequently, if
  \begin{equation}\label{eq:class-product-log-bound-input}
    \sum_{n\le N}\frac{p_{n,C}}{\lambda^n}
    =
    \frac{|C|}{|G|}\log N+\mathcal M_C(A,\tau;G)+O(N^{-1}),
  \end{equation}
  then
  \begin{equation}\label{eq:class-product-log-bound-output}
    \log P_C(N)=
    -\frac{|C|}{|G|}\log N
    -\mathcal M_C(A,\tau;G)
    -\sum_{r\ge2}\frac{\Pi_{\mathbf1_C}(\lambda^{-r})}{r}
    +O(N^{-1}).
  \end{equation}
\end{lemma}

\begin{proof}
  The explicit class bound \eqref{eq:class-primitive-explicit-bound} gives a
  uniform constant
  \[
    B_{\mathrm{prod}}:=
    1+C_{\mathrm{cls}}(A,\tau;G)
  \]
  such that
  \[
    0\le p_{n,C}
    \le
    B_{\mathrm{prod}}\frac{\lambda^n}{n}
    \qquad(n\ge1,
    \ C\subset G),
  \]
  after increasing the constant harmlessly over the finite set of classes.  In
  fact the displayed choice follows directly from
  \(|C|/|G|\le1\), \(\Lambda_*\le\lambda\), and non-negativity of
  \(p_{n,C}\).  Hence \(\sum_np_{n,C}\lambda^{-rn}\) converges absolutely for
  every \(r\ge2\), uniformly in \(C\), and the full tail
  \(T_C(\infty)-T_C(N)\) is bounded by
  \(B_{\mathrm{prod}}\) times
  \[
    \sum_{r\ge2}\frac1r\sum_{n>N}\frac{\lambda^{(1-r)n}}{n}.
  \]
  Since \(n\ge N+1\) on the inner tail,
  \[
    \sum_{r\ge2}\frac1r\sum_{n>N}\frac{\lambda^{(1-r)n}}{n}
    \le
    \frac{1}{N+1}
    \sum_{r\ge2}\frac{\lambda^{(1-r)(N+1)}}{r(1-\lambda^{1-r})}.
  \]
  Write \(q=\lambda^{-1}<1\).  The last sum equals
  \[
    \sum_{r\ge2}\frac{q^{(r-1)(N+1)}}{r(1-q^{r-1})}
    \le
    \frac{1}{2(1-q)}\sum_{s\ge1}q^{s(N+1)}
    =O(q^{N+1}),
  \]
  with a constant independent of \(C\).  This proves
  \eqref{eq:class-product-tail-bound}.  Taking logarithms in the finite
  Euler product gives
  \[
    \log P_C(N)=
    -\sum_{n\le N}\frac{p_{n,C}}{\lambda^n}-T_C(N).
  \]
  Substituting \eqref{eq:class-product-log-bound-input} and
  \eqref{eq:class-product-tail-bound} gives
  \eqref{eq:class-product-log-bound-output}, because
  \(O(\lambda^{-N}/N)\) is absorbed by \(O(N^{-1})\).
\end{proof}

\begin{proposition}[Uniform remainders for the class Mertens and product formulae]
  \label{prop:main-ii-iii-remainder-audit}
  Under the hypotheses of \Cref{thm:class-mertens-explicit}, the
  \(O(N^{-1})\) remainders in
  \Cref{cor:class-mertens-log-form,thm:class-euler-product-mertens} may be
  chosen uniformly over all conjugacy classes.  More precisely, the additive
  logarithmic remainder is bounded by
  \eqref{eq:class-mertens-log-explicit}, the repetition-tail remainder is
  bounded by the explicit geometric estimate in
  \Cref{lem:class-euler-product-tail-bound}, and exponentiation changes a
  class-uniform logarithmic bound \(B/N\) into a class-uniform multiplicative
  bound \(eB/N\) for all sufficiently large \(N\).  All constants depend only
  on the finite determinant model \((A,\tau,G)\), the Perron root \(\lambda\),
  and the strict gap margin through \(C_{\mathrm{cls}}(A,\tau;G)\) and
  \(\theta=\Lambda_*/\lambda<1\), and not on the class \(C\) or on \(N\).
\end{proposition}

\begin{proof}
  The additive part is exactly the explicit estimate
  \eqref{eq:class-mertens-log-explicit}.  Since \(|C|/|G|\le1\), its right-hand
  side is bounded by
  \[
    \frac{1}{N+1}
    +
    \frac{C_{\mathrm{cls}}(A,\tau;G)\theta^{N+1}}
      {(N+1)(1-\theta)},
  \]
  which is uniform in \(C\) and is \(O(N^{-1})\).  No floor term is hidden here:
  the sums are over integer lengths \(n\le N\), and the only passage from
  \(H_N\) to \(\log N\) is the displayed bound
  \(\abs{H_N-\log N-\gamma}\le (N+1)^{-1}\).

  For the product repetition tail, the proof of
  \Cref{lem:class-euler-product-tail-bound} first uses the class-uniform bound
  \(p_{n,C}\le B_{\mathrm{prod}}\lambda^n/n\), with
  \(B_{\mathrm{prod}}:=1+C_{\mathrm{cls}}(A,\tau;G)\), which follows from
  \eqref{eq:class-primitive-explicit-bound}, \(|C|/|G|\le1\), and
  \(\Lambda_*\le\lambda\).  The omitted part after length \(N\) is therefore
  bounded by
  \[
    \frac{B_{\mathrm{prod}}}{N+1}
    \sum_{r\ge2}\frac{\lambda^{(1-r)(N+1)}}{r(1-\lambda^{1-r})}
    \le
    \frac{B_{\mathrm{prod}}}{N+1}\,
    \frac{1}{2(1-
    \lambda^{-1})}\,
    \frac{\lambda^{-(N+1)}}{1-\lambda^{-(N+1)}}
  \]
  for all \(N\ge1\).  This is stronger than a class-uniform
  \(O(N^{-1})\) bound.  Substituting the additive and tail estimates into the
  exact logarithmic identity for \(P_C(N)\) gives a real logarithmic remainder
  \(R_{C,N}\) with \(\abs{R_{C,N}}\le B/N\), where \(B\) is independent of
  \(C\).  For all large \(N\), \(\abs{R_{C,N}}\le1\); the elementary inequality
  \(\abs{e^x-1}\le e\abs{x}\) for \(\abs{x}\le1\), as recorded in
  \Cref{lem:class-product-log-to-product-error}, gives the same
  class-uniform \(1+O(N^{-1})\) product remainder.  These are the only estimates
  used in the product proof, so the stated uniformity and dependence of the
  constants follow.
\end{proof}

\begin{lemma}[Exponentiating the class product logarithm]
  \label{lem:class-product-log-to-product-error}
  Let \(a\in\RR\), \(L\in\RR\), and let \(R_N=O(N^{-1})\) be a real sequence.
  If a positive sequence \(Q_N\) satisfies
  \[
    \log Q_N=-a\log N+L+R_N,
  \]
  then
  \[
    Q_N=e^L N^{-a}\bigl(1+O(N^{-1})\bigr).
  \]
  If the implied constant in \(R_N=O(N^{-1})\) is uniform over a finite set of
  indices, then the resulting multiplicative error is uniform over that set.
\end{lemma}

\begin{proof}
  Choose \(B>0\) with \(|R_N|\le B/N\) for all large \(N\).  Then
  \[
    Q_N=e^L N^{-a}e^{R_N}.
  \]
  For all large \(N\), \(|R_N|\le1\), and the elementary estimate
  \(|e^x-1|\le e|x|\) for \(|x|\le1\) gives
  \(e^{R_N}=1+O(N^{-1})\).  The same bound with the maximum of the finitely
  many constants gives the uniform assertion.
\end{proof}

\begin{proposition}[Product-tail reduction from the additive Mertens formula]
  \label{prop:class-euler-product-main-chain-reduction}
  Assume the hypotheses of \Cref{thm:class-mertens-explicit}.  For a fixed
  conjugacy class \(C\), put
  \[
    P_C(N):=
    \prod_{\substack{\gamma\in\cP\,:\\ |\gamma|\le N\\
      \tau(\gamma)\in C}}
      \left(1-\lambda^{-|\gamma|}\right).
  \]
  Then the logarithmic Main Theorem~II estimate
  \eqref{eq:class-mertens-log}, the \(r\ge2\) product-tail identity
  \eqref{eq:class-product-tail-bound}, and
  \Cref{lem:class-product-log-to-product-error} imply
  \begin{equation}\label{eq:class-euler-product-reduction-asymptotic}
    P_C(N)=
    \exp\!\left(
      -\mathcal M_C(A,\tau;G)
      -\sum_{r\ge2}\frac{\Pi_{\mathbf1_C}(\lambda^{-r})}{r}
    \right)
    N^{-|C|/|G|}\bigl(1+O(N^{-1})\bigr).
  \end{equation}
  Moreover the two constant formulas in Main Theorem~III are equivalent
  exactly through the fixed-label normalisation identity
  \begin{equation}\label{eq:class-euler-product-reduction-normalisation}
    \mathcal M_C(A,\tau;G)
    +\sum_{r\ge2}\frac{\Pi_{\mathbf1_C}(\lambda^{-r})}{r}
    =
    \frac{|C|}{|G|}
    \left(
      \gamma+
      \log C(A)+
      \sum_{\varrho\ne\mathbf1}
        \chi_\varrho^\ast(C)F_\varrho(\lambda^{-1})
    \right).
  \end{equation}
  Thus \Cref{thm:class-euler-product-mertens} separates two inputs: the
  logarithmic class-Mertens formula and the absolutely convergent
  \(r\ge2\) repetition tail determine the displayed constant, while the
  classical Frobenius-class product theorems cited around
  \Cref{thm:class-euler-product-mertens} supply the inherited existence and
  exponent layer.  The Adams-corrected value of \(K_C\) is identified by the
  local determinant calculation.
\end{proposition}

\begin{proof}
  The logarithm of the finite Euler product is
  \[
    \log P_C(N)
    =
    -\sum_{n\le N}\frac{p_{n,C}}{\lambda^n}
    -
    \sum_{r\ge2}\frac1r\sum_{n\le N}p_{n,C}\lambda^{-rn}.
  \]
  Substituting \eqref{eq:class-mertens-log} into the first term and
  \eqref{eq:class-product-tail-bound} into the second term gives
  \[
    \log P_C(N)
    =
    -\frac{|C|}{|G|}\log N
    -\mathcal M_C(A,\tau;G)
    -\sum_{r\ge2}\frac{\Pi_{\mathbf1_C}(\lambda^{-r})}{r}
    +O(N^{-1}).
  \]
  Since \(P_C(N)>0\), \Cref{lem:class-product-log-to-product-error}
  exponentiates this estimate and proves
  \eqref{eq:class-euler-product-reduction-asymptotic}.

  The constants in \eqref{eq:class-euler-product-k-mc} and
  \eqref{eq:class-euler-product-k-f} are both written on the positive real
  branch.  Taking real logarithms of those two displayed expressions shows
  that their equality is precisely
  \eqref{eq:class-euler-product-reduction-normalisation}.  Conversely,
  substituting \eqref{eq:class-euler-product-reduction-normalisation} into
  the exponent of \eqref{eq:class-euler-product-k-mc} gives exactly the
  exponent of \eqref{eq:class-euler-product-k-f}.  Hence the two formulas are
  equivalent through the stated normalisation and no additional analytic input
  is hidden in the comparison.
\end{proof}

\begin{proposition}[Fixed-label formula for the displayed product constant]
  \label{prop:class-product-internal-k-certificate}
  Assume the hypotheses of \Cref{thm:class-mertens-explicit}.  For each
  conjugacy class \(C\subset G\), define
  \begin{equation}\label{eq:class-product-internal-k-certificate}
    K_C^{\mathrm{int}}:=
    \exp\!\left(
      -\frac{|C|}{|G|}
      \left(
        \gamma+
        \log C(A)+
        \sum_{\varrho\ne\mathbf1}
          \chi_\varrho^\ast(C)F_\varrho(\lambda^{-1})
      \right)
    \right).
  \end{equation}
  Then, using only the additive class-Mertens theorem
  \Cref{thm:class-mertens-explicit,cor:class-mertens-log-form}, the explicit
  \(r\ge2\) product-tail bound \Cref{lem:class-euler-product-tail-bound}, and
  the fixed-label normalisation identity
  \eqref{eq:class-euler-product-reduction-normalisation}, one has
  \[
    P_C(N)=K_C^{\mathrm{int}}N^{-|C|/|G|}\bigl(1+O(N^{-1})\bigr).
  \]
  In particular the displayed Main Theorem~III formula for \(K_C\) follows
  from Main Theorem~II plus the product tail and fixed-label normalisation; the
  external Frobenius-class Mertens product theorem supplies the separated
  product exponent and existence layer.
\end{proposition}

\begin{proof}
  By \Cref{prop:class-euler-product-main-chain-reduction}, the additive
  logarithmic estimate and the \(r\ge2\) tail give
  \[
    P_C(N)=
    \exp\!\left(
      -\mathcal M_C(A,\tau;G)
      -\sum_{r\ge2}\frac{\Pi_{\mathbf1_C}(\lambda^{-r})}{r}
    \right)
    N^{-|C|/|G|}\bigl(1+O(N^{-1})\bigr).
  \]
  The normalisation identity
  \eqref{eq:class-euler-product-reduction-normalisation} rewrites the exponent
  of the leading constant as
  \[
    -\frac{|C|}{|G|}
    \left(
      \gamma+
      \log C(A)+
      \sum_{\varrho\ne\mathbf1}
        \chi_\varrho^\ast(C)F_\varrho(\lambda^{-1})
    \right),
  \]
  which is exactly \(\log K_C^{\mathrm{int}}\) on the positive real branch.
  This proves the product asymptotic with the stated constant.  The final
  sentence is logical bookkeeping: no cited external product theorem appears in
  the three inputs just used, so those citations can only calibrate the
  classical exponent/existence layer rather than the Adams-corrected constant formula.
\end{proof}

\begin{proposition}[Three logarithmic objects at the product constant]
  \label{prop:three-logarithm-product-interface}
  Assume the hypotheses of \Cref{thm:class-mertens-explicit}, and let
  \(\phi:G\to\CC\) be a class function orthogonal to the trivial
  character.  The three logarithmic objects used in Main Theorem~III have
  the following distinct roles:
  \begin{center}
  \begin{tabular}{@{}L{0.25\textwidth}L{0.36\textwidth}L{0.29\textwidth}@{}}
  \toprule
  Object & Formula & Product-constant role \\
  \midrule
  Periodic Artin logarithm
  & \(\mathcal L_\phi(z)=\sum_{r\ge1}\Pi_{\psi^r\phi}(z^r)/r\)
  & Encodes periodic determinant traces, where repetitions replace
    \(\phi\) by \(\psi^r\phi\). \\
  Primitive channel
  & \(\Pi_\phi(z)=\sum_{\gamma\in\cP}\phi(\tau(\gamma))z^{|\gamma|}\)
  & Is recovered from determinant logarithms by Adams--M\"obius inversion. \\
  Fixed-label Euler transform
  & \(F_\phi(z)=\sum_{r\ge1}\Pi_\phi(z^r)/r\)
  & Is the quantity evaluated at \(z=\lambda^{-1}\) in the non-trivial
    Fourier coordinates of \(K_C\). \\
  \bottomrule
  \end{tabular}
  \end{center}
  More precisely, \(F_\phi(\lambda^{-1})\) is absolutely convergent and is
  determined by the determinant data through
  \begin{equation}\label{eq:three-logarithm-product-interface}
    F_\phi(\lambda^{-1})
    =
    \sum_{r,m\ge1}\frac{\mu(m)}{rm}\,
      \mathcal L_{\psi^m\phi}(\lambda^{-rm}),
  \end{equation}
  using the Perron-boundary branches of \Cref{thm:fixed-label-euler-transform}.
  In the determinant expansion of the right-hand side the scalar Perron
  logarithm is not evaluated: the term \((r,m,\pi)=(1,1,\mathbf1)\) has zero
  coefficient because \(\phi\perp\mathbf1\), and it is deleted before the
  boundary value \(z=\lambda^{-1}\) is taken.  All remaining terms are either
  non-trivial Abel-boundary values made regular by \(\eta<\lambda\), or
  interior values with \(rm\ge2\).
  Consequently the non-trivial term in the class-product constant is a
  fixed-label Euler transform, and it may be replaced by the periodic
  Artin logarithm only in the channels satisfying the coefficient
  obstruction \eqref{eq:channel-separation-obstruction}.
  The fixed-coordinate fibre corollary
  \Cref{thm:class-product-minimal-obstruction-skeleton} below records this
  last sentence inside the chosen branch-compatible chart: after the
  scalar coordinate is fixed, every bad replacement of the determinant
  boundary data is exactly a non-zero vector in the \((q-1)\)-dimensional
  zero-mean class-function coordinate space, where \(q\) is the number of
  conjugacy classes of \(G\), and no scalar, exponent-level, or
  notation-level record can hide it there.
\end{proposition}

\begin{proof}
  The displayed formulas are the definitions of the primitive channel and
  fixed-label Euler transform together with
  \Cref{lem:class-logarithm-primitive-expansion}, which gives the Artin
  logarithm formula.  Since \(\phi\) is orthogonal to the trivial character
  and \(\eta<\lambda\), \Cref{thm:fixed-label-euler-transform} gives
  absolute convergence of \(F_\phi(\lambda^{-1})\) and the determinant
  expression \eqref{eq:three-logarithm-product-interface}.  In the expanded
  determinant expression the coefficient of the single scalar boundary term is
  \(\langle\phi,\mathbf1\rangle_G=0\); hence that singular logarithm is omitted
  before the Abel limit is formed.  The remaining boundary terms have
  non-trivial twist and are regular by the strict twisted gap, while every
  term with \(rm\ge2\) lies strictly inside the Perron disc.  This is exactly
  the branch-compatible continuation used here.
  Finally, \Cref{prop:logarithmic-channel-separation} proves that equality
  between \(F_\phi\) and \(\mathcal L_\phi\) is equivalent to
  \eqref{eq:channel-separation-obstruction}.  Main Theorem~III uses
  precisely the values \(F_\varrho(\lambda^{-1})\) in its non-trivial
  Fourier coordinates, so the stated product-constant roles follow.  The
  final comparison sentence is the specialisation of
  \Cref{thm:class-product-minimal-obstruction-skeleton} to the boundary-error
  vector
  \(B_\varrho=\mathcal L_\varrho(\lambda^{-1})\), or more generally to any
  branch-compatible substitute \((B_\varrho)_{\varrho\ne\mathbf1}\).
\end{proof}

The product theorem has two ingredients.  The exponent \(|C|/|G|\),
Chebotarev equidistribution, and the existence scale of Frobenius-class Mertens
products are the standard symbolic-dynamical inputs of Parry--Pollicott,
Noorani--Parry, Sharp, and Mesliza--Noorani
\cite{ParryPollicott1986Chebotarev,NooraniParry1992ChebotarevShifts,Sharp1991Mertens,MeslizaNoorani1999FrobeniusMertens}.
In the present finite SFT model, closed orbits are primitive periodic edge-shift
orbits, the Frobenius datum is the conjugacy class of the ordered edge-cocycle
product, and the product weight is
\(\lambda^{-|\gamma|}=\exp(-h|\gamma|)\), with \(h=\log\lambda\).  A change from
right to left homogeneous-extension conventions only replaces \(C\) by
\(C^{-1}\); the Hermitian coefficient \(\chi_\varrho^\ast(C)\) fixes the
orientation used below.  The finite determinant contribution of this paper is
the evaluation of the non-trivial constant coordinates as the fixed-label Euler
transform values \(F_\varrho(\lambda^{-1})\), not as the periodic Artin
logarithms \(\mathcal L_\varrho(\lambda^{-1})\).

\subsubsection{Main Theorem III: fixed-label Euler-product constants}
\label{subsubsec:main-iii-fixed-label-products}

The notation in the theorem is the fixed-label notation of
\Cref{subsec:fixed-label-vs-artin-logarithm}.  Here \(K_C\) is the positive
real product constant, \(C(A)\) is the scalar finite-part normalisation,
\(\chi_\varrho^\ast(C)=\overline{\chi_\varrho(C)}\) is the Hermitian class
coefficient, and the non-trivial coordinates are
\(F_\varrho(\lambda^{-1})\).  The additive theorem supplies the \(r=1\)
logarithm, the primitive product-tail lemma supplies the \(r\ge2\)
repetitions, character orthogonality gives the non-trivial coordinates, and the
determinant formula evaluates those coordinates.

\begin{proposition}[Product-constant factorisation for Main Theorem~III]
  \label{prop:main-iii-product-constant-factorisation}
  Assume the hypotheses of \Cref{thm:class-mertens-explicit}.  Put
  \(\alpha_A:=\gamma+\log C(A)\), and, for a positive class-product profile
  \((K_C)_C\), put
  \[
    \kappa_K(C):=-\frac{|G|}{|C|}\log K_C-\alpha_A .
  \]
  Then the Main Theorem~III product constants are obtained by the following
  three steps.
  \begin{enumerate}
    \item The additive class-Mertens formula and the explicit \(r\ge2\)
    product tail give
    \begin{equation}\label{eq:main-iii-factorisation-mc-constant}
      K_C=
      \exp\!\left(
        -\mathcal M_C(A,\tau;G)
        -\sum_{r\ge2}\frac{\Pi_{\mathbf1_C}(\lambda^{-r})}{r}
      \right)
    \end{equation}
    and hence
    \(P_C(N)=K_CN^{-|C|/|G|}(1+O(N^{-1}))\).
    \item The Hermitian class-indicator expansion converts the same constants
    into the fixed-label coordinates by
    \begin{equation}\label{eq:main-iii-factorisation-fourier-profile}
      \kappa_K(C)=
      \sum_{\varrho\ne\mathbf1}
        \chi_\varrho^\ast(C)F_\varrho(\lambda^{-1}) .
    \end{equation}
    Equivalently,
    \begin{equation}\label{eq:main-iii-factorisation-inverse-profile}
      F_\pi(\lambda^{-1})=
      \sum_{C\subset G}\frac{|C|}{|G|}\kappa_K(C)\chi_\pi(C),
      \qquad \pi\ne\mathbf1 .
    \end{equation}
    \item For every non-trivial irreducible \(\pi\), the value
    \(F_\pi(\lambda^{-1})\) in \eqref{eq:main-iii-factorisation-fourier-profile} is
    the branch-compatible determinant value of
    \Cref{prop:fixed-label-determinant-perron}: the scalar Perron triple is
    deleted before the boundary value is taken, the remaining \(rm=1\) terms
    use the Abel branch, and all \(rm\ge2\) terms are absolutely convergent.
  \end{enumerate}
  Consequently the product asymptotic and the determinant value of \(K_C\) are
  determined by
  \eqref{eq:main-iii-factorisation-mc-constant}--\eqref{eq:main-iii-factorisation-inverse-profile};
  the later rigidity and comparison statements concern this already determined
  coordinate vector.
\end{proposition}

\begin{proof}
  Step \textup{(1)} is exactly
  \Cref{prop:class-euler-product-main-chain-reduction}: taking logarithms of
  the finite Euler product separates the \(r=1\) contribution, which is
  \Cref{thm:class-mertens-explicit,cor:class-mertens-log-form}, from the
  absolutely convergent \(r\ge2\) repetition tail of
  \Cref{lem:class-euler-product-tail-bound};
  \Cref{lem:class-product-log-to-product-error} then exponentiates the
  logarithmic estimate on the positive branch and gives the stated
  multiplicative error.

  For Step \textup{(2)}, use the class-indicator identity
  \[
    \mathbf1_C=\frac{|C|}{|G|}\mathbf1+\frac{|C|}{|G|}
      \sum_{\varrho\ne\mathbf1}\chi_\varrho^\ast(C)\chi_\varrho .
  \]
  Applying it to the additive constant and to the absolutely convergent tail
  gives
  \[
  \begin{aligned}
    \mathcal M_C(A,\tau;G)&+
      \sum_{r\ge2}\frac{\Pi_{\mathbf1_C}(\lambda^{-r})}{r} \\
    &=\frac{|C|}{|G|}
      \left(
        \gamma+\log C(A)+
        \sum_{\varrho\ne\mathbf1}
          \chi_\varrho^\ast(C)F_\varrho(\lambda^{-1})
      \right),
  \end{aligned}
  \]
  where the scalar parenthesis is the scalar finite-part product
  normalisation and the non-trivial parentheses are the fixed-label Euler
  transforms.  Substituting this identity into
  \eqref{eq:main-iii-factorisation-mc-constant} and taking positive real logarithms
  gives \eqref{eq:main-iii-factorisation-fourier-profile}.  Orthogonality of
  irreducible characters for
  \(\langle U,V\rangle_G=\sum_C(|C|/|G|)U(C)\overline{V(C)}\) gives the inverse
  formula \eqref{eq:main-iii-factorisation-inverse-profile}.

  Step \textup{(3)} is the substitution of
  \Cref{prop:fixed-label-determinant-perron} into the non-trivial coordinates
  in \eqref{eq:main-iii-factorisation-fourier-profile}.  That proposition is precisely
  the branch-compatible determinant evaluation of the fixed-label Euler
  transform at \(\lambda^{-1}\), with the scalar singular term removed before
  evaluation and all lower spectral-radius tails convergent.  Thus the product
  theorem depends only on the additive Mertens estimate, the product-tail
  reduction, character orthogonality, and the determinant evaluation.  The
  rigidity and obstruction results cited later compare alternative coordinate
  vectors with the already determined vector, so they are consequences rather
  than inputs.
\end{proof}

The proof of Main Theorem~III separates the following dependencies.  The
Frobenius-class product existence statement and exponent \(|C|/|G|\) are the
classical symbolic-dynamical input recalled above.  The scalar normalisation is
the finite-part constant \(C(A)\) from the trivial channel.  The new finite-state
part is the fixed-label calculation: the additive Mertens logarithm and the
absolutely convergent \(r\ge2\) product tail are converted by class Fourier
orthogonality into the values \(F_\varrho(\lambda^{-1})\), and
\Cref{prop:fixed-label-determinant-perron,prop:main-iii-fixed-label-coordinate-factorisation}
evaluate these values from the Adams--M\"obius determinant formula.  The
rigidity statements later in this section are consequences of this evaluated
coordinate vector, while the finite \(S_3\) interval computations in the
appendices verify the Artin-substitution obstruction in one model.  The strict
twisted gap \(\eta<\lambda\) leaves only the scalar Perron pole; the non-trivial
branches and the displayed value of \(K_C\) are therefore fixed by the
Adams-corrected determinant calculation rather than by the periodic Artin
logarithms.

\begin{theorem}[Main Theorem III: Adams-corrected Frobenius-class
  Euler-product constants]
  \label{thm:class-euler-product-mertens}
  Assume the hypotheses of \Cref{thm:class-mertens-explicit}, namely
  \(\lambda>1\) and the strict twisted spectral gap \(\eta<\lambda\). For a
  conjugacy class \(C\subset G\), set
  \begin{equation}\label{eq:class-euler-product-def}
    P_C(N):=
    \prod_{\substack{\gamma\in\cP\,:\\ |\gamma|\le N\\
      \tau(\gamma)\in C}}
      \left(1-\lambda^{-|\gamma|}\right).
  \end{equation}
  Then
  \begin{equation}\label{eq:class-euler-product-asymptotic}
    P_C(N)=K_C\,N^{-|C|/|G|}\bigl(1+O(N^{-1})\bigr),
  \end{equation}
  The exponent \(|C|/|G|\) and the existence of a Frobenius-class product
  asymptotic are supplied, in this normalisation, by the Frobenius-class
  Mertens product theorem of Mesliza--Noorani together with Sharp's
  symbolic-dynamical Mertens normalisation and the Parry--Pollicott/Noorani--Parry
  Chebotarev setting
  \cite{MeslizaNoorani1999FrobeniusMertens,Sharp1991Mertens,ParryPollicott1986Chebotarev,NooraniParry1992ChebotarevShifts}.
  The constant \(K_C\) is then identified locally by the Adams-corrected
  determinant expression in the fixed finite-state cocycle model.
  Here the positive branch of the constant is
  \begin{equation}\label{eq:class-euler-product-k-mc}
    K_C=
    \exp\!\left(
      -\mathcal M_C(A,\tau;G)
      -\sum_{r\ge2}\frac{\Pi_{\mathbf1_C}(\lambda^{-r})}{r}
    \right).
  \end{equation}
  Equivalently, the same positive constant is given by the branch-compatible
  character formula
  \begin{equation}\label{eq:class-euler-product-k-f}
    K_C=
    \exp\!\left(
      -\frac{|C|}{|G|}
      \left(
        \gamma+\log C(A)
        +\sum_{\varrho\neq\mathbf1}
          \chi_\varrho^\ast(C)F_\varrho(\lambda^{-1})
      \right)
    \right).
  \end{equation}
  The summation in \eqref{eq:class-euler-product-k-mc} is absolutely
  convergent, and \(F_\varrho(\lambda^{-1})\) is the fixed-label Euler
  transform of \eqref{eq:fixed-label-euler-transform-def}, rather than the
  Artin logarithm \(\mathcal L_\varrho(\lambda^{-1})\).  The comparison with
  periodic Artin logarithms and the resulting uniqueness of the Adams coordinate
  are recorded in
  \Cref{prop:three-logarithm-product-interface,prop:class-product-comparison,cor:class-product-classical-layer-uniqueness,thm:main-iii-certificate-chain}.
  The individual summands
  \(\chi_\varrho^\ast(C)F_\varrho(\lambda^{-1})\) need not be real when
  \(\varrho\) is not self-conjugate.  The bracket is real only after the
  conjugate irreducible pairs are combined; equivalently it is the value of a
  real class function under the Hermitian character expansion.  The
  exponential in \eqref{eq:class-euler-product-k-f} is therefore taken in the
  ordinary positive real branch.  The factor \(|C|/|G|\) in
  \eqref{eq:class-euler-product-k-f} multiplies the entire Hermitian
  class-indicator bracket, including the non-trivial fixed-label Fourier sum.
  The determinant version of the non-trivial values
  \(F_\varrho(\lambda^{-1})\) is exactly the branch-compatible one in
  \Cref{prop:fixed-label-determinant-perron}: the scalar Perron determinant
  \((r,m,\pi)=(1,1,\mathbf1)\) is deleted before evaluation, non-trivial
  \(rm=1\) determinants are taken by the Abel branch, and the \(rm\ge2\)
  determinant tail is absolutely convergent at \(z=\lambda^{-1}\).
\end{theorem}

\begin{proof}
  We spell out the short proof chain given by
  \Cref{prop:class-euler-product-main-chain-reduction,prop:class-product-internal-k-certificate}.
  Those propositions reduce the displayed product asymptotic and the displayed
  \(K_C\)-formula to the logarithmic form of Main Theorem~II, the explicit
  \(r\ge2\) product tail, and the fixed-label normalisation.  It remains here
  only to verify those inputs with the branch and positivity conventions stated
  in the theorem.

  Taking logarithms in \eqref{eq:class-euler-product-def} gives
  \[
    \log P_C(N)
    =
    \sum_{\substack{n\le N}}
      p_{n,C}\log(1-\lambda^{-n})
    =
    -\sum_{n\le N}\frac{p_{n,C}}{\lambda^n}
      -
      \sum_{r\ge2}\frac{1}{r}
        \sum_{n\le N}p_{n,C}\lambda^{-rn}.
  \]
  The full \(r\ge2\) tail, its truncation error, and the passage from the
  additive class-Mertens estimate to the product logarithm are exactly
  \Cref{lem:class-euler-product-tail-bound}.  Substituting the logarithmic
  form \eqref{eq:class-mertens-log} from \Cref{cor:class-mertens-log-form}
  into \eqref{eq:class-product-log-bound-output} gives
  \[
    \log P_C(N)
    =
    -\frac{|C|}{|G|}\log N
    -\mathcal M_C(A,\tau;G)
    -\sum_{r\ge2}\frac{\Pi_{\mathbf1_C}(\lambda^{-r})}{r}
    +O(N^{-1}).
  \]
  Exponentiating gives \eqref{eq:class-euler-product-asymptotic} and
  \eqref{eq:class-euler-product-k-mc} by
  \Cref{lem:class-product-log-to-product-error}: take
  \(a=|C|/|G|\) and
  \[
    L_C=-\mathcal M_C(A,\tau;G)
    -\sum_{r\ge2}\frac{\Pi_{\mathbf1_C}(\lambda^{-r})}{r}.
  \]
  The product \(P_C(N)\) is positive, the logarithmic remainder in
  \Cref{lem:class-euler-product-tail-bound} is uniform over the finite set of
  conjugacy classes, and the cited lemma converts that additive
  \(O(N^{-1})\) remainder into the stated multiplicative
  \(1+O(N^{-1})\) remainder.  This also fixes the positive branch uniformly for
  all classes.

  We now record the normalization calculation that produces the single
  global factor \(|C|/|G|\) in the character formula.  The Hermitian
  class-indicator expansion is
  \[
    \mathbf1_C
    =
    \frac{|C|}{|G|}\mathbf1+
    \frac{|C|}{|G|}
      \sum_{\varrho\neq\mathbf1}\chi_\varrho^\ast(C)\chi_\varrho .
  \]
  Applying this identity first to the additive class-Mertens constant and
  then to the absolutely convergent Euler-product tail gives
  \[
  \begin{aligned}
    &\mathcal M_C(A,\tau;G)+
      \sum_{r\ge2}\frac{\Pi_{\mathbf1_C}(\lambda^{-r})}{r} \\
    &\quad=
      \frac{|C|}{|G|}
      \left(
        \mathcal M(A)+\sum_{r\ge2}\frac{\Pi_{\mathbf1}(\lambda^{-r})}{r}
      \right) \\
    &\qquad+
      \frac{|C|}{|G|}
      \sum_{\varrho\neq\mathbf1}\chi_\varrho^\ast(C)
      \left(
        \Pi_\varrho(\lambda^{-1})+
        \sum_{r\ge2}\frac{\Pi_\varrho(\lambda^{-r})}{r}
      \right).
  \end{aligned}
  \]
  This displayed identity is an equality of real class-function coordinates,
  not a termwise assertion about individual complex characters: conjugate
  irreducible representations occur in conjugate pairs, and the Hermitian
  coefficients recombine to the real class indicator \(\mathbf1_C\).  The
  scalar coefficient is \(|C|/|G|\), and the same coefficient multiplies each
  non-trivial Fourier coordinate because it is already present in the
  class-indicator expansion before either the Mertens constant or the tail is
  evaluated.  Thus no sign or factor is introduced when the product logarithm
  is exponentiated.
  The non-trivial parenthesis is exactly
  \(F_\varrho(\lambda^{-1})\), while the scalar parenthesis is
  \(\gamma+\log C(A)\) by the scalar finite-part product normalization.
  Thus the factor \(|C|/|G|\) comes from the class-indicator Fourier
  coefficient and multiplies the scalar term and every fixed-label Fourier
  contribution before the outer minus sign is applied.

  \Cref{lem:class-product-constant-equivalence} rewrites the exponent in
  \eqref{eq:class-euler-product-k-mc} as the exponent in
  \eqref{eq:class-euler-product-k-f}. This proves the equivalent
  character formula.  The same computation is isolated in
   \Cref{lem:class-product-hermitian-normalisation} to make the
   \(|C|/|G|\) normalisation explicit.  Finally,
   \Cref{prop:fixed-label-determinant-perron} supplies the determinant
   evaluation of each non-trivial \(F_\varrho(\lambda^{-1})\) with the local
   singular-term deletion, branch convention, and absolute tail convergence
   stated in the theorem.
\end{proof}

\begin{remark}[Determinant inputs for Main Theorem~III]
  \label{rem:main-iii-dependency-map}
  Under the hypotheses of \Cref{thm:class-euler-product-mertens}, the
  determinant part of the product theorem uses the following mathematical
  inputs:
  \begin{enumerate}[label=\textup{(\roman*)},leftmargin=*]
    \item the classical Frobenius-class product layer supplies the existence
      of a product constant and the exponent \(|C|/|G|\);
    \item \Cref{lem:class-euler-product-tail-bound} converts the logarithmic
      class-Mertens estimate and the absolutely convergent \(r\ge2\) product
      tail into \eqref{eq:class-euler-product-k-mc};
    \item \Cref{lem:class-product-constant-equivalence,lem:class-product-hermitian-normalisation}
      convert the primitive-tail constant into the fixed-label character
      expression \eqref{eq:class-euler-product-k-f};
    \item \Cref{thm:fixed-label-euler-transform,prop:fixed-label-determinant-perron}
      identify each non-trivial coordinate
      \(F_\varrho(\lambda^{-1})\) with its Adams--M\"obius determinant
      boundary value, with the scalar Perron triple deleted before evaluation;
    \item \Cref{prop:main-iii-fixed-label-coordinate-factorisation,thm:class-product-minimal-obstruction-skeleton,prop:class-product-comparison}
      record the reversible fixed-coordinate form and the separation between
      the inherited classical product theorem and the Adams-corrected
      determinant constant proved here.
  \end{enumerate}
  Consequently the Main Theorem~III product constants are determined by the
  scalar normalisation \(C(A)\), the finite character table and power maps of
  \(G\), and the finitely many twisted determinant polynomials
  \(\det(I-zA_\pi)\); replacing the fixed-label vector by any other
  branch-compatible vector changes the class constants exactly by the
  Peter--Weyl error measured in \Cref{prop:class-product-comparison}.
\end{remark}

\begin{proposition}[Closed fixed-label coordinate equivalence]
  \label{prop:main-iii-fixed-label-coordinate-factorisation}
  Assume the hypotheses of \Cref{thm:class-mertens-explicit}.  The product
  constants in \Cref{thm:class-euler-product-mertens} are determined by the
  following three finite coordinate maps, and each arrow is reversible once the scalar
  normalisation \(\gamma+\log C(A)\) is fixed:
  \begin{enumerate}[label=\textup{(\roman*)},leftmargin=*]
    \item the primitive-label logarithm for a non-trivial character is the
      fixed-label Euler transform
      \[
        F_\varrho(\lambda^{-1})
        =\sum_{r\ge1}\frac{\Pi_\varrho(\lambda^{-r})}{r};
      \]
    \item the same number is the branch-normalised determinant boundary value
      \[
        \sum_{r,m\ge1}\frac{\mu(m)}{rm}
        \sum_{\substack{\pi\in\Irr(G)\\(r,m,\pi)\ne(1,1,\mathbf1)}}
          a_{\varrho,\pi}^{(m)}
          \Log\det(I-\lambda^{-rm}A_\pi)^{-1};
      \]
      the exclusion in the inner sum is part of the definition at the Perron
      boundary: the singular scalar term
      \(\Log\det(I-\lambda^{-1}A_{\mathbf1})^{-1}\) is deleted before
      evaluation and is not recovered by cancellation;
    \item the class-product constants are the positive real exponentials
      \[
        K_C=
        \exp\!\left(
        -\frac{|C|}{|G|}
        \left(
          \gamma+\log C(A)+
          \sum_{\varrho\ne\mathbf1}\chi_\varrho^\ast(C)
            F_\varrho(\lambda^{-1})
        \right)
        \right).
      \]
  \end{enumerate}
  Consequently the finite twisted determinant polynomials
  \(\det(I-zA_\pi)\), the finite character table, the Adams coefficients, and
  the scalar determinant constant \(C(A)\) form a complete fixed-coordinate
  record of \((K_C)_{C\subset G}\).  Conversely, the positive constants
  \((K_C)_C\), together with \(C(A)\), recover all non-trivial values
  \(F_\varrho(\lambda^{-1})\).  In this sense the fixed-label transform,
  determinant Perron formula, and \(K_C\)-formula form a single closed
  finite-dimensional coordinate system rather than three independent
  computational layers.
\end{proposition}

\begin{proof}
  Step \textup{(i)} is the second identity in
  \eqref{eq:fixed-label-euler-transform-vs-artin}, evaluated at
  \(z=\lambda^{-1}\); the value exists for non-trivial \(\varrho\) by the
  Perron-boundary assertion of \Cref{thm:fixed-label-euler-transform}.  Step
  \textup{(ii)} is exactly \Cref{prop:fixed-label-determinant-perron}: the
  scalar Perron triple \((1,1,\mathbf1)\) is excluded before evaluation,
  non-trivial \(rm=1\) terms use the Abel branch, and all \(rm\ge2\) terms are
  ordinary determinant-germ values inside the Perron disc.  Step
  \textup{(iii)} is formula \eqref{eq:class-euler-product-k-f} in
  \Cref{thm:class-euler-product-mertens}.  Substituting \textup{(ii)} into
  \textup{(iii)} gives the advertised finite determinant record, since
  \(G\) has finitely many irreducible characters and each twisted matrix
  \(A_\pi\) is finite dimensional.

  It remains only to prove the reversibility assertion.  With
  \(\alpha_A:=\gamma+\log C(A)\), define
  \[
    H_K(C):=-\frac{|G|}{|C|}\log K_C-\alpha_A .
  \]
  Formula \textup{(iii)} gives
  \[
    H_K(C)=
    \sum_{\varrho\ne\mathbf1}\chi_\varrho^\ast(C)
      F_\varrho(\lambda^{-1}).
  \]
  Character orthogonality on class functions therefore recovers, for every
  non-trivial \(\pi\),
  \[
    F_\pi(\lambda^{-1})
    =\sum_{C\subset G}\frac{|C|}{|G|}H_K(C)\chi_\pi(C).
  \]
  Thus the constants and the fixed scalar normalisation determine the
  non-trivial fixed-label vector, while the preceding substitutions determine
  the constants from that vector.  This proves the claimed closed coordinate equivalence.
\end{proof}

The exponent \(|C|/|G|\) and the existence of a Frobenius-class product
constant are the inherited classical part of \Cref{thm:class-euler-product-mertens},
parallel to the dynamical Mertens and Chebotarev product theorems of Sharp,
Parry--Pollicott, Noorani--Parry, and especially Mesliza--Noorani's
Frobenius-class Mertens product theorem
\cite{Sharp1991Mertens,ParryPollicott1986Chebotarev,NooraniParry1992ChebotarevShifts,MeslizaNoorani1999FrobeniusMertens}.
In applying those results here, the abstract objects in the cited theorems are
identified with the primitive finite-state data fixed in
\Cref{sec:preliminaries}: the mixing base is the primitive edge shift
\((X_A^+,\sigma)\), the finite extension is the skew-product graph on
\(V\times G\), the Frobenius class is the intrinsic conjugacy class of
\(\tau(\gamma)\), and the entropy normalisation is \(h=\log\lambda\).  The
possible opposite-action convention in a homogeneous extension only replaces
\(C\) by \(C^{-1}\); the exponent is unchanged and our Hermitian coefficient
\(\chi_\varrho^\ast(C)\) fixes the displayed class orientation.  The
strict inequality \(\eta<\lambda\) supplies the required absence of additional
unit-modulus or Perron-boundary twisted channels in this finite-dimensional
model.  No broader smooth, Markov-extension, or arbitrary-cocycle hypothesis is
used in the sequel.
The derivation above proves the displayed \(K_C\)-formula from Main
Theorem~II and the \(r\ge2\) product tail; the new assertion used in the
finite-state main chain is the branch-compatible Adams--M\"obius determinant
formula for that constant, and hence the obstruction to replacing the
fixed-label Euler transform by an ordinary Artin logarithm.




\begin{corollary}[Branch-compatible finite-determinant evaluation of the product constants]
  \label{cor:class-euler-product-determinant-evaluation}
  Assume the hypotheses of \Cref{thm:class-mertens-explicit}.  For each
  non-trivial \(\varrho\in\Irr(G)\), put
  \begin{equation}\label{eq:class-euler-product-determinant-F}
    \mathfrak F_\varrho:=
    \sum_{r,m\ge1}\frac{\mu(m)}{rm}
      \sum_{\substack{\pi\in\Irr(G)\\(r,m,\pi)\neq(1,1,\mathbf1)}}
        a_{\varrho,\pi}^{(m)}
        \Log\det(I-\lambda^{-rm}A_\pi)^{-1},
  \end{equation}
  with the same deletion and Abel-path branch convention as
  \Cref{prop:fixed-label-determinant-perron}: the scalar Perron triple
  \((1,1,\mathbf1)\) is removed before any endpoint branch is evaluated, the
  only boundary evaluations left are the non-trivial blocks at
  \(z=\lambda^{-1}\), and scalar Adams terms with \(rm\ge2\) are retained as
  ordinary interior determinant logarithms.  Then the product constants in
  \Cref{thm:class-euler-product-mertens} are recovered directly from the
  finite determinant family by
  \begin{equation}\label{eq:class-euler-product-k-det-adjacent}
    K_C=
    \exp\!\left(
      -\frac{|C|}{|G|}
      \left(
        \gamma+\log C(A)
        +\sum_{\varrho\neq\mathbf1}
          \chi_\varrho^\ast(C)\mathfrak F_\varrho
      \right)
    \right).
  \end{equation}
  Thus the classical product theorem supplies the asymptotic exponent and the
  positive constants' existence branch, while this branch-compatible determinant
  readout supplies the Adams-corrected non-trivial coordinates.  No
  cocycle-level orbit enumeration beyond the finite matrices \(A_\pi\), the
  character table, and the Adams coefficients is needed to compute
  \((K_C)_{C\subset G}\) once the scalar normalisation \(C(A)\) is fixed.
\end{corollary}

\begin{proof}
  By \Cref{prop:fixed-label-determinant-perron}, the determinant expression
  \eqref{eq:class-euler-product-determinant-F} is exactly the Perron-point
  fixed-label value \(F_\varrho(\lambda^{-1})\) for every non-trivial
  irreducible \(\varrho\).  Substituting these equalities into the
  branch-compatible character formula
  \eqref{eq:class-euler-product-k-f} of
  \Cref{thm:class-euler-product-mertens} gives
  \eqref{eq:class-euler-product-k-det-adjacent}.  The branch convention and
  the exclusion of the scalar Perron determinant are precisely those of
  \Cref{prop:fixed-label-determinant-perron}; hence every determinant in
  the displayed expression is either a non-trivial Abel-boundary value made
  regular by \(\eta<\lambda\), or an ordinary interior value with
  \(rm\ge2\).  The last assertion follows because the right-hand side uses
  only these determinants, the finite character table, and the finite Adams
  coefficient tensors.
\end{proof}


\begin{proposition}[Adams-corrected constant readout and replacement obstruction]
  \label{prop:classical-layer-adams-corrected-constant-separation}
  Assume the hypotheses of \Cref{thm:class-mertens-explicit}.  Fix the
  positive branch of the product constants in
  \Cref{thm:class-euler-product-mertens}, and put
  \[
    \alpha_A:=\gamma+\log C(A).
  \]
  Then the following records are equivalent once the Frobenius-class product
  exponent \(|C|/|G|\) and the positive constants \((K_C)_C\) are fixed:
  \begin{enumerate}[label=\textup{(\roman*)},leftmargin=*]
    \item the non-trivial fixed-label vector
      \((F_\varrho(\lambda^{-1}))_{\varrho\neq\mathbf1}\);
    \item the non-trivial determinant vector
      \((\mathfrak F_\varrho)_{\varrho\neq\mathbf1}\) of
      \eqref{eq:class-euler-product-determinant-F};
    \item the scalar-normalised class-product constant profile
      \[
        C\longmapsto
        -\frac{|G|}{|C|}\log K_C-\alpha_A .
      \]
  \end{enumerate}
  More explicitly,
  \begin{equation}\label{eq:classical-layer-forward-calibration}
    K_C=
    \exp\!\left(
      -\frac{|C|}{|G|}
      \left(
        \alpha_A+
        \sum_{\varrho\neq\mathbf1}
          \chi_\varrho^\ast(C)F_\varrho(\lambda^{-1})
      \right)
    \right),
  \end{equation}
  while every non-trivial coordinate is recovered from the constants by
  \begin{equation}\label{eq:classical-layer-inverse-calibration}
    F_\pi(\lambda^{-1})
    =
    \sum_{C\subset G}
      \frac{|C|}{|G|}
      \left(
        -\frac{|G|}{|C|}\log K_C-\alpha_A
      \right)
      \chi_\pi(C),
    \qquad \pi\neq\mathbf1 .
  \end{equation}
  Consequently the cited Frobenius-class Mertens theorems may be used for the
  leading exponent and existence branch, but they do not determine, and cannot
  replace, the Adams-corrected determinant coordinates unless they also supply
  the fixed-label vector in \textup{(i)}.  More sharply, every
  branch-compatible replacement vector
  \((B_\varrho)_{\varrho\neq\mathbf1}\) has a completely measured obstruction:
  it changes the scalar-normalised constants by the class function
  \begin{equation}\label{eq:classical-layer-replacement-error}
    C\longmapsto
    \sum_{\varrho\neq\mathbf1}
      \chi_\varrho^\ast(C)
      \bigl(F_\varrho(\lambda^{-1})-B_\varrho\bigr),
  \end{equation}
  and satisfies the Parseval identity
  \begin{equation}\label{eq:classical-layer-replacement-parseval}
    \sum_{C\subset G}\frac{|C|}{|G|}
    \left|\sum_{\varrho\neq\mathbf1}
      \chi_\varrho^\ast(C)
      \bigl(F_\varrho(\lambda^{-1})-B_\varrho\bigr)\right|^2
    =
    \sum_{\varrho\neq\mathbf1}
      \left|F_\varrho(\lambda^{-1})-B_\varrho\right|^2 .
  \end{equation}
  Hence this error is identically zero if and only if
  \(B_\varrho=F_\varrho(\lambda^{-1})\) for every non-trivial
  \(\varrho\).  In particular, whenever the periodic Artin values are regular,
  the naive replacement
  \(B_\varrho=\mathcal L_\varrho(\lambda^{-1})\) gives the true product
  constants for all conjugacy classes if and only if all non-trivial Adams
  discrepancies
  \(F_\varrho(\lambda^{-1})-\mathcal L_\varrho(\lambda^{-1})\) vanish.
\end{proposition}

\begin{proof}
  The equality between \textup{(i)} and \textup{(ii)} is precisely
  \Cref{cor:class-euler-product-determinant-evaluation}: its proof identifies
  \(\mathfrak F_\varrho\) with the Perron-point fixed-label value
  \(F_\varrho(\lambda^{-1})\), using the Abel-boundary branch convention and
  deleting the singular scalar Perron triple before evaluation.

  Formula \eqref{eq:classical-layer-forward-calibration} is the constant
  formula \eqref{eq:class-euler-product-k-f} of
  \Cref{thm:class-euler-product-mertens}.  Taking positive real logarithms and
  subtracting the scalar term gives the Hermitian Fourier expansion
  \[
    -\frac{|G|}{|C|}\log K_C-\alpha_A
    =
    \sum_{\varrho\neq\mathbf1}
      \chi_\varrho^\ast(C)F_\varrho(\lambda^{-1}).
  \]
  The irreducible characters form an orthonormal basis for class functions with
  inner product
  \[
    \langle U,V\rangle_G
    =\sum_{C\subset G}\frac{|C|}{|G|}U(C)\overline{V(C)}.
  \]
  Taking the \(\pi\)-coefficient of the displayed expansion gives
  \eqref{eq:classical-layer-inverse-calibration}.  Thus the forward and inverse
  maps between \textup{(i)} and \textup{(iii)} are inverse maps on the
  non-trivial class-function subspace.

  Finally, the same subtraction with a replacement vector
  \((B_\varrho)_{\varrho\neq\mathbf1}\) gives exactly
  \eqref{eq:classical-layer-replacement-error}.  Character orthogonality shows
  that Parseval for the same inner product gives
  \eqref{eq:classical-layer-replacement-parseval}.  Thus this class function
  vanishes identically if and only if all of its non-trivial Fourier
  coefficients vanish, namely
  \(B_\varrho=F_\varrho(\lambda^{-1})\) for every non-trivial
  \(\varrho\).  Taking
  \(B_\varrho=\mathcal L_\varrho(\lambda^{-1})\), when those Artin values are
  regular, gives the stated non-abelian Adams obstruction.  This proves both
  the equivalence of the three records and the claimed comparison: the
  Frobenius-class product theorem fixes the exponent and positive branch, while
  the branch-compatible Adams--M\"obius determinant vector supplies the
  finite-state coordinate record.
\end{proof}
