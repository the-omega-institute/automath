\section{An exact \texorpdfstring{\(S_3\)}{S3} contradiction}
\label{sec:s3-witness}

We now give a strict-gap example in which the corrected and asserted Artin
coordinates are provably different.  The example is finite-dimensional; no
numerical spectral approximation enters the argument.

Let
\[
  A=\begin{pmatrix}1&1\\1&1\end{pmatrix},
  \qquad \lambda=2,
\]
and label the four directed edges by
\begin{equation}\label{eq:s3-labels}
 \begin{aligned}
  \tau(0\to0)&=(23),&\qquad \tau(0\to1)&=e,\\
  \tau(1\to0)&=(12),&\qquad \tau(1\to1)&=(123).
 \end{aligned}
\end{equation}
Let \(\varepsilon\) and \(\varrho\) denote the sign and standard
representations of \(S_3\).

\begin{lemma}[Twisted determinants in the witness]\label{lem:s3-determinants}
For the cocycle \eqref{eq:s3-labels},
\begin{equation}\label{eq:s3-polynomials}
  P_{\mathbf1}(z)=1-2z,
  \qquad P_\varepsilon(z)=1,
  \qquad P_\varrho(z)=1+z.
\end{equation}
Consequently,
\begin{equation}\label{eq:s3-artin-values}
  L_\varepsilon(z)\equiv0,
  \qquad L_\varrho(z)=-\log(1+z),
\end{equation}
and the strict twisted gap holds with \(\eta=1<2\).
\end{lemma}

\begin{proof}
In the sign representation,
\[
  A_\varepsilon=
  \begin{pmatrix}-1&1\\-1&1\end{pmatrix},
  \qquad A_\varepsilon^2=0,
\]
so \(P_\varepsilon(z)=1\).  In a real model of the standard
representation, direct substitution of the four labels gives
\[
  A_\varrho=
  \begin{pmatrix}
    \frac12&\frac32&1&0\\
    \frac12&-\frac12&0&1\\
    -1&0&-\frac12&-\frac32\\
    0&1&\frac12&-\frac12
  \end{pmatrix}.
\]
Row reduction, or a direct determinant calculation, gives
\[
  \det(tI-A_\varrho)=t^3(t+1).
\]
Thus \(P_\varrho(z)=1+z\) and \(\rad(A_\varrho)=1\); the sign block has
spectral radius zero.  The logarithm identities follow from
\eqref{eq:artin-determinant-log}.
\end{proof}

The vanishing of \(L_\varepsilon\) does not imply the vanishing of the
primitive or fixed-label sign coordinates.  Indeed, the power map on the
quadratic character satisfies
\[
  \psi^m\varepsilon=
  \begin{cases}
    \varepsilon,&m\text{ odd},\\
    \mathbf1,&m\text{ even}.
  \end{cases}
\]

\begin{lemma}[Exact fixed-label sign coordinate]\label{lem:s3-fixed-sign}
For \(|z|<1/2\), with radial continuation to \(z=1/2\),
\begin{align}
  \Pi_\varepsilon(z)
   &=-\frac12\sum_{\substack{m\geq1\\m\text{ odd}}}
       \frac{\mu(m)}m\log(1-2z^{2m})^{-1},
       \label{eq:s3-pi-sign}\\
  F_\varepsilon(z)
   &=-\frac12\sum_{j\geq0}\frac1{2^j}
       \log\!\left(1-2z^{2^{j+1}}\right)^{-1}.
       \label{eq:s3-f-sign}
\end{align}
In particular,
\begin{equation}\label{eq:s3-strict-negative}
  F_\varepsilon(1/2)
  =-\frac12\sum_{j\geq0}\frac1{2^j}
       \log\!\left(1-2^{1-2^{j+1}}\right)^{-1}<0.
\end{equation}
The exact rational certificate described in \Cref{sec:exactness} sharpens
this to
\begin{equation}\label{eq:s3-rational-window}
  -\frac{381}{1000}<F_\varepsilon(1/2)<-\frac{380}{1000}.
\end{equation}
\end{lemma}

\begin{proof}
Insert the two Adams cases above into
\eqref{eq:primitive-adams-mobius}.  The odd terms vanish because
\(L_\varepsilon=0\).  A squarefree even integer has the form \(2m\) with
\(m\) odd and \(\mu(2m)=-\mu(m)\), while
\(L_{\mathbf1}(z)=-\log(1-2z)\).  This gives
\eqref{eq:s3-pi-sign}.

Now insert \eqref{eq:s3-pi-sign} into
\(F_\varepsilon(z)=\sum_{q\geq1}\Pi_\varepsilon(z^q)/q\).  For a fixed
\(k=mq\), the coefficient multiplying
\(-\frac12\log(1-2z^{2k})^{-1}\) is
\[
  \frac1k\sum_{\substack{m\mid k\\m\text{ odd}}}\mu(m).
\]
This divisor sum is one precisely when the odd part of \(k\) is one, and is
zero otherwise.  Hence \(k=2^j\), proving \eqref{eq:s3-f-sign}.  The series
converges at \(z=1/2\), and every summand in
\eqref{eq:s3-strict-negative} is strictly negative.

For completeness, the rational window uses
\[
  2\sum_{k=0}^{3}\frac{u^{2k+1}}{2k+1}
  <\log q<
  2\sum_{k=0}^{3}\frac{u^{2k+1}}{2k+1}
    +\frac{2u^9}{9(1-u^2)},
  \qquad u=\frac{q-1}{q+1},
\]
on the first four logarithms in \eqref{eq:s3-strict-negative}; the remaining
tail is less than \((2^{35}-8)^{-1}\).  Substitution and clearing the positive
denominators gives \eqref{eq:s3-rational-window}.  These are precisely the
operations reproduced by the exact-arithmetic certificate cited in
\Cref{sec:exactness}.
\end{proof}

\begin{theorem}[The \(S_3\) contradiction and its resolution]
\label{thm:s3-contradiction}
For the strict-gap extension \eqref{eq:s3-labels},
\begin{equation}\label{eq:s3-contradiction-coordinate}
  L_\varepsilon(1/2)=0,
  \qquad
  F_\varepsilon(1/2)\in
     \left(-\frac{381}{1000},-\frac{380}{1000}\right).
\end{equation}
Let \(C_t\) be the conjugacy class of transpositions, and let
\(K_{C_t}^{\mathrm{Art}}\) denote the value obtained from
\eqref{eq:correct-product-fourier} by replacing every \(F_\rho(1/2)\) with
\(L_\rho(1/2)\).  Then
\begin{equation}\label{eq:s3-class-contradiction}
  \log\frac{K_{C_t}}{K_{C_t}^{\mathrm{Art}}}
  =\frac12F_\varepsilon(1/2)\ne 0.
\end{equation}
Thus the 1999 Artin-\(L\) constant and the exact primitive-class product
cannot both be correct.  The contradiction is resolved by the replacement
\(L_\rho\mapsto F_\rho\) in \Cref{thm:formal-correction}.
\end{theorem}

\begin{proof}
The first assertion combines \Cref{lem:s3-determinants,lem:s3-fixed-sign}.
On the three classes \((\{e\},C_t,C_3)\), the non-trivial character rows are
\[
  \chi_\varepsilon=(1,-1,1),
  \qquad \chi_\varrho=(2,0,-1).
\]
Since \(|C_t|/|S_3|=1/2\), the standard coordinate vanishes from the
\(C_t\) Fourier component.  Subtracting the Artin-replaced logarithm from
\eqref{eq:correct-product-fourier} therefore gives
\[
  -\frac12\chi_\varepsilon(C_t)
    \bigl(F_\varepsilon(1/2)-L_\varepsilon(1/2)\bigr)
  =\frac12F_\varepsilon(1/2),
\]
which is non-zero by \eqref{eq:s3-contradiction-coordinate}.
\end{proof}
