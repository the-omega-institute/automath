\section{Finite quadratic arithmetic}
\label{app:quadratic-arithmetic}

\subsection{The positive-recurrence integer exclusion}
\label{app:positive-integer-exclusion}

We prove Lemma~\ref{lem:positive-integer-exclusion}.  Put
\[
S=a^2+a+b,\qquad T=a^3+a^2+2ab+b=aS+b(a+1).
\]
Then
\[
U_+-a=\frac{b(a+1)}S,\qquad L_+-a=\frac{bS}T.
\]
For $h=be-ak>0$, divide
\[
be=aq+r,\qquad qb=ap+t,\qquad 0\le r,t<a.
\]
Here $0\le q\le b$, and $q\le b-1$ when $r>0$.  If $k\le q-1$, then
$h\ge a+r$ and
\[
hL_+-kb>(a+r)a-(b-1)b\ge a^2-b(b-1)\ge1.
\]
The inequality $k\ge q+1$ would give $h<0$.  Hence $k=q$ and $h=r$.
Writing $j=kb-ah=a(p-r)+t$, the cases $p<r$ and $p>r$ give respectively
\[
rL_+-qb=a(r-p)-t+r(L_+-a)>a-t\ge1
\]
and
\[
qb-rU_+=a(p-r)+t-r(U_+-a)
>a-\frac{rb}{a}\ge a-b+\frac ba\ge1.
\]

It remains to treat $p=r$.  Put $w=rb-at$.  The divisions give
\begin{equation}
b^2e=r(a^2+b)+at,\qquad w=r(a^2+2b)-b^2e.
\label{eq:app-terminal-divisions}
\end{equation}
The integer $w$ cannot vanish.  Indeed, if $a=gA$, $b=gB$ with
$(A,B)=1$, then $at=rb$ gives $r=AR$, $t=BR$ with $0<R<g$; reduction of
the first two divisions modulo $g$ and Bezout's identity force $g\mid R$.

Make one further division
\[
rb=au+v,\qquad 0\le v<a.
\]
Then $w=a(u-t)+v$.  Substitution in the three reversed divisions shows
\[
\begin{array}{c|c|c}
\text{sign of }w&\text{quotient relation}&\text{remainder inequality}\\ \hline
w>0&u-t\ge2&\text{automatic}\\
w>0&u-t=1&av\ge b(t+1)\\
w<0&t-u\ge2&\text{automatic}\\
w<0&t-u=1&b(t-1)\ge v(a+1).
\end{array}
\]
For the two boundary rows, substitute $u=t+1$ or $u=t-1$ successively into
\[
r=\frac{au+v}{b},\qquad q=\frac{ar+t}{b},\qquad
e=\frac{aq+r}{b}.
\]
After multiplication by $b,b^2,b^3$, failure of the stated remainder
inequality forces $r\ge a$, $q>b$, or a nonintegral final quotient.  If an
equality occurs before the last division, write again $a=gA$, $b=gB$; the
remaining quotient lies strictly between $0$ and $g$, while the three
divisibilities force it to be divisible by $g$.  This establishes the four
alternatives without assuming $(a,b)=1$.

Define
\begin{align*}
\Delta_L&=rbS-(t+1)T=T\{r(L_+-a)-t-1\},\\
\Delta_U&=tS-rb(a+1)-S=S\{t-r(U_+-a)-1\}.
\end{align*}
Using $T=aS+b(a+1)$ gives
\[
\Delta_L=wS-T-tb(a+1),\qquad
\Delta_U=tb+(-w)(a+1)-S.
\]
If $w>0$ and $u-t\ge2$, then
\[
\Delta_L\ge2aS-T-(a-1)b(a+1)=a^2(a-b+1)>0.
\]
If $w>0$ and $u-t=1$, the remainder inequality gives
$\Delta_L=vS-b(a+1)(t+1)\ge0$.  Thus
$r(L_+-a)-t\ge1$.

If $w<0$ and $t-u\ge2$, then
\[
\Delta_U\ge(a+1)^2-S=a+1-b\ge1.
\]
If $w<0$ and $t-u=1$, the last remainder inequality gives
$\Delta_U=b(t-1)-v(a+1)\ge0$.  Thus
$t-r(U_+-a)\ge1$.  Since $w\ne0$, one of these alternatives applies and
proves Lemma~\ref{lem:positive-integer-exclusion}.

\subsection{The aperture-three triple classification}
\label{app:m3-triples}

For $m=3$, write each congruence in \eqref{eq:sliding-kernel} as
\[
e_t+Q_1e_{t+1}+Q_2e_{t+2}=k_tQ_3,\qquad
k_t\in\{-1,0,1\}.
\]

In the negative-conjugate chamber,
\[
(Q_1,Q_2,Q_3)=(a+1,a^2+a+b,a^3+a^2+2ab+b),\qquad |e_j|\le a.
\]
For one equation $x+Q_1y+Q_2z=kQ_3$, sign symmetry reduces to $k=0,1$.
When $k=0$, the bound $2Q_2>a(1+Q_1)$ gives $|z|\le1$; $z=0$ gives
$(x,y)=(0,0)$, and $z=1$ gives $(x,y)=(-b,-a)$.  When $k=1$, the bound
at $z\le a-1$ forces $z=a$, after which
$x+(a+1)y=b(a+1)$ gives $(x,y)=(0,b)$.  Hence the solution list is
\[
(x,y,z;k)\in
\{(0,0,0;0),\ \pm(b,a,-1;0),\ \pm(0,b,a;1)\}.
\]
Its terminal pairs are $(0,0)$, $\pm(a,-1)$, and $\pm(b,a)$.  A final
triple of type $\pm(0,b,a)$ has initial pair $\pm(0,b)$, which is not a
terminal pair.  A final triple of type $\pm(b,a,-1)$ forces the preceding
triple to be $\pm(0,b,a)$, whose initial pair again cannot be terminal.

In the positive-conjugate chamber put $C=a-b$ and $d=a-1$.  Then
\[
(Q_1,Q_2,Q_3)=(a,a^2-b,a^3-2ab),\qquad 2\le C\le d.
\]
For $x+ay+Q_2z=kQ_3$, the same bounds give
\[
(x,y,z;k)\in
\{(0,0,0;0),\ \pm(C,d,-1;0),
\ \pm(-b,C,d;1),\ \pm(C,C-1,d;1)\}.
\]
Indeed, for $k=0$ one has $|z|\le1$, and $z=1$ forces
$(x,y)=(-C,-d)$.  For $k=1$, the inequality at $z\le a-2$ forces $z=d$;
reducing $x+ay=aC-b$ modulo $a$ leaves precisely
$(x,y)=(-b,C)$ and $(C,C-1)$.  The terminal pairs are
$(0,0)$, $\pm(d,-1)$, $\pm(C,d)$, and $\pm(C-1,d)$.  Neither
$\pm(-b,C)$ nor $\pm(C,C-1)$ is terminal.  The remaining type
$\pm(C,d,-1)$ forces a preceding $\pm(-b,C,d)$, whose initial pair is not
terminal.

Thus, in either chamber, a nonzero triple cannot persist through all three
overlaps.  Reading the three equations backwards forces
$e_2=e_3=e_4=0$, then $e_1=0$, and finally $e_0=0$.
