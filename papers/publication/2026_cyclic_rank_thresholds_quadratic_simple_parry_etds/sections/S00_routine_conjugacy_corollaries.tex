\section{Routine dynamical consequences}\label{sec:supp-routine-dynamics}

This section records standard consequences removed from the main article.
They are not inputs to the arithmetic threshold or Fischer-cover arguments.

\begin{proposition}[Dynamical corollaries above threshold]
\label{prop:conjugacy-consequences}
For $m\ge3$, the conjugacy $\Phi_m$ identifies invariant Borel probability
measures on the full two-shift and on $Y_m$, preserving ergodicity, mixing
properties, Bernoulli-ness, the $K$-property, and entropy.  Moreover,
\[
\#\operatorname{Fix}(\sigma^n|_{Y_m})=2^n,
\qquad
\zeta_{Y_m}(z)=\frac1{1-2z}.
\]
\end{proposition}

\begin{proof}
Theorem~\ref{thm:two-sided-conjugacy} supplies a topological conjugacy and a
sliding-block inverse.  The measure correspondence and the listed
measure-theoretic properties are standard conjugacy invariants.  Period-$n$
points correspond bijectively to the $2^n$ period-$n$ points of the full
two-shift, and substitution in the Artin--Mazur definition gives
\[
\exp\left(\sum_{n\ge1}\frac{2^nz^n}{n}\right)=(1-2z)^{-1}.
\]
\end{proof}

\begin{proposition}[Entropy preservation for every aperture]
\label{prop:measure-entropy-preservation}
Let $m\ge2$, let $\nu$ be shift invariant on $\{0,1\}^{\mathbb Z}$, and put
$\eta=(\Phi_m)_*\nu$.  Then $h_\eta(\sigma)=h_\nu(\sigma)$.
\end{proposition}

\begin{proof}
For a raw block $A_{0:n+m-2}$ and its output block $Y_{0:n-1}$,
determinism and the resolving lift in Theorem~\ref{thm:finite-state-lift} give
\[
0\le H_\nu(A_{0:n+m-2})-H_\eta(Y_{0:n-1})
\le(m-1)\log2.
\]
Indeed, after the initial $(m-1)$-bit overlap state is specified, the output
block determines the raw block.  Division by $n$ and passage to the limit
prove the assertion.
\end{proof}

\begin{corollary}[Topological entropy]
\label{cor:entropy-preservation}
For every $m\ge2$, $h_{\rm top}(Y_m)=\log2$.
\end{corollary}

\begin{proof}
The factor inequality gives the upper bound.  Applying
Proposition~\ref{prop:measure-entropy-preservation} to the fair Bernoulli law
and then the variational principle gives the lower bound.
\end{proof}
