\subsection{Bounded sliding congruences}

The proof separates a recurrence estimate from one finite integer exclusion.
For $a\ge b\ge1$ put
\[
L_+=a+\frac{b}{a+b/(a+b/(a+1))},\qquad
U_+=a+\frac{b}{a+b/(a+1)}.
\]

\begin{lemma}[Positive-recurrence integer exclusion]
\label{lem:positive-integer-exclusion}
Let $1\le e\le a$, $0\le k\le b$, and $h=be-ak$.  If $h>0$, then
\[
hL_+-kb\ge1\qquad\hbox{or}\qquad kb-hU_+\ge1.
\]
\end{lemma}

\begin{proof}
The Euclidean divisions, including the non-coprime endpoint cases, are given
in Appendix~\ref{app:positive-integer-exclusion}.
\end{proof}

\begin{lemma}[Nearest-multiple separation]
\label{lem:nearest-multiple-separation}
Let $Q_0,Q_1,\ldots$ satisfy either \eqref{eq:qminus} or \eqref{eq:qplus},
write $s=1$ or $s=-1$ accordingly, and put
\[
d=\begin{cases}a,&s=1,\\a-1,&s=-1.\end{cases}
\]
Then, for $r\ge4$ and $1\le |e|\le d$,
\begin{equation}
\operatorname{dist}(beQ_{r-1},Q_r\mathbb Z)\ge Q_{r-2}.
\label{eq:quadratic-separation}
\end{equation}
\end{lemma}

\begin{proof}
Distance is unchanged by replacing $e$ with $-e$, so assume $e>0$, and put
$R_r=Q_{r-1}/Q_{r-2}$.

\emph{Recurrence ratios.}
The recurrence $R_{r+1}=a+sb/R_r$ and the initial values give
\begin{align}
a+\frac{b}{a+1}&\le R_r<a+\frac ba &&(s=1, r\ge3),
\label{eq:ratio-plus}\\
a-\frac{b}{a-1}&<R_r\le a-\frac ba &&(s=-1, r\ge3).
\label{eq:ratio-minus}
\end{align}
Applying the recurrence once more yields, for $r\ge4$,
\begin{equation}
L_+\le R_r\le U_+\quad(s=1),\qquad
L_-<R_r\le U_-\quad(s=-1),
\label{eq:sharpened-ratio-bounds}
\end{equation}
where
\[
L_-=a-\frac{b}{a-1},\qquad
U_-=a-\frac{b}{a-b/a}.
\]
For $s=1$ the defining fractional-linear map reverses order and exchanges
the two endpoint iterates; for $s=-1$ it preserves order on $(a-1,a)$.

\emph{Nearest multiple.}
Choose a nearest integer $k$ to $beQ_{r-1}/Q_r=be/R_{r+1}$.  The ratio
bounds give $0<be/R_{r+1}<b$, hence
\begin{equation}
0\le k\le b.
\label{eq:nearest-k-bound}
\end{equation}
With $h=be-ak$, the recurrence gives the exact error identity
\begin{equation}
beQ_{r-1}-kQ_r=(hR_r-skb)Q_{r-2}.
\label{eq:nearest-error}
\end{equation}

\emph{Chamber separation.}
For $s=1$, a negative $h$ gives $|hR_r-kb|>1$, while $h=0$ gives
$kb\ge1$.  If $h>0$, Lemma~\ref{lem:positive-integer-exclusion} and
$L_+\le R_r\le U_+$ give $|hR_r-kb|\ge1$.

For $s=-1$, only $h<0$ can cancel.  Let $\beta>1$ and $0<\alpha<1$ be the
roots of $x^2-ax+b$, put $g=-h=ak-be>0$, and consider
\[
\nu=k^2-aek+be^2=(k-e\alpha)(k-e\beta)\in\mathbb Z\setminus\{0\}.
\]
If $\nu<0$, then
\[
g\beta-kb=\beta^2(k-e\alpha)
=\frac{\beta^2|\nu|}{e\beta-k}>1,
\]
because $0<e\beta-k<\beta^2$.  Since $R_r>\beta$, this gives
$gR_r-kb>1$.

If $\nu>0$, set $\delta=e\alpha-k>0$, $D=a^2-b$, and
$H=(a-1)(\beta-\alpha)+1$.  Then
\[
g=e\alpha^2-a\delta>0,\qquad
\nu=\delta\{e(\beta-\alpha)+\delta\}\ge1,
\]
so $\delta>H^{-1}$.  Using $\alpha-ab/D=\alpha^3/D$ gives
\[
kb-gU_-
=\delta\left(\beta^2+\frac{a\alpha^3}{D}\right)
-\frac{e\alpha^5}{D}
>\frac{\beta^2}{H}-\frac{(a-1)\alpha^5}{D}>1.
\]
For the last inequality, direct expansion gives
\[
D-H=2\alpha^2+\alpha(\beta-1)+(\beta-1)>0
\]
and
\[
\beta^2-H=\beta-1-\alpha(1-\alpha)>(a-1)\alpha^5;
\]
indeed $\alpha^5<\alpha$ and $b\le a-2$ implies
$\beta(1-\alpha)\ge2-\alpha>1$.  Since $R_r\le U_-$, this gives
$kb-gR_r>1$.  The cases $h>0$ and $h=0$ have no cancellation and again
give $|hR_r+kb|\ge1$.

\emph{Distance estimate.}
Thus $|hR_r-skb|\ge1$ in both chambers.  Equation
\eqref{eq:nearest-error}, with $k$ chosen nearest, now gives
\[
\operatorname{dist}(beQ_{r-1},Q_r\mathbb Z)
=|beQ_{r-1}-kQ_r|\ge Q_{r-2}.
\]
\end{proof}

\begin{lemma}[Quadratic bounded sliding congruences]
\label{lem:quadratic-sliding-congruences}
Take either parameter chamber in
Lemma~\ref{lem:quadratic-pisot-chambers}, let $d=\lfloor\beta\rfloor$, and
let $(Q_j)$ be \eqref{eq:qminus} or \eqref{eq:qplus}.  If $m\ge3$ and
$e_0,\ldots,e_{2m-2}\in[-d,d]\cap\mathbb Z$ satisfy
\begin{equation}
\sum_{j=0}^{m-1}Q_j e_{t+j}\equiv0\pmod{Q_m}
\qquad(0\le t<m),
\label{eq:sliding-kernel}
\end{equation}
then $e_0=\cdots=e_{2m-2}=0$.
\end{lemma}

\begin{proof}
Write $s=1$ in the negative chamber and $s=-1$ in the positive chamber, so
$Q_{j+2}=aQ_{j+1}+sbQ_j$, and put $\delta=Q_1-a$.  Subtracting $a$ times
the congruence at $t+1$ and $sb$ times that at $t+2$ from the congruence at
$t$ gives
\begin{equation}
e_t+\delta e_{t+1}-sbQ_{m-1}e_{t+m+1}\equiv0\pmod{Q_m}.
\label{eq:quadratic-elimination}
\end{equation}

For $m\ge4$, the small part has modulus at most $d(1+\delta)$, whereas
\[
Q_{m-2}\ge Q_2=
\begin{cases}
a^2+a+b>2a=d(1+\delta),&s=1,\\
a^2-b>a-1=d(1+\delta),&s=-1.
\end{cases}
\]
Lemma~\ref{lem:nearest-multiple-separation} therefore forces
$e_{t+m+1}=0$ for $0\le t\le m-3$.  The last original congruence then gives
$e_{m-1}+Q_1e_m=0$; since $Q_1>d$ and the left side has modulus below
$Q_m$, both digits vanish.  Back substitution gives the remaining digits.

At $m=3$, each window sum is $k_tQ_3$ with
$k_t\in\{-1,0,1\}$.  Appendix~\ref{app:m3-triples} classifies the bounded
solutions of one window equation in each chamber and compares their
overlapping endpoint pairs.  The only three-window overlap is the zero one,
so $e_0=\cdots=e_4=0$.
\end{proof}

\subsection{Sharp threshold and extremal bases}

\begin{theorem}[Exact quadratic-Pisot threshold]
\label{thm:full-quadratic-pisot-threshold}
Let $\beta$ be an arbitrary quadratic Pisot number and use exactly the
greedy language, Parry data, rank, cyclic modulus and fold of
Definition~\ref{def:quadratic-parry-fold}.  Then
\[
m_*(\beta)=
\begin{cases}
3,&\beta\text{ is the root of }x^2-ax-a,\quad a\ge1,\\
3,&\beta\text{ is the root of }x^2-ax+1,\quad a\ge3,\\
2,&\text{otherwise}.
\end{cases}
\]
More strongly, for every $m\ge m_*(\beta)$ and every $n\ge m$, the
finite-block sliding fold is injective and
\begin{equation}
|\mathcal L_n(Y_{\beta,m})|
=(\lfloor\beta\rfloor+1)^{n+m-1}.
\label{eq:quadratic-pisot-block-count}
\end{equation}
At these apertures $\Phi_{\beta,m}$ is a topological conjugacy from the full
$(\lfloor\beta\rfloor+1)$-shift onto the SFT $Y_{\beta,m}$, with inverse of
memory zero and anticipation $m-1$.

Consequently the maximum of the threshold function on the full quadratic
Pisot family is three.  Its extremizers are exactly the negative-conjugate
diagonal $P^-_{a,a}$ and the positive-conjugate norm-one family
$P^+_{a,1}$.  The golden mean is the least extremizer, but it is not the
unique extremizer after passage from metallic units to all quadratic Pisot
bases.
\end{theorem}

The extremal equalities $d=\kappa$ identify the two constant raw points at
aperture two.  When $d>\kappa$, the formulas below decode every raw triple
from two output symbols.

\begin{proof}
Equality of two folded output blocks is, by
Proposition~\ref{prop:parry-rank-bijection}, equivalent to the sliding
congruences \eqref{eq:sliding-kernel} for their digit differences.  For
$m\ge3$, Lemma~\ref{lem:quadratic-sliding-congruences} makes every
length-$m$ finite-block map injective.  Consecutive decoded blocks cover a
longer raw block, proving injectivity for every $n\ge m$ and the count
\eqref{eq:quadratic-pisot-block-count}.  Overlapping the unique decoded
length-$m$ blocks gives a local inverse, and compatibility of consecutive
decoded blocks gives an SFT description by forbidden output words of length
at most $m+1$.

It remains to classify $m=2$.  Put
\[
\kappa=Q_2-dQ_1=
\begin{cases}b,&P^-_{a,b},\\ a-b,&P^+_{a,b}.\end{cases}
\]
Proposition~\ref{prop:parry-rank-bijection} gives the same rule in both
chambers:
\[
\Fold_{\beta,2}(u,v)=
\begin{cases}
(u,v),&v<d\text{ or }u<\kappa,\\
(u-\kappa,0),&v=d\text{ and }u\ge\kappa.
\end{cases}
\]
If $d=\kappa$, the constant points $0^{\mathbb Z}$ and $d^{\mathbb Z}$
have the constant zero image, so conjugacy fails.  This occurs exactly for
$P^-_{a,a}$ and $P^+_{a,1}$.

Assume $d>\kappa$.  Let a raw triple $(a_0,a_1,a_2)$ produce
$y_i=(p_i,q_i)=\Fold_{\beta,2}(a_i,a_{i+1})$.  The central digit is
\begin{equation}
a_1=
\begin{cases}
q_0,&q_0>0,\\
0,&q_0=0\text{ and }p_1=0,\\
d,&q_0=0\text{ and }p_1\in\{d-\kappa,d\},
\end{cases}
\label{eq:m2-central-decoder}
\end{equation}
because $q_0=0$ leaves only $a_1=0$ or $d$.  Then
\begin{equation}
a_0=
\begin{cases}
p_0,&q_0>0,\\
p_0,&q_0=0\text{ and }a_1=0,\\
p_0+\kappa,&q_0=0\text{ and }a_1=d,
\end{cases}
\label{eq:m2-left-decoder}
\end{equation}
and
\begin{equation}
a_2=
\begin{cases}
q_1,&q_1>0,\\
0,&q_1=0\text{ and }p_1=a_1,\\
d,&q_1=0\text{ and }p_1=a_1-\kappa.
\end{cases}
\label{eq:m2-right-decoder}
\end{equation}
The alternatives are disjoint because $0<\kappa<d$.  Thus two consecutive
outputs decode the raw triple.  Sliding this decoder proves finite-block
injectivity for every $n\ge2$, the block count, and a memory-zero inverse of
anticipation one.

At the two extremal loci aperture three succeeds by
Lemma~\ref{lem:quadratic-sliding-congruences}.  Lemma
\ref{lem:quadratic-pisot-chambers} covers all quadratic Pisot parameters.
\end{proof}
