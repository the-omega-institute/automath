\subsection{The two-window branch locus}
\label{sec:stab:branch-and-conjugacy}

\begin{theorem}[Two-window branch locus]\label{thm:m2-branch-locus}
Let
\[
y^*=(\ldots,00,00,00,\ldots)\in Y_2.
\]
Then
\[
\Phi_2^{-1}(y^*)=\{0^{\ZZ},1^{\ZZ}\},
\]
and every other point of $Y_2$ has a singleton preimage.
\end{theorem}

\begin{proof}
For $m=2$ the modulus is $F_4=3$, the weights are $1,2$, and
\[
X_2=\{00,10,01\}.
\]
The four raw windows give
\[
\Fold_2(00)=\Fold_2(11)=00,
\qquad \Fold_2(10)=10,
\qquad \Fold_2(01)=01.
\]
Thus $10$ and $01$ determine their raw pair, while $00$ says that the two
raw bits are equal.  An all-zero output consequently has exactly the two
constant lifts.

If $y\ne y^*$, choose an occurrence of $10$ or $01$.  It determines one raw
pair.  Moving right and left, each $00$ propagates equality and each nonzero
symbol specifies its pair.  This determines the whole lift.  Existence
follows from $y\in Y_2$.
\end{proof}

\begin{remark}\label{rem:sofic-comparison}
The graph $\mathcal G_m$ is a bi-resolving presentation of $Y_m$ with degree
at most $2^{m-1}$.  General resolving-map theory supplies this framework
\cite{LindMarcus1995,Ashley1991,Boyle1983}; the degree-one threshold is the
arithmetic content of Lemma~\ref{lem:m3-decoder} and Theorem
\ref{thm:block-invertibility}.
\end{remark}
