\section{Finite-window Fibonacci stabilization}\label{sec:stabilization}

The Fibonacci case makes the arithmetic visible before the two quadratic
chambers enter.  Reduction modulo $F_{m+2}$ selects a unique admissible word,
and overlap then resolves the ambiguity of isolated windows.

\subsection{Language, valuation, and cyclic fold}\label{sec:stab:golden}

Let $(F_n)_{n\ge1}$ be the Fibonacci sequence with $F_1=F_2=1$ and
$F_{n+2}=F_{n+1}+F_n$.

\begin{definition}[Golden-mean language and valuation]\label{def:Xm}
For $m\ge1$, let
\[
X_m=\{x\in\{0,1\}^m:x_ix_{i+1}=0\text{ for }1\le i<m\}.
\]
For $\omega=(\omega_1,\ldots,\omega_m)\in\{0,1\}^m$, define
\[
\Val_m(\omega)=\sum_{k=1}^m\omega_kF_{k+1}.
\]
\end{definition}

\paragraph{Finite-window Zeckendorf range.}\label{prop:finite-window-range}
For every $m\ge1$, the restriction
\[
\Val_m|_{X_m}:X_m\longrightarrow\{0,1,\ldots,F_{m+2}-1\}
\]
is a bijection.

\begin{proof}
By Zeckendorf's theorem, every nonnegative integer has a unique expansion in
nonconsecutive Fibonacci weights \cite{Zeckendorf1972,Lothaire2002}.  For
$N<F_{m+2}$ its largest weight is at most $F_{m+1}$, so the expansion gives
an element of $X_m$; uniqueness gives injectivity.
\end{proof}

\begin{definition}[Cyclic Fibonacci fold]\label{def:fold}
For $m\ge1$, define
\[
\Fold_m(\omega)=
(\Val_m|_{X_m})^{-1}\bigl(\Val_m(\omega)\bmod F_{m+2}\bigr),
\qquad \omega\in\{0,1\}^m,
\]
where the residue is represented in $\{0,\ldots,F_{m+2}-1\}$.
\end{definition}

\paragraph{Basic properties of $\Fold_m$.}\label{thm:fold-suite}
For every $m\ge1$, the map $\Fold_m$ is well defined, fixes every word in
$X_m$, is onto $X_m$, and its nonempty fibers partition $\{0,1\}^m$.

\begin{proof}
The finite-window range bijection gives a unique admissible representative
of every residue; admissible words represent themselves, and the remaining
claims follow immediately.
\end{proof}
