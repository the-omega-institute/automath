\begin{definition}[Dominant exact-kernel family]\label{def:kernel-chebotarev-dominant-exact-kernel-family}
Let \(G\) be a finite abelian group, let \(u\in(0,1]\), and assume Assumption \ref{thm:kernel-chebotarev-exp}. We say that the nontrivial twisted spectrum is dominated by the exact kernel \(H_\star\le G\) if there exist
\[
0\le \Gamma_\star(u)<\eta_\star(u)<\lambda_1(u),
\]
a nonempty set \(S_\star(u)\subseteq \widehat G\setminus\{\ind\}\), and nonzero complex amplitudes \(a_\chi(u)\) and dominant spectral values \(\omega_\chi(u)\) for \(\chi\in S_\star(u)\) such that
\[
\ker\chi=H_\star
\qquad(\chi\in S_\star(u)),
\]
\[
\abs{\omega_\chi(u)}=\eta_\star(u)
\qquad(\chi\in S_\star(u)),
\]
and
\[
B_n(\chi;u)
=
\frac{a_\chi(u)\omega_\chi(u)^n}{n}
+
O\!\left(\frac{\Gamma_\star(u)^n}{n}\right)
\qquad(\chi\in S_\star(u)),
\]
\[
B_n(\psi;u)
=
O\!\left(\frac{\Gamma_\star(u)^n}{n}\right)
\qquad(\psi\in \widehat G\setminus(S_\star(u)\cup\{\ind\})).
\]
\end{definition}

\begin{theorem}[Auxiliary abstract dominant-kernel bookkeeping]\label{thm:kernel-chebotarev-dominant-kernel-detection}
Assume the hypotheses of Definition \ref{def:kernel-chebotarev-dominant-exact-kernel-family}. Let
\[
\mu_{n,u}(c):=\frac{B_{n,c}(u)}{B_n(\ind;u)}
\qquad(c\in G)
\]
for all sufficiently large \(n\), so that \(B_n(\ind;u)>0\), and define
\[
\mathcal D_{n,u}(H)
:=
D\!\bigl((\pi_H)_\ast\mu_{n,u}\,\|\,\mathrm{unif}_{G/H}\bigr)
\qquad(H\le G).
\]
Set
\[
C_\star(u):=\sum_{\chi\in S_\star(u)}\abs{a_\chi(u)}^2>0,
\]
\[
\rho_\star(u)
:=
\max\!\left\{
\frac{\eta_\star(u)\Gamma_\star(u)}{\lambda_1(u)^2},
\left(\frac{\Gamma_\star(u)}{\lambda_1(u)}\right)^2,
\tau(u)^3
\right\},
\qquad
\tau(u):=\frac{\Lambda(u)}{\lambda_1(u)}.
\]
Then \(0\le \rho_\star(u)<1\), and for every subgroup \(H\le G\),
\[
\mathcal D_{n,u}(H)
=
\frac{C_\star(u)}{2}
\left(\frac{\eta_\star(u)}{\lambda_1(u)}\right)^{2n}
+
O\!\bigl(\rho_\star(u)^n\bigr)
\qquad\text{if }H\subseteq H_\star,
\]
whereas
\[
\mathcal D_{n,u}(H)=O\!\bigl(\rho_\star(u)^n\bigr)
\qquad\text{if }H\not\subseteq H_\star.
\]
Thus the quotients on which the leading entropy survives are exactly those with \(H\subseteq H_\star\), or equivalently those admitting the natural quotient map \(G/H\twoheadrightarrow G/H_\star\). In this sense \(G/H_\star\) is the coarsest quotient carrying the dominant second-order entropy.

Moreover,
\[
\sum_{J\supseteq H_\star}\mu_{\mathrm{sub}}(H_\star,J)\,\mathcal D_{n,u}(J)
=
\frac{C_\star(u)}{2}
\left(\frac{\eta_\star(u)}{\lambda_1(u)}\right)^{2n}
+
O\!\bigl(\rho_\star(u)^n\bigr).
\]
\end{theorem}

\begin{proof}
The inequalities \(\Gamma_\star(u)<\eta_\star(u)<\lambda_1(u)\) and \(\tau(u)<1\) show immediately that \(0\le \rho_\star(u)<1\).

For \(H\le G\), Theorem \ref{thm:kernel-chebotarev-quotient-lattice-tomography} gives
\[
\mathcal E_{n,u}(H)
:=
\chi^2\!\bigl((\pi_H)_\ast\mu_{n,u}\,\|\,\mathrm{unif}_{G/H}\bigr)
=
\frac{1}{B_n(\ind;u)^2}
\sum_{\chi\in H^\perp\setminus\{\ind\}}
\abs{B_n(\chi;u)}^2.
\]
On the other hand, Assumption \ref{thm:kernel-chebotarev-exp} implies
\[
B_n(\ind;u)
=
\frac{\lambda_1(u)^n}{n}\left(1+O\!\bigl(\tau(u)^n\bigr)\right),
\qquad
\frac{1}{B_n(\ind;u)^2}
=
\frac{n^2}{\lambda_1(u)^{2n}}\left(1+O\!\bigl(\tau(u)^n\bigr)\right).
\]

Fix \(\chi\in S_\star(u)\). By definition,
\[
\abs{B_n(\chi;u)}^2
=
\frac{\abs{a_\chi(u)}^2\eta_\star(u)^{2n}}{n^2}
+
O\!\left(\frac{\eta_\star(u)^n\Gamma_\star(u)^n}{n^2}\right)
+
O\!\left(\frac{\Gamma_\star(u)^{2n}}{n^2}\right).
\]
Summing over \(\chi\in S_\star(u)\) yields
\[
\sum_{\chi\in S_\star(u)}\abs{B_n(\chi;u)}^2
=
\frac{C_\star(u)\eta_\star(u)^{2n}}{n^2}
+
O\!\left(\frac{\eta_\star(u)^n\Gamma_\star(u)^n}{n^2}\right)
+
O\!\left(\frac{\Gamma_\star(u)^{2n}}{n^2}\right).
\]
For every \(\psi\notin S_\star(u)\cup\{\ind\}\), the defining hypothesis gives
\[
\abs{B_n(\psi;u)}^2
=
O\!\left(\frac{\Gamma_\star(u)^{2n}}{n^2}\right).
\]

Suppose first that \(H\subseteq H_\star\). Since every \(\chi\in S_\star(u)\) has kernel \(H_\star\), these characters all lie in \(H^\perp\). Therefore
\[
\sum_{\chi\in H^\perp\setminus\{\ind\}}\abs{B_n(\chi;u)}^2
=
\frac{C_\star(u)\eta_\star(u)^{2n}}{n^2}
+
O\!\left(\frac{\eta_\star(u)^n\Gamma_\star(u)^n}{n^2}\right)
+
O\!\left(\frac{\Gamma_\star(u)^{2n}}{n^2}\right).
\]
Multiplying by \(1/B_n(\ind;u)^2\) gives
\[
\mathcal E_{n,u}(H)
=
C_\star(u)\left(\frac{\eta_\star(u)}{\lambda_1(u)}\right)^{2n}
+
O\!\left(\left(\frac{\eta_\star(u)\Gamma_\star(u)}{\lambda_1(u)^2}\right)^n\right)
+
O\!\left(\left(\frac{\Gamma_\star(u)}{\lambda_1(u)}\right)^{2n}\right)
+
O\!\bigl(\tau(u)^{3n}\bigr).
\]
Here we used the elementary inequality
\[
\tau(u)^n\left(\frac{\eta_\star(u)}{\lambda_1(u)}\right)^{2n}\le \tau(u)^{3n},
\]
which holds because \(\eta_\star(u)\le \eta(u)\le \Lambda(u)\).

Now suppose that \(H\not\subseteq H_\star\). Then no character in \(S_\star(u)\) belongs to \(H^\perp\), since \(H\subseteq\ker\chi\) would force \(H\subseteq H_\star\). Hence only the error terms remain and
\[
\mathcal E_{n,u}(H)
=
O\!\left(\left(\frac{\Gamma_\star(u)}{\lambda_1(u)}\right)^{2n}\right).
\]

Theorem \ref{thm:kernel-chebotarev-entropy-tomography} now yields
\[
\mathcal D_{n,u}(H)=\frac12\,\mathcal E_{n,u}(H)+O\!\bigl(\tau(u)^{3n}\bigr)
\]
uniformly in \(H\). Combining this with the two estimates for \(\mathcal E_{n,u}(H)\) gives the claimed asymptotics for \(\mathcal D_{n,u}(H)\).

Finally, Theorem \ref{thm:kernel-chebotarev-entropy-tomography} also gives
\[
\sum_{J\supseteq H_\star}\mu_{\mathrm{sub}}(H_\star,J)\,\mathcal D_{n,u}(J)
=
\frac12\,\mathcal W_{n,u}(H_\star)+O\!\bigl(\tau(u)^{3n}\bigr).
\]
By Theorem \ref{thm:kernel-chebotarev-quotient-lattice-tomography}, \(\mathcal W_{n,u}(H_\star)\) is precisely the exact-kernel spectral mass at \(H_\star\), and the preceding computation shows that only the characters in \(S_\star(u)\) contribute at dominant size. Hence
\[
\mathcal W_{n,u}(H_\star)
=
C_\star(u)\left(\frac{\eta_\star(u)}{\lambda_1(u)}\right)^{2n}
+
O\!\bigl(\rho_\star(u)^n\bigr),
\]
which is the asserted formula.
\end{proof}

\begin{theorem}[Finite-field exact-kernel visibility obstruction and recovery]\label{thm:finite-field-exact-kernel-visibility-obstruction}
Let \(V\to U\) be a geometrically connected finite \etale{} abelian Galois
cover over \(\FF_q\) with group \(G\), and use the notation of Definition
\ref{def:finite-field-quotient-notation}.  Fix \(n\ge n_0(V/U)\), and set
\[
Y_n(H):=\frac{n^2N_n^2}{q^n}
\chi^2\!\left(\mu_n^H\middle\|\mathrm{unif}_{G/H}\right)
\qquad(H\le G).
\]
Then
\[
Y_n(H)=\sum_{K\supseteq H}E_n(K)
\qquad(H\le G).
\]
Consequently the full quotient \(\chi^2\)-profile recovers the exact-kernel
energy function by the subgroup-lattice inversion
\[
E_n(K)=\sum_{J\supseteq K}\mu_{\mathrm{sub}}(K,J)Y_n(J).
\]

More generally, let \(\mathcal O\) be any collection of observed subgroups.
The observation map
\[
E:K\longmapsto E(K)
\quad\rightsquigarrow\quad
\left(\sum_{K\supseteq H}E(K)\right)_{H\in\mathcal O}
\]
cannot distinguish two exact kernels \(K_1,K_2\le G\) for which
\[
\{H\in\mathcal O:H\subseteq K_1\}
=
\{H\in\mathcal O:H\subseteq K_2\}.
\]
Indeed, any transfer of energy from \(K_1\) to \(K_2\) leaves every observed
quotient-visible energy unchanged.  Thus exact-kernel detection from a
restricted quotient family is possible only after this finite subgroup-lattice
obstruction has been removed; for \(\mathcal O=\{H:H\le G\}\), it is removed
exactly by the M\"obius formula above.
\end{theorem}

\begin{proof}
The first identity is Lemma \ref{lem:finite-field-exact-quotient-normalization}
multiplied by \(n^2N_n^2/q^n\):
\[
Y_n(H)=
\sum_{\chi\in H^\perp\setminus\{\ind\}}|A_n(\chi)|^2.
\]
For a character \(\chi\), the condition \(\chi\in H^\perp\) is equivalent to
\(H\subseteq\ker\chi\).  Grouping characters by their exact kernel gives
\[
Y_n(H)=
\sum_{K\supseteq H}
\sum_{\substack{\chi\ne\ind\\ \ker\chi=K}}|A_n(\chi)|^2
=\sum_{K\supseteq H}E_n(K).
\]
M\"obius inversion on the finite subgroup lattice gives the displayed recovery
formula.

For the obstruction, write
\(\mathbf v_K(H)=1\) if \(H\subseteq K\) and \(0\) otherwise, for
\(H\in\mathcal O\).  The observation vector of a unit energy supported at
\(K\) is precisely \(\mathbf v_K\).  The stated equality of lower shadows in
\(\mathcal O\) says \(\mathbf v_{K_1}=\mathbf v_{K_2}\).  Hence replacing an
energy function \(E\) by
\[
E'(K_1)=E(K_1)-t,
\qquad
E'(K_2)=E(K_2)+t,
\qquad
E'(K)=E(K)\quad(K\notin\{K_1,K_2\}),
\]
for any real \(t\) for which the displayed values remain nonnegative, changes
the exact-kernel energies but not the observed vector
\((\sum_{K\supseteq H}E(K))_{H\in\mathcal O}\).  Thus the restricted quotient
family has a genuine finite-lattice indistinguishability obstruction.  When
all subgroups are observed, the M\"obius formula just proved provides an
explicit inverse, so no such obstruction remains.
\end{proof}

\begin{theorem}[Persistent exact-kernel energy detection]\label{thm:kernel-chebotarev-persistent-energy-detection}
Let \(V\to U\) be a geometrically connected finite \etale{} abelian Galois cover over \(\FF_q\) with group \(G\), and use the notation of Definition \ref{def:finite-field-quotient-notation}. For \(K\le G\), put
\[
\overline E_N(K):=\frac1N\sum_{n=1}^N E_n(K),
\]
and define the persistent support
\[
\mathcal P:=\left\{K\le G:\limsup_{N\to\infty}\overline E_N(K)>0\right\}.
\]
If \(\mathcal P\ne\emptyset\) and
\[
K_{\mathrm{per}}:=\bigcap_{K\in\mathcal P}K,
\]
then \(G/K_{\mathrm{per}}\) is the coarsest quotient through which every persistently energetic exact-kernel character is visible. For each \(K\in\mathcal P\) there is a subsequence \(n_r\to\infty\) and a constant \(c_K>0\) such that
\[
E_{n_r}(K)\ge c_K,
\]
and along this subsequence the entropy inversion formula of Theorem \ref{thm:finite-field-chebotarev-entropy-tomography} detects the same kernel:
\[
2q^{n_r}\sum_{J\supseteq K}\mu_{\mathrm{sub}}(K,J)
D\!\left(\mu_{n_r}^J\middle\|\mathrm{unif}_{G/J}\right)
\ge \frac{c_K}{2}
\]
for all sufficiently large \(r\).

Moreover, let
\[
A_n(\chi)=\sum_{\zeta}b_{\chi,\zeta}\zeta^n+e_n(\chi),
\qquad |\zeta|=1,
\qquad
\frac1N\sum_{n\le N}|e_n(\chi)|^2\longrightarrow0,
\]
be the finite unitary expansion supplied by Proposition \ref{prop:finite-field-unitary-artin-expansion}, after combining equal frequencies for each \(\chi\). Then the explicit criterion
\[
\lim_{N\to\infty}\overline E_N(K)
=
\sum_{\substack{\chi\ne\ind\\ \ker\chi=K}}
\sum_{\zeta}\left|b_{\chi,\zeta}\right|^2
\]
holds. Thus \(K\in\mathcal P\) exactly when the unitary Artin signal, including the possible quadratic imprimitive term, is not identically canceled for at least one character of exact kernel \(K\).
\end{theorem}

\begin{proof}
If \(K\in\mathcal P\), then \(\limsup_N\overline E_N(K)>0\). Hence there is a constant \(c_K>0\) and infinitely many \(n\) with \(E_n(K)\ge c_K\); otherwise all sufficiently large values would be below every positive threshold and the Cesaro limsup would vanish. This gives the subsequence. A character with kernel \(K\) is visible on the quotient \(G/H\) exactly when \(H\subseteq K\). Therefore all persistent exact-kernel characters are visible precisely on quotients \(G/H\) with \(H\subseteq\bigcap_{K\in\mathcal P}K=K_{\mathrm{per}}\), and \(G/K_{\mathrm{per}}\) is the coarsest such quotient. Theorem \ref{thm:finite-field-chebotarev-entropy-tomography} gives
\[
2q^n\sum_{J\supseteq K}\mu_{\mathrm{sub}}(K,J)
D(\mu_n^J\|\mathrm{unif}_{G/J})
=E_n(K)+O(q^{-n/2}),
\]
so the lower bound \(c_K/2\) holds along the chosen subsequence for all large \(r\).

For the explicit criterion, use the elementary mean-square identity for finite unitary exponential sums. If
\[
f(n)=\sum_{a=1}^m b_a\zeta_a^n,
\qquad |\zeta_a|=1,
\]
then
\[
\frac1N\sum_{n=1}^N |f(n)|^2
=\sum_{a,b}b_a\overline{b_b}\frac1N\sum_{n=1}^N(\zeta_a\overline{\zeta_b})^n.
\]
The inner Cesaro average tends to \(1\) when \(\zeta_a=\zeta_b\) and to \(0\) otherwise. After combining equal frequencies, the limit is therefore \(\sum_\zeta |b_\zeta|^2\). The Cesaro-square negligibility of \(e_n(\chi)\) proved in Proposition \ref{prop:finite-field-unitary-artin-expansion}, together with Cauchy--Schwarz, shows that replacing \(A_n(\chi)\) by its unitary exponential part does not change the limiting mean square. Summing over the finite set of characters with exact kernel \(K\) gives the displayed formula and the stated noncancellation criterion.
\end{proof}
