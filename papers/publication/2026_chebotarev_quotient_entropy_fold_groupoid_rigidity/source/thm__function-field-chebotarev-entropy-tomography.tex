\begin{definition}[Finite-field quotient notation]\label{def:finite-field-quotient-notation}
Let \(U/\FF_q\) be a smooth geometrically connected curve and let \(V\to U\) be a geometrically connected finite \etale abelian Galois cover with group \(G\). For \(c\in G\) put
\[
B_n(c):=\#\{v\in |U|:\deg v=n,\ \mathrm{Frob}_v=c\},
\qquad
N_n:=\sum_{c\in G}B_n(c),
\]
and, for all \(n\) with \(N_n>0\), define the primitive Frobenius-label law
\[
\mu_n(c):=\frac{B_n(c)}{N_n}\qquad(c\in G).
\]
For a subgroup \(H\le G\) write
\[
\pi_H:G\twoheadrightarrow G/H,
\qquad
\mu_n^H:=(\pi_H)_\ast\mu_n.
\]
For \(\chi\in\widehat G\) define the twisted degree-\(n\) prime sum and its square-root normalization by
\[
B_n(\chi):=\sum_{c\in G}\chi(c)B_n(c),
\qquad
A_n(\chi):=\frac{nB_n(\chi)}{q^{n/2}}.
\]
Finally, for \(K\le G\), define the exact-kernel Artin energy
\[
E_n(K):=
\sum_{\substack{\chi\in\widehat G\setminus\{\ind\}\\ \ker\chi=K}}
|A_n(\chi)|^2,
\]
and denote by \(\mu_{\mathrm{sub}}(K,J)\) the M\"obius function of the finite subgroup lattice of \(G\), ordered by inclusion.
\end{definition}

\begin{proposition}[Normalized reciprocal-zero formula]\label{prop:finite-field-normalized-reciprocal-zero-formula}
Assume the notation of Definition \ref{def:finite-field-quotient-notation}. Let \(\chi\in\widehat G\setminus\{\ind\}\), and write the Artin \(L\)-function of \(\chi\) on \(U\) in the form
\[
L(U,\chi,T)=\prod_{j=1}^{m_\chi}(1-\alpha_{\chi,j}T)
\]
after the trivial factors have been separated. Then
\[
A_n(\chi)
=
-q^{-n/2}\sum_{j=1}^{m_\chi}\alpha_{\chi,j}^{\,n}
+
R_n(\chi),
\]
where
\[
R_n(\chi):=
-q^{-n/2}
\sum_{\substack{d\mid n\\ d<n}}
 d\sum_{\substack{v\in |U|\\ \deg v=d}}
 \chi\!\left(\mathrm{Frob}_v^{n/d}\right).
\]
The proper-divisor term satisfies \(R_n(\chi)=O_U(1)\), uniformly in \(\chi\) and \(n\). If \(U=C\) is projective, there is no boundary contribution. For a nonprojective open \(U\subset C\), the finitely many omitted boundary points change the corresponding projective formula by at most \(O_U(nq^{-n/2})\) after the same normalization.
\end{proposition}

\begin{proof}
The logarithmic derivative of the Euler product gives, for every \(r\ge1\),
\[
\sum_{d\mid r}d\sum_{\substack{v\in |U|\\ \deg v=d}}
\chi\!\left(\mathrm{Frob}_v^{r/d}\right)
=
-\sum_{j=1}^{m_\chi}\alpha_{\chi,j}^{\,r}.
\]
Taking \(r=n\) and isolating the term \(d=n\) gives
\[
nB_n(\chi)
=
-\sum_{j=1}^{m_\chi}\alpha_{\chi,j}^{\,n}
-
\sum_{\substack{d\mid n\\ d<n}}
 d\sum_{\substack{v\in |U|\\ \deg v=d}}
 \chi\!\left(\mathrm{Frob}_v^{n/d}\right).
\]
Division by \(q^{n/2}\) is the displayed formula. Since \(|\chi(\mathrm{Frob}_v^{n/d})|=1\), the proper-divisor sum is bounded in absolute value by
\[
\sum_{\substack{d\mid n\\d<n}}d\#\{v\in |U|:\deg v=d\}
\le
\sum_{d\le n/2}\#U(\FF_{q^d})
=O_U(q^{n/2}),
\]
which proves the uniform bound for \(R_n(\chi)\). If \(U\subset C\) is obtained by removing a finite set \(S\), then each boundary point of degree dividing \(n\) contributes at most its degree to the closed-point sum, so the total boundary correction in \(nB_n(\chi)\) is \(O_U(n)\), hence is \(O_U(nq^{-n/2})\) after normalization. In the projective case \(S=\emptyset\).
\end{proof}

\begin{theorem}[Finite-field Chebotarev quotient entropy tomography]\label{thm:finite-field-chebotarev-entropy-tomography}
Let \(V\to U\) be a geometrically connected finite \etale abelian Galois cover of a smooth geometrically connected curve over \(\FF_q\), with group \(G\). Use the notation of Definition \ref{def:finite-field-quotient-notation}. Then the quantities \(A_n(\chi)\) are bounded uniformly in \(n\) and \(\chi\ne\ind\), and, uniformly for all subgroups \(H\le G\),
\[
\chi^2\!\left(\mu_n^H\,\middle\|\,\mathrm{unif}_{G/H}\right)
=
q^{-n}\sum_{\chi\in H^\perp\setminus\{\ind\}}|A_n(\chi)|^2
+O(q^{-3n/2}),
\]
where \(H^\perp:=\{\chi\in\widehat G:\chi|_H\equiv1\}\). Moreover
\[
D\!\left(\mu_n^H\,\middle\|\,\mathrm{unif}_{G/H}\right)
=
\frac12 q^{-n}\sum_{\chi\in H^\perp\setminus\{\ind\}}|A_n(\chi)|^2
+O(q^{-3n/2})
\]
uniformly in \(H\). Consequently, for every \(K\le G\),
\[
E_n(K)
=
2q^n\sum_{J\supseteq K}\mu_{\mathrm{sub}}(K,J)
D\!\left(\mu_n^J\,\middle\|\,\mathrm{unif}_{G/J}\right)
+O(q^{-n/2}),
\]
uniformly in \(K\). At the quadratic level the exact identity is
\[
\sum_{J\supseteq K}\mu_{\mathrm{sub}}(K,J)
\chi^2\!\left(\mu_n^J\,\middle\|\,\mathrm{unif}_{G/J}\right)
=
\frac{q^n}{n^2N_n^2}\,E_n(K).
\]
Equivalently, since \(N_n=q^n/n+O(q^{n/2}/n)\), the left-hand side is
\[
q^{-n}E_n(K)+O(q^{-3n/2}).
\]
\end{theorem}

\begin{proof}
Proposition \ref{prop:finite-field-normalized-reciprocal-zero-formula}, together with the Riemann hypothesis for curves over finite fields, gives \(|\alpha_{\chi,j}|\le q^{1/2}\) and hence \(A_n(\chi)=O(1)\), uniformly over the finite set \(\widehat G\setminus\{\ind\}\). The prime number theorem for the trivial channel gives
\[
N_n=\frac{q^n}{n}+O\!\left(\frac{q^{n/2}}{n}\right),
\qquad
\varepsilon_n:=\frac{nN_n}{q^n}-1=O(q^{-n/2}).
\]
For a fixed quotient \(Q_H=G/H\), Proposition \ref{thm:kernel-chebotarev-visible-hidden-character-energy} applied to \(\pi_H:G\to Q_H\) gives the exact Parseval identity
\[
\chi^2\!\left(\mu_n^H\middle\|\mathrm{unif}_{Q_H}\right)
=
\frac{1}{N_n^2}\sum_{\chi\in H^\perp\setminus\{\ind\}}|B_n(\chi)|^2.
\]
Since \(B_n(\chi)=q^{n/2}A_n(\chi)/n\), this becomes
\[
\chi^2\!\left(\mu_n^H\middle\|\mathrm{unif}_{Q_H}\right)
=
q^{-n}(1+\varepsilon_n)^{-2}
\sum_{\chi\in H^\perp\setminus\{\ind\}}|A_n(\chi)|^2.
\]
The sums over characters are uniformly bounded, because \(G\) is fixed and the normalized amplitudes are uniformly bounded. Expanding \((1+\varepsilon_n)^{-2}=1+O(q^{-n/2})\) proves the asserted \(\chi^2\)-asymptotic.

It remains to pass from \(\chi^2\) to relative entropy uniformly over the subgroup lattice. Put
\[
r_n^H(\bar c):=|Q_H|\mu_n^H(\bar c)=1+\delta_n^H(\bar c).
\]
Fourier inversion on \(Q_H\), the preceding bound for \(B_n(\chi)/N_n\), and the finiteness of \(G\) give
\[
\max_{H\le G}\max_{\bar c\in Q_H}|\delta_n^H(\bar c)|=O(q^{-n/2}).
\]
For \(|x|\le1/2\),
\[
(1+x)\log(1+x)=x+\frac{x^2}{2}+O(|x|^3).
\]
For all sufficiently large \(n\) this expansion may be applied to every \(\delta_n^H(\bar c)\), and \(
|Q_H|^{-1}\sum_{\bar c}\delta_n^H(\bar c)=0\). Hence
\[
D(\mu_n^H\|\mathrm{unif}_{Q_H})
=
\frac12\frac1{|Q_H|}\sum_{\bar c\in Q_H}|\delta_n^H(\bar c)|^2
+O(q^{-3n/2})
=
\frac12\chi^2(\mu_n^H\|\mathrm{unif}_{Q_H})+O(q^{-3n/2}),
\]
uniformly in \(H\). This proves the relative-entropy expansion.

For the inversion formula, the visibility condition \(\chi\in H^\perp\) is equivalent to \(H\subseteq\ker\chi\). Thus
\[
\sum_{\chi\in H^\perp\setminus\{\ind\}}|A_n(\chi)|^2
=
\sum_{J\supseteq H}E_n(J).
\]
M\"obius inversion on the finite subgroup lattice gives
\[
E_n(K)=
\sum_{J\supseteq K}\mu_{\mathrm{sub}}(K,J)
\sum_{\chi\in J^\perp\setminus\{\ind\}}|A_n(\chi)|^2.
\]
Substituting the exact Parseval identity before the expansion gives the displayed exact \(\chi^2\)-identity. Substituting the entropy expansion and multiplying by \(2q^n\) gives the relative-entropy recovery formula, because the subgroup lattice is finite and \(2q^nO(q^{-3n/2})=O(q^{-n/2})\).
\end{proof}

\begin{example}[Cyclic Kummer cover]\label{ex:cyclic-kummer-quotient-tomography}
Assume \(\ell\) is an odd prime with \(\ell\mid(q-1)\), and consider the connected Kummer cover
\[
V=\Spec \FF_q[z,z^{-1}]\longrightarrow U=\mathbb{G}_{m,\FF_q},
\qquad t=z^\ell.
\]
It is finite \etale over \(U\) with Galois group \(G\cong\mu_\ell\cong\ZZ/\ell\ZZ\). The subgroup lattice has only \(\{1\}\subset G\), and every nontrivial character has exact kernel \(\{1\}\). Therefore
\[
E_n(\{1\})=\sum_{k=1}^{\ell-1}|A_n(\chi_k)|^2,
\qquad
E_n(G)=0.
\]
The full quotient \(G/\{1\}\) is the only nontrivial quotient, while the quotient by \(G\) is a point. Since
\[
\mu_{\mathrm{sub}}(\{1\},\{1\})=1,
\qquad
\mu_{\mathrm{sub}}(\{1\},G)=-1,
\]
and \(D((\pi_G)_\ast\mu_n\|\mathrm{unif}_{G/G})=0\), Theorem \ref{thm:finite-field-chebotarev-entropy-tomography} gives the concrete recovery formula
\[
E_n(\{1\})
=
2q^nD(\mu_n\|\mathrm{unif}_G)+O(q^{-n/2}).
\]
Thus, in this cyclic Kummer case, the single nontrivial quotient entropy recovers exactly the total nontrivial Artin square-root energy.
\end{example}

\begin{example}[Biquadratic Artin--Schreier quotient lattice]\label{ex:biquadratic-artin-schreier-quotient-lattice}
Let \(q\) be odd and let \(V\to U\) be a geometrically connected finite \etale biquadratic Kummer cover obtained by adjoining two independent quadratic characters on a smooth affine curve \(U/\FF_q\). Then \(G\cong(\ZZ/2\ZZ)^2\). There are three order-two subgroups \(H_1,H_2,H_3\), and the three nontrivial characters \(\chi_i\) have kernels \(H_i\). Hence
\[
E_n(H_i)=|A_n(\chi_i)|^2,
\qquad
E_n(\{1\})=E_n(G)=0.
\]
For each \(i\), the quotient by \(H_i\) sees only \(\chi_i\), so the exact \(\chi^2\)-identity gives
\[
\chi^2\!\left((\pi_{H_i})_\ast\mu_n\middle\|\mathrm{unif}_{G/H_i}\right)
=
\frac{q^n}{n^2N_n^2}|A_n(\chi_i)|^2.
\]
The full quotient by \(\{1\}\) sees all three nontrivial characters and gives the cumulative identity
\[
\chi^2(\mu_n\|\mathrm{unif}_G)
=
\frac{q^n}{n^2N_n^2}
\sum_{i=1}^3|A_n(\chi_i)|^2.
\]
Thus the subgroup lattice separates the three exact-kernel energies individually from the three index-two quotient entropies, while the full quotient records their sum.
\end{example}
