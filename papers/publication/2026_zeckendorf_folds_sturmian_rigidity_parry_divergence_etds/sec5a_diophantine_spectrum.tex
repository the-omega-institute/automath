\subsection{Diophantine R\'enyi Spectrum Beyond Bounded Type}

The bounded-type estimate above suppresses a second phenomenon. For a general
irrational slope, a large continued-fraction coefficient creates many short
Sturmian cylinders together with a smaller family of long cylinders. The
relative influence of these two families depends on the R\'enyi order. The
three-gap multiplicities permit this dependence to be determined exactly on
the logarithmic scale.

We first record the continued-fraction form of the factor-frequency formula.
Write
\[
\alpha=[0;a_1,a_2,\ldots],
\qquad
\frac{p_n}{q_n}=[0;a_1,\ldots,a_n],
\]
with \(q_{-1}:=0\), \(p_0:=0\), \(q_0:=1\), and set
\[
\delta_{-1}:=1,
\qquad
\delta_n:=|q_n\alpha-p_n|\quad(n\ge0).
\]

\begin{lemma}[Exact Sturmian factor-frequency profile]
\label{lem:exact-sturmian-frequency-profile}
Let \(N=m+1\). There are unique integers \(n\ge0\),
\(1\le r\le a_{n+1}\), and \(0\le s<q_n\) such that
\begin{equation}\label{eq:three-gap-ostrowski-block}
N=r q_n+q_{n-1}+s.
\end{equation}
The positive masses of the length-\(m\) Sturmian factor law \(\mu_m\) are
\[
\begin{array}{c|c}
\text{mass}&\text{multiplicity}\\ \hline
\delta_n&N-q_n\\
\delta_{n-1}-r\delta_n&s\\
\delta_{n-1}-(r-1)\delta_n&q_n-s,
\end{array}
\]
where a row of zero multiplicity is omitted. Consequently, for
\(q\in(0,\infty)\setminus\{1\}\),
\begin{align}
\exp\bigl((1-q)H_q(\mu_m)\bigr)
={}&(N-q_n)\delta_n^q
+s(\delta_{n-1}-r\delta_n)^q \notag\\
&+(q_n-s)(\delta_{n-1}-(r-1)\delta_n)^q.
\label{eq:exact-sturmian-renyi-sum}
\end{align}
For \(q=1\), the corresponding formula is obtained by summing
\(-k x\log x\) over the three mass--multiplicity pairs \((x,k)\).
\end{lemma}

\begin{proof}
The length-\(m\) cylinders are the intervals cut out on the circle by the
\(N=m+1\) points
\[
\{0,-\alpha,\ldots,-m\alpha\}.
\]
The continued-fraction
form of the three-distance theorem gives exactly the three lengths and
multiplicities displayed above; see
\cite{Slater1967GapsSteps,Berthe1996SturmianFrequencies}. The intervals
between consecutive blocks
\[
q_n+q_{n-1}\le N<q_{n+1}+q_n
\]
are partitioned uniquely by~\eqref{eq:three-gap-ostrowski-block}, which gives
the uniqueness of \(n,r,s\). Since cylinder mass equals interval length,
substitution into the definition of R\'enyi entropy gives
\eqref{eq:exact-sturmian-renyi-sum} and its Shannon limit.
\end{proof}

Let the irrationality exponent of \(\alpha\) be
\[
\mu(\alpha)
:=
\sup\left\{\zeta>0:
\left|\alpha-\frac pq\right|<q^{-\zeta}
\text{ for infinitely many }(p,q)\in\mathbb Z\times\mathbb N
\right\}.
\]
The standard convergent estimates give
\begin{equation}\label{eq:irrationality-exponent-denominators}
\tau(\alpha)
:=
\limsup_{n\to\infty}\frac{\log q_{n+1}}{\log q_n}
=\mu(\alpha)-1\in[1,\infty].
\end{equation}
For \(\tau<\infty\), define
\[
\kappa_q(\tau)
:=
\begin{cases}
\dfrac{1}{1+q(\tau-1)},&0\le q\le1,\\[0.9em]
\dfrac{1}{\tau},&1\le q\le\infty.
\end{cases}
\]
At \(\tau=\infty\), set
\(\kappa_0(\infty):=1\) and
\(\kappa_q(\infty):=0\) for \(q>0\).

\begin{theorem}[Diophantine R\'enyi spectrum and Parry correction]
\label{thm:diophantine-renyi-spectrum}
Assume \(\beta=\alpha\), and let \(\alpha\in(0,1)\) be irrational. Then for
every \(q\in[0,\infty]\),
\begin{equation}\label{eq:sturmian-renyi-spectrum}
\liminf_{m\to\infty}
\frac{H_q(\mu_m)}{\log m}
=
\kappa_q\bigl(\tau(\alpha)\bigr).
\end{equation}
The same spectrum is the logarithmic correction to both optimal injective
placement and the canonical folded Parry divergence:
\begin{equation}\label{eq:parry-diophantine-spectrum}
\liminf_{m\to\infty}
\frac{m\log\varphi-L_q(m)}{\log m}
=
\liminf_{m\to\infty}
\frac{m\log\varphi-D_q(\pi_m^{\mathrm{fold}}\|\pi_m)}{\log m}
=
\kappa_q\bigl(\tau(\alpha)\bigr).
\end{equation}
Equivalently, if \(2\le\mu(\alpha)<\infty\), the coefficient in
\eqref{eq:sturmian-renyi-spectrum} and
\eqref{eq:parry-diophantine-spectrum} is
\[
\begin{cases}
\bigl(1+q(\mu(\alpha)-2)\bigr)^{-1},&0\le q\le1,\\
\bigl(\mu(\alpha)-1\bigr)^{-1},&1\le q\le\infty.
\end{cases}
\]
Thus every slope of irrationality exponent two has the full logarithmic
coefficient at every order. If \(\alpha\) is Liouville, the coefficient is
one at \(q=0\) and zero at every \(q>0\).
\end{theorem}

\begin{proof}
Put \(N=m+1\), and use the unique representation
\eqref{eq:three-gap-ostrowski-block}. Throughout the proof write
\[
Q:=q_n,\qquad Q^+:=q_{n+1},\qquad A:=a_{n+1}.
\]
The standard continued-fraction identities and estimates are
\begin{equation}\label{eq:continued-fraction-gap-estimates}
\delta_{n-1}=A\delta_n+\delta_{n+1},\qquad
\frac{1}{Q^++Q}<\delta_n<\frac{1}{Q^+}.
\end{equation}
Moreover, the largest of the three gaps in
Lemma~\ref{lem:exact-sturmian-frequency-profile} is smaller than
\(\delta_{n-1}+\delta_n<2/Q\). Hence
\begin{equation}\label{eq:entropy-mesh-lower-bound}
H_q(\mu_m)\ge H_\infty(\mu_m)>\log Q-\log2
\qquad(q\in[0,\infty]).
\end{equation}

We first prove the lower bound in~\eqref{eq:sturmian-renyi-spectrum} when
\(\tau:=\tau(\alpha)<\infty\). Fix \(q\in(0,1)\). If \(N\ge2Q\), the first
row of Lemma~\ref{lem:exact-sturmian-frequency-profile} and
\eqref{eq:continued-fraction-gap-estimates} give
\begin{equation}\label{eq:small-gap-entropy-lower-bound}
H_q(\mu_m)
\ge
\frac{\log N-\log2-q\log(Q^++Q)}{1-q}.
\end{equation}
If \(N<2Q\), estimate~\eqref{eq:entropy-mesh-lower-bound} already gives
\(H_q(\mu_m)/\log N\to1\) along any such subsequence.

Given \(\varepsilon>0\), for all sufficiently large \(n\),
\[
\log(Q^++Q)\le(\tau+\varepsilon)\log Q.
\]
Combining~\eqref{eq:entropy-mesh-lower-bound} and
\eqref{eq:small-gap-entropy-lower-bound}, and writing
\(x:=\log N/\log Q\), yields
\[
\frac{H_q(\mu_m)}{\log N}
\ge
\max\left\{
\frac1x,
\frac{x-q(\tau+\varepsilon)}{(1-q)x}
\right\}-o(1).
\]
The minimum of the maximum on the right over \(x\ge1\) is attained where
the two terms agree and equals
\[
\frac{1}{1+q(\tau+\varepsilon-1)}.
\]
Letting \(\varepsilon\downarrow0\) proves the required lower bound for
\(q\in(0,1)\). For \(q\in[1,\infty]\),
\eqref{eq:entropy-mesh-lower-bound} and \(N<Q^++Q\) give directly
\[
\liminf_{m\to\infty}\frac{H_q(\mu_m)}{\log m}\ge\frac1\tau.
\]
Finally, \(H_0(\mu_m)=\log(m+1)\), so the order-zero lower bound is an
equality. If \(\tau=\infty\), these arguments are needed only at \(q=0\),
since the asserted lower bound is zero for every positive order.

It remains to attain the lower bounds. The case \(\tau=1\) follows at once
from \(H_q(\mu_m)\le H_0(\mu_m)=\log(m+1)\), so suppose \(\tau>1\). Choose
a subsequence of convergents along which
\[
\frac{\log Q^+}{\log Q}\longrightarrow\tau.
\]
Along this subsequence \(A\to\infty\), and
\begin{equation}\label{eq:partial-quotient-log-scale}
\frac{\log A}{\log Q}\longrightarrow\tau-1,
\end{equation}
with the evident extended-real interpretation when \(\tau=\infty\).

Fix \(q\in(0,1)\), choose \(r=\lfloor A^q\rfloor\), \(s=0\), and set
\(N=rQ+q_{n-1}\). Lemma~\ref{lem:exact-sturmian-frequency-profile} then
has two nonzero rows. From~\eqref{eq:continued-fraction-gap-estimates},
\[
\begin{aligned}
\exp\bigl((1-q)H_q(\mu_{N-1})\bigr)
&=(N-Q)\delta_n^q
+Q\bigl(\delta_{n-1}-(r-1)\delta_n\bigr)^q\\
&\le (r+1)Q(AQ)^{-q}+2^q Q^{1-q}
\le (2+2^q)Q^{1-q}
\end{aligned}
\]
for all sufficiently large \(n\). Thus
\(H_q(\mu_{N-1})\le\log Q+O_q(1)\), while
\[
\log N=\log Q+q\log A+o(\log Q+\log A).
\]
Together with~\eqref{eq:partial-quotient-log-scale}, this gives the matching
upper bound
\[
\liminf_{m\to\infty}\frac{H_q(\mu_m)}{\log m}
\le\frac{1}{1+q(\tau-1)}.
\]

For \(q\in[1,\infty]\), choose instead
\[
r=\left\lfloor\frac{A}{\log A}\right\rfloor,\qquad
s=0,\qquad
N=rQ+q_{n-1}.
\]
There are \(N-Q\) gaps of length \(\delta_n\) and \(Q\) gaps of the other
length. If
\(\theta:=(N-Q)\delta_n\) is the total mass of the first family, then
\(\theta=O(1/\log A)\). The Shannon entropy therefore decomposes as
\[
H_1(\mu_{N-1})
=h(\theta)+\theta\log(N-Q)+(1-\theta)\log Q
\le\log Q+O(1),
\]
where \(h(t)=-t\log t-(1-t)\log(1-t)\). By monotonicity of R\'enyi entropy,
the same upper bound holds for every \(q\in[1,\infty]\). Since
\[
\log N=\log Q+\log A-\log\log A+o(\log Q+\log A),
\]
equation~\eqref{eq:partial-quotient-log-scale} gives the upper coefficient
\(1/\tau\). This completes the proof of~\eqref{eq:sturmian-renyi-spectrum}.

For \(m\ge6\), Theorem~\ref{thm:optimal-injective-relabeling} gives
\[
m\log\varphi-L_q(m)
=H_q(\mu_m)+\log\varphi+\log\pi(0),
\]
while Theorem~\ref{thm:full-renyi-defect-formula} gives
\[
0\le D_q(\pi_m^{\mathrm{fold}}\|\pi_m)-L_q(m)
\le2\log\varphi.
\]
The two quantities in~\eqref{eq:parry-diophantine-spectrum} therefore differ
from \(H_q(\mu_m)\) by bounded terms, uniformly in \(m\). Dividing by
\(\log m\) proves~\eqref{eq:parry-diophantine-spectrum}.
\end{proof}

\begin{remark}[Order-one transition]
For \(2<\mu(\alpha)<\infty\), the spectrum is strictly decreasing on
\([0,1]\) and constant on \([1,\infty]\). The change at \(q=1\) separates the
orders that retain a contribution from the numerous short cylinders from the
orders controlled by the sparse long cylinders. At irrationality exponent
two the two scales have the same logarithmic exponent, which is why the
spectrum collapses to the constant value one even when the partial quotients
are unbounded.
\end{remark}
