\section{One-position overlap chains of bounded zero representations}
\label{sec:bounded-zero-transients}

Let $U=(u_n)_{n\ge0}$ be a strictly increasing sequence of positive
integers with $u_0=1$.  We assume that $U$ satisfies the integral recurrence
whose characteristic polynomial is the minimal polynomial of a nonintegral
Pisot number $\beta$.  Thus, if
$\beta=\beta_0,\beta_1,\ldots,\beta_{s-1}$ are the roots of that polynomial,
then $\beta>1$ and $|\beta_i|<1$ for $i\ge1$.  The recurrence and the initial
values effectively give algebraic numbers $c_i$ such that
\begin{equation}
u_n=\sum_{i=0}^{s-1}c_i\beta_i^n,
\qquad c_0>0.
\label{eq:pisot-binet}
\end{equation}
Effective root isolation gives positive algebraic bounds $b_U,A_U$ with
\begin{equation}
b_U\beta^n\le u_n\le A_U\beta^n\qquad(n\ge0).
\label{eq:pisot-effective-growth}
\end{equation}
Indeed, one isolates an index after which the total conjugate contribution
in \eqref{eq:pisot-binet} is at most $c_0\beta^n/2$ and inspects the
remaining finite prefix.

Fix $D\ge1$.  For $m\ge2$, let $G_m(U,D)$ be the directed graph with vertex
set
\[
([-D,D]\cap\mathbb Z)^{m-1}.
\]
There is an edge
\[
(e_0,\ldots,e_{m-2})\longrightarrow
(e_1,\ldots,e_{m-2},e_{m-1})
\]
if and only if there is an integer $c$ such that
\begin{equation}
e_0u_0+e_1u_1+\cdots+e_{m-1}u_{m-1}+cu_m=0.
\label{eq:bounded-zero-overlap-edge}
\end{equation}
An edge is therefore a bounded $U$-representation of zero, organized so
that consecutive representations overlap in one position.  A path is a
one-position overlap chain.  We distinguish the initial vertices whose
first coordinate is nonzero because an arbitrary nonzero chain can be
shifted until its first exposed nonzero coefficient occupies that position.

Put
\begin{equation}
K_{U,D}=\left\lceil\frac{DA_U}{b_U(\beta-1)}\right\rceil.
\label{eq:bounded-zero-carry-bound}
\end{equation}
The coefficient $c$ completing an edge is unique and satisfies
$|c|\le K_{U,D}$: uniqueness follows from $u_m>0$, and
\[
|c|\le D\frac{\sum_{j<m}u_j}{u_m}
 <\frac{DA_U}{b_U(\beta-1)}.
\]
The proof compares adjacent edges through all embeddings of the Pisot field.
The following constants bound the conjugate contributions and separate the
finitely many possible dominant-embedding defects from zero.
Define the effective positive separation
\begin{equation}
\delta_{U,D}=\min\left\{|a+k-\beta l|:
 |a|\le D,\ |k|,|l|\le K_{U,D},\ a+k-\beta l\ne0\right\}.
\label{eq:bounded-zero-separation}
\end{equation}
The variables in this minimum are integers.  Since $\beta$ is irrational,
$\delta_{U,D}>0$, and exact algebraic-number comparison computes it.  Set
\begin{align}
M_{U,D}&=\frac1{c_0}\sum_{i=1}^{s-1}|c_i|
 \left(\frac{D}{1-|\beta_i|}+K_{U,D}\right),\nonumber\\
B_{U,D}&=D+(1+\beta)M_{U,D},
\label{eq:bounded-zero-conjugate-bound}
\end{align}
and
\begin{equation}
m_0(U,D)=\min\{m\ge2:\beta^m\delta_{U,D}>B_{U,D}\}.
\label{eq:bounded-zero-contraction-threshold}
\end{equation}
All these constants are effective from $U$ and $D$.

\begin{lemma}[Adjacent collapse for an arbitrary bounded coefficient alphabet]
\label{lem:bounded-zero-arbitrary-alphabet-collapse}
Let $U$ satisfy the integral recurrence whose characteristic polynomial is
the minimal polynomial of a nonintegral Pisot number $\beta$, and fix
$D\ge1$.  If $m\ge m_0(U,D)$, coefficients
$e_t,\ldots,e_{t+m}\in[-D,D]\cap\mathbb Z$, and integers $k_t,k_{t+1}$
satisfy
\[
\sum_{j=0}^{m-1}u_je_{t+j}=k_tu_m,
\qquad
\sum_{j=0}^{m-1}u_je_{t+1+j}=k_{t+1}u_m,
\]
then
\[
k_{t+1}=0,\qquad e_{t+m}=-k_t.
\]
\end{lemma}

\begin{proof}
The growth bounds give, for either row,
\[
|k_t|\le D\frac{\sum_{j<m}u_j}{u_m}
 <\frac{DA_U}{b_U(\beta-1)}\le K_{U,D},
\]
and similarly $|k_{t+1}|\le K_{U,D}$.  For every embedding set
\[
A_{t,i}=\sum_{j=0}^{m-1}e_{t+j}\beta_i^j-k_t\beta_i^m.
\]
The recurrence and its Binet expansion give $\sum_i c_iA_{t,i}=0$.
For $i\ge1$,
\[
|A_{t,i}|\le\frac{D}{1-|\beta_i|}+K_{U,D},
\]
so $|A_{t,0}|\le M_{U,D}$, and the same holds at $t+1$.
Direct subtraction of the adjacent dominant-embedding sums gives the exact
identity
\begin{equation}
\beta^m(e_{t+m}+k_t-\beta k_{t+1})
=\beta A_{t+1,0}-A_{t,0}+e_t.
\label{eq:bounded-zero-adjacent-identity}
\end{equation}
The right side has modulus at most $B_{U,D}$.  If the parenthesis on the
left were nonzero, \eqref{eq:bounded-zero-separation} and
\eqref{eq:bounded-zero-contraction-threshold} would make the left side
strictly larger than $B_{U,D}$, a contradiction.  Hence
$e_{t+m}+k_t=\beta k_{t+1}$.  Irrationality of $\beta$ and integrality of
the three coefficients force $k_{t+1}=0$ and $e_{t+m}=-k_t$.
\end{proof}

\begin{theorem}[Linear transient bound for bounded $U$-representations of zero]
\label{thm:bounded-zero-linear-transient}
Let $U=(u_n)_{n\ge0}$ be a strictly increasing sequence of positive integers
with $u_0=1$, satisfying the integral recurrence whose characteristic
polynomial is the minimal polynomial of a nonintegral Pisot number $\beta$.
Fix $D\ge1$, and let $G_m(U,D)$ be the graph defined above.  For every edge,
the completing coefficient is unique and necessarily satisfies
$|c|\le K_{U,D}$.
Then there is an effectively computable $C(U,D)<\infty$ such that, for every
$m\ge2$, if no directed cycle of $G_m(U,D)$ is reachable from a vertex whose
first coordinate is nonzero, every directed path beginning at such a vertex
has fewer than $C(U,D)m$ edges.

Equivalently, after shifting a chain so that its initial exposed coefficient
is nonzero, either it reaches a directed cycle, or it has fewer than
$C(U,D)m$ further overlaps.  The graphs are finite-state recognizers of these
overlap chains, and their edges are bounded $U$-representations of zero by
\eqref{eq:bounded-zero-overlap-edge}.
\end{theorem}

\begin{proof}
For an edge at time $t$, write $k_t=-c$ in
\eqref{eq:bounded-zero-overlap-edge}.  The growth estimate gives
$|k_t|\le K_{U,D}$ uniformly in $m$.  Suppose first that
$m\ge m_0(U,D)$.  Along any path of at least two edges, Lemma
\ref{lem:bounded-zero-arbitrary-alphabet-collapse}, applied to each adjacent
pair, gives
\[
k_1=k_2=\cdots=0,\qquad
e_m=-k_0,\qquad e_{m+1}=e_{m+2}=\cdots=0.
\]
A path of $m+1$ edges therefore reaches the all-zero vertex.  That vertex
has a loop, so such a path would make a directed cycle reachable from its
initial vertex.  Under the hypothesis of the theorem, every path has at
most $m$ edges for $m\ge m_0(U,D)$.

For each of the finitely many $2\le m<m_0(U,D)$, the graph is finite.  When
no cycle is reachable from the first-coordinate-nonzero vertices, exact
acyclic longest-path search computes the maximum $p_{U,D}(m)$; set it to
zero otherwise.  Thus, for example,
\[
C(U,D)=\max\left(2,
 \max_{2\le m<m_0(U,D)}
 \left\lceil\frac{1+p_{U,D}(m)}m\right\rceil\right)
\]
is effective and has the required property, with the inner maximum understood
as zero when $m_0(U,D)=2$.
\end{proof}

\begin{corollary}[Global chain dichotomy]
\label{cor:bounded-zero-global-dichotomy}
Under the hypotheses of Theorem~\ref{thm:bounded-zero-linear-transient}, fix
$m\ge2$ and consider all bounded zero words of length $m+1$ that overlap one
digit at a time.  Either some chain with a nonzero coefficient has a finite
prefix followed by a periodic bounded zero representation, or every nonzero
chain has fewer than $(C(U,D)+1)m$ rows.
\end{corollary}

\begin{proof}
A reachable cycle gives the periodic tail.  In the absence of such a cycle,
at most $m-1$ initial shifts pass before a nonzero coordinate occupies the
first position, after which Theorem~\ref{thm:bounded-zero-linear-transient}
bounds the remaining overlaps.  The alternatives are mutually exclusive
because a periodic tail can be repeated to make arbitrarily long chains.
\end{proof}

\begin{remark}[Arithmetic scope]
\label{rem:bounded-zero-standard-object-boundary}
Theorem~\ref{thm:bounded-zero-linear-transient} is independent of the cyclic
rank construction of Section~\ref{sec:cyclic-rank-application}.  Its
essential arithmetic features are the distinguished terminal weight $u_m$
and the use of consecutive recurrence weights; replacing $u_m$ by an
unrelated modulus removes the identity underlying Lemma
\ref{lem:bounded-zero-arbitrary-alphabet-collapse}.  These features are what
make the linear estimate available for the overlap graphs considered here.
\end{remark}
