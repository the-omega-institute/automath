\section{\texorpdfstring{\(L0\)}{L0}: expression universality}\label{sec:L0}

This section fixes the single primitive and the precise boundary of the imported
expression theorem.  Throughout, \(\operatorname{Log}\) is the principal complex
logarithm.

\begin{definition}[EML operator]\label{def:eml}
The Exp-Minus-Log operation is the partial binary operation
\[
  \eml\colon\CC\times\CC^*\to\CC,
  \qquad \eml(x,y)=e^x-\operatorname{Log}y .
\]
An EML term is interpreted only where every second argument is nonzero and the
principal-branch expressions used in the term are defined.
\end{definition}

\begin{proposition}[Imported EML representation theorem]
\label{thm:eml-universality}
Odrzywo{\l}ek's public arXiv v2 record
\cite{Odrzywolek2026SingleBinaryOperator} asserts that every primitive in the
scientific-calculator basis \(\mathcal B_{36}\) displayed there has a finite
representative over \((\eml,1)\) on its stated principal-branch domain.  This is
used here only as an \(L0\) representative-existence theorem: it supplies no
normal form, no equality algorithm for EML terms, no semantic compiler or
normalizer, and no Richardson witness identity except through the separate
interface of \Cref{thm:separation}.
\end{proposition}

\begin{proof}
The result is imported from the versioned public record
arXiv:2603.21852v2, DOI \texttt{10.48550/arXiv.2603.21852}.  The ancillary file
named in \Cref{ass:HW} is used only for the optional printed-string hypothesis;
the main theorem chain treats the full \(\mathcal B_{36}\) result as the single
existence assertion stated here.
\end{proof}

\begin{theorem}[Exact scope of the imported \texorpdfstring{\(\mathcal B_{36}\)}{B36} theorem]
\label{thm:B36-import-exact-scope}
\label{thm:no-self-contained-B36-representation-proof}
Within this article the assertion that the whole scientific-calculator basis
\(\mathcal B_{36}\) has finite representatives over \((\eml,1)\) is precisely the
external existence import \Cref{thm:eml-universality}.  The self-contained part
of the paper proves only the printed core witnesses of
\Cref{prop:internal-core-eml-witnesses}.  Hence every theorem in the main chain
remains valid if the phrase ``the versioned public record asserts finite representatives for the displayed basis''
is replaced by the single hypothesis ``finite representatives for the displayed
basis exist on their stated principal-branch domains.''  No later theorem may use
the full \(\mathcal B_{36}\) statement as though a self-contained proof of all
thirty-six representatives had been reproduced here, or as a hidden compiler,
normalizer, or verified Richardson package; the last exclusion is the
proof-graph cut of \Cref{thm:richardson-proof-graph-no-hidden-witness}.
\end{theorem}

\begin{proof}
\Cref{thm:eml-universality} states the full \(\mathcal B_{36}\) result by citing
the versioned public record.  The proof of that proposition explicitly says that
the result is imported, not rederived.  The only EML representatives whose
real-domain identities are calculated inside this manuscript are the finite
macros in \Cref{lem:eml-printed-core,lem:eml-real-witness-library}; these are
collected in \Cref{prop:internal-core-eml-witnesses}.  Inspecting the dependency
cut in \Cref{prop:eml-import-dependency-cut}, the positive protocol theorem, the
certificate theorem, and the Richardson obstruction cite only this internal core
or the separate verified-triple interface, and therefore do not require a
self-contained derivation of every element of \(\mathcal B_{36}\).  Thus the full
calculator-basis statement has exactly the external-import scope asserted above:
representative existence on the stated domains.  If one replaces the cited
theorem by any other theorem with that same existence content, no proof in
Sections~\ref{sec:L1}--\ref{sec:strictness} changes.  In particular,
\Cref{thm:richardson-proof-graph-no-hidden-witness,thm:no-core-only-richardson-translation}
shows that the imported basis cannot be used to bypass the verified-triple
hypotheses for \(\pi\), sine, and nonnegative square root.
\end{proof}

\begin{corollary}[External compiler isolation]
\label{cor:external-compiler-isolation}
The imported \(L0\) theorem cannot be used as a hidden proof of an \(L1\)
normalizer, an \(L2\) certificate theorem, a semantic compiler into decidable
protocol objects, or the Richardson normalizer obstruction.
\end{corollary}

\begin{proof}
\Cref{thm:eml-universality} asserts existence of representatives only.  The
\(L1\) normalizer is constructed in \Cref{thm:zeckendorf-protocol-univ}; the
\(L2\) certificates are finite-fibre data from \Cref{prop:zeckendorf-cert-univ};
and the Richardson obstruction is \Cref{thm:separation}.
\end{proof}

\begin{theorem}[Public-record audit]
\label{thm:eml-public-record-audit}
\label{thm:pinned-L0-external-hypothesis}
\label{prop:external-record-metadata-nonloadbearing}
The \(L0\) import is pinned to arXiv:2603.21852v2 and the bibliographic metadata
in \cite{Odrzywolek2026SingleBinaryOperator}; the Zenodo record
\cite{Odrzywolek2026ZenodoCode} is cited only as provenance.  No proof in this
paper depends on executing archived code or on an unversioned mutable table.
\end{theorem}

\begin{proof}
The theorem statement records the exact citation used in
\Cref{thm:eml-universality}.  All later proofs cite either that proposition as
an existence import or the independent theorem interfaces in
\Cref{sec:L1,sec:L2,sec:strictness}.
\end{proof}

\begin{proposition}[EML import dependency cut]
\label{prop:eml-import-dependency-cut}
\label{prop:expression-import-isolation}
\label{prop:L0-dependency-localization}
If the full external \(\mathcal B_{36}\) import is removed, the internal core
EML identities, including the square-root witness only on \((0,\infty)\), the
Zeckendorf \(L1\) protocol, the one-orbit obstruction, the finite-fibre \(L2\)
certificates, and the verified-triple Richardson obstruction remain unchanged.
\end{proposition}

\begin{proof}
The internal core EML identities are \Cref{prop:internal-core-eml-witnesses} and
\Cref{lem:eml-printed-core}.  The other four claims are proved in
\Cref{sec:L1,sec:L2,sec:strictness}; none has \Cref{thm:eml-universality} as a
hypothesis.
\end{proof}

\begin{theorem}[Replacement-invariant \texorpdfstring{\(L0\)}{L0} cut]
\label{thm:L0-import-replacement-invariance}
\label{thm:B36-import-proof-graph-cut}
Let \(\mathbf B\) be any external theorem whose only asserted content for this
article is finite representative existence for the displayed calculator basis on
the stated principal-branch domains.  If \Cref{thm:eml-universality} is replaced
everywhere in the proof of \Cref{thm:main-hierarchy} by \(\mathbf B\), then every
positive \(L1\) claim, every finite-fibre \(L2\) certificate claim, the one-orbit
obstruction, and the Richardson verified-triple obstruction remain unchanged.
Conversely, any use of the imported \(\mathcal B_{36}\) record to prove equality
of arbitrary EML terms, to choose normal forms, or to verify \(\pi\), sine, or
nonnegative square root for Richardson is an additional hypothesis not present in
this paper.
\end{theorem}

\begin{proof}
The forward assertion is a proof-graph check.  The \(L1\) protocol is constructed
from Zeckendorf normal forms and value transport in
\Cref{thm:zeckendorf-protocol-univ}; the one-orbit obstruction uses only the free
monogenic orbit classification in \Cref{thm:restricted-level-zero-protocol-dichotomy};
the fixed-resolution finite-fibre certificates are
\Cref{prop:zeckendorf-cert-univ}; and the finite-state transfer obstruction is
the cover construction of \Cref{thm:finite-state-strictness}.  None of those
statements has the full calculator-basis import as a hypothesis.  The
Richardson part enters the main hierarchy through \Cref{thm:separation}, whose
hypotheses are a finite triple \((\mathsf P,\mathsf S,\mathsf R)\) and the three
proved real-domain identities for \(\pi\), sine on \(\RR\), and nonnegative square
root on \(\RR_{\ge0}\), not the existence of all \(\mathcal B_{36}\)
representatives.  Thus replacing \Cref{thm:eml-universality} by any theorem with
the same representative-existence content leaves those proofs verbatim.

For the converse, equality of arbitrary EML terms, canonical choice of normal
forms, and semantic verification of the three Richardson witnesses are not part
of \Cref{def:expr-univ} or \Cref{thm:eml-universality}.  They are explicitly
excluded by \Cref{thm:B36-import-exact-scope,cor:external-compiler-isolation}.
Using the public \(\mathcal B_{36}\) record for any of these stronger tasks
therefore adds information beyond the imported \(L0\) theorem.  Such information
is not available in the main proof chain, so the stated cut is exact.
\end{proof}

\begin{proposition}[Internal core EML witnesses]
\label{prop:internal-core-eml-witnesses}
In the principal-branch EML semantics, pure EML terms over \((\eml,1)\) represent
the constant \(1\), the identity input, \(\exp\), positive logarithm, negation,
addition, subtraction, multiplication, division by a nonzero real, rational
constants, \(\log2\), and square root on the positive domain \((0,\infty)\).
No value at \(0\) is asserted by this internal core package.
\end{proposition}

\begin{proof}
The explicit macros and real-domain calculations are printed in
\Cref{lem:eml-printed-core,lem:eml-real-witness-library}.  Those proofs use only
the definition of \(\eml\), positivity of the relevant inputs, and elementary
real algebra.
\end{proof}

\begin{definition}[Expression universality]\label{def:expr-univ}
A system is \(L0\)-expression-universal for a signature if every primitive
operation in the signature has at least one term representative over the chosen
primitive operations on its stated domain.  No normal form, equality procedure,
or fibre certificate is part of the definition.
\end{definition}

\begin{remark}\label{rem:eml-variants}
All EML claims in this article use the principal-branch convention of
\Cref{def:eml}; no branch-free complex calculus is asserted.
\end{remark}

\begin{remark}\label{rem:no-canon}
Expression universality alone does not choose canonical representatives.  This
is the gap separated by the Richardson obstruction in \Cref{thm:separation}.
\end{remark}
