\section{Introduction}\label{sec:intro}

Universality statements for a single binary primitive split into different
assertions.  We use three levels.  Level \(L0\), \emph{expression universality},
asserts existence of terms.  Level \(L1\), \emph{protocol universality}, asserts
computable normal forms, compiled operations, and semantic readout identities.
Level \(L2\), \emph{certificate universality}, adds finite certificates over the
fibres of a normalizing map.

The motivating \(L0\) primitive is Odrzywo{\l}ek's operation
\[
  \eml(x,y)=e^x-\operatorname{Log}y
\]
with the constant \(1\), cited from the public arXiv v2 record
\cite{Odrzywolek2026SingleBinaryOperator}.  We use that record only as a pinned
representative-existence theorem for its displayed scientific-calculator basis,
not as a normal form or equality algorithm for EML terms.  The positive \(L1\)
protocol is internal: Zeckendorf normal forms, valuation transport through
\(\NN\), and Church iteration give compiled addition and multiplication on
\(\NN\).

The Richardson obstruction is stated as a relative theorem.  Given a finite pure
EML triple \((\mathsf P,\mathsf S,\mathsf R)\) whose principal-branch real-domain
identities
\[
  \mathsf P=\pi,
  \qquad \mathsf S(x)=\sin x\quad(x\in\RR),
  \qquad \mathsf R(t)=\sqrt t\quad(t\in\RR_{\ge0})
\]
have been proved, \Cref{thm:separation} reduces Richardson's identity-to-zero
problem to zero-identity in the translated pure EML fragment.  Thus the fragment
has no computable semantic normalizer into finite decidable objects, and no
equality-reflecting compiler into finite decidable protocol objects.  The article
prints and proves the core EML arithmetic library needed by the translation, but
it does not prove an unconditional concrete triple for \(\pi\), sine, and
nonnegative square root.

\begin{assumption}[Conditional printed-string Richardson package]
\label{ass:HW}
The notation \((H_{\mathcal W})\) denotes the external analytic assertion that
the source-locked printed EML strings
\[
  W_{\pi}=U_{47},\qquad W_{\sin}=U_{111},
  \qquad W_{\sqrt{}}=U_{126}
\]
from Odrzywo{\l}ek's arXiv:2603.21852v2 ancillary file
\texttt{SupplementaryInformation.pdf}, Part~I, Sec.~1.3, Table~S2 and Part~II,
Secs.~2.1--2.5, satisfy
\[
  W_{\pi}=\pi,
  \qquad W_{\sin}(x)=\sin x\quad(x\in\RR),
  \qquad W_{\sqrt{}}(t)=\sqrt t\quad(t\in\RR_{\ge0})
\]
in the principal-branch semantics of \Cref{def:eml}.  The public identifiers are
the versioned arXiv record \cite{Odrzywolek2026SingleBinaryOperator}, DOI
\texttt{10.48550/arXiv.2603.21852}, and the provenance snapshot
\cite{Odrzywolek2026ZenodoCode}, DOI \texttt{10.5281/zenodo.19762318}.  This
assumption is used only for the optional printed-string specialization of the
relative theorem; the main hierarchy uses the verified-triple branch.
\end{assumption}

\begin{theorem}[Obstruction spine]
\label{thm:obstruction-spine}
The strictness part of the paper has two primary obstructions and one subordinate
certificate addendum:
\begin{enumerate}
\item the relative Richardson obstruction of
  \Cref{thm:separation,thm:proc-ams-relative-richardson-theorem,thm:relative-richardson-main-spine-closure};
\item the one-free-monogenic-orbit multiplication obstruction of
  \Cref{thm:level-zero-obstruction};
\item the cover-relative finite-fibre/finite-state separation of
  \Cref{thm:finite-state-strictness}.
\end{enumerate}
The named strings \((U_{47},U_{111},U_{126})\) are not part of the unconditional
spine; they specialize the first item only under \((H_{\mathcal W})\) or an
independent proof of the three displayed identities.
\end{theorem}

\begin{proof}
The first item is exactly \Cref{thm:separation}, whose proof factors through the
verified replacement-triple theorem
\Cref{thm:replacement-triple-richardson-interface} and its Proc. AMS formulation
\Cref{thm:proc-ams-relative-richardson-theorem}.  Those theorems assume a finite
triple with proved real-domain identities and do not use the printed strings.
The second item is \Cref{thm:level-zero-obstruction}, refined by the exponent
skeleton and faithful-submonoid classification.  The third item is
\Cref{thm:finite-state-strictness}; it is formulated for computable finite-fibre
covers over the fixed Zeckendorf normal-form sets.  The final sentence follows
from \Cref{prop:printed-witness-certificate-boundary,thm:concrete-richardson-instance-exact-data}.
\end{proof}

\begin{theorem}[Single-primitive hierarchy]\label{thm:main-hierarchy}
With the definitions of \(L0\), \(L1\), and \(L2\) in
\Cref{sec:L0,sec:L1,sec:L2}, the following hierarchy holds.
\begin{enumerate}
\item The pair \((\eml,1)\) has the cited \(L0\) representative-existence property
  for the pinned public \(\mathcal B_{36}\) calculator basis on its stated
  principal-branch domains.  This is only an external expression-existence
  anchor, not an equality algorithm, normal form, compiler, or Richardson witness
  package.
\item Zeckendorf normal forms and valuation transport give an internal \(L1\)
  protocol for the operations \(\{+,\times\}\) on \(\NN\).
\item The Richardson clause is relative in the following explicit sense.  For
  every finite pure EML triple \((\mathsf P,\mathsf S,\mathsf R)\) whose identities
  \(\mathsf P=\pi\), \(\mathsf S(x)=\sin x\) on \(\RR\), and
  \(\mathsf R(t)=\sqrt t\) on \(\RR_{\ge0}\) have been proved, Richardson's
  identity-to-zero problem many-one reduces to zero-identity in the associated
  pure EML fragment.  Hence that fragment has neither a computable semantic
  normalizer into finite decidable objects nor an equality-reflecting semantic
  compiler into finite decidable protocol objects.  This article proves the
  core arithmetic EML library, but it does not prove a finite triple for
  \(\pi\), sine on \(\RR\), and nonnegative square root on \(\RR_{\ge0}\).  The
  printed strings \((U_{47},U_{111},U_{126})\) enter only under the optional
  hypothesis \((H_{\mathcal W})\); they are not an unconditional concrete
  Richardson instance.
\item Multiplication on \(\NN_{\ge1}\) cannot be injectively read from the ambient
  product on one free monogenic orbit.  In that model every readout factors
  through the prime-weight exponent skeleton, and the faithful restrictions are
  exactly the trivial monoid and a single-prime-power ray.  A scoped finite-rank
  refinement records the corresponding rank budget, but the load-bearing
  obstruction is the one-free-monogenic-orbit theorem.
\item The finite-fibre \(L2\) layer gives fixed-resolution Hankel--Prony
  certificates, but the strict finite-state transfer separation proved here is
  cover-relative: in the computable finite-fibre cover category over the fixed
  Zeckendorf normal-form sets, such certificates do not force a uniform
  finite-state transfer formula.  No intrinsic asymptotic failure theorem for the
  original \(\Fold_m\) fibre multiplicities is asserted.
\item After deleting the unprinted \(\mathcal B_{36}\) representative table and
  the optional printed-string hypothesis \((H_{\mathcal W})\), the internal core
  witness package, the verified-triple Richardson implication, the Zeckendorf
  protocol, the one-orbit obstruction, and the cover-relative \(L2\) obstruction
  remain proved.  Only the global calculator-basis headline and the concrete
  named-string Richardson specialization are removed.
\end{enumerate}
\end{theorem}

\begin{proof}
The \(L0\) assertion is exactly the pinned external anchor
\Cref{thm:eml-universality}, with its domain and dependency boundary fixed by
\Cref{thm:L0-citation-domain-support,thm:L0-external-anchor-criterion,thm:B36-import-exact-scope}.
The \(L1\) assertion is the Zeckendorf protocol theorem
\Cref{thm:zeckendorf-protocol-univ}.  The Richardson assertion is
\Cref{thm:separation}, proved from the verified replacement-triple interface
\Cref{thm:replacement-triple-richardson-interface,thm:relative-richardson-main-spine-closure}; the exact relative-versus-concrete alternative and the data needed for a
concrete printed-string instance are isolated in
\Cref{thm:submission-richardson-dichotomy,thm:concrete-richardson-instance-exact-data}.  The one-orbit and
finite-rank product obstructions are
\Cref{thm:restricted-level-zero-protocol-dichotomy,thm:one-orbit-faithful-submonoid-classification,thm:finite-rank-product-readout-lower-bound}.
Finally, the fixed-resolution certificate facts and the cover-relative
finite-state separation are
\Cref{thm:L2-finite-window-transfer-obstruction,thm:L2-subordinate-cover-classification,thm:L2-cover-relative-closure,thm:finite-state-strictness},
while \Cref{thm:L2-cover-intrinsic-nonimplication} excludes interpreting the
cover example as an intrinsic asymptotic theorem for the original fibres.  The
last item is the dependency-cut theorem
\Cref{thm:load-bearing-core-dependency-cut}; it proves that replacing the full
calculator-basis import by the self-contained core, and omitting the optional
printed-string hypothesis, preserves the stated internal theorem chain and
removes only the two advertised optional headlines.
\end{proof}

\begin{remark}[Scoped strictness gate]
\label{thm:scoped-separation-quantifiers}
The two primary strictness assertions have different quantifiers.  The
Richardson obstruction is universal over already verified finite EML triples;
the printed strings give a branch only under \((H_{\mathcal W})\).  The
multiplication obstruction is unconditional but is restricted to readout from one
free monogenic orbit.  The \(L2\) separation is a subordinate cover-relative
addendum.
\end{remark}

\begin{corollary}[Main \texorpdfstring{\(L2\)}{L2} addendum]
\label{cor:main-L2-addendum}
Cover finite-fibre certificate data are strictly stronger than fixed finite-state
transfer in the computable cover category used in this article.
\end{corollary}

\begin{proof}
This is \Cref{thm:finite-state-strictness}, read with the cover category and
non-intrinsic boundary of
\Cref{thm:L2-subordinate-cover-classification,thm:L2-cover-intrinsic-nonimplication}.
\end{proof}

\begin{theorem}[Main-spine scope projection]
\label{thm:main-spine-scope-closure}
\label{thm:main-theorem-scope-projection}
The theorem chain of \Cref{thm:main-hierarchy} is the image of the following
five load-bearing interfaces, and no larger interface is used:
\begin{enumerate}
\item the cited \(L0\) representative-existence anchor for the public
  \(\mathcal B_{36}\) basis;
\item the value-transported Zeckendorf \(L1\) protocol;
\item the relative Richardson obstruction for supplied finite EML triples whose
  three real-domain identities have already been proved;
\item the one-free-monogenic-orbit multiplication obstruction; and
\item the cover-relative finite-fibre/finite-state separation over the fixed
  normal-form sets \(X_m\).
\end{enumerate}
Consequently any attempted strengthening of \Cref{thm:main-hierarchy} by adding
one of the following conclusions is not a consequence of the paper's proof
chain: a term-by-term self-contained \(\mathcal B_{36}\) catalogue, an
unconditional concrete Richardson triple for \((U_{47},U_{111},U_{126})\), an
intrinsic finite-state theorem for the original \(\Fold_m\) fibres, a
bounded-window or digit-local Zeckendorf arithmetic theorem, or a global rigidity
theorem for all single-primitive protocols realizing \(\{+,\times\}\).
\end{theorem}

\begin{proof}
Each positive or negative assertion in \Cref{thm:main-hierarchy} is proved by one
of the labels cited in its proof.  Item~(1) is exactly the external anchor
\Cref{thm:eml-universality}, and its proof boundary is fixed by
\Cref{thm:B36-import-exact-scope,thm:L0-citation-domain-support}; those results
state representative existence only and do not print or verify the whole table.
Item~(2) is \Cref{thm:zeckendorf-protocol-univ}, together with
\Cref{prop:value-transport-exact-interface,cor:value-transported-minimal-witness},
which identify the operations as decode--compute--encode transports through
\(\Val\) rather than local digit algorithms.  Item~(3) is precisely
\Cref{thm:separation,thm:proc-ams-relative-richardson-theorem,thm:richardson-triple-submission-alternative}:
the proof assumes a finite verified triple and, by
\Cref{thm:concrete-richardson-instance-exact-data,thm:no-unconditional-printed-richardson-edge},
does not verify the named strings unconditionally.  Item~(4) is
\Cref{thm:level-zero-obstruction}, with the faithful-submonoid classification and
the explicitly scoped finite-rank refinement cited in \Cref{thm:main-hierarchy};
the negative inference needed for the hierarchy uses the one-free-monogenic-orbit
hypotheses.  Item~(5) is the cover-category
result \Cref{thm:finite-state-strictness}, whose nonintrinsic boundary is recorded
in \Cref{thm:L2-cover-relative-closure,thm:L2-cover-intrinsic-nonimplication}.

For the exclusions, suppose one of the five stronger conclusions in the final
sentence were a consequence of the displayed proof chain.  Its proof would have
to factor through one of the five interfaces above, because those are the only
interfaces just enumerated.  The \(L0\) interface is citation-only beyond the
internal core, so it cannot supply a printed table.  The Richardson interface is
universal over already verified triples, so it cannot supply the missing analytic
verification of \((U_{47},U_{111},U_{126})\).  The Zeckendorf interface is
explicitly value transported, so it cannot supply a bounded-window or digit-local
algorithm.  The finite-fibre interface changes only auxiliary cover
multiplicities over fixed normal forms, so it cannot imply an intrinsic theorem
for the original \(\Fold_m\) fibres.  Finally the multiplication obstruction is
proved only on one free monogenic orbit, so it cannot classify all possible
single-primitive protocols.  This proves both the stated projection and the
listed nonconsequences.
\end{proof}

\begin{remark}[Compact hierarchy form]\label{cor:compact-hierarchy-form}
The submission-facing chain is
\[
  \Cref{thm:eml-universality}\Rightarrow L0,
  \qquad
  \Cref{thm:zeckendorf-protocol-univ}\Rightarrow L1,
\]
with strictness from the relative Richardson theorem
\Cref{thm:separation}, the one-orbit obstruction
\Cref{thm:level-zero-obstruction}, and the optional \(L2\) addendum
\Cref{cor:main-L2-addendum}.
\end{remark}

\begin{remark}[Richardson submission boundary]\label{cor:richardson-submission-boundary}
The paper proves a Richardson obstruction interface, not an unconditional
principal-branch verification of \((U_{47},U_{111},U_{126})\).  Once any finite
pure EML triple is specified and the three displayed real-domain identities are
proved, \Cref{thm:separation} gives the transported EML normalizer obstruction.
\end{remark}

\begin{theorem}[Richardson obstruction interface]\label{thm:richardson-obstruction-interface}
The Richardson input to the main theorem is exactly the relative verified
interface of \Cref{thm:proc-ams-relative-richardson-theorem}: a finite pure-EML
triple with proved identities for \(\pi\), sine on \(\RR\), and the nonnegative
square root on \(\RR_{\ge0}\).  Under this interface, identity to zero for
Richardson expressions reduces effectively to equality with the translated zero
EML term, so a computable semantic normalizer would decide Richardson's
undecidable problem.  Without those three identities for a named finite triple,
including \((U_{47},U_{111},U_{126})\), the paper asserts no concrete
printed-string Richardson instance.
\end{theorem}

\begin{proof}
This is \Cref{thm:proc-ams-relative-richardson-theorem} in interface form.  The
proof there constructs the translation by structural recursion and applies
Richardson's undecidability theorem recorded in \Cref{lem:richardson-fragment}.
\end{proof}

\begin{corollary}[Positive realizations]\label{cor:main-positive-realizations}
The positive parts of the hierarchy are constructive: \((\eml,1)\) gives the
imported expression representatives, Zeckendorf valuation transport gives the
compiled \(L1\) arithmetic operations, and finite fibre-size multisets give the
\(L2\) certificates.
\end{corollary}

\begin{proof}
The three constructions are respectively
\Cref{thm:eml-universality,thm:zeckendorf-protocol-univ,prop:zeckendorf-cert-univ}.
\end{proof}
