\section{Introduction}\label{sec:intro}

Universality statements can mean different things.  A single operation may
represent a large class of expressions, while still giving no effective way to
choose or compare representatives.  We separate three levels for this
phenomenon.  Level \(L0\), \emph{expression universality}, asks only that terms
exist.  Level \(L1\), \emph{protocol universality}, asks for computable normal
forms, compiled operations, and semantic readout identities.  Level \(L2\),
\emph{certificate universality}, adds finite certificates attached to the
fibres of a normalizing map.

The motivating \(L0\) example is Odrzywo{\l}ek's operation
\[
  \eml(x,y)=e^x-\operatorname{Log}y,
\]
with the constant \(1\), which represents the operations in a standard
scientific-calculator basis \cite{Odrzywolek2026SingleBinaryOperator}.  We use
that public record only as a representative-existence theorem.  The positive
\(L1\) protocol in this paper is internal: Zeckendorf normal forms, valuation
transport through \(\NN\), and Church iteration give compiled addition and
multiplication on \(\NN\).  The main obstructions show that these two assertions
should not be conflated.

The Richardson obstruction is used in a verified-triple form.  If a finite pure
EML triple \((\mathsf P,\mathsf S,\mathsf R)\) is supplied with proved real-domain
identities
\[
  \mathsf P=\pi,
  \qquad \mathsf S(x)=\sin x\quad(x\in\RR),
  \qquad \mathsf R(t)=\sqrt t\quad(t\in\RR_{\ge0}),
\]
then \Cref{thm:separation} gives an unconditional Richardson normalizer
obstruction for that verified triple.  The named printed roots
\(U_{47},U_{111},U_{126}\) are retained only as a conditional specialization of
this theorem, not as a load-bearing input to the main hierarchy.

the source-locked printed EML strings

from Odrzywo{\l}ek's arXiv:2603.21852v2 ancillary file

Secs.~2.1--2.5, satisfy
\[
  W_{\pi}=\pi,
  \qquad W_{\sin}(x)=\sin x\quad(x\in\RR),
  \qquad W_{\sqrt{}}(t)=\sqrt t\quad(t\in\RR_{\ge0})
\]
in the principal-branch semantics of \Cref{def:eml}.  The public identifiers are
the versioned arXiv record \cite{Odrzywolek2026SingleBinaryOperator}, DOI
\texttt{10.48550/arXiv.2603.21852}, and the provenance snapshot
\cite{Odrzywolek2026ZenodoCode}, DOI \texttt{10.5281/zenodo.19762318}.  This

\Cref{thm:separation,thm:printed-richardson-witnesses-interface}; the main

\end{assumption}

\begin{proposition}[Public-record boundary]
\label{prop:companion-reference-audit}
The only public-record imports in the theorem-note spine are the \(L0\)
representative-existence statement for \((\eml,1)\), Richardson's undecidability
theorem, the standard Zeckendorf normal-form theorem, and the classical
finite-atomic moment facts cited in \Cref{sec:L2}.  No companion manuscript,
unversioned source table, or hidden compiler output is used as a proof input.
\end{proposition}

\begin{proof}
The imported \(L0\) statement is isolated in \Cref{thm:eml-universality}.  The
Richardson step in \Cref{thm:separation} cites Richardson's theorem through
\Cref{lem:richardson-fragment} and then uses only the verified EML witness
interface stated there.  The \(L1\) protocol uses the greedy Zeckendorf
normalizer and valuation transport proved in \Cref{sec:L1}.  The \(L2\) facts
are finite sums over finite fibres, with classical references supplied in
\Cref{sec:L2}.  These are exactly the four listed imports.
\end{proof}

\end{dependencyremark}

\begin{theorem}[Obstruction spine]
\label{thm:obstruction-spine}
The hierarchy has two primary obstructions and one subordinate certificate
addendum.
\begin{enumerate}
\item A verified finite EML triple for \(\pi\), sine, and nonnegative square root
  transports Richardson's identity problem into pure EML terms, so no computable
  semantic normalizer with decidable equality exists on that translated
  fragment.
\item Multiplication on \(\NN_{\ge1}\) cannot be read injectively from the
  ambient product on a single free monogenic orbit; every such readout factors
  through a prime-weight exponent skeleton and has a prime-power collision.
\item Finite-fibre certificates give Hankel positivity and finite Prony
  reconstruction at each fixed resolution, but they do not force a uniform
  finite-state transfer formula in the resolution variable.
\end{enumerate}
\end{theorem}

\begin{proof}
The first assertion is \Cref{thm:separation}.  The second is
\Cref{thm:level-zero-obstruction}, with the skeleton classification in
\Cref{lem:level-zero-exponent-skeleton,thm:one-orbit-readout-classification}.
The third is \Cref{thm:L2-finite-fiber-no-transfer-content,thm:finite-state-strictness}.
The three proofs use disjoint data: the Richardson witness interface, the free
monogenic orbit, and computable finite covers, respectively.
\end{proof}

\begin{theorem}[Proc. AMS hierarchy spine]\label{thm:main-hierarchy}
With the definitions in \Cref{sec:L0,sec:L1,sec:L2}, the single-primitive
hierarchy has the following theorem-note form.
\begin{enumerate}
\item The operation \((\eml,1)\) gives an imported \(L0\) expression system for
  the public calculator basis, but that import gives no semantic normalizer.
\item Zeckendorf normal forms and valuation transport give an internal \(L1\)
  protocol for \(\{+,\times\}\) on \(\NN\).
\item The \(L0\) representative statement does not promote to such a semantic
  normalizer or semantic compiler on the Richardson fragment, by
  \Cref{thm:separation,cor:compile-scope}.
\item The multiplication readout in the \(L1\) protocol cannot be collapsed to
  the ambient product on one free monogenic orbit, by
  \Cref{thm:level-zero-obstruction}.
\item The finite-fibre \(L2\) layer supplies fixed-resolution certificates, but
  finite-fibre certificates and finite-state transfer are separated by
  \Cref{thm:finite-state-strictness}.
\end{enumerate}
\end{theorem}

\begin{proof}
The \(L0\) clause is \Cref{thm:eml-universality}; the dependency cut is
\Cref{prop:eml-import-dependency-cut}.  The positive \(L1\) clause is
\Cref{thm:zeckendorf-protocol-univ}; its source-compilation boundary is
\Cref{cor:compile-scope}.  The Richardson and one-orbit obstructions
are the two clauses of \Cref{thm:obstruction-spine}.  The \(L2\) assertion is
the finite-fibre certificate taxonomy of \Cref{prop:zeckendorf-cert-univ},
together with the transfer separation in \Cref{thm:finite-state-strictness}.
\end{proof}

\begin{corollary}[Subordinate finite-fibre addendum]\label{cor:main-L2-addendum}
The \(L2\) part of \Cref{thm:main-hierarchy} is independent of the two primary
obstructions: Hankel positivity and Prony reconstruction are fixed-resolution
finite-atomic facts, while the strict \(L2\) content is the separate
finite-state growth obstruction.
\end{corollary}

\begin{proof}
Apply \Cref{prop:classical-finite-atomic-skeleton} to the fibre-size multiset at
each resolution.  This uses neither Richardson expressions nor the one-orbit
readout.  The absence of any forced finite-state transfer is exactly
\Cref{thm:L2-finite-fiber-no-transfer-content}.
\end{proof}

\begin{theorem}[Three-anchor formulation]
\label{thm:three-anchor-excision}
\label{prop:main-dependency-splitting}
The submission depends only on three anchors: the public \(L0\) expression
import, the internal Zeckendorf \(L1\) protocol, and the finite-fibre \(L2\)
certificate construction.  All Richardson conclusions pass through the explicit
verified-triple interface of \Cref{thm:separation}; the printed-root package is
only the conditional specialization \((H_{\mathcal W})\).
\end{theorem}

\begin{proof}
The anchors are the three sections \Cref{sec:L0,sec:L1,sec:L2}.  The only
cross-anchor obstruction using EML analytic identities is \Cref{thm:separation},
whose statement separates the unconditional verified-triple branch from the
conditional named-root branch.  Hence no further source table or companion
result is a hidden premise.
\end{proof}

\begin{corollary}[Compact hierarchy form]\label{cor:compact-hierarchy-form}
The main theorem chain is the chain
\[
  \Cref{thm:eml-universality}
  \Longrightarrow L0,
  \qquad
  \Cref{thm:zeckendorf-protocol-univ}
  \Longrightarrow L1,
\]
with strictness supplied by \Cref{thm:separation,thm:level-zero-obstruction}
and the \(L2\) addendum supplied by \Cref{cor:main-L2-addendum}.
\end{corollary}

\begin{proof}
This is a restatement of \Cref{thm:main-hierarchy} with the cited theorem labels
substituted for the five clauses.
\end{proof}

\begin{corollary}[Richardson submission boundary]\label{cor:richardson-submission-boundary}
The paper proves an unconditional Richardson obstruction for every verified
finite EML triple satisfying the three displayed real-domain identities.  It
does not assert an unconditional named-root theorem for
\(U_{47},U_{111},U_{126}\) unless \((H_{\mathcal W})\) is assumed.
\end{corollary}

\begin{proof}
The first sentence is the first branch of \Cref{thm:separation}.  The second is
the final sentence of the same theorem and the definition of \((H_{\mathcal W})\)
in \Cref{ass:HW}.
\end{proof}

\begin{theorem}[Richardson obstruction interface]\label{thm:richardson-obstruction-interface}
The Richardson input to the main theorem is exactly the following interface: a
finite pure-EML triple with proved identities for \(\pi\), sine on \(\RR\), and
the nonnegative square root on \(\RR_{\ge0}\).  Under this interface, identity
to zero for Richardson expressions reduces effectively to equality with the
translated zero EML term, so a computable semantic normalizer would decide
Richardson's undecidable problem.
\end{theorem}

\begin{proof}
This is \Cref{thm:replacement-triple-richardson-unconditional} in interface
form.  The proof there constructs the translation by structural recursion and
then applies Richardson's undecidability theorem recorded in
\Cref{lem:richardson-fragment}.
\end{proof}

\begin{corollary}[Positive realizations]\label{cor:main-positive-realizations}
The positive parts of the hierarchy are constructive: \((\eml,1)\) gives the
imported expression representatives, Zeckendorf valuation transport gives the
compiled \(L1\) arithmetic operations, and finite fibre-size multisets give the
\(L2\) certificates.
\end{corollary}

\begin{proof}
The three constructions are respectively
\Cref{thm:eml-universality,thm:zeckendorf-protocol-univ,prop:zeckendorf-cert-univ}.
\end{proof}
