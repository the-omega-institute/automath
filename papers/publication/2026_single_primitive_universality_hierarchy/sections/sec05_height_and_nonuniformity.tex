\section{Fibre height and nonuniformity in the degree}
\label{sec:height}

For \(n\ge0\) and \(N\in\mathbb Z\), let
\[
 R_n(N)=\#\{A\subseteq\{1,\ldots,n\}:\sum_{i\in A}F_i=N\},
\]
with \(R_n(N)=0\) outside the attainable range.  Then
\begin{equation}\label{eq:R-recursion}
 R_n(N)=R_{n-1}(N)+R_{n-1}(N-F_n).
\end{equation}

\begin{lemma}[Fold fibre--representation pairing]
\label{lem:fold-fibre-representation-pairing}
If \(x\in X_m\) and \(r=\Val_m(x)\), then
\[
 d_m(x)=R_{m+1}(r)+R_{m+1}(r+F_{m+2})
       =R_{m+2}(r+F_{m+2}).
\]
\end{lemma}
\begin{proof}
The possible word values lie in
\([0,F_{m+3}-2]\), which is contained in \([0,2F_{m+2})\).
Together with \eqref{eq:fold-residue}, a word over \(x\) therefore has value
\(r\) or \(r+F_{m+2}\), giving the first equality.  Splitting
representations of \(r+F_{m+2}\) according as they use \(F_{m+2}\) gives
the second equality by \eqref{eq:R-recursion}.
\end{proof}

\begin{lemma}[Ordinary interval maximum input]
\label{lem:ordinary-interval-maximum-input}
Let
\(C_k=\max\{R(N):F_k\le N<F_{k+1}\}\).  We use the cited theorem
\cite[Thm.~4.7]{KocabovaMasakovaPelantova2005}, after converting its convention
\(\mathcal F_0=\mathcal F_1=1\) to \(F_1=1,F_2=2\).  Its positive-index
statement is
\[
 C_{2t+1}=F_{t+1}\quad(t\ge0),\qquad
 C_{2t+2}=2F_t\quad(t\ge1),\qquad C_2=1.
\]
No assertion about fold fibres is imported from that source.
\end{lemma}

\begin{lemma}[Shifted half-interval maximum]
\label{thm:half-interval-fibonacci-representation-max}
For
\[
 H_n=\max_{0\le r<F_{n+1}}R_{n+1}(F_{n+1}+r),
\]
one has
\[
 H_{2s}=F_{s+1},\qquad H_{2s+1}=2F_s\qquad(s\ge1),
\]
and \(H_1=1\).
\end{lemma}
\begin{proof}
On \([F_{n+1},F_{n+2})\), the truncated count \(R_{n+1}\) is the ordinary
count, so its maximum is \(C_{n+1}\).  For the remaining tail write
\(N=F_{n+2}+u\), \(0\le u<F_{n-1}\).  Equation \eqref{eq:R-recursion} gives
\[
 R_{n+1}(N)=R_n(F_{n+2}+u)+R_n(F_n+u)=R_n(F_n+u),
\]
because the largest sum from \(F_1,\ldots,F_n\) is \(F_{n+2}-2\).
The last term is at most \(C_n\le C_{n+1}\), using the displayed formulas for
\(C_k\).  Hence \(H_n=C_{n+1}\), and substitution gives the parity formulas.
The case \(n=1\) is direct.
\end{proof}

\begin{theorem}[Exact intrinsic fold height]
\label{thm:intrinsic-fold-fibre-height}
Let \(M_m=\max_{x\in X_m}d_m(x)\).  Then \(M_m=H_{m+1}\), and hence
\[
 M_{2s-1}=F_{s+1},\qquad M_{2s}=2F_s\qquad(s\ge1).
\]
Equivalently, \(M_1,\ldots,M_{10}=2,2,3,4,5,6,8,10,13,16\).
\end{theorem}
\begin{proof}
By \Cref{lem:fold-fibre-representation-pairing}, the fibre sizes are exactly
\(R_{m+2}(F_{m+2}+r)\) for \(0\le r<F_{m+2}\).  Since \(\Val_m\) is a
bijection onto that interval, maximizing gives \(M_m=H_{m+1}\).  Apply the
preceding lemma.
\end{proof}

\begin{corollary}[Height entropy]
\label{cor:linfty-fold-entropy}
With \(\varphi=(1+\sqrt5)/2\),
\[
 M_m=\Theta(\varphi^{m/2}),\qquad
 \lim_{m\to\infty}\frac1m\log M_m=\frac12\log\varphi .
\]
\end{corollary}
\begin{proof}
Use the parity formulas in
\Cref{thm:intrinsic-fold-fibre-height} and \(F_s=\Theta(\varphi^s)\).
\end{proof}

\begin{corollary}[No rational transfer law uniform in the degree]
\label{cor:intrinsic-all-degree-transfer-obstruction}
Define
\[
 \mathcal M(z,w)=\sum_{m\ge1}\sum_{q\ge0}S_q(m)z^mw^q .
\]
Then \(\mathcal M(z,w)\) is not rational.  In particular there are no
\(r\ge1\), \(A(w)\in M_r(\mathbb C(w))\), and
\(u(w),v(w)\in\mathbb C(w)^r\) for which
\[
 \sum_{q\ge0}S_q(m)w^q=u(w)^{\mathsf T}A(w)^mv(w)
\]
holds for all sufficiently large \(m\) as a formal series at \(w=0\).
\end{corollary}
\begin{proof}
Let \(\Delta_m=S_m(m)\).  A maximal fibre gives
\(\Delta_m\ge M_m^m\).  By
\Cref{cor:linfty-fold-entropy},
\[
 \log\Delta_m\ge \tfrac12m^2\log\varphi-O(m).
\]
If \(\mathcal M=P/Q\) were rational with \(Q(0,0)\ne0\), it would be analytic
on a small closed polydisc.  Cauchy's estimate would give
\(S_q(m)\le C R_z^{-m}R_w^{-q}\).  On \(q=m\) this is only exponential in
\(m\), contradicting the quadratic lower bound.  The displayed matrix law
would make the tail
\(z^{m_0}u(w)^{\mathsf T}A(w)^{m_0}(I-zA(w))^{-1}v(w)\) rational; adding finitely
many initial terms would contradict the first part.
\end{proof}

\begin{remark}
The preceding corollary is a quantifier statement, not a second transfer
construction.  Fixed \(q\) permits a matrix whose size depends on \(q\);
the height theorem rules out one rational law that is uniform in \(q\).

\end{remark}
