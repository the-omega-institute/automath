\section{\texorpdfstring{\(L2\)}{L2}: finite certificates}\label{sec:L2}

An \(L1\) protocol supplies normal forms.  Level \(L2\) records computable
finite invariants of the normalizing fibres.  For Zeckendorf words these fibres
are the fibres of \(\Fold_m\colon\Omega_m\to X_m\).  The point of this section is
short: fixed-resolution finite fibres give the usual finite-atomic moment
certificates, but they do not imply a uniform finite-state transfer formula in
the resolution variable.

For \(x\in X_m\), put
\[
  d_m(x)=|\Fold_m^{-1}(x)|,
  \qquad
  S_q(m)=\sum_{x\in X_m} d_m(x)^q\quad(q\ge0).
\]
Thus \(S_0(m)=|X_m|=F_{m+2}\) and \(S_1(m)=2^m\).

\begin{definition}[Collision moment]\label{def:collision-moment}
For any finite normal-form set \(N_m\) with finite fibres, the \(q\)-th
collision moment is
\begin{equation}\label{eq:Sq}
  S_q^{(m)}=\sum_{n\in N_m}|\mathrm{NF}_m^{-1}(n)|^q .
\end{equation}
\end{definition}

\begin{proposition}[Classical finite-atomic certificate skeleton]
\label{prop:classical-finite-atomic-skeleton}
Let \(D\) be a finite multiset of positive integers and set
\(M_q(D)=\sum_{d\in D}d^q\).  Then the Hankel matrices
\([M_{i+j}(D)]_{0\le i,j\le r}\) are positive semidefinite for every \(r\), and
the moment segment \(M_0(D),\ldots,M_{2s-1}(D)\), where \(s\) is the number of
distinct values in \(D\), determines \(D\) uniquely.  Consequently fixed-fibre
certificate data determine exactly a finite atomic measure at each resolution
and impose no recurrence or finite-state transfer law as the resolution varies.
\end{proposition}

\begin{proof}
Write the distinct values as \(a_1<\cdots<a_s\), with multiplicities
\(n_1,\ldots,n_s\).  Then \(M_q(D)=\sum_i n_i a_i^q\).  For
\(Q(t)=\sum_{j=0}^r c_jt^j\),
\[
  \sum_{i,j=0}^r c_i c_j M_{i+j}(D)=\sum_{a\in D} Q(a)^2\ge0,
\]
which proves Hankel positivity; this is the finite positive-measure instance of
the moment problem \cite{Hamburger1920,ShohatTamarkin1943,Akhiezer1965}.  For
reconstruction, the sequence \((M_q)\) satisfies the order-\(s\) recurrence whose
characteristic polynomial is \(\prod_i(T-a_i)\).  The first \(2s\) moments give
the minimal recurrence by the usual Hankel/Prony linear system; its roots are
the \(a_i\), and the Vandermonde system
\(M_q=
\sum_i n_i a_i^q\) for \(0\le q<s\) gives the multiplicities
\cite{Prony1795,BatenkovYomdin2013}.  All arguments are internal to the fixed
finite multiset \(D\), so they do not relate different resolutions.
\end{proof}

\begin{theorem}[Hankel positivity]\label{thm:hankel-psd}
For every \(m\), the Zeckendorf collision moments satisfy
\([S_{i+j}(m)]_{0\le i,j\le r}\succeq0\) for all \(r\).  The same statement holds
for every computable finite-fibre cover after replacing \(d_m(x)\) by the
covered fibre size.
\end{theorem}

\begin{proof}
Apply \Cref{prop:classical-finite-atomic-skeleton} to the multiset
\(D_m=\{d_m(x):x\in X_m\}\), or to the corresponding covered multiset.
\end{proof}

\begin{theorem}[Finite Prony reconstruction]\label{thm:prony}
At a fixed resolution \(m\), the finite list of collision moments through
degree \(2s_m-1\), where \(s_m\) is the number of distinct fibre sizes, recovers
the fibre-size multiset.  Hence any two finite-fibre systems with the same such
moment segment have the same fixed-resolution certificate data.
\end{theorem}

\begin{proof}
This is the reconstruction part of
\Cref{prop:classical-finite-atomic-skeleton} applied to the fixed multiset of
fibre sizes.
\end{proof}

\begin{theorem}[Star-kernel certificate]\label{thm:star-kernel}
For fixed \(m\), the kernel
\[
  K_m(i,j)=S_{i+j}(m)
\]
is a positive semidefinite Gram kernel.  It is finite-rank, and its rank is the
number of distinct fibre sizes at resolution \(m\).
\end{theorem}

\begin{proof}
With \(D_m=\{a_1^{(n_1)},\ldots,a_s^{(n_s)}\}\), the vector
\(v_i=(a_1^i\sqrt{n_1},\ldots,a_s^i\sqrt{n_s})\) satisfies
\(K_m(i,j)=\langle v_i,v_j\rangle\).  The Vandermonde vectors for distinct
positive \(a_i\) are independent, so the rank is \(s\).
\end{proof}

\begin{definition}[Effective resolution data]\label{def:effective-resolution-data}
An effective finite-resolution datum is a computable sequence of finite
surjections \(\mathrm{NF}_m\colon\Omega_m\to N_m\) with finite fibres and
computable fibre-size lists.
\end{definition}

\begin{definition}[Certificate universality]\label{def:cert-univ}
An \(L1\) protocol is \(L2\)-certificate-universal if it is equipped with
effective finite-resolution data whose fibre-size multisets provide the fixed
finite certificates of \Cref{prop:classical-finite-atomic-skeleton}.  A
finite-state transfer formula is the stronger extra condition that, for each
fixed \(q\), there are a finite matrix \(A_q\) and vectors \(u_q,v_q\) with
\(S_q^{(m)}=u_q^{\mathsf T}A_q^m v_q\) for all \(m\).
\end{definition}

\begin{proposition}[Finite-state coefficient budget]
\label{prop:finite-state-coefficient-budget}
If a sequence \((b_m)_{m\ge1}\) has a fixed finite-state transfer formula
\(b_m=u^{\mathsf T}A^m v\), then \((b_m)\) satisfies a linear recurrence with
constant coefficients.  Consequently its generating series is rational, and
there are constants \(C,\rho\) such that \(|b_m|\le C\rho^m\).
\end{proposition}

\begin{proof}
The Cayley--Hamilton theorem gives a linear recurrence of order at most the
size of \(A\) for the powers \(A^m\), hence for \(u^{\mathsf T}A^m v\).  A
sequence satisfying a constant-coefficient recurrence has rational generating
series.  The exponential bound follows from any matrix norm: choose
\(\rho>\|A\|\) and absorb the finitely many initial terms into \(C\).
\end{proof}

\begin{proposition}[Finite-fibre certificate taxonomy]
\label{prop:finite-fiber-certificate-taxonomy}
For any effective finite-resolution datum, the finite-fibre certificate at
resolution \(m\) factors through the multiset of fibre sizes.  Thus Hankel
positivity, Prony reconstruction, and the star-kernel rank are fixed-resolution
finite-atomic consequences and are invariant under replacing the datum by any
other datum with the same fibre-size multiset.
\end{proposition}

\begin{proof}
Each listed invariant is a function only of the moments
\(\sum d^q\) of the finite fibre-size multiset.  The conclusions are precisely
\Cref{prop:classical-finite-atomic-skeleton,thm:star-kernel}.
\end{proof}

\begin{proposition}[Zeckendorf certificate universality]
\label{prop:zeckendorf-cert-univ}
The Zeckendorf \(L1\) protocol of \Cref{thm:zeckendorf-protocol-univ} is
\(L2\)-certificate-universal: its finite data are the computable surjections
\(\Fold_m\colon\Omega_m\to X_m\), and their fibre-size multisets give the
certificates of \Cref{prop:classical-finite-atomic-skeleton}.
\end{proposition}

\begin{proof}
For fixed \(m\), both \(\Omega_m\) and \(X_m\) are finite and effectively
enumerable, and \Cref{lem:greedy-zeckendorf-normalizer} computes the fold map.
Counting preimages gives the fibre-size list; the certificate assertions then
follow from \Cref{prop:classical-finite-atomic-skeleton}.
\end{proof}

\begin{theorem}[Finite-fibre skeleton extremal]\label{thm:finite-fiber-skeleton-extremal}
Among data visible to fixed-resolution finite-fibre certificates, the fibre-size
multiset is complete: two systems have the same certificate output for all
polynomial moment, Hankel, Prony, and star-kernel tests if and only if their
fibre-size multisets agree.
\end{theorem}

\begin{proof}
If the multisets agree, every listed test agrees by definition.  Conversely,
the Prony test in \Cref{thm:prony} reconstructs the multiset from finitely many
moments, so agreement of all certificate output forces agreement of the
multiset.
\end{proof}

\begin{proposition}[L2 novelty is finite-state separation]
\label{prop:L2-novelty-finite-state-separation}
The generic finite-fibre certificates of \Cref{prop:zeckendorf-cert-univ} do not
contain a finite-state transfer formula.  Such a formula is an additional
uniform-in-\(m\) structure and is constrained by
\Cref{prop:finite-state-coefficient-budget}.
\end{proposition}

\begin{proof}
The certificates at resolution \(m\) are finite sums over one finite multiset.
They make no assertion about \(m+1\).  A finite-state transfer formula, by
contrast, imposes one fixed matrix model for all \(m\), hence the recurrence and
growth constraints of \Cref{prop:finite-state-coefficient-budget}.
\end{proof}

\begin{corollary}[L2 load split]\label{cor:L2-load-split}
The \(L2\) positive clause is the finite-fibre certificate construction; the
strictness clause is the separate transfer obstruction of \Cref{thm:finite-state-strictness}.
\end{corollary}

\begin{proof}
The positive clause is \Cref{prop:zeckendorf-cert-univ}.  The obstruction is the
failure of the finite-state budget, which is not a fixed-resolution moment
claim by \Cref{prop:L2-novelty-finite-state-separation}.
\end{proof}

\begin{corollary}[Subordinate role of L2 finite-fibre facts]
\label{cor:L2-subordinate-role}
The finite-fibre facts do not prove the Richardson obstruction, the one-orbit
multiplication obstruction, or the Zeckendorf \(L1\) protocol.  They only attach
finite certificates after a normalizer has already been chosen.
\end{corollary}

\begin{proof}
Every result in this section starts from finite fibres of a given normalizing
map.  The \(L1\), Richardson, and one-orbit arguments are proved in
\Cref{sec:L1,sec:strictness} and do not use these certificates.
\end{proof}

\begin{theorem}[Finite-fibre certificates do not imply transfer]
\label{thm:L2-finite-fiber-no-transfer-content}
Finite-fibre certificate data alone impose no recurrence, rational generating
series, exponential growth bound, or fixed finite-matrix transfer formula on
\((S_q^{(m)})_{m\ge1}\).  In the Zeckendorf cover category, for every computable
sequence \(a_m\ge1\) there is a computable finite-fibre cover preserving the
normal forms and readouts whose second moment is
\[
  S_2^\sharp(m)=(a_m+1)^2+F_{m+2}-1 .
\]
Taking \(a_m=m!\) gives a computable finite-fibre certificate system with no
finite-state transfer formula for \(S_2^\sharp(m)\).
\end{theorem}

\begin{proof}
At resolution \(m\), choose one normal form and replace its singleton fibre by a
fibre of size \(a_m+1\), leaving every other fibre unchanged.  This finite cover
does not change the normal-form set or any readout on normal forms, and it is
computable because \((a_m)\) is computable.  The displayed formula follows by
adding the square of the enlarged fibre and the squares of the remaining
singleton fibres.  If \(a_m=m!\), then \((S_2^\sharp(m))\) grows faster than
\(C\rho^m\) for every \(C,
\rho\), contradicting the exponential budget in
\Cref{prop:finite-state-coefficient-budget}.  Hence no fixed finite-state
transfer formula exists.
\end{proof}

\begin{theorem}[Fibre multiset factorization]\label{thm:L2-fiber-multiset-factorization}
Every fixed-resolution \(L2\) certificate considered here factors through the
fibre-size multiset \(\mathcal D_m\).  Thus finite-fibre certificates are exactly
finite-multiset invariants until a uniform transfer structure is added.
\end{theorem}

\begin{proof}
This is \Cref{prop:finite-fiber-certificate-taxonomy} with the notation
\(\mathcal D_m\) emphasized.
\end{proof}

\begin{theorem}[Finite-fibre transfer pairing]\label{thm:L2-finite-fiber-transfer-pairing}
If a finite-fibre certificate system also has a finite-state transfer formula
for \(S_q^{(m)}\), then its fixed-resolution certificate data and its
resolution-wise growth data are paired by a constant-coefficient recurrence.
Conversely, the one-spike covers of \Cref{thm:L2-finite-fiber-no-transfer-content}
show that the fixed-resolution certificates need not admit such a pairing.
\end{theorem}

\begin{proof}
The first sentence is \Cref{prop:finite-state-coefficient-budget}.  The second
is the factorial one-spike construction in
\Cref{thm:L2-finite-fiber-no-transfer-content}.
\end{proof}

\begin{corollary}[Finite-fibre certificates are independent of transfer]
\label{cor:hankel-psd-independent}
Hankel positivity, finite Prony reconstruction, and the star-kernel certificate
can hold at every resolution while finite-state transfer fails.
\end{corollary}

\begin{proof}
They hold for every finite-fibre system by
\Cref{thm:hankel-psd,thm:prony,thm:star-kernel}; finite-state transfer fails for
the factorial one-spike cover in \Cref{thm:L2-finite-fiber-no-transfer-content}.
\end{proof}
