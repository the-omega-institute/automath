\section{Strictness of the hierarchy}\label{sec:strictness}

This section proves the two strictness mechanisms used by
\Cref{thm:main-hierarchy}.  The Richardson part is stated with a verified
finite EML triple as its load-bearing hypothesis.  The printed strings
\(U_{47},U_{111},U_{126}\) appear only as the conditional cited specialization
\((H_{\mathcal W})\) from \Cref{ass:HW}.

\subsection{The Richardson obstruction}

Let \(\mathcal R\) be Richardson's real elementary language from
\cite[pp.~514, 520, Theorem~2]{Richardson1968}: one real variable, rational
constants, \(\log2\), \(\pi\), \(+,-,\times\), composition, \(\exp\), \(\sin\),
and a unary symbol \(\mu\) satisfying \(\mu(x)=|x|\) for \(x\ne0\).  We use the
concrete representative \(\mu_0(x)=\sqrt{x^2}\).

\begin{lemma}[Richardson absolute-value interface]
\label{lem:richardson-punctured-mu-reduction}
Richardson's theorem applies to the concrete total completion
\(\mu_0(x)=\sqrt{x^2}\).  Hence any exact EML translation of rational constants,
\(\log2\), \(\pi\), \(+,-,\times\), \(\exp\), \(\sin\), and
\(x\mapsto\sqrt{x^2}\) carries Richardson's identity-to-zero problem into the
translated EML fragment.
\end{lemma}

\begin{proof}
Richardson assumes only that \(\mu(x)=|x|\) for \(x\ne0\)
\cite[pp.~514, 520]{Richardson1968}.  The function \(\mu_0(x)=\sqrt{x^2}\)
satisfies this condition.  Therefore his undecidability theorem applies to the
language with this concrete completion.  Replacing each \(\mu(F)\) by the EML
translation of \(\sqrt{F^2}\) is an effective syntactic reduction once the
listed primitive translations are exact on their stated real domains.
\end{proof}

\begin{lemma}[Richardson identity-to-zero theorem]
\label{lem:richardson-fragment}
The identity-to-zero problem for \(\mathcal R\) with \(\mu=
\mu_0\) is undecidable.
\end{lemma}

\begin{proof}
This is Richardson's Theorem~2, instantiated by
\Cref{lem:richardson-punctured-mu-reduction}; see
\cite[Theorem~2, p.~520]{Richardson1968}.
\end{proof}

\begin{theorem}[Abstract normalizer obstruction]
\label{thm:abstract-normalizer-obstruction}
Let \(\tau\colon\mathcal R\to\mathcal T\) be an effective translation into a
computable term class whose real semantics preserve domains and values:
\(\dom\llbracket\tau(E)\rrbracket=\dom(E)\) and
\(\llbracket\tau(E)\rrbracket(x)=E(x)
\) on that domain.  Then \(\tau(\mathcal R)\) admits no total computable
normalizer into finite objects with decidable equality that identifies exactly
equal transported real functions.
\end{theorem}

\begin{proof}
If such a normalizer existed, compute \(\tau(E)\) and compare its normal form
with that of \(\tau(0)\).  The comparison would decide whether \(E\) is zero on
its Richardson domain, contradicting \Cref{lem:richardson-fragment}.
\end{proof}

\begin{lemma}[Transported zero-test obstruction]
\label{lem:transported-partial-normalizer-obstruction}
The conclusion of \Cref{thm:abstract-normalizer-obstruction} holds for every
domain-preserving, value-preserving EML translation of \(\mathcal R\).
\end{lemma}

\begin{proof}
Apply \Cref{thm:abstract-normalizer-obstruction} with \(\mathcal T\) equal to the
translated EML fragment.
\end{proof}

\begin{lemma}[Printed EML core witnesses]
\label{lem:eml-printed-core}
In the principal-branch EML semantics of \Cref{def:eml}, the following pure EML
macros represent the indicated real functions:
\[
  E(u)=\eml(u,1)=e^u,
  \qquad
  L(u)=\eml\bigl(1,E(\eml(1,u))\bigr)=\log u\quad(u>0),
\]
\[
  D(a,b)=\eml(L(a),E(b))=a-b\quad(a>0,
b\in\RR),
\]
\[
  P(u)=\eml(u,E(u))=e^u-u>0,
  \quad
  N(u)=D(P(u),E(u))=-u,
\]
\[
  A(u,v)=D(E(u),D(P(u),v))=u+v,
  \qquad S(u,v)=A(u,N(v))=u-v .
\]
\end{lemma}

\begin{proof}
The EML semantics are \(\eml(x,y)=e^x-\operatorname{Log}y\) when the principal
logarithm is defined.  Since the second input of \(E(u)\) is \(1\),
\(E(u)=e^u\).  If \(u>0\), then \(\eml(1,u)=e-\log u\) and \(E(\eml(1,u))=e^e/u>0\), so
\(L(u)=e-\log(e^e/u)=\log u\).  Thus
\(D(a,b)=\exp(\log a)-\log(e^b)=a-b\) for \(a>0\).  The function
\(P(u)=e^u-u\) has minimum \(1\) at \(u=0\), hence is positive.  Substituting in
\(D\) gives \(N(u)=-u\), then \(A(u,v)=u+v\), and finally \(S(u,v)=u-v\).
\end{proof}

\begin{theorem}[Self-contained core EML witness package]
\label{lem:eml-real-witness-library}
The macros of \Cref{lem:eml-printed-core} extend internally to finite pure EML
terms representing multiplication, division by a nonzero real, rational
constants, \(\log2\), and the positive square root:
\[
\begin{aligned}
  T(u,v)&=E(A(L(u),L(v)))=uv\quad(u,v>0),\\
  R_+(u)&=E(N(L(u)))=1/u\quad(u>0),
\end{aligned}
\]
\[
  Q(u)=E(u)>0,
  \qquad P(u)=E(u)-u>0,
\]
\[
\begin{aligned}
  M(u,v)=S\Bigl(&A\bigl(T(Q(u),Q(v)),T(P(u),P(v))\bigr),\\
  &A\bigl(T(Q(u),P(v)),T(P(u),Q(v))\bigr)\Bigr)=uv,
\end{aligned}
\]
for all real \(u,v\).  Indeed all four inputs to \(T\) in this displayed
definition are positive.  With this expanded pure-EML multiplication term,
\[
  J(u,v)=M\bigl(u,M(v,R_+(M(v,v)))\bigr)=u/v\quad(v\ne0),
\]
and
\[
  G(u)=E(J(L(u),A(1,1)))=\sqrt u\quad(u>0).
\]
The value at \(0\) for the nonnegative square root, the constant \(\pi\), and
\(\sin\) on \(\RR\) are not supplied by this core package.
\end{theorem}

\begin{proof}
For positive inputs, \(T(u,v)=\exp(\log u+\log v)=uv\) and
\(R_+(u)=\exp(-\log u)=1/u\).  Since \(Q(u)>0\) and \(P(u)>0\), all four calls
to \(T\) in the definition of \(M\) are made on its positive domain.  Using
\(Q(u)-P(u)=u\), the displayed term gives
\[
  M(u,v)=Q(u)Q(v)+P(u)P(v)-Q(u)P(v)-P(u)Q(v)=uv .
\]
If \(v\ne0\), then \(M(v,v)=v^2>0\), so
\(J(u,v)=u\,v/v^2=u/v\).  Finally
\(G(u)=\exp((\log u)/2)=\sqrt u\) for \(u>0\).  Rational constants are obtained
from \(1\) by repeated addition, negation, and division by nonzero integers;
\(\log2=L(A(1,1))\).
\end{proof}

\begin{theorem}[Positive-core versus nonnegative Richardson root]
\label{thm:positive-core-nonnegative-root-boundary}
The self-contained core EML package supplies a square-root witness only on the
positive domain \((0,\infty)\).  A domain-preserving Richardson translation of
the concrete absolute-value constructor
\(\mu_0(x)=\sqrt{x^2}\) on all real inputs is obtained from the core package
exactly after adding a supplied finite witness \(\mathsf R\) satisfying
\[
  \mathsf R(t)=\sqrt t\qquad(t\in\RR_{\ge0}).
\]
In particular, the positive witness \(G\) of
\Cref{lem:eml-real-witness-library} is not a replacement for the
nonnegative-square-root member of the Richardson triple.
\end{theorem}

\begin{proof}
The proof of \Cref{lem:eml-real-witness-library} constructs
\(G(u)=\sqrt u\) by applying \(L(u)=\log u\).  The term \(L(u)\) is defined only
for \(u>0\), hence the displayed construction proves no semantic value for
\(G(0)\).  For the Richardson absolute-value constructor the translated input to
the square root is \(M(u,u)=u^2\).  This is positive when \(u\ne0\) and is equal
to \(0\) when \(u=0\).  Thus substituting the positive witness \(G\) would leave
the translated term undefined at the real input \(u=0\), so it would not preserve
the domain of the total completion \(\mu_0\).

If a finite witness \(\mathsf R\) is supplied with
\(\mathsf R(t)=\sqrt t\) on all \(t\ge0\), then
\(\mathsf R(M(u,u))=\sqrt{u^2}=|u|\) for every real \(u\).  This is precisely the
zero-domain step used in \Cref{lem:eml-zero-absolute-witness}.  Hence the
positive core and the nonnegative Richardson triple member have the stated
separate roles.
\end{proof}

\begin{lemma}[Quantitative core witness table]
\label{lem:eml-core-witness-table}
The self-contained EML part of the Richardson translation uses the following
finite printed witness table.  Each entry is a pure EML macro from
\Cref{lem:eml-printed-core,lem:eml-real-witness-library}; the domain column is
the exact real domain on which the identity is proved.
\[
\begin{array}{@{}c@{\quad}c@{\quad}c@{}}
\hbox{constructor} & \hbox{witness} & \hbox{domain}\\ \hline
0,1,\mathbb Q,\log2 &
\hbox{finite arithmetic from }1,\ L(A(1,1)) & \RR\hbox{ for constants}\\
u+v & A(u,v) & \RR^2\\
u-v & S(u,v) & \RR^2\\
uv & M(u,v) & \RR^2\\
u/v & J(u,v) & \{(u,v):v\ne0\}\\
e^u & E(u) & \RR\\
\sqrt u & G(u) & (0,\infty)
\end{array}
\]
For any verified Richardson triple \((\mathsf P,\mathsf S,\mathsf R)\), adjoining
the three rows
\[
\begin{array}{@{}c@{\quad}c@{\quad}c@{}}
\pi&\mathsf P&\RR^0\\
\sin u&\mathsf S(u)&\RR\\
\sqrt u&\mathsf R(u)&[0,\infty)
\end{array}
\]
gives the complete finite witness table used by
\Cref{lem:richardson-translation,lem:richardson-translation-bookkeeping}.  The
only overlap between the two tables is the square-root row: the core row is
proved only on \((0,\infty)\), while the Richardson row must include \(0\).
\end{lemma}

\begin{proof}
The first table is exactly the list proved in
\Cref{lem:eml-printed-core,lem:eml-real-witness-library}.  Rational constants are
finite because each fixed rational is obtained by a finite arithmetic expansion
from \(1\), and \(\log2=L(A(1,1))\).  The domains are the domains checked in the
proofs of those two results: the auxiliary logarithm and reciprocal calls occur
only on positive or nonzero inputs, and the multiplication witness is all-real
because its four internal products use \(Q(u)>0\) and \(P(u)>0\).  The second
table is not constructed by the core package; it is precisely the three
verified-triple hypotheses.  The distinction between \(G\) and \(\mathsf R\) is
\Cref{thm:positive-core-nonnegative-root-boundary}.  Therefore these finitely
many displayed rows are exactly the witness library and domain data used in the
structural recursion.
\end{proof}

\begin{theorem}[No core-only Richardson translation]
\label{thm:no-core-only-richardson-translation}
\label{thm:core-root-domain-obstruction}
Let \(\mathcal E_{\mathrm{core}}\) be the pure EML fragment generated by the
self-contained witnesses of \Cref{lem:eml-printed-core,lem:eml-real-witness-library}.
There is no domain-preserving, value-preserving translation of Richardson's
language \(\mathcal R\) with \(\mu=\mu_0\) into
\(\mathcal E_{\mathrm{core}}\) obtained by replacing the Richardson absolute-value
symbol with the core positive-root construction \(G(M(F,F))\).  Consequently the
core package does not by itself instantiate Richardson's theorem; the missing
load-bearing datum is a finite square-root witness defined at \(0\), together
with the separate witnesses for \(\pi\) and sine on all of \(\RR\).
\end{theorem}

\begin{proof}
Take the Richardson term \(F(x)=0\).  The concrete completion used here has
\(\mu_0(F(x))=|0|=0\) for every real input, so a domain-preserving translation of
this term must be defined on all of \(\RR\).  In the core library the only
printed square-root construction is
\[
  G(u)=E(J(L(u),A(1,1))),
\]
and \Cref{lem:eml-real-witness-library} proves its identity only on
\((0,\infty)\).  The proof of \Cref{thm:positive-core-nonnegative-root-boundary}
identifies the reason: the subterm \(L(u)=\log u\) is undefined at \(u=0\).
For the proposed absolute-value replacement, the input to this root is
\(M(F,F)=M(0,0)=0\), since \Cref{lem:eml-real-witness-library} gives
\(M(u,v)=uv\) on all real inputs.  Thus \(G(M(F,F))\) is not defined on the same
real domain as \(\mu_0(F)\).  This already prevents a domain-preserving
translation of \(\mathcal R\) using only the positive-root construction.

The theorem does not assert that no other pure EML string could ever represent
the missing functions.  It asserts exactly the manuscript-level dependency cut:
inside the self-contained core library, no proof supplies a nonnegative
square-root value at \(0\), and the same library also supplies neither \(\pi\) nor
sine on all real inputs.  Therefore the Richardson reduction in this article can
only run after the verified triple required in
\Cref{thm:replacement-triple-richardson-interface} has been supplied, or after
the optional printed-string hypothesis \((H_{\mathcal W})\) supplies that triple
for the named strings.
\end{proof}

\begin{theorem}[Core EML witness boundary]
\label{thm:core-eml-witness-boundary}
\label{thm:no-self-contained-concrete-richardson-eml-instance}
The self-contained EML library proved in this article is exactly a core witness
package for rational constants, \(\log2\), arithmetic operations, composition,
\(\exp\), and positive square root.  It is not a self-contained concrete
Richardson witness package.  In particular, no theorem in this article derives
from that core library all three identities
\[
  \mathsf P=\pi,
  \qquad \mathsf S(x)=\sin x\quad(x\in\RR),
  \qquad \mathsf R(t)=\sqrt t\quad(t\in\RR_{\ge0}),
\]
for any printed pure EML triple.  Hence every Richardson obstruction in the main
chain of \Cref{thm:main-hierarchy} either factors through an externally verified
replacement triple or, for the names \((U_{47},U_{111},U_{126})\), through the
optional hypothesis \((H_{\mathcal W})\).
\end{theorem}

\begin{proof}
The positive content of the internal library is precisely the construction in
\Cref{lem:eml-real-witness-library}: it proves rational constants, \(\log2\),
arithmetic, composition, \(\exp\), and a square-root term on \((0,\infty)\).
The same theorem explicitly does not supply \(\pi\), sine on \(\RR\), or the
value of square root at \(0\).  The domain obstruction at \(0\) is not cosmetic:
\Cref{thm:positive-core-nonnegative-root-boundary,thm:no-core-only-richardson-translation}
show that the Richardson absolute-value constructor requires a square-root
witness on all of \(\RR_{\ge0}\).  The remaining two missing constructors are
the displayed \(\pi\) and sine clauses in
\Cref{lem:richardson-translation,thm:richardson-interface-proof-graph-cut}.
Thus the core package cannot instantiate Richardson's theorem by itself.

For a specified finite pure EML triple whose three displayed identities have
already been proved, \Cref{thm:replacement-triple-richardson-interface} supplies
the Richardson normalizer obstruction.  For the printed names
\((U_{47},U_{111},U_{126})\), \Cref{prop:printed-witness-certificate-boundary}
gives only a syntactic source lock, while
\Cref{thm:printed-richardson-witnesses-interface} supplies the semantic
identities only after assuming \((H_{\mathcal W})\).  Therefore the main theorem
chain has exactly the two routes stated in the theorem and no self-contained
concrete Richardson EML instantiation.
\end{proof}

\begin{lemma}[Source-locked printed strings]
\label{lem:eml-ancillary-certificate}
\label{lem:eml-source-level-check}
\label{lem:eml-expanded-witness-strings}
The symbols \(U_{47},U_{111},U_{126}\) denote fixed finite pure EML strings from
the cited arXiv v2 ancillary record in \Cref{ass:HW}.  This syntactic lock does
not prove their real-domain identities.
\end{lemma}

\begin{proof}
The strings are specified by their public versioned table locations in
\Cref{ass:HW}.  A finite string specification proves only syntactic identity of
the strings so named.  Its semantic content is exactly the separate analytic
assertion \((H_{\mathcal W})\).
\end{proof}

\begin{proposition}[Printed witness certificate boundary]
\label{prop:printed-witness-certificate-boundary}
\label{prop:printed-table-needs-analytic-interface}
\label{prop:printed-root-conditional-only}
\label{prop:printed-root-no-hidden-oracle}
The finite table positions and string locks for \(U_{47},U_{111},U_{126}\) are
not a proof that these strings denote \(\pi\), sine, and nonnegative square root.
The printed-string Richardson route is valid only after the three real-domain
identities are supplied as hypotheses; \((H_{\mathcal W})\) is the optional
specialization retained for these names.
\end{proposition}

\begin{proof}
A finite EML string is syntax.  The Richardson translation requires the three
real-domain semantic identities displayed in \Cref{ass:HW}.  Therefore table
membership can identify the strings but cannot replace the analytic identities.
With \((H_{\mathcal W})\), those identities are hypotheses; without it, the
printed-string branch lacks the required witness interface.
\end{proof}

\begin{theorem}[No unconditional concrete Richardson triple]
\label{thm:no-unconditional-concrete-richardson-triple}
The manuscript does not exhibit an unconditional concrete finite pure EML triple
for \(\pi\), sine on \(\RR\), and nonnegative square root on \(\RR_{\ge0}\).
Consequently no named-string Richardson instance, including
\((U_{47},U_{111},U_{126})\), is a load-bearing unconditional theorem of this
article.  The Richardson branch of \Cref{thm:main-hierarchy} is unconditional
only in the following relative sense: for each separately specified finite pure
EML triple whose three real-domain identities have already been proved,
\Cref{thm:replacement-triple-richardson-interface} applies.
\end{theorem}

\begin{proof}
The complete self-contained witness table available in the manuscript is
\Cref{lem:eml-core-witness-table}.  Its core rows explicitly omit \(\pi\), sine
on \(\RR\), and square root at \(0\); the only rows providing those three
constructors are the three verified-triple rows, which are hypotheses rather
than internally proved strings.  By
\Cref{prop:printed-witness-certificate-boundary}, the source lock for
\((U_{47},U_{111},U_{126})\) proves only that the finite strings have been named,
not that they satisfy the required real-domain identities.  Hence a concrete
printed-string instance follows only after adding those identities, for these
names through \((H_{\mathcal W})\) and
\Cref{thm:printed-richardson-witnesses-interface}.  Without such an added
verification, the only proved Richardson statement is the relative interface
theorem for arbitrary already verified triples.
\end{proof}

\begin{theorem}[Conditional printed-string witness interface]
\label{thm:printed-richardson-witnesses-interface}
Assume \((H_{\mathcal W})\).  Then the source-locked printed strings
\((W_{\pi},W_{\sin},W_{\sqrt{}})=(U_{47},U_{111},U_{126})\) form a finite EML
witness triple for \(\pi\), sine on \(\RR\), and nonnegative square root on
\(\RR_{\ge0}\).  This theorem is only the conditional unpacking of
\((H_{\mathcal W})\); it is not an unconditional proof of the three semantic
identities.
\end{theorem}

\begin{proof}
This is precisely the content of \((H_{\mathcal W})\), with the strings locked
by \Cref{lem:eml-ancillary-certificate}.
\end{proof}

\begin{theorem}[Printed strings are non-load-bearing]
\label{thm:printed-roots-nonloadbearing}
The Richardson obstruction instantiated in this paper is only the
verified-interface theorem \Cref{thm:replacement-triple-richardson-interface}:
it applies after a finite replacement triple
\((\mathsf P,\mathsf S,\mathsf R)\) and the three semantic identities have already
been supplied.  It does not use the three semantic identities asserted by
\((H_{\mathcal W})\).  More precisely, remove
\Cref{ass:HW,lem:eml-ancillary-certificate,prop:printed-witness-certificate-boundary,thm:printed-richardson-witnesses-interface}
from the proof graph.  The interface theorem still proves the normalizer
obstruction for every separately verified finite replacement triple, but it does
not produce a triple for \(\pi\) and sine.  Consequently the printed-string route is
optional unless a stable theorem proving \((H_{\mathcal W})\) is supplied.
\end{theorem}

\begin{proof}
The proof of \Cref{lem:richardson-translation} has as non-core hypotheses only
the three real-domain identities for the specified replacement triple:
\(\mathsf P=\pi\), \(\mathsf S(x)=\sin x\) on \(\RR\), and
\(\mathsf R(t)=\sqrt t\) on \(\RR_{\ge0}\).  It does not cite the printed strings.
Combining that translation with
\Cref{lem:transported-partial-normalizer-obstruction} gives
\Cref{thm:replacement-triple-richardson-interface}, and the proof of
\Cref{thm:main-hierarchy} cites that verified-triple interface for the
Richardson clause.  Thus the proof path contains no occurrence of
\((H_{\mathcal W})\) or the named strings.  The printed strings enter only through
\Cref{thm:printed-richardson-witnesses-interface}, whose statement begins with
an assumption.  Hence, without an external theorem proving that assumption, the
printed-string route is optional and non-load-bearing.
\end{proof}

\begin{theorem}[Exact Richardson proof-graph cut]
\label{thm:richardson-interface-proof-graph-cut}
Let the unconditional proof graph of this article be obtained by deleting the
optional hypothesis \((H_{\mathcal W})\) and every edge whose only purpose is to
turn the names \(U_{47},U_{111},U_{126}\) into analytic witnesses.  In that graph,
every Richardson obstruction used by \Cref{thm:main-hierarchy,thm:obstruction-spine}
factors through the following interface and through no printed-string instantiation:
a finite pure EML triple \((\mathsf P,\mathsf S,\mathsf R)\) together with proofs
of
\[
  \mathsf P=\pi,
  \qquad \mathsf S(x)=\sin x\ (x\in\RR),
  \qquad \mathsf R(t)=\sqrt t\ (t\in\RR_{\ge0}).
\]
Consequently the unconditional manuscript proves the Richardson normalizer
obstruction for each already verified replacement triple, but it proves no
instantiated theorem asserting that \((U_{47},U_{111},U_{126})\) is such a triple.
If \((H_{\mathcal W})\) is added, the only additional conclusion is the optional
specialization obtained by substituting those three printed strings into this
same interface.
\end{theorem}

\begin{proof}
The construction of the transported Richardson fragment is
\Cref{lem:richardson-translation}.  Its structural recursion uses the internal
core witnesses of \Cref{lem:eml-real-witness-library} for rational constants,
\(\log2\), arithmetic, composition, and \(\exp\), and it uses an external triple
only at the three displayed clauses for \(\pi\), sine, and nonnegative square
root.  The absolute-value step is then forced by
\Cref{lem:eml-zero-absolute-witness}.  Hence the translation, and therefore the
zero-test reduction and normalizer obstruction of
\Cref{thm:replacement-triple-richardson-interface}, require exactly the verified
triple interface displayed in the statement.

The printed names enter only through \Cref{ass:HW} and the syntactic lock
\Cref{lem:eml-ancillary-certificate}.  By
\Cref{prop:printed-witness-certificate-boundary}, the syntactic lock is not a
proof of the three analytic identities.  Removing \((H_{\mathcal W})\) therefore
removes the only edge that identifies the printed strings with the verified
triple interface.  The remaining proof graph still contains
\Cref{thm:replacement-triple-richardson-interface}, so every already verified
replacement triple gives the obstruction, but no conclusion about the particular
triple \((U_{47},U_{111},U_{126})\) follows.  Adding \((H_{\mathcal W})\) supplies
exactly those three missing identities for the printed strings, and
\Cref{thm:printed-richardson-witnesses-interface} then gives precisely the
conditional specialization and nothing more.
\end{proof}

\begin{corollary}[Verified-interface criterion]
\label{cor:verified-richardson-interface-criterion}
A Richardson obstruction instance belongs to the verified-interface branch of
\Cref{thm:main-hierarchy} exactly when its finite pure EML triple has already
been accompanied by proofs of the three real-domain identities for \(\pi\), sine
on \(\RR\), and the nonnegative square root on \(\RR_{\ge0}\).  The source-locked
triple \((U_{47},U_{111},U_{126})\) satisfies this criterion in this paper only
under the additional assumption \((H_{\mathcal W})\).
\end{corollary}

\begin{proof}
If the three identities have been proved for a finite pure EML triple, then
\Cref{thm:replacement-triple-richardson-interface} applies and supplies the
verified-interface branch.  Conversely, the translation of
\Cref{lem:richardson-translation} needs witnesses for the source constructors
\(\pi\), \(\sin\), and \(x\mapsto\sqrt{x^2}\), while
\Cref{prop:printed-witness-certificate-boundary} says that the printed table
alone is not such a semantic interface.  For \((U_{47},U_{111},U_{126})\),
\Cref{thm:printed-richardson-witnesses-interface} provides that interface only
after assuming \((H_{\mathcal W})\).  Hence the named triple is not an
unconditional instance in this manuscript.
\end{proof}

\begin{theorem}[Printed-string real-domain upgrade criterion]
\label{thm:printed-string-real-domain-upgrade-criterion}
\label{thm:unconditional-printed-string-instance-criterion}
Let \(W_{\pi},W_{\sin},W_{\sqrt{}}\) be any three printed pure EML strings,
including the source-locked names \((U_{47},U_{111},U_{126})\).  The following are
equivalent for this manuscript's Richardson theorem chain.
\begin{enumerate}
\item The printed triple is an unconditional concrete Richardson instance of
  \Cref{thm:main-hierarchy}.
\item The manuscript supplies, or cites as an explicit theorem, proofs of the
  three principal-branch real-domain identities
  \[
    W_{\pi}=\pi,
    \qquad W_{\sin}(x)=\sin x\quad(x\in\RR),
    \qquad W_{\sqrt{}}(t)=\sqrt t\quad(t\in\RR_{\ge0}).
  \]
\item The triple is an element of the verified-interface family to which
  \Cref{thm:replacement-triple-richardson-interface} applies.
\end{enumerate}
Consequently the present unconditional submission, which does not prove these
three identities for \((U_{47},U_{111},U_{126})\), has no unconditional concrete
Richardson instance for those names.  Under \((H_{\mathcal W})\), and only under
that additional analytic input, the same theorem gives the conditional
printed-string specialization.
\end{theorem}

\begin{proof}
Assume first that the printed triple is an unconditional concrete Richardson
instance.  The Richardson translation used in \Cref{lem:richardson-translation}
replaces the source symbols \(\pi\), \(\sin\), and \(t\mapsto\sqrt t\) by the
three printed EML terms.  For the translated zero-test to be a many-one reduction
from Richardson's real identity problem, the replacement must preserve the
domains and values of those three source symbols on the displayed real domains.
Thus the three real-domain identities in (2) are necessary.

If (2) holds, the triple is a finite pure EML triple accompanied by exactly the
semantic proofs required in the definition of the verified-interface family;
therefore (3) holds.  If (3) holds, then
\Cref{thm:replacement-triple-richardson-interface} applies to the triple and
supplies the Richardson many-one reduction, the normalizer obstruction, and the
compiler obstruction.  This is precisely an unconditional concrete instance
when the three proofs are present in the submission or in an explicitly cited
theorem, so (1) follows.

For \((U_{47},U_{111},U_{126})\), the manuscript records only a syntactic
source lock and the optional analytic assumption \((H_{\mathcal W})\).
\Cref{prop:printed-witness-certificate-boundary} shows that the syntactic lock is
not a proof of the identities, and \Cref{cor:verified-richardson-interface-criterion}
identifies those identities as the necessary and sufficient interface.  Hence
the named strings are not an unconditional concrete instance here.  If
\((H_{\mathcal W})\) is added, it supplies exactly the identities in (2), so the
equivalence places the named triple in the verified family and yields precisely
the conditional specialization.
\end{proof}

\begin{proposition}[Locked three-witness interface]
\label{prop:locked-three-witness-interface}
\label{prop:three-witness-stability}
\label{prop:eml-certificate-reproducibility}
Fix the internal core EML library of \Cref{prop:internal-core-eml-witnesses}.
Let \((\mathsf P,\mathsf S,\mathsf R)\) and
\((\mathsf P',\mathsf S',\mathsf R')\) be two finite pure EML triples satisfying
the three real-domain identities for \(\pi\), sine on \(\RR\), and square root on
\(\RR_{\ge0}\).  Then the two Richardson translations obtained from these triples
have the same real domains and values on every source expression in
\(\mathcal R\).  Consequently the Richardson many-one reduction and the excluded
normalizer/compiler conclusions of
\Cref{thm:replacement-triple-richardson-interface,thm:separation} depend only on
the three verified semantic identities, not on the printed syntax of the chosen
triple.  In particular, replacing the optional printed triple
\((U_{47},U_{111},U_{126})\) by any other verified finite triple preserves the
obstruction, while using the printed triple without \((H_{\mathcal W})\) supplies
no obstruction instance.
\end{proposition}

\begin{proof}
For a source expression \(E\in\mathcal R\), form the two translated EML terms
\(\tau(E)\) and \(\tau'(E)\) by the same structural recursion.  We prove by
induction on \(E\) that the transported real partial functions have identical
domains and values.  The variables, rational constants, \(\log2\), arithmetic
operations, composition, and \(\exp\) are translated in both recursions by the
same internal core witnesses of \Cref{prop:internal-core-eml-witnesses}; hence
the induction step for those constructors is immediate from the already proved
core identities.  For the three non-core constructors, the hypotheses give
\(\mathsf P=\mathsf P'=\pi\),
\(\mathsf S(x)=\mathsf S'(x)=\sin x\) for every real \(x\), and
\(\mathsf R(t)=\mathsf R'(t)=\sqrt t\) for every \(t\ge0\), with the same stated
domains.  Thus these base and unary cases also preserve exactly the source
domain and source value.  The partial-domain clauses for composition and the
arithmetic constructors are determined only by the domains and values of their
subterms, so the induction closes.

Therefore every source expression is sent by the two triples to extensionally
equal transported real partial functions.  The zero-test reduction in
\Cref{thm:replacement-triple-richardson-interface} uses only this extensional
domain-and-value preservation and Richardson's theorem, so the reduction is
unchanged by replacing one verified triple with another.  The normalizer and
compiler obstructions in that theorem are consequences of the same reduction;
they are therefore stable under the same replacement.  If
\((H_{\mathcal W})\) is assumed, \Cref{thm:printed-richardson-witnesses-interface}
turns \((U_{47},U_{111},U_{126})\) into one such verified triple.  Without that
hypothesis, \Cref{prop:printed-witness-certificate-boundary} gives only fixed
finite strings and none of the three semantic identities, so the proposition
supplies no instantiated printed-string obstruction.
\end{proof}

\begin{proof}

\(\mathsf{Rich}(U_{47},U_{111},U_{126})\) is not an unconditional consequence of

It remains to check that no hidden source of printed-name semantics enters the

\begin{proposition}[Verified-triple replacement theorem]
\label{prop:verified-triple-replacement-theorem}
Let \((\mathsf P,\mathsf S,\mathsf R)\) and
\((\mathsf P',\mathsf S',\mathsf R')\) be finite pure EML triples.  Suppose that
\((\mathsf P,\mathsf S,\mathsf R)\) satisfies the three real-domain identities
\[
  \mathsf P=\pi,
  \qquad \mathsf S(x)=\sin x\quad(x\in\RR),
  \qquad \mathsf R(t)=\sqrt t\quad(t\in\RR_{\ge0}),
\]
in the principal-branch semantics of \Cref{def:eml}, while no proof of those
three identities is supplied for
\((\mathsf P',\mathsf S',\mathsf R')\).  Then every Richardson obstruction,
normalizer obstruction, and protocol-compiler obstruction used in
\Cref{thm:main-hierarchy} is obtained from the first triple and cannot be
transferred to the second triple merely because the second triple is printed or
source-locked.  In particular, after deleting \((H_{\mathcal W})\), the printed
triple \((U_{47},U_{111},U_{126})\) is not a Richardson witness triple in this
article, while any independently verified replacement triple is.
\end{proposition}

\begin{proof}
The hypotheses on \((\mathsf P,\mathsf S,\mathsf R)\) are exactly the hypotheses
of \Cref{thm:replacement-triple-richardson-interface}.  Applying that theorem
gives Richardson's many-one zero-test reduction for the translated fragment and
also the nonexistence of a computable semantic normalizer into finite decidable
objects and of an equality-reflecting compiler into finite protocol objects.
Those are precisely the Richardson consequences cited later through
\Cref{thm:separation,cor:strict-L0-L1,prop:L1-compiles-L0} in the proof of
\Cref{thm:main-hierarchy}.

For \((\mathsf P',\mathsf S',\mathsf R')\), the same application is impossible
unless its own three principal-branch real-domain identities have first been
proved.  The translation lemma treats \(\pi\), sine on all of \(\RR\), and the
nonnegative square root as semantic inputs, not as names; hence replacing one
finite triple by another is valid only after the replacement triple has the same
proved identities.  A source lock supplies a finite string and no analytic
identity by \Cref{prop:printed-witness-certificate-boundary}.  Therefore the
obstruction cannot be transported from a verified triple to an unverified
printed triple by notation alone.

Specializing \((\mathsf P',\mathsf S',\mathsf R')\) to
\((U_{47},U_{111},U_{126})\), the only article-level assertion of the three
identities is the optional hypothesis \((H_{\mathcal W})\) recorded in
\Cref{ass:HW,thm:printed-richardson-witnesses-interface}.  Removing that
hypothesis leaves exactly the verified replacement-triple theorem and removes
the printed-string specialization.  Conversely, any finite replacement triple
whose three identities are proved satisfies the first paragraph and therefore
is a valid Richardson witness for this article's hierarchy theorem.
\end{proof}

\begin{theorem}[No unconditional printed-string Richardson instance]
\label{thm:no-unconditional-printed-root-richardson-instance}
\label{thm:printed-root-gap-is-interface-gap}

unconditional instantiated branch whose witness triple is
\((U_{47},U_{111},U_{126})\).  More precisely, the statement
\(

\), in the sense of \Cref{thm:richardson-scope-exact-main-clause}, is available
in this manuscript if and only if the printed strings have first been supplied
with the verified-triple interface.  In the present proof graph that can happen
only through one of the following printed-string inputs:
\begin{enumerate}
\item the external hypothesis \((H_{\mathcal W})\), which asserts the three
  semantic identities for those printed strings; or
\item a separate proof, not present in this article, that these same three
  printed strings satisfy the real-domain identities for \(\pi\), sine on
  \(\RR\), and nonnegative square root on \(\RR_{\ge0}\), all in the
  principal-branch real-domain EML semantics of \Cref{def:eml}.
\end{enumerate}
Consequently, if \((H_{\mathcal W})\) is not assumed and no such principal-branch
real-domain proof for the printed strings is supplied, the unconditional
Richardson statement remaining in this paper is exactly the verified-interface
theorem \Cref{thm:replacement-triple-richardson-interface}, not a concrete
pure-EML triple theorem for \(\pi\), sine, and square root.  Separately verified
finite replacement triples still give their own interface instances, but those
are not instances for the names \((U_{47},U_{111},U_{126})\) unless the separate
proof is about those same strings.  This is the status used by
\Cref{thm:main-hierarchy}: the main hierarchy has a universal verified-interface
Richardson clause and no unconditional printed-string specialization.
\end{theorem}

\begin{proof}
The printed table contributes only finite syntax: by
\Cref{lem:eml-ancillary-certificate}, the strings
\(U_{47},U_{111},U_{126}\) are source-locked strings, while
\Cref{prop:printed-witness-certificate-boundary} says that such locking is not a
proof that they denote \(\pi\), sine, and nonnegative square root.  The only
statement that turns these particular strings into a semantic witness triple is
\Cref{thm:printed-richardson-witnesses-interface}, and its first hypothesis is
\((H_{\mathcal W})\).

Conversely, if either listed printed-string input is present, then the hypotheses
of \Cref{thm:replacement-triple-richardson-interface} hold
with \((\mathsf P,\mathsf S,\mathsf R)=(U_{47},U_{111},U_{126})\).  Substitution in
that interface theorem gives exactly
\(\mathsf{Rich}(U_{47},U_{111},U_{126})\).  Thus the printed-string specialization
is equivalent, inside this article, to first supplying the three real-domain
identities for those printed strings; it is not a conclusion drawn from their
finite syntax alone.  Because the semantics of \Cref{def:eml} use the principal
complex logarithm, the required identities are principal-branch identities on the
real domains displayed in the statement; a formal equality of strings, or a
calculation in a different branch convention, is not the verified-triple input

Finally, \Cref{prop:verified-triple-replacement-theorem,thm:printed-roots-nonloadbearing} and
\Cref{thm:finite-eml-richardson-trichotomy} record the same dependency cut in
the main proof graph: removing \((H_{\mathcal W})\) and the printed-string witness

asserting that \((U_{47},U_{111},U_{126})\) is a concrete EML witness triple.
Hence the unconditional conclusion is exactly the interface theorem and not an

\begin{theorem}[Anchor-erasure for the unconditional Richardson separation]
\label{thm:root-name-erasure-unconditional-separation}
\label{thm:separation-is-interface-only}
\label{thm:public-anchor-erasure-richardson}
Remove the optional hypothesis \((H_{\mathcal W})\) from the hypotheses of this

\((U_{47},U_{111},U_{126})\) by arbitrary fixed finite pure EML strings.  In

\begin{proof}

  gives the transported Richardson zero-test, normalizer, and compiler

\item no unconditional theorem of this article instantiates those obstructions
  with the particular names \((U_{47},U_{111},U_{126})\);
\item replacing \(\mathbf B\) by another cited representative-existence theorem
  with the same \(\mathcal B_{36}\) content, or replacing \(\mathbf U\) by any

\end{enumerate}
Thus the Proc. AMS spine proves an unconditional verified-triple interface and
an optional printed-string specialization only under \((H_{\mathcal W})\); it
does not prove or use a self-contained \(\mathcal B_{36}\) representation theorem
or an unconditional concrete Richardson triple for the printed strings.
\end{theorem}

\begin{proof}
The first assertion is exactly
\Cref{thm:replacement-triple-richardson-interface}.  Its proof constructs
\(\tau_{\mathsf P,\mathsf S,\mathsf R}\) by the structural recursion of
\Cref{lem:richardson-translation}.  The clauses for rational constants,
\(\log2\), arithmetic operations, composition, and \(\exp\) use only the printed

three remaining Richardson clauses use exactly the hypotheses
\[
  \mathsf P=\pi,
  \qquad \mathsf S(x)=\sin x\quad(x\in\RR),
  \qquad \mathsf R(t)=\sqrt t\quad(t\in\RR_{\ge0}).
\]
Once those hypotheses are present, \Cref{lem:richardson-translation} preserves

reduces to zero-identity on the translated pure EML fragment.  The normalizer
and compiler obstructions follow by the decision-procedure argument in
\Cref{thm:abstract-normalizer-obstruction,thm:replacement-triple-richardson-interface}.

For the second assertion, \Cref{lem:eml-ancillary-certificate} gives only a
source lock for the three printed strings.  By
\Cref{prop:printed-witness-certificate-boundary}, such a lock is not a proof of
the three real-domain identities.  \Cref{thm:no-unconditional-printed-root-richardson-instance}
then states the exact alternative: the names become a Richardson instance only

for those same strings has been added.  Both inputs are excluded by the

For the third assertion, inspect the proof of the main hierarchy.  Its \(L0\)
clause cites \(\mathcal B_{36}\) only through the representative-existence scope

\end{proof}

\begin{theorem}[Largest verified Richardson fragment]

Richardson's undecidable identity-to-zero problem on \(\mathcal R\), fixed in
\Cref{lem:richardson-fragment}, many-one reduces to zero-identity on

\begin{theorem}[Relative finite Richardson fragment]
\label{thm:verified-finite-richardson-fragment}

\((\mathsf P,\mathsf S,\mathsf R)\) satisfying the three real-domain identities

\end{theorem}

\begin{proof}

\begin{lemma}[Structural-recursion Richardson translation with domain clauses]
\label{lem:richardson-translation}
Let \((\mathsf P,\mathsf S,
\mathsf R)\) be a finite pure EML triple satisfying the three real-domain
identities.  There is an effective translation \(\tau_{\mathsf P,\mathsf S,
\mathsf R}\colon\mathcal R\to\) pure EML terms such that, for every
\(E\in\mathcal R\),
\[
  \dom\llbracket\tau_{\mathsf P,\mathsf S,
  \mathsf R}(E)\rrbracket_{\RR}=\dom(E),
  \qquad
  \llbracket\tau_{\mathsf P,\mathsf S,
  \mathsf R}(E)\rrbracket_{\RR}(x)=E(x)
\]
for all \(x\in\dom(E)\).  Moreover \(|\tau(E)|\le K^{2|E|}\) and
\(\operatorname{depth}\tau(E)\le K|E|\) for a constant \(K\) depending only on
the finite witness library and the rational-constant expansion convention.  The
recursion uses the supplied triple only in the clauses for the source
constructors \(\pi\), \(\sin|_{\RR}\), and
\(t\mapsto\sqrt t\) on \(\RR_{\ge0}\); all other clauses are supplied by the
internal core library.
\end{lemma}

\begin{proof}
Define \(
\tau\) by structural recursion.  Variables map to variables.  Rational
constants and \(\log2\) use \Cref{lem:eml-real-witness-library}; \(\pi\) uses
\(\mathsf P\).  The constructors \(+,-,\times,\exp\) use the corresponding core
witnesses, \(\sin\) uses \(\mathsf S\), composition is syntactic substitution,
and \(\mu(F)\) is translated as \(\mathsf R(M(\tau(F),\tau(F)))\).  The
translation is effective because all witnesses are finite.

The domain and value assertion is proved by induction on the expression tree.
The primitive cases are exactly the proved real-domain identities in
\Cref{lem:eml-printed-core,lem:eml-real-witness-library} and in the supplied
triple.  The composition step is the definition of partial-function
composition.  The \(\mu\)-step is \Cref{lem:eml-zero-absolute-witness}.  This
proves exact agreement on real domains.

For the size bound, choose \(K\) larger than the size, depth, and input-leaf
multiplicity of every witness in the finite library and large enough for the
chosen rational-constant expansions.  Substituting translated children into a
fixed witness multiplies size by at most a factor bounded by \(K\) and adds at most \(K\) to depth;
composition multiplies sizes and adds depths.  Induction on \(|E|\) gives the
displayed bounds after increasing \(K\) once.
\end{proof}

\begin{lemma}[Quantitative Richardson translation]
\label{lem:richardson-translation-bookkeeping}
\label{prop:richardson-translation-bookkeeping}
Fix the printed core EML witness table of \Cref{lem:eml-core-witness-table} and a verified triple
\((\mathsf P,\mathsf S,\mathsf R)\) satisfying
\[
  \mathsf P=\pi,
  \qquad
  \mathsf S(x)=\sin x\quad(x\in\RR),
  \qquad
  \mathsf R(t)=\sqrt t\quad(t\in\RR_{\ge0}).
\]
Measure the size of a Richardson expression by its expression-tree size after
expanding rational constants according to the fixed convention recorded in that
table.  Then the translation
\(\tau_{\mathsf P,\mathsf S,\mathsf R}\) of
\Cref{lem:richardson-translation} can be chosen with the following uniform
bounds and admissibility properties.  There is a constant \(C\), depending only
on the fixed finite witness list and on the rational-expansion convention, such
that for every Richardson expression \(E\),
\[
  |\tau_{\mathsf P,\mathsf S,\mathsf R}(E)|\le C^{|E|},
  \qquad
  \operatorname{depth}\tau_{\mathsf P,\mathsf S,\mathsf R}(E)
  \le C|E|,
\]
and the number of nested source-substitution steps used on any root-to-leaf path
in constructing \(\tau_{\mathsf P,\mathsf S,\mathsf R}(E)\) is at most \(|E|\).
Moreover every principal-branch call introduced by the core library is made on
its stated real domain on \(\dom(E)\), and the only non-core admissibility inputs
are exactly the three displayed identities for \(\mathsf P\), \(\mathsf S\), and
\(\mathsf R\).  In particular the \(\mu_0\)-clause calls \(\mathsf R\) only on
values of the form \(F(x)^2\ge0\), including the possible value \(0\).
\end{lemma}

\begin{proof}
Let \(\mathcal L\) be the finite list of core witnesses for
\(0,1\), \(\log2\), addition, subtraction, multiplication, reciprocal on nonzero
denominators, division, \(\exp\), and positive square root displayed in
\Cref{lem:eml-core-witness-table}; rational constants are
first expanded from \(0,1\) using this same fixed arithmetic library.  Adjoin the
three verified strings \(\mathsf P,\mathsf S,\mathsf R\).  Let \(C\) be larger
than twice the maximum, over this finite list of primitive witnesses, of the tree
size, syntactic depth, and number of input leaves of any witness; enlarge \(C\)
once more so that each fixed rational-leaf expansion contributes at most \(C\)
times its counted expanded size.

The translation is the structural recursion of
\Cref{lem:richardson-translation}.  A variable, \(\pi\), \(\log2\), a rational
constant, or the zero constant is replaced by the corresponding fixed witness.
The constructors \(+,-,\times,\exp,\sin\) are translated by substituting the
already translated children into the fixed witness for that constructor, with
\(\sin\) using \(\mathsf S\).  Composition is syntactic substitution of the
translated inner expression into the translated outer expression.  Finally
\(\mu(F)\) is translated as \(\mathsf R(M(\tau(F),\tau(F)))\), where \(M\) is the
all-real multiplication witness of \Cref{lem:eml-real-witness-library}.

We prove the two bounds by induction on the expanded expression tree.  In a
primitive leaf case the assertion follows from the choice of \(C\).  At a fixed
constructor with translated children \(T_1,\ldots,T_r\), substituting the
children into a witness from \(\mathcal L\cup\{\mathsf P,\mathsf S,\mathsf R\}\)
adds at most a constant amount to depth and increases size by at most a constant
multiple of \(1+|T_1|+\cdots+|T_r|\).  Increasing \(C\) if necessary, the
induction hypotheses therefore give
\(|T|\le C^{1+|E_1|+\cdots+|E_r|}\) and
\(\operatorname{depth}T\le C(1+|E_1|+\cdots+|E_r|)\).  A composition node is the
same estimate with one additional source-substitution step: its tree size is
bounded by the product of the outer translated size and the largest substituted
inner size, hence by \(C^{|E|}\) after increasing \(C\), while depths add and so
remain bounded by \(C|E|\).  Along a root-to-leaf path the recursion passes
through at most one source node at each step, so the nested source-substitution
count is at most the number of nodes of \(E\).

It remains to record admissibility.  The primitive and arithmetic core clauses
are admissible on precisely the domains proved in
\Cref{lem:eml-printed-core,lem:eml-real-witness-library}; the induction
hypothesis supplies those domains for the translated children.  The supplied
identity \(\mathsf S(x)=\sin x\) is total on \(\RR\), and \(\mathsf P\) is a
constant.  For the \(\mu_0\)-clause, the multiplication witness gives
\(M(\tau(F)(x),\tau(F)(x))=F(x)^2\) for every \(x\in\dom(F)\), and this value is
nonnegative.  The hypothesis on \(\mathsf R\) therefore applies also when
\(F(x)=0\).  Thus no hidden principal-branch or square-root-at-zero assumption is
introduced beyond the three displayed verified-triple identities, and the
domain and value conclusions of \Cref{lem:richardson-translation} hold with the
quantitative bounds just proved.
\end{proof}

\begin{theorem}[Verified-triple Richardson obstruction interface]
\label{thm:replacement-triple-richardson-interface}
Let \((\mathsf P,\mathsf S,\mathsf R)\) be a specified finite pure EML triple
for which the real-domain identities
\[
  \mathsf P=\pi,
  \qquad
  \mathsf S(x)=\sin x\quad(x\in\RR),
  \qquad
  \mathsf R(t)=\sqrt t\quad(t\in\RR_{\ge0})
\]
have been proved, and let
\(\tau=\tau_{\mathsf P,\mathsf S,\mathsf R}\) be the translation of
\Cref{lem:richardson-translation}.  Then the Richardson identity-to-zero problem
many-one reduces to the zero-identity problem on the translated pure EML
fragment \(\tau(\mathcal R)\):
\[
  E\equiv 0\hbox{ on }\dom(E)
  \quad\Longleftrightarrow\quad
  \tau(E)\equiv \tau(0)
  \hbox{ on the transported real domain}.
\]
Consequently the translated fragment admits none of the following computable
structures:
\begin{enumerate}
\item a zero-identity decision procedure;
\item a semantic normalizer into finite objects with decidable equality that
  identifies exactly equal transported real functions;
\item a semantic compiler into any finite decidable protocol object whose
  equality reflects and preserves equality of the transported real functions on
  this fragment.
\end{enumerate}
This is an interface theorem: it is conditional on the displayed finite triple
and its proved identities.  It does not construct \(\mathsf P\),
\(\mathsf S\), or \(\mathsf R\), and it does not use \((H_{\mathcal W})\),
\(U_{47}\), \(U_{111}\), or \(U_{126}\).
\end{theorem}

\begin{proof}
By \Cref{lem:richardson-translation,lem:richardson-translation-bookkeeping}, the
verified triple gives an effective, quantitatively bounded,
domain-preserving, value-preserving translation of \(\mathcal R\) into pure EML
terms.  Hence for every source expression \(E\),
\[
  \dom\llbracket\tau(E)\rrbracket_{\RR}=\dom(E),
  \qquad
  \llbracket\tau(E)\rrbracket_{\RR}(x)=E(x)
  \quad(x\in\dom(E)).
\]
The same identities hold for the constant zero expression.  Therefore
\(E\) is identically zero on its Richardson domain if and only if the two
translated EML terms \(\tau(E)\) and \(\tau(0)\) define the same transported real
partial function.  Since \(\tau\) is effective, this is a many-one reduction from
Richardson's identity-to-zero problem of \Cref{lem:richardson-fragment} to the
zero-identity problem on \(\tau(\mathcal R)\).  The first excluded structure would
therefore decide Richardson's undecidable problem.

If a semantic normalizer into finite objects with decidable equality existed,
then computing the two normal forms of \(\tau(E)\) and \(\tau(0)\) and comparing
them would be a zero-identity decision procedure on \(\tau(\mathcal R)\), again
contradicting \Cref{lem:richardson-fragment}.  This is precisely
\Cref{lem:transported-partial-normalizer-obstruction} applied to the verified
translation.

Finally suppose there were a computable semantic compiler from
\(\tau(\mathcal R)\) into finite decidable protocol objects whose equality
reflects and preserves equality of the transported real functions on the
fragment.  Applying the compiler to \(\tau(E)\) and \(\tau(0)\), then deciding
target equality, would again decide whether \(E\) is identically zero.  Thus the
compiler is impossible as well.  The only non-core analytic inputs used anywhere
in this argument are the three displayed identities for the specified triple, so
the theorem is an interface theorem and contains no printed-string instantiation.
\end{proof}


\begin{theorem}[Richardson interface and endpoint obstruction]
\label{thm:submission-richardson-dichotomy}
\label{thm:proc-ams-relative-richardson-theorem}
\label{thm:relative-richardson-main-spine-closure}
\label{thm:concrete-richardson-instance-exact-data}
\label{thm:richardson-triple-submission-alternative}
\label{thm:printed-root-richardson-conditional-clause}
\label{thm:conditional-richardson-witness-package}
\label{thm:exact-richardson-witness-conditionality}
\label{thm:exact-source-boundary-richardson-branch}
\label{cor:richardson-witness-alternative}
\label{cor:conditional-three-witness-obstruction}
\label{cor:no-instantiated-named-root-consequence}
\label{thm:closed-main-interface}
\label{thm:B36-and-printed-root-scope-closure}
\label{thm:no-B36-Richardson-instantiation}
\label{thm:verified-triple-only-richardson-adjoining}
\label{cor:public-provenance-free-richardson-closure}
\label{thm:exact-unconditional-richardson-scope}
\label{thm:richardson-interface-instantiation-criterion}
\label{cor:concrete-richardson-instance-alternative}
\label{thm:unconditional-richardson-factorization}
\label{thm:no-unconditional-printed-richardson-edge}
\label{thm:separation}
\label{thm:relative-richardson-branch-closure}
\label{thm:no-principal-branch-eml-nonnegative-root}
Fix the principal-branch pure EML semantics of \Cref{def:eml}.  The Richardson
part of the main theorem is the following closed relative interface.
\begin{enumerate}
\item If a finite pure EML triple \((\mathsf P,\mathsf S,\mathsf R)\) is supplied
with proved real-domain identities
\[
  \mathsf P=\pi,
  \qquad \mathsf S(x)=\sin x\quad(x\in\RR),
  \qquad \mathsf R(t)=\sqrt t\quad(t\in\RR_{\ge0}),
\]
then Richardson's identity-to-zero problem many-one reduces to zero-identity on
the translated pure EML fragment.  Consequently that fragment admits no
zero-identity algorithm, no computable semantic normalizer into finite objects
with decidable equality, and no equality-reflecting semantic compiler into finite
decidable protocol objects.
\item The named strings \((U_{47},U_{111},U_{126})\) give such a concrete
printed-string instance only under \((H_{\mathcal W})\) or after an independent
proof of the same three real-domain identities.  Source-locking or printing the
strings is not a proof of those identities.
\item In the ordinary partial principal-branch semantics, no finite pure EML term
which is defined on a right-neighbourhood of \(0\) can represent
\(t\mapsto\sqrt t\) on \(\RR_{\ge0}\).  Hence this article cannot honestly replace
the verified-triple hypothesis by a self-contained pure-EML nonnegative-square-root
witness.  Its self-contained square-root witness is exactly the positive-domain
term \(G(t)\) of \Cref{lem:eml-real-witness-library}; the full Richardson edge is
therefore relative to a supplied verified triple.
\end{enumerate}
This theorem is the Richardson clause used in
\Cref{thm:obstruction-spine,thm:main-hierarchy}.  The EML witness-catalogue
problem for \(\pi\), sine, and endpoint square root is a separate witness-library
problem and is not silently imported into the Proc. AMS proof chain.
\end{theorem}

\begin{proof}
For (1), \Cref{lem:richardson-translation,lem:richardson-translation-bookkeeping}
give an effective, domain-preserving, value-preserving translation
\(\tau=\tau_{\mathsf P,\mathsf S,\mathsf R}\) of Richardson expressions into pure
EML terms.  Thus, for every Richardson expression \(E\),
\[
  E\equiv0\hbox{ on }\dom(E)
  \quad\Longleftrightarrow\quad
  \tau(E)\equiv\tau(0)
  \hbox{ on the transported real domain}.
\]
Since Richardson's identity-to-zero problem for \(\mathcal R\) is undecidable by
\Cref{lem:richardson-fragment}, this effective equivalence is a many-one
reduction.  A zero-identity algorithm on the translated fragment would decide the
Richardson problem.  A computable semantic normalizer into finite objects with
decidable equality would decide the same problem by normalizing \(\tau(E)\) and
\(\tau(0)\) and comparing the results; this is the transported-normalizer
obstruction of \Cref{lem:transported-partial-normalizer-obstruction}.  Likewise,
an equality-reflecting semantic compiler into any finite decidable protocol
object would decide the problem by compiling \(\tau(E)\) and \(\tau(0)\) and then
checking target equality.  This proves all obstruction conclusions in (1).

For (2), inspect the construction just used.  Its non-core inputs are exactly the
three displayed identities for \(\mathsf P\), \(\mathsf S\), and \(\mathsf R\).
The optional hypothesis \((H_{\mathcal W})\) asserts those identities for the
printed names \((U_{47},U_{111},U_{126})\), so under that hypothesis (1)
specializes to those strings.  Without \((H_{\mathcal W})\) or an independent
proof of the same analytic identities, the printed names contribute only finite
syntax and provenance; finite syntax alone contains no semantic verification.
Therefore the unconditional paper proves the universal verified-triple
implication, not a concrete printed-string Richardson instance.

It remains to prove the endpoint obstruction in (3).  Let \(T(t)\) be a finite
pure EML term defined for all \(t\in[0,\epsilon)\).  We claim that, after possibly
shrinking \(\epsilon\), the real-domain value of every subterm of \(T\) is a
real-analytic function of \(t\) on a neighbourhood of \(0\) whenever that subterm
is defined at \(0\) and is continuous on the right-neighbourhood under
consideration.  This is a structural induction.  The constant \(1\) and the input
\(t\) are analytic.  If a subterm has the form \(\eml(A(t),B(t))\), then by
induction \(A\) and \(B\) are analytic near \(0\).  Since the term is defined at
\(0\), the second input satisfies \(B(0)\ne0\) and lies in the principal-logarithm
domain.  After shrinking the neighbourhood, \(B\) stays inside one local branch
domain of the principal logarithm.  Hence
\(e^{A(t)}-\operatorname{Log}B(t)\) is analytic there.  This completes the
induction.

If such a term represented \(\sqrt t\) on \([0,\epsilon)\), its value would
therefore be real-analytic at \(0\).  Write its Taylor expansion as
\(a_0+a_1t+a_2t^2+\cdots\).  Equality with \(\sqrt t\) for \(t>0\) forces
\(a_0=0\).  Dividing by \(t\) gives
\(a_1+a_2t+\cdots=t^{-1/2}\), impossible as \(t\downarrow0\) because the left
side remains bounded while the right side diverges.  Equivalently,
\(\sqrt t\) is not differentiable at \(0\).  Thus no finite pure EML term defined
at the endpoint can be the nonnegative square root on all of \(\RR_{\ge0}\).
The positive-domain witness \(G(t)=\exp((\log t)/2)\) remains valid on
\((0,\infty)\) by \Cref{lem:eml-real-witness-library}, but it is undefined at
\(0\) and so cannot supply the Richardson absolute-value constructor at zeros.
\end{proof}

\begin{corollary}[Verified-triple expression/protocol separation]
\label{cor:strict-L0-L1}
For every verified finite EML Richardson triple
\((\mathsf P,\mathsf S,\mathsf R)\), the imported \(L0\)
expression-universality result for \((\eml,1)\) does not provide a computable
semantic normalizer for the Richardson fragment translated through that triple.
For the named printed strings \((U_{47},U_{111},U_{126})\), this is only the
conditional specialization obtained after assuming \((H_{\mathcal W})\).
\end{corollary}

\begin{proof}
Fix a verified triple.  A computable semantic normalizer for the translated
fragment would contradict the normalizer obstruction in \Cref{thm:separation}.
The hypothesis ``verified'' is load-bearing: \Cref{thm:separation} first requires
the three real-domain identities for \(\pi\), sine, and nonnegative square root.
Thus the corollary is a universal implication over verified triples.  The printed
names instantiate that implication only when \((H_{\mathcal W})\) supplies the
same three identities, exactly as in
\Cref{thm:printed-root-richardson-conditional-clause}.
\end{proof}

\begin{proposition}[Protocol compilation boundary]
\label{prop:L1-compiles-L0}
Let \(\mathcal E\) be an effective expression grammar with semantic equivalence
relation \(\equiv\).  Suppose an \(L1\) protocol has computable normal forms with
decidable equality.  If there is a computable semantic compiler
\(C\colon\mathcal E\to N\) into those normal forms such that
\[
  E\equiv E' \quad\Longleftrightarrow\quad C(E)=C(E')
\]
for all source expressions in the fragment under consideration, then semantic
equivalence in \(\mathcal E\) is decidable.  Consequently no Richardson-undecidable
source fragment admits such a compiler into the Zeckendorf \(L1\) protocol.
\end{proposition}

\begin{proof}
Given two source expressions \(E,E'\), compute \(C(E)
\) and \(C(E')\).  Since equality of target normal forms is decidable in an
\(L1\) protocol, decide whether \(C(E)=C(E')\).  The displayed reflection and
preservation condition makes this decision equivalent to deciding
\(E\equiv E'\).  If the source fragment is Richardson-undecidable, this would
contradict \Cref{lem:richardson-fragment} after the effective transported
translation of \Cref{lem:richardson-translation}.  The Zeckendorf target has
decidable normal-form equality by \Cref{prop:fold-confluence} and is an \(L1\)
protocol by \Cref{thm:zeckendorf-protocol-univ}, so it is covered by the general
argument.
\end{proof}

\begin{corollary}[Compilation scope]
\label{cor:compile-scope}
The Zeckendorf \(L1\) protocol is a computable protocol for natural-number
arithmetic after value transport, but it is not a computable semantic compiler
or normalizer for an \(L0\) expression grammar whose source equivalence is
Richardson-undecidable.
\end{corollary}

\begin{proof}
The positive protocol scope is \Cref{thm:zeckendorf-protocol-univ}.  The excluded
compiler scope is exactly \Cref{prop:L1-compiles-L0}: a compiler reflecting
semantic equivalence into decidable Zeckendorf normal-form equality would decide
the Richardson-undecidable source equivalence relation.
\end{proof}

\begin{remark}[Transported Richardson boundary]
The Richardson obstruction concerns transported real semantics on the verified
Richardson fragment only.  It is not a decidability statement for arbitrary raw
EML terms on complex principal-branch domains.
Every equality used in \Cref{lem:richardson-translation} is an equality of
partial real functions on the domains of the source Richardson expressions.
The proof never constructs or compares arbitrary EML terms outside that
translated fragment.
\end{remark}

\begin{proposition}[External universality versus verified obstruction]
\label{prop:external-versus-locked-package}
\label{thm:no-hidden-L0-semantic-normalizer}
The Odrzywo{\l}ek \(L0\) import and the Richardson obstruction have different
logical roles: the former supplies expression representatives for a calculator
basis, while the latter requires the explicit three-witness interface of
\Cref{thm:minimal-eml-witness-interface}.
\end{proposition}

\begin{proof}
The \(L0\) import is \Cref{thm:eml-universality}; the obstruction is
\Cref{thm:separation}.  The proof of \Cref{thm:separation} cites only the core
EML witnesses, the three-witness interface, and Richardson's theorem.
\end{proof}

\subsection{Finite-fibre transfer strictness}

\begin{proposition}[Finite-fibre certificates and transfer obstruction]
\label{prop:L2-generic-vs-structural}
Finite-fibre certificates are fixed-resolution finite-atomic data.  A
finite-state transfer formula in the resolution variable is extra structure.
\end{proposition}

\begin{proof}
This is \Cref{prop:L2-novelty-finite-state-separation}.
\end{proof}

\begin{theorem}[Finite-state transfer is the \texorpdfstring{\(L2\)}{L2} novelty]
\label{thm:L2-finite-state-novelty-spine}
The non-generic \(L2\) strictness claim in this article is the cover-relative
separation, inside the computable finite-fibre cover category over the fixed
Zeckendorf normal forms, between finite-fibre certificates and finite-state
transfer.  It is not Hankel positivity, Prony reconstruction, or an intrinsic
claim about the original \(\Fold_m\) fibre multiplicities.
\end{theorem}

\begin{proof}
Hankel positivity and Prony reconstruction hold for all finite fibres by
\Cref{thm:hankel-psd,thm:prony}.  Finite-state transfer can fail by
\Cref{thm:finite-state-strictness}.
\end{proof}

\begin{definition}[Computable finite-fibre cover]
\label{def:computable-cover}
A computable finite-fibre cover of the Zeckendorf resolution data is auxiliary
finite certificate data over the fixed normal-form set \(X_m\): over each normal
form one attaches a computably specified nonempty finite set, without changing
\(X_m\), the original maps \(\Fold_m\), or any readout on normal forms.
\end{definition}

\begin{lemma}[Covers preserve readouts]
\label{lem:cover-preserves-protocol}
Computable finite-fibre covers preserve the Zeckendorf normal forms and the
compiled \(L1\) readouts.
\end{lemma}

\begin{proof}
The cover changes only the number of certificate points over a fixed normal
form.  Projection to the normal form is unchanged, so every operation and
readout defined on normal forms is unchanged.
\end{proof}

\begin{theorem}[Computable covers are computable fibre-size data]
\label{thm:finite-fiber-cover-realization}
Specifying a computable finite-fibre cover is equivalent to specifying a
computable positive integer fibre size over each normal form at each resolution.
\end{theorem}

\begin{proof}
Given a cover, count the finite certificate fibres effectively.  Given
computable positive sizes, attach the auxiliary certificate fibre
\(\{1,\ldots,d\}\) over the corresponding normal form.  These
constructions are inverse up to relabelling of certificate points.
\end{proof}

\begin{theorem}[Cover skeleton used for \texorpdfstring{\(L2\)}{L2} strictness]
\label{thm:L2-cover-skeleton-strictness}
Fix the Zeckendorf normal-form system \((X_m,\Fold_m)\).  In the class of
computable finite-fibre covers of \Cref{def:computable-cover}, the
finite-state-transfer obstruction proved in this article is witnessed entirely
by the computable fibre-size function over the fixed sets \(X_m\).  More
precisely, the cover-relative strictness theorem proved here follows from the
following two-part skeleton:
\begin{enumerate}
\item finite-state transfer sequences are exponentially bounded in the
  resolution parameter; and
\item a computable cover may enlarge a single certificate fibre over a fixed
  normal form at resolution \(m\) to any prescribed computable size \(a_m\).
\end{enumerate}
No assertion about the intrinsic fibre multiplicities of the original maps
\(\Fold_m\) is used or implied by this skeleton.
\end{theorem}

\begin{proof}
The first part is \Cref{prop:finite-state-coefficient-budget}: any fixed
finite-state transfer formula gives a sequence bounded by \(C\lambda^m\) for
some constants \(C,\lambda\).  For the second part, use
\Cref{thm:finite-fiber-cover-realization}.  Choose one normal form in each
nonempty \(X_m\) and assign its cover fibre the computable size \(a_m\), assigning
size \(1\) to all other cover fibres.  This is a computable finite-fibre cover,
and by \Cref{lem:cover-preserves-protocol} the projection to \(X_m\), the normal
forms, and all \(L1\) readouts are unchanged.  Its second collision moment is
therefore altered only by the chosen fibre-size data; taking a superexponential
computable sequence such as \(a_m=m!\) violates the exponential budget and gives
\Cref{thm:finite-state-strictness}.  Since the construction changes only cover
certificate multiplicities above already fixed normal forms, it makes no claim
about the unmodified fibres of \(\Fold_m\).
\end{proof}

\begin{theorem}[Main-article boundary for finite-fibre refinements]
\label{thm:L2-main-article-boundary}
Only computable finite-fibre cover data that leave normal forms fixed are used
in this article's \(L2\) strictness theorem.  Any intrinsic asymptotic theory of
the unmodified Zeckendorf \(\Fold_m\) fibre multiplicities is outside the theorem
chain proved here; the article proves no recurrence, nonrecurrence, rationality,
or finite-state-transfer classification for those intrinsic multiplicities.
\end{theorem}

\begin{proof}
All results here use \Cref{def:computable-cover}, whose projection to
\(X_m\) is fixed.  The cover-size function is independent finite certificate data
over the already chosen normal forms, while the intrinsic multiplicities are the
fibre sizes of the unmodified maps \(\Fold_m\).  By
\Cref{thm:L2-cover-intrinsic-nonimplication}, a transfer statement for the cover
moment has no consequence for the intrinsic fold moment without an additional
comparison theorem.  No proof in this article supplies such a comparison or uses
an asymptotic formula for intrinsic Zeckendorf fibres.
\end{proof}

\begin{theorem}[Computable-cover-only boundary of finite-state obstruction]
\label{prop:cover-only-strictness-boundary}
\label{thm:one-spike-finite-state-boundary}
The one-spike construction is sharp for the computable finite-fibre cover model:
it preserves all normal-form readouts and changes only one added finite
certificate fibre at each resolution, yet it can realize any computable
second-moment spike of the form \((a_m+1)^2+F_{m+2}-1\).  The theorem is silent
about intrinsic \(\Fold_m\) fibre multiplicities.
\end{theorem}

\begin{proof}
This is the construction in \Cref{thm:L2-finite-fiber-no-transfer-content}.
Only one fibre is enlarged, and \Cref{lem:cover-preserves-protocol} proves that
readouts are preserved.
\end{proof}

\begin{theorem}[Intrinsic fold asymptotics are not inferred]
\label{thm:intrinsic-fold-asymptotics-split}
The factorial one-spike obstruction does not assert an intrinsic asymptotic law
for the original Zeckendorf fibres.  It is a statement about computable
finite-fibre covers, so no recurrence, nonrecurrence, rational generating series,
growth classification, or finite-state-transfer classification for intrinsic
\(\Fold_m\) fibre multiplicities follows from \Cref{thm:finite-state-strictness}.
\end{theorem}

\begin{proof}
The proof of \Cref{thm:L2-finite-fiber-no-transfer-content} deliberately alters
cover multiplicities.  It leaves the original fold map untouched.  The formal
nonimplication is \Cref{thm:L2-cover-intrinsic-nonimplication}.
\end{proof}


\begin{theorem}[Cover-relative closure of the \texorpdfstring{\(L2\)}{L2} strictness clause]
\label{thm:L2-cover-relative-closure}
The \(L2\) strictness clause used by \Cref{thm:main-hierarchy} is fully proved
inside the computable finite-fibre cover category of \Cref{def:computable-cover}.
It neither requires nor implies any recurrence, nonrecurrence, rationality,
growth classification, or finite-state-transfer classification for the intrinsic
multiplicity sequence of the original maps \(\Fold_m\).
\end{theorem}

\begin{proof}
The positive finite-fibre part of \(L2\) is fixed-resolution certificate data:
\Cref{thm:hankel-psd,thm:prony,prop:zeckendorf-cert-univ} give Hankel positivity,
log-convexity, and finite Prony reconstruction from finite fibres at each fixed
resolution.  A finite-state transfer law is additional cross-resolution
structure.  By \Cref{prop:finite-state-coefficient-budget}, every fixed
finite-state transfer sequence has an exponential bound in the resolution
parameter.

Now work in the cover category.  By \Cref{thm:finite-fiber-cover-realization}, a
computable cover is exactly a computable positive integer fibre-size function
over the unchanged normal-form sets \(X_m\), and by
\Cref{lem:cover-preserves-protocol} such a cover preserves the Zeckendorf normal
forms and all \(L1\) readouts.  The one-spike skeleton of
\Cref{thm:L2-cover-skeleton-strictness,thm:L2-one-spike-skeleton} chooses one
normal form at each resolution and assigns it a computable size \(a_m\), assigning
size \(1\) elsewhere.  Taking, for example, \(a_m=m!\) gives a computable cover
whose second collision moment violates every exponential bound, and therefore
cannot come from a fixed finite-state transfer formula.  This proves the
cover-relative strictness theorem.

The proof has not used the intrinsic fibre sizes of the original maps
\(\Fold_m\).  Indeed \Cref{thm:L2-cover-intrinsic-nonimplication} records the
formal nonimplication: altering auxiliary cover multiplicities over fixed normal
forms gives no comparison theorem for unmodified fold fibres.  Hence the main
article's \(L2\) strictness is exactly cover-relative, and all intrinsic
\(\Fold_m\) asymptotic classifications listed in the statement are outside the
proved theorem chain.
\end{proof}

\begin{theorem}[Cover-relative finite-fibre certificates do not force finite-state transfer]
\label{thm:finite-state-strictness}
There is a computable finite-fibre cover of the Zeckendorf \(L1\) protocol whose
finite-fibre certificates exist at every resolution but whose second collision
moment over the fixed normal-form sets \(X_m\) has no fixed finite-state transfer
formula.  This is a cover-relative strictness theorem in the computable
finite-fibre cover category; it is not an intrinsic finite-state failure theorem
for the original \(\Fold_m\) fibre multiplicities.
\end{theorem}

\begin{proof}
Apply the cover skeleton \Cref{thm:L2-cover-skeleton-strictness} with
\(a_m=m!\), equivalently apply \Cref{thm:L2-finite-fiber-no-transfer-content} to
that one-spike sequence.  The cover is computable and preserves readouts by
\Cref{lem:cover-preserves-protocol}.  Its second moment grows faster than every
exponential, whereas every finite-state transfer sequence is exponentially
bounded by \Cref{prop:finite-state-coefficient-budget}.  The obstruction is
therefore cover-relative: it concerns the added finite certificate fibres over
the unchanged normal-form sets, not intrinsic multiplicities of the original
\(\Fold_m\) fibres.
\end{proof}

\begin{proposition}[Finite prefixes do not certify transfer]
\label{prop:finite-prefix-no-transfer-certification}
In the computable finite-fibre cover category, no bounded initial window of
finite-fibre certificate data can certify finite-state transfer for all later
resolutions.
\end{proposition}

\begin{proof}
Given any bound \(M\), choose a one-spike cover with \(a_m=1\) for
\(m\le M\) and \(a_m=m!\) for \(m>M\).  It agrees with the unspiked data on the
initial window but violates the finite-state exponential budget later.
\end{proof}

\begin{theorem}[Budget spine used for \texorpdfstring{\(L2\)}{L2} strictness]
\label{thm:L2-budget-spine}
The cover-relative \(L2\) strictness obstruction proved in this article is a
growth-budget obstruction: finite-state transfer sequences obey the exponential
bound of \Cref{prop:finite-state-coefficient-budget}, while computable
finite-fibre covers can violate it, and the superexponential cover obstructions
used here have the one-spike skeleton of
\Cref{thm:L2-cover-skeleton-strictness,thm:L2-one-spike-skeleton}.
\end{theorem}

\begin{proof}
The budget is \Cref{prop:finite-state-coefficient-budget}; the violating cover
is \Cref{thm:finite-state-strictness}.  The one-spike bad-example skeleton used
for the cover-relative strictness theorem is \Cref{thm:L2-one-spike-skeleton}.
\end{proof}

\begin{corollary}[Finite-state growth budget]
\label{cor:finite-state-growth-budget}
Every finite-state transfer formula gives at most exponential growth in the
resolution parameter.
\end{corollary}

\begin{proof}
This is the exponential-bound clause of \Cref{prop:finite-state-coefficient-budget}.
\end{proof}
