\section{Strictness of the hierarchy}\label{sec:strictness}

This section proves the two strictness mechanisms used by
\Cref{thm:main-hierarchy}.  The Richardson part is stated with a verified
finite EML triple as its load-bearing hypothesis.  The printed roots
\(U_{47},U_{111},U_{126}\) are treated only as source-locked strings; no
unconditional theorem below instantiates the obstruction with them.

\subsection{The Richardson obstruction}

Let \(\mathcal R\) be Richardson's real elementary language from
\cite[pp.~514, 520, Theorem~2]{Richardson1968}: one real variable, rational
constants, \(\log2\), \(\pi\), \(+,-,\times\), composition, \(\exp\), \(\sin\),
and a unary symbol \(\mu\) satisfying \(\mu(x)=|x|\) for \(x\ne0\).  We use the
concrete representative \(\mu_0(x)=\sqrt{x^2}\).

\begin{lemma}[Richardson absolute-value interface]
\label{lem:richardson-punctured-mu-reduction}
Richardson's theorem applies to the concrete total completion
\(\mu_0(x)=\sqrt{x^2}\).  Hence any exact EML translation of rational constants,
\(\log2\), \(\pi\), \(+,-,\times\), \(\exp\), \(\sin\), and
\(x\mapsto\sqrt{x^2}\) carries Richardson's identity-to-zero problem into the
translated EML fragment.
\end{lemma}

\begin{proof}
Richardson assumes only that \(\mu(x)=|x|\) for \(x\ne0\)
\cite[pp.~514, 520]{Richardson1968}.  The function \(\mu_0(x)=\sqrt{x^2}\)
satisfies this condition.  Therefore his undecidability theorem applies to the
language with this concrete completion.  Replacing each \(\mu(F)\) by the EML
translation of \(\sqrt{F^2}\) is an effective syntactic reduction once the
listed primitive translations are exact on their stated real domains.
\end{proof}

\begin{lemma}[Richardson identity-to-zero theorem]
\label{lem:richardson-fragment}
The identity-to-zero problem for \(\mathcal R\) with \(\mu=
\mu_0\) is undecidable.
\end{lemma}

\begin{proof}
This is Richardson's Theorem~2, instantiated by
\Cref{lem:richardson-punctured-mu-reduction}; see
\cite[Theorem~2, p.~520]{Richardson1968}.
\end{proof}

\begin{theorem}[Abstract normalizer obstruction]
\label{thm:abstract-normalizer-obstruction}
Let \(\tau\colon\mathcal R\to\mathcal T\) be an effective translation into a
computable term class whose real semantics preserve domains and values:
\(\dom\llbracket\tau(E)\rrbracket=\dom(E)\) and
\(\llbracket\tau(E)\rrbracket(x)=E(x)
\) on that domain.  Then \(\tau(\mathcal R)\) admits no total computable
normalizer into finite objects with decidable equality that identifies exactly
equal transported real functions.
\end{theorem}

\begin{proof}
If such a normalizer existed, compute \(\tau(E)\) and compare its normal form
with that of \(\tau(0)\).  The comparison would decide whether \(E\) is zero on
its Richardson domain, contradicting \Cref{lem:richardson-fragment}.
\end{proof}

\begin{lemma}[Transported zero-test obstruction]
\label{lem:transported-partial-normalizer-obstruction}
The conclusion of \Cref{thm:abstract-normalizer-obstruction} holds for every
domain-preserving, value-preserving EML translation of \(\mathcal R\).
\end{lemma}

\begin{proof}
Apply \Cref{thm:abstract-normalizer-obstruction} with \(\mathcal T\) equal to the
translated EML fragment.
\end{proof}

\begin{lemma}[Printed EML core witnesses]
\label{lem:eml-printed-core}
In the principal-branch EML semantics of \Cref{def:eml}, the following pure EML
macros represent the indicated real functions:
\[
  E(u)=\eml(u,1)=e^u,
  \qquad
  L(u)=\eml\bigl(1,E(\eml(1,u))\bigr)=\log u\quad(u>0),
\]
\[
  D(a,b)=\eml(L(a),E(b))=a-b\quad(a>0,
b\in\RR),
\]
\[
  P(u)=\eml(u,E(u))=e^u-u>0,
  \quad
  N(u)=D(P(u),E(u))=-u,
\]
\[
  A(u,v)=D(E(u),D(P(u),v))=u+v,
  \qquad S(u,v)=A(u,N(v))=u-v .
\]
\end{lemma}

\begin{proof}
The EML semantics are \(\eml(x,y)=e^x-\operatorname{Log}y\) when the principal
logarithm is defined.  Since the second input of \(E(u)\) is \(1\),
\(E(u)=e^u\).  If \(u>0\), then \(\eml(1,u)=e-\log u\) and \(E(\eml(1,u))=e^e/u>0\), so
\(L(u)=e-\log(e^e/u)=\log u\).  Thus
\(D(a,b)=\exp(\log a)-\log(e^b)=a-b\) for \(a>0\).  The function
\(P(u)=e^u-u\) has minimum \(1\) at \(u=0\), hence is positive.  Substituting in
\(D\) gives \(N(u)=-u\), then \(A(u,v)=u+v\), and finally \(S(u,v)=u-v\).
\end{proof}

\begin{theorem}[Self-contained core EML witness package]
\label{lem:eml-real-witness-library}
The macros of \Cref{lem:eml-printed-core} extend internally to finite pure EML
terms representing multiplication, division by a nonzero real, rational
constants, \(\log2\), and the positive square root:
\[
\begin{aligned}
  T(u,v)&=E(A(L(u),L(v)))=uv\quad(u,v>0),\\
  R_+(u)&=E(N(L(u)))=1/u\quad(u>0),
\end{aligned}
\]
\[
  Q(u)=E(u)>0,
  \qquad P(u)=E(u)-u>0,
\]
\[
\begin{aligned}
  M(u,v)=S\Bigl(&A\bigl(T(Q(u),Q(v)),T(P(u),P(v))\bigr),\\
  &A\bigl(T(Q(u),P(v)),T(P(u),Q(v))\bigr)\Bigr)=uv,
\end{aligned}
\]
for all real \(u,v\).  Indeed all four inputs to \(T\) in this displayed
definition are positive.  With this expanded pure-EML multiplication term,
\[
  J(u,v)=M\bigl(u,M(v,R_+(M(v,v)))\bigr)=u/v\quad(v\ne0),
\]
and
\[
  G(u)=E(J(L(u),A(1,1)))=\sqrt u\quad(u>0).
\]
The value at \(0\) for the nonnegative square root, the constant \(\pi\), and
\(\sin\) on \(\RR\) are not supplied by this core package.
\end{theorem}

\begin{proof}
For positive inputs, \(T(u,v)=\exp(\log u+\log v)=uv\) and
\(R_+(u)=\exp(-\log u)=1/u\).  Since \(Q(u)>0\) and \(P(u)>0\), all four calls
to \(T\) in the definition of \(M\) are made on its positive domain.  Using
\(Q(u)-P(u)=u\), the displayed term gives
\[
  M(u,v)=Q(u)Q(v)+P(u)P(v)-Q(u)P(v)-P(u)Q(v)=uv .
\]
If \(v\ne0\), then \(M(v,v)=v^2>0\), so
\(J(u,v)=u\,v/v^2=u/v\).  Finally
\(G(u)=\exp((\log u)/2)=\sqrt u\) for \(u>0\).  Rational constants are obtained
from \(1\) by repeated addition, negation, and division by nonzero integers;
\(\log2=L(A(1,1))\).
\end{proof}

\begin{theorem}[Positive-core versus nonnegative Richardson root]
\label{thm:positive-core-nonnegative-root-boundary}
The self-contained core EML package supplies a square-root witness only on the
positive domain \((0,\infty)\).  A domain-preserving Richardson translation of
the concrete absolute-value constructor
\(\mu_0(x)=\sqrt{x^2}\) on all real inputs is obtained from the core package
exactly after adding a supplied finite witness \(\mathsf R\) satisfying
\[
  \mathsf R(t)=\sqrt t\qquad(t\in\RR_{\ge0}).
\]
In particular, the positive witness \(G\) of
\Cref{lem:eml-real-witness-library} is not a replacement for the
nonnegative-square-root member of the Richardson triple.
\end{theorem}

\begin{proof}
The proof of \Cref{lem:eml-real-witness-library} constructs
\(G(u)=\sqrt u\) by applying \(L(u)=\log u\).  The term \(L(u)\) is defined only
for \(u>0\), hence the displayed construction proves no semantic value for
\(G(0)\).  For the Richardson absolute-value constructor the translated input to
the square root is \(M(u,u)=u^2\).  This is positive when \(u\ne0\) and is equal
to \(0\) when \(u=0\).  Thus substituting the positive witness \(G\) would leave
the translated term undefined at the real input \(u=0\), so it would not preserve
the domain of the total completion \(\mu_0\).

If a finite witness \(\mathsf R\) is supplied with
\(\mathsf R(t)=\sqrt t\) on all \(t\ge0\), then
\(\mathsf R(M(u,u))=\sqrt{u^2}=|u|\) for every real \(u\).  This is precisely the
zero-domain step used in \Cref{lem:eml-zero-absolute-witness}.  Hence the
positive core and the nonnegative Richardson triple member have the stated
separate roles.
\end{proof}

\begin{lemma}[Source-locked printed strings]
\label{lem:eml-ancillary-certificate}
\label{lem:eml-source-level-check}
\label{lem:eml-expanded-witness-strings}
The symbols \(U_{47},U_{111},U_{126}\) denote fixed finite pure EML strings from
the cited arXiv v2 ancillary record in \Cref{ass:HW}.  This syntactic lock does
not prove their real-domain identities.
\end{lemma}

\begin{proof}
The strings are specified by their public versioned table locations in
\Cref{ass:HW}.  A finite string specification proves only syntactic identity of
the strings so named.  Its semantic content is exactly the separate analytic
assertion \((H_{\mathcal W})\).
\end{proof}

\begin{proposition}[Printed witness certificate boundary]
\label{prop:printed-witness-certificate-boundary}
\label{prop:printed-table-needs-analytic-interface}
\label{prop:printed-root-conditional-only}
\label{prop:printed-root-no-hidden-oracle}
The finite table positions and string locks for \(U_{47},U_{111},U_{126}\) are
not a proof that these strings denote \(\pi\), sine, and nonnegative square root.
Thus these names do not supply a Richardson witness triple in this article.
\end{proposition}

\begin{proof}
A finite EML string is syntax.  The Richardson translation requires the three
real-domain semantic identities displayed in \Cref{ass:HW}.  Therefore table
membership can identify the strings but cannot replace the analytic identities.
Since the article does not prove those identities, the printed names remain
outside the verified Richardson interface used below.
\end{proof}

\begin{theorem}[No printed-string witness interface]
\label{thm:printed-richardson-witnesses-interface}
The source-locked printed roots
\((W_{\pi},W_{\sin},W_{\sqrt{}})=(U_{47},U_{111},U_{126})\) are not a finite EML
witness triple in the theorem stream of this article.  They would become such a
triple only after an external analytic proof of the three real-domain identities
recorded in \Cref{ass:HW}; that proof is not supplied here.
\end{theorem}

\begin{proof}
\Cref{lem:eml-ancillary-certificate} locks only finite syntax, and
\Cref{prop:printed-witness-certificate-boundary} proves that finite syntax is not
a semantic verification.  The identities required by the Richardson translation
are exactly the three identities displayed in \Cref{ass:HW}.  Since they are not
proved in this article, the printed strings cannot be used as the verified
triple in \Cref{lem:richardson-translation} or
\Cref{thm:replacement-triple-richardson-interface}.
\end{proof}

\begin{theorem}[Printed roots are non-load-bearing]
\label{thm:printed-roots-nonloadbearing}
The Richardson obstruction instantiated in this paper is only the
verified-interface theorem \Cref{thm:replacement-triple-richardson-interface}:
it applies after a finite replacement triple
\((\mathsf P,\mathsf S,\mathsf R)\) and the three semantic identities have already
been supplied.  It does not use the three semantic identities asserted by
  the external analytic assertion recorded in \Cref{ass:HW}.  More precisely,
  remove
\Cref{ass:HW,lem:eml-ancillary-certificate,prop:printed-witness-certificate-boundary,thm:printed-richardson-witnesses-interface}
from the proof graph.  The interface theorem still proves the normalizer
obstruction for every separately verified finite replacement triple, but it does
  not produce a triple for \(\pi\) and sine.  Consequently the named-root route is
absent from this article.
\end{theorem}

\begin{proof}
The proof of \Cref{lem:richardson-translation} has as non-core hypotheses only
the three real-domain identities for the specified replacement triple:
\(\mathsf P=\pi\), \(\mathsf S(x)=\sin x\) on \(\RR\), and
\(\mathsf R(t)=\sqrt t\) on \(\RR_{\ge0}\).  It does not cite the printed strings.
Combining that translation with
\Cref{lem:transported-partial-normalizer-obstruction} gives
\Cref{thm:replacement-triple-richardson-interface}, and the proof of
\Cref{thm:main-hierarchy} cites that verified-triple interface for the
Richardson clause.  Thus the proof path contains no occurrence of
\((H_{\mathcal W})\) or the named roots.  The printed roots are discussed only in
the exclusion \Cref{thm:printed-richardson-witnesses-interface}.  Hence the
named-root route is absent from the unconditional proof stream.
\end{proof}

\begin{theorem}[Exact Richardson proof-graph cut]
\label{thm:richardson-interface-proof-graph-cut}
Let the unconditional proof graph of this article be obtained by deleting the
optional hypothesis \((H_{\mathcal W})\) and every edge whose only purpose is to
turn the names \(U_{47},U_{111},U_{126}\) into analytic witnesses.  In that graph,
every Richardson obstruction used by \Cref{thm:main-hierarchy,thm:obstruction-spine}
factors through the following interface and through no named-root instantiation:
a finite pure EML triple \((\mathsf P,\mathsf S,\mathsf R)\) together with proofs
of
\[
  \mathsf P=\pi,
  \qquad \mathsf S(x)=\sin x\ (x\in\RR),
  \qquad \mathsf R(t)=\sqrt t\ (t\in\RR_{\ge0}).
\]
Consequently the unconditional manuscript proves the Richardson normalizer
obstruction for each already verified replacement triple, but it proves no
instantiated theorem asserting that \((U_{47},U_{111},U_{126})\) is such a triple.
The printed names can be substituted into this interface only after an external
analytic verification of those three identities, which is not part of this
article.
\end{theorem}

\begin{proof}
The construction of the transported Richardson fragment is
\Cref{lem:richardson-translation}.  Its structural recursion uses the internal
core witnesses of \Cref{lem:eml-real-witness-library} for rational constants,
\(\log2\), arithmetic, composition, and \(\exp\), and it uses an external triple
only at the three displayed clauses for \(\pi\), sine, and nonnegative square
root.  The absolute-value step is then forced by
\Cref{lem:eml-zero-absolute-witness}.  Hence the translation, and therefore the
zero-test reduction and normalizer obstruction of
\Cref{thm:replacement-triple-richardson-interface}, require exactly the verified
triple interface displayed in the statement.

The printed names enter only through \Cref{ass:HW} and the syntactic lock
\Cref{lem:eml-ancillary-certificate}.  By
\Cref{prop:printed-witness-certificate-boundary}, the syntactic lock is not a
proof of the three analytic identities.  Removing \((H_{\mathcal W})\) therefore
removes the only edge that identifies the printed strings with the verified
triple interface.  The remaining proof graph still contains
\Cref{thm:replacement-triple-richardson-interface}, so every already verified
replacement triple gives the obstruction, but no conclusion about the particular
triple \((U_{47},U_{111},U_{126})\) follows.  The missing edge is precisely an
analytic verification of the three identities for the printed strings; by
\Cref{thm:printed-richardson-witnesses-interface}, that edge is not supplied in
this article.
\end{proof}

\begin{corollary}[Verified-interface criterion]
\label{cor:verified-richardson-interface-criterion}
A Richardson obstruction instance belongs to the verified-interface branch of
\Cref{thm:main-hierarchy} exactly when its finite pure EML triple has already
been accompanied by proofs of the three real-domain identities for \(\pi\), sine
on \(\RR\), and the nonnegative square root on \(\RR_{\ge0}\).  The source-locked
triple \((U_{47},U_{111},U_{126})\) does not satisfy this criterion in this
paper.
\end{corollary}

\begin{proof}
If the three identities have been proved for a finite pure EML triple, then
\Cref{thm:replacement-triple-richardson-interface} applies and supplies the
verified-interface branch.  Conversely, the translation of
\Cref{lem:richardson-translation} needs witnesses for the source constructors
\(\pi\), \(\sin\), and \(x\mapsto\sqrt{x^2}\), while
\Cref{prop:printed-witness-certificate-boundary} says that the printed table
alone is not such a semantic interface.  For \((U_{47},U_{111},U_{126})\),
\Cref{thm:printed-richardson-witnesses-interface} records that this article
provides no such interface.  Hence the named triple is not an instance in this
manuscript.
\end{proof}

\begin{proposition}[Locked three-witness interface]
\label{prop:locked-three-witness-interface}
\label{prop:three-witness-stability}
\label{prop:eml-certificate-reproducibility}
The Richardson translation uses only the internal core witnesses and one finite
triple \((\mathsf P,\mathsf S,\mathsf R)\) satisfying the three real-domain
identities for \(\pi\), sine, and nonnegative square root.
\end{proposition}

\begin{proof}
The construction in \Cref{lem:richardson-translation} translates each source
constructor by the corresponding core witness, except for \(\pi\), \(\sin\), and
\(\sqrt{\,}\), where it uses \((\mathsf P,\mathsf S,\mathsf R)\).  No other EML
identity is referenced.
\end{proof}

\begin{theorem}[Richardson proof graph has no hidden witness]
\label{thm:richardson-proof-graph-no-hidden-witness}
\label{thm:verified-triple-is-necessary-interface}
Fix the internal core EML library of \Cref{prop:internal-core-eml-witnesses} and
the imported \(L0\) representative theorem \Cref{thm:eml-universality}.  In this
article these data alone do not instantiate Richardson's EML normalizer
obstruction.  Any proof of such an instantiated obstruction in the manuscript
must additionally contain one of the following two inputs:
\begin{enumerate}
\item a specified finite pure EML triple
  \((\mathsf P,\mathsf S,\mathsf R)\) together with proved real-domain
  identities
  \[
    \mathsf P=\pi,
    \qquad \mathsf S(x)=\sin x\quad(x\in\RR),
    \qquad \mathsf R(t)=\sqrt t\quad(t\in\RR_{\ge0});
  \]
\item for the particular printed strings
  \((U_{47},U_{111},U_{126})\), an external analytic proof of the identities
  recorded in \Cref{ass:HW}; no such proof is supplied in this article.
\end{enumerate}
Equivalently, after deleting these two inputs from the proof graph, the
remaining unconditional Richardson assertion is exactly the abstract interface
\Cref{thm:replacement-triple-richardson-interface}; no concrete finite pure-EML
triple for \(\pi\), sine on \(\RR\), and nonnegative square root on
\(\RR_{\ge0}\) is produced or used.
\end{theorem}

\begin{proof}
The only place where Richardson's source language is converted to pure EML is
\Cref{lem:richardson-translation}.  Its structural recursion sends the internal
constructors--rational constants, \(\log2\), \(+\), \(-\), \(\times\), and
\(\exp\)--to the core witnesses of
\Cref{lem:eml-printed-core,lem:eml-real-witness-library}.  Those core witnesses
are summarized in \Cref{prop:internal-core-eml-witnesses}.  That proposition
does not include \(\pi\), does not include sine on \(\RR\), and gives square root
only on the positive domain \((0,\infty)\), with no value asserted at \(0\).

Richardson's undecidable fragment, as fixed in
\Cref{lem:richardson-punctured-mu-reduction,lem:richardson-fragment}, also uses
the constructors \(\pi\), \(\sin\), and the absolute-value representative
\(x\mapsto\sqrt{x^2}\).  A domain-preserving translation of the last constructor
requires a square-root witness at the input \(0\), because the subterm
\(F(x)^2\) can vanish on its real domain.  This is precisely the boundary proved
in \Cref{thm:positive-core-nonnegative-root-boundary}: the positive-domain core
root cannot by itself serve as Richardson's nonnegative square-root witness.
Thus the core library and the \(L0\) import leave exactly three missing semantic
interfaces, namely \(\pi\), sine on all real inputs, and square root on
\(\RR_{\ge0}\).

The first listed input supplies those three interfaces directly, and then
\Cref{thm:replacement-triple-richardson-interface} gives the normalizer
obstruction.  The second listed input is precisely what is missing for the
printed names: by \Cref{thm:printed-richardson-witnesses-interface}, this article
does not turn \((U_{47},U_{111},U_{126})\) into such a verified triple.
Conversely, without a verified replacement triple, the translation has no
manuscript-proved witnesses for the three Richardson constructors just
identified.  The external \(\mathcal B_{36}\) theorem cannot fill the gap,
because \Cref{thm:B36-import-exact-scope,prop:eml-import-dependency-cut} restrict
it here to representative existence and explicitly exclude its use as a hidden
verified Richardson package.  Hence deleting the two listed inputs leaves only
the abstract interface theorem and no instantiated concrete triple.
\end{proof}

\begin{theorem}[Finite EML Richardson trichotomy]
\label{thm:minimal-eml-witness-interface}
\label{thm:richardson-interface-boundary}
\label{thm:no-unverified-richardson-instance}
\label{thm:finite-eml-richardson-trichotomy}
\label{prop:richardson-witness-dependency-cut}
\label{prop:no-third-richardson-interface}
The Richardson obstruction for pure EML terms has exactly the following status
in this article.
\begin{enumerate}
\item Every finite pure EML triple \((\mathsf P,\mathsf S,\mathsf R)\) with proved
  real-domain identities
  \[
    \mathsf P=\pi,
    \qquad \mathsf S(x)=\sin x\quad(x\in\RR),
    \qquad \mathsf R(t)=\sqrt t\quad(t\in\RR_{\ge0})
  \]
  gives the Richardson normalizer obstruction of
  \Cref{thm:replacement-triple-richardson-interface}.
\item The named finite strings \((U_{47},U_{111},U_{126})\) do not give such an
  instance in this article.
\item Without a proved replacement triple, this
  manuscript proves no unconditional finite pure-EML triple for \(\pi\), sine on
  \(\RR\), and nonnegative square root on \(\RR_{\ge0}\); the Richardson result is
  then only the interface theorem of the first clause.
\end{enumerate}
Consequently, the Proc. AMS Richardson clause in \Cref{thm:main-hierarchy} is
only relative to a verified triple: the paper proves the interface theorem and
supplies no concrete named-root instance.  The proof graph has no hidden third
source of witnesses, by
\Cref{thm:richardson-proof-graph-no-hidden-witness}.
\end{theorem}

\begin{proof}
The first clause is exactly
\Cref{thm:replacement-triple-richardson-interface}.  For the second clause,
\Cref{thm:printed-richardson-witnesses-interface} says that the source-locked
strings do not form a verified triple in this article, and
\Cref{thm:printed-root-richardson-conditional-clause} keeps them outside the
interface theorem.

It remains only to justify that no third route is being asserted.  Inspect the
translation scheme in \Cref{lem:richardson-translation}.  Richardson's language
contains the source constructors \(\pi\), \(\sin\), and the punctured
absolute-value symbol, represented here as \(x\mapsto\sqrt{x^2}\).  The internal
core package \Cref{lem:eml-printed-core,lem:eml-real-witness-library} supplies
the other constructors but explicitly does not supply \(\pi\), sine on all of
  \(\RR\), or the square-root value at \(0\); this is the domain boundary of
  \Cref{thm:positive-core-nonnegative-root-boundary}.  The sharper dependency
  statement is \Cref{thm:richardson-proof-graph-no-hidden-witness}: without the
  verified replacement-triple input, no other manuscript result supplies these
  three identities.
  Therefore correctness of the transported Richardson fragment requires exactly
  the displayed semantic interface for those three source constructors.  This
  identifies the Richardson clause of \Cref{thm:main-hierarchy} with the
  verified-triple interface and proves the stated Proc. AMS scope restriction.
  The printed table contributes
only fixed syntax by \Cref{lem:eml-ancillary-certificate}, and the imported
\(L0\) representation theorem is only representative existence by
\Cref{thm:eml-universality,prop:eml-import-dependency-cut}.  Hence, absent a
proved replacement triple, the manuscript asserts no finite triple and retains
only the interface theorem.
\end{proof}

\begin{theorem}[Scope-exact Richardson clause]
\label{thm:richardson-scope-exact-main-clause}
Let \(\mathsf{Rich}(\mathsf P,\mathsf S,\mathsf R)\) denote the conclusion of
\Cref{thm:replacement-triple-richardson-interface} for a finite pure EML triple
\((\mathsf P,\mathsf S,\mathsf R)\).  The unconditional Richardson clause used in
\Cref{thm:main-hierarchy} is exactly
\[
  \forall(\mathsf P,\mathsf S,\mathsf R)
  \bigl(\mathsf P=\pi,\ \mathsf S=\sin|_{\RR},\
  \mathsf R=\sqrt{\ }|_{\RR_{\ge0}}\bigr)
  \Longrightarrow \mathsf{Rich}(\mathsf P,\mathsf S,\mathsf R),
\]
where the three displayed identities mean manuscript-proved real-domain
identities for the specified finite strings.  It is not the existential or
printed-name specialization
\(\mathsf{Rich}(U_{47},U_{111},U_{126})\).  Every theorem in the
main proof chain remains true if the three printed names are replaced by any
other fixed finite pure EML strings.
\end{theorem}

\begin{proof}
The quantified implication is precisely
\Cref{thm:replacement-triple-richardson-interface}: its hypotheses are a
specified finite pure EML triple and the three real-domain identities displayed
in the statement, and its conclusion is \(\mathsf{Rich}(\mathsf P,\mathsf S,
\mathsf R)\).  The proof of \Cref{thm:main-hierarchy} cites the Richardson
obstruction through \Cref{thm:separation}, and \Cref{thm:separation} cites
\Cref{thm:replacement-triple-richardson-interface} with exactly the same
verified-triple hypotheses.  Thus the Richardson input to the main hierarchy is
the universal interface implication just displayed.

For the printed strings, \Cref{lem:eml-ancillary-certificate} supplies only fixed
syntax and \Cref{prop:printed-witness-certificate-boundary} says that this
syntax is not a proof of the three real-domain identities.
\Cref{thm:printed-richardson-witnesses-interface} records that this article does
not turn \((U_{47},U_{111},U_{126})\) into a verified triple.  Hence
\(\mathsf{Rich}(U_{47},U_{111},U_{126})\) is not a consequence of the manuscript.

It remains to check that no hidden source of printed-name semantics enters the
main proof chain.  This is the dependency cut proved in

\begin{lemma}[Effective Richardson translation]
\label{lem:richardson-translation}
Let \((\mathsf P,\mathsf S,
\mathsf R)\) be a finite pure EML triple satisfying the three real-domain
identities.  There is an effective translation \(\tau_{\mathsf P,\mathsf S,
\mathsf R}\colon\mathcal R\to\) pure EML terms such that, for every
\(E\in\mathcal R\),
\[
  \dom\llbracket\tau_{\mathsf P,\mathsf S,
  \mathsf R}(E)\rrbracket_{\RR}=\dom(E),
  \qquad
  \llbracket\tau_{\mathsf P,\mathsf S,
  \mathsf R}(E)\rrbracket_{\RR}(x)=E(x)
\]
for all \(x\in\dom(E)\).  Moreover \(|\tau(E)|\le K^{2|E|}\) and
\(\operatorname{depth}\tau(E)\le K|E|\) for a constant \(K\) depending only on
the finite witness library and the rational-constant expansion convention.
\end{lemma}

\begin{proof}
Define \(
\tau\) by structural recursion.  Variables map to variables.  Rational
constants and \(\log2\) use \Cref{lem:eml-real-witness-library}; \(\pi\) uses
\(\mathsf P\).  The constructors \(+,-,\times,\exp\) use the corresponding core
witnesses, \(\sin\) uses \(\mathsf S\), composition is syntactic substitution,
and \(\mu(F)\) is translated as \(\mathsf R(M(\tau(F),\tau(F)))\).  The
translation is effective because all witnesses are finite.

The domain and value assertion is proved by induction on the expression tree.
The primitive cases are exactly the proved real-domain identities in
\Cref{lem:eml-printed-core,lem:eml-real-witness-library} and in the supplied
triple.  The composition step is the definition of partial-function
composition.  The \(\mu\)-step is \Cref{lem:eml-zero-absolute-witness}.  This
proves exact agreement on real domains.

For the size bound, choose \(K\) larger than the size, depth, and input-leaf
multiplicity of every witness in the finite library and large enough for the
chosen rational-constant expansions.  Substituting translated children into a
fixed witness multiplies size by at most a factor bounded by \(K\) and adds at most \(K\) to depth;
composition multiplies sizes and adds depths.  Induction on \(|E|\) gives the
displayed bounds after increasing \(K\) once.
\end{proof}

\begin{proposition}[Quantitative Richardson interface]
\label{prop:richardson-translation-bookkeeping}
The translation of \Cref{lem:richardson-translation} has exponential tree-size
growth and linear substitution-depth growth in the source expression size, with
constants depending only on the finite witness library.
\end{proposition}

\begin{proof}
This is the final paragraph of the proof of \Cref{lem:richardson-translation}.
\end{proof}

\begin{theorem}[Verified-triple Richardson obstruction interface]
\label{thm:replacement-triple-richardson-interface}
Let \((\mathsf P,\mathsf S,\mathsf R)\) be a specified finite pure EML triple
for which the real-domain identities
\[
  \mathsf P=\pi,
  \qquad
  \mathsf S(x)=\sin x\quad(x\in\RR),
  \qquad
  \mathsf R(t)=\sqrt t\quad(t\in\RR_{\ge0})
\]
have been proved, and let
\(\tau=\tau_{\mathsf P,\mathsf S,\mathsf R}\) be the translation of
\Cref{lem:richardson-translation}.  Then the Richardson identity-to-zero problem
many-one reduces to the zero-identity problem on the translated pure EML
fragment \(\tau(\mathcal R)\):
\[
  E\equiv 0\hbox{ on }\dom(E)
  \quad\Longleftrightarrow\quad
  \tau(E)\equiv \tau(0)
  \hbox{ on the transported real domain}.
\]
Consequently the translated fragment admits none of the following computable
structures:
\begin{enumerate}
\item a zero-identity decision procedure;
\item a semantic normalizer into finite objects with decidable equality that
  identifies exactly equal transported real functions;
\item a semantic compiler into any finite decidable protocol object whose
  equality reflects and preserves equality of the transported real functions on
  this fragment.
\end{enumerate}
This is an interface theorem: it is conditional on the displayed finite triple
and its proved identities.  It does not construct \(\mathsf P\),
\(\mathsf S\), or \(\mathsf R\), and it does not use \((H_{\mathcal W})\),
\(U_{47}\), \(U_{111}\), or \(U_{126}\).
\end{theorem}

\begin{proof}
By \Cref{lem:richardson-translation}, the verified triple gives an effective,
domain-preserving, value-preserving translation of \(\mathcal R\) into pure EML
terms.  Hence for every source expression \(E\),
\[
  \dom\llbracket\tau(E)\rrbracket_{\RR}=\dom(E),
  \qquad
  \llbracket\tau(E)\rrbracket_{\RR}(x)=E(x)
  \quad(x\in\dom(E)).
\]
The same identities hold for the constant zero expression.  Therefore
\(E\) is identically zero on its Richardson domain if and only if the two
translated EML terms \(\tau(E)\) and \(\tau(0)\) define the same transported real
partial function.  Since \(\tau\) is effective, this is a many-one reduction from
Richardson's identity-to-zero problem of \Cref{lem:richardson-fragment} to the
zero-identity problem on \(\tau(\mathcal R)\).  The first excluded structure would
therefore decide Richardson's undecidable problem.

If a semantic normalizer into finite objects with decidable equality existed,
then computing the two normal forms of \(\tau(E)\) and \(\tau(0)\) and comparing
them would be a zero-identity decision procedure on \(\tau(\mathcal R)\), again
contradicting \Cref{lem:richardson-fragment}.  This is precisely
\Cref{lem:transported-partial-normalizer-obstruction} applied to the verified
translation.

Finally suppose there were a computable semantic compiler from
\(\tau(\mathcal R)\) into finite decidable protocol objects whose equality
reflects and preserves equality of the transported real functions on the
fragment.  Applying the compiler to \(\tau(E)\) and \(\tau(0)\), then deciding
target equality, would again decide whether \(E\) is identically zero.  Thus the
compiler is impossible as well.  The only non-core analytic inputs used anywhere
in this argument are the three displayed identities for the specified triple, so
the theorem is an interface theorem and contains no named-root instantiation.
\end{proof}

\begin{proposition}[Protocol compilation boundary]
\label{prop:L1-compiles-L0}
Let \(\mathcal E\) be an effective expression grammar with semantic equivalence
relation \(\equiv\).  Suppose an \(L1\) protocol has computable normal forms with
decidable equality.  If there is a computable semantic compiler
\(C\colon\mathcal E\to N\) into those normal forms such that
\[
  E\equiv E' \quad\Longleftrightarrow\quad C(E)=C(E')
\]
for all source expressions in the fragment under consideration, then semantic
equivalence in \(\mathcal E\) is decidable.  Consequently no Richardson-undecidable
source fragment admits such a compiler into the Zeckendorf \(L1\) protocol.
\end{proposition}

\begin{proof}
Given two source expressions \(E,E'\), compute \(C(E)
\) and \(C(E')\).  Since equality of target normal forms is decidable in an
\(L1\) protocol, decide whether \(C(E)=C(E')\).  The displayed reflection and
preservation condition makes this decision equivalent to deciding
\(E\equiv E'\).  If the source fragment is Richardson-undecidable, this would
contradict \Cref{lem:richardson-fragment} after the effective transported
translation of \Cref{lem:richardson-translation}.  The Zeckendorf target has
decidable normal-form equality by \Cref{prop:fold-confluence} and is an \(L1\)
protocol by \Cref{thm:zeckendorf-protocol-univ}, so it is covered by the general
argument.
\end{proof}

\begin{corollary}[Compilation scope]
\label{cor:compile-scope}
The Zeckendorf \(L1\) protocol is a computable protocol for natural-number
arithmetic after value transport, but it is not a computable semantic compiler
or normalizer for an \(L0\) expression grammar whose source equivalence is
Richardson-undecidable.
\end{corollary}

\begin{proof}
The positive protocol scope is \Cref{thm:zeckendorf-protocol-univ}.  The excluded
compiler scope is exactly \Cref{prop:L1-compiles-L0}: a compiler reflecting
semantic equivalence into decidable Zeckendorf normal-form equality would decide
the Richardson-undecidable source equivalence relation.
\end{proof}

Hankel positivity or Prony reconstruction.
\end{theorem}

\begin{proof}
Hankel positivity and Prony reconstruction hold for all finite fibres by
\Cref{thm:hankel-psd,thm:prony}.  Finite-state transfer can fail by
\Cref{thm:finite-state-strictness}.
\end{proof}

\begin{definition}[Computable finite-fibre cover]
\label{def:computable-cover}
A computable finite-fibre cover of the Zeckendorf resolution data is obtained by
replacing each finite fibre of \(\Fold_m\) by a computably specified nonempty
finite set, without changing the normal-form set \(X_m\) or any readout on
normal forms.
\end{definition}

\begin{lemma}[Covers preserve readouts]
\label{lem:cover-preserves-protocol}
Computable finite-fibre covers preserve the Zeckendorf normal forms and the
compiled \(L1\) readouts.
\end{lemma}

\begin{proof}
The cover changes only the number of certificate points over a fixed normal
form.  Projection to the normal form is unchanged, so every operation and
readout defined on normal forms is unchanged.
\end{proof}

\begin{theorem}[Computable covers are computable fibre-size data]
\label{thm:finite-fiber-cover-realization}
Specifying a computable finite-fibre cover is equivalent to specifying a
computable positive integer fibre size over each normal form at each resolution.
\end{theorem}

\begin{proof}
Given a cover, count the finite fibres effectively.  Given computable positive
sizes, replace the fibre over each normal form by \(\{1,
\ldots,d\}\).  These
constructions are inverse up to relabelling of certificate points.
\end{proof}

\begin{theorem}[Main-article boundary for finite-fibre refinements]
\label{thm:L2-main-article-boundary}
Only cover data that leave normal forms fixed are used in this article.  Any
intrinsic asymptotic theory of Zeckendorf fibre multiplicities is outside the
main theorem chain.
\end{theorem}

\begin{proof}
All results here use \Cref{def:computable-cover}, whose projection to
\(X_m\) is fixed.  No proof changes the stable normal-form system or uses an
asymptotic formula for intrinsic Zeckendorf fibres.
\end{proof}

\begin{theorem}[Cover-only boundary of finite-state obstruction]
\label{prop:cover-only-strictness-boundary}
\label{thm:one-spike-finite-state-boundary}
The one-spike construction is sharp for the cover model: it preserves all
normal-form readouts and changes only one finite certificate fibre at each
resolution, yet it can realize any computable second-moment spike of the form
\((a_m+1)^2+F_{m+2}-1\).
\end{theorem}

\begin{proof}
This is the construction in \Cref{thm:L2-finite-fiber-no-transfer-content}.
Only one fibre is enlarged, and \Cref{lem:cover-preserves-protocol} proves that
readouts are preserved.
\end{proof}

\begin{theorem}[Intrinsic fold asymptotics are split]
\label{thm:intrinsic-fold-asymptotics-split}
The factorial one-spike obstruction does not assert an intrinsic asymptotic law
for the original Zeckendorf fibres.  It is a statement about computable
finite-fibre covers.
\end{theorem}

\begin{proof}
The proof of \Cref{thm:L2-finite-fiber-no-transfer-content} deliberately alters
cover multiplicities.  It leaves the original fold map untouched.
\end{proof}

\begin{theorem}[Cover finite-fibre certificates do not force finite-state transfer]
\label{thm:finite-state-strictness}
There is a computable finite-fibre cover of the Zeckendorf \(L1\) protocol whose
finite-fibre certificates exist at every resolution but whose second collision
moment has no fixed finite-state transfer formula.
\end{theorem}

\begin{proof}
Apply \Cref{thm:L2-finite-fiber-no-transfer-content} with \(a_m=m!\).  The cover
is computable and preserves readouts by \Cref{lem:cover-preserves-protocol}.  Its
second moment grows faster than every exponential, whereas every finite-state
transfer sequence is exponentially bounded by
\Cref{prop:finite-state-coefficient-budget}.
\end{proof}

\begin{proposition}[Finite prefixes do not certify transfer]
\label{prop:finite-prefix-no-transfer-certification}
No bounded initial window of finite-fibre certificate data can certify
finite-state transfer for all later resolutions.
\end{proposition}

\begin{proof}
Given any bound \(M\), choose a one-spike cover with \(a_m=1\) for
\(m\le M\) and \(a_m=m!\) for \(m>M\).  It agrees with the unspiked data on the
initial window but violates the finite-state exponential budget later.
\end{proof}

\begin{theorem}[Budget spine for \texorpdfstring{\(L2\)}{L2} strictness]
\label{thm:L2-budget-spine}
The cover-relative \(L2\) strictness obstruction is exactly a growth-budget
obstruction: finite-state transfer sequences obey the exponential bound of
\Cref{prop:finite-state-coefficient-budget}, while computable finite-fibre
covers can violate it, and every superexponential cover obstruction has the
one-spike skeleton of \Cref{thm:L2-one-spike-skeleton}.
\end{theorem}

\begin{proof}
The budget is \Cref{prop:finite-state-coefficient-budget}; the violating cover
is \Cref{thm:finite-state-strictness}.  The reduction from arbitrary
superexponential cover growth to a one-spike bad-example skeleton is
\Cref{thm:L2-one-spike-skeleton}.
\end{proof}

\begin{corollary}[Finite-state growth budget]
\label{cor:finite-state-growth-budget}
Every finite-state transfer formula gives at most exponential growth in the
resolution parameter.
\end{corollary}

\begin{proof}
This is the exponential-bound clause of \Cref{prop:finite-state-coefficient-budget}.
\end{proof}
