\section{Transfer matrices and the second moment}
\label{sec:transfer}

For \(q=0\), \(S_0(m)=F_{m+2}\) and the Fibonacci matrix suffices.  Assume
\(q\ge1\).  The audited interface gives the fixed matrix formula.

\begin{theorem}[Intrinsic fixed-degree transfer]
\label{thm:intrinsic-fold-moment-transfer}
For every fixed \(q\ge0\), there are effectively constructible integer
matrices \(T_q\) and integer vectors \(u_q,v_q\), independent of \(m\), with
\[
 S_q(m)=u_q^{\mathsf T}T_q^m v_q\qquad(m\ge1).
\]
Consequently the ordinary generating function of \(S_q\) is rational and the
sequence satisfies a constant-coefficient recurrence.
\end{theorem}
\begin{proof}
For \(q=0\), take the usual two-state Fibonacci matrix.  For \(q\ge1\), take
\(T_q=M_q\), \(u_q=M_q^{\mathsf T}e_0\), and \(v_q=b_q\) from
\Cref{thm:audited-fixed-degree-transfer-interface}.  Then
\[
 S_q(m)=e_0^{\mathsf T}M_q^{m+1}b_q
       =u_q^{\mathsf T}T_q^m v_q .
\]
The resolvent identity
\[
 \sum_{m\ge1}S_q(m)z^m
 =z\,u_q^{\mathsf T}(I-zT_q)^{-1}v_q
\]
proves rationality, and Cayley--Hamilton gives the recurrence.
\end{proof}

\begin{proposition}[Exponential budget for fixed transfer]
\label{prop:fixed-transfer-interface-budget}
If \(b_m=u^{\mathsf T}A^mv\) for a fixed complex matrix \(A\), then
there are \(C>0\) and \(\rho>0\) such that \(|b_m|\le C\rho^m\).
\end{proposition}
\begin{proof}
Put \(A\) in Jordan form, or use any submultiplicative matrix norm:
\(|b_m|\le\|u\|\,\|A\|^m\|v\|\).
\end{proof}

\begin{theorem}[Primitive intrinsic transfer]
\label{thm:primitive-intrinsic-moment-transfer}
For every fixed \(q\ge1\), let \(\mathcal T_q\) be the states lying on a
positive-weight path from the initial state to an accepting state, and let
\(P_q\) be the induced matrix.  Then \(P_q\) is primitive.  If \(\lambda_q\)
is its Perron root, then
\[
 S_q(m)=c_q\lambda_q^m+O(\theta_q^m),
 \qquad c_q>0,\quad 0\le\theta_q<\lambda_q,
\]
and
\[
 \lim_{m\to\infty}\frac1m\log S_q(m)=\log\lambda_q .
\]
\end{theorem}
\begin{proof}
The positive-weight column alphabet is
\[
 \mathcal A_q=\{0,1\}^{q-1}\cup\{-1,0\}^{q-1}.
\]
If a reachable accepting marker \(\sigma\) had both a \(1\) and a \(-1\)
coordinate, the corresponding word values would differ by \(2F_{n+1}\),
whereas all word values lie in an interval of length
\(F_{n+2}-2<2F_{n+1}\).  Thus every reachable marker lies in
\(\mathcal A_q\).  For a state \(c\in\mathcal T_q\), choose a positive path
from the initial state to \(c\), followed by one to \((\sigma,0)\).  Since
\(-\sigma\in\mathcal A_q\),
\[
 \tau_{-\sigma}(\sigma,0)=(0,0).
\]
Therefore every state of \(\mathcal T_q\) reaches the initial state, while
the initial state reaches every state by definition.  The support graph is
strongly connected.  The zero column has weight \(w_q(0)=2\) and gives a loop
at \((0,0)\), so the period is one.  Hence \(P_q\) is primitive.  Perron--
Frobenius applied to the nonzero restrictions of the initial and accepting
vectors gives the stated asymptotic and pressure.
\end{proof}

\begin{lemma}[Initial second-moment enumeration]
\label{lem:initial-second-moment-enumeration}
For \(C_2(n)\), the number of ordered pairs of length-\(n\) binary words with
equal residue modulo \(F_{n+1}\), one has
\[
 C_2(1)=2,\qquad C_2(2)=6,\qquad C_2(3)=14.
\]
\end{lemma}
\begin{proof}
For \(n=1,2,3\), the residue multiplicity lists are respectively
\((1,1)\), \((2,1,1)\), and \((2,2,1,2,1)\).  Squaring and summing gives the
displayed values.
\end{proof}

\begin{proposition}[Exact fold-pairing self-similarity]
\label{prop:exact-fold-pairing-self-similarity}
For \(m\ge3\), put \(M=F_{m+2}\), let \(R(n)\) count representations of
\(n\) by distinct parts from \(F_1,\ldots,F_{m+1}\), and define
\[
 T_2(m)=\sum_{n\in\mathbb Z}R(n)^2,
 \qquad
 C(m)=\sum_{0\le v<M}R(v)R(v+M).
\]
Then
\begin{equation}\label{eq:cross-term-self-similarity}
 C(m)=S_2(m-2),
\end{equation}
and consequently
\begin{equation}\label{eq:fold-untruncated-self-similarity}
 S_2(m)=T_2(m)+2S_2(m-2).
\end{equation}
\end{proposition}
\begin{proof}
Write
\[
 G=F_m,\qquad H=F_{m+1},\qquad M=G+H,\qquad
 \mathcal L=\{F_1,\ldots,F_{m-1}\},
\]
and write \(\|A\|\) for the sum of the parts in a subset \(A\).
The cross term \(C(m)\) counts ordered pairs \((A,B)\) of subsets of
\(\mathcal L\cup\{G,H\}\) such that
\[
 \|B\|=\|A\|+M,\qquad \|A\|<M.
\]
The largest sum omitting \(H\) is
\(\sum_{i=1}^{m}F_i=M-2\), so \(H\in B\).  Write
\(B=\{H\}\cup B'\).  Then
\(\|B'\|=\|A\|+G\).  Since \(\|B'\|\le M-2\), one has
\(\|A\|\le H-2<H\), so \(H\notin A\).  Thus \(A\) and \(B'\) are subsets of
\(\mathcal L\cup\{G\}\) whose sums differ by \(G\).

On the other hand, \(S_2(m-2)\) counts ordered pairs \((P,Q)\) of subsets of
\(\mathcal L\) whose sums are congruent modulo \(G=F_m\).  Their sums lie in
\([0,H-2]\), and \(H-2<2G\), so their difference is \(0\), \(G\), or \(-G\).
Define \((A,B')\) from \((P,Q)\) by
\[
\begin{array}{c|c|c}
 \|P\|=\|Q\| & A=P & B'=Q\cup\{G\}\\
 \|Q\|=\|P\|+G & A=P & B'=Q\\
 \|P\|=\|Q\|+G & A=Q\cup\{G\} & B'=P\cup\{G\}.
\end{array}
\]
In every case \(\|B'\|=\|A\|+G\), and adjoining \(H\) to \(B'\) produces a
pair counted by \(C(m)\).  This map is bijective.  Indeed, membership of \(G\)
in \((A,B')\) is respectively \((0,1)\), \((0,0)\), or \((1,1)\), and these
three patterns recover the displayed cases.  The remaining pattern \((1,0)\)
would require two subsets of \(\mathcal L\) to have sums differing by \(2G\),
which is impossible because both sums are at most \(H-2<2G\).  This proves
\eqref{eq:cross-term-self-similarity}.

Finally, the total attainable sum from \(F_1,\ldots,F_{m+1}\) is
\(F_{m+3}-2<2M\).  Hence
\[
 \sum_{0\le v<M}\bigl(R(v)^2+R(v+M)^2\bigr)=T_2(m).
\]
Expanding the squared fibre identity of
\Cref{lem:fold-fibre-representation-pairing} and using
\eqref{eq:cross-term-self-similarity} gives
\eqref{eq:fold-untruncated-self-similarity}.
\end{proof}

\begin{theorem}[Sharp second-moment recurrence]
\label{thm:intrinsic-second-moment-recurrence}
Let \(C_2(n)\) be as in the preceding lemma.  For \(n\ge4\),
\[
 C_2(n)=2C_2(n-1)+2C_2(n-2)-2C_2(n-3).
\]
Since \(S_2(m)=C_2(m+1)\),
\[
 S_2(1)=6,\quad S_2(2)=14,\quad S_2(3)=36,
\]
and the same recurrence holds for \(S_2(m)\) for \(m\ge4\).  In particular
\[
 \sum_{m\ge1}S_2(m)z^m
 =\frac{6z+2z^2-4z^3}{1-2z-2z^2+2z^3}.
\]
\end{theorem}
\begin{proof}
For \(q=2\), the signed alphabet is \(\{-1,0,1\}\) with weights
\(1,2,1\).  Let \(W_n(p,r)\) be the total weight of paths of length \(n\),
and set
\[
 A_n=W_n(0,0),\quad B_n=W_n(1,0)=W_n(-1,0),\quad
 D_n=W_n(0,1)=W_n(0,-1).
\]
An induction from the initial state shows that \((1,1)\) and \((-1,-1)\) are
unreachable, so they contribute no additional incoming terms to \(B_{n+1}\).
Then \(C_2(n)=A_n+2B_n\), and the transition rule gives
\[
 A_{n+1}=2A_n+2B_n,\qquad B_{n+1}=D_n,\qquad
 D_{n+1}=A_n+2B_n=C_2(n).
\]
The untruncated moment in
\Cref{prop:exact-fold-pairing-self-similarity} is \(T_2(m)=A_{m+1}\).  Also
\(D_n=A_n-A_{n-1}\), so
\[
 A_{n+2}=2A_{n+1}+2D_n
        =2A_{n+1}+2A_n-2A_{n-1}.
\]
Thus \(T_2\) satisfies the bare cubic recurrence for \(m\ge3\).  Iterating the
same three state equations from \((A_0,B_0,D_0)=(1,0,0)\) gives, in addition
to the initial values in \Cref{lem:initial-second-moment-enumeration},
\[
 S_2(3)=C_2(4)=36,\qquad S_2(4)=88,\qquad S_2(5)=220.
\]

Let
\(\mathcal Q a(m)=a(m)-2a(m-1)-2a(m-2)+2a(m-3)\).  Applying \(\mathcal Q\)
to \eqref{eq:fold-untruncated-self-similarity} gives, for \(m\ge6\),
\[
 \mathcal Q S_2(m)=\mathcal Q T_2(m)+2\mathcal Q S_2(m-2)
                  =2\mathcal Q S_2(m-2).
\]
The displayed values give \(\mathcal Q S_2(4)=\mathcal Q S_2(5)=0\), so
induction on the two parity classes proves the recurrence for every \(m\ge4\).
The case \(n=4\) of the stated \(C_2\) recurrence is the displayed computation
\(C_2(4)=36\); the cases \(n\ge5\) are the just-proved \(S_2\) recurrence with
\(n=m+1\).  The other initial values are supplied by
\Cref{lem:initial-second-moment-enumeration}, and the generating function
follows.
\end{proof}

\begin{corollary}[Polynomial fibre statistics]
\label{cor:polynomial-intrinsic-fiber-statistics}
For every fixed polynomial \(P\in\mathbb Z[t]\), the sequence
\[
 m\longmapsto\sum_{x\in X_m}P(d_m(x))
\]
has a rational ordinary generating function and a constant-coefficient
recurrence.
\end{corollary}
\begin{proof}
Expand \(P\) into finitely many monomials and apply
\Cref{thm:intrinsic-fold-moment-transfer}.
\end{proof}
