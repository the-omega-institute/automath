\section{Visible Quotients and Quantitative Recovery}
\label{sec:visible-quotient}

Let $A$ be a finite abelian group, $\mathfrak{U}$ a finite cylinder-algebra cover, and
$[\alpha]\in\check{H}^2(\mathfrak{U},A)$ a defect class as in
Section~\ref{sec:descent}.  We always regard $A$ as the automorphism band of
the chosen local sections on $\mathfrak{U}$ furnished by
Definition~\ref{def:abelian-band}.

For an abelian group $B$, normalized \v{C}ech cohomology of the cover with
constant coefficients is the simplicial cohomology of its ordered nerve.
The universal coefficient theorem \cite[Theorem~3.2]{HatcherAlgebraicTopology}
therefore gives a natural exact sequence
\begin{equation}\label{eq:uct-visible-quotient}
0\longrightarrow
\operatorname{Ext}^1\!\bigl(H_1(N(\mathfrak{U});\ZZ),B\bigr)
\longrightarrow
\check{H}^2(\mathfrak{U},B)
\xrightarrow{\ \operatorname{ev}\ }
\Hom\!\bigl(H_2(N(\mathfrak{U});\ZZ),B\bigr)
\longrightarrow 0.
\end{equation}
The evaluation term determines the only possible kernel of a universal
quotient.  The next theorem identifies the residual obstruction and thereby
classifies the individual defects for which that quotient exists.

\begin{theorem}[Classification of universal visible quotients]
\label{thm:intrinsic-visible-quotient}
Let $\mathfrak{U}$ be a finite cylinder-algebra cover, let $A$ be a finite
abelian group, and let
$[\alpha]\in\check{H}^2(\mathfrak{U},A)$.  Write
\[
h_\alpha:=\operatorname{ev}([\alpha])\colon
H_2(N(\mathfrak{U});\ZZ)\longrightarrow A,
\qquad
I_\alpha:=\im(h_\alpha),
\]
and let $q_0\colon A\twoheadrightarrow A/I_\alpha$ be the quotient map.
Naturality of evaluation gives
$\operatorname{ev}((q_0)_*[\alpha])=0$, so exactness of
\eqref{eq:uct-visible-quotient} determines a unique class
\begin{equation}\label{eq:residual-extension-class}
\epsilon_\alpha\in
\operatorname{Ext}^1\!\bigl(H_1(N(\mathfrak{U});\ZZ),A/I_\alpha\bigr)
\end{equation}
whose image in $\check{H}^2(\mathfrak{U},A/I_\alpha)$ is
$(q_0)_*[\alpha]$.  The following are equivalent.
\begin{enumerate}[label=\textup{(\arabic*)}]
\item There is a quotient $q_\alpha\colon A\twoheadrightarrow\Avis$ such that
$(q_\alpha)_*[\alpha]=0$ and such that every homomorphism
$\varphi\colon A\to B$ to an abelian group satisfying
$\varphi_*[\alpha]=0$ factors uniquely through $q_\alpha$.
\item $(q_0)_*[\alpha]=0$.
\item $\epsilon_\alpha=0$.
\end{enumerate}
When these conditions hold, $q_0$ is the universal quotient; in particular,
\begin{equation}\label{eq:visible-kernel-cycle-image}
\Avis=A/I_\alpha,
\qquad
\ker(q_\alpha)=I_\alpha.
\end{equation}

Consequently, every finite abelian band and every degree-$2$ defect on
$\mathfrak{U}$ admit a universal killing quotient if and only if
$H_1(N(\mathfrak{U});\ZZ)$ is torsion-free.  If this homology group has a
cyclic torsion summand $\ZZ/n\ZZ$, failure already occurs for
$A=\ZZ/n\ZZ$ and a class with zero evaluation.
\end{theorem}

\begin{proof}
The equivalence of \textup{(2)} and \textup{(3)} follows from the injectivity
of the left-hand map in \eqref{eq:uct-visible-quotient}.  Suppose first that
\textup{(2)} holds.  If $\varphi_*[\alpha]=0$, naturality of evaluation gives
\[
0=\operatorname{ev}(\varphi_*[\alpha])=\varphi\circ h_\alpha.
\]
Thus $I_\alpha\subseteq\ker\varphi$, and $\varphi$ factors uniquely through
$q_0$.  Since $(q_0)_*[\alpha]=0$, this proves \textup{(1)} with
$q_\alpha=q_0$ and establishes \eqref{eq:visible-kernel-cycle-image}.

Conversely, assume \textup{(1)} and represent $\epsilon_\alpha$ by an
extension
\begin{equation}\label{eq:residual-extension-representative}
0\longrightarrow A/I_\alpha
\xrightarrow{\ j\ } E\longrightarrow
H_1(N(\mathfrak{U});\ZZ)\longrightarrow0.
\end{equation}
The coefficient map $j$ sends $\epsilon_\alpha$ to zero.  To verify this,
consider an extension
$0\to M\xrightarrow{i}E'\xrightarrow{p}H\to0$ and a map
$u\colon M\to B$.  Its pushout has middle group
\[
P=(B\oplus E')/\langle(u(m),-i(m)):m\in M\rangle.
\]
If $\widetilde u\colon E'\to B$ extends $u$, then
\[
p(e)\longmapsto[-\widetilde u(e),e]
\]
is a well-defined splitting of $P\to H$.  Conversely, if
$s\colon H\to P$ is a splitting, the difference between the canonical map
$e\mapsto[0,e]$ and $s\circ p$ takes values in the embedded copy of $B$ and
therefore defines a homomorphism $\widetilde u\colon E'\to B$; the relation
$[0,i(m)]=[u(m),0]$ gives $\widetilde u\circ i=u$.  Thus the pushout splits
if and only if $u$ extends to $E'$.  Taking $u=i$ and the identity of $E'$
proves the assertion.

By naturality of \eqref{eq:uct-visible-quotient}, the composite
\[
A\xrightarrow{\ q_0\ }A/I_\alpha\xrightarrow{\ j\ }E
\]
therefore kills $[\alpha]$.  Universality of $q_\alpha$ implies
\[
\ker(q_\alpha)\subseteq\ker(jq_0)=I_\alpha.
\]
On the other hand, $(q_\alpha)_*[\alpha]=0$ implies
$q_\alpha h_\alpha=0$, hence
$I_\alpha\subseteq\ker(q_\alpha)$.  Thus
$\ker(q_\alpha)=I_\alpha$, and $q_\alpha$ is $q_0$ followed by an
isomorphism.  Since $q_\alpha$ kills $[\alpha]$, so does $q_0$, proving
\textup{(2)}.

It remains to prove the uniform assertion.  Since the nerve is finite,
$H_1(N(\mathfrak{U});\ZZ)$ is finitely generated.  If it is torsion-free,
it is free, and the $\operatorname{Ext}^1$ group in
\eqref{eq:residual-extension-class} vanishes for every coefficient group;
hence every $\epsilon_\alpha$ is zero.  Conversely, suppose that first
homology has a cyclic summand $\ZZ/n\ZZ$.  Set $A=\ZZ/n\ZZ$, and choose the
element of
\[
\operatorname{Ext}^1(H_1(N(\mathfrak{U});\ZZ),A)
\]
whose component on this summand is represented by the nonsplit extension
\[
0\longrightarrow\ZZ/n\ZZ
\xrightarrow{\ [x]\mapsto[nx]\ }\ZZ/n^2\ZZ
\longrightarrow\ZZ/n\ZZ\longrightarrow0
\]
and whose other components are zero.  Its image $[\alpha]$ under the
left-hand injection in \eqref{eq:uct-visible-quotient} is nonzero and has
$h_\alpha=0$.  Hence $I_\alpha=0$ and
$(q_0)_*[\alpha]=[\alpha]\ne0$, so the equivalent conditions fail.
\end{proof}

\begin{remark}\label{rem:cocycle-vs-pointwise}
Both parts of the criterion are intrinsic to $[\alpha]$.  The subgroup
$I_\alpha$ is generated by evaluations on integral $2$-cycles, rather than by
the values of a chosen cocycle representative, and
$\epsilon_\alpha$ is then fixed by exactness.  Torsion in first homology is
therefore an obstruction to uniform existence, not to every individual
visible quotient.
\end{remark}

\begin{lemma}[Register lower bound for full-resolution recovery]
\label{thm:prefix-site-register-lower-bound}
Assume the equivalent conditions of
Theorem~\ref{thm:intrinsic-visible-quotient}, and let
$\Avis=A/I_\alpha$ be the visible quotient attached to the fixed cover and
class.  Suppose one wishes to recover
the full $A$-resolution from $\Avis$ by adjoining a finite register set $R$ in
such a way that there exists an injection
\[
A\hookrightarrow \Avis\times R.
\]
Then
\[
\lvert R\rvert \ge \lvert I_\alpha\rvert.
\]
This bound is sharp: equality can be attained with $R=I_\alpha$.
\end{lemma}

\begin{proof}
Every fiber of the quotient map $A\twoheadrightarrow \Avis$ is a coset of
$I_\alpha$, hence has cardinality $\lvert I_\alpha\rvert$.  If
$A\hookrightarrow \Avis\times R$, then restricting the injection to any one of
these fibers yields an injection into $R$.  Therefore
\[
\lvert R\rvert\ge \lvert I_\alpha\rvert.
\]

For sharpness, take $R:=I_\alpha$.  Choose a set-theoretic section
$s:\Avis\to A$ of the quotient map.  Then every $a\in A$ can be written
uniquely as
\[
a = s(\bar a)+h
\qquad
(\bar a\in\Avis,\ h\in I_\alpha),
\]
and the map $a\mapsto (\bar a,h)$ is injective.
\end{proof}

\begin{definition}[Character-based spectral readout]
\label{def:character-based-spectral-readout}
Let $\chi\in \widehat A:=\Hom(A,\TT)$.  Pushing an $A$-banded comparison datum
forward along $\chi$ produces a $\TT$-banded comparison datum, and hence a
pushed-forward defect class $\chi_*[\alpha]$.  The resulting $\TT$-valued
descent problem is called the \emph{character-based spectral readout}
associated to $\chi$.  It is said to be \emph{spectrally definable} if
$\chi_*[\alpha]=0$ in
$\check{H}^2(\mathfrak{U},\TT)$.
\end{definition}

\begin{proposition}[Spectral definability occurs only on the visible quotient]
\label{thm:prefix-site-spectral-definability-avis}
Assume the equivalent conditions of
Theorem~\ref{thm:intrinsic-visible-quotient}.  Let
$\widehat A:=\Hom(A,\TT)$ be the Pontryagin dual, and define the
annihilator
\[
I_\alpha^\perp
:=
\{\chi\in \widehat A:\ \chi(k)=1\ \text{for all }k\in I_\alpha\}.
\]
Then $I_\alpha^\perp\cong \widehat{\Avis}$, and the character-based spectral
readout of Definition~\ref{def:character-based-spectral-readout} is spectrally
definable if and only if the chosen character lies in $I_\alpha^\perp$.
\end{proposition}

\begin{proof}
Because $A$ is finite abelian, a character of $A$ factors through the quotient
$A\twoheadrightarrow \Avis\cong A/I_\alpha$ if and only if it is trivial on
$I_\alpha$.
This identifies $\widehat{\Avis}$ with the subgroup
$I_\alpha^\perp\le \widehat A$.

Now let $\chi\in\widehat A$.  By
Theorem~\ref{thm:intrinsic-visible-quotient}, the pushed-forward class
$\chi_*[\alpha]\in\check{H}^2(\mathfrak{U},\TT)$ vanishes if and only if $\chi$
factors through $q\colon A\twoheadrightarrow \Avis$.  By the first paragraph,
that happens exactly when $\chi$ is trivial on $I_\alpha$.  Hence the spectral
readout is definable if and only if $\chi\in I_\alpha^\perp$.
\end{proof}
