
\subsection{Realisation of prescribed cyclic zero divisors by cyclic blocks}

\Cref{cor:primitive-symbolic-summability-exclusion} shows that the primitive-orbit
data \((n,1,p_n(A))\) attached to a nontrivial primitive finite-state
shift do not satisfy the summability hypothesis used below. Accordingly,
this subsection treats the abstract realisation problem for summable
prescribed cyclic zero divisors rather than the divergent symbolic
primitive-orbit packaging itself.
The bridge in \Cref{prop:fredholm-witt-bridge} starts from the Witt
coordinates
\[
  p_n(T)=\frac{1}{n}\sum_{d\mid n}\mu(d)\,\Tr(T^{n/d}),
\]
which need not be integers or non-negative, whereas the data here are
triples \((n_j,\beta_j,m_j)\) with \(m_j\in\NN\) and
\(\beta_j\in\CC\setminus\{0\}\). The cyclic-block construction therefore
starts from an independently prescribed entire divisor
\[
  D_{\mathcal D}(z)=\prod_{j\in J}(1-\beta_j z^{n_j})^{m_j}
\]
with integer multiplicities and the summable root budget
\[
  \sum_{j\in J}m_j n_j\abs{\beta_j}^{1/n_j}<\infty.
\]
By
\Cref{prop:cyclic-witt-compatibility}, these two parametrisations agree
only in the special unit-root integral case
\(\mathcal D=\{(n,1,m_n)\}_{n\ge1}\); in general the cyclic theorem below
is not a converse to the Fredholm--Witt bridge and does not assert that
every Fredholm--Witt expansion can be realised by positive-integer cyclic
blocks.

For \(n\ge 1\), let \(\Pi_n\) denote the cyclic permutation matrix on
\(\CC^n\). For \(\alpha\in\CC\) define the cyclic block
\[
  C(n,\alpha):=\alpha\Pi_n.
\]
Since the eigenvalues of \(\Pi_n\) are the \(n\)th roots of unity and
\(\prod_{\omega^n=1}(t-\omega)=t^n-1\), one has
\begin{equation}\label{eq:cycle-block-det}
  \det(I-zC(n,\alpha))
  =\prod_{\omega^n=1}(1-z\alpha\omega)
  =1-\alpha^n z^n.
\end{equation}

\begin{definition}[Cyclic data, trace budget, and support size]
  \label{def:cyclic-data-trace-budget}
  A countable cyclic datum is a family
  \[
    \mathcal D=\{(n_j,\beta_j,m_j)\}_{j\in J},
    \qquad
    n_j,m_j\in\NN,\quad \beta_j\in\CC\setminus\{0\},
  \]
  indexed by a finite or countably infinite set \(J\). Its trace budget,
  formal positive determinant, and numerical support size are
  \[
    \tau_{\mathcal D}
    :=\sum_{j\in J}m_j n_j\abs{\beta_j}^{1/n_j}\in[0,\infty],
  \]
  \[
    D_{\mathcal D}(z)
    :=\prod_{j\in J}(1-\beta_jz^{n_j})^{m_j},
    \qquad
    N_{\mathcal D}:=\sum_{j\in J}m_jn_j\in\NN\cup\{\infty\}.
  \]
  The associated cyclic zero divisor is the divisor on
  \(\CC^\times\)
  \[
    Z_{\mathcal D}
    :=\sum_{j\in J}m_j\sum_{\lambda^{n_j}=\beta_j}[\lambda].
  \]
  Thus a point \(\lambda\in\CC^\times\) has divisor multiplicity
  \[
    M_{\mathcal D}(\lambda)
    :=\sum_{\{j\in J:\lambda^{n_j}=\beta_j\}}m_j,
  \]
  possibly \(+\infty\) before summability or realisability hypotheses are
  imposed. Equivalently, the factor
  \((1-\beta_jz^{n_j})^{m_j}\) contributes multiplicity \(m_j\) at every
  root \(\lambda\) of \(\lambda^{n_j}=\beta_j\).
  The symbol \(N_{\mathcal D}\) is a count of cyclic spectral slots. When
  \(J\) is infinite it is used only as the statement that there are
  infinitely many such slots; the Hilbert-space dimension of the closed
  support of the associated separable cyclic model is then \(\aleph_0\),
  not the extended integer \(\infty\).

  If \(\tau_{\mathcal D}<\infty\), choose roots
  \(\alpha_j^{n_j}=\beta_j\) and set
  \[
    H_{\mathcal D}
    :=\bigoplus_{j\in J}\bigl(\CC^{n_j}\otimes\CC^{m_j}\bigr),
    \qquad
    L_{\mathcal D}
    :=\bigoplus_{j\in J}\bigl(C(n_j,\alpha_j)\otimes I_{m_j}\bigr).
  \]
\end{definition}

\begin{lemma}[Single cyclic-block spectrum]
  \label{lem:single-cyclic-block-spectrum}
  Let \(n,m\in\NN\), \(\beta\in\CC\setminus\{0\}\), and
  \(\alpha^n=\beta\). Then the non-zero eigenvalues of
  \(C(n,\alpha)\otimes I_m\) are
  \[
    \alpha\omega,\qquad \omega^n=1,
  \]
  each with multiplicity \(m\), and
  \[
    \det\!\bigl(I-z(C(n,\alpha)\otimes I_m)\bigr)
    =(1-\beta z^n)^m .
  \]
\end{lemma}

\begin{proof}
  The Fourier basis diagonalizes \(\Pi_n\) with eigenvalues
  \(\omega\), \(\omega^n=1\). Hence \(C(n,\alpha)\otimes I_m\) has
  eigenvalues \(\alpha\omega\), each repeated \(m\) times. Therefore
  \[
    \det\!\bigl(I-z(C(n,\alpha)\otimes I_m)\bigr)
    =\prod_{\omega^n=1}(1-z\alpha\omega)^m
    =\left(\prod_{\omega^n=1}(1-z\alpha\omega)\right)^m.
  \]
  For \(x=\alpha z\), the identity
  \[
    \prod_{\omega^n=1}(1-x\omega)=1-x^n
  \]
  follows from \(\prod_{\omega^n=1}(t-\omega)=t^n-1\) by substituting
  \(t=x^{-1}\) and multiplying by \(x^n\). Hence
  \[
    \det\!\bigl(I-z(C(n,\alpha)\otimes I_m)\bigr)
    =(1-\alpha^nz^n)^m
    =(1-\beta z^n)^m .
  \]
\end{proof}

\begin{lemma}[Finite cyclic truncation determinant with positive exponents]
  \label{lem:finite-cyclic-truncation-determinant}
  Let \(F\) be a finite set of indices and put
  \[
    L_F:=\bigoplus_{j\in F}\bigl(C(n_j,\alpha_j)\otimes I_{m_j}\bigr),
    \qquad \alpha_j^{n_j}=\beta_j.
  \]
  Then, as a finite-dimensional determinant identity,
  \[
    \det(I-zL_F)
    =\prod_{j\in F}(1-\beta_j z^{n_j})^{m_j},
    \qquad
    \det(I-zL_F)^{-1}
    =\prod_{j\in F}(1-\beta_j z^{n_j})^{-m_j}
  \]
  wherever the reciprocal is defined.
\end{lemma}

\begin{proof}
  Because \(L_F\) is a finite orthogonal direct sum, the determinant of
  \(I-zL_F\) factors as the product of the determinants of its block
  restrictions:
  \[
    \det(I-zL_F)
    =\prod_{j\in F}
      \det\!\bigl(I-z(C(n_j,\alpha_j)\otimes I_{m_j})\bigr).
  \]
  Applying \Cref{lem:single-cyclic-block-spectrum} to each block gives
  \begin{align*}
    \det(I-zL_F)
    &=\prod_{j\in F}(1-\alpha_j^{n_j}z^{n_j})^{m_j} \\
    &=\prod_{j\in F}(1-\beta_j z^{n_j})^{m_j}.
  \end{align*}
  Equivalently, one may write the same computation eigenvalue by
  eigenvalue as
  \begin{align*}
    \det(I-zL_F)
    &=\prod_{j\in F}\prod_{\omega^{n_j}=1}
      (1-z\alpha_j\omega)^{m_j} \\
    &=\prod_{j\in F}(1-\alpha_j^{n_j}z^{n_j})^{m_j}
      =\prod_{j\in F}(1-\beta_j z^{n_j})^{m_j}.
  \end{align*}
  Taking reciprocals gives the second identity on the complement of the
  finite zero set of the determinant.
\end{proof}

\begin{lemma}[Direct normal convergence of the cyclic product]
  \label{lem:direct-cyclic-weierstrass-product}
  Assume
  \[
    \sum_{j\in J}m_j n_j\abs{\beta_j}^{1/n_j}<\infty,
    \qquad
    a_j:=\abs{\beta_j}^{1/n_j}.
  \]
  Then for every \(R>0\),
  \[
    \sum_{j\in J} m_j \sup_{\abs z\le R}\abs{\beta_j z^{n_j}}
    \le \sum_{j\in J}m_j\abs{\beta_j}R^{n_j}
    =\sum_{j\in J}m_j(Ra_j)^{n_j}
    <\infty.
  \]
  Consequently the infinite product
  \[
    \prod_{j\in J}(1-\beta_j z^{n_j})^{m_j}
  \]
  converges normally on compact subsets of \(\CC\). In particular it
  converges locally uniformly, defines an entire function, and its value
  is independent of the enumeration of \(J\).
\end{lemma}

\begin{proof}
  Since \(m_jn_j\ge1\) for every \(j\),
  \[
    \sum_{j\in J} a_j
    \le \sum_{j\in J} m_jn_j a_j
    <\infty,
  \]
  so \(a_j\to0\). Fix \(R>0\). For all sufficiently large \(j\) one has
  \(Ra_j\le\frac12\), hence
  \[
    \sup_{\abs z\le R}\abs{\beta_j z^{n_j}}
    =\abs{\beta_j}R^{n_j}
    =(Ra_j)^{n_j}
    \le Ra_j.
  \]
  Writing \(J_{\mathrm{small}}(R)\subset J\) for a finite set containing
  all remaining indices, one obtains
  \begin{align*}
    \sum_{j\in J} m_j \sup_{\abs z\le R}\abs{\beta_j z^{n_j}}
    &\le C_R+R\sum_{j\notin J_{\mathrm{small}}(R)}m_j a_j \\
    &\le C_R+R\sum_{j\notin J_{\mathrm{small}}(R)}m_jn_j a_j
    <\infty,
  \end{align*}
  where
  \[
    C_R:=\sum_{j\in J_{\mathrm{small}}(R)}
    m_j\abs{\beta_j}R^{n_j}<\infty.
  \]
  Therefore
  \[
    \sum_{j\in J} m_j \sup_{\abs z\le R}\abs{\beta_j z^{n_j}}<\infty.
  \]

  Let \(K_R:=\{z\in\CC:\abs z\le R\}\). After discarding finitely many
  indices, the estimate above gives
  \[
    \sup_{z\in K_R}\abs{\beta_j z^{n_j}}
    \le \abs{\beta_j}R^{n_j}
    \le \frac12.
  \]
  On \(\{w\in\CC:\abs w\le \frac12\}\), the principal logarithm satisfies
  \(\abs{\log(1-w)}\le 2\abs w\). Hence, for all large \(j\),
  \[
    \sup_{z\in K_R}
    \abs{m_j\log(1-\beta_j z^{n_j})}
    \le 2m_j\abs{\beta_j}R^{n_j}.
  \]
  The right-hand side is summable in \(j\), so the series
  \[
    \sum_j m_j\log(1-\beta_j z^{n_j})
  \]
  converges absolutely and uniformly on \(K_R\) after a finite initial
  segment. Equivalently, the product converges normally on \(K_R\).
  Exponentiating shows that the tail product converges uniformly on
  \(K_R\), hence the full product
  \(\prod_j(1-\beta_j z^{n_j})^{m_j}\) converges locally uniformly on
  \(K_R\). Since \(R\) was arbitrary, the product defines an entire
  function on \(\CC\). The absolute uniform convergence of the logarithmic
  series also makes the sum, and therefore the product, independent of the
  chosen enumeration of \(J\).
\end{proof}

\begin{lemma}[Trace-norm limit convention for cyclic products]
  \label{lem:cyclic-product-trace-norm-limit}
  Assume
  \[
    \sum_{j\in J}m_j n_j\abs{\beta_j}^{1/n_j}<\infty
  \]
  and choose \(\alpha_j^{n_j}=\beta_j\). Let
  \[
    H_{\mathcal D}
    :=\bigoplus_{j\in J}\bigl(\CC^{n_j}\otimes\CC^{m_j}\bigr),
    \qquad
    L_{\mathcal D}
    :=\bigoplus_{j\in J}\bigl(C(n_j,\alpha_j)\otimes I_{m_j}\bigr)
  \]
  on \(H_{\mathcal D}\), and let \(L_{\mathcal D}^{(N)}\) be any finite
  truncation along an enumeration of \(J\). Then
  \(L_{\mathcal D}^{(N)}\to L_{\mathcal D}\) in trace norm and
  \[
    \prod_{j\in J}(1-\beta_j z^{n_j})^{m_j}
    =\lim_{N\to\infty}
      \prod_{j\le N}(1-\beta_j z^{n_j})^{m_j}
  \]
  locally uniformly for \(z\in\CC\), and
  \[
    \det(I-zL_{\mathcal D})
    =\lim_{N\to\infty}
      \prod_{j\le N}(1-\beta_j z^{n_j})^{m_j}
  \]
  locally uniformly for \(z\in\CC\). On any connected zero-free
  neighbourhood \(\Omega\ni0\) for this determinant,
  \[
    \det(I-zL_{\mathcal D})^{-1}
    =\lim_{N\to\infty}
      \prod_{j\le N}(1-\beta_j z^{n_j})^{-m_j}
  \]
  locally uniformly on \(\Omega\). Thus the negative-exponent product is
  used only as the reciprocal of the non-vanishing Fredholm determinant on
  such a domain.
\end{lemma}

\begin{proof}
  The singular values of the \(j\)-th block are all
  \(\abs{\alpha_j}=\abs{\beta_j}^{1/n_j}\), with multiplicity
  \(m_jn_j\). Hence
  \[
    \norm{L_{\mathcal D}-L_{\mathcal D}^{(N)}}_1
    =\sum_{j>N}m_j n_j\abs{\beta_j}^{1/n_j}\to0 .
  \]
  Equivalently, the net of finite partial direct sums over finite
  subsets of \(J\) converges unconditionally in trace norm; no ordering of
  the factors is part of the determinant definition.
  By \Cref{lem:direct-cyclic-weierstrass-product}, the positive-exponent
  product
  \[
    P_{\mathcal D}(z):=\prod_{j\in J}(1-\beta_j z^{n_j})^{m_j}
  \]
  is already well defined as an entire locally uniform limit of the finite
  products.
  For each fixed \(N\), \Cref{lem:finite-cyclic-truncation-determinant}
  gives
  \[
    \det(I-zL_{\mathcal D}^{(N)})
    =\prod_{j\le N}(1-\beta_j z^{n_j})^{m_j}.
  \]
  Trace-norm continuity of the Fredholm determinant, in the form of
  Simon's estimate \cite[Theorem~3.4]{Simon2005TraceIdeals} and in line
  with the finite-rank approximation approach of
  \cite{Simon1977InfiniteDeterminants} and the continuity-extension
  approach of \cite{GohbergGoldbergKrupnik1996TracesDeterminants},
  \[
    \abs{\det(I+K)-\det(I+L)}
    \le\norm{K-L}_1\exp\!\bigl(1+\norm{K}_1+\norm{L}_1\bigr),
  \]
  applied to \(K=-zL_{\mathcal D}^{(N)}\) and
  \(L=-zL_{\mathcal D}\), gives locally uniform convergence in \(z\).
  Since the same finite products already converge locally uniformly to
  \(P_{\mathcal D}\), uniqueness of locally uniform limits yields
  \[
    \det(I-zL_{\mathcal D})=P_{\mathcal D}(z)
    =\prod_{j\in J}(1-\beta_j z^{n_j})^{m_j}.
  \]
  If the limiting determinant is non-zero on \(\Omega\), then its
  reciprocal is holomorphic there; local uniform convergence of the
  reciprocals follows as follows. On a compact set
  \(K\Subset\Omega\), the limiting determinant has a positive minimum
  modulus. The locally uniform convergence of the finite determinants
  therefore makes all sufficiently large finite determinants non-zero on
  \(K\), and the reciprocal difference is controlled by the usual
  identity
  \[
    f_N^{-1}-f^{-1}=\frac{f-f_N}{f_Nf}.
  \]
  Hence the reciprocal finite products converge locally uniformly to
  \(D_{\mathcal D}^{-1}\) on \(\Omega\).
\end{proof}

\begin{lemma}[Choice of roots and unitary relabelling]\label{lem:choice-of-roots}
  Let \(n,m\in\NN\), let \(\beta\in\CC\setminus\{0\}\), and let
  \(\alpha^n=(\alpha')^n=\beta\). Then
  \(C(n,\alpha)\otimes I_m\) and \(C(n,\alpha')\otimes I_m\) are
  unitarily equivalent. More explicitly, if
  \(\eta:=\alpha'/\alpha\), so that \(\eta^n=1\), and
  \((e_0,\dots,e_{n-1})\) is the cyclic basis in which
  \(\Pi_n e_k=e_{k+1\!\!\pmod n}\), then
  \[
    U_\eta e_k:=\eta^k e_k
    \qquad (0\le k<n)
  \]
  defines a unitary on \(\CC^n\) satisfying
  \[
    U_\eta C(n,\alpha)U_\eta^{-1}=C(n,\alpha').
  \]
  Consequently any two root choices in the construction of
  \(L_{\mathcal D}\) give unitarily equivalent cyclic models, via the
  block direct sum of the maps \(U_{\eta_j}\otimes I_{m_j}\). The
  determinant data therefore determine the cyclic model only up to this
  explicit relabelling and the addition of zero summands.
\end{lemma}

\begin{proof}
  The operator \(U_\eta\) is unitary because it is diagonal in the cyclic
  basis with unimodular diagonal entries. For \(0\le k<n\),
  \[
    U_\eta\Pi_nU_\eta^{-1}e_k
    =U_\eta\Pi_n(\eta^{-k} e_k)
    =\eta^{-k}U_\eta e_{k+1}
    =\eta^{-k}\eta^{k+1}e_{k+1}
    =\eta\Pi_ne_k
  \]
  if \(k<n-1\), and the same computation at \(k=n-1\) gives
  \(U_\eta\Pi_nU_\eta^{-1}e_{n-1}=\eta^{-(n-1)}e_0=\eta e_0\), since
  \(\eta^n=1\). Thus \(U_\eta\Pi_nU_\eta^{-1}=\eta\Pi_n\). Hence
  \[
    U_\eta C(n,\alpha)U_\eta^{-1}
    =\alpha\eta\Pi_n
    =\alpha'\Pi_n
    =C(n,\alpha').
  \]
  Tensoring with \(I_m\) proves the finite-multiplicity statement, and the
  Hilbert direct sum of the resulting block unitaries proves the assertion
  for \(L_{\mathcal D}\).
\end{proof}

\begin{proposition}[Acyclic construction interface for cyclic data]
  \label{prop:cyclic-data-construction-interface}
  Let \(\mathcal D\) be a countable cyclic datum in the sense of
  \Cref{def:cyclic-data-trace-budget} with
  \(\tau_{\mathcal D}<\infty\), and choose roots
  \(\alpha_j^{n_j}=\beta_j\). For each finite
  \(F\Subset J\), set
  \[
    H_F:=\bigoplus_{j\in F}(\CC^{n_j}\otimes\CC^{m_j}),
    \qquad
    L_F:=\bigoplus_{j\in F}(C(n_j,\alpha_j)\otimes I_{m_j}),
  \]
  and regard \(L_F\) as acting by zero on the orthogonal complement of
  \(H_F\) in \(H_{\mathcal D}\). Then the net
  \((L_F)_{F\Subset J}\), directed by inclusion, converges
  unconditionally in trace norm to a normal operator
  \(L_{\mathcal D}\in\sone(H_{\mathcal D})\). This limit satisfies
  \[
    \norm{L_{\mathcal D}}_1=\tau_{\mathcal D},
    \qquad
    \dim H_{\mathcal D}=
    \begin{cases}
      N_{\mathcal D},& J\text{ finite},\\
      \aleph_0,& J\text{ infinite},
    \end{cases}
  \]
  where the second equality is a Hilbert-space dimension identity.
  Moreover \(D_{\mathcal D}\) is an entire function and
  \[
    \det(I-zL_{\mathcal D})=D_{\mathcal D}(z)
  \]
  locally uniformly for \(z\in\CC\). The unitary equivalence class of
  \(L_{\mathcal D}\) is independent of the chosen roots; more precisely,
  replacing \(\alpha_j\) by another root \(\alpha_j'\) conjugates the
  \(j\)-th block by the diagonal relabelling from
  \(\Cref{lem:choice-of-roots}\), and the orthogonal direct sum of these
  relabellings conjugates the two cyclic models.
\end{proposition}

\begin{proof}
  If \(E\subset F\) are finite subsets of \(J\), then the singular values
  of \(L_F-L_E\) are the numbers
  \(\abs{\alpha_j}=\abs{\beta_j}^{1/n_j}\), each repeated \(m_jn_j\)
  times, with \(j\in F\setminus E\). Hence
  \[
    \norm{L_F-L_E}_1
    =\sum_{j\in F\setminus E}m_jn_j\abs{\beta_j}^{1/n_j}.
  \]
  Since the full sum defining \(\tau_{\mathcal D}\) is finite, the finite
  partial sums are an unconditionally Cauchy net in trace norm. Their
  trace-norm limit is the block direct sum
  \(L_{\mathcal D}\). Trace-norm continuity of the norm gives
  \[
    \norm{L_{\mathcal D}}_1
    =\sup_{\substack{F\subset J\\ F\ \mathrm{finite}}}
      \sum_{j\in F}m_jn_j\abs{\beta_j}^{1/n_j}
    =\tau_{\mathcal D}.
  \]
  Normality follows by passing to the strong, hence weak, limit in the
  block diagonal representation: each summand
  \(C(n_j,\alpha_j)\otimes I_{m_j}\) is a scalar multiple of a unitary and
  different summands act on orthogonal reducing subspaces, so
  \(L_{\mathcal D}L_{\mathcal D}^*=L_{\mathcal D}^*L_{\mathcal D}\) on the
  finite-support algebraic direct sum and hence on its closure.
  \Cref{lem:direct-cyclic-weierstrass-product,lem:cyclic-product-trace-norm-limit}
  gives the entire locally uniform convergence of \(D_{\mathcal D}\) and
  the displayed Fredholm determinant identity. These inputs all precede
  the exact threshold and normal-classification theorems, so this
  proposition supplies their construction step without using either
  theorem.

  The Hilbert-space dimension statement is the dimension of the orthogonal
  direct sum
  \(\bigoplus_{j\in J}(\CC^{n_j}\otimes\CC^{m_j})\): finite \(J\) gives the
  finite integer \(N_{\mathcal D}\), while countably infinite \(J\) gives
  the separable infinite dimension \(\aleph_0\).

  Finally, let \((\alpha_j')_{j\in J}\) be another choice of roots with
  \((\alpha_j')^{n_j}=\beta_j\). By \Cref{lem:choice-of-roots}, there is a
  blockwise unitary
  \(U_j:\CC^{n_j}\otimes\CC^{m_j}\to\CC^{n_j}\otimes\CC^{m_j}\) conjugating
  \(C(n_j,\alpha_j)\otimes I_{m_j}\) to
  \(C(n_j,\alpha_j')\otimes I_{m_j}\). The orthogonal direct sum
  \(\bigoplus_{j\in J}U_j\) is unitary and conjugates the two direct-sum
  limits, proving root-choice independence.
\end{proof}

\noindent
\Cref{prop:cyclic-data-construction-interface} is the acyclic
construction interface used by the exact threshold theorem and by the
later normal classification: it depends only on the cyclic block
calculus and determinant continuity, while the classification theorems
use it as an input.

\noindent\emph{Determinant convention.}
In the infinite cyclic-block products below, the positive-exponent product
is first defined directly by
\Cref{lem:direct-cyclic-weierstrass-product} as an entire locally uniform
limit of finite products, and \Cref{lem:cyclic-product-trace-norm-limit}
then identifies that entire function with the Fredholm determinant of the
trace-class cyclic-block model. The product with negative exponents is not
an independently defined infinite product. It always means the reciprocal
of that non-vanishing Fredholm determinant on a specified connected
zero-free neighbourhood of the origin. Thus the positive-exponent product
exists as a genuine Weierstrass product, while reciprocals are taken only
after a zero-free domain has been fixed.

\noindent\emph{Reciprocal-product realisation.}
For data \(\mathcal D=\{(n_j,\beta_j,m_j)\}_{j\in J}\) satisfying the
summability hypothesis, write
\[
  D_{\mathcal D}(z):=\prod_{j\in J}(1-\beta_j z^{n_j})^{m_j}
\]
for the locally uniform limit of the positive-exponent finite products.
A trace-class operator \(L\) will be said to realise the reciprocal Euler
product
\(\prod_{j\in J}(1-\beta_j z^{n_j})^{-m_j}\) only in the following
determinant-side sense: on some connected zero-free neighbourhood
\(\Omega\ni0\) for \(D_{\mathcal D}\),
\[
  \begin{aligned}
    \det(I-zL)&=D_{\mathcal D}(z),\\
    \det(I-zL)^{-1}
      &=D_{\mathcal D}(z)^{-1}
        =\prod_{j\in J}(1-\beta_j z^{n_j})^{-m_j}
        \quad (z\in\Omega).
  \end{aligned}
\]
This terminology never reverses the finite-dimensional calculation:
for every finite truncation \(F\), the Fredholm determinant is
\(\prod_{j\in F}(1-\beta_j z^{n_j})^{m_j}\), and the negative-exponent
expression is its reciprocal where the determinant is non-zero.

\begin{definition}[Closed non-zero spectral support]
  \label{def:closed-nonzero-support}
For a normal trace-class operator \(L\), write
\[
  \mathsf S(L):=
  \overline{\mathrm{span}}
  \{\,\ker(L-\lambda I):\lambda\in\spec(L)\setminus\{0\}\,\}.
\]
This notation is the support convention used throughout the normal
construction and classification sections; the model-specific range formula
is recorded later in \Cref{lem:cyclic-sum-range-closure}.
\end{definition}

\begin{lemma}[Closed support is \(\overline{\Ran L}\)]
  \label{lem:normal-support-range-closure}
  If \(L\) is normal and trace-class, then
  \[
    \mathsf S(L)=(\ker L)^\perp=\overline{\Ran(L)}.
  \]
  This identifies the closed support with \(\overline{\Ran L}\). It does
  not assert that \(\Ran L\) is closed.
\end{lemma}

\begin{proof}
  A trace-class normal operator is compact. Hence the spectral theorem
  yields the orthogonal decomposition
  \[
    H=\ker L\oplus
    \overline{\mathrm{span}}
    \{\,\ker(L-\lambda I):\lambda\in\spec(L)\setminus\{0\}\,\},
  \]
  with possible spectral accumulation only at \(0\). This shows that the
  closed span of the non-zero eigenspaces is \((\ker L)^\perp\).

  The algebraic range \(\Ran L\) contains every non-zero eigenspace because
  \(L\) is invertible on that eigenspace, and \(\Ran L\) is orthogonal to
  \(\ker L\).
  Therefore
  \[
    \mathsf S(L)\subseteq \overline{\Ran(L)}
    \subseteq (\ker L)^\perp=\mathsf S(L),
  \]
  which proves the claim.
\end{proof}

\begin{proposition}[Sharp trace-norm budget for cyclic zero divisors]
  \label{prop:sharp-trace-budget-support-obstruction}
  Let \(\mathcal D=\{(n_j,\beta_j,m_j)\}_{j\in J}\) be a countable
  family with \(n_j,m_j\in\NN\) and
  \(\beta_j\in\CC\setminus\{0\}\). Choose roots
  \(\alpha_j^{n_j}=\beta_j\), and let
  \[
    \Lambda_{\mathcal D}
    :=
    \bigl\{\!\bigl\{\alpha_j\omega:
      j\in J,\ \omega^{n_j}=1,\text{ repeated }m_j\text{ times}
    \bigr\}\!\bigr\}
  \]
  be the prescribed non-zero spectral multiset, with multiplicities added
  when the same complex number occurs more than once.

  Suppose \(L\in\sone(H)\) is normal and its non-zero spectrum, counted
  with algebraic multiplicity, is exactly \(\Lambda_{\mathcal D}\). Then
  the trace-norm budget is forced:
  \[
    \|L\|_1
    =
    \sum_{\lambda\in\Lambda_{\mathcal D}}\abs{\lambda}
    =
    \sum_{j\in J}m_j n_j\abs{\beta_j}^{1/n_j}
    <\infty .
  \]
  Moreover
  \[
    \dim \mathsf S(L)=
    \begin{cases}
      N_{\mathcal D},& J\text{ finite},\\
      \aleph_0,& J\text{ infinite},
    \end{cases}
  \]
  as a Hilbert-space dimension; equivalently, the finite numerical support
  count is \(N_{\mathcal D}=\sum_{j\in J}m_jn_j\) when \(J\) is finite,
  while a countably infinite datum gives separable infinite support
  dimension. If \(J\) is infinite, then \(\Ran L\) is a proper
  dense subspace of its closed support:
  \[
    \Ran(L)\subsetneq \mathsf S(L)
    =\overline{\Ran(L)} .
  \]
  Conversely, when the displayed budget is finite, the cyclic direct sum
  \(L_{\mathcal D}\) supplied by
  \Cref{prop:cyclic-data-construction-interface}, with trace-norm control
  from \Cref{lem:cyclic-product-trace-norm-limit}, attains exactly this
  trace norm and this closed support.
\end{proposition}

\begin{proof}
  Since \(L\) is compact and normal, the spectral theorem diagonalizes it
  on the orthogonal sum of its non-zero eigenspaces and its kernel. For a
  normal trace-class operator, the trace norm is the sum of the moduli of
  the non-zero eigenvalues, counted with multiplicity. Therefore
  \[
    \|L\|_1
    =\sum_{\lambda\in\spec(L)\setminus\{0\}}
      \abs{\lambda}\,\dim\ker(L-\lambda I).
  \]
  By the assumed cyclic zero divisor, this is the sum over the multiset
  \(\Lambda_{\mathcal D}\). Each \(j\)-th cyclic factor contributes
  \(n_j\) eigenvalues of modulus
  \(\abs{\alpha_j}=\abs{\beta_j}^{1/n_j}\), each repeated \(m_j\) times.
  Hence
  \[
    \sum_{\lambda\in\Lambda_{\mathcal D}}\abs{\lambda}
    =\sum_{j\in J}m_j n_j\abs{\beta_j}^{1/n_j}.
  \]
  The left side is \(\|L\|_1<\infty\), so the budget is necessary.

  The space \(\mathsf S(L)\) is the Hilbert direct sum of the non-zero
  eigenspaces. Its dimension is therefore the total multiplicity of
  \(\Lambda_{\mathcal D}\), namely
  \[
    \sum_{\lambda\in\Lambda_{\mathcal D}}1
    =N_{\mathcal D},
  \]
  with the same convention that multiplicities are counted. For finite
  \(J\) this is the Hilbert-space dimension itself. For infinite \(J\) the
  displayed equality records the countably infinite spectral-slot count,
  and the Hilbert-space dimension is \(\aleph_0\). This remains true when
  different cyclic factors contribute the same eigenvalue, because for
  normal operators the algebraic and geometric multiplicities coincide, so
  the eigenspace dimension is the sum of the corresponding algebraic
  multiplicities.

  By \Cref{lem:normal-support-range-closure},
  \(\overline{\Ran(L)}=\mathsf S(L)\). If \(J\) is infinite, then
  the multiset \(\Lambda_{\mathcal D}\) is infinite and the trace-class
  condition just proved forces its elements to have moduli tending to
  \(0\). Choose an orthonormal sequence of eigenvectors \(e_r\) with
  eigenvalues \(\lambda_r\ne0\) and \(\lambda_r\to0\). The closed-range
  theorem would give a constant \(c>0\) with
  \(\norm{Lx}\ge c\norm{x}\) for all
  \(x\in(\ker L)^\perp=\mathsf S(L)\) if \(\Ran(L)\) were closed.
  Applying this to \(e_r\) gives \(\abs{\lambda_r}\ge c\), a
  contradiction. Thus \(\Ran L\) is not closed. Since its closure
  is \(\mathsf S(L)\), it is a proper dense subspace of that support.

  Finally assume
  \(\sum_jm_jn_j\abs{\beta_j}^{1/n_j}<\infty\). The block sum
  \(L_{\mathcal D}\) is trace-class by
  \Cref{lem:cyclic-product-trace-norm-limit}. Its blocks have exactly the
  multiset \(\Lambda_{\mathcal D}\), so the equalities above give both the
  asserted trace norm and the asserted closed support.
\end{proof}

\begin{proposition}[Collapsed spectral multiplicities of Euler data]
  \label{prop:collapsed-spectral-multiplicities}
  Let \(\mathcal D=\{(n_j,\beta_j,m_j)\}_{j\in J}\) satisfy
  \[
    n_j,m_j\in\NN,\qquad
    \beta_j\in\CC\setminus\{0\},\qquad
    \sum_{j\in J}m_j n_j\abs{\beta_j}^{1/n_j}<\infty,
  \]
  and choose roots \(\alpha_j^{n_j}=\beta_j\). For each
  \(\lambda\in\CC\setminus\{0\}\), define the collapsed multiplicity
  \[
    M_{\mathcal D}(\lambda)
    :=\sum_{\{j\in J:\lambda^{n_j}=\beta_j\}} m_j.
  \]
  Then each \(M_{\mathcal D}(\lambda)\) is finite, the support
  \[
    \Sigma_{\mathcal D}
    :=\{\lambda\in\CC\setminus\{0\}:M_{\mathcal D}(\lambda)\neq 0\}
  \]
  is countable, and for the cyclic model
  \[
    L_{\mathcal D}
    =\bigoplus_{j\in J}\bigl(C(n_j,\alpha_j)\otimes I_{m_j}\bigr)
  \]
  one has
  \[
    \dim\ker(L_{\mathcal D}-\lambda I)=M_{\mathcal D}(\lambda)
    \qquad (\lambda\in\CC\setminus\{0\}).
  \]
  Consequently,
  \[
    L_{\mathcal D}
    \cong \bigoplus_{\lambda\in\Sigma_{\mathcal D}}
      \lambda I_{M_{\mathcal D}(\lambda)},
  \]
  and
  \[
    \det(I-zL_{\mathcal D})
    =\prod_{\lambda\in\Sigma_{\mathcal D}}
      (1-\lambda z)^{M_{\mathcal D}(\lambda)}
  \]
  as an entire function, where the right-hand side is the locally uniform
  limit of the corresponding finite products. Equivalently, the multiset
  \(\Lambda_{\mathcal D}\) of non-zero eigenvalues is exactly the multiset
  in which \(\lambda\) appears with multiplicity \(M_{\mathcal D}(\lambda)\).
  This is a collapse statement: it does not reconstruct the chosen Euler
  packaging \((n_j,\beta_j,m_j)\) from the determinant without auxiliary
  presentation data.
\end{proposition}

\begin{proof}
  For each \(j\), the Fourier basis diagonalizes \(\Pi_{n_j}\), so
  \(C(n_j,\alpha_j)\otimes I_{m_j}\) is unitarily equivalent to
  \[
    \bigoplus_{\omega^{n_j}=1}\alpha_j\omega\,I_{m_j}.
  \]
  Since \(\lambda^{n_j}=\beta_j\) if and only if
  \(\lambda=\alpha_j\omega\) for some \(n_j\)-th root of unity \(\omega\),
  the \(j\)-th block contributes an \(m_j\)-dimensional eigenspace to
  \(\ker(L_{\mathcal D}-\lambda I)\) exactly when \(\lambda^{n_j}=\beta_j\),
  and otherwise contributes nothing. Summing over \(j\) gives
  \[
    \dim\ker(L_{\mathcal D}-\lambda I)=M_{\mathcal D}(\lambda).
  \]

  Summing over \(\lambda\neq 0\) and then over the indices contributing to
  each \(\lambda\) gives
  \begin{align*}
    \sum_{\lambda\in\Sigma_{\mathcal D}}\abs{\lambda}M_{\mathcal D}(\lambda)
    &=\sum_{\lambda\in\Sigma_{\mathcal D}}
      \sum_{\{j:\lambda^{n_j}=\beta_j\}} m_j\abs{\lambda} \\
    &=\sum_{j\in J}\sum_{\lambda^{n_j}=\beta_j} m_j\abs{\lambda} \\
    &=\sum_{j\in J}m_jn_j\abs{\beta_j}^{1/n_j}<\infty.
  \end{align*}
  In particular, each \(M_{\mathcal D}(\lambda)\) is finite and
  \(\Sigma_{\mathcal D}\) is countable.

  The blockwise Fourier transforms therefore assemble to a unitary
  equivalence between \(L_{\mathcal D}\) and the diagonal operator
  \[
    \bigoplus_{\lambda\in\Sigma_{\mathcal D}}
      \lambda I_{M_{\mathcal D}(\lambda)}.
  \]
  For any finite truncation \(F\subset J\), the corresponding finite model
  has determinant
  \[
    \prod_{\lambda\in\Sigma_{\mathcal D,F}}
      (1-\lambda z)^{M_{\mathcal D,F}(\lambda)},
  \]
  obtained by grouping equal eigenvalues after diagonalization. By
  \Cref{lem:cyclic-product-trace-norm-limit}, these finite determinants
  converge locally uniformly to \(\det(I-zL_{\mathcal D})\). This gives the
  displayed determinant formula and the equivalent multiset statement.
\end{proof}

\subsection{Schatten determinants}
\label{subsec:schatten-regularized-cyclic}

The trace-class threshold above has an exact analogue for cyclic-block
models in every Schatten class. The determinant must then be replaced by
the regularized Fredholm determinant of integer order. For
\(1<p<\infty\), put \(r=\lceil p\rceil\); thus
\(\mathcal S_p\subseteq\mathcal S_r\), and \(\det_r(I-\cdot)\) is defined
on \(\mathcal S_p\). The following normalization and chain proposition are
part of the theorem sequence for the Schatten extension: they specify which
entire factors change under regularization and which zero-divisor data are
unchanged.

\begin{definition}[Regularized determinant normalization]
  \label{def:regularized-determinant-normalization}
  For an integer \(r\ge1\) and a finite-dimensional operator \(A\), this
  paper uses
  \[
    \det_r(I-zA)
    :=
    \det(I-zA)
    \exp\!\left(\sum_{q=1}^{r-1}\frac{z^q}{q}\operatorname{tr}(A^q)\right),
  \]
  with the empty sum interpreted as \(0\) when \(r=1\). This is the sign
  convention corresponding to
  \(\det_r(I+A)=\det(I+A)\exp(\sum_{q=1}^{r-1}(-1)^q\operatorname{tr}(A^q)/q)\)
  applied to \(A=-zT\).
\end{definition}

\begin{lemma}[Regularized determinant extension and cyclic-block factor]
  \label{lem:regularized-determinant-extension-cyclic-factor}
  The normalization of \Cref{def:regularized-determinant-normalization}
  extends to \(A\in\mathcal S_r\) as the standard regularized Fredholm
  determinant of order \(r\). It is the locally uniform limit, along
  finite-rank approximants converging in \(\mathcal S_r\), of the finite-rank
  expression in \Cref{def:regularized-determinant-normalization}; in
  particular, if \(T_n\to T\) in \(\mathcal S_r\), then
  \(\det_r(I-zT_n)\to\det_r(I-zT)\) locally uniformly in \(z\). Moreover
  \(\det_1(I-zA)=\det(I-zA)\) on trace-class operators, and for a finite
  cyclic block satisfying \(B^n=\beta I\),
  \[
    \det_r(I-zB)
    =(1-\beta z^n)
      \exp\!\left(\sum_{1\le q\le (r-1)/n}
        \frac{\beta^q z^{qn}}{q}\right).
  \]
\end{lemma}

\begin{proof}
  The extension and continuity statements are the standard construction and
  continuity theorem for regularized determinants; see Simon's construction in
  \cite[Chapter~9, Theorems~9.2--9.3]{Simon2005TraceIdeals}. The precise
  finite-rank normalization is
  \cite[eq.~(2.80), pp.~462--463]{GesztesyLatushkinMitreaZinchenko2005}, and
  local uniform continuity under finite-rank approximation in the Schatten
  norm is recorded in
  \cite[Lemma~2.14, eqs.~(2.77)--(2.79), p.~462]{GesztesyLatushkinMitreaZinchenko2005},
  with the source references there to Gohberg--Krein
  \cite[pp.~162--163]{GohbergKrein1969NonSelfAdjoint} and Simon
  \cite[Chapter~9]{Simon2005TraceIdeals}. The case \(r=1\) is the ordinary
  Fredholm determinant by the empty-sum convention.

  For the cyclic block, the eigenvalues are the \(n\)-th roots of \(\beta\),
  so \(\det(I-zB)=1-\beta z^n\). Also
  \(\operatorname{tr}(B^k)=0\) unless \(n\mid k\), while
  \(\operatorname{tr}(B^{qn})=n\beta^q\). Substitution in the preceding
  normalization gives the displayed factor after the cancellation of the
  factor \(n\) against the denominator \(qn\).
\end{proof}

\begin{proposition}[Regularized zero-divisor chain for cyclic blocks]
  \label{prop:regularized-zero-divisor-chain}
  Let \(r\ge1\), let \(\mathcal D=\{(n_j,\beta_j,m_j)\}_{j\in J}\) satisfy
  \[
    \sum_{j\in J}m_jn_j\abs{\beta_j}^{p/n_j}<\infty
  \]
  for some \(1<p\le r\), and let \(P_F^{(r)}\) be the finite-subset products
  determined by \Cref{lem:regularized-determinant-extension-cyclic-factor}.
  Then, for
  every finite \(F\Subset J\), the zero divisor of \(P_F^{(r)}
  \) on \(\CC\) is
  \[
    \sum_{j\in F}m_j\sum_{\lambda^{n_j}=\beta_j}[\lambda^{-1}],
  \]
  and the locally uniform limit supplied by
  \Cref{lem:regularized-cyclic-finite-subset-products} has finite zero order at
  \(z=\lambda^{-1}\) equal to
  \[
    M_{\mathcal D}(\lambda)
    =\sum_{\{j\in J:\lambda^{n_j}=\beta_j\}}m_j
  \]
  for every \(\lambda\in\CC^\times\). Consequently, for the cyclic model
  \(L_{\mathcal D}\), the normalization of
  \Cref{def:regularized-determinant-normalization} changes only the
  zero-free entire factor in the determinant formula; the regularized
  determinant \(\det_r(I-zL_{\mathcal D})\) and the ordinary Fredholm
  determinant in the trace-class case have the same prescribed non-zero
  cyclic zero divisor.
\end{proposition}

\begin{proof}
  Fix a finite set \(F\Subset J\). By
  \Cref{lem:regularized-determinant-extension-cyclic-factor}, the \(j\)-th finite
  cyclic contribution to \(P_F^{(r)}\) is
  \[
    (1-\beta_jz^{n_j})^{m_j}
    \exp\!\left(
      m_j\sum_{1\le q\le(r-1)/n_j}
        \frac{\beta_j^qz^{qn_j}}{q}
    \right).
  \]
  The exponential factor is everywhere non-zero. Hence the zeros of this
  contribution are exactly the roots of \(1-\beta_jz^{n_j}\), namely
  \(z=\lambda^{-1}\) with \(\lambda^{n_j}=\beta_j\), each with order
  \(m_j\). Multiplying over the finite set \(F\) gives the displayed finite
  zero divisor.

  Now fix \(\lambda\in\CC^\times\). The Schatten budget implies local
  finiteness of the collapsed support away from zero. Indeed, for
  \(\eta=\abs{\lambda}/2\), only finitely many cyclic roots can satisfy
  \(\abs{\mu}\ge\eta\), because otherwise the summable quantity
  \(\sum_jm_jn_j\abs{\beta_j}^{p/n_j}\) would dominate an infinite sum of
  positive terms at least \(\eta^p\). In particular
  \(M_{\mathcal D}(\lambda)<\infty\), and after shrinking a disc around
  \(\lambda^{-1}\) no zero from the finite set of other roots with
  \(\abs{\mu}\ge\eta\) lies in the disc.

  On this disc, divide the finite products by
  \((1-\lambda z)^{M_{\mathcal D}(\lambda)}\). The factors belonging to
  indices with \(\lambda^{n_j}=\beta_j\) then have non-zero value at
  \(\lambda^{-1}\), and all remaining factors are zero-free on the disc.
  Their logarithms still form the uniformly absolutely convergent family
  provided by \Cref{lem:regularized-cyclic-finite-subset-products}. The
  quotient therefore converges locally uniformly to a holomorphic function
  whose value at \(\lambda^{-1}\) is non-zero. The limiting zero order is
  exactly \(M_{\mathcal D}(\lambda)\).

  Finally, for \(L_{\mathcal D}\),
  \Cref{lem:regularized-cyclic-finite-subset-products} gives the locally
  uniform regularized product and determinant continuity identifies it with
  \(\det_r(I-zL_{\mathcal D})\). Since the preceding paragraphs show that
  all exponential corrections are zero-free and only the factors
  \((1-\beta_jz^{n_j})^{m_j}\) contribute zeros, the regularized determinant
  has exactly the same prescribed non-zero cyclic zero divisor as the
  ordinary Fredholm determinant formula when \(r=1\).
\end{proof}

\begin{lemma}[Finite-subset convention for regularized cyclic products]
  \label{lem:regularized-cyclic-finite-subset-products}
  Let \(1<p\le r\) with \(r\in\NN\), and let
  \(\mathcal D=\{(n_j,\beta_j,m_j)\}_{j\in J}\) satisfy
  \[
    \sum_{j\in J}m_jn_j\abs{\beta_j}^{p/n_j}<\infty.
  \]
  For \(F\Subset J\), define
  \[
    P_F^{(r)}(z):=
    \prod_{j\in F}(1-\beta_jz^{n_j})^{m_j}
    \exp\!\left(
      m_j\sum_{1\le q\le (r-1)/n_j}
        \frac{\beta_j^qz^{qn_j}}{q}
    \right).
  \]
  Then \(P_F^{(r)}\) converges locally uniformly on \(\CC\), as
  \(F\Subset J\), to an entire function independent of the enumeration of
  \(J\). More precisely, for every \(R>0\), all but finitely many
  indices satisfy \(R\abs{\beta_j}^{1/n_j}\le1/2\), and for such \(j\)
  and \(\abs z\le R\),
  \[
    \left|
      \log\!\left[
        (1-\beta_jz^{n_j})^{m_j}
        \exp\!\left(
          m_j\sum_{1\le q\le (r-1)/n_j}
            \frac{\beta_j^qz^{qn_j}}{q}
        \right)
      \right]
    \right|
    \le C_{p,r,R}\,m_jn_j\abs{\beta_j}^{p/n_j}.
  \]
\end{lemma}

\begin{proof}
  Put \(a_j=\abs{\beta_j}^{1/n_j}\). The Schatten budget implies
  \(a_j\to0\). Fix \(R>0\), and discard the finite set for which
  \(Ra_j>1/2\). On the remaining tail, use the principal logarithm. The
  logarithm of the \(j\)-th regularized factor is
  \[
    m_j\log(1-\beta_jz^{n_j})
    +m_j\sum_{1\le q\le(r-1)/n_j}
      \frac{\beta_j^qz^{qn_j}}{q}
    =
    -m_j\sum_{\substack{q\ge1\\ qn_j\ge r}}
      \frac{\beta_j^qz^{qn_j}}{q}.
  \]
  Since \(qn_j\ge r\ge p\) on this tail and \(Ra_j\le1/2\),
  \[
    \sum_{\substack{q\ge1\\ qn_j\ge r}}
      \frac{(Ra_j)^{qn_j}}{q}
    \le
    C_{p,r,R}\,n_j a_j^p
  \]
  after increasing \(C_{p,r,R}\) to cover the finitely many possibilities
  for \(n_j<r\); if \(n_j\ge r\), the first remaining exponent is already
  \(n_j\ge p\). Multiplying by \(m_j\) gives the displayed estimate. The
  right side is summable by hypothesis. Hence, on each disk, the tail
  logarithms form an unconditionally and uniformly absolutely convergent
  family. The finite-subset logarithmic sums therefore satisfy the Cauchy
  criterion uniformly on compact sets, their limit is holomorphic, and
  exponentiation gives locally uniform convergence of \(P_F^{(r)}\) to an
  entire function. Unconditional convergence of the logarithmic family
  also proves that the limit is independent of the enumeration of \(J\).
\end{proof}

\begin{theorem}[Exact Schatten membership threshold for the cyclic model]
  \label{thm:schatten-cyclic-model-membership}
  Let \(1<p<\infty\), let \(r=\lceil p\rceil\), and let
  \(\mathcal D=\{(n_j,\beta_j,m_j)\}_{j\in J}\) be countable cyclic data.
  The integer \(r\) is recorded only for the later regularized
  determinant statement; the present theorem classifies membership of the
  cyclic model in the smaller class \(\mathcal S_p\), not existence or
  uniqueness of \(\det_r\) on its natural domain \(\mathcal S_r\).
  Choose roots \(\alpha_j^{n_j}=\beta_j\), and form the cyclic direct sum
  \(L_{\mathcal D}=\bigoplus_j(C(n_j,\alpha_j)\otimes I_{m_j})\) whenever
  it is defined as a compact operator. Then
  \[
    L_{\mathcal D}\in\mathcal S_p
    \quad\Longleftrightarrow\quad
    \sum_{j\in J}m_jn_j\abs{\beta_j}^{p/n_j}<\infty,
  \]
  and in this case
  \[
    \norm{L_{\mathcal D}}_p^p
    =\sum_{j\in J}m_jn_j\abs{\beta_j}^{p/n_j}.
  \]
\end{theorem}

\begin{proof}
  On the \(j\)-th cyclic block the singular values are all
  \(\abs{\alpha_j}=\abs{\beta_j}^{1/n_j}\), with multiplicity
  \(m_jn_j\). Hence the finite block sums are Cauchy in
  \(\mathcal S_p\) exactly when
  \[
    \sum_jm_jn_j\abs{\beta_j}^{p/n_j}<\infty,
  \]
  and the same computation gives the formula for
  \(\norm{L_{\mathcal D}}_p^p\).
\end{proof}

\begin{theorem}[Regularized cyclic determinant on a prescribed \(\mathcal S_p\) budget]
  \label{thm:schatten-cyclic-regularized-determinant}
  Let \(1<p<\infty\), let \(r=\lceil p\rceil\), and let
  \(\mathcal D=\{(n_j,\beta_j,m_j)\}_{j\in J}\) be countable cyclic data
  satisfying
  \[
    \tau_{\mathcal D,p}
    :=\sum_{j\in J}m_jn_j\abs{\beta_j}^{p/n_j}<\infty .
  \]
  This is a prescribed cyclic-data theorem in the smaller class
  \(\mathcal S_p\); it does not classify all regularized determinant fibers in
  the natural \(\mathcal S_r\)-domain.  Choose roots
  \(\alpha_j^{n_j}=\beta_j\), and let
  \(L_{\mathcal D}\) be the cyclic model from
  \Cref{thm:schatten-cyclic-model-membership}. Then the finite-subset
  regularized cyclic products
  \(P_F^{(r)}\) from
  \Cref{lem:regularized-cyclic-finite-subset-products} converge locally
  uniformly on \(\CC\), independently of the enumeration of \(J\), to
  \[
    \det_r(I-zL_{\mathcal D})
    =
    \prod_{j\in J}
      (1-\beta_jz^{n_j})^{m_j}
      \exp\!\left(
        m_j\sum_{1\le q\le (r-1)/n_j}
          \frac{\beta_j^qz^{qn_j}}{q}
      \right).
  \]
  Equivalently, if \(p\) is an integer, this is the sharp
  \(\det_p\)-formula
  \[
    \det_p(I-zL_{\mathcal D})
    =
    \prod_{j\in J}
      (1-\beta_jz^{n_j})^{m_j}
      \exp\!\left(
        m_j\sum_{1\le q\le (p-1)/n_j}
          \frac{\beta_j^qz^{qn_j}}{q}
      \right).
  \]

  Conversely, if \(T\in\mathcal S_p(H)\) and the finite products on the
  right-hand side above, directed by finite subsets of \(J\), converge
  locally uniformly to \(\det_r(I-zT)\), then
  \[
    \sum_{j\in J}m_jn_j\abs{\beta_j}^{p/n_j}
    \le \norm{T}_p^p.
  \]
  Thus the displayed Schatten budget is the exact threshold for
  \(\mathcal S_p\)-membership of the normal cyclic model and the sharp
  lower bound in the prescribed cyclic-product regularized determinant
  problem attached to \(\mathcal D\). This is a prescribed-budget statement;
  it is not a classification of existence of \(\det_r\) on its full natural
  domain \(\mathcal S_r\).
\end{theorem}

\begin{proof}
  By \Cref{thm:schatten-cyclic-model-membership},
  \(L_{\mathcal D}\in\mathcal S_p\) and
  \(\norm{L_{\mathcal D}}_p^p=\tau_{\mathcal D,p}\). Since \(p\le r\),
  convergence in
  \(\mathcal S_p\) implies convergence in \(\mathcal S_r\), so the
  regularized determinant is continuous along these finite truncations in
  the precise sense fixed in
  \Cref{lem:regularized-determinant-extension-cyclic-factor}: convergence in
  \(\mathcal S_r\) gives locally uniform convergence of
  \(z\mapsto\det_r(I-zT)\). This is the standard continuity theorem for
  regularized determinants; in the form used here it is
  \cite[Lemma~2.14, eqs.~(2.77)--(2.79), p.~462]{GesztesyLatushkinMitreaZinchenko2005},
  based on Gohberg--Krein \cite[pp.~162--163]{GohbergKrein1969NonSelfAdjoint}
  and Simon \cite[Chapter~9, Theorems~9.2--9.3]{Simon2005TraceIdeals}.
  For determinant product formulas in this normalization see also
  Britz--Carey--Gesztesy--Nichols--Sukochev--Zanin
  \cite[eq.~(1.3) and the product theorem, eq.~(D.38A)]{BritzCareyGesztesyNicholsSukochevZanin2021}.

  For a finite-dimensional operator \(A\), the regularized determinant is
  \[
    \det_r(I-zA)
    =\det(I-zA)
      \exp\!\left(\sum_{\ell=1}^{r-1}
        \frac{z^\ell\Tr(A^\ell)}{\ell}\right).
  \]
  Applying this to one block
  \(B=C(n,\alpha)\otimes I_m\), with \(\beta=\alpha^n\), gives
  \[
    \det(I-zB)=(1-\beta z^n)^m.
  \]
  Also
  \[
    \Tr(B^\ell)
    =m\alpha^\ell\sum_{\omega^n=1}\omega^\ell
    =
    \begin{cases}
      mn\,\beta^q,& \ell=qn,\\
      0,& n\nmid \ell.
    \end{cases}
  \]
  Therefore
  \[
    \det_r(I-zB)
    =(1-\beta z^n)^m
      \exp\!\left(
        m\sum_{1\le q\le (r-1)/n}
          \frac{\beta^qz^{qn}}{q}
      \right).
  \]
  Multiplying over finite subsets gives the stated finite products.

  The locally uniform convergence of the finite regularized products is
  exactly \Cref{lem:regularized-cyclic-finite-subset-products}. Determinant
  continuity in \(\mathcal S_r\) identifies that locally uniform limit with
  \(\det_r(I-zL_{\mathcal D})\).

  Finally assume \(T\in\mathcal S_p\) has the stated product-side
  regularized determinant limit. The exponential corrections in the finite
  cyclic factors are non-vanishing entire functions, so they do not affect
  zero orders. At \(z=\lambda^{-1}\), each factor with
  \(\lambda^{n_j}=\beta_j\) contributes exactly \(m_j\) to the vanishing
  order, and all other cyclic factors are non-zero there. The same
  derivative argument used in
  \Cref{lem:product-side-cyclic-divisor-multiplicity} applies verbatim to
  the finite regularized products: an infinite contribution would make
  \(\det_r(I-zT)\) vanish to arbitrarily high finite order, contradicting
  \(\det_r(I)=1\), while a finite contribution forces vanishing order at
  least \(M_{\mathcal D}(\lambda)\). The standard zero-divisor theorem for
  regularized Fredholm determinants in this normalization is the
  determinant-divisor equality encoded in
  \cite[Lemma~2.15, eq.~(2.85), p.~463]{GesztesyLatushkinMitreaZinchenko2005},
  with determinant construction from \cite[Chapter~9]{Simon2005TraceIdeals}
  and \cite[Section~4.2, pp.~162--163]{GohbergKrein1969NonSelfAdjoint}; it
  says that zeros of \(\det_r(I-zT)\) are exactly reciprocals of the
  non-zero eigenvalues of \(T\), with the same algebraic multiplicities.
  Hence every root
  \(\lambda\) of the cyclic divisor contributes at least
  \(M_{\mathcal D}(\lambda)\) to the algebraic multiplicity of \(T\).
  Weyl's eigenvalue--singular-value majorisation, in its Schatten form
  \[
    \sum_k\abs{\lambda_k(T)}^p\le\sum_ks_k(T)^p=\norm{T}_p^p,
  \]
  due to Weyl and Fan
  \cite{Weyl1949EigenvalueInequalities,Fan1951MaximumProperties}, then
  yields
  \[
    \sum_{j\in J}m_jn_j\abs{\beta_j}^{p/n_j}
    =\sum_{\lambda\ne0}\abs{\lambda}^pM_{\mathcal D}(\lambda)
    \le\norm{T}_p^p.
  \]
  This proves the converse lower bound and the sharpness follows from the
  cyclic model.
\end{proof}

\begin{corollary}[Controlled scope of the regularized cyclic package]
  \label{cor:regularized-controlled-extension-scope}
  Let \(1<p<\infty\), put \(r=\lceil p\rceil\), and let
  \(\mathcal D\) be countable cyclic data satisfying
  \[
    \sum_{j\in J}m_jn_j\abs{\beta_j}^{p/n_j}<\infty .
  \]
  Then the regularized product formula of
  \Cref{thm:schatten-cyclic-regularized-determinant} is a controlled
  extension of the trace-class Fredholm determinant realization theorem in
  the following precise sense.
  \begin{enumerate}[label=\textup{(\roman*)}]
    \item It first asserts the smaller-budget membership
    \(L_{\mathcal D}\in\mathcal S_p\) and computes
    \(\det_r(I-zL_{\mathcal D})\) for that model.
    \item Its converse lower bound applies only to operators
    \(T\in\mathcal S_p\) whose regularized determinant is obtained as the
    locally uniform prescribed cyclic product limit appearing in the
    theorem.
    \item It does not assert that every operator in the natural
    \(\mathcal S_r\)-domain of \(\det_r\), or every possible
    \(\det_r\)-zero divisor in that domain, is classified by the
    \(\mathcal S_p\)-budget.
  \end{enumerate}
\end{corollary}

\begin{proof}
  Assertion (i) is exactly
  \Cref{thm:schatten-cyclic-model-membership} together with the first half
  of \Cref{thm:schatten-cyclic-regularized-determinant}. Assertion (ii) is
  the converse half of
  \Cref{thm:schatten-cyclic-regularized-determinant}, whose hypotheses
  explicitly include \(T\in\mathcal S_p\) and the stated locally uniform
  product-side realization of \(\det_r(I-zT)\). Assertion (iii) follows
  because \(\mathcal S_p\subseteq\mathcal S_r\) is a proper strengthening
  when \(p<r\), and because the theorem never removes that stronger
  membership hypothesis from the converse statement. The explicit
  boundary example in \Cref{prop:regularized-domain-boundary} shows that
  this distinction is substantive, not merely terminological.
\end{proof}

\begin{proposition}[Regularized determinant domain boundary]
  \label{prop:regularized-domain-boundary}
  Let \(1<p<\infty\), put \(r=\lceil p\rceil\), and let
  \(\mathcal D\) be cyclic data satisfying
  \[
    \sum_{j\in J}m_jn_j\abs{\beta_j}^{p/n_j}<\infty .
  \]
  The conclusions of
  \Cref{thm:schatten-cyclic-regularized-determinant} are conclusions in
  the smaller Schatten class \(\mathcal S_p\): they give membership of the
  normal cyclic model in \(\mathcal S_p\), the explicit
  \(\det_r\)-factorization for that model, and the sharp lower bound for
  \(\mathcal S_p\)-operators whose regularized determinant has the
  prescribed cyclic product limit. They do not classify all operators in
  the natural determinant domain \(\mathcal S_r\), nor all
  \(\det_r\)-fibers with the same zero divisor but no prescribed cyclic
  product presentation.

  More precisely, whenever \(p<r\) there are cyclic data for which
  \[
    \sum_jm_jn_j\abs{\beta_j}^{r/n_j}<\infty,
    \qquad
    \sum_jm_jn_j\abs{\beta_j}^{p/n_j}=\infty .
  \]
  For those data the normal
  cyclic model belongs to \(\mathcal S_r\), and hence has a well-defined
  \(\det_r\), but it is not in \(\mathcal S_p\) and is outside the
  \(\mathcal S_p\)-budget theorem.
\end{proposition}

\begin{proof}
  The first paragraph is exactly the content of
  \Cref{thm:schatten-cyclic-regularized-determinant}: its hypotheses and
  lower bound are stated for \(T\in\mathcal S_p\), while the regularized
  determinant \(\det_r\) is defined on the larger class \(\mathcal S_r\).
  Thus the theorem supplies a minimization and rigidity statement after
  the stronger \(\mathcal S_p\)-membership assumption has been imposed.

  Assume now that \(p<r\). Choose
  \(J=\{j\ge2\}\), \(n_j=m_j=1\), and
  \(\beta_j=j^{-1/p}\). Then
  \[
    \sum_{j\ge2}m_jn_j\abs{\beta_j}^{p/n_j}
    =\sum_{j\ge2}j^{-1}=\infty,
  \]
  whereas
  \[
    \sum_{j\ge2}m_jn_j\abs{\beta_j}^{r/n_j}
    =\sum_{j\ge2}j^{-r/p}<\infty
  \]
  because \(r/p>1\). By
  \Cref{thm:schatten-cyclic-model-membership}, the corresponding cyclic
  diagonal model is in \(\mathcal S_r\) and is not in \(\mathcal S_p\).
  The standard regularized determinant construction therefore gives
  \(\det_r(I-zL_{\mathcal D})\) on the natural \(\mathcal S_r\)-domain,
  but the \(\mathcal S_p\)-budget appearing in
  \Cref{thm:schatten-cyclic-regularized-determinant} is infinite. Hence no
  conclusion of that theorem can be read as a classification of the whole
  \(\mathcal S_r\)-domain.
\end{proof}
