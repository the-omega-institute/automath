\subsection{Spectral rigidity}

The determinant rigidity lemma used here was stated before
\Cref{thm:normal-realizability-threshold}. In the next theorem, the lower
bound itself is the immediate combination of that lemma with the classical
Weyl/Ky Fan inequality; the additional content is the equality and
near-equality analysis, which shows how the trace-norm defect controls
the Schur off-diagonal entries of a triangular representative.

\begin{lemma}[Ordered Weyl--Ky Fan--Lidskii inequality for trace class]
  \label{lem:weyl-ky-fan-trace-class}
Let \(T\in\sone(H)\). Let \((\lambda_k(T))_{k\ge1}\) be the eigenvalues
of \(T\), repeated according to algebraic multiplicity and ordered so
that
\[
  \abs{\lambda_1(T)}\ge \abs{\lambda_2(T)}\ge \cdots,
\]
padding by zeros once the non-zero spectrum is exhausted. Let
\((s_k(T))_{k\ge1}\) be the singular values of \(T\), ordered so that
\[
  s_1(T)\ge s_2(T)\ge \cdots \ge 0.
\]
Then for every \(N\ge1\),
\[
  \sum_{k=1}^N \abs{\lambda_k(T)}
  \le \sum_{k=1}^N s_k(T).
\]
Consequently,
\[
  \sum_{k\ge1}\abs{\lambda_k(T)}\le \|T\|_1.
\]
\end{lemma}

\begin{proof}
This is the classical Weyl--Ky Fan majorisation inequality for compact
operators, in the trace-class form descended from Lidskii's trace theorem:
the partial sums of the moduli of the eigenvalues, ordered decreasingly
and counted with algebraic multiplicity, are dominated by the corresponding
partial sums of the singular values. See Lidskii's original trace theorem
\cite[p.~485]{Lidskii1959}, K\"onig's Weyl-inequality chapter
\cite{Weyl1949EigenvalueInequalities,Fan1951MaximumProperties},
K\"onig's Weyl-inequality chapter
\cite[Chapter~2, Section~2.a]{Konig1986EigenvalueDistribution}, and the
treatment of \(s\)-numbers and Weyl--Ky Fan inequalities in
\cite[Chapter~1]{Simon2005TraceIdeals} and
\cite[Chapter~VI, Section~1]{GohbergGoldbergKaashoek1990ClassesI}.
Since \(T\in\sone(H)\), the
singular values are summable and
\(\sum_{k\ge1}s_k(T)=\|T\|_1\). Passing to the monotone limit in the
partial-sum inequality therefore yields the final claim.
\end{proof}

\begin{lemma}[Ringrose triangularisation with prescribed diagonal]
  \label{lem:ringrose-prescribed-diagonal}
  Let \(T\) be compact on a Hilbert space \(H\). After replacing \(H\)
  by the reducing subspace
  \[
    H_T:=\overline{\mathrm{span}}\{u_r,v_r:r\ge1\}
  \]
  coming from any singular-value expansion
  \(T=\sum_{r\ge1}s_r(T)\langle\,\cdot\,,u_r\rangle v_r\), the operator
  acts on a separable Hilbert space and vanishes on \(H_T^\perp\).
  On this separable reducing part there is an orthonormal basis
  \((e_k)_{k\ge1}\) in which the matrix of \(T\) is upper triangular and
  the diagonal entries are precisely the non-zero eigenvalues of \(T\),
  repeated according to algebraic multiplicity, followed by zeros.
\end{lemma}

\begin{proof}
  The separable reduction follows directly from the singular-value
  expansion: both \(H_T\) and \(H_T^\perp\) reduce \(T\), and
  \(T|_{H_T^\perp}=0\). Ringrose's compact triangular representation
  theorem for compact operators on a separable Hilbert space gives a
  maximal triangularizing chain whose successive one-dimensional quotients
  carry exactly the characteristic values of the compact operator,
  counted according to algebraic multiplicity, with zero repeated after
  the non-zero characteristic values have been exhausted; see
  \cite[Chapter~1, Theorem~3, pp.~6--8]{Ringrose1971CompactNonSelfAdjoint}.
  Applying this theorem to \(T|_{H_T}\) gives the stated orthonormal
  Schur basis and prescribed diagonal.
\end{proof}

\begin{lemma}[Two-by-two trace-norm pinching identity]
  \label{lem:two-by-two-trace-norm-pinching}
  For \(a,b,c\in\CC\), let
  \[
    B=\begin{pmatrix}a&c\\0&b\end{pmatrix}.
  \]
  Then
  \[
    \|B\|_1^2=(\abs{a}+\abs{b})^2+\abs{c}^2.
  \]
  In particular,
  \[
    \|B\|_1\ge \abs{a}+\abs{b},
  \]
  and equality holds if and only if \(c=0\).
\end{lemma}

\begin{proof}
  Let \(\sigma_1,\sigma_2\ge0\) be the singular values of \(B\). Then
  \[
    \|B\|_1^2=(\sigma_1+\sigma_2)^2
    =\sigma_1^2+\sigma_2^2+2\sigma_1\sigma_2.
  \]
  The squared singular values are the eigenvalues of \(B^*B\), so
  \[
    \sigma_1^2+\sigma_2^2=\Tr(B^*B)
    =\abs{a}^2+\abs{b}^2+\abs{c}^2,
  \]
  and
  \[
    \sigma_1\sigma_2=\sqrt{\det(B^*B)}
    =\abs{\det B}
    =\abs{ab}.
  \]
  Therefore
  \[
    \|B\|_1^2
    =\abs{a}^2+\abs{b}^2+\abs{c}^2+2\abs{a}\,\abs{b}
    =(\abs{a}+\abs{b})^2+\abs{c}^2.
  \]
  The equality characterization follows immediately, since the square
  root is strictly increasing on \([0,\infty)\).
\end{proof}

\begin{lemma}[Two-by-two Schatten pinching rigidity]
  \label{lem:two-by-two-schatten-pinching}
  Let \(1<p<\infty\). For \(a,b,c\in\CC\), put
  \[
    B=\begin{pmatrix}a&c\\0&b\end{pmatrix}
  \]
  and define
  \[
    \Phi_p(a,b,c):=\|B\|_p^p-\abs a^p-\abs b^p .
  \]
  Then
  \[
    \Phi_p(a,b,c)\ge0,
  \]
  Equality holds if and only if \(c=0\). For fixed \(a,b\), if
  \[
    \Phi_p(a,b,c_\nu)\longrightarrow0,
  \]
  then \(c_\nu\to0\).
\end{lemma}

\begin{proof}
  Let \(\sigma_1\ge\sigma_2\ge0\) be the singular values of \(B\), and let
  \(\mu_1\ge\mu_2\ge0\) be the decreasing rearrangement of
  \(\abs a,\abs b\). The two eigenvalues of \(B\) are \(a\) and \(b\), so
  Weyl--Ky Fan majorisation gives
  \[
    (\mu_1,\mu_2)\prec_w(\sigma_1,\sigma_2).
  \]
  Since \(t\mapsto t^p\) is increasing and strictly convex on
  \([0,\infty)\), weak majorisation implies
  \[
    \mu_1^p+\mu_2^p\le \sigma_1^p+\sigma_2^p=\|B\|_p^p.
  \]
  This is exactly \(\Phi_p(a,b,c)\ge0\).

  If equality holds in the displayed inequality, the strict convexity
  equality case for weak majorisation forces
  \((\sigma_1,\sigma_2)=(\mu_1,\mu_2)\). Therefore
  \[
    \abs a^2+\abs b^2+\abs c^2
    =\Tr(B^*B)
    =\sigma_1^2+\sigma_2^2
    =\mu_1^2+\mu_2^2
    =\abs a^2+\abs b^2,
  \]
  so \(c=0\). The converse is immediate because \(B\) is then diagonal.
  The final assertion follows from continuity of the singular values as
  functions of the entries of a \(2\times2\) matrix.
\end{proof}

\begin{lemma}[Trace-norm convergence of Schur compressions]
  \label{lem:trace-norm-schur-compressions}
  Let \(T\in\sone(H)\), and let \((P_N)_{N\ge1}\) be finite-rank
  orthogonal projections with \(P_N\to I\) strongly. Then
  \[
    P_NTP_N\longrightarrow T
    \qquad\text{in }\sone\text{-norm}.
  \]
\end{lemma}

\begin{proof}
  Fix \(\varepsilon>0\), and choose a finite-rank operator \(F\) with
  \(\|T-F\|_1<\varepsilon\). Since multiplication by contractions is
  contractive on \(\sone\),
  \[
    \|T-P_NTP_N\|_1
    \le 2\|T-F\|_1+\|F-P_NFP_N\|_1.
  \]
  It remains to treat \(F\). Write \(F=\sum_{\ell=1}^r x_\ell\otimes
  y_\ell\), where \((x\otimes y)h=\langle h,y\rangle x\). For rank-one
  operators, \(\|x\otimes y\|_1=\|x\|\,\|y\|\), and therefore
  \[
    \|x_\ell\otimes y_\ell-P_Nx_\ell\otimes P_Ny_\ell\|_1
    \le \|x_\ell-P_Nx_\ell\|\,\|y_\ell\|
      +\|P_Nx_\ell\|\,\|y_\ell-P_Ny_\ell\|.
  \]
  Strong convergence \(P_N\to I\) makes the right-hand side tend to zero
  for each \(\ell\). Hence \(P_NFP_N\to F\) in trace norm, and the displayed
  estimate gives \(\limsup_N\|T-P_NTP_N\|_1\le 2\varepsilon\). Since
  \(\varepsilon\) is arbitrary, the claim follows.
\end{proof}

\begin{lemma}[Schatten-norm convergence of Schur compressions]
  \label{lem:schatten-schur-compressions}
  Let \(1\le p<\infty\), let \(T\in\mathcal S_p(H)\), and let
  \((P_N)_{N\ge1}\) be finite-rank orthogonal projections with
  \(P_N\to I\) strongly. Then
  \[
    P_NTP_N\longrightarrow T
    \qquad\text{in }\mathcal S_p\text{-norm}.
  \]
\end{lemma}

\begin{proof}
  The proof is the same finite-rank approximation argument as in
  \Cref{lem:trace-norm-schur-compressions}. Given \(\varepsilon>0\),
  choose a finite-rank operator \(F\) with \(\|T-F\|_p<\varepsilon\).
  Multiplication by contractions is contractive on every Schatten class,
  and hence
  \[
    \|T-P_NTP_N\|_p
    \le 2\|T-F\|_p+\|F-P_NFP_N\|_p.
  \]
  Writing \(F\) as a finite sum of rank-one operators
  \(x_\ell\otimes y_\ell\), and using
  \(\|x\otimes y\|_p=\|x\|\,\|y\|\), strong convergence of \(P_N\) gives
  \(P_NFP_N\to F\) in \(\mathcal S_p\)-norm. Letting
  \(\varepsilon\downarrow0\) proves the claim.
\end{proof}

\begin{theorem}[Global trace-norm minimum and minimizer classification in a Fredholm-determinant fiber]
  \label{thm:determinant-fiber-trace-minimality}
  Let \(\mathcal D=\{(n_j,\beta_j,m_j)\}_{j\in J}\) satisfy the
  summability hypothesis of \Cref{thm:cyclic-fredholm-realisation}, and
  write
  \[
    D_{\mathcal D}(z)
    :=\prod_{j\in J}(1-\beta_j z^{n_j})^{m_j}.
  \]
  Let \(M_{\mathcal D}\) and \(\Sigma_{\mathcal D}\) be the collapsed
  multiplicity function and its support from
  \Cref{prop:collapsed-spectral-multiplicities}. If \(T\in \sone(H)\)
  satisfies
  \[
    \det(I-zT)=D_{\mathcal D}(z)
  \]
  as a holomorphic germ at \(z=0\), then
  \[
    \|T\|_1
    \ge \sum_{\lambda\in\Sigma_{\mathcal D}}
      \abs{\lambda}M_{\mathcal D}(\lambda)
    =\sum_{j\in J}m_j n_j\abs{\beta_j}^{1/n_j}.
  \]
  The cyclic model \(L_{\mathcal D}\) from
  \Cref{thm:cyclic-fredholm-realisation} attains equality. Consequently,
  \[
    \inf\bigl\{\|T\|_1:T\in \sone,\ \det(I-zT)=D_{\mathcal D}(z)
      \text{ near }0\bigr\}
    =\sum_{j\in J}m_j n_j\abs{\beta_j}^{1/n_j}.
  \]
  Moreover, equality
  \[
    \|T\|_1
    =\sum_{j\in J}m_j n_j\abs{\beta_j}^{1/n_j}
  \]
  holds if and only if \(T\) is normal and there exist a Hilbert space
  \(H_0\) and a unitary \(U:H\to H_0\oplus H_{\mathcal D}\) such that
  \[
    UTU^{-1}=0_{H_0}\oplus L_{\mathcal D}.
  \]
  Thus the canonical cyclic model is the entire trace-norm minimizer class
  in the full determinant fiber, not only in the normal category.
\end{theorem}

\begin{proof}
  By \Cref{lem:fredholm-spectral-rigidity}, the non-zero eigenvalues of
  \(T\), counted with algebraic multiplicity, are exactly the non-zero
  eigenvalues encoded by \(D_{\mathcal D}\). Equivalently, by
  \Cref{prop:collapsed-spectral-multiplicities}, the non-zero eigenvalue
  multiset of \(T\) is the multiset in which each
  \(\lambda\in\Sigma_{\mathcal D}\) appears with multiplicity
  \(M_{\mathcal D}(\lambda)\).

  Let \((\lambda_k(T))_{k\ge1}\) be the eigenvalues of \(T\), repeated
  according to algebraic multiplicity and ordered so that
  \[
    \abs{\lambda_1(T)}\ge \abs{\lambda_2(T)}\ge \cdots,
  \]
  with zeros appended after the non-zero spectrum, and let
  \((s_k(T))_{k\ge1}\) be the singular values ordered so that
  \(s_1(T)\ge s_2(T)\ge \cdots\). By the ordered Weyl--Ky Fan lemma just
  stated, for every \(N\ge1\),
  \[
    \sum_{k=1}^N \abs{\lambda_k(T)}
    \le \sum_{k=1}^N s_k(T);
  \]
  Passing to the monotone limit \(N\to\infty\) yields
  \[
    \sum_{k\ge1}\abs{\lambda_k(T)}
    \le \sum_{k\ge1}s_k(T)
    =\|T\|_1.
  \]
  Applying this to the fixed eigenvalue multiset above gives
  \[
    \|T\|_1
    \ge \sum_{\lambda\in\Sigma_{\mathcal D}}
      \abs{\lambda}M_{\mathcal D}(\lambda).
  \]
  The second equality is the summability identity from
  \Cref{prop:collapsed-spectral-multiplicities}.

  For the cyclic model \(L_{\mathcal D}\), normality implies that its
  singular values are exactly the moduli of its non-zero eigenvalues,
  counted with multiplicity. Hence
  \[
    \|L_{\mathcal D}\|_1
    =\sum_{\lambda\in\Sigma_{\mathcal D}}
      \abs{\lambda}M_{\mathcal D}(\lambda)
    =\sum_{j\in J}m_j n_j\abs{\beta_j}^{1/n_j}.
  \]
  The last equality is again
  \Cref{prop:collapsed-spectral-multiplicities}; equivalently, one may
  read off the same formula from
  \Cref{prop:sharp-trace-budget-support-obstruction}.
  Hence the lower bound is sharp and equals the stated infimum.

  Suppose now that equality holds:
  \[
    \|T\|_1
    =\sum_{\lambda\in\Sigma_{\mathcal D}}
      \abs{\lambda}M_{\mathcal D}(\lambda).
  \]
  Choose a singular-value expansion
  \[
    T=\sum_{r\ge1}s_r(T)\,\langle \,\cdot\,,u_r\rangle v_r
  \]
  with orthonormal sets \((u_r)\), \((v_r)\). Then
  \[
    H_T:=\overline{\mathrm{span}}\{u_r,v_r:r\ge1\}
  \]
  is separable and reduces \(T\), while \(T\) vanishes on \(H_T^\perp\).
  Replacing \(H\) by \(H_T\) and \(T\) by \(T|_{H_T}\) therefore loses no
  information, so we may assume from now on that \(H\) is separable.

  Applying \Cref{lem:ringrose-prescribed-diagonal} to \(T\), let
  \((e_k)_{k\ge1}\) be such a basis and let \(A=[a_{ij}]\) be the matrix
  of \(T\) in this basis. Since those non-zero eigenvalues are exactly the
  multiset encoded by
  \(D_{\mathcal D}\),
  \[
    \sum_{k\ge1}\abs{a_{kk}}
    =\sum_{\lambda\in\Sigma_{\mathcal D}}
      \abs{\lambda}M_{\mathcal D}(\lambda)
    =\|T\|_1.
  \]

  Let \(P_N\) be the orthogonal projection onto
  \(\mathrm{span}\{e_1,\ldots,e_N\}\), and set \(A_N:=P_NTP_N\). Since
  \(P_N\to I\) strongly, \Cref{lem:trace-norm-schur-compressions} gives
  \(A_N\to T\) in trace norm, so
  \[
    \|A_N\|_1\longrightarrow \|T\|_1.
  \]
  Also \(A_N\) is upper triangular with diagonal
  \(a_{11},\dots,a_{NN}\), hence
  \[
    \delta_N
    :=\|A_N\|_1-\sum_{k=1}^N\abs{a_{kk}}
    \ge0
  \]
  and
  \[
    \delta_N\longrightarrow0.
  \]

  Fix \(1\le i<j\), and let \(N\ge j\). Write \(R_{ij}\) for the
  projection onto \(\mathrm{span}\{e_i,e_j\}\) and
  \(S_{ij}:=P_N-R_{ij}\). On the finite-dimensional space \(P_NH\), set
  \(U_{ij}:=R_{ij}-S_{ij}\), which is unitary. Then
  \[
    R_{ij}A_NR_{ij}\oplus S_{ij}A_NS_{ij}
    =\frac12(A_N+U_{ij}A_NU_{ij}),
  \]
  so the triangle inequality and unitary invariance of \(\|\cdot\|_1\)
  give the pinching estimate
  \[
    \|A_N\|_1
    \ge \|R_{ij}A_NR_{ij}\|_1+\|S_{ij}A_NS_{ij}\|_1.
  \]
  Because \(A_N\) is upper triangular,
  \[
    R_{ij}A_NR_{ij}
    =
    \begin{pmatrix}
      a_{ii} & a_{ij} \\
      0 & a_{jj}
    \end{pmatrix},
  \]
  while \(S_{ij}A_NS_{ij}\) is upper triangular with diagonal entries
  \((a_{kk})_{1\le k\le N,\ k\neq i,j}\). Applying the basic inequality
  \(\sum |\lambda_k|\le \|\cdot\|_1\) to the second block therefore yields
  \[
    \delta_N
    \ge
    \left\|
      \begin{pmatrix}
        a_{ii} & a_{ij} \\
        0 & a_{jj}
      \end{pmatrix}
    \right\|_1
    -\abs{a_{ii}}-\abs{a_{jj}}.
  \]

  Applying the two-by-two pinching identity
  \Cref{lem:two-by-two-trace-norm-pinching} to the displayed matrix
  \(B_{ij}:=R_{ij}A_NR_{ij}\) gives
  \[
    \abs{a_{ij}}^2
    =
    \bigl(\|B_{ij}\|_1+\abs{a_{ii}}+\abs{a_{jj}}\bigr)
    \bigl(\|B_{ij}\|_1-\abs{a_{ii}}-\abs{a_{jj}}\bigr)
    \le
    2\|A_N\|_1\,\delta_N .
  \]
  Since \(A_N\to T\) in trace norm and \(\delta_N\to0\), this quantitative
  estimate forces \(a_{ij}=0\).
  As \(i<j\) were arbitrary, every strict upper-triangular entry of \(A\)
  vanishes. Thus \(A\) is diagonal in an orthonormal basis, so \(T\) is
  normal.

  Now \(T\) is a normal trace-class operator with
  \(\det(I-zT)=D_{\mathcal D}(z)\). Applying
  \Cref{thm:cyclic-fredholm-realisation} yields a Hilbert space \(H_0\)
  and a unitary \(U:H\to H_0\oplus H_{\mathcal D}\) such that
  \[
    UTU^{-1}=0_{H_0}\oplus L_{\mathcal D}.
  \]
  The converse implication is immediate from the already proved equality
  \(\|L_{\mathcal D}\|_1=\sum_jm_jn_j\abs{\beta_j}^{1/n_j}\).
\end{proof}

\begin{corollary}[Schur-entry stability for trace-norm near-minimizers]
  \label{cor:schur-entry-stability}
  Under the hypotheses of
  \Cref{thm:determinant-fiber-trace-minimality}, let \(T\in\sone(H)\)
  satisfy
  \[
    \det(I-zT)=D_{\mathcal D}(z)
  \]
  near \(0\), and put
  \[
    \tau_{\mathcal D}
    :=\sum_{\lambda\in\Sigma_{\mathcal D}}
      \abs{\lambda}M_{\mathcal D}(\lambda).
  \]
  Choose the Ringrose basis from
  \Cref{lem:ringrose-prescribed-diagonal}, ordered so that the diagonal
  entries are the prescribed eigenvalues followed by zeros, and write
  \(A=[a_{ij}]\) for the resulting upper triangular matrix. For
  \(P_N\) the projection onto
  \(\operatorname{span}\{e_1,\ldots,e_N\}\), set
  \[
    \delta_N(T):=\|P_NTP_N\|_1-\sum_{k=1}^N\abs{a_{kk}}.
  \]
  Then \(\delta_N(T)\ge0\),
  \[
    \limsup_{N\to\infty}\delta_N(T)
    \le \|T\|_1-\tau_{\mathcal D},
  \]
  and for every \(1\le i<j\le N\),
  \[
    \abs{a_{ij}}^2
    \le 2\|P_NTP_N\|_1\,\delta_N(T).
  \]
  In particular, if \(\|T\|_1\le\tau_{\mathcal D}+\varepsilon\), then for
  each fixed \(i<j\),
  \[
    \abs{a_{ij}}
    \le \sqrt{2(\tau_{\mathcal D}+\varepsilon)\varepsilon}
    \qquad(\varepsilon\downarrow0 \text{ along any near-minimizing family}).
  \]
\end{corollary}

\begin{proof}
  The proof is the quantitative part of the preceding argument, without
  setting the final defect equal to zero. The compressions
  \(P_NTP_N\to T\) in trace norm by
  \Cref{lem:trace-norm-schur-compressions}, and the prescribed diagonal
  is absolutely summable with total \(\tau_{\mathcal D}\). Hence
  \[
    \limsup_{N\to\infty}\delta_N(T)
    \le \|T\|_1-\tau_{\mathcal D}.
  \]
  The pinching estimate applied to the \(i,j\) coordinates of
  \(P_NTP_N\) gives
  \[
    \delta_N(T)
    \ge
    \left\|
      \begin{pmatrix}a_{ii}&a_{ij}\\0&a_{jj}\end{pmatrix}
    \right\|_1-\abs{a_{ii}}-\abs{a_{jj}}.
  \]
  Using \Cref{lem:two-by-two-trace-norm-pinching} and
  \(\|B\|_1+\abs{a_{ii}}+\abs{a_{jj}}\le2\|P_NTP_N\|_1\) yields the
  displayed entry estimate. If
  \(\|T\|_1\le\tau_{\mathcal D}+\varepsilon\), then
  \(\|P_NTP_N\|_1\le\|T\|_1\) and, after taking \(N\to\infty\) with
  \(N\ge j\), the bound for the fixed entry follows.
\end{proof}

\begin{lemma}[Collapsed multiplicities under a Schatten budget]
  \label{lem:schatten-collapsed-multiplicities}
  Let \(1<p<\infty\), and let
  \(\mathcal D=\{(n_j,\beta_j,m_j)\}_{j\in J}\) satisfy
  \[
    n_j,m_j\in\NN,\qquad
    \beta_j\in\CC\setminus\{0\},\qquad
    \sum_{j\in J}m_jn_j\abs{\beta_j}^{p/n_j}<\infty .
  \]
  Define
  \[
    M_{\mathcal D}(\lambda)
    :=\sum_{\{j\in J:\lambda^{n_j}=\beta_j\}}m_j,
    \qquad
    \Sigma_{\mathcal D}
    :=\{\lambda\neq0:M_{\mathcal D}(\lambda)\neq0\}.
  \]
  Then each \(M_{\mathcal D}(\lambda)\) is finite,
  \(\Sigma_{\mathcal D}\) is countable, and
  \[
    \sum_{\lambda\in\Sigma_{\mathcal D}}
      \abs{\lambda}^pM_{\mathcal D}(\lambda)
    =\sum_{j\in J}m_jn_j\abs{\beta_j}^{p/n_j}.
  \]
  Moreover the non-zero eigenvalue multiset of the cyclic
  \(\mathcal S_p\)-model \(L_{\mathcal D}\) is exactly the multiset in
  which \(\lambda\) occurs with multiplicity \(M_{\mathcal D}(\lambda)\).
\end{lemma}

\begin{proof}
  Choose roots \(\alpha_j^{n_j}=\beta_j\). The block
  \(C(n_j,\alpha_j)\otimes I_{m_j}\) has eigenvalues
  \(\alpha_j\omega\), \(\omega^{n_j}=1\), each repeated \(m_j\) times.
  Hence it contributes to the \(\lambda\)-eigenspace precisely when
  \(\lambda^{n_j}=\beta_j\), and the multiplicity contributed is \(m_j\).
  This gives the stated multiset description.

  Since every solution of \(\lambda^{n_j}=\beta_j\) has
  \(\abs{\lambda}^p=\abs{\beta_j}^{p/n_j}\), summing over the multiset
  gives
  \[
    \sum_{\lambda\in\Sigma_{\mathcal D}}
      \abs{\lambda}^pM_{\mathcal D}(\lambda)
    =
    \sum_{j\in J}\sum_{\lambda^{n_j}=\beta_j}
      m_j\abs{\lambda}^p
    =
    \sum_{j\in J}m_jn_j\abs{\beta_j}^{p/n_j}.
  \]
  The right-hand side is finite, so no non-zero \(\lambda\) can have
  infinite multiplicity and the support is countable.
\end{proof}

\begin{theorem}[Schatten minimizer rigidity for regularized determinant fibers]
  \label{thm:schatten-determinant-fiber-minimality}
  Let \(1<p<\infty\), put \(r=\lceil p\rceil\), and let
  \(\mathcal D=\{(n_j,\beta_j,m_j)\}_{j\in J}\) satisfy
  \[
    \tau_{\mathcal D,p}
    :=\sum_{j\in J}m_jn_j\abs{\beta_j}^{p/n_j}<\infty .
  \]
  Choose roots \(\alpha_j^{n_j}=\beta_j\), and let
  \(L_{\mathcal D}\) be the cyclic \(\mathcal S_p\)-model from
  \Cref{thm:schatten-cyclic-regularized-determinant}. If
  \(T\in\mathcal S_p(H)\) satisfies
  \[
    \det_r(I-zT)=\det_r(I-zL_{\mathcal D})
  \]
  as entire functions, then
  \[
    \|T\|_p^p\ge \tau_{\mathcal D,p}.
  \]
  Equality holds if and only if \(T\) is normal and there exist a Hilbert
  space \(H_0\) and a unitary \(U:H\to H_0\oplus H_{\mathcal D}\) such
  that
  \[
    UTU^{-1}=0_{H_0}\oplus L_{\mathcal D}.
  \]
  Thus the cyclic model is the unique \(\mathcal S_p\)-norm minimizer
  class in the full regularized determinant fiber.

  More quantitatively, let \(M_{\mathcal D}\) and
  \(\Sigma_{\mathcal D}\) be the collapsed multiplicity function and its
  support from \Cref{lem:schatten-collapsed-multiplicities}. Choose the
  Ringrose basis from
  \Cref{lem:ringrose-prescribed-diagonal}, ordered so that the diagonal
  entries are the prescribed non-zero eigenvalues, repeated according to
  \(M_{\mathcal D}\), followed by zeros, and write \(A=[a_{ij}]\). For
  \(P_N\) the projection onto
  \(\operatorname{span}\{e_1,\ldots,e_N\}\), set
  \[
    \Delta_{N,p}(T)
    :=\|P_NTP_N\|_p^p-\sum_{k=1}^N\abs{a_{kk}}^p .
  \]
  Then \(\Delta_{N,p}(T)\ge0\),
  \[
    \limsup_{N\to\infty}\Delta_{N,p}(T)
    \le \|T\|_p^p-\tau_{\mathcal D,p},
  \]
  and for every \(1\le i<j\le N\),
  \[
    \left\|
      \begin{pmatrix}
        a_{ii} & a_{ij}\\
        0 & a_{jj}
      \end{pmatrix}
    \right\|_p^p-\abs{a_{ii}}^p-\abs{a_{jj}}^p
    \le \Delta_{N,p}(T).
  \]
  Consequently, along any family in this regularized determinant fiber
  with \(\|T\|_p^p\downarrow\tau_{\mathcal D,p}\), every fixed Schur
  off-diagonal entry tends to \(0\).
\end{theorem}

\begin{proof}
  In finite dimension, regularization multiplies the ordinary determinant
  by a non-vanishing entire exponential factor. Hence the zeros of the
  regularized determinant, with multiplicities, are precisely the
  reciprocals of the non-zero eigenvalues, counted with algebraic
  multiplicity. The identity
  \[
    \det_r(I-zT)=\det_r(I-zL_{\mathcal D})
  \]
  and \Cref{lem:schatten-collapsed-multiplicities} therefore show that the
  non-zero eigenvalue multiset of \(T\) is exactly the multiset in which
  \(\lambda\in\Sigma_{\mathcal D}\) occurs
  \(M_{\mathcal D}(\lambda)\) times.

  Weyl--Ky Fan majorisation in Schatten form gives
  \[
    \sum_k\abs{\lambda_k(T)}^p
    \le \sum_ks_k(T)^p
    =\|T\|_p^p.
  \]
  The fixed eigenvalue multiset gives
  \[
    \sum_k\abs{\lambda_k(T)}^p
    =\sum_{\lambda\in\Sigma_{\mathcal D}}
      \abs{\lambda}^pM_{\mathcal D}(\lambda)
    =\sum_{j\in J}m_jn_j\abs{\beta_j}^{p/n_j}
    =\tau_{\mathcal D,p},
  \]
  The middle equality is \Cref{lem:schatten-collapsed-multiplicities}.
  This proves the lower bound.

  The normal cyclic model attains the bound by
  \Cref{thm:schatten-cyclic-regularized-determinant}. Suppose conversely
  that \(\|T\|_p^p=\tau_{\mathcal D,p}\). Apply
  \Cref{lem:ringrose-prescribed-diagonal} to the compact operator \(T\)
  after the separable reduction coming from a singular-value expansion.
  In the resulting upper triangular matrix \(A=[a_{ij}]\), the diagonal
  is the prescribed non-zero eigenvalue multiset followed by zeros, and
  therefore
  \[
    \sum_{k\ge1}\abs{a_{kk}}^p=\tau_{\mathcal D,p}=\|T\|_p^p.
  \]

  Let \(P_N\) project onto \(\operatorname{span}\{e_1,\ldots,e_N\}\). By
  \Cref{lem:schatten-schur-compressions},
  \[
    P_NTP_N\to T\quad\text{in }\mathcal S_p\text{-norm}.
  \]
  Thus
  \[
    \Delta_{N,p}(T)
    :=\|P_NTP_N\|_p^p-\sum_{k=1}^N\abs{a_{kk}}^p
    \longrightarrow0.
  \]
  Nonnegativity of \(\Delta_{N,p}(T)\) follows from the finite-dimensional
  Weyl--Ky Fan inequality applied to \(P_NTP_N\).

  Fix \(1\le i<j\), and take \(N\ge j\). Let \(R_{ij}\) be the projection
  onto \(\operatorname{span}\{e_i,e_j\}\) and \(S_{ij}:=P_N-R_{ij}\).
  With \(U_{ij}:=R_{ij}-S_{ij}\) on \(P_NH\), the block pinching identity
  \[
    R_{ij}P_NTP_NR_{ij}\oplus S_{ij}P_NTP_NS_{ij}
    =\frac12(P_NTP_N+U_{ij}P_NTP_NU_{ij})
  \]
  and convexity of the Schatten norm imply
  \[
    \|P_NTP_N\|_p^p
    \ge
    \|R_{ij}P_NTP_NR_{ij}\|_p^p
    +\|S_{ij}P_NTP_NS_{ij}\|_p^p.
  \]
  The second block is upper triangular with diagonal entries
  \((a_{kk})_{1\le k\le N,\ k\neq i,j}\), so the finite-dimensional
  Weyl--Ky Fan inequality gives
  \[
    \|S_{ij}P_NTP_NS_{ij}\|_p^p
    \ge
    \sum_{\substack{1\le k\le N\\k\neq i,j}}\abs{a_{kk}}^p.
  \]
  Since the first block is
  \[
    \begin{pmatrix}
      a_{ii} & a_{ij}\\
      0 & a_{jj}
    \end{pmatrix},
  \]
  the displayed Schur-entry defect inequality follows. By
  \Cref{lem:two-by-two-schatten-pinching}, the left-hand side is
  nonnegative and can vanish only when \(a_{ij}=0\). Since
  \(\Delta_{N,p}(T)\to0\), every fixed strict upper-triangular entry
  vanishes. Hence \(T\) is diagonal in an orthonormal basis, and therefore
  normal.

  A normal compact operator is classified, up to unitary equivalence and
  arbitrary zero summands, by its non-zero eigenvalues with multiplicity.
  Since this multiset is the one encoded by \(L_{\mathcal D}\), the
  claimed unitary equivalence follows. The converse equality implication
  has already been noted for \(L_{\mathcal D}\).

  Finally, the same compression argument without assuming equality gives
  the stated estimates. Indeed
  \(P_NTP_N\to T\) in \(\mathcal S_p\) and the prescribed diagonal has
  \(p\)-sum \(\tau_{\mathcal D,p}\), so
  \[
    \limsup_{N\to\infty}\Delta_{N,p}(T)
    \le \|T\|_p^p-\tau_{\mathcal D,p}.
  \]
  The pinching estimate above gives the displayed two-by-two defect
  bound. If a family has
  \(\|T\|_p^p-\tau_{\mathcal D,p}\to0\), then the corresponding
  two-by-two defects tend to \(0\) for every fixed pair \(i<j\), and
  \Cref{lem:two-by-two-schatten-pinching} forces the fixed Schur
  off-diagonal entries to tend to \(0\).
\end{proof}

\begin{remark}[What is classical and what is not in the determinant-fiber theorem]
  \label{rem:minimizer-classification-novelty}
  The lower bound in
  \Cref{thm:determinant-fiber-trace-minimality} is exactly the standard
  combination of \Cref{lem:fredholm-spectral-rigidity} with the classical
  inequality \(\sum_k\abs{\lambda_k(T)}\le \|T\|_1\). The additional
  trace-norm information is the quantitative pinching estimate in
  \Cref{cor:schur-entry-stability}: near-minimal defect forces every fixed
  Schur off-diagonal entry in the prescribed Ringrose triangularisation to
  be small, and zero defect forces all such entries to vanish. The normal
  classification theorem then identifies the equality case with the cyclic
  model up to zero summands. The Schatten theorem
  \Cref{thm:schatten-determinant-fiber-minimality} proves the analogous
  statement for regularized determinant fibers in every
  \(\mathcal S_p\), \(1<p<\infty\): equality in the sharp
  \(\mathcal S_p\)-budget is again rigid, and the same Schur-compression
  mechanism gives a quantitative near-minimizer constraint.
\end{remark}

Trace-norm minimality fixes the normal minimizers, but the determinant
still permits arbitrary nilpotent Jordan structure at each prescribed
non-zero eigenvalue, subject only to algebraic multiplicity. The next
result records this residual freedom and shows that it can be approached
at the sharp trace-norm bound, while it is attained only in the diagonal
case.

\begin{theorem}[Jordan profiles away from zero]
  \label{thm:nonnormal-jordan-classification}
  Let \(\mathcal D=\{(n_j,\beta_j,m_j)\}_{j\in J}\) satisfy the
  summability hypothesis of \Cref{thm:cyclic-fredholm-realisation}, and
  let \(L_{\mathcal D}\) be the associated block-cyclic model. Set
  \[
    D_{\mathcal D}(z):=\det(I-zL_{\mathcal D}).
  \]
  Let \(M_{\mathcal D}\) and \(\Sigma_{\mathcal D}\) be the collapsed
  multiplicity function and its support from
  \Cref{prop:collapsed-spectral-multiplicities}. For each
  \(\lambda\in \Sigma_{\mathcal D}\), let
  \[
    \pi_\lambda=(r_{\lambda,1},\dots,r_{\lambda,\ell_\lambda})
  \]
  be a finite tuple of positive integers. Then the following are
  equivalent:
  \begin{enumerate}[label=\textup{(\roman*)}]
    \item There exists \(T\in \sone(H)\) such that
      \[
        \det(I-zT)=D_{\mathcal D}(z)
      \]
      near \(z=0\), and for every \(\lambda\in \Sigma_{\mathcal D}\)
      the restriction of \(T\) to the generalized eigenspace at
      \(\lambda\) has Jordan block sizes exactly \(\pi_\lambda\).
    \item For every \(\lambda\in \Sigma_{\mathcal D}\),
      \[
        M_{\mathcal D}(\lambda)
        =\sum_{q=1}^{\ell_\lambda}r_{\lambda,q}.
      \]
  \end{enumerate}
  If these conditions hold, then
  \[
    \inf\Bigl\{\|T\|_1:T\text{ satisfies \textup{(i)}}\Bigr\}
    =\sum_{j\in J}m_jn_j\abs{\beta_j}^{1/n_j}.
  \]
  For this fixed profile \((\pi_\lambda)_{\lambda\in\Sigma_{\mathcal D}}\),
  this sharp infimum is attained precisely when every prescribed block has
  size one, i.e. precisely when \(r_{\lambda,q}=1\) for every
  \(\lambda\in \Sigma_{\mathcal D}\) and every
  \(1\le q\le \ell_\lambda\). If the fixed profile contains a nontrivial
  Jordan block, then the displayed bound is approached by the
  approximating operators constructed in the proof but is not attained
  within that fixed profile class.
\end{theorem}

\begin{proof}
  Assume \textup{(i)}. By \Cref{lem:fredholm-spectral-rigidity}, the
  non-zero spectrum of \(T\), counted with algebraic multiplicity, agrees
  with that of \(L_{\mathcal D}\). By
  \Cref{prop:collapsed-spectral-multiplicities}, the algebraic
  multiplicity of each \(\lambda\in \Sigma_{\mathcal D}\) in
  \(L_{\mathcal D}\) is \(M_{\mathcal D}(\lambda)\). On the
  finite-dimensional generalized eigenspace at \(\lambda\), the sum of
  the Jordan block sizes is exactly the algebraic multiplicity. Therefore
  \textup{(ii)} follows.

  Assume now that \textup{(ii)} holds. Let \(N_r\) denote the nilpotent
  Jordan block of size \(r\), with ones on the superdiagonal and zeros
  elsewhere. Since
  \[
    \sum_{\lambda\in \Sigma_{\mathcal D}}
      M_{\mathcal D}(\lambda)\abs{\lambda}
    =\sum_{j\in J}m_jn_j\abs{\beta_j}^{1/n_j}
    <\infty
  \]
  by \Cref{prop:collapsed-spectral-multiplicities}, the set
  \[
    \mathcal B
    :=\{(\lambda,q):\lambda\in\Sigma_{\mathcal D},
      \ 1\le q\le \ell_\lambda\}
  \]
  is countable. Fix \(\varepsilon>0\), and choose numbers
  \(\delta_{\lambda,q}>0\) such that
  \[
    \sum_{(\lambda,q)\in\mathcal B}\delta_{\lambda,q}<\varepsilon.
  \]
  For each \((\lambda,q)\in\mathcal B\), continuity of the trace norm on
  the finite-dimensional matrix algebra \(M_{r_{\lambda,q}}(\CC)\) yields
  a number \(t_{\lambda,q}\ge 0\) such that
  \[
    t_{\lambda,q}>0 \text{ if } r_{\lambda,q}\ge 2,
    \qquad
    \left\|\lambda I_{r_{\lambda,q}}
      +t_{\lambda,q}N_{r_{\lambda,q}}\right\|_1
    \le r_{\lambda,q}\abs{\lambda}+\delta_{\lambda,q}.
  \]
  Set
  \[
    B_{\lambda,q}
    :=\lambda I_{r_{\lambda,q}}+t_{\lambda,q}N_{r_{\lambda,q}}.
  \]
  If \(r_{\lambda,q}\ge 2\), then \(B_{\lambda,q}\) is similar to the
  Jordan block \(J_{r_{\lambda,q}}(\lambda)=\lambda I_{r_{\lambda,q}}
  +N_{r_{\lambda,q}}\), because
  \[
    D_t^{-1}J_r(\lambda)D_t=\lambda I_r+tN_r,
    \qquad
    D_t:=\operatorname{diag}(1,t,\dots,t^{r-1}),
  \]
  for every \(t>0\). If \(r_{\lambda,q}=1\), then \(B_{\lambda,q}\) is
  the \(1\times 1\) Jordan block \((\lambda)\).

  Define
  \[
    H_\varepsilon
    :=\bigoplus_{(\lambda,q)\in\mathcal B}\CC^{r_{\lambda,q}},
    \qquad
    T_\varepsilon
    :=\bigoplus_{(\lambda,q)\in\mathcal B}B_{\lambda,q}.
  \]
  The estimate above and \textup{(ii)} give
  \begin{align*}
    \|T_\varepsilon\|_1
    &\le \sum_{(\lambda,q)\in\mathcal B}
      \left(r_{\lambda,q}\abs{\lambda}+\delta_{\lambda,q}\right) \\
    &= \sum_{\lambda\in \Sigma_{\mathcal D}}
      M_{\mathcal D}(\lambda)\abs{\lambda}
      +\sum_{(\lambda,q)\in\mathcal B}\delta_{\lambda,q} \\
    &< \sum_{j\in J}m_jn_j\abs{\beta_j}^{1/n_j}+\varepsilon.
  \end{align*}
  Hence \(T_\varepsilon\in \sone(H_\varepsilon)\). By construction, for
  each \(\lambda\in \Sigma_{\mathcal D}\) the generalized eigenspace of
  \(T_\varepsilon\) at \(\lambda\) is the direct sum of the blocks
  \(B_{\lambda,q}\), so the Jordan block sizes are exactly
  \(\pi_\lambda\).

  Let \(\mathcal B_N\subset \mathcal B\) be an increasing exhaustion by
  finite sets, and set
  \[
    T_\varepsilon^{(N)}
    :=\bigoplus_{(\lambda,q)\in\mathcal B_N}B_{\lambda,q}.
  \]
  Then
  \[
    \|T_\varepsilon-T_\varepsilon^{(N)}\|_1\to 0.
  \]
  Since each \(B_{\lambda,q}\) is upper triangular with diagonal entry
  \(\lambda\), one has
  \[
    \det(I-zB_{\lambda,q})=(1-\lambda z)^{r_{\lambda,q}}.
  \]
  Therefore
  \[
    \det(I-zT_\varepsilon^{(N)})
    =\prod_{(\lambda,q)\in\mathcal B_N}(1-\lambda z)^{r_{\lambda,q}}.
  \]
  By trace-norm continuity of the Fredholm determinant
  \cite[Theorem~3.4]{Simon2005TraceIdeals},
  \[
    \det(I-zT_\varepsilon)
    =\prod_{\lambda\in\Sigma_{\mathcal D}}
      (1-\lambda z)^{M_{\mathcal D}(\lambda)}
  \]
  as a holomorphic germ at \(z=0\). The same product is
  \(\det(I-zL_{\mathcal D})\), by
  \Cref{prop:collapsed-spectral-multiplicities}. Thus
  \[
    \det(I-zT_\varepsilon)=D_{\mathcal D}(z)
  \]
  near \(z=0\). This proves \textup{(i)}.

  The construction above gives, for every \(\varepsilon>0\), an operator
  \(T_\varepsilon\) satisfying \textup{(i)} and
  \[
    \|T_\varepsilon\|_1
    <\sum_{j\in J}m_jn_j\abs{\beta_j}^{1/n_j}+\varepsilon.
  \]
  On the other hand, every operator satisfying \textup{(i)} obeys the
  lower bound from \Cref{thm:determinant-fiber-trace-minimality}.
  Consequently,
  \[
    \inf\Bigl\{\|T\|_1:T\text{ satisfies \textup{(i)}}\Bigr\}
    =\sum_{j\in J}m_jn_j\abs{\beta_j}^{1/n_j}.
  \]

  If \(r_{\lambda,q}=1\) for every \(\lambda\) and \(q\), then
  \(L_{\mathcal D}\) itself satisfies \textup{(i)} and attains the
  infimum by \Cref{thm:cyclic-fredholm-realisation}. Conversely, assume
  that the infimum is attained by some operator \(T\) satisfying
  \textup{(i)}. Then \(\|T\|_1\) equals the lower bound from
  \Cref{thm:determinant-fiber-trace-minimality}, so \(T\) is normal. For
  each \(\lambda\in \Sigma_{\mathcal D}\), the restriction of \(T\) to
  the generalized eigenspace at \(\lambda\) is a finite-dimensional normal
  operator with the single eigenvalue \(\lambda\), hence equals
  \(\lambda I\). Therefore every Jordan block has size \(1\).
\end{proof}

\begin{remark}[Freedom at zero]
  \label{rem:zero-spectrum-freedom}
  If \(Q\in \sone(K)\) is quasinilpotent, the trace-class eigenvalue
  product formula gives \(\det(I-zQ)\equiv 1\). Thus \(T\oplus Q\) has
  the same Fredholm determinant as \(T\).
  \Cref{thm:nonnormal-jordan-classification} therefore identifies the
  complete determinant-constrained Jordan geometry away from \(0\), while
  the quasinilpotent sector remains completely free.
\end{remark}

\begin{theorem}[Quasinilpotent stabilization preserves non-zero root-space invariants]
  \label{thm:quasinilpotent-stabilization-nonzero-invariants}
  Let \(T\in\sone(H)\), \(T'\in\sone(H')\), and let
  \(Q\in\sone(K)\), \(Q'\in\sone(K')\) be quasinilpotent. Suppose that
  there is a unitary
  \[
    U:H\oplus K\longrightarrow H'\oplus K'
  \]
  such that
  \[
    U(T\oplus Q)U^{-1}=T'\oplus Q'.
  \]
  Then for every \(\lambda\in\CC\setminus\{0\}\) and every \(k\ge1\),
  \(U\) maps
  \[
    \ker((T-\lambda I)^k)\oplus\{0\}
  \]
  unitarily onto
  \[
    \ker((T'-\lambda I)^k)\oplus\{0\},
  \]
  and the restricted root-space operators are unitarily equivalent:
  \[
    U\bigl((T-\lambda I)|_{\ker((T-\lambda I)^k)}\oplus0\bigr)U^{-1}
    =
    (T'-\lambda I)|_{\ker((T'-\lambda I)^k)}\oplus0 .
  \]
  Consequently the stabilized unitary-equivalence class preserves the
  Jordan block profile at every non-zero eigenvalue. In particular, a
  determinant-fiber element with a nontrivial Jordan block at some
  \(\lambda\neq0\) cannot become unitarily equivalent to the normal cyclic
  model \(L_{\mathcal D}\), even after adjoining arbitrary trace-class
  quasinilpotent summands at zero.
\end{theorem}

\begin{proof}
  Since \(Q\) and \(Q'\) are quasinilpotent,
  \[
    \sigma(Q)=\sigma(Q')=\{0\}.
  \]
  Hence \(Q-\lambda I\) and \(Q'-\lambda I\) are invertible for every
  \(\lambda\neq0\). Therefore, for every \(k\ge1\),
  \[
    \ker((T\oplus Q-\lambda I)^k)
    =
    \ker((T-\lambda I)^k)\oplus\{0\},
  \]
  and similarly
  \[
    \ker((T'\oplus Q'-\lambda I)^k)
    =
    \ker((T'-\lambda I)^k)\oplus\{0\}.
  \]
  The intertwining relation
  \(U(T\oplus Q)=(T'\oplus Q')U\) implies
  \[
    U(T\oplus Q-\lambda I)^k
    =(T'\oplus Q'-\lambda I)^kU.
  \]
  Thus \(U\) carries the first kernel onto the second. Because \(U\) is
  unitary on the ambient Hilbert spaces, its restriction is a unitary
  between these finite-dimensional root spaces with their inherited inner
  products.

  The same intertwining relation, restricted to the displayed kernels,
  gives the asserted unitary equivalence of the nilpotent root-space
  restrictions. For compact operators the root space at a non-zero
  eigenvalue is finite-dimensional and stabilizes after finitely many
  powers; the dimensions of the kernels
  \(\ker((T-\lambda I)^k)\) for \(k\ge1\), equivalently the unitary
  equivalence class of the restricted nilpotent operator on the stabilized
  root space, determine the Jordan block partition at \(\lambda\). Hence
  these profiles survive quasinilpotent stabilization at zero. The final
  assertion follows because the normal cyclic model has only one-by-one
  Jordan blocks at its non-zero eigenvalues.
\end{proof}

\begin{corollary}[Nonnormal determinant fibers are trace-norm unbounded]
  \label{cor:nonnormal-fiber-unbounded}
  Under the hypotheses of
  \Cref{thm:determinant-fiber-trace-minimality}, let
  \[
    \tau_{\mathcal D}
    :=\sum_{j\in J}m_j n_j\abs{\beta_j}^{1/n_j}.
  \]
  Then for every \(r>0\) there exists a trace-class nonnormal operator
  \(T_r\) such that
  \[
    \det(I-zT_r)=D_{\mathcal D}(z)
    \qquad\text{and}\qquad
    \|T_r\|_1=\tau_{\mathcal D}+r.
  \]
  In particular, the determinant fiber over \(D_{\mathcal D}\) is
  unbounded in trace norm once nonnormal realisations are allowed.
\end{corollary}

\begin{proof}
  Let \((e_n)_{n\ge1}\) be the standard orthonormal basis of
  \(\ell^2(\NN)\), and define the weighted shift \(Q\) by
  \[
    Qe_n:=2^{-n}e_{n+1}
    \qquad (n\ge1).
  \]
  Then \(Q^*Qe_n=4^{-n}e_n\), so \(|Q|e_n=2^{-n}e_n\) and hence
  \[
    \|Q\|_1=\Tr|Q|=\sum_{n\ge1}2^{-n}=1.
  \]
  Thus \(Q\in\sone\). Moreover,
  \[
    Q^m e_n
    =2^{-(n+(n+1)+\cdots+(n+m-1))}e_{n+m}
    =2^{-(mn+\frac{m(m-1)}2)}e_{n+m},
  \]
  whence
  \[
    \|Q^m\|
    \le 2^{-\frac{m(m+1)}2}
    \qquad (m\ge1).
  \]
  Therefore
  \[
    r(Q)=\lim_{m\to\infty}\|Q^m\|^{1/m}=0,
  \]
  so \(\sigma(Q)=\{0\}\). Since \(Q^m e_n\perp e_n\) for every \(m\ge1\)
  and every \(n\), one has \(\Tr(Q^m)=0\) for all \(m\ge1\). The
  Fredholm--Witt expansion from \Cref{prop:fredholm-witt-bridge} therefore gives
  \[
    \log\det(I-zQ)^{-1}
    =\sum_{m\ge1}\frac{\Tr(Q^m)}{m}z^m=0
  \]
  near \(z=0\), hence \(\det(I-zQ)=1\) on all of \(\CC\). Now set
  \(Q_r:=rQ\). Then \(Q_r\) is trace-class and nonnormal, with
  \[
    \sigma(Q_r)=\{0\},\qquad
    \det(I-zQ_r)=1,\qquad
    \|Q_r\|_1=r.
  \]
  Therefore
  \[
    T_r:=L_{\mathcal D}\oplus Q_r
  \]
  is trace-class and nonnormal, and multiplicativity of Fredholm
  determinants under direct sums gives
  \[
    \det(I-zT_r)
    =\det(I-zL_{\mathcal D})\det(I-zQ_r)
    =D_{\mathcal D}(z).
  \]
  Since trace norm is additive on orthogonal direct sums,
  \[
    \|T_r\|_1=\|L_{\mathcal D}\|_1+\|Q_r\|_1=\tau_{\mathcal D}+r,
  \]
  using \Cref{thm:determinant-fiber-trace-minimality} for
  \(\|L_{\mathcal D}\|_1=\tau_{\mathcal D}\).
\end{proof}

\begin{remark}[The same obstruction persists in the full determinant fiber]
  \label{rem:weighted-shift-obstruction}
  The concrete weighted-shift example was already given in
  \Cref{rem:nonnormal-obstruction-after-realisation}. The present
  corollary strengthens that observation: after adjoining the scaled
  quasinilpotent shifts \(Q_r=rQ\), one obtains nonnormal realisations of
  the same determinant with arbitrarily large trace norm. Thus the
  normal classifications in
  \Cref{thm:cyclic-fredholm-realisation,thm:cyclic-data-classification}
  cannot extend to all trace-class operators.
\end{remark}

\begin{theorem}[Determinant classification by collapsed spectral data]
  \label{thm:cyclic-data-classification}
  Let \(\mathcal D=\{(n_j,\beta_j,m_j)\}_{j\in J}\) and
  \(\mathcal D'=\{(n_k',\beta_k',m_k')\}_{k\in K}\) satisfy the summability
  hypothesis of \Cref{thm:cyclic-fredholm-realisation}, and choose roots
  \(\alpha_j^{n_j}=\beta_j\) and \((\alpha_k')^{n_k'}=\beta_k'\). Define
  \[
    M_{\mathcal D}(\lambda)
    :=\sum_{\{j\in J:\lambda^{n_j}=\beta_j\}} m_j,
    \qquad
    M_{\mathcal D'}(\lambda)
    :=\sum_{\{k\in K:\lambda^{n_k'}=\beta_k'\}} m_k',
  \]
  and write
  \(L_{\mathcal D}\), \(L_{\mathcal D'}\) for the associated cyclic models,
  and let
  \[
    \Lambda_{\mathcal D}
    :=
    \bigl\{\!\bigl\{\alpha_j\omega:
      j\in J,\ \omega^{n_j}=1,\text{ repeated }m_j\text{ times}
    \bigr\}\!\bigr\},
  \]
  with \(\Lambda_{\mathcal D'}\) defined analogously. Then the following are
  equivalent:
  \begin{enumerate}[label=\textup{(\roman*)}]
    \item \(\det(I-zL_{\mathcal D})=\det(I-zL_{\mathcal D'})\) as holomorphic
      germs at \(z=0\).
    \item \(M_{\mathcal D}(\lambda)=M_{\mathcal D'}(\lambda)\) for every
      \(\lambda\in\CC\setminus\{0\}\).
    \item \(\Lambda_{\mathcal D}=\Lambda_{\mathcal D'}\) as multisets.
    \item There exist Hilbert spaces \(H_0\), \(H_0'\) and a unitary
      \(V:H_0\oplus H_{\mathcal D}\to H_0'\oplus H_{\mathcal D'}\) such that
      \[
        V(0_{H_0}\oplus L_{\mathcal D})V^{-1}
        =0_{H_0'}\oplus L_{\mathcal D'}.
      \]
  \end{enumerate}
  In this situation the determinant germ determines, in the normal
  trace-class category, a unique block-cyclic realisation class up to zero
  summands. In particular,
  \[
    \dim \mathsf S(L_{\mathcal D})
    =\dim \mathsf S(L_{\mathcal D'}),
  \]
  with dimension interpreted as a Hilbert-space dimension. Since \(J\) and
  \(K\) are countable, this common value lies in
  \(\NN\cup\{\aleph_0\}\). Separately,
  \[
    \|L_{\mathcal D}\|_1=\|L_{\mathcal D'}\|_1,
    \qquad
    \sum_{j\in J}m_jn_j\abs{\beta_j}^{1/n_j}
    =\sum_{k\in K}m_k'n_k'\abs{\beta_k'}^{1/n_k'}.
  \]
\end{theorem}

\begin{proof}
  The implication \textup{(iv)}\(\Rightarrow\)\textup{(i)} is immediate,
  because unitary equivalence and zero summands do not change the Fredholm
  determinant. For \textup{(i)}\(\Rightarrow\)\textup{(ii)}, apply
  \Cref{lem:fredholm-spectral-rigidity} to \(L_{\mathcal D}\) and
  \(L_{\mathcal D'}\). Their non-zero spectra therefore agree with
  algebraic multiplicity. By
  \Cref{prop:collapsed-spectral-multiplicities}, the multiplicity of each
  \(\lambda\neq0\) in those spectra is exactly
  \(M_{\mathcal D}(\lambda)\) and \(M_{\mathcal D'}(\lambda)\),
  respectively. Hence \(M_{\mathcal D}=M_{\mathcal D'}\).

  The equivalence
  \textup{(ii)}\(\Leftrightarrow\)\textup{(iii)} is exactly the multiset
  reformulation supplied by
  \Cref{prop:collapsed-spectral-multiplicities}: \(\lambda\) appears in
  \(\Lambda_{\mathcal D}\) with multiplicity \(M_{\mathcal D}(\lambda)\),
  and similarly for \(\mathcal D'\).

  Assume \textup{(ii)}. By
  \Cref{prop:collapsed-spectral-multiplicities}, both cyclic models are
  unitarily equivalent to the same diagonal operator
  \[
    \bigoplus_{\lambda\in\Sigma}\lambda I_{M_{\mathcal D}(\lambda)},
    \qquad
    \Sigma:=\{\lambda\neq0:M_{\mathcal D}(\lambda)\neq0\}.
  \]
  After adjoining arbitrary zero summands, this gives \textup{(iv)}.

  The closing statements are the support-dimension and trace-norm formulas
  from \Cref{prop:sharp-trace-budget-support-obstruction} applied to either
  model. Thus
  \[
    \dim \mathsf S(L_{\mathcal D})=\dim \mathsf S(L_{\mathcal D'})
  \]
  is a Hilbert-space dimension identity with value in
  \(\NN\cup\{\aleph_0\}\), while
  \[
    \|L_{\mathcal D}\|_1=\|L_{\mathcal D'}\|_1
  \]
  and the displayed trace-budget equality are genuine numerical identities
  in \([0,\infty)\).
\end{proof}

\begin{corollary}[Determinant-intrinsic trace budget and minimizer class]
  \label{cor:determinant-intrinsic-trace-budget}
  Let \(\mathcal D\) and \(\mathcal D'\) be summable cyclic data as in
  \Cref{thm:cyclic-data-classification}. If
  \[
    \det(I-zL_{\mathcal D})=\det(I-zL_{\mathcal D'})
  \]
  as holomorphic germs at \(0\), then the determinant germ fixes the same
  trace budget in both presentations:
  \[
    \sum_{j\in J}m_jn_j\abs{\beta_j}^{1/n_j}
    =
    \sum_{k\in K}m_k'n_k'\abs{\beta_k'}^{1/n_k'}.
  \]
  Moreover this common number is the infimum of \(\|T\|_1\) over the full
  trace-class determinant fiber
  \[
    \{\,T\in\sone:\det(I-zT)=\det(I-zL_{\mathcal D})\text{ near }0\,\},
  \]
  and the equality cases are exactly the normal cyclic model class
  \(0_{H_0}\oplus L_{\mathcal D}\), equivalently
  \(0_{H_0'}\oplus L_{\mathcal D'}\), up to unitary equivalence and zero
  summands.
\end{corollary}

\begin{proof}
  By \Cref{thm:cyclic-data-classification}, equality of the determinant
  germs is equivalent to equality of the collapsed multiplicity functions
  \(M_{\mathcal D}=M_{\mathcal D'}\), and it gives unitary equivalence of
  the two cyclic models after adjoining zero summands. The same theorem
  gives the trace-budget identity displayed above. Applying
  \Cref{thm:determinant-fiber-trace-minimality} to either presentation of
  the common determinant germ identifies this common trace budget with the
  full trace-class infimum and classifies equality exactly as the normal
  cyclic model, with arbitrary zero summands. The unitary equivalence from
  \Cref{thm:cyclic-data-classification} makes the two descriptions of the
  minimizer class identical.
\end{proof}

\begin{remark}[Finite-dimensional specialization]
  \label{rem:realisability-vs-rigidity}
  Let \(A\) be a finite matrix and let \(T\in \sone\) satisfy
  \[
    \det(I-zT)=\det(I-zA).
  \]
  Then \(T\) and \(A\) have the same non-zero eigenvalues with
  algebraic multiplicity. Thus cyclic-block constructions may enlarge
  the ambient operator category, but they do not create spectral
  freedom once the determinant is prescribed. This is just
  \Cref{lem:fredholm-spectral-rigidity} applied to the finite-rank
  operator \(A\).
\end{remark}
