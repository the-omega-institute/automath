\section{Motivation from finite-state zeta functions}
\label{sec:preliminaries}

\begin{proposition}[Classical finite-state zeta determinant and Euler product]
  \label{thm:primitive-symbolic-exclusion}
  Let \(A\) be a finite square matrix with entries in \(\ZZ_{\ge0}\), regarded
  as the adjacency matrix of a finite directed graph with possible multiple
  edges; an entry equal to zero means that the corresponding directed edge is
  absent. Put \(a_n(A)=\Tr(A^n)\), and let
  \[
    p_n(A)=\frac{1}{n}\sum_{d\mid n}\mu(d)\,a_{n/d}(A)
  \]
  be the primitive closed-orbit counts. Equivalently,
  \begin{equation}\label{eq:trace-primitive-identity}
    a_n(A)=\sum_{d\mid n}d\,p_d(A).
  \end{equation}
  Then \(p_n(A)\in\ZZ_{\ge0}\) for every \(n\), and
  \[
    \zeta_A(z)
    :=\exp\!\left(\sum_{n\ge 1}\frac{a_n(A)}{n}z^n\right)
    =\det(I-zA)^{-1}
    =\prod_{n\ge 1}(1-z^n)^{-p_n(A)}
  \]
  as holomorphic germs at \(z=0\); equivalently, as formal power series.
  If \(\rho(A)>0\), the displayed identity is analytic for
  \(|z|<\rho(A)^{-1}\).
\end{proposition}

\begin{proof}
  The formula for \(p_n(A)\) is the usual necklace/Witt inversion formula
  \cite{MetropolisRota1983}. Combinatorially, \(a_n(A)=\Tr(A^n)\) counts
  based closed directed walks of length \(n\), with multiplicity when the
  graph has multiple edges. Each primitive closed orbit of least period
  \(d\) contributes exactly \(d\) based closed walks to \(a_n(A)\) whenever
  \(d\mid n\), and contributes none otherwise. This proves
  \eqref{eq:trace-primitive-identity}; Möbius inversion gives the displayed
  formula for \(p_n(A)\) and also proves that the numbers \(p_n(A)\) are
  non-negative integers.

  The determinant formula for shifts of finite type is classical; see
  Bowen--Lanford \cite{BowenLanford1970Zeta}, Ruelle's zeta-function work
  \cite{Ruelle1976ZetaExpanding,Ruelle1978Thermodynamic}, and
  Parry--Pollicott \cite{ParryPollicott1990Zeta}; for standard symbolic
  dynamics references see also \cite{ArtinMazur1965,LindMarcus1995}.
  Equivalently,
  in finite dimension one may sum \(-\log(1-z\lambda_i)\) over the
  eigenvalues \(\lambda_i\) of \(A\), counted algebraically, and use
  \(\sum_i\lambda_i^n=\Tr(A^n)\). This gives
  \[
    -\log\det(I-zA)
    =\sum_{n\ge1}\frac{a_n(A)}{n}z^n
  \]
  as a holomorphic identity for \(|z|<\rho(A)^{-1}\) when \(\rho(A)>0\),
  and as a formal identity in all cases. Substituting
  \eqref{eq:trace-primitive-identity} gives
  \[
    \sum_{n\ge1}\frac{a_n(A)}{n}z^n
    =\sum_{m\ge1}\sum_{r\ge1}\frac{p_m(A)}{r}z^{mr}
    =-\sum_{m\ge1}p_m(A)\log(1-z^m)
  \]
  as a germ at \(0\), which is precisely the Euler product.
\end{proof}

\begin{corollary}[Primitive symbolic Euler data are not trace-class summable]
  \label{cor:primitive-symbolic-summability-exclusion}
  Let \(A\) be a primitive \(0\)-\(1\) adjacency matrix for a finite
  one-sided subshift of finite type. Thus multiple edges are not encoded in
  this corollary. With \(p_n(A)\) as in
  \Cref{thm:primitive-symbolic-exclusion},
  \[
    \sum_{n\ge1} n\,p_n(A)<\infty
    \qquad\Longleftrightarrow\qquad
    A=[1].
  \]
\end{corollary}

\begin{proof}
  If \(A=[1]\), then \(a_n(A)=1\) for all \(n\), so \(p_1(A)=1\) and
  \(p_n(A)=0\) for \(n\ge2\). Conversely suppose \(A\ne[1]\). Under the
  present \(0\)-\(1\) SFT convention, a one-symbol primitive matrix is
  necessarily \([1]\); hence the alphabet has at least two symbols. Since
  \(A\) is primitive, some power \(A^k\) has strictly positive entries, and
  therefore every row sum of \(A^k\) is at least \(2\). Hence the Perron root
  satisfies \(\lambda^k=\rho(A^k)\ge2\),
  so \(\lambda>1\). Perron--Frobenius theory gives
  \(a_n(A)=\lambda^n+O(r^n)\) for some \(r<\lambda\), and therefore
  \(a_n(A)\) is unbounded. If \(\sum_n n\,p_n(A)<\infty\), then the
  non-negativity of the primitive counts and
  \eqref{eq:trace-primitive-identity} would give
  \(a_n(A)\le\sum_d d\,p_d(A)\) for all \(n\), a contradiction. Thus the
  summability condition holds only for \(A=[1]\).
\end{proof}

\begin{remark}[Role of the symbolic product]
  \label{rem:symbolic-product-role}
  For primitive finite matrices, the numbers \(p_n(A)\) are non-negative
  integers because they count primitive periodic orbits. This arithmetic
  feature belongs to the finite-state category. The proposition records the
  classical finite-state determinant/Euler-product identity, while
  \Cref{cor:primitive-symbolic-summability-exclusion} shows that,
  except for the one-symbol fixed-point shift, the primitive symbolic Euler
  product
  \[
    \prod_{n\ge1}(1-z^n)^{-p_n(A)}
  \]
  is used in this paper only as a germ-level motivation and not as a
  summable cyclic zero divisor for
  \Cref{thm:cyclic-fredholm-realisation}. For instance, for the full
  \(k\)-shift with \(k\ge2\), \(a_n(A)=k^n\),
  \(\zeta_A(z)=(1-kz)^{-1}\), and
  \(\sum_n n\,p_n(A)=\infty\).
\end{remark}
