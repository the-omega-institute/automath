\section{Finite-State Symbolic Motivation}
\label{sec:preliminaries}

\begin{proposition}[Primitive symbolic Euler data are not trace-class summable]
  \label{thm:primitive-symbolic-exclusion}
  Let \(A\) be a primitive adjacency matrix for a finite one-sided subshift
  of finite type. Put \(a_n(A)=\Tr(A^n)\), and let
  \[
    p_n(A)=\frac{1}{n}\sum_{d\mid n}\mu(d)\,a_{n/d}(A)
  \]
  be the primitive-orbit counts; equivalently,
  \begin{equation}\label{eq:trace-primitive-identity}
    a_n(A)=\sum_{d\mid n}d\,p_d(A).
  \end{equation}
  Then, for the Perron root \(\lambda\) of \(A\),
  \[
    \zeta_A(z)
    :=\exp\!\left(\sum_{n\ge 1}\frac{a_n(A)}{n}z^n\right)
    =\det(I-zA)^{-1}
    =\prod_{n\ge 1}(1-z^n)^{-p_n(A)}
  \]
  for every \(|z|<\lambda^{-1}\). The Euler product identity is an
  equality of holomorphic germs at \(z=0\); equivalently, it is an
  identity of formal power series. Moreover,
  \[
    \sum_{n\ge1} n\,p_n(A)<\infty
    \qquad\Longleftrightarrow\qquad
    A=[1].
  \]
\end{proposition}

\begin{proof}
  The formulas for \(p_n(A)\) and \eqref{eq:trace-primitive-identity} are
  the usual necklace/Witt inversion identities
  \cite{MetropolisRota1983}. The determinant formula for shifts of finite
  type is classical; see
  \cite{ArtinMazur1965,BowenLanford1970Zeta,LindMarcus1995}. Since \(A\)
  is finite-dimensional, it also follows by summing
  \(-\log(1-z\lambda_i)\) over the eigenvalues \(\lambda_i\) of \(A\) and
  using \(\sum_i\lambda_i^n=\Tr(A^n)\). Substituting
  \eqref{eq:trace-primitive-identity} gives
  \[
    \sum_{n\ge1}\frac{a_n(A)}{n}z^n
    =\sum_{m\ge1}\sum_{r\ge1}\frac{p_m(A)}{r}z^{mr}
    =-\sum_{m\ge1}p_m(A)\log(1-z^m)
  \]
  as a germ at \(0\), which is the stated Euler product.

  If \(A=[1]\), then \(a_n(A)=1\) for all \(n\), so \(p_1(A)=1\) and
  \(p_n(A)=0\) for \(n\ge2\). Conversely suppose \(A\ne[1]\). Since \(A\)
  is primitive, some power \(A^k\) has strictly positive entries; because
  the alphabet has at least two symbols, every row sum of \(A^k\) is at
  least \(2\). Hence the Perron root satisfies \(\lambda^k=\rho(A^k)\ge2\),
  so \(\lambda>1\). Perron--Frobenius theory gives
  \(a_n(A)=\lambda^n+O(r^n)\) for some \(r<\lambda\), and therefore
  \(a_n(A)\) is unbounded. If \(\sum_n n\,p_n(A)<\infty\), then the
  non-negativity of the primitive counts and
  \eqref{eq:trace-primitive-identity} would give
  \(a_n(A)\le\sum_d d\,p_d(A)\) for all \(n\), a contradiction. Thus the
  summability condition holds only for \(A=[1]\).
\end{proof}

\begin{remark}[Role of the symbolic product]
  \label{rem:symbolic-product-role}
  For primitive finite matrices, the numbers \(p_n(A)\) are non-negative
  integers because they count primitive periodic orbits. This arithmetic
  feature belongs to the finite-state category. The proposition shows that,
  except for the one-symbol fixed-point shift, the primitive symbolic Euler
  product
  \[
    \prod_{n\ge1}(1-z^n)^{-p_n(A)}
  \]
  is used in this paper only as a germ-level motivation and not as a
  summable cyclic zero divisor for
  \Cref{thm:cyclic-fredholm-realisation}. For instance, for the full
  \(k\)-shift with \(k\ge2\), \(a_n(A)=k^n\),
  \(\zeta_A(z)=(1-kz)^{-1}\), and
  \(\sum_n n\,p_n(A)=\infty\).
\end{remark}
