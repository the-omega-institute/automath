\subsection{Imported symbol criterion and the Poisson integer-orbit symbol}%
\label{subsec:lattice-background-poisson-symbol}

\paragraph{Imported shift-invariant-space criterion.}
This paragraph is the only abstract sampling input used in the integer-orbit
argument.  The background lemma records the classical principal
shift-invariant-space/Riesz-translate criterion under the hypotheses displayed
in its statement: a Toeplitz Gram matrix with $a\in\ell^1(\ZZ)$, a principal
symbol bounded above and below away from zero, and a restriction map whose
Riesz-coordinate action is the Gram operator
\cite{deBoorDeVoreRon1994,RonShen1995,Bownik2000,AldroubiGrochenig2001}; it is
compatible with the RKHS sampling and Galerkin frameworks of
\cite{NashedWalter1991,NashedSun2010,ChengJiangSun2016}.  The Poisson-specific
inputs are the closed strip-kernel symbol in
Proposition~\ref{prop:poisson-lattice-symbol} and the observation-channel
identification in Theorem~\ref{thm:poisson-channel-rkhs}.  The Riesz bounds,
cardinal Fourier formula and norm, and projection/quotient theorem then follow
as closed-form consequences of these inputs and the imported criterion; related
integer-shift and nod-function computations appear in
\cite{KiselevMininNovikov2016,KiselevMininNovikov2019}.

\smallskip
\begin{lemma}[Classical imported principal-symbol sampling criterion]%
\label{lem:imported-principal-symbol-sampling}
This is a background result in the principal shift-invariant-space and
RKHS sampling literature, used here in the form of
\cite{deBoorDeVoreRon1994,RonShen1995,Bownik2000,AldroubiGrochenig2001,NashedWalter1991,NashedSun2010,ChengJiangSun2016}.
Let $\mathcal H$ be a Hilbert space and let
$(\varphi_n)_{n\in\ZZ}$ be a family whose Gram matrix is Toeplitz,
\[
\langle\varphi_m,\varphi_n\rangle_{\mathcal H}=a_{m-n},
\qquad a\in\ell^1(\ZZ).
\]
Put
\[
\sigma(\theta):=\sum_{k\in\ZZ}a_ke^{-\ii k\theta},
\qquad -\pi\le\theta\le\pi,
\]
and suppose that $0<A\le\sigma(\theta)\le B<\infty$ a.e.  Then the
synthesis map
\[
C:c_{00}(\ZZ)\to\mathcal H,
\qquad Cc:=\sum_{n\in\ZZ}c_n\varphi_n,
\]
extends to a topological isomorphism from $\ell^2(\ZZ)$ onto the closed
orbit space $V:=\overline{\operatorname{span}}\{\varphi_n:n\in\ZZ\}$, with
\[
A\|c\|_{\ell^2}^2\le\|Cc\|_{\mathcal H}^2
\le B\|c\|_{\ell^2}^2.
\]
Let $G$ be the Toeplitz Gram operator on $\ell^2(\ZZ)$ with matrix
$(a_{m-n})_{m,n\in\ZZ}$.  The discrete Fourier transform diagonalizes
$G$ as multiplication by~$\sigma$, and $G^{-1}$ is multiplication by
$\sigma^{-1}$.  If, in addition, a bounded restriction map
$R:V\to\ell^2(\ZZ)$ satisfies $R(Cc)=Gc$ for all finitely supported $c$,
then $R$ is an isomorphism.  For every datum $\alpha\in\ell^2(\ZZ)$ the
unique interpolant $R^{-1}\alpha$ satisfies
\begin{equation}\label{eq:imported-reciprocal-symbol-norm}
\|R^{-1}\alpha\|_{\mathcal H}^2
=\frac1{2\pi}\int_{-\pi}^{\pi}
\frac{|\widetilde\alpha(\theta)|^2}{\sigma(\theta)}\dd\theta .
\end{equation}
The cardinal element $L:=R^{-1}\delta_0$ is synthesized by
$d=G^{-1}\delta_0$, equivalently $\widetilde d=\sigma^{-1}$ a.e.
Thus the statement supplies the functional-analytic passage from a positive
principal symbol to Riesz bounds, bounded interpolation, and the
reciprocal-symbol norm.  In the Poisson setting,
Proposition~\ref{prop:poisson-lattice-symbol} and
Theorem~\ref{thm:poisson-channel-rkhs} verify these displayed hypotheses for
the strip kernel and observation profiles
\cite{deBoorDeVoreRon1994,RonShen1995,AldroubiGrochenig2001}.
\end{lemma}
\begin{proof}
The standard Fourier proof in the notation of this paper is recalled in
Appendix~\ref{app:background-symbol-criterion}.
\end{proof}

Appendix~\ref{app:background-symbol-criterion} recalls the standard Fourier
proof in the present notation.  Lemma~\ref{lem:imported-principal-symbol-sampling}
is used only as imported background; below, the article-specific input is the
computation of the particular Poisson-strip symbol \eqref{eq:lattice-symbol}
and its use for the observation-channel projection.

\paragraph{Poisson-specific integer-orbit symbol.}
Lemma~\ref{lem:imported-principal-symbol-sampling} becomes an
article-specific statement only after
the Poisson strip kernel is inserted.  The next proposition is the calculation
that foregrounds the integer-orbit contribution used by the observation-channel
projection theorem below.

\begin{proposition}[Explicit Poisson integer-orbit symbol]%
\label{prop:poisson-lattice-symbol}
Let
\[
K(a,b):=\frac{2}{4+(a-b)^2},
\qquad
\sigma(\theta):=\sum_{n\in\ZZ}K(n,0)e^{-\ii n\theta}
=\sum_{n\in\ZZ}\frac{2}{4+n^2}e^{-\ii n\theta}.
\]
Then $\sigma$ is continuous and $2\pi$-periodic, and for
$\theta\in[-\pi,\pi]$ one has
\begin{equation}\label{eq:lattice-symbol}
\sigma(\theta)
=\frac{\pi\cosh(2(\pi-|\theta|))}{\sinh(2\pi)}.
\end{equation}
Consequently the sharp lattice-symbol bounds are
\[
A_{\ZZ}:=\frac{\pi}{\sinh(2\pi)}
\le
\sigma(\theta)
\le
\pi\coth(2\pi)=:B_{\ZZ}
\qquad(\theta\in[-\pi,\pi]).
\]
Moreover, for every finitely supported scalar sequence
$c=(c_n)_{n\in\ZZ}$, with
\[
\widetilde c(\theta):=\sum_{n\in\ZZ}c_ne^{-\ii n\theta},
\]
one has the exact Gram identity
\begin{equation}\label{eq:lattice-symbol-form}
\sum_{m,n\in\ZZ}c_m\overline{c_n}\,K(m,n)
=\frac{1}{2\pi}\int_{-\pi}^{\pi}
\sigma(\theta)|\widetilde c(\theta)|^2\dd\theta .
\end{equation}
\end{proposition}

\begin{proof}
Set $k(x):=K(x,0)=2/(4+x^2)$.  With the Fourier convention used in this
paper,
\[
\widehat k(\xi)=\pi e^{-2|\xi|}.
\]
The defining series for $\sigma$ is absolutely and uniformly convergent,
so $\sigma$ is continuous and $2\pi$-periodic.
The Poisson summation formula \cite[Chapter~III]{Stein1971} gives
\[
\sigma(\theta)=\sum_{m\in\ZZ}\widehat k(\theta+2\pi m)
=\pi\sum_{m\in\ZZ}e^{-2|\theta+2\pi m|}.
\]
For $0\le\theta\le\pi$ this geometric series equals
\[
\pi\left(e^{-2\theta}
+\sum_{j=1}^{\infty}e^{-2(\theta+2\pi j)}
+\sum_{j=1}^{\infty}e^{-2(2\pi j-\theta)}\right)
=\pi\,\frac{\cosh(2(\pi-\theta))}{\sinh(2\pi)}.
\]
The expression is even in~$\theta$, so \eqref{eq:lattice-symbol} follows
on $[-\pi,\pi]$, and periodicity gives the full symbol.
The minimum and maximum of the displayed formula occur respectively at
$|\theta|=\pi$ and $\theta=0$, which gives the sharp constants
$A_{\ZZ}$ and $B_{\ZZ}$.

Finally, since $K(m,n)=k(m-n)$, Parseval's identity for the discrete
Fourier transform gives
\[
\sum_{m,n\in\ZZ}c_m\overline{c_n}\,K(m,n)
=\frac{1}{2\pi}\int_{-\pi}^{\pi}
\left(\sum_{\ell\in\ZZ}k(\ell)e^{-\ii\ell\theta}\right)
|\widetilde c(\theta)|^2\dd\theta,
\]
which is exactly \eqref{eq:lattice-symbol-form}.
\end{proof}

\paragraph{Consequences of the imported criterion for the Poisson symbol.}
Define the closed shift-generated Hilbert subspaces
\[
\mathcal H_K(\ZZ)
:=
\overline{\operatorname{span}}\{K(\cdot,n):n\in\ZZ\}
\subset \mathcal H_K,
\]
\[
\mathcal S_{\ZZ}
:=
\overline{\operatorname{span}}\{\Psi_n:n\in\ZZ\}
\subset L^2_0(\omega).
\]
Thus $\mathcal H_K(\ZZ)$ and $\mathcal S_{\ZZ}$ are the Hilbert closures
of finite linear combinations of the corresponding kernel sections and
secants.
Set
\begin{equation}\label{eq:lattice-sampling-constants}
A_{\ZZ}:=\frac{\pi}{\sinh(2\pi)},
\qquad
B_{\ZZ}:=\pi\coth(2\pi).
\end{equation}
We apply the imported principal-symbol criterion
Lemma~\ref{lem:imported-principal-symbol-sampling} once, with the Toeplitz
kernel and symbol computed in
Proposition~\ref{prop:poisson-lattice-symbol}.  This gives the following
standard Riesz and sampling consequences.  The passage from a positive
Toeplitz symbol to Riesz bases, restriction isomorphisms, and cardinal
interpolation is background shift-invariant/RKHS sampling theory; the
article-specific input is the closed Poisson symbol
$\sigma(\theta)=\pi\cosh(2(\pi-|\theta|))/\sinh(2\pi)$ and the induced
constants $A_{\ZZ},B_{\ZZ}$.  The families
$\{K(\cdot,n)\}_{n\in\ZZ}$ and $\{\Psi_n\}_{n\in\ZZ}$ are Riesz bases for
$\mathcal H_K(\ZZ)$ and $\mathcal S_{\ZZ}$, respectively.  More
precisely, for every
finitely supported scalar sequence $(c_n)_{n\in\ZZ}$ one has
\begin{equation}\label{eq:lattice-riesz-kernel}
A_{\ZZ}\sum_{n\in\ZZ}|c_n|^2
\le
\left\|
\sum_{n\in\ZZ}c_nK(\cdot,n)
\right\|_{\mathcal H_K}^2
\le
B_{\ZZ}\sum_{n\in\ZZ}|c_n|^2,
\end{equation}
\begin{equation}\label{eq:lattice-riesz-secant}
A_{\ZZ}\sum_{n\in\ZZ}|c_n|^2
\le
\left\|
\sum_{n\in\ZZ}c_n\Psi_n
\right\|_{L^2(\omega)}^2
\le
B_{\ZZ}\sum_{n\in\ZZ}|c_n|^2.
\end{equation}
Consequently, the synthesis maps
\[
\mathcal C_K(c):=\sum_{n\in\ZZ}c_nK(\cdot,n),
\qquad
\mathcal C_S(c):=\sum_{n\in\ZZ}c_n\Psi_n,
\qquad c\in c_{00}(\ZZ),
\]
extend uniquely to topological isomorphisms
\[
\mathcal C_K:\ell^2(\ZZ)\xrightarrow{\sim}\mathcal H_K(\ZZ),
\qquad
\mathcal C_S:\ell^2(\ZZ)\xrightarrow{\sim}\mathcal S_{\ZZ}.
\]
If $F\in\mathcal H_K(\ZZ)$, then
\begin{equation}\label{eq:lattice-sampling-HK}
A_{\ZZ}\|F\|_{\mathcal H_K}^2
\le
\sum_{n\in\ZZ}|F(n)|^2
\le
B_{\ZZ}\|F\|_{\mathcal H_K}^2.
\end{equation}
Consequently, the restriction map
\[
R_{\ZZ}:\mathcal H_K(\ZZ)\to \ell^2(\ZZ),
\qquad
R_{\ZZ}F:=(F(n))_{n\in\ZZ},
\]
is an isomorphism of Hilbert spaces.  Equivalently, every
$\alpha\in\ell^2(\ZZ)$ admits a unique interpolant
$F_\alpha\in\mathcal H_K(\ZZ)$ satisfying
\[
F_\alpha(n)=\alpha_n
\qquad(n\in\ZZ).
\]

Indeed, the hypotheses of the imported criterion are verified as follows.
For finitely supported $c=(c_n)_{n\in\ZZ}$ set
\[
F_c:=\sum_{n\in\ZZ}c_nK(\cdot,n),
\qquad
\widetilde c(\theta):=\sum_{n\in\ZZ}c_ne^{-\ii n\theta},
\qquad
\theta\in[-\pi,\pi].
\]
By Theorem~\ref{thm:mode-gram-kernel},
\[
\|F_c\|_{\mathcal H_K}^2
=\sum_{m,n\in\ZZ}c_m\overline{c_n}\,K(m,n)
=\sum_{m,n\in\ZZ}c_m\overline{c_n}\,\frac{2}{4+(m-n)^2}.
\]
Thus \eqref{eq:lattice-symbol-form} from
Proposition~\ref{prop:poisson-lattice-symbol} gives
\[
\|F_c\|_{\mathcal H_K}^2
=\frac{1}{2\pi}\int_{-\pi}^{\pi}\sigma(\theta)
|\widetilde c(\theta)|^2\dd\theta.
\]
Combining the sharp symbol bounds from
Proposition~\ref{prop:poisson-lattice-symbol} with Parseval for
$\widetilde c$ proves \eqref{eq:lattice-riesz-kernel}.  Since
$\mathcal U\Psi_n=K(\cdot,n)$ by
Theorem~\ref{thm:mode-space-rkhs}, \eqref{eq:lattice-riesz-secant}
follows immediately.
The two inequalities show that $\mathcal C_K$ and $\mathcal C_S$
are bounded above and below on the dense subspace $c_{00}(\ZZ)$ of
$\ell^2(\ZZ)$.  Hence they extend uniquely to bounded topological
isomorphisms from $\ell^2(\ZZ)$ onto their closed ranges.  By definition
those closed ranges are precisely $\mathcal H_K(\ZZ)$ and
$\mathcal S_{\ZZ}$, so the stated Riesz-basis assertions follow.

Now let $G$ be the Toeplitz Gram operator on $\ell^2(\ZZ)$ associated
with the matrix $(K(m,n))_{m,n\in\ZZ}$.  For finitely supported $c$ one
has $R_{\ZZ}F_c=Gc$, so
\[
\sum_{n\in\ZZ}|F_c(n)|^2
=\|Gc\|_{\ell^2}^2
=\frac{1}{2\pi}\int_{-\pi}^{\pi}\sigma(\theta)^2
|\widetilde c(\theta)|^2\dd\theta.
\]
Since $A_{\ZZ}\le \sigma\le B_{\ZZ}$, comparison with
\eqref{eq:lattice-symbol-form} yields
\[
A_{\ZZ}\|F_c\|_{\mathcal H_K}^2
\le
\sum_{n\in\ZZ}|F_c(n)|^2
\le
B_{\ZZ}\|F_c\|_{\mathcal H_K}^2.
\]
Passing to limits in $\mathcal H_K(\ZZ)$ proves
\eqref{eq:lattice-sampling-HK}.

The symbol bounds imply that $G$ extends to a boundedly invertible
operator on $\ell^2(\ZZ)$.  Thus for every $\alpha\in\ell^2(\ZZ)$ there is
a unique $c\in\ell^2(\ZZ)$ with $Gc=\alpha$.  By
\eqref{eq:lattice-riesz-kernel}, the series $\sum_n c_nK(\cdot,n)$
converges in $\mathcal H_K(\ZZ)$ to some $F_\alpha$, and then
$R_{\ZZ}F_\alpha=\alpha$.  Uniqueness follows from the lower bound in
\eqref{eq:lattice-sampling-HK}.
