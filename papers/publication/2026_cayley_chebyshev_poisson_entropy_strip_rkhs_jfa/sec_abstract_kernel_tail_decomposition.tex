\subsection{A sufficient kernel criterion for positive tail jets}
\label{subsec:abstract-positive-tail-jet-kernel}

We now separate the kernel properties from their stable implementation.  The
following is a sufficient kernel criterion; no necessity or global
minimality assertion is made.  Fix an integer \(r\ge1\).  A strictly positive
probability density \(p\in C^r(\RR^d)\) will be called \emph{tail-jet
admissible of order \(r\)} if the following two conditions hold.
\begin{enumerate}[label=\textup{(K\arabic*)},leftmargin=2.8em]
\item For every multi-index \(\gamma\) with \(1\le|\gamma|\le r\),
\[
 \left\|\frac{\partial^\gamma p}{p}\right\|_\infty<\infty.
\]
\item The top normalized derivative has quotient translation modulus
\[
 \omega_{p,r}(t):=
 \max_{|\gamma|=r}\ \sup_{|h|\le t}
 \left\|\frac{\partial^\gamma p(\mathord\cdot-h)
                   -\partial^\gamma p}{p}\right\|_\infty,
 \qquad 0<t\le1,
\]
and \(\omega_{p,r}(t)\to0\) as \(t\downarrow0\).
\end{enumerate}
The omitted finiteness clause is automatic.  Indeed, if
\(M_r=\max_{|\gamma|=r}\|\partial^\gamma p/p\|_\infty\) and
\(L=\|\nabla\log p\|_\infty\), then, for \(|h|\le1\),
\[
 \frac{|\partial^\gamma p(y-h)-\partial^\gamma p(y)|}{p(y)}
 \le M_r\{e^{L|h|}+1\}.
\]

\begin{unnumberedlemma}[Zero mass of integrable density derivatives]
If \(p\in C^r(\RR^d)\) and \(\partial^\gamma p\in L^1\) for
\(|\gamma|\le r\), then
\[
 \int_{\RR^d}\partial^\gamma p(y)\dd y=0,
 \qquad 1\le|\gamma|\le r.
\]
\end{unnumberedlemma}

\begin{proof}
Choose \(j\) with \(\gamma_j>0\), put
\(f=\partial^{\gamma-e_j}p\in W^{1,1}\), and test
\(\partial_jf\) against cutoffs \(\chi_R\to1\) with
\(\|\nabla\chi_R\|_\infty=O(R^{-1})\).  Then
\(\int\chi_R\partial_jf=-\int f\partial_j\chi_R\to0\), while dominated
convergence sends the left side to \(\int\partial_jf\).
\end{proof}

\begin{unnumberedlemma}[Bounded-score translate entropy]
If \(p>0\), \(p\in C^1(\RR^d)\), and
\(\|\nabla\log p\|_\infty\le L\), then, for every \(z\in\RR^d\),
\begin{equation}\label{eq:bounded-score-translate-entropy}
 0\le D_{\rm KL}\bigl(p(\mathord\cdot-z)\|p\bigr)\le L|z|.
\end{equation}
\end{unnumberedlemma}

\begin{proof}
After the change of variables \(w=y-z\), the divergence is
\(\int p(w)\log\{p(w)/p(w+z)\}\dd w\).  The fundamental theorem of
calculus bounds the absolute value of the logarithm by \(L|z|\); integration
and nonnegativity of relative entropy give
\eqref{eq:bounded-score-translate-entropy}.
\end{proof}

Put \(p_s(x)=s^{-d}p(x/s)\),
\(\Omega_p(\dd y)=p(y)\dd y\), and
\[
 F_y^p(z)=\frac{p(y-z)}{p(y)}.
\]
For a law \(\lambda\) with finite \(r\)th absolute moment define
\[
 B_{p,j}^\lambda(y):=
 \frac{(-1)^j}{p(y)}
 \sum_{|\gamma|=j}\frac{m_\gamma(\lambda)}{\gamma!}
 \partial^\gamma p(y),\qquad 1\le j\le r.
\]

\begin{theorem}[Abstract positive tail-jet kernel theorem]
\label{thm:abstract-positive-tail-jet-kernel}
Let \(p\) be tail-jet admissible of order \(r\), and let \(\nu,\eta\) be
probability laws on \(\RR^d\) such that
\[
 m_\gamma(\nu)=m_\gamma(\eta)\quad(|\gamma|<r),
 \qquad \int |x|^r(\nu+\eta)(\dd x)<\infty.
\]
For \(\lambda\in\{\nu,\eta\}\), set
\[
 \tau_s^\lambda=\lambda(|x|>s),\qquad
 V_{p,s}^\lambda(y)=
 \int_{|x|>s}F_y^p(x/s)\lambda(\dd x),
\]
and let the common retained jet and the two proxies be
\begin{align*}
 C_{p,r-1,s}
 &=\sum_{j=1}^{r-1}s^{-j}B_{p,j}^\nu
  =\sum_{j=1}^{r-1}s^{-j}B_{p,j}^\eta,\\
 A_{p,s}^\lambda
 &=1+C_{p,r-1,s}+V_{p,s}^\lambda-\tau_s^\lambda.
\end{align*}
The sum is empty when \(r=1\).  For all sufficiently large \(s\), both
\(A_{p,s}^\lambda\) are positive probability densities relative to
\(\Omega_p\).  Thus
\[
 k_{p,r,s}^\lambda(x)=s^{-d}p(x/s)A_{p,s}^\lambda(x/s)
\]
are probability densities and the finite, nonnegative proxy energy
\[
 \mathcal E_{p,r,s}(\nu,\eta)
 :=D_{\rm KL}(k_{p,r,s}^\nu\|k_{p,r,s}^\eta)
\]
is well defined.  If
\(\Delta_r=\int x^{\otimes r}(\nu-\eta)(\dd x)\), put
\[
 B_{p,\Delta_r}(y)=
 \frac{(-1)^r}{r!p(y)}\langle\Delta_r,D^rp(y)\rangle,
 \qquad
 \mathcal Q_{p,r}(\Delta_r)
 =\frac12\int B_{p,\Delta_r}^2\dd\Omega_p.
\]
Then
\begin{equation}\label{eq:abstract-tail-defect-decomposition}
 D_{\rm KL}(p_s*\nu\|p_s*\eta)
 =\mathcal Q_{p,r}(\Delta_r)s^{-2r}
  +\mathcal E_{p,r,s}(\nu,\eta)+o(s^{-2r}).
\end{equation}
Moreover \(\mathcal Q_{p,r}\) is finite and positive definite on
\(\operatorname{Sym}^r(\RR^d)\), and hence, law by law,
\[
 s^{2r}D_{\rm KL}(p_s*\nu\|p_s*\eta)
 \longrightarrow\mathcal Q_{p,r}(\Delta_r)
 \quad\Longleftrightarrow\quad
 s^{2r}\mathcal E_{p,r,s}(\nu,\eta)\longrightarrow0.
\]
\end{theorem}

\begin{proof}
Let
\[
 \mathsf B_{p,j}(y,z)=
 \frac{(-1)^j}{j!p(y)}\langle D^jp(y),z^{\otimes j}\rangle,
 \qquad \mathsf B_{p,0}=1.
\]
Condition \textup{(K1)} gives, for \(1\le j\le r\),
\begin{equation}\label{eq:abstract-tail-mode-bounds}
 \|\mathsf B_{p,j}(\mathord\cdot,z)\|_\infty
 +\|\mathsf B_{p,j}(\mathord\cdot,z)\|_{L^1(\Omega_p)}
 \le C_{p,r}|z|^j.
\end{equation}
It also puts every derivative through order \(r\) in \(L^1\), so the
zero-mass lemma applies.  Ordinary translation continuity in \(L^1\), the
integral Taylor formula, and \textup{(K2)} imply that, for \(|z|\le1\),
\begin{align}
 &\left\|F_\cdot^p(z)-\sum_{j=0}^r
       \mathsf B_{p,j}(\mathord\cdot,z)\right\|_\infty
 +\left\|F_\cdot^p(z)-\sum_{j=0}^r
       \mathsf B_{p,j}(\mathord\cdot,z)\right\|_{L^1(\Omega_p)}\notag\\
 &\hspace{35mm}\le \varepsilon_{p,r}(|z|)|z|^r,
 \label{eq:abstract-tail-local-remainder}
\end{align}
where \(\varepsilon_{p,r}\) is bounded on \([0,1]\) and tends to zero at
zero.  The uniform remainder is bounded by a constant times
\(|z|^r\omega_{p,r}(|z|)\).  After multiplication by \(p(y)\), the
\(L^1(\Omega_p)\) remainder is bounded by a constant times
\[
 |z|^r\max_{|\gamma|=r}\sup_{|h|\le|z|}
 \|\partial^\gamma p(\mathord\cdot-h)-\partial^\gamma p\|_1,
\]
which is \(o(|z|^r)\).  Thus only \(C^r\), not \(C^{r+1}\), is used.

Define the actual quotient
\[
 q_{p,s}^\lambda(y)=\int F_y^p(x/s)\lambda(\dd x).
\]
Splitting at \(|x|=s\), retaining the exterior translate exactly, and using
the moments gives
\begin{equation}\label{eq:abstract-tail-exact-quotient-split}
 q_{p,s}^\lambda
 =A_{p,s}^\lambda+s^{-r}B_{p,r}^\lambda+R_{p,s}^\lambda,
 \qquad
 \|R_{p,s}^\lambda\|_\infty
 +\|R_{p,s}^\lambda\|_{L^1(\Omega_p)}=o(s^{-r}).
\end{equation}
After multiplication by \(s^r\), the interior Taylor remainder is dominated
by \(C|x|^r\) and converges pointwise to zero.  Every discarded exterior
mode satisfies
\[
 s^{-j}\int_{|x|>s}|x|^j\lambda(\dd x)
 \le s^{-r}\int_{|x|>s}|x|^r\lambda(\dd x)=o(s^{-r}),
 \qquad j\le r,
\]
and the same estimate gives \(\tau_s^\lambda=o(s^{-r})\).

Translation preserves mass, so \(V_{p,s}^\lambda\ge0\) and
\(\int V_{p,s}^\lambda\dd\Omega_p=\tau_s^\lambda\).  The zero-mass lemma
gives \(\int B_{p,j}^\lambda\dd\Omega_p=0\).  Therefore each proxy has mass
one, and
\[
 \inf_yA_{p,s}^\lambda(y)
 \ge1-\|C_{p,r-1,s}\|_\infty-\tau_s^\lambda\longrightarrow1.
\]
This proves normalization and positivity without a tail comparison.

We next verify entropy finiteness using only \textup{(K1)}.  Put
\(L=\|\nabla\log p\|_\infty\).  When \(\tau_s^\lambda>0\), convexity of
relative entropy and the bounded-score entropy lemma give, for
\(Q_{p,s}^\lambda=V_{p,s}^\lambda/\tau_s^\lambda\),
\begin{equation}\label{eq:abstract-tail-normalized-proxy-entropy}
 \int Q_{p,s}^\lambda\log Q_{p,s}^\lambda\dd\Omega_p
 \le\frac{L}{s\tau_s^\lambda}
 \int_{|x|>s}|x|\lambda(\dd x)<\infty.
\end{equation}
The first moment is finite because \(r\ge1\).  The inequalities
\(\Phi(\tau q)\le\tau q\log(1+q)\) for \(0\le\tau\le1\), and
\[
 (u+v)\log^+(u+v)\le C\{1+v+\Phi(v)\},
 \qquad 0\le u\le3/2,\quad v\ge0,
\]
then show
\(\int A_{p,s}^\lambda\log^+A_{p,s}^\lambda\dd\Omega_p<\infty\).
The case \(\tau_s^\lambda=0\) is immediate.  Since the two proxies have a
common positive lower bound, their relative entropy is finite.

The bounded score also gives
\[
 F_y^p(z)\ge e^{-L|z|},\qquad
 q_{p,s}^\lambda(y)
 \ge e^{-L\int|x|\lambda(\dd x)/s}\longrightarrow1
\]
uniformly in \(y\).  The backgrounds tend to one in \(L^1(\Omega_p)\), and
their common jet cancels, so
\begin{align*}
 \|A_{p,s}^\lambda-1\|_1
 &\le\sum_{j=1}^{r-1}s^{-j}\|B_{p,j}^\lambda\|_1
      +2\tau_s^\lambda=o(1),\\
 \|A_{p,s}^\nu-A_{p,s}^\eta\|_1
 &\le2(\tau_s^\nu+\tau_s^\eta)=o(s^{-r}).
\end{align*}
All hypotheses of the two-background KL perturbation lemma are satisfied
with \(\epsilon_s=s^{-r}\).  Its Hessian and cubic-remainder calculation
gives
\[
 D_{\rm KL}(p_s*\nu\|p_s*\eta)
 =\mathcal E_{p,r,s}(\nu,\eta)
  +\frac{s^{-2r}}2\int
       (B_{p,r}^\nu-B_{p,r}^\eta)^2\dd\Omega_p+o(s^{-2r}).
\]
The moment identity makes the difference in the integral exactly
\(B_{p,\Delta_r}\), proving \eqref{eq:abstract-tail-defect-decomposition}.

Finiteness of \(\mathcal Q_{p,r}\) follows from \textup{(K1)}.  If it
vanishes, then
\[
 \sum_{|\gamma|=r}\frac{\Delta m_\gamma}{\gamma!}
 \partial^\gamma p=0
\]
almost everywhere.  Fourier transformation gives
\(P(i\xi)\widehat p(\xi)=0\), where \(P\) is the corresponding homogeneous
polynomial.  Since \(\widehat p(0)=1\), continuity makes \(\widehat p\)
nonzero on a neighborhood of zero.  Thus \(P\) vanishes on an open set and
is identically zero.  Hence \(\Delta_r=0\), proving positive definiteness.
Nonnegativity of the proxy KL energy proves the final equivalence.
\end{proof}

\begin{proposition}[Closure and verified kernel classes]
\label{prop:tail-jet-kernel-closure-and-examples}
For each fixed order \(r\), tail-jet admissibility is preserved by:
\begin{enumerate}[label=\textup{(\roman*)},leftmargin=2.8em]
\item invertible affine images;
\item finite tensor products, allowing different block dimensions; and
\item finite positive mixtures of admissible densities on the same space.
\end{enumerate}
The criterion of Theorem~\ref{thm:abstract-positive-tail-jet-kernel}
therefore applies to every kernel obtained by finitely combining these
operations.  Its base members include, at every order, each isotropic
strictly \(\alpha\)-stable density, \(0<\alpha<2\), the Poisson kernel in
every dimension, and every multivariate Student kernel
\[
 p_{\beta,d}(y)=c_{\beta,d}(1+|y|^2)^{-\beta/2},\qquad \beta>d.
\]
In particular, affine mixtures and products of these kernels give generally
nonradial, nonstable members of the criterion.
\end{proposition}

\begin{proof}
Write \(R_{p,\gamma}=\partial^\gamma p/p\).  Under \textup{(K1)}, every
\(R_{p,\gamma}\) with \(|\gamma|<r\) is globally Lipschitz, since
\[
 \partial_jR_{p,\gamma}
 =R_{p,\gamma+e_j}-R_{p,\gamma}R_{p,e_j}.
\]
For \(|\gamma|=r\), \textup{(K2)} and the bounded score imply uniform
continuity of \(R_{p,\gamma}\): for \(|h|\le t\),
\[
 |R_{p,\gamma}(y-h)-R_{p,\gamma}(y)|
 \le\omega_{p,r}(t)+\|R_{p,\gamma}\|_\infty(e^{Lt}-1).
\]
Conversely, uniform continuity of the bounded top normalized derivatives,
together with \(p(y-h)/p(y)\in[e^{-L|h|},e^{L|h|}]\), implies
\textup{(K2)}.  We use this equivalent form for products.

If \(q(x)=|\det A|^{-1}p(A^{-1}(x-b))\), the chain rule expresses every
normalized derivative of \(q\) as a fixed linear combination of normalized
derivatives of \(p\).  A translation by \(h\) for \(q\) becomes a
translation by \(A^{-1}h\) for \(p\).  This proves affine closure.

For a product \(q(x_1,\ldots,x_N)=\prod_{i=1}^Np_i(x_i)\), a block
multi-index \(\gamma=(\gamma_1,\ldots,\gamma_N)\) satisfies
\[
 R_{q,\gamma}(x)=\prod_{i=1}^N R_{p_i,\gamma_i}(x_i),
 \qquad R_{p_i,0}=1.
\]
These factors are bounded and uniformly continuous: they are Lipschitz below
order \(r\), and the preceding equivalence handles order \(r\).  Hence the
product satisfies \textup{(K1)} and \textup{(K2)}.

Finally let \(q=\sum_{i=1}^Na_ip_i\), where \(a_i>0\) and
\(\sum_i a_i=1\).  With \(w_i(y)=a_ip_i(y)/q(y)\),
\[
 R_{q,\gamma}(y)=\sum_iw_i(y)R_{p_i,\gamma}(y),
 \qquad \sum_iw_i(y)=1,
\]
so \textup{(K1)} follows.  Moreover,
\[
 \omega_{q,r}(t)
 \le\max_i\omega_{p_i,r}(t)\longrightarrow0,
\]
which proves mixture closure.

For isotropic stable kernels, Proposition~\ref{prop:stable-kernel-input-package}
gives, with \(\beta=d+\alpha\),
\[
 p(y)\asymp(1+|y|)^{-\beta},\qquad
 |\partial^\gamma p(y)|\le C_\gamma
 (1+|y|)^{-\beta-|\gamma|}.
\]
This proves \textup{(K1)}.  The bound one order higher, the mean-value
theorem, and \(1+|y|\le2(1+|y-th|)\) give
\(\omega_{p,r}(t)\le Ct\), proving \textup{(K2)}.  The bounded-score lemma
then supplies translate entropy without an additional hypothesis.  The
stable notation is identified explicitly in
Corollary~\ref{thm:stable-arbitrary-order-law-by-law-decomposition} below.

At unit height the Poisson kernel is the Student member
\(P_1^{(d)}(y)=c_d(1+|y|^2)^{-(d+1)/2}\).  For general \(\beta>d\), repeated
differentiation shows that every
\(\partial^\gamma p_{\beta,d}/p_{\beta,d}\) is a finite sum of terms
\[
 c_{\gamma,k}\frac{y^{\delta_{\gamma,k}}}{(1+|y|^2)^k},
 \qquad |\delta_{\gamma,k}|\le2k,
\]
and is bounded.  The same statement one order higher, together with
\(p_{\beta,d}(y-th)/p_{\beta,d}(y)\le C_{\beta,d}\) for \(|h|\le1\), gives
\(\omega_{p,r}(t)\le Ct\).  Thus the Student and Poisson kernels satisfy
\textup{(K1)}--\textup{(K2)}, completing the verification.
\end{proof}

\begin{proposition}[Gaussian mechanism boundary]
\label{prop:gaussian-additive-proxy-mechanism-boundary}
The standard Gaussian is not tail-jet admissible at any order.  More
strongly, at order two there are moment-matched inputs for which the
prescribed additive common-jet proxies are negative.  Thus
Theorem~\ref{thm:abstract-positive-tail-jet-kernel} describes the
bounded-score/additive branch of the proxy geometry.  Gaussian smoothing
belongs to a different, unbounded-score/multiplicative or exponential-score
branch requiring weighted Hermite control; this proposition is a mechanism
boundary, not a theorem covering Gaussian smoothing.
\end{proposition}

\begin{proof}
For the standard Gaussian \(g\),
\[
 \partial^\gamma g/g=(-1)^{|\gamma|}H_\gamma,
 \qquad \nabla\log g(y)=-y,
\]
so \textup{(K1)} fails already at first order.  The quotient
\[
 F_y^g(z)=\exp\{y\mathbin\cdot z-|z|^2/2\}
\]
has infimum zero for every \(z\ne0\), and the corresponding difference
quotients in \textup{(K2)} are unbounded for every nonzero translation.
Hence the uniform lower background used by the additive two-background
expansion is unavailable.

The proxy failure is explicit in dimension one at \(r=2\).  Take
\(\nu=\delta_m\) and
\(\eta=(\delta_{m-a}+\delta_{m+a})/2\), where \(m\ne0\) and \(a>0\).
They have the same mass and first moment.  For \(s>|m|+a\), the retained
tails vanish, while the common first Gaussian mode is \(my\); both
prescribed proxies therefore equal \(1+my/s\), which is negative on a
half-line.  Consequently, the uniformly bounded cubic-remainder estimate in
the two-background additive-jet expansion is unavailable.  Gaussian
translate entropy is not the obstruction at orders \(r\ge2\), since
\(D_{\rm KL}(g(\mathord\cdot-z)\|g)=|z|^2/2\).  Weighted Hermite arguments
can prove Gaussian first-unmatched-moment asymptotics under stronger input
hypotheses, but they use the different proxy geometry stated above.
\end{proof}
