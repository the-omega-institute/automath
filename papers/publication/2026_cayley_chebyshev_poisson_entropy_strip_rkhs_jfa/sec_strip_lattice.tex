\section{The Strip RKHS and Lattice Consequences}%
\label{sec:strip-lattice}

\paragraph{Complex Hilbert space convention.}
From this point onward, we work with the complexification of the real
Hilbert space from Section~\ref{sec:gram-space}, still denoted
$\mathcal H_K$.  Its inner product is linear in the first variable and
conjugate-linear in the second:
\[
\langle f,g\rangle_{\mathcal H_K}
:=\frac{1}{2\pi^2}\int_{\RR}e^{2|\xi|}\widehat f(\xi)\,
\overline{\widehat g(\xi)}\,\dd\xi.
\]
All reproducing kernels satisfy the Hermitian symmetry
$K(z,w)=\overline{K(w,z)}$, and evaluation representers depend on
$\overline w$, not on~$w$, in the second slot.

\paragraph{Role of the lattice results.}
This section develops the Hilbert and lattice side of the same secant mode
family used in the entropy expansion.  Proposition~\ref{prop:gram-space-poisson-image}
identifies $\mathcal H_K$ with the Poisson image space
$\sqrt{\pi}\,e^{-|D|}L^2(\RR)$ and provides the strip model.  The subsequent
Hardy splitting is included as classical model identification: it names the
positive and negative Fourier-support summands of this strip RKHS and fixes
their kernels, but it is not a separate sampling assertion.  For the lattice part
we use the standard principal shift-invariant-space mechanism on the closed
integer-orbit spaces $\mathcal H_K(\ZZ)$ and $\mathcal S_{\ZZ}$.
The imported part is the positive-symbol/Riesz-basis/cardinal machinery.  The
Poisson-specific part is the closed hyperbolic symbol
\eqref{eq:lattice-symbol}, the resulting cardinal kernel formula
\eqref{eq:cardinal-kernel-fourier} and norm identity
\eqref{eq:cardinal-norm-formula}, and the final projection dichotomy for the
actual observation profiles from Theorem~\ref{thm:poisson-channel-rkhs}:
integer samples recover exactly the lattice projection of the observation
channel, and recover the full channel precisely in the lattice-subspace case.
The $h\ZZ$ fiberization below is used only to identify that integer projection
and its quotient; the headline sampling assertions of the section are the
Poisson integer-symbol theorem, the cardinal reconstruction formula, and the
observation-channel obstruction theorem.  In this sense the lattice package is
the integer-orbit realization of the entropy paper's observation channel: the
background sampling theorem supplies the abstract inverse for a positive
symbol, while the new work computes the Poisson symbol and interprets its
projection and invisible component for the profiles coming from the
Cayley--Poisson quotient.

\subsection{A Poisson image realization of the mode space}

\begin{proposition}[The mode Gram space as a Poisson image space]%
\label{prop:gram-space-poisson-image}
Let
\[
K(a,b):=\frac{2}{4+(a-b)^2}=\pi\,P_2(a-b),
\qquad a,b\in\RR,
\]
and let $\mathcal H_K$ be the reproducing-kernel Hilbert space generated
by~$K$.  Define
\[
\mathcal X
:=
\left\{
f\in L^2(\RR):
\int_{\RR}e^{2|\xi|}\abs{\widehat f(\xi)}^2\dd\xi<\infty
\right\},
\]
with norm
\[
\|f\|_{\mathcal X}^2
:=\frac{1}{2\pi^2}\int_{\RR}e^{2|\xi|}\abs{\widehat f(\xi)}^2\dd\xi.
\]
Then $\mathcal H_K=\mathcal X$ isometrically.  Equivalently,
\[
\mathcal H_K=\sqrt{\pi}\,T_1(L^2(\RR)),
\qquad
\|\sqrt{\pi}\,T_1 g\|_{\mathcal H_K}=\|g\|_{L^2(\RR)}
\quad(g\in L^2(\RR)).
\]
Every $f\in\mathcal H_K$ admits a unique holomorphic extension to the
strip
\[
\mathbb S:=\{z\in\CC:\ |\Im z|<1\},
\]
given by
\[
f(z)=\frac{1}{2\pi}\int_{\RR}\widehat f(\xi)e^{\ii z\xi}\dd\xi,
\qquad z\in\mathbb S,
\]
and satisfying the sharp pointwise estimate
\[
|f(x+\ii y)|\le \frac{1}{\sqrt{2(1-y^2)}}\,\|f\|_{\mathcal H_K}
\qquad(x\in\RR,\ |y|<1).
\]
\end{proposition}

\begin{proof}
The Fourier transform of the translation kernel $k(x):=2/(4+x^2)$ is
\[
\widehat k(\xi)=\pi e^{-2|\xi|}.
\]
Hence
\[
\widehat{K(\cdot,a)}(\xi)=\pi e^{-2|\xi|}e^{-\ii a\xi}.
\]
For $f\in\mathcal X$ one therefore has
\begin{align*}
\langle f,K(\cdot,a)\rangle_{\mathcal X}
&=
\frac{1}{2\pi^2}\int_{\RR}
e^{2|\xi|}\widehat f(\xi)\,
\overline{\widehat{K(\cdot,a)}(\xi)}\dd\xi \\
&=
\frac{1}{2\pi}\int_{\RR}\widehat f(\xi)e^{\ii a\xi}\dd\xi
=f(a)
\end{align*}
by Fourier inversion.  Thus $\mathcal X$ is a reproducing-kernel Hilbert
space with kernel~$K$, and the Moore--Aronszajn uniqueness theorem yields
$\mathcal X=\mathcal H_K$ \cite{Aronszajn1950}.

Now let $g\in L^2(\RR)$ and set $f:=\sqrt{\pi}\,T_1 g$.  Since
\[
\widehat f(\xi)=\sqrt{\pi}\,e^{-|\xi|}\widehat g(\xi),
\]
Plancherel gives
\[
\|f\|_{\mathcal X}^2
=\frac{1}{2\pi^2}\int_{\RR}e^{2|\xi|}\pi e^{-2|\xi|}
\abs{\widehat g(\xi)}^2\dd\xi
=\frac{1}{2\pi}\int_{\RR}\abs{\widehat g(\xi)}^2\dd\xi
=\|g\|_{L^2}^2.
\]
Conversely, if $f\in\mathcal X$, define
\[
\widehat g(\xi):=\pi^{-1/2}e^{|\xi|}\widehat f(\xi).
\]
Then $g\in L^2(\RR)$ and
$\widehat f=\sqrt{\pi}\,e^{-|\xi|}\widehat g$, so
$f=\sqrt{\pi}\,T_1 g$.  This proves the image-space description.

If $z=x+\ii y$ with $|y|<1$, Cauchy--Schwarz gives
\begin{align*}
|f(z)|
&\le \frac{1}{2\pi}
\left(\int_{\RR}e^{2|\xi|}\abs{\widehat f(\xi)}^2\dd\xi\right)^{1/2}
\left(\int_{\RR}e^{-2|\xi|}e^{-2y\xi}\dd\xi\right)^{1/2} \\
&=
\frac{1}{\sqrt{2(1-y^2)}}\,\|f\|_{\mathcal X},
\end{align*}
because
\[
\int_{\RR}e^{-2|\xi|}e^{-2y\xi}\dd\xi=\frac{1}{1-y^2}.
\]
The constant is sharp: at each $z=x+\ii y$ equality is attained by the
normalized reproducing kernel whose Fourier transform is proportional to
$e^{-2|\xi|}e^{-\ii\overline z\xi}$.
Hence the integral defining $f(z)$ converges absolutely and locally
uniformly on $\mathbb S$, so the extension is holomorphic there.  The same
estimate shows uniqueness by continuity from the real axis.
\end{proof}

\begin{remark}\label{rem:poisson-image-framework}
The identity $K(a,b) = \pi P_2(a-b)$ means that the Gram kernel is,
up to the factor~$\pi$, the Poisson kernel at time~$2$.  This explains
why the RKHS is a Poisson image space: the kernel itself arises from the
semigroup.  The holomorphic extension and pointwise estimate above are
the standard $H^2$ Poisson-boundary statements transported to a
translation-invariant strip model; see
\cite{Duren1970,Rudin1987}.
\end{remark}

\subsection{Hardy splitting, lattice sampling, and cardinal reconstruction}

\begin{proposition}[Classical Hardy splitting of the Poisson strip model]%
\label{prop:poisson-hardy-splitting}
Let
\[
\mathcal H_K^+
:=
\left\{
f\in\mathcal H_K:\ \widehat f(\xi)=0 \text{ for a.e. }\xi<0
\right\},
\]
\[
\mathcal H_K^-
:=
\left\{
f\in\mathcal H_K:\ \widehat f(\xi)=0 \text{ for a.e. }\xi>0
\right\}.
\]
Then the classical positive/negative Hardy decomposition transported through
the Poisson image model gives
\[
\mathcal H_K=\mathcal H_K^+\oplus \mathcal H_K^-
\]
orthogonally.  Moreover, each $f_\pm\in\mathcal H_K^\pm$ extends
holomorphically to $\mathbb S$, and $\mathcal H_K^\pm$ is a
reproducing-kernel Hilbert space on~$\mathbb S$ with kernel
\begin{equation}\label{eq:hardy-splitting-kernels}
K_+(z,w)=\frac{1}{2(2-\ii(z-\overline w))},
\qquad
K_-(z,w)=\frac{1}{2(2+\ii(z-\overline w))}.
\end{equation}
For $a,b\in\RR$ one has
\begin{equation}\label{eq:hardy-splitting-real-kernel}
K(a,b)=K_+(a,b)+K_-(a,b).
\end{equation}
Thus the proposition is a background identification of the two Hardy
half-strip components of the already constructed Poisson image space; the
Poisson-specific data retained for the lattice theorem are the normalization
$K(a,b)=\pi P_2(a-b)$ and the explicit kernels in
\eqref{eq:hardy-splitting-kernels}.
\end{proposition}

\begin{proof}
This is the standard Paley--Wiener/Hardy support splitting for a strip,
written in the normalization of Proposition~\ref{prop:gram-space-poisson-image};
see the classical Hardy-space references in
Remark~\ref{rem:poisson-image-framework}.
The norm formula from Proposition~\ref{prop:gram-space-poisson-image}
shows that $\mathcal H_K^+$ and $\mathcal H_K^-$ are closed subspaces of
$\mathcal H_K$, and they are orthogonal because their Fourier supports are
disjoint.  The decomposition
\[
\widehat f
=\widehat f\,\mathbf 1_{[0,\infty)}
+\widehat f\,\mathbf 1_{(-\infty,0]}
\]
therefore yields the asserted orthogonal sum.

For $w\in\mathbb S$, define $K_\pm(\cdot,w)$ by
\[
\widehat{K_+(\cdot,w)}(\xi)
:=
\pi e^{-2\xi}e^{-\ii\overline w\,\xi}\mathbf 1_{[0,\infty)}(\xi),
\]
\[
\widehat{K_-(\cdot,w)}(\xi)
:=
\pi e^{2\xi}e^{-\ii\overline w\,\xi}\mathbf 1_{(-\infty,0]}(\xi).
\]
Since $|\Im w|<1$, these functions belong to $\mathcal H_K^\pm$.  If
$f_+\in\mathcal H_K^+$, then
\begin{align*}
\langle f_+,K_+(\cdot,w)\rangle_{\mathcal H_K}
&=
\frac{1}{2\pi^2}\int_0^\infty
e^{2\xi}\widehat f_+(\xi)\,
\overline{\pi e^{-2\xi}e^{-\ii\overline w\,\xi}}\dd\xi \\
&=
\frac{1}{2\pi}\int_0^\infty \widehat f_+(\xi)e^{\ii w\xi}\dd\xi
=f_+(w).
\end{align*}
The same argument gives the reproducing identity for $\mathcal H_K^-$.  By
Fourier inversion,
\[
K_+(z,w)
=\frac{1}{2\pi}\int_0^\infty
\pi e^{-2\xi}e^{\ii(z-\overline w)\xi}\dd\xi
=\frac{1}{2(2-\ii(z-\overline w))},
\]
\[
K_-(z,w)
=\frac{1}{2\pi}\int_{-\infty}^0
\pi e^{2\xi}e^{\ii(z-\overline w)\xi}\dd\xi
=\frac{1}{2(2+\ii(z-\overline w))},
\]
which proves \eqref{eq:hardy-splitting-kernels}.  Adding the two kernels
on the real axis yields \eqref{eq:hardy-splitting-real-kernel}.
\end{proof}

We next record a fiberization formula for the unshifted family $h\ZZ$.  This
is an auxiliary calculation inside the strip model, not a separate sampling
theorem of the paper.  Its only role in the main theorem chain is to supply the
closed Poisson weight and the orthogonal projection formula used at the
integer orbit; the reconstruction result used here is the integer
specialization in
Corollary~\ref{cor:integer-orbit-projection}, the quotient
Proposition~\ref{prop:integer-orbit-scoped-quotient}, and the Poisson
observation theorem which follows them.  No shifted-lattice, sublattice,
derivative-sampling, or nonuniform sampling assertion is part of this
article.  The general principal shift-invariant-space mechanism is standard;
see \cite{deBoorDeVoreRon1994,RonShen1995,AldroubiGrochenig2001}.  In the
present model both the periodized weight and the projector can be written in
closed form.

\begin{theorem}[Auxiliary unshifted lattice projection and integer quotient rigidity]%
\label{thm:uniform-lattice-projection}
Let $h>0$ and set
\[
I_h:=\Bigl[-\frac{\pi}{h},\frac{\pi}{h}\Bigr],
\qquad
\Sigma_h(\xi):=\sum_{m\in\ZZ}e^{-2\abs{\xi+2\pi m/h}}
\qquad(\xi\in I_h).
\]
Then
\begin{equation}\label{eq:lattice-periodized-weight}
\Sigma_h(\xi)
=\frac{\cosh(2(\pi/h-\abs{\xi}))}{\sinh(2\pi/h)}
\qquad(\xi\in I_h).
\end{equation}
Define
\[
\mathcal H_K(h\ZZ)
:=
\overline{\operatorname{span}}\{K(\cdot,hn):n\in\ZZ\}
\subset\mathcal H_K
\]
and
\[
\mathcal N_h
:=
\{f\in\mathcal H_K:\ f(hn)=0\ \text{for every }n\in\ZZ\}.
\]
For $f\in\mathcal H_K$ write
\[
q_f(\xi):=e^{2\abs{\xi}}\widehat f(\xi).
\]
Then the following hold.
\begin{enumerate}[label=(\roman*)]
\item
      One has $f\in\mathcal H_K(h\ZZ)$ if and only if $q_f$ is
      $2\pi/h$-periodic and
      \[
      \int_{I_h}\Sigma_h(\xi)\abs{q_f(\xi)}^2\dd\xi<\infty.
      \]
      In that case,
      \begin{equation}\label{eq:lattice-subspace-norm}
      \|f\|_{\mathcal H_K}^2
      =\frac{1}{2\pi^2}\int_{I_h}\Sigma_h(\xi)\abs{q_f(\xi)}^2\dd\xi.
      \end{equation}
\item
      The orthogonal complement of~$\mathcal H_K(h\ZZ)$ is exactly
      $\mathcal N_h$, and
      \begin{equation}\label{eq:lattice-alias-cancellation}
      f\in\mathcal N_h
      \iff
      \sum_{m\in\ZZ}\widehat f\!\left(\xi+\frac{2\pi m}{h}\right)=0
      \quad\text{for a.e. }\xi\in I_h.
      \end{equation}
      Consequently,
      \[
      \mathcal H_K=\mathcal H_K(h\ZZ)\oplus \mathcal N_h
      \]
      orthogonally.
\item
      Let $\Pi_h:\mathcal H_K\to\mathcal H_K(h\ZZ)$ denote the orthogonal
      projection.  Then, for almost every $\xi\in I_h$, the series
      \[
      \sum_{m\in\ZZ}\widehat f\!\left(\xi+\frac{2\pi m}{h}\right)
      \]
      converges absolutely and
      \begin{equation}\label{eq:lattice-alias-projection}
      q_{\Pi_hf}(\xi)
      =\frac{\displaystyle\sum_{m\in\ZZ}
      \widehat f\!\left(\xi+\frac{2\pi m}{h}\right)}
      {\Sigma_h(\xi)},
      \end{equation}
      with $q_{\Pi_hf}$ extended $2\pi/h$-periodically to~$\RR$.
      Equivalently,
      \[
      \widehat{\Pi_hf}(\xi)=e^{-2\abs{\xi}}q_{\Pi_hf}(\xi)
      \qquad\text{for a.e. }\xi\in\RR.
      \]
      Moreover,
      \begin{equation}\label{eq:lattice-projection-norm}
      \|\Pi_hf\|_{\mathcal H_K}^2
      =\frac{1}{2\pi^2}\int_{I_h}
      \frac{\left|\sum_{m\in\ZZ}
      \widehat f\!\left(\xi+\frac{2\pi m}{h}\right)\right|^2}
      {\Sigma_h(\xi)}\,\dd\xi,
      \end{equation}
      and $\Pi_hf$ is the unique element of~$\mathcal H_K(h\ZZ)$
      satisfying
      \[
      (\Pi_hf)(hn)=f(hn)
      \qquad(n\in\ZZ).
      \]
\item
      The $h\ZZ$ samples determine exactly the projection $\Pi_hf$ and no
      finer ambient datum: for $f,g\in\mathcal H_K$,
      \begin{equation}\label{eq:uniform-lattice-quotient-rigidity}
      f(hn)=g(hn)\quad(n\in\ZZ)
      \qquad\Longleftrightarrow\qquad
      \Pi_hf=\Pi_hg .
      \end{equation}
      Equivalently,
      \[
      \mathcal H_K/\mathcal N_h\longrightarrow\mathcal H_K(h\ZZ),
      \qquad
      f+\mathcal N_h\longmapsto\Pi_hf,
      \]
      is a well-defined isometric Hilbert-space isomorphism.  Thus full
      reconstruction from the $h\ZZ$ orbit holds for a given
      $f\in\mathcal H_K$ if and only if $f\in\mathcal H_K(h\ZZ)$.
\item
      At the only specialization used below, namely $h=1$, put
      $\Pi_{\ZZ}:=\Pi_1$.  Any reconstruction rule
      \[
      \mathcal R:\mathcal H_K\longrightarrow \mathcal H_K(\ZZ)
      \]
      whose value depends only on the integer samples and which is exact on
      $\mathcal H_K(\ZZ)$ must equal the integer projection:
      \[
      \mathcal Rf=\Pi_{\ZZ}f\qquad(f\in\mathcal H_K).
      \]
      Thus the auxiliary $h\ZZ$ fiberization contributes to the paper only
      through the integer projection and its quotient; the Poisson-specific
      main-chain inputs are the integer symbol in
      Proposition~\ref{prop:poisson-lattice-symbol} and the observation-channel
      theorem, Theorem~\ref{thm:poisson-observation-lattice-sampling}.
\end{enumerate}
\end{theorem}

Only the specialization $h=1$ is used in the Poisson observation-channel
theorems below.  The parameter $h$ is retained in
Theorem~\ref{thm:uniform-lattice-projection} only to derive the unshifted
integer-orbit projection theorem from one formula.  It is not used later as an
invitation to vary the observation set: the reconstruction result of this paper
is the integer-orbit projection and quotient theorem, sharpened by the
Poisson-specific symbol and observation-channel results below.

\begin{proof}
Let
\[
a:=\frac{2\pi}{h}.
\]
For $\xi\in I_h$ one has $\abs{\xi}\le a/2$, so
\begin{align*}
\Sigma_h(\xi)
&=e^{-2\abs{\xi}}
+\sum_{m=1}^{\infty}e^{-2(\abs{\xi}+ma)}
+\sum_{m=1}^{\infty}e^{-2(ma-\abs{\xi})} \\
&=\frac{e^{-2\abs{\xi}}+e^{-2a}e^{2\abs{\xi}}}{1-e^{-2a}}
=\frac{\cosh(a-2\abs{\xi})}{\sinh a},
\end{align*}
which is \eqref{eq:lattice-periodized-weight}.

If
\[
F=\sum_{n=-N}^{N}c_nK(\cdot,hn),
\]
then
\[
\widehat F(\xi)=\pi e^{-2\abs{\xi}}\sum_{n=-N}^{N}c_ne^{-\ii hn\xi},
\]
hence
\[
q_F(\xi)=\pi\sum_{n=-N}^{N}c_ne^{-\ii hn\xi},
\]
which is $a$-periodic.  Using the norm formula from
Proposition~\ref{prop:gram-space-poisson-image} and partitioning~$\RR$
into the translates $I_h+ma$ gives
\[
\|F\|_{\mathcal H_K}^2
=\frac{1}{2\pi^2}\int_{I_h}
\sum_{m\in\ZZ}e^{-2\abs{\xi+ma}}\abs{q_F(\xi)}^2\dd\xi
=\frac{1}{2\pi^2}\int_{I_h}\Sigma_h(\xi)\abs{q_F(\xi)}^2\dd\xi.
\]
Passing to the closure proves one direction of~(i).

Conversely, let $q$ be $a$-periodic with
\[
\int_{I_h}\Sigma_h(\xi)\abs{q(\xi)}^2\dd\xi<\infty.
\]
Since $\Sigma_h$ is continuous and strictly positive on the compact
interval~$I_h$, the spaces
$L^2(I_h,\Sigma_h(\xi)\dd\xi)$ and $L^2(I_h,\dd\xi)$ coincide with
equivalent norms.  The trigonometric polynomials
\[
\sum_{n=-N}^{N}d_ne^{-\ii hn\xi}
\]
are therefore dense in $L^2(I_h,\Sigma_h(\xi)\dd\xi)$.  Choose such a
sequence $p_N$ with $p_N\to \pi^{-1}q$ in that weighted norm, and define
$F_N\in\mathcal H_K(h\ZZ)$ by
\[
F_N:=\sum_{n=-N}^{N}d_nK(\cdot,hn).
\]
Then $q_{F_N}=\pi p_N$, so the preceding norm identity gives
\[
\|F_N-F_M\|_{\mathcal H_K}^2
=\frac{1}{2\pi^2}\int_{I_h}\Sigma_h(\xi)
\abs{q_{F_N}(\xi)-q_{F_M}(\xi)}^2\dd\xi.
\]
Hence $(F_N)$ is Cauchy in~$\mathcal H_K$ and converges to some
$F\in\mathcal H_K(h\ZZ)$ with $q_F=q$.  This proves~(i).

For every $f\in\mathcal H_K$ and almost every $\xi\in I_h$,
Cauchy--Schwarz gives
\[
\sum_{m\in\ZZ}\abs{\widehat f(\xi+ma)}
\le
\Sigma_h(\xi)^{1/2}
\left(
\sum_{m\in\ZZ}e^{2\abs{\xi+ma}}
\abs{\widehat f(\xi+ma)}^2
\right)^{1/2},
\]
so the alias sum is absolutely convergent for almost every~$\xi$.

If $g\in\mathcal H_K(h\ZZ)$, then by~(i) its profile $q_g$ is
$a$-periodic.  Using the Fourier norm formula again,
\begin{align*}
\langle f,g\rangle_{\mathcal H_K}
&=
\frac{1}{2\pi^2}\int_{I_h}
\sum_{m\in\ZZ}e^{-2\abs{\xi+ma}}q_f(\xi+ma)\,
\overline{q_g(\xi)}\dd\xi \\
&=
\frac{1}{2\pi^2}\int_{I_h}
\left(
\sum_{m\in\ZZ}\widehat f(\xi+ma)
\right)\overline{q_g(\xi)}\dd\xi.
\end{align*}
Since the periodic profiles $q_g$ form a dense subspace of
$L^2(I_h,\Sigma_h(\xi)\dd\xi)$ and $\Sigma_h$ is bounded above and below
on~$I_h$, the same set is dense in $L^2(I_h,\dd\xi)$.  Therefore
$f\perp\mathcal H_K(h\ZZ)$ if and only if the alias sum in
\eqref{eq:lattice-alias-cancellation} vanishes almost everywhere on~$I_h$.

By the reproducing property,
\[
f(hn)=\langle f,K(\cdot,hn)\rangle_{\mathcal H_K}
\qquad(n\in\ZZ),
\]
and the closed linear span of the kernel sections $K(\cdot,hn)$ is
exactly $\mathcal H_K(h\ZZ)$.  Consequently,
\[
f\in\mathcal N_h
\iff
f\perp\mathcal H_K(h\ZZ).
\]
This proves~(ii).

Let $g\in\mathcal H_K(h\ZZ)$ and write $q_g$ for its periodic profile.
Then
\[
\|f-g\|_{\mathcal H_K}^2
=\frac{1}{2\pi^2}\int_{I_h}
\sum_{m\in\ZZ}e^{-2\abs{\xi+ma}}
\abs{q_f(\xi+ma)-q_g(\xi)}^2\dd\xi.
\]
For each $\xi\in I_h$, the integrand is minimized when $q_g(\xi)$ is the
weighted average of the fiber values $q_f(\xi+ma)$ with weights
$e^{-2\abs{\xi+ma}}$.  Since
\[
e^{-2\abs{\xi+ma}}q_f(\xi+ma)=\widehat f(\xi+ma),
\]
the minimizing profile is exactly the one in
\eqref{eq:lattice-alias-projection}.  This proves the projection formula.
Substituting the minimizing value into the norm identity gives
\eqref{eq:lattice-projection-norm}.  Finally,
\[
f-\Pi_hf\in\mathcal N_h
\]
by~(ii), so $(\Pi_hf)(hn)=f(hn)$ for every $n\in\ZZ$.  If
$G\in\mathcal H_K(h\ZZ)$ satisfies $G(hn)=f(hn)$ for all $n$, then
$G-\Pi_hf\in\mathcal H_K(h\ZZ)\cap\mathcal N_h=\{0\}$, hence
$G=\Pi_hf$.  This proves~(iii).

It remains to identify the quotient determined by the samples.  The forward
implication in \eqref{eq:uniform-lattice-quotient-rigidity} follows because
equal $h\ZZ$ samples give $f-g\in\mathcal N_h$, and by~(ii) this is the
orthogonal complement of $\mathcal H_K(h\ZZ)$, so the two orthogonal
projections coincide.  The reverse implication follows from~(iii), since
$\Pi_hf$ and $\Pi_hg$ interpolate respectively the samples of $f$ and~$g$.
The displayed quotient map is well defined because adding an element of
$\mathcal N_h=\mathcal H_K(h\ZZ)^\perp$ does not change the projection.  It is
surjective by applying it to $F+\mathcal N_h$ with $F\in\mathcal H_K(h\ZZ)$,
injective by \eqref{eq:uniform-lattice-quotient-rigidity}, and isometric by
the orthogonal decomposition
\[
\|f+\mathcal N_h\|_{\mathcal H_K/\mathcal N_h}
=\inf_{u\in\mathcal N_h}\|\Pi_hf+(f-\Pi_hf)+u\|_{\mathcal H_K}
=\|\Pi_hf\|_{\mathcal H_K}.
\]
  Finally, full reconstruction means $f$ equals the unique lattice-subspace
  interpolant $\Pi_hf$, and this is equivalent to
  $f\in\mathcal H_K(h\ZZ)$.  This proves~(iv).

For~(v), set $h=1$ in~(iv).  If $\mathcal R$ depends only on the integer
samples, then $f$ and $\Pi_{\ZZ}f$ have the same samples by~(iii), so
\[
\mathcal Rf=\mathcal R(\Pi_{\ZZ}f).
\]
Since $\Pi_{\ZZ}f\in\mathcal H_K(\ZZ)$ and $\mathcal R$ is exact on that
subspace, the right-hand side is $\Pi_{\ZZ}f$.  Hence
$\mathcal R=\Pi_{\ZZ}$.  The final sentence records precisely this dependence
of the later integer-orbit theorems on the present auxiliary calculation.
\end{proof}

\begin{corollary}[Integer-orbit projection]%
\label{cor:integer-orbit-projection}
For the integer lattice $h=1$, let
\[
\Pi_{\ZZ}:=\Pi_1,
\qquad
\mathcal N_{\ZZ}:=\{f\in\mathcal H_K:\ f(n)=0\text{ for every }n\in\ZZ\}.
\]
Then
\[
\mathcal H_K=\mathcal H_K(\ZZ)\oplus\mathcal N_{\ZZ}
\]
orthogonally, and $\Pi_{\ZZ}f$ is the unique element of
$\mathcal H_K(\ZZ)$ satisfying
\[
(\Pi_{\ZZ}f)(n)=f(n)\qquad(n\in\ZZ).
\]
If $q_f(\xi):=e^{2|\xi|}\widehat f(\xi)$, then for almost every
$\theta\in[-\pi,\pi]$,
\begin{equation}\label{eq:integer-orbit-projection-symbol}
q_{\Pi_{\ZZ}f}(\theta)
=
\frac{\displaystyle\sum_{m\in\ZZ}\widehat f(\theta+2\pi m)}
{\displaystyle\sum_{m\in\ZZ}e^{-2|\theta+2\pi m|}}
=
\frac{\pi\displaystyle\sum_{m\in\ZZ}\widehat f(\theta+2\pi m)}
{\sigma(\theta)},
\end{equation}
extended $2\pi$-periodically, where the Poisson symbol is exactly
\[
\sigma(\theta)=\frac{\pi\cosh(2(\pi-|\theta|))}{\sinh(2\pi)}
\qquad(-\pi\le \theta\le \pi).
\]
Thus the only lattice specialization used in
Theorem~\ref{thm:poisson-observation-lattice-sampling} is the integer-orbit
projection with this explicit Poisson symbol.
Moreover, for every $f\in\mathcal H_K$, the integer sample sequence
$(f(n))_{n\in\ZZ}$ determines exactly the projection $\Pi_{\ZZ}f$: it is
the unique element of $\mathcal H_K(\ZZ)$ with those integer samples, and
\[
f-\Pi_{\ZZ}f\in\mathcal N_{\ZZ}.
\]
Consequently the integer orbit determines the full function $f$ if and only
if $f\in\mathcal H_K(\ZZ)$.
\end{corollary}

\begin{proof}
The first two assertions are Theorem~\ref{thm:uniform-lattice-projection}
with $h=1$.  In the notation of that proposition,
\[
\Sigma_1(\theta)=\sum_{m\in\ZZ}e^{-2|\theta+2\pi m|}
\qquad(-\pi\le\theta\le\pi).
\]
Multiplying the closed form \eqref{eq:lattice-periodized-weight} for
$\Sigma_1$ by~$\pi$ gives the stated Poisson symbol
\[
\sigma(\theta):=\pi\Sigma_1(\theta)
=\frac{\pi\cosh(2(\pi-|\theta|))}{\sinh(2\pi)}.
\]
Substituting $h=1$ in the projection formula
\eqref{eq:lattice-alias-projection} and replacing $\Sigma_1$ by
$\sigma/\pi$ gives \eqref{eq:integer-orbit-projection-symbol}.  The final
sentence records the dependence of
Theorem~\ref{thm:poisson-observation-lattice-sampling} on this integer
case.  The uniqueness assertion and the residual inclusion are the last
part of Theorem~\ref{thm:uniform-lattice-projection} with $h=1$.  If
$f\in\mathcal H_K(\ZZ)$, then $\Pi_{\ZZ}f=f$, so the integer orbit recovers
the full function.  Conversely, if the integer orbit recovers $f$ as an
element of the reconstruction space $\mathcal H_K(\ZZ)$, then
$f=\Pi_{\ZZ}f\in\mathcal H_K(\ZZ)$.
\end{proof}

\begin{corollary}[Integer-orbit reconstruction rigidity]%
\label{cor:integer-orbit-reconstruction-rigidity}
Let $Y$ be a set, and let $\mathcal R:\mathcal H_K\to Y$ be any rule whose
value depends only on the integer orbit, in the sense that
\[
f(n)=g(n)\quad(n\in\ZZ)\qquad\Longrightarrow\qquad
\mathcal R(f)=\mathcal R(g).
\]
Then $\mathcal R$ factors uniquely through the integer projection
$\Pi_{\ZZ}$:
\[
\mathcal R=\widetilde{\mathcal R}\circ\Pi_{\ZZ}
\qquad\text{for a unique }\widetilde{\mathcal R}:\mathcal H_K(\ZZ)\to Y.
\]
In particular, if a reconstruction map
$\mathcal R:\mathcal H_K\to\mathcal H_K(\ZZ)$ depends only on integer samples
and is exact on the lattice subspace, meaning
$\mathcal R(F)=F$ for every $F\in\mathcal H_K(\ZZ)$, then
$\mathcal R=\Pi_{\ZZ}$.  Thus the integer-orbit chain in this paper has a
single possible ambient reconstruction: the lattice projection.
\end{corollary}

\begin{proof}
Define
$\widetilde{\mathcal R}(F):=\mathcal R(F)$ for
$F\in\mathcal H_K(\ZZ)$.  For any $f\in\mathcal H_K$,
Corollary~\ref{cor:integer-orbit-projection} gives
$(\Pi_{\ZZ}f)(n)=f(n)$ for every $n\in\ZZ$.  Since $\mathcal R$ depends only
on these samples, $\mathcal R(f)=\mathcal R(\Pi_{\ZZ}f)$, proving the stated
factorization.  Uniqueness follows because $\Pi_{\ZZ}$ is onto
$\mathcal H_K(\ZZ)$.

If $\mathcal R$ takes values in $\mathcal H_K(\ZZ)$ and is exact on
$\mathcal H_K(\ZZ)$, then
\[
\mathcal R(f)=\mathcal R(\Pi_{\ZZ}f)=\Pi_{\ZZ}f
\]
for every $f\in\mathcal H_K$.  Hence $\mathcal R=\Pi_{\ZZ}$.
\end{proof}

\begin{proposition}[Integer orbit and reconstruction quotient]%
\label{prop:integer-orbit-scoped-quotient}
The integer samples define exactly the quotient of $\mathcal H_K$ by the
nullspace $\mathcal N_{\ZZ}$ and no finer reconstruction on the ambient RKHS.
More precisely, for $f,g\in\mathcal H_K$,
\[
f(n)=g(n)\quad(n\in\ZZ)
\qquad\Longleftrightarrow\qquad
\Pi_{\ZZ}f=\Pi_{\ZZ}g .
\]
Equivalently, the map
\[
\mathcal H_K/\mathcal N_{\ZZ}\longrightarrow \mathcal H_K(\ZZ),
\qquad
f+\mathcal N_{\ZZ}\longmapsto \Pi_{\ZZ}f,
\]
is a well-defined isometric Hilbert-space isomorphism.  Therefore any
integer-orbit reconstruction from these samples factors uniquely through
$\Pi_{\ZZ}$ as in
Corollary~\ref{cor:integer-orbit-reconstruction-rigidity}; it reconstructs
the full ambient function if and only if that function already lies in
$\mathcal H_K(\ZZ)$.
\end{proposition}

\begin{proof}
By Corollary~\ref{cor:integer-orbit-projection},
$f-\Pi_{\ZZ}f\in\mathcal N_{\ZZ}$ and $\Pi_{\ZZ}f$ has the same integer
samples as~$f$.  Hence if $\Pi_{\ZZ}f=\Pi_{\ZZ}g$, then $f$ and $g$ have the
same integer samples.  Conversely, if $f(n)=g(n)$ for every $n\in\ZZ$, then
$f-g\in\mathcal N_{\ZZ}$.  Since
\[
\mathcal H_K=\mathcal H_K(\ZZ)\oplus\mathcal N_{\ZZ}
\]
orthogonally, the projection of $f-g$ onto $\mathcal H_K(\ZZ)$ is zero, so
$\Pi_{\ZZ}f=\Pi_{\ZZ}g$.

The displayed map is well defined because changing $f$ by an element of
$\mathcal N_{\ZZ}=\mathcal H_K(\ZZ)^\perp$ does not change its orthogonal
projection.  It is injective by the first paragraph, surjective because every
$F\in\mathcal H_K(\ZZ)$ is the image of its own coset, and isometric for the
quotient norm because the orthogonal decomposition gives
\[
\|f+\mathcal N_{\ZZ}\|_{\mathcal H_K/\mathcal N_{\ZZ}}
=\inf_{u\in\mathcal N_{\ZZ}}\|\Pi_{\ZZ}f+(f-\Pi_{\ZZ}f)+u\|_{\mathcal H_K}
=\|\Pi_{\ZZ}f\|_{\mathcal H_K}.
\]
Thus the only data-dependent object determined by integer samples in the
ambient RKHS is the projection $\Pi_{\ZZ}f$.  Full reconstruction is therefore
equivalent to $f=\Pi_{\ZZ}f$, namely to $f\in\mathcal H_K(\ZZ)$.
\end{proof}

\subsection{Imported symbol criterion and the Poisson integer-orbit symbol}%
\label{subsec:lattice-background-poisson-symbol}

\paragraph{Imported shift-invariant-space criterion.}
This paragraph is the only abstract sampling input used in the integer-orbit
argument.  The following unnumbered background criterion fixes notation for the
standard positive-symbol mechanism used later.  All hypotheses needed for the
import are displayed in the statement: a Toeplitz Gram matrix with
\(a\in\ell^1(\ZZ)\), a principal symbol bounded above and below away from
zero, and a restriction map whose action in Riesz coordinates is the Gram
operator.  Under precisely these hypotheses it is the classical principal
shift-invariant-space/Riesz-translate argument
\cite{deBoorDeVoreRon1994,RonShen1995,Bownik2000,AldroubiGrochenig2001}
and is compatible with the RKHS sampling and Galerkin frameworks of
\cite{NashedWalter1991,NashedSun2010,ChengJiangSun2016}.  The paper's
Poisson-specific work begins after this criterion, and it has exactly two inputs:
the closed symbol for the strip kernel in
Proposition~\ref{prop:poisson-lattice-symbol} and the observation-channel
identification in Theorem~\ref{thm:poisson-channel-rkhs}.  The later Riesz
bounds, cardinal norm, Fourier formula, and observation-channel projection
theorem are the resulting closed-form consequences of these inputs plus the
standard criterion.  Existing Cauchy/Lorentz integer-shift and nod-function
computations are recorded for background in
\cite{KiselevMininNovikov2016,KiselevMininNovikov2019}; the point here is not
to reprove the general cardinal theory but to insert the Poisson-strip secant
mode space into it through the explicit formulas
\eqref{eq:lattice-symbol}, \eqref{eq:cardinal-kernel-fourier}, and
\eqref{eq:cardinal-norm-formula}.

\smallskip
\noindent\textit{Classical imported principal-symbol sampling criterion.}
Let $\mathcal H$ be a Hilbert space and let
$(\varphi_n)_{n\in\ZZ}$ be a family whose Gram matrix is Toeplitz,
\[
\langle\varphi_m,\varphi_n\rangle_{\mathcal H}=a_{m-n},
\qquad a\in\ell^1(\ZZ).
\]
Put
\[
\sigma(\theta):=\sum_{k\in\ZZ}a_ke^{-\ii k\theta},
\qquad -\pi\le\theta\le\pi,
\]
and suppose that $0<A\le\sigma(\theta)\le B<\infty$ a.e.  Then the
synthesis map
\[
C:c_{00}(\ZZ)\to\mathcal H,
\qquad Cc:=\sum_{n\in\ZZ}c_n\varphi_n,
\]
extends to a topological isomorphism from $\ell^2(\ZZ)$ onto the closed
orbit space $V:=\overline{\operatorname{span}}\{\varphi_n:n\in\ZZ\}$, with
\[
A\|c\|_{\ell^2}^2\le\|Cc\|_{\mathcal H}^2
\le B\|c\|_{\ell^2}^2.
\]
Let $G$ be the Toeplitz Gram operator on $\ell^2(\ZZ)$ with matrix
$(a_{m-n})_{m,n\in\ZZ}$.  The discrete Fourier transform diagonalizes
$G$ as multiplication by~$\sigma$, and $G^{-1}$ is multiplication by
$\sigma^{-1}$.  If, in addition, a bounded restriction map
$R:V\to\ell^2(\ZZ)$ satisfies $R(Cc)=Gc$ for all finitely supported $c$,
then $R$ is an isomorphism.  For every datum $\alpha\in\ell^2(\ZZ)$ the
unique interpolant $R^{-1}\alpha$ satisfies
\begin{equation}\label{eq:imported-reciprocal-symbol-norm}
\|R^{-1}\alpha\|_{\mathcal H}^2
=\frac1{2\pi}\int_{-\pi}^{\pi}
\frac{|\widetilde\alpha(\theta)|^2}{\sigma(\theta)}\dd\theta .
\end{equation}
The cardinal element $L:=R^{-1}\delta_0$ is synthesized by
$d=G^{-1}\delta_0$, equivalently $\widetilde d=\sigma^{-1}$ a.e.
Thus this statement is an imported background criterion: it supplies only the
functional-analytic passage from a positive principal symbol to Riesz bounds,
bounded interpolation, and the reciprocal-symbol norm.  The article's new
Poisson input is the verification, in Proposition~\ref{prop:poisson-lattice-symbol}
and Theorem~\ref{thm:poisson-channel-rkhs}, that the strip kernel and Poisson
observation profiles satisfy these displayed hypotheses.  The criterion itself
is not asserted as a new theorem of this article
\cite{deBoorDeVoreRon1994,RonShen1995,AldroubiGrochenig2001}.
\par\smallskip

This criterion is used below only as this classical background.  The new
Poisson-strip input is the computation of the particular symbol
\eqref{eq:lattice-symbol}; no originality is claimed for the general
Toeplitz/Riesz-orbit sampling mechanism itself.

For reference, Appendix~\ref{app:background-symbol-criterion} recalls the
one-line Fourier proof of this imported criterion; it is separated from the
main text to keep the theorem chain below attached to the Poisson symbol and
the observation quotient.

\paragraph{Poisson-specific integer-orbit symbol.}
The imported criterion above becomes an article-specific statement only after
the Poisson strip kernel is inserted.  The next proposition is the calculation
that foregrounds the integer-orbit contribution used by the observation-channel
projection, quotient, and minimax results below.

\begin{proposition}[Explicit Poisson integer-orbit symbol]%
\label{prop:poisson-lattice-symbol}
Let
\[
K(a,b):=\frac{2}{4+(a-b)^2},
\qquad
\sigma(\theta):=\sum_{n\in\ZZ}K(n,0)e^{-\ii n\theta}
=\sum_{n\in\ZZ}\frac{2}{4+n^2}e^{-\ii n\theta}.
\]
Then $\sigma$ is continuous and $2\pi$-periodic, and for
$\theta\in[-\pi,\pi]$ one has
\begin{equation}\label{eq:lattice-symbol}
\sigma(\theta)
=\frac{\pi\cosh(2(\pi-|\theta|))}{\sinh(2\pi)}.
\end{equation}
Consequently the sharp lattice-symbol bounds are
\[
A_{\ZZ}:=\frac{\pi}{\sinh(2\pi)}
\le
\sigma(\theta)
\le
\pi\coth(2\pi)=:B_{\ZZ}
\qquad(\theta\in[-\pi,\pi]).
\]
Moreover, for every finitely supported scalar sequence
$c=(c_n)_{n\in\ZZ}$, with
\[
\widetilde c(\theta):=\sum_{n\in\ZZ}c_ne^{-\ii n\theta},
\]
one has the exact Gram identity
\begin{equation}\label{eq:lattice-symbol-form}
\sum_{m,n\in\ZZ}c_m\overline{c_n}\,K(m,n)
=\frac{1}{2\pi}\int_{-\pi}^{\pi}
\sigma(\theta)|\widetilde c(\theta)|^2\dd\theta .
\end{equation}
\end{proposition}

\begin{proof}
Set $k(x):=K(x,0)=2/(4+x^2)$.  With the Fourier convention used in this
paper,
\[
\widehat k(\xi)=\pi e^{-2|\xi|}.
\]
The defining series for $\sigma$ is absolutely and uniformly convergent,
so $\sigma$ is continuous and $2\pi$-periodic.
The Poisson summation formula \cite[Chapter~III]{Stein1971} gives
\[
\sigma(\theta)=\sum_{m\in\ZZ}\widehat k(\theta+2\pi m)
=\pi\sum_{m\in\ZZ}e^{-2|\theta+2\pi m|}.
\]
For $0\le\theta\le\pi$ this geometric series equals
\[
\pi\left(e^{-2\theta}
+\sum_{j=1}^{\infty}e^{-2(\theta+2\pi j)}
+\sum_{j=1}^{\infty}e^{-2(2\pi j-\theta)}\right)
=\pi\,\frac{\cosh(2(\pi-\theta))}{\sinh(2\pi)}.
\]
The expression is even in~$\theta$, so \eqref{eq:lattice-symbol} follows
on $[-\pi,\pi]$, and periodicity gives the full symbol.
The minimum and maximum of the displayed formula occur respectively at
$|\theta|=\pi$ and $\theta=0$, which gives the sharp constants
$A_{\ZZ}$ and $B_{\ZZ}$.

Finally, since $K(m,n)=k(m-n)$, Parseval's identity for the discrete
Fourier transform gives
\[
\sum_{m,n\in\ZZ}c_m\overline{c_n}\,K(m,n)
=\frac{1}{2\pi}\int_{-\pi}^{\pi}
\left(\sum_{\ell\in\ZZ}k(\ell)e^{-\ii\ell\theta}\right)
|\widetilde c(\theta)|^2\dd\theta,
\]
which is exactly \eqref{eq:lattice-symbol-form}.
\end{proof}

\begin{proposition}[Poisson specialization of the standard symbol sampling criterion]%
\label{prop:poisson-lattice-sampling}
Define the closed shift-generated Hilbert subspaces
\[
\mathcal H_K(\ZZ)
:=
\overline{\operatorname{span}}\{K(\cdot,n):n\in\ZZ\}
\subset \mathcal H_K,
\]
\[
\mathcal S_{\ZZ}
:=
\overline{\operatorname{span}}\{\Psi_n:n\in\ZZ\}
\subset L^2_0(\omega).
\]
Thus $\mathcal H_K(\ZZ)$ and $\mathcal S_{\ZZ}$ are the Hilbert closures
of finite linear combinations of the corresponding kernel sections and
secants.
Set
\begin{equation}\label{eq:lattice-sampling-constants}
A_{\ZZ}:=\frac{\pi}{\sinh(2\pi)},
\qquad
B_{\ZZ}:=\pi\coth(2\pi).
\end{equation}
Applying the background principal-symbol criterion above to the explicit
Poisson-strip symbol of Proposition~\ref{prop:poisson-lattice-symbol} gives
the following Riesz and sampling conclusions.  The criterion and the passage
from a positive Toeplitz symbol to Riesz bases, restriction isomorphisms, and
cardinal interpolation are standard; the specialization made here is the
closed Poisson symbol
$\sigma(\theta)=\pi\cosh(2(\pi-|\theta|))/\sinh(2\pi)$ and the induced
constants $A_{\ZZ},B_{\ZZ}$.  The families
$\{K(\cdot,n)\}_{n\in\ZZ}$ and $\{\Psi_n\}_{n\in\ZZ}$ are Riesz bases for
$\mathcal H_K(\ZZ)$ and $\mathcal S_{\ZZ}$, respectively.  More
precisely, for every
finitely supported scalar sequence $(c_n)_{n\in\ZZ}$ one has
\begin{equation}\label{eq:lattice-riesz-kernel}
A_{\ZZ}\sum_{n\in\ZZ}|c_n|^2
\le
\left\|
\sum_{n\in\ZZ}c_nK(\cdot,n)
\right\|_{\mathcal H_K}^2
\le
B_{\ZZ}\sum_{n\in\ZZ}|c_n|^2,
\end{equation}
\begin{equation}\label{eq:lattice-riesz-secant}
A_{\ZZ}\sum_{n\in\ZZ}|c_n|^2
\le
\left\|
\sum_{n\in\ZZ}c_n\Psi_n
\right\|_{L^2(\omega)}^2
\le
B_{\ZZ}\sum_{n\in\ZZ}|c_n|^2.
\end{equation}
Consequently, the synthesis maps
\[
\mathcal C_K(c):=\sum_{n\in\ZZ}c_nK(\cdot,n),
\qquad
\mathcal C_S(c):=\sum_{n\in\ZZ}c_n\Psi_n,
\qquad c\in c_{00}(\ZZ),
\]
extend uniquely to topological isomorphisms
\[
\mathcal C_K:\ell^2(\ZZ)\xrightarrow{\sim}\mathcal H_K(\ZZ),
\qquad
\mathcal C_S:\ell^2(\ZZ)\xrightarrow{\sim}\mathcal S_{\ZZ}.
\]
If $F\in\mathcal H_K(\ZZ)$, then
\begin{equation}\label{eq:lattice-sampling-HK}
A_{\ZZ}\|F\|_{\mathcal H_K}^2
\le
\sum_{n\in\ZZ}|F(n)|^2
\le
B_{\ZZ}\|F\|_{\mathcal H_K}^2.
\end{equation}
Consequently, the restriction map
\[
R_{\ZZ}:\mathcal H_K(\ZZ)\to \ell^2(\ZZ),
\qquad
R_{\ZZ}F:=(F(n))_{n\in\ZZ},
\]
is an isomorphism of Hilbert spaces.  Equivalently, every
$\alpha\in\ell^2(\ZZ)$ admits a unique interpolant
$F_\alpha\in\mathcal H_K(\ZZ)$ satisfying
\[
F_\alpha(n)=\alpha_n
\qquad(n\in\ZZ).
\]
\end{proposition}

\begin{proof}
We spell out the Poisson specialization of the background principal-symbol
criterion because the explicit constants and later observation-channel
bounds come from Proposition~\ref{prop:poisson-lattice-symbol}; no new
abstract sampling theorem is being asserted here.
For finitely supported $c=(c_n)_{n\in\ZZ}$ set
\[
F_c:=\sum_{n\in\ZZ}c_nK(\cdot,n),
\qquad
\widetilde c(\theta):=\sum_{n\in\ZZ}c_ne^{-\ii n\theta},
\qquad
\theta\in[-\pi,\pi].
\]
By Theorem~\ref{thm:mode-gram-kernel},
\[
\|F_c\|_{\mathcal H_K}^2
=\sum_{m,n\in\ZZ}c_m\overline{c_n}\,K(m,n)
=\sum_{m,n\in\ZZ}c_m\overline{c_n}\,\frac{2}{4+(m-n)^2}.
\]
Thus \eqref{eq:lattice-symbol-form} from
Proposition~\ref{prop:poisson-lattice-symbol} gives
\[
\|F_c\|_{\mathcal H_K}^2
=\frac{1}{2\pi}\int_{-\pi}^{\pi}\sigma(\theta)
|\widetilde c(\theta)|^2\dd\theta.
\]
Combining the sharp symbol bounds from
Proposition~\ref{prop:poisson-lattice-symbol} with Parseval for
$\widetilde c$ proves \eqref{eq:lattice-riesz-kernel}.  Since
$\mathcal U\Psi_n=K(\cdot,n)$ by
Theorem~\ref{thm:mode-space-rkhs}, \eqref{eq:lattice-riesz-secant}
follows immediately.
The two inequalities show that $\mathcal C_K$ and $\mathcal C_S$
are bounded above and below on the dense subspace $c_{00}(\ZZ)$ of
$\ell^2(\ZZ)$.  Hence they extend uniquely to bounded topological
isomorphisms from $\ell^2(\ZZ)$ onto their closed ranges.  By definition
those closed ranges are precisely $\mathcal H_K(\ZZ)$ and
$\mathcal S_{\ZZ}$, so the stated Riesz-basis assertions follow.

Now let $G$ be the Toeplitz Gram operator on $\ell^2(\ZZ)$ associated
with the matrix $(K(m,n))_{m,n\in\ZZ}$.  For finitely supported $c$ one
has $R_{\ZZ}F_c=Gc$, so
\[
\sum_{n\in\ZZ}|F_c(n)|^2
=\|Gc\|_{\ell^2}^2
=\frac{1}{2\pi}\int_{-\pi}^{\pi}\sigma(\theta)^2
|\widetilde c(\theta)|^2\dd\theta.
\]
Since $A_{\ZZ}\le \sigma\le B_{\ZZ}$, comparison with
\eqref{eq:lattice-symbol-form} yields
\[
A_{\ZZ}\|F_c\|_{\mathcal H_K}^2
\le
\sum_{n\in\ZZ}|F_c(n)|^2
\le
B_{\ZZ}\|F_c\|_{\mathcal H_K}^2.
\]
Passing to limits in $\mathcal H_K(\ZZ)$ proves
\eqref{eq:lattice-sampling-HK}.

The symbol bounds imply that $G$ extends to a boundedly invertible
operator on $\ell^2(\ZZ)$.  Thus for every $\alpha\in\ell^2(\ZZ)$ there is
a unique $c\in\ell^2(\ZZ)$ with $Gc=\alpha$.  By
\eqref{eq:lattice-riesz-kernel}, the series $\sum_n c_nK(\cdot,n)$
converges in $\mathcal H_K(\ZZ)$ to some $F_\alpha$, and then
$R_{\ZZ}F_\alpha=\alpha$.  Uniqueness follows from the lower bound in
\eqref{eq:lattice-sampling-HK}.
\end{proof}

\subsection{Poisson cardinal reconstruction and observation-channel sampling}%
\label{subsec:poisson-cardinal-observation}

The preceding symbol theorem is now applied only at the integer orbit.  The
cardinal kernel below interpolates data inside the closed lattice subspace
$\mathcal H_K(\ZZ)$, and the following observation-channel theorem applies the
same projection to the Poisson secant profiles from the entropy comparison.
Thus the samples determine the lattice projection and its quotient class; they
determine the full ambient strip function only when that function already lies
in $\mathcal H_K(\ZZ)$.

\begin{corollary}[Explicit lattice cardinal kernel and norm]%
\label{cor:poisson-cardinal-reconstruction}
Under the hypotheses of Proposition~\ref{prop:poisson-lattice-sampling}, let
\[
L:=R_{\ZZ}^{-1}\delta_0\in\mathcal H_K(\ZZ).
\]
Here $\mathcal H_K(\ZZ)$ is the closed span of the kernel sections
$\{K(\cdot,n):n\in\ZZ\}$.
Then $L(n)=\delta_{0n}$ for every $n\in\ZZ$.  For every
$\alpha=(\alpha_n)_{n\in\ZZ}\in\ell^2(\ZZ)$, the series
\[
F_\alpha(z):=\sum_{n\in\ZZ}\alpha_nL(z-n)
\]
converges in $\mathcal H_K(\ZZ)$ and uniformly on compact subsets of the
strip $\mathbb S$, and it is the unique function in $\mathcal H_K(\ZZ)$
satisfying
\[
F_\alpha(n)=\alpha_n
\qquad(n\in\ZZ).
\]
If
\[
\widetilde\alpha(\theta):=\sum_{n\in\ZZ}\alpha_ne^{-\ii n\theta},
\qquad \theta\in[-\pi,\pi],
\]
then
\begin{equation}\label{eq:cardinal-norm-formula}
\|F_\alpha\|_{\mathcal H_K}^2
=\frac{1}{2\pi}\int_{-\pi}^{\pi}
\frac{|\widetilde\alpha(\theta)|^2}{\sigma(\theta)}\dd\theta,
\end{equation}
where $\sigma$ is the lattice symbol from \eqref{eq:lattice-symbol}.  If
$d\in\ell^2(\ZZ)$ is the unique sequence satisfying
\[
\widetilde d(\theta)=\sigma(\theta)^{-1}
\qquad\text{for a.e. }\theta\in[-\pi,\pi],
\]
then
\[
L(z)=\sum_{n\in\ZZ}d_nK(z,n)
\]
in $\mathcal H_K$, and for almost every $\theta\in[-\pi,\pi]$ and every
$m\in\ZZ$ one has
\begin{equation}\label{eq:cardinal-kernel-fourier}
\widehat L(\theta+2\pi m)
=\frac{\pi e^{-2|\theta+2\pi m|}}{\sigma(\theta)}.
\end{equation}
\end{corollary}

\begin{proof}
For $n\in\ZZ$ define the translation operator
$\tau_nF(z):=F(z-n)$.  Since
\[
K(z-n,m)=K(z,m+n),
\]
integer translations preserve $\mathcal H_K(\ZZ)$.  By definition,
$R_{\ZZ}L=\delta_0$.  If $S_n$ denotes the shift operator on
$\ell^2(\ZZ)$ given by $(S_na)_m:=a_{m-n}$, then
$R_{\ZZ}\tau_n=S_nR_{\ZZ}$, so for every $n\in\ZZ$
one has
\[
R_{\ZZ}(\tau_nL)=S_n\delta_0=\delta_n.
\]
Hence $\tau_nL$ is the unique interpolant of $\delta_n$, and therefore
$L(n)=\delta_{0n}$.

If $\alpha$ has finite support, then
\[
R_{\ZZ}\left(\sum_{n\in\ZZ}\alpha_nL(\cdot-n)\right)
=\sum_{n\in\ZZ}\alpha_n\delta_n
=\alpha.
\]
By the uniqueness part of
Proposition~\ref{prop:poisson-lattice-sampling}, the corresponding interpolant
is exactly
\[
F_\alpha=\sum_{n\in\ZZ}\alpha_nL(\cdot-n).
\]
For general $\alpha\in\ell^2(\ZZ)$, let $\alpha^{(N)}$ be the truncation
to $\{-N,\dots,N\}$.  Then $\alpha^{(N)}\to\alpha$ in $\ell^2(\ZZ)$, and
the continuity of $R_{\ZZ}^{-1}$ implies
\[
F_{\alpha^{(N)}}=R_{\ZZ}^{-1}\alpha^{(N)}
\longrightarrow
R_{\ZZ}^{-1}\alpha=:F_\alpha
\]
in $\mathcal H_K(\ZZ)$.  Since each $F_{\alpha^{(N)}}$ is a finite
cardinal sum, the series for $F_\alpha$ converges in $\mathcal H_K(\ZZ)$.
Proposition~\ref{prop:gram-space-poisson-image} gives the uniform estimate
\[
|F(z)|\le \frac{1}{\sqrt{2(1-r^2)}}\,\|F\|_{\mathcal H_K}
\qquad(|\Im z|\le r<1),
\]
so $\mathcal H_K$-norm convergence implies uniform convergence on every
compact subset of~$\mathbb S$.  The interpolation identity for $F_\alpha$
follows from the continuity of $R_{\ZZ}$.

The norm formula is the reciprocal-symbol identity
\eqref{eq:imported-reciprocal-symbol-norm} from the background
principal-symbol criterion with the Poisson symbol
\eqref{eq:lattice-symbol}.  Concretely, first suppose that $\alpha$
has finite support and write
\[
F_\alpha=\sum_{n\in\ZZ}c_nK(\cdot,n).
\]
Then $Gc=\alpha$, where $G$ is the Toeplitz Gram operator from the proof
of Proposition~\ref{prop:poisson-lattice-sampling}.  In the discrete Fourier
representation this becomes
\[
\widetilde\alpha(\theta)=\sigma(\theta)\widetilde c(\theta)
\qquad\text{for a.e. }\theta\in[-\pi,\pi].
\]
Using \eqref{eq:lattice-symbol-form}, we obtain
\[
\|F_\alpha\|_{\mathcal H_K}^2
=\frac{1}{2\pi}\int_{-\pi}^{\pi}
\sigma(\theta)|\widetilde c(\theta)|^2\dd\theta
=\frac{1}{2\pi}\int_{-\pi}^{\pi}
\frac{|\widetilde\alpha(\theta)|^2}{\sigma(\theta)}\dd\theta.
\]
Since $1/\sigma\in L^\infty([-\pi,\pi])$, the right-hand side is
continuous in $\alpha\in\ell^2(\ZZ)$, so the formula extends to all
$\alpha$ by density.

Now let $d:=G^{-1}\delta_0$.  Then $R_{\ZZ}^{-1}\delta_0=L$ gives
\[
L=\sum_{n\in\ZZ}d_nK(\cdot,n)
\]
in $\mathcal H_K$, and the discrete Fourier transform diagonalizes $G$,
so
\[
\widetilde d(\theta)=\sigma(\theta)^{-1}
\qquad\text{for a.e. }\theta\in[-\pi,\pi].
\]
Taking Fourier transforms on~$\RR$ and using
$\widehat{K(\cdot,n)}(\xi)=\pi e^{-2|\xi|}e^{-\ii n\xi}$, we find
\[
\widehat L(\xi)
=\pi e^{-2|\xi|}\sum_{n\in\ZZ}d_ne^{-\ii n\xi}.
\]
Writing $\xi=\theta+2\pi m$ with $\theta\in[-\pi,\pi]$ and $m\in\ZZ$,
the periodicity of the discrete Fourier series yields
\[
\sum_{n\in\ZZ}d_ne^{-\ii n(\theta+2\pi m)}
=\widetilde d(\theta)
=\sigma(\theta)^{-1},
\]
which proves \eqref{eq:cardinal-kernel-fourier}.
\end{proof}

\begin{theorem}[Integer-orbit sampling of Poisson observation channels]%
\label{thm:poisson-observation-lattice-sampling}
Let $\nu$ be a probability measure on~$\RR$ with finite mean
$\bar\gamma$, let $t>0$, and let $\widetilde\delta_t$ be the centred
fluctuation profile from Theorem~\ref{thm:poisson-channel-rkhs}.  Put
\[
F_t:=\mathcal U\widetilde\delta_t\in\mathcal H_K,
\]
and let $\Pi_{\ZZ}$ denote the orthogonal projection of $\mathcal H_K$ onto
$\mathcal H_K(\ZZ)$.  Define the integer-orbit observation samples
\[
\alpha_n(t):=F_t(n),\qquad n\in\ZZ .
\]
These samples are not abstract translate samples: by
Theorem~\ref{thm:poisson-channel-rkhs} they are the evaluations of the
centred Poisson observation profile against the secant modes at the integer
orbit.  The projection $\Pi_{\ZZ}$ is therefore the lattice component of the
same observation channel used in the entropy comparison, as isolated in
Corollary~\ref{cor:integer-orbit-projection}; the constants and
norm formula below use the exact Poisson symbol \eqref{eq:lattice-symbol}.
Then
\begin{equation}\label{eq:poisson-observation-integer-samples}
\alpha_n(t)
=\langle \widetilde\delta_t,\Psi_n\rangle_{L^2(\omega)}
=\frac1t\int_{\RR}
(\gamma-\bar\gamma)\,
K\!\left(n,\frac{\gamma-\bar\gamma}{t}\right)\dd\nu(\gamma),
\qquad n\in\ZZ .
\end{equation}
Moreover $\alpha(t):=(\alpha_n(t))_{n\in\ZZ}\in\ell^2(\ZZ)$ and
\begin{equation}\label{eq:poisson-observation-lattice-stability}
A_{\ZZ}\|\Pi_{\ZZ}F_t\|_{\mathcal H_K}^2
\le
\sum_{n\in\ZZ}|\alpha_n(t)|^2
\le
B_{\ZZ}\|\Pi_{\ZZ}F_t\|_{\mathcal H_K}^2
\le
B_{\ZZ}\|\widetilde\delta_t\|_{L^2(\omega)}^2.
\end{equation}
The cardinal reconstruction with these observable samples is exactly the
lattice projection, not a reconstruction of the ambient channel unless the
ambient channel already lies in the lattice subspace:
\begin{equation}\label{eq:poisson-observation-cardinal-projection}
\Pi_{\ZZ}F_t(z)
=\sum_{n\in\ZZ}\alpha_n(t)L(z-n)
\qquad(z\in\mathbb S),
\end{equation}
with convergence in $\mathcal H_K(\ZZ)$ and uniformly on compact subsets
of~$\mathbb S$, and
\begin{equation}\label{eq:poisson-observation-cardinal-norm}
\|\Pi_{\ZZ}F_t\|_{\mathcal H_K}^2
=\frac1{2\pi}\int_{-\pi}^{\pi}
\frac{|\widetilde\alpha_t(\theta)|^2}{\sigma(\theta)}\dd\theta,
\qquad
\widetilde\alpha_t(\theta):=
\sum_{n\in\ZZ}\alpha_n(t)e^{-\ii n\theta}.
\end{equation}
Finally,
\[
F_t-\Pi_{\ZZ}F_t\in\mathcal H_K(\ZZ)^\perp
\qquad\text{and}\qquad
(F_t-\Pi_{\ZZ}F_t)(n)=0\quad(n\in\ZZ).
\]
Thus the integer samples reconstruct $\Pi_{\ZZ}F_t$ from the data
$(F_t(n))_{n\in\ZZ}$.  They reconstruct the full Poisson observation channel
$F_t$ if and only if $F_t\in\mathcal H_K(\ZZ)$; otherwise they recover exactly
the observable lattice component and nothing from the orthogonal component.
\end{theorem}

\begin{proof}
Theorem~\ref{thm:poisson-channel-rkhs} gives
$F_t=\mathcal U\widetilde\delta_t\in\mathcal H_K$ and the formula
\[
F_t(a)=\frac1t\int_{\RR}
(\gamma-\bar\gamma)\,
K\!\left(a,\frac{\gamma-\bar\gamma}{t}\right)\dd\nu(\gamma).
\]
Taking $a=n\in\ZZ$ gives the last expression in
\eqref{eq:poisson-observation-integer-samples}.  Since
$\mathcal U\Psi_n=K(\cdot,n)$ and evaluation in $\mathcal H_K$ is
represented by $K(\cdot,n)$,
\[
F_t(n)
=\langle F_t,K(\cdot,n)\rangle_{\mathcal H_K}
=\langle \widetilde\delta_t,\Psi_n\rangle_{L^2(\omega)}.
\]
This proves \eqref{eq:poisson-observation-integer-samples}.

For every $n\in\ZZ$, the vector $K(\cdot,n)$ belongs to
$\mathcal H_K(\ZZ)$.  Hence
$F_t-\Pi_{\ZZ}F_t$ is orthogonal to $K(\cdot,n)$, and therefore
\[
(\Pi_{\ZZ}F_t)(n)
=\langle \Pi_{\ZZ}F_t,K(\cdot,n)\rangle_{\mathcal H_K}
=\langle F_t,K(\cdot,n)\rangle_{\mathcal H_K}
=F_t(n)=\alpha_n(t).
\]
Applying the sampling inequality
\eqref{eq:lattice-sampling-HK} from
Proposition~\ref{prop:poisson-lattice-sampling} to
$\Pi_{\ZZ}F_t\in\mathcal H_K(\ZZ)$ proves that
$\alpha(t)\in\ell^2(\ZZ)$ and gives the first two inequalities in
\eqref{eq:poisson-observation-lattice-stability}.  The last inequality
follows from the contractivity of the orthogonal projection and the
unitarity of~$\mathcal U$.

By Corollary~\ref{cor:poisson-cardinal-reconstruction}, there is a unique
element of $\mathcal H_K(\ZZ)$ with lattice data $\alpha(t)$, and it is
the cardinal sum in
\eqref{eq:poisson-observation-cardinal-projection}.  Since
$\Pi_{\ZZ}F_t$ has exactly those samples, uniqueness identifies the
cardinal sum with $\Pi_{\ZZ}F_t$.  The convergence assertion and the norm
identity \eqref{eq:poisson-observation-cardinal-norm} are precisely the
convergence and norm statements of that corollary applied to
$\alpha=\alpha(t)$.

The residual belongs to $\mathcal H_K(\ZZ)^\perp$ by definition of
$\Pi_{\ZZ}$, and the displayed computation above already shows that it
vanishes at every integer.  If $F_t\in\mathcal H_K(\ZZ)$, then
$\Pi_{\ZZ}F_t=F_t$, so the cardinal reconstruction is the full channel.
Conversely, if the integer-orbit cardinal reconstruction is the full
channel, then $F_t=\Pi_{\ZZ}F_t\in\mathcal H_K(\ZZ)$.
\end{proof}

\begin{corollary}[Exact quotient obstruction for integer observations]%
\label{cor:poisson-integer-observation-quotient}
Let
\[
\mathcal N_{\ZZ}:=\{F\in\mathcal H_K:F(n)=0\text{ for every }n\in\ZZ\}.
\]
Then
\[
\mathcal N_{\ZZ}=\mathcal H_K(\ZZ)^\perp,
\]
and the integer restriction map
\[
R_{\ZZ}:\mathcal H_K\longrightarrow\ell^2(\ZZ),
\qquad R_{\ZZ}F:=(F(n))_{n\in\ZZ},
\]
is a bounded surjection with kernel~$\mathcal N_{\ZZ}$.  Hence it induces an
isomorphism
\[
\overline R_{\ZZ}:\mathcal H_K/\mathcal N_{\ZZ}\xrightarrow{\sim}\ell^2(\ZZ).
\]
For $\alpha\in\ell^2(\ZZ)$, the unique element of the coset
$\overline R_{\ZZ}^{-1}\alpha$ with minimal $\mathcal H_K$-norm is the
cardinal interpolant $F_\alpha\in\mathcal H_K(\ZZ)$ from
Corollary~\ref{cor:poisson-cardinal-reconstruction}, and its norm is
\[
\inf\{\|F\|_{\mathcal H_K}^2:R_{\ZZ}F=\alpha\}
=\|F_\alpha\|_{\mathcal H_K}^2
=\frac{1}{2\pi}\int_{-\pi}^{\pi}
\frac{|\widetilde\alpha(\theta)|^2}{\sigma(\theta)}\dd\theta .
\]
Consequently the integer observations of a Poisson channel
$F_t=\mathcal U\widetilde\delta_t$ determine exactly its quotient class
$F_t+\mathcal N_{\ZZ}$ and its minimal-norm representative
$\Pi_{\ZZ}F_t$.  They determine the full channel precisely when
$F_t\in\mathcal H_K(\ZZ)$; otherwise the nonzero component in
$\mathcal N_{\ZZ}=\mathcal H_K(\ZZ)^\perp$ is an intrinsic invisible
obstruction to full recovery from integer samples.
\end{corollary}

\begin{proof}
For $F\in\mathcal H_K$ and $n\in\ZZ$, the reproducing property gives
\[
F(n)=\langle F,K(\cdot,n)\rangle_{\mathcal H_K}.
\]
Thus $F$ vanishes on every integer if and only if it is orthogonal to every
kernel section $K(\cdot,n)$, and this is equivalent to orthogonality to their
closed span $\mathcal H_K(\ZZ)$.  Hence
$\mathcal N_{\ZZ}=\mathcal H_K(\ZZ)^\perp$.

For arbitrary $F\in\mathcal H_K$, the preceding equality gives
\[
R_{\ZZ}F=R_{\ZZ}\Pi_{\ZZ}F,
\]
because $F-\Pi_{\ZZ}F\in\mathcal N_{\ZZ}$.  Applying
Proposition~\ref{prop:poisson-lattice-sampling} to
$\Pi_{\ZZ}F\in\mathcal H_K(\ZZ)$ yields
\[
\|R_{\ZZ}F\|_{\ell^2}^2
=\|R_{\ZZ}\Pi_{\ZZ}F\|_{\ell^2}^2
\le B_{\ZZ}\|\Pi_{\ZZ}F\|_{\mathcal H_K}^2
\le B_{\ZZ}\|F\|_{\mathcal H_K}^2,
\]
so $R_{\ZZ}$ is bounded on the full ambient RKHS.  Its kernel is exactly
$\mathcal N_{\ZZ}$ by definition.  Surjectivity follows because
Corollary~\ref{cor:poisson-cardinal-reconstruction} supplies, for every
$\alpha\in\ell^2(\ZZ)$, an element $F_\alpha\in\mathcal H_K(\ZZ)$ with
$R_{\ZZ}F_\alpha=\alpha$.  The first isomorphism theorem for Hilbert spaces
therefore gives the displayed quotient isomorphism.

Let $F$ satisfy $R_{\ZZ}F=\alpha$.  Then $F-F_\alpha\in\mathcal N_{\ZZ}$, so
the already proved identity
$\mathcal N_{\ZZ}=\mathcal H_K(\ZZ)^\perp$ gives the orthogonal decomposition
\[
F=F_\alpha+(F-F_\alpha),
\qquad F_\alpha\in\mathcal H_K(\ZZ),\quad
F-F_\alpha\in\mathcal H_K(\ZZ)^\perp .
\]
Consequently
\[
\|F\|_{\mathcal H_K}^2
=\|F_\alpha\|_{\mathcal H_K}^2+
\|F-F_\alpha\|_{\mathcal H_K}^2,
\]
which proves both minimality and uniqueness of the minimal-norm
representative.  The reciprocal-symbol norm formula is exactly
\eqref{eq:cardinal-norm-formula}.  Applying the same quotient statement to
the Poisson observation samples in
Theorem~\ref{thm:poisson-observation-lattice-sampling} gives the final
assertions about $F_t$, including the full-recovery criterion and the
invisible orthogonal obstruction.
\end{proof}





\begin{corollary}[Lattice-projection invisible budget]
\label{thm:poisson-integer-invisible-obstruction}
The subspace
\[
\mathcal N_{\ZZ}=\mathcal H_K(\ZZ)^\perp
\]
is infinite dimensional.  More sharply, if $E\subset[0,2\pi)$ has positive
measure, then there exists a nonzero $F_E\in\mathcal N_{\ZZ}$ whose Fourier
transform is supported in $E\cup(E+2\pi)$.  Consequently, for every integer
data sequence $\alpha\in\ell^2(\ZZ)$ and every
$\rho\ge \|F_\alpha\|_{\mathcal H_K}$, the observed fiber inside the RKHS ball
\[
\mathbb B_\rho:=\{F\in\mathcal H_K:
\|F\|_{\mathcal H_K}\le\rho\}
\]
is exactly
\[
\{F\in\mathbb B_\rho:R_{\ZZ}F=\alpha\}
=F_\alpha+
\{G\in\mathcal N_{\ZZ}:
\|G\|_{\mathcal H_K}^2
\le \rho^2-\|F_\alpha\|_{\mathcal H_K}^2
\},
\]
where $F_\alpha\in\mathcal H_K(\ZZ)$ is the minimal-norm cardinal interpolant
from Corollary~\ref{cor:poisson-cardinal-reconstruction}.  If
$\rho>\|F_\alpha\|_{\mathcal H_K}$ this set contains infinitely many mutually
distinct functions with the same integer samples.

Moreover, as a consequence of this lattice projection decomposition, its exact
Chebyshev radius in~$\mathcal H_K$ is
\begin{equation}\label{eq:integer-orbit-minimax-radius}
r_{\mathrm{obs}}(\alpha,\rho)
:=\bigl(\rho^2-\|F_\alpha\|_{\mathcal H_K}^2\bigr)^{1/2}.
\end{equation}
Equivalently, after the integer samples have selected the quotient class and
the minimal-norm lattice projection $F_\alpha$, any reconstruction rule whose
output depends only on the integer data~$\alpha$ has minimax error on
\[
\{F\in\mathbb B_\rho:R_{\ZZ}F=\alpha\}
\]
at least $r_{\mathrm{obs}}(\alpha,\rho)$; this bound is attained by the
cardinal representative $F_\alpha$ and is therefore sharp.  Applied to the
Poisson observation samples
$\alpha(t)=(F_t(n))_{n\in\ZZ}$ in
Theorem~\ref{thm:poisson-observation-lattice-sampling}, the cardinal
reconstruction $\Pi_{\ZZ}F_t=F_{\alpha(t)}$ is therefore the projection selected
by the exact symbol/cardinal theorem; the minimax statement is the budget form
of the same quotient obstruction, while every nonzero component in
~$\mathcal N_{\ZZ}$ is intrinsically unrecoverable from the integer orbit.
\end{corollary}

\begin{proof}
Choose a nonzero $q\in L^2(E)$, regard $q$ as extended by zero to all of
$\RR$, and set
\[
\widehat F_E(\xi)
:=q(\xi)\mathbf 1_E(\xi)
-q(\xi-2\pi)\mathbf 1_{E+2\pi}(\xi).
\]
Equivalently, on the second branch write $\xi=\eta+2\pi$ with
$\eta\in E$ and put $\widehat F_E(\eta+2\pi)=-q(\eta)$; outside
$E\cup(E+2\pi)$ the transform is zero.
Since $E\cup(E+2\pi)$ has finite measure and is bounded, the norm identity in
Proposition~\ref{prop:gram-space-poisson-image} gives
\[
\|F_E\|_{\mathcal H_K}^2
=\frac{1}{2\pi^2}\int_E
\bigl(e^{2|\xi|}+e^{2|\xi+2\pi|}\bigr)|q(\xi)|^2\dd\xi<\infty,
\]
so $F_E\in\mathcal H_K$ and $F_E\ne0$.  For every $n\in\ZZ$, the kernel-section
Fourier formula from Proposition~\ref{prop:gram-space-poisson-image} gives
\[
\langle F_E,K(\cdot,n)\rangle_{\mathcal H_K}
=\frac1{2\pi}\int_{\RR}
\widehat F_E(\xi)e^{\ii n\xi}\dd\xi .
\]
Substituting the definition of~$\widehat F_E$ and changing variables in the
second integral yield
\[
\begin{split}
\langle F_E,K(\cdot,n)\rangle_{\mathcal H_K}
&=\frac1{2\pi}\int_E q(\xi)e^{\ii n\xi}\dd\xi
 -\frac1{2\pi}\int_E q(\xi)e^{\ii n(\xi+2\pi)}\dd\xi \\
&=0.
\end{split}
\]
Hence $F_E$ vanishes at every integer and belongs to
$\mathcal N_{\ZZ}$ by Corollary~\ref{cor:poisson-integer-observation-quotient}.
Taking disjoint positive-measure subsets $E_j\subset[0,2\pi)$ produces
nonzero invisible vectors with disjoint Fourier supports, and therefore
linearly independent invisible vectors.  Thus $\mathcal N_{\ZZ}$ is infinite
dimensional.

Now fix $\alpha\in\ell^2(\ZZ)$.  Corollary~\ref{cor:poisson-integer-observation-quotient}
identifies the unrestricted fiber over~$\alpha$ as
\[
F_\alpha+\mathcal N_{\ZZ},
\]
with the orthogonal decomposition
$F_\alpha\perp\mathcal N_{\ZZ}$.  Hence, for $G\in\mathcal N_{\ZZ}$,
\[
\|F_\alpha+G\|_{\mathcal H_K}^2
=\|F_\alpha\|_{\mathcal H_K}^2+\|G\|_{\mathcal H_K}^2.
\]
The displayed identity proves the exact description of the intersection with
$\mathbb B_\rho$.  If $\rho>\|F_\alpha\|_{\mathcal H_K}$, its invisible radius
$r_{\mathrm{obs}}(\alpha,\rho)$ is positive, and the infinite-dimensionality of
$\mathcal N_{\ZZ}$ gives infinitely many distinct $G$ in the invisible ball.

It remains to prove the exact minimax radius.  Let
\[
\mathcal F_{\alpha,\rho}:=
F_\alpha+
\{G\in\mathcal N_{\ZZ}:
\|G\|_{\mathcal H_K}\le r_{\mathrm{obs}}(\alpha,\rho)\}.
\]
The cardinal representative satisfies
\[
\sup_{F\in\mathcal F_{\alpha,\rho}}
\|F-F_\alpha\|_{\mathcal H_K}
=r_{\mathrm{obs}}(\alpha,\rho),
\]
so the Chebyshev radius is at most the displayed value.  Conversely take any
candidate output $Y\in\mathcal H_K$.  Decompose
\[
Y-F_\alpha=Y_0+Y_1,
\qquad Y_0\in\mathcal H_K(\ZZ),\quad Y_1\in\mathcal N_{\ZZ}.
\]
Orthogonality gives
\[
\sup_{\|G\|\le r_{\mathrm{obs}}}
\|F_\alpha+G-Y\|_{\mathcal H_K}^2
=
\|Y_0\|_{\mathcal H_K}^2+
\sup_{\|G\|\le r_{\mathrm{obs}}}
\|G-Y_1\|_{\mathcal H_K}^2 .
\]
The last supremum is at least $r_{\mathrm{obs}}^2$: if $Y_1=0$, choose any
invisible vector of norm $r_{\mathrm{obs}}$; if $Y_1\ne0$, take
$G=-r_{\mathrm{obs}}Y_1/\|Y_1\|_{\mathcal H_K}$.  If $Y\ne F_\alpha$, this
choice gives a strictly larger worst-case error: either $\|Y_0\|>0$, or
$Y_1\ne0$ and the displayed invisible supremum is at least
$(r_{\mathrm{obs}}+\|Y_1\|_{\mathcal H_K})^2$ when $r_{\mathrm{obs}}>0$, while
for $r_{\mathrm{obs}}=0$ the singleton fiber already forces $Y=F_\alpha$ for
zero error.  Thus every possible output has worst-case error at least
$r_{\mathrm{obs}}(\alpha,\rho)$, with equality only at $F_\alpha$.  Since all
procedures depending only on the integer samples must use a single output for
the common datum~$\alpha$, the same lower bound holds for all such procedures,
and the preceding paragraph shows that $F_\alpha$ uniquely attains it.  For the Poisson samples $\alpha(t)$, Theorem~\ref{thm:poisson-observation-lattice-sampling}
identifies $F_{\alpha(t)}$ with $\Pi_{\ZZ}F_t$, yielding the stated
observation-channel specialization.
\end{proof}

\begin{corollary}[Local-to-global dichotomy for integer samples]
\label{cor:poisson-integer-local-global-dichotomy}
Let $R_{\ZZ}:\mathcal H_K\to\ell^2(\ZZ)$ be the integer restriction map and
let $\mathcal N_{\ZZ}=\ker R_{\ZZ}$.  If the admissible class is the lattice
subspace $\mathcal H_K(\ZZ)$, then $R_{\ZZ}$ is one-to-one and onto
$\ell^2(\ZZ)$, so the integer data determine the unique lattice element with
those samples.  If the admissible class is the full ambient RKHS, then for
every $F\in\mathcal H_K$ with data $\alpha=R_{\ZZ}F$ one has
\[
R_{\ZZ}^{-1}\alpha=\Pi_{\ZZ}F+\mathcal N_{\ZZ},
\]
and $\mathcal N_{\ZZ}$ is infinite dimensional.  Hence the same data determine
exactly the quotient class $F+\mathcal N_{\ZZ}$ and its minimal-norm
representative $\Pi_{\ZZ}F$, but not the full channel.  Applied to the Poisson
channel $F_t=\mathcal U\widetilde\delta_t$, full recovery from the integer
orbit is therefore exactly the additional lattice-subspace condition
$F_t\in\mathcal H_K(\ZZ)$; otherwise every nonzero vector in
$\mathcal N_{\ZZ}$ gives a distinct globally different channel with identical
local integer observations.
\end{corollary}

\begin{proof}
Corollary~\ref{cor:poisson-integer-observation-quotient} identifies
$\mathcal N_{\ZZ}$ with the orthogonal complement
$\mathcal H_K(\ZZ)^\perp$ and gives
\[
R_{\ZZ}^{-1}\alpha=F_\alpha+\mathcal N_{\ZZ},
\]
where $F_\alpha=\Pi_{\ZZ}F$ is the cardinal minimal-norm representative.  On
the lattice subspace $\mathcal H_K(\ZZ)$,
Proposition~\ref{prop:poisson-lattice-sampling} makes $R_{\ZZ}$ a bounded
isomorphism onto $\ell^2(\ZZ)$, so the samples determine the unique lattice
element $F_\alpha$.  In the full ambient RKHS,
Corollary~\ref{thm:poisson-integer-invisible-obstruction} proves that
$\mathcal N_{\ZZ}$ is infinite dimensional.  Therefore the full ambient fiber
is the nontrivial affine space $F_\alpha+\mathcal N_{\ZZ}$, while the quotient
class and the orthogonal projection are uniquely determined.  The final
assertion is the same statement applied to $F_t=\mathcal U\widetilde\delta_t$
in Theorem~\ref{thm:poisson-observation-lattice-sampling}.
\end{proof}

\begin{corollary}[Poisson integer-orbit normal form and energy identity]%
\label{cor:poisson-lattice-normal-form}
Let $\nu$ be a probability measure on $\RR$ with finite mean $\bar\gamma$,
let $t>0$, and set
\[
F_t:=\mathcal U\widetilde\delta_t\in\mathcal H_K,
\qquad
\alpha_n(t):=F_t(n)
\quad(n\in\ZZ).
\]
Then the integer-orbit data of the Poisson channel have the exact normal form
\begin{equation}\label{eq:poisson-orbit-normal-form}
F_t=F_{\alpha(t)}+G_t,
\qquad
F_{\alpha(t)}:=\sum_{n\in\ZZ}\alpha_n(t)L(\cdot-n)
=\Pi_{\ZZ}F_t,
\qquad
G_t\in\mathcal N_{\ZZ},
\end{equation}
where the series converges in $\mathcal H_K(\ZZ)$ and uniformly on compact
subsets of the strip.  Moreover
\begin{equation}\label{eq:poisson-orbit-energy-identity}
\|F_t\|_{\mathcal H_K}^2
=
\frac1{2\pi}\int_{-\pi}^{\pi}
\frac{|\widetilde\alpha_t(\theta)|^2}{\sigma(\theta)}\dd\theta
+\|G_t\|_{\mathcal H_K}^2,
\end{equation}
with
\[
\sigma(\theta)=\frac{\pi\cosh(2(\pi-|\theta|))}{\sinh(2\pi)},
\qquad -\pi\le\theta\le\pi .
\]
Consequently the following are equivalent:
\begin{enumerate}[label=\textup{(\roman*)},leftmargin=*]
\item the integer orbit reconstructs the full Poisson observation channel;
\item $F_t\in\mathcal H_K(\ZZ)$;
\item $G_t=0$;
\item equality holds in the energy identity
\[
\|F_t\|_{\mathcal H_K}^2
=
\frac1{2\pi}\int_{-\pi}^{\pi}
\frac{|\widetilde\alpha_t(\theta)|^2}{\sigma(\theta)}\dd\theta .
\]
\end{enumerate}
If these equivalent conditions fail, then
$F_t+\mathcal N_{\ZZ}$ is the full ambiguity class of ambient RKHS channels
with the same integer samples, and every datum-dependent reconstruction map
from integer samples alone has sharp worst-case invisible error given by
Corollary~\ref{thm:poisson-integer-invisible-obstruction}.  This consequence uses no
new sampling criterion beyond the imported background criterion stated in
Subsection~\ref{subsec:lattice-background-poisson-symbol}; the Poisson-specific
inputs are the closed symbol \eqref{eq:lattice-symbol}, the cardinal formula
\eqref{eq:cardinal-kernel-fourier}, the norm formula
\eqref{eq:cardinal-norm-formula}, and the channel realization in
Theorem~\ref{thm:poisson-channel-rkhs}.
\end{corollary}

\begin{proof}
Proposition~\ref{prop:poisson-lattice-symbol} gives a continuous positive
Toeplitz symbol $\sigma$ for the Gram matrix
$(K(m,n))_{m,n\in\ZZ}$ and the sharp bounds
$A_{\ZZ}\le\sigma\le B_{\ZZ}$.  The proof of
Proposition~\ref{prop:poisson-lattice-sampling} uses only this symbol
identity, Parseval's theorem on $\ell^2(\ZZ)$, and the bounded inverse
multiplier $1/\sigma$; hence its Riesz bounds, sampling inequality, and
restriction isomorphism are precisely the standard principal-symbol
consequences of the positive symbol.  The proof of
Corollary~\ref{cor:poisson-cardinal-reconstruction} then applies that same
restriction isomorphism to the datum $\delta_0$ and to general
$\alpha\in\ell^2(\ZZ)$, so the cardinal kernel and reciprocal-symbol norm
are also consequences of the same symbol calculation.

Now insert the Poisson observation channel.  By
Theorem~\ref{thm:poisson-channel-rkhs},
$F_t=\mathcal U\widetilde\delta_t\in\mathcal H_K$, and
Theorem~\ref{thm:poisson-observation-lattice-sampling} identifies the sample
sequence $\alpha(t)=(F_t(n))_{n\in\ZZ}$ as an element of $\ell^2(\ZZ)$.  The
same theorem gives
\[
\Pi_{\ZZ}F_t(z)=\sum_{n\in\ZZ}\alpha_n(t)L(z-n)
\]
with the stated convergence.  Define
\[
G_t:=F_t-\Pi_{\ZZ}F_t .
\]
Since $\Pi_{\ZZ}$ is the orthogonal projection onto $\mathcal H_K(\ZZ)$,
Corollary~\ref{cor:poisson-integer-observation-quotient} gives
$G_t\in\mathcal H_K(\ZZ)^\perp=\mathcal N_{\ZZ}$.  This proves the normal
form \eqref{eq:poisson-orbit-normal-form}.

The two summands in \eqref{eq:poisson-orbit-normal-form} are orthogonal.
Therefore
\[
\|F_t\|_{\mathcal H_K}^2
=\|\Pi_{\ZZ}F_t\|_{\mathcal H_K}^2+
\|G_t\|_{\mathcal H_K}^2.
\]
Applying the reciprocal-symbol norm formula
\eqref{eq:poisson-observation-cardinal-norm} to
$\Pi_{\ZZ}F_t=F_{\alpha(t)}$ gives
\eqref{eq:poisson-orbit-energy-identity}.

The equivalence of the four reconstruction criteria is now immediate:
$F_t$ is recovered by the integer-orbit cardinal reconstruction if and only if
$F_t=\Pi_{\ZZ}F_t$, which is equivalent to
$F_t\in\mathcal H_K(\ZZ)$, to $G_t=0$, and, by the orthogonal energy identity,
to equality in \eqref{eq:poisson-orbit-energy-identity} without the
invisible term.

Finally, if $R_{\ZZ}F=\alpha(t)$, then
Corollary~\ref{cor:poisson-integer-observation-quotient} gives
$F-F_{\alpha(t)}\in\mathcal N_{\ZZ}$.  Conversely every element of
$F_{\alpha(t)}+\mathcal N_{\ZZ}$ has the same integer samples.  Since
$F_t-F_{\alpha(t)}=G_t\in\mathcal N_{\ZZ}$, this fiber is also the coset
$F_t+\mathcal N_{\ZZ}$.  The sharp worst-case invisible-error statement is
exactly Corollary~\ref{thm:poisson-integer-invisible-obstruction} applied to
this coset.  Thus the nonstandard, Poisson-specific inputs in the lattice
chain are the explicit symbol, the channel realization, the cardinal norm, and
the quotient-by-invisible-components obstruction; the abstract positive-symbol
criterion remains imported background.
\end{proof}

\begin{theorem}[Classification and optimality of integer Poisson observations]%
\label{thm:integer-poisson-observation-classification}
Let $\alpha\in\ell^2(\ZZ)$ be an integer observation sequence, and let
$F_\alpha\in\mathcal H_K(\ZZ)$ be the cardinal representative from
Corollary~\ref{cor:poisson-cardinal-reconstruction}.  Then the following
statements hold.
\begin{enumerate}[label=\textup{(\roman*)},leftmargin=*]
\item The complete class of ambient strip RKHS channels with integer data
      $\alpha$ is the affine Hilbert space
      \[
      \mathfrak F(\alpha)
      :=\{F\in\mathcal H_K:F(n)=\alpha_n\text{ for all }n\in\ZZ\}
      =F_\alpha+\mathcal N_{\ZZ}.
      \]
      Thus $F_\alpha$ is the unique element of this class lying in
      $\mathcal H_K(\ZZ)$ and the unique element of minimal
      $\mathcal H_K$-norm.
\item The exact excess-energy budget of any $F\in\mathfrak F(\alpha)$ is
      \begin{equation}\label{eq:classified-integer-excess-budget}
      \|F-F_\alpha\|_{\mathcal H_K}^2
      =\|F\|_{\mathcal H_K}^2
      -\frac1{2\pi}\int_{-\pi}^{\pi}
      \frac{|\widetilde\alpha(\theta)|^2}{\sigma(\theta)}\dd\theta .
      \end{equation}
      In particular, for a Poisson observation channel
      $F_t=\mathcal U\widetilde\delta_t$ with integer data
      $\alpha(t)$, the right-hand side is exactly the invisible energy
      $\|G_t\|_{\mathcal H_K}^2$ in
      Corollary~\ref{cor:poisson-lattice-normal-form}.
\item If $\mathcal Q:\mathcal H_K\to Y$ is any map into a Hausdorff space
      whose value depends only on the integer samples, then
      $\mathcal Q$ is constant on each affine class
      $F_\alpha+\mathcal N_{\ZZ}$.  Hence every datum-dependent
      reconstruction or invariant of the ambient channel factors through the
      quotient
      \[
      \mathcal H_K/\mathcal N_{\ZZ}
      \simeq \mathcal H_K(\ZZ)
      \simeq \ell^2(\ZZ),
      \]
      and no such procedure can distinguish two channels that differ by a
      nonzero vector of~$\mathcal N_{\ZZ}$.
\item Among all maps from integer data into~$\mathcal H_K$, the cardinal map
      $\alpha\mapsto F_\alpha$ is the unique map that is exact on the lattice
      subspace $\mathcal H_K(\ZZ)$ and has minimal worst-case error on every
      fiber cut by a ball
      $\{F\in\mathfrak F(\alpha):\|F\|_{\mathcal H_K}\le\rho\}$,
      $\rho\ge\|F_\alpha\|_{\mathcal H_K}$.  Its sharp error radius is
      \[
      \bigl(\rho^2-\|F_\alpha\|_{\mathcal H_K}^2\bigr)^{1/2}.
      \]
\end{enumerate}
Consequently the lattice endpoint of the Cayley--Poisson theorem chain is not
an additional sampling hypothesis: after the Cayley--Haar normalization fixes
the Poisson strip kernel, integer data have exactly one canonical output, the
cardinal projection, and the only obstruction to full recovery is the
classified invisible summand~$\mathcal N_{\ZZ}$.
\end{theorem}

\begin{proof}
Corollary~\ref{cor:poisson-integer-observation-quotient} gives
$\mathcal N_{\ZZ}=\ker R_{\ZZ}=\mathcal H_K(\ZZ)^\perp$ and shows that
$R_{\ZZ}:\mathcal H_K\to\ell^2(\ZZ)$ is onto.  It also identifies, for each
$\alpha$, the cardinal interpolant $F_\alpha\in\mathcal H_K(\ZZ)$ as the
unique minimal-norm element with samples~$\alpha$.  Therefore
$R_{\ZZ}^{-1}\alpha=F_\alpha+\ker R_{\ZZ}=F_\alpha+\mathcal N_{\ZZ}$, proving
the classification in (i).

Since $F_\alpha\in\mathcal H_K(\ZZ)$ and
$F-F_\alpha\in\mathcal N_{\ZZ}=\mathcal H_K(\ZZ)^\perp$, Pythagoras gives
\[
\|F\|_{\mathcal H_K}^2
=\|F_\alpha\|_{\mathcal H_K}^2+
\|F-F_\alpha\|_{\mathcal H_K}^2 .
\]
The reciprocal-symbol norm formula
\eqref{eq:cardinal-norm-formula} gives
\[
\|F_\alpha\|_{\mathcal H_K}^2
=\frac1{2\pi}\int_{-\pi}^{\pi}
\frac{|\widetilde\alpha(\theta)|^2}{\sigma(\theta)}\dd\theta,
\]
which proves \eqref{eq:classified-integer-excess-budget}.  Taking
$F=F_t$ and $\alpha=\alpha(t)$ gives the asserted identification with
$\|G_t\|_{\mathcal H_K}^2$ by
Corollary~\ref{cor:poisson-lattice-normal-form}.

If $\mathcal Q$ depends only on integer samples and
$F_1-F_2\in\mathcal N_{\ZZ}$, then $R_{\ZZ}F_1=R_{\ZZ}F_2$, hence
$\mathcal Q(F_1)=\mathcal Q(F_2)$.  Thus $\mathcal Q$ is constant on the
cosets of~$\mathcal N_{\ZZ}$ and factors through
$\mathcal H_K/\mathcal N_{\ZZ}$.  The quotient is identified with
$\mathcal H_K(\ZZ)$ by orthogonal projection and with $\ell^2(\ZZ)$ by
Corollary~\ref{cor:poisson-integer-observation-quotient}, proving (iii).

Finally fix $\alpha$ and $\rho\ge\|F_\alpha\|_{\mathcal H_K}$.  By (i) and
orthogonality,
\[
\{F\in\mathfrak F(\alpha):\|F\|_{\mathcal H_K}\le\rho\}
=F_\alpha+
\{G\in\mathcal N_{\ZZ}:
\|G\|_{\mathcal H_K}
\le(\rho^2-
\|F_\alpha\|_{\mathcal H_K}^2)^{1/2}\}.
\]
The sharp Chebyshev-radius computation for this ball in the invisible
subspace is exactly the minimax part of
Corollary~\ref{thm:poisson-integer-invisible-obstruction}; it shows that the
center $F_\alpha$ attains the displayed radius and that no other output can
do better, with uniqueness under the stated exactness on
$\mathcal H_K(\ZZ)$.  This proves (iv).  The final sentence is the
combination of the classification, the explicit Poisson symbol used in the
cardinal norm, and the observation-channel realization from
Theorem~\ref{thm:poisson-channel-rkhs}.
\end{proof}

\begin{remark}
The sampling step just used is standard once a principal shift-invariant
symbol is bounded above and below.  This is the role of the Riesz-translate
and RKHS sampling criteria in
\cite{deBoorDeVoreRon1994,RonShen1995,AldroubiGrochenig2001,NashedWalter1991,NashedSun2010,ChengJiangSun2016}.
Related nod-function, derivative-sampling, Gaussian, and Lorentz/Cauchy
examples appear in
\cite{RomeroUlanovskiiZlotnikov2024,BaranovBelovGrochenig2022,GrochenigRomeroStockler2020,KiselevMininNovikov2016,KiselevMininNovikov2019}.
The Poisson-specific contribution here is not a new abstract sampling
criterion.  It is the strip realization that makes the symbol
$\sigma(\theta)=\pi\cosh(2(\pi-|\theta|))/\sinh(2\pi)$, the cardinal
kernel, and the interpolation norm explicit for the Poisson observation
channel; in particular it yields the norm identity
\eqref{eq:cardinal-norm-formula} and the quotient obstruction used above.
\end{remark}
