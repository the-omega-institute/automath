\subsection{Poisson cardinal projection and observation-channel samples}%
\label{subsec:poisson-cardinal-observation}

After inserting the Poisson symbol of
Proposition~\ref{prop:poisson-lattice-symbol} into the imported
principal-symbol criterion, the cardinal kernel is an explicit consequence of
standard interpolation in $\mathcal H_K(\ZZ)$.  The Poisson observation theorem
below is this lattice projection applied to secant profiles: the samples
determine the lattice projection and quotient class, and its statement also
identifies the exact law-side condition for full ambient recovery.

\ifdefined\JFAArticleMain
The secondary explicit-cardinal-kernel corollary is in the Supplementary
Material, Corollary ``Explicit lattice cardinal kernel and norm.''  It defines
\(L=R_{\ZZ}^{-1}\delta_0\), proves cardinal interpolation and compact-uniform
convergence, and gives the reciprocal-symbol norm and Fourier formulas there.
\else
\begin{corollary}[Explicit lattice cardinal kernel and norm]%
\label{cor:poisson-cardinal-reconstruction}
With the Poisson symbol consequences recorded immediately after
Proposition~\ref{prop:poisson-lattice-symbol}, let
\[
L:=R_{\ZZ}^{-1}\delta_0\in\mathcal H_K(\ZZ).
\]
Here $\mathcal H_K(\ZZ)$ is the closed span of the kernel sections
$\{K(\cdot,n):n\in\ZZ\}$.
Then $L(n)=\delta_{0n}$ for every $n\in\ZZ$.  For every
$\alpha=(\alpha_n)_{n\in\ZZ}\in\ell^2(\ZZ)$, the series
\[
F_\alpha(z):=\sum_{n\in\ZZ}\alpha_nL(z-n)
\]
converges in $\mathcal H_K(\ZZ)$ and uniformly on compact subsets of the
strip $\mathbb S$, and it is the unique function in $\mathcal H_K(\ZZ)$
satisfying
\[
F_\alpha(n)=\alpha_n
\qquad(n\in\ZZ).
\]
If
\[
\widetilde\alpha(\theta):=\sum_{n\in\ZZ}\alpha_ne^{-\ii n\theta},
\qquad \theta\in[-\pi,\pi],
\]
then
\begin{equation}\label{eq:cardinal-norm-formula}
\|F_\alpha\|_{\mathcal H_K}^2
=\frac{1}{2\pi}\int_{-\pi}^{\pi}
\frac{|\widetilde\alpha(\theta)|^2}{\sigma(\theta)}\dd\theta,
\end{equation}
where $\sigma$ is the lattice symbol from \eqref{eq:lattice-symbol}.  If
$d\in\ell^2(\ZZ)$ is the unique sequence satisfying
\[
\widetilde d(\theta)=\sigma(\theta)^{-1}
\qquad\text{for a.e. }\theta\in[-\pi,\pi],
\]
then
\[
L(z)=\sum_{n\in\ZZ}d_nK(z,n)
\]
in $\mathcal H_K$, and for almost every $\theta\in[-\pi,\pi]$ and every
$m\in\ZZ$ one has
\begin{equation}\label{eq:cardinal-kernel-fourier}
\widehat L(\theta+2\pi m)
=\frac{\pi e^{-2|\theta+2\pi m|}}{\sigma(\theta)}.
\end{equation}
\end{corollary}

\begin{proof}
For $n\in\ZZ$ define the translation operator
$\tau_nF(z):=F(z-n)$.  Since
\[
K(z-n,m)=K(z,m+n),
\]
integer translations preserve $\mathcal H_K(\ZZ)$.  By definition,
$R_{\ZZ}L=\delta_0$.  If $S_n$ denotes the shift operator on
$\ell^2(\ZZ)$ given by $(S_na)_m:=a_{m-n}$, then
$R_{\ZZ}\tau_n=S_nR_{\ZZ}$, so for every $n\in\ZZ$
one has
\[
R_{\ZZ}(\tau_nL)=S_n\delta_0=\delta_n.
\]
Hence $\tau_nL$ is the unique interpolant of $\delta_n$, and therefore
$L(n)=\delta_{0n}$.

If $\alpha$ has finite support, then
\[
R_{\ZZ}\left(\sum_{n\in\ZZ}\alpha_nL(\cdot-n)\right)
=\sum_{n\in\ZZ}\alpha_n\delta_n
=\alpha.
\]
By the uniqueness part of
the standard sampling consequences of
Lemma~\ref{lem:imported-principal-symbol-sampling}, the corresponding
interpolant
is exactly
\[
F_\alpha=\sum_{n\in\ZZ}\alpha_nL(\cdot-n).
\]
For general $\alpha\in\ell^2(\ZZ)$, let $\alpha^{(N)}$ be the truncation
to $\{-N,\dots,N\}$.  Then $\alpha^{(N)}\to\alpha$ in $\ell^2(\ZZ)$, and
the continuity of $R_{\ZZ}^{-1}$ implies
\[
F_{\alpha^{(N)}}=R_{\ZZ}^{-1}\alpha^{(N)}
\longrightarrow
R_{\ZZ}^{-1}\alpha=:F_\alpha
\]
in $\mathcal H_K(\ZZ)$.  Since each $F_{\alpha^{(N)}}$ is a finite
cardinal sum, the series for $F_\alpha$ converges in $\mathcal H_K(\ZZ)$.
Proposition~\ref{prop:gram-space-poisson-image} gives the uniform estimate
\[
|F(z)|\le \frac{1}{\sqrt{2(1-r^2)}}\,\|F\|_{\mathcal H_K}
\qquad(|\Im z|\le r<1),
\]
so $\mathcal H_K$-norm convergence implies uniform convergence on every
compact subset of~$\mathbb S$.  The interpolation identity for $F_\alpha$
follows from the continuity of $R_{\ZZ}$.

The norm formula is the reciprocal-symbol identity
\eqref{eq:imported-reciprocal-symbol-norm} from
Lemma~\ref{lem:imported-principal-symbol-sampling} with the Poisson symbol
\eqref{eq:lattice-symbol}.  Concretely, first suppose that $\alpha$
has finite support and write
\[
F_\alpha=\sum_{n\in\ZZ}c_nK(\cdot,n).
\]
Then $Gc=\alpha$, where $G$ is the Toeplitz Gram operator from the proof
of the same imported criterion.  In the discrete Fourier
representation this becomes
\[
\widetilde\alpha(\theta)=\sigma(\theta)\widetilde c(\theta)
\qquad\text{for a.e. }\theta\in[-\pi,\pi].
\]
Using \eqref{eq:lattice-symbol-form}, we obtain
\[
\|F_\alpha\|_{\mathcal H_K}^2
=\frac{1}{2\pi}\int_{-\pi}^{\pi}
\sigma(\theta)|\widetilde c(\theta)|^2\dd\theta
=\frac{1}{2\pi}\int_{-\pi}^{\pi}
\frac{|\widetilde\alpha(\theta)|^2}{\sigma(\theta)}\dd\theta.
\]
Since $1/\sigma\in L^\infty([-\pi,\pi])$, the right-hand side is
continuous in $\alpha\in\ell^2(\ZZ)$, so the formula extends to all
$\alpha$ by density.

Now let $d:=G^{-1}\delta_0$.  Then $R_{\ZZ}^{-1}\delta_0=L$ gives
\[
L=\sum_{n\in\ZZ}d_nK(\cdot,n)
\]
in $\mathcal H_K$, and the discrete Fourier transform diagonalizes $G$,
so
\[
\widetilde d(\theta)=\sigma(\theta)^{-1}
\qquad\text{for a.e. }\theta\in[-\pi,\pi].
\]
Taking Fourier transforms on~$\RR$ and using
$\widehat{K(\cdot,n)}(\xi)=\pi e^{-2|\xi|}e^{-\ii n\xi}$, we find
\[
\widehat L(\xi)
=\pi e^{-2|\xi|}\sum_{n\in\ZZ}d_ne^{-\ii n\xi}.
\]
Writing $\xi=\theta+2\pi m$ with $\theta\in[-\pi,\pi]$ and $m\in\ZZ$,
the periodicity of the discrete Fourier series yields
\[
\sum_{n\in\ZZ}d_ne^{-\ii n(\theta+2\pi m)}
=\widetilde d(\theta)
=\sigma(\theta)^{-1},
\]
which proves \eqref{eq:cardinal-kernel-fourier}.
\end{proof}

\ifdefined\JFAArticleMain
The secondary Fourier-uniqueness lemma showing that the full continuous
profile determines the centred law is in the Supplementary Material, Lemma
``Observation profile determines the centred law.''
\else
\begin{observationprofilelemma}[Observation profile determines the centred law]%
\label{lem:observation-profile-determines-law}
Write \(X_c:=\gamma-\bar{\gamma}\).  Let \(\kappa_t\) be the distribution of
\(X_c/t\), and define the finite signed
measure
\[
\lambda_t(\dd s):=s\,\kappa_t(\dd s).
\]
For the Poisson observation channel in
Theorem~\ref{thm:poisson-observation-lattice-sampling},
\[
q_{F_t}(\xi)=e^{2|\xi|}\widehat F_t(\xi)=\pi\,\widehat{\lambda_t}(\xi)
\]
with the Fourier convention used throughout this section.  Hence \(q_{F_t}\)
determines \(\lambda_t\) by Fourier uniqueness.  It determines \(\kappa_t\) on
\(\RR\setminus\{0\}\) by
\[
\kappa_t(B)=\int_B s^{-1}\lambda_t(\dd s),
\qquad 0\notin\overline B,
\]
and then determines the atom at the origin from total mass:
\[
\kappa_t(\{0\})=1-\kappa_t(\RR\setminus\{0\}).
\]
Thus the full continuous profile \(q_{F_t}\) determines the centred law of
\(X_c/t\).  For the integer-orbit cardinal reconstruction formula to
represent this full channel, \(q_{F_t}\) must be \(2\pi\)-periodic,
equivalently \(\kappa_t\) must be supported on \(\ZZ\).
\end{observationprofilelemma}

\begin{proof}
The Fourier transform calculation in the proof of
Theorem~\ref{thm:poisson-observation-lattice-sampling} gives
\[
\widehat F_t(\xi)
=\pi e^{-2|\xi|}\int_{\RR}s e^{-\ii s\xi}\,\kappa_t(\dd s).
\]
This is exactly \(q_{F_t}(\xi)=\pi\widehat{\lambda_t}(\xi)\).  Fourier
uniqueness for finite signed measures gives \(\lambda_t\).  On any Borel set
bounded away from the origin, multiplication by \(s^{-1}\) is bounded, so the
displayed inversion recovers \(\kappa_t\) there.  Increasing such sets to
\(\RR\setminus\{0\}\) recovers all off-origin mass, and the remaining mass is
the atom at \(0\) because \(\kappa_t\) is a probability measure.

The last assertion is the periodicity criterion proved in
Theorem~\ref{thm:poisson-observation-lattice-sampling}: the integer-orbit
cardinal formula first recovers the projection \(\Pi_{\ZZ}F_t\), and this
specific reconstruction formula represents the full channel precisely when
the law-side profile is \(2\pi\)-periodic.  Since \(\lambda_t=s\kappa_t\), the
periodicity condition is equivalent to
\((e^{-\ii2\pi s}-1)s\,\kappa_t(\dd s)=0\), hence to support of \(\kappa_t\) on
\(\ZZ\) after allowing the harmless atom at \(0\).
\end{proof}

\begin{theorem}[Integer-orbit projection of Poisson observation channels]%
\label{thm:poisson-observation-lattice-sampling}
\label{prop:poisson-law-lattice-periodicity}
Let $\nu$ be a probability measure on~$\RR$ with finite mean
$\bar{\gamma}$, let $t>0$, and let $\mathfrak d_t$ be the centred
fluctuation profile from Theorem~\ref{thm:poisson-channel-rkhs}.  Put
\[
  X_c:=\gamma-\bar{\gamma},
  \qquad
F_t:=\mathcal U\mathfrak d_t\in\mathcal H_K,
\]
and let $\Pi_{\ZZ}$ denote the orthogonal projection of $\mathcal H_K$ onto
$\mathcal H_K(\ZZ)$.  Define the integer-orbit observation samples
\[
\alpha_n(t):=F_t(n),\qquad n\in\ZZ .
\]
By Theorem~\ref{thm:poisson-channel-rkhs}, these samples are evaluations of
the centred Poisson observation profile against the secant modes at the
integer orbit.  The projection $\Pi_{\ZZ}$ is the lattice component of that
observation channel, with constants and norm formula determined by the exact
Poisson symbol \eqref{eq:lattice-symbol}.  The reconstruction of the lattice
component uses the standard principal-symbol projection mechanism recorded in
Lemma~\ref{lem:imported-principal-symbol-sampling}.  Full ambient recovery is
equivalent to the law-side condition
\[
\nu\bigl(\{\gamma\in\RR:X_c\in t\ZZ\}\bigr)=1;
\]
the theorem lists the equivalent RKHS and periodicity formulations below.
Then
\begin{equation}\label{eq:poisson-observation-integer-samples}
\alpha_n(t)
=\langle \mathfrak d_t,\Psi_n\rangle_{L^2(\omega)}
=\frac1t\int_{\RR}
(\gamma-\bar{\gamma})\,
K\!\left(n,\frac{\gamma-\bar{\gamma}}{t}\right)\dd\nu(\gamma),
\qquad n\in\ZZ .
\end{equation}
Moreover $\alpha(t):=(\alpha_n(t))_{n\in\ZZ}\in\ell^2(\ZZ)$ and
\begin{equation}\label{eq:poisson-observation-lattice-stability}
A_{\ZZ}\|\Pi_{\ZZ}F_t\|_{\mathcal H_K}^2
\le
\sum_{n\in\ZZ}|\alpha_n(t)|^2
\le
B_{\ZZ}\|\Pi_{\ZZ}F_t\|_{\mathcal H_K}^2
\le
B_{\ZZ}\|\mathfrak d_t\|_{L^2(\omega)}^2.
\end{equation}
The cardinal reconstruction with these observable samples is the lattice
projection:
\begin{equation}\label{eq:poisson-observation-cardinal-projection}
\Pi_{\ZZ}F_t(z)
=\sum_{n\in\ZZ}\alpha_n(t)L(z-n)
\qquad(z\in\mathbb S),
\end{equation}
with convergence in $\mathcal H_K(\ZZ)$ and uniformly on compact subsets
of~$\mathbb S$, and
\begin{equation}\label{eq:poisson-observation-cardinal-norm}
\|\Pi_{\ZZ}F_t\|_{\mathcal H_K}^2
=\frac1{2\pi}\int_{-\pi}^{\pi}
\frac{|\widetilde\alpha_t(\theta)|^2}{\sigma(\theta)}\dd\theta,
\qquad
\widetilde\alpha_t(\theta):=
\sum_{n\in\ZZ}\alpha_n(t)e^{-\ii n\theta}.
\end{equation}
Finally,
\[
F_t-\Pi_{\ZZ}F_t\in\mathcal H_K(\ZZ)^\perp
\qquad\text{and}\qquad
(F_t-\Pi_{\ZZ}F_t)(n)=0\quad(n\in\ZZ).
\]
Thus the integer samples reconstruct $\Pi_{\ZZ}F_t$ from the data
$(F_t(n))_{n\in\ZZ}$.  The next display gives the law-side criterion for this
projection to equal the full observation channel.
Define the law-side spectral profile
\begin{equation}\label{eq:poisson-law-side-profile}
Q_t(\xi)
:=\frac{\pi}{t}\int_{\RR}X_c
e^{-\ii X_c\xi/t}\,\dd\nu(\gamma),
\qquad \xi\in\RR .
\end{equation}
Then $Q_t$ is bounded and continuous, and
\begin{equation}\label{eq:poisson-law-side-q-profile}
q_{F_t}(\xi):=e^{2|\xi|}\widehat F_t(\xi)=Q_t(\xi)
\qquad\text{for a.e. }\xi\in\RR .
\end{equation}
Consequently, the following are equivalent:
\begin{enumerate}[label=(\roman*)]
\item the integer-orbit cardinal reconstruction in
      \eqref{eq:poisson-observation-cardinal-projection} recovers the full
      Poisson observation channel $F_t$;
\item $F_t\in\mathcal H_K(\ZZ)$;
\item the law-side profile $Q_t$ is $2\pi$-periodic, equivalently
      \begin{equation}\label{eq:poisson-law-side-periodicity}
      \int_{\RR}X_c e^{-\ii X_c(\xi+2\pi)/t}\,\dd\nu(\gamma)
      =
      \int_{\RR}X_c e^{-\ii X_c\xi/t}\,\dd\nu(\gamma)
      \qquad(\xi\in\RR).
      \end{equation}
\item the centred law has no mass off the lattice $t\ZZ$, equivalently
      \begin{equation}\label{eq:poisson-law-side-lattice-support}
      \nu\bigl(\{\gamma\in\RR:\gamma-\bar{\gamma}\in t\ZZ\}\bigr)=1 .
      \end{equation}
\end{enumerate}
In this case the lattice norm of the full channel is
\begin{equation}\label{eq:poisson-law-side-lattice-norm}
\|F_t\|_{\mathcal H_K}^2
=\frac{1}{2\pi^2}\int_{-\pi}^{\pi}
\Sigma_1(\xi)|Q_t(\xi)|^2\,\dd\xi
=\frac{1}{2\pi^3}\int_{-\pi}^{\pi}
\sigma(\xi)|Q_t(\xi)|^2\,\dd\xi,
\end{equation}
where
\[
\Sigma_1(\xi):=\sum_{m\in\ZZ}e^{-2|\xi+2\pi m|}
=\frac{\sigma(\xi)}{\pi}.
\]
Equivalently, if $\phi_{\nu_c}(s):=\int_{\RR}e^{\ii sX_c}\dd\nu(\gamma)$
is the characteristic function of the centred law, then, with the Fourier
convention $\widehat f(\xi)=\int f(x)e^{-\ii x\xi}\dd x$,
\[
Q_t(\xi)=-\frac{\pi\ii}{t}\,\phi_{\nu_c}'(-\xi/t),
\]
so the full-recovery condition is precisely the $2\pi$-periodicity of this
signed first characteristic-function channel.  Negative frequencies are read
with the usual conjugate symmetry
$\phi_{\nu_c}(-s)=\overline{\phi_{\nu_c}(s)}$, and no symmetry of the law is
assumed.  Equivalently, the centred law is supported on the affine lattice
$\bar{\gamma}+t\ZZ$.
\end{theorem}

\begin{proof}
Theorem~\ref{thm:poisson-channel-rkhs} gives
$F_t=\mathcal U\mathfrak d_t\in\mathcal H_K$ and the formula
\[
F_t(a)=\frac1t\int_{\RR}
(\gamma-\bar{\gamma})\,
K\!\left(a,\frac{\gamma-\bar{\gamma}}{t}\right)\dd\nu(\gamma).
\]
Taking $a=n\in\ZZ$ gives the last expression in
\eqref{eq:poisson-observation-integer-samples}.  Since
$\mathcal U\Psi_n=K(\cdot,n)$ and evaluation in $\mathcal H_K$ is
represented by $K(\cdot,n)$,
\[
F_t(n)
=\langle F_t,K(\cdot,n)\rangle_{\mathcal H_K}
=\langle \mathfrak d_t,\Psi_n\rangle_{L^2(\omega)}.
\]
This proves \eqref{eq:poisson-observation-integer-samples}.

For every $n\in\ZZ$, the vector $K(\cdot,n)$ belongs to
$\mathcal H_K(\ZZ)$.  Hence
$F_t-\Pi_{\ZZ}F_t$ is orthogonal to $K(\cdot,n)$, and therefore
\[
(\Pi_{\ZZ}F_t)(n)
=\langle \Pi_{\ZZ}F_t,K(\cdot,n)\rangle_{\mathcal H_K}
=\langle F_t,K(\cdot,n)\rangle_{\mathcal H_K}
=F_t(n)=\alpha_n(t).
\]
Applying the sampling inequality proved in the Supplementary Material,
paragraph ``Consequences of the
imported criterion for the Poisson symbol,'' to
$\Pi_{\ZZ}F_t\in\mathcal H_K(\ZZ)$ proves that
$\alpha(t)\in\ell^2(\ZZ)$ and gives the first two inequalities in
\eqref{eq:poisson-observation-lattice-stability}.  The last inequality
follows from the contractivity of the orthogonal projection and the
unitarity of~$\mathcal U$.

By the explicit-cardinal-kernel corollary in the Supplementary Material,
cited above, there is a unique
element of $\mathcal H_K(\ZZ)$ with lattice data $\alpha(t)$, and it is
the cardinal sum in
\eqref{eq:poisson-observation-cardinal-projection}.  Since
$\Pi_{\ZZ}F_t$ has exactly those samples, uniqueness identifies the
cardinal sum with $\Pi_{\ZZ}F_t$.  The convergence assertion and the norm
identity \eqref{eq:poisson-observation-cardinal-norm} are precisely the
convergence and norm statements of that corollary applied to
$\alpha=\alpha(t)$.

The residual belongs to $\mathcal H_K(\ZZ)^\perp$ by definition of
$\Pi_{\ZZ}$, and the displayed computation above already shows that it
vanishes at every integer.

The finite first moment assumption gives
\[
|Q_t(\xi)|\le \frac{\pi}{t}\int_{\RR}|X_c|\,\dd\nu(\gamma),
\]
and dominated convergence gives continuity of~$Q_t$.  By
Theorem~\ref{thm:poisson-channel-rkhs},
\[
F_t(a)=\frac1t\int_{\RR}X_c\,K(a,X_c/t)\dd\nu(\gamma).
\]
The Fourier transform of the Poisson strip kernel in the first variable is
\[
\widehat{K(\cdot,b)}(\xi)=\pi e^{-2|\xi|}e^{-\ii b\xi},
\]
as used in the Fourier formula of the explicit-cardinal-kernel corollary.
Taking the Fourier transform
under the Bochner integral therefore gives
\[
\widehat F_t(\xi)
=\frac{\pi}{t}e^{-2|\xi|}
\int_{\RR}X_c e^{-\ii X_c\xi/t}\dd\nu(\gamma)
=e^{-2|\xi|}Q_t(\xi)
\]
in the distributional, hence a.e., Fourier representative sense.  This proves
\eqref{eq:poisson-law-side-q-profile}.

The equivalence of (i) and (ii) follows directly from the projection formula:
if $F_t\in\mathcal H_K(\ZZ)$, then $\Pi_{\ZZ}F_t=F_t$, so the cardinal
reconstruction is the full channel; conversely, if the integer-orbit cardinal
reconstruction is the full channel, then
$F_t=\Pi_{\ZZ}F_t\in\mathcal H_K(\ZZ)$.  By item~(i) of the integer
projection lemma in the Supplementary Material, a function
$f\in\mathcal H_K$ belongs to $\mathcal H_K(\ZZ)$ if and only if
$q_f(\xi)=e^{2|\xi|}\widehat f(\xi)$ has a $2\pi$-periodic representative
and
\[
\int_{-\pi}^{\pi}\Sigma_1(\xi)|q_f(\xi)|^2\dd\xi<\infty .
\]
For $f=F_t$, the representative is the continuous function $Q_t$.  If
$F_t\in\mathcal H_K(\ZZ)$, the a.e. periodicity from the lemma is therefore
everywhere periodic, because $Q_t(\xi+2\pi)-Q_t(\xi)$ is continuous.  Conversely,
if $Q_t$ is $2\pi$-periodic, the displayed weighted integral is finite since
$Q_t$ is bounded and $\Sigma_1$ is continuous on $[-\pi,\pi]$.  The same lemma
then gives $F_t\in\mathcal H_K(\ZZ)$.  Thus (ii) and (iii) are equivalent, and
\eqref{eq:poisson-law-side-periodicity} is just the definition of the
periodicity of~$Q_t$.

It remains to identify what this periodicity says about the law.  Let
$\kappa$ be the finite positive measure which is the distribution of
$X_c/t$.  Since $\nu$ has finite first moment, the signed measure
\[
\lambda(\dd s):=s\,\kappa(\dd s)
\]
has finite total variation and total mass zero.  Condition
\eqref{eq:poisson-law-side-periodicity} is equivalent to
\[
\widehat\lambda(\xi+2\pi)=\widehat\lambda(\xi)
\qquad(\xi\in\RR).
\]
Thus the finite signed measure $(e^{-\ii2\pi s}-1)\lambda(\dd s)$ has
Fourier transform identically zero, and uniqueness of Fourier transforms of
finite measures gives
\[
(e^{-\ii2\pi s}-1)s\,\kappa(\dd s)=0.
\]
Hence $\kappa$ is supported on $\ZZ\cup\{0\}$; this is just~$\ZZ$.  Therefore
\eqref{eq:poisson-law-side-lattice-support} follows.  Conversely, if
$X_c/t\in\ZZ$ almost surely, then the exponential factor
$e^{-\ii2\pi X_c/t}$ is one almost surely and
\eqref{eq:poisson-law-side-periodicity} follows immediately.  This proves the
equivalence of (iii) and (iv).

When these equivalent conditions hold, the norm formula in item~(i) of the
integer projection lemma in the Supplementary Material gives the first equality
in \eqref{eq:poisson-law-side-lattice-norm}; the identity
$\Sigma_1=\sigma/\pi$ gives the second.  Finally, finite first moment implies
that $\phi_{\nu_c}$ is differentiable and
\[
\phi_{\nu_c}'(s)=\ii\int_{\RR}X_c e^{\ii sX_c}\dd\nu(\gamma),
\]
which yields the characteristic-function expression for~$Q_t$ at
$s=-\xi/t$.  This signed argument is consistent with the conjugate
characteristic-function symmetry used in
Proposition~\ref{prop:poisson-central-two-channel} and keeps the
positive- and negative-frequency data distinguished.
\end{proof}
