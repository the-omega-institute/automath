\section{KL Tail Identity for the Poisson Doob Quotient}%
\label{sec:doob-phi-entropy}

Let \(\nu\) be a probability measure on~\(\RR\) with mean \(\bar\gamma\).
For \(t>0\) put
\[
h_t:=P_t*\nu,\qquad
g_t(x):=P_t(x-\bar\gamma),\qquad
u_t:=h_t/g_t .
\]
The Cauchy generator is normalized by
\[
(Lf)(x)=\frac1\pi\operatorname{p.v.}\int_{\RR}
\frac{f(y)-f(x)}{(x-y)^2}\dd y,
\]
so that \(\partial_tP_t=LP_t\).  This section records only the
KL-specialized consequence of the Doob transform.  The range-local
\(C^{1,1}\) convex-entropy closure used to justify the differentiated identity
is in Appendix~\ref{app:doob-bregman-domain}.  The surrounding literature on
nonlocal and discrete entropy-gradient structures includes the finite-chain
entropy convexity theory of Erbar--Maas~\cite{ErbarMaas2012RicciMarkov},
Maas~\cite{Maas2011EntropyMarkov}, Mielke~\cite{Mielke2013EntropyMarkov},
and Erbar's jump-process entropy gradient flows
\cite{Erbar2014JumpEntropy}.  The identity below is not a curvature statement
in those metric structures; it is the Cauchy Doob quotient's cutoff-closed
Bregman dissipation formula.

\begin{lemma}[Second-moment KL decay to the translated Poisson profile]%
\label{lem:poisson-second-moment-kl-decay}
Let $\nu$ be a probability measure on~$\RR$ with mean $\bar\gamma$ and
finite centred second moment.  Put
\[
h_t:=P_t*\nu,
\qquad
g_t(x):=P_t(x-\bar\gamma).
\]
Then
\[
D_{\mathrm{KL}}(h_t\|g_t)\longrightarrow0
\qquad(t\to\infty).
\]
\end{lemma}

\begin{proof}
Write $X_c=\gamma-\bar\gamma$ and $x=\bar\gamma+ty$.  Since
$g_t(\bar\gamma+ty)=(\pi t)^{-1}(1+y^2)^{-1}$,
\[
u_t(\bar\gamma+ty):=\frac{h_t(\bar\gamma+ty)}{g_t(\bar\gamma+ty)}
=\int_{\RR}\frac{1+y^2}{1+(y-X_c/t)^2}\dd\nu(\gamma).
\]
For $A>0$ decompose $\nu=\nu_A+\nu_A^c$ according to
$|X_c|\le A$ and $|X_c|>A$.  The elementary kernel bound
\begin{equation}\label{eq:second-moment-kl-kernel-bound}
\frac{1+y^2}{1+(y-\varepsilon)^2}
\le 2(1+\varepsilon^2)
\qquad(y,\varepsilon\in\RR)
\end{equation}
implies
\[
0\le u_{t,A}^c(y):=\int_{|X_c|>A}
\frac{1+y^2}{1+(y-X_c/t)^2}\dd\nu(\gamma)
\le 2\int_{|X_c|>A}\left(1+\frac{X_c^2}{t^2}\right)\dd\nu(\gamma).
\]
Hence, for $t\ge1$,
\begin{equation}\label{eq:second-moment-kl-tail-sup}
\sup_y u_{t,A}^c(y)
\le 2\nu(|X_c|>A)+2\int_{|X_c|>A}X_c^2\dd\nu.
\end{equation}

The bounded part has total mass $m_A:=\nu(|X_c|\le A)$ and first centred
moment $b_A:=\int_{|X_c|\le A}X_c\dd\nu$.  The exact expansion
\[
\frac{1+y^2}{1+(y-\varepsilon)^2}
=1+\frac{2y}{1+y^2}\varepsilon
+\frac{(3y^2-1)\varepsilon^2-2y\varepsilon^3}
{(1+y^2)(1+(y-\varepsilon)^2)}
\]
shows that, for each fixed $A$,
\begin{equation}\label{eq:bounded-truncated-quotient-limit}
u_{t,A}(y):=\int_{|X_c|\le A}
\frac{1+y^2}{1+(y-X_c/t)^2}\dd\nu(\gamma)
=m_A+\frac{2yb_A}{t(1+y^2)}+O_A(t^{-2})
\end{equation}
uniformly in~$y$.  Since
$b_A=-\int_{|X_c|>A}X_c\dd\nu$, Cauchy--Schwarz gives
\[
|b_A|\le \left(\int_{|X_c|>A}X_c^2\dd\nu\right)^{1/2}.
\]
Combining this with
\eqref{eq:second-moment-kl-tail-sup} and
\eqref{eq:bounded-truncated-quotient-limit} yields
\[
\limsup_{t\to\infty}\|u_t-1\|_{L^\infty(\RR)}
\le 1-m_A+2\nu(|X_c|>A)+2\int_{|X_c|>A}X_c^2\dd\nu.
\]
Letting $A\to\infty$ gives $\|u_t-1\|_\infty\to0$.

For all sufficiently large $t$ we therefore have
$|u_t-1|\le1/2$.  With $\delta_t=u_t-1$ and using
$\int g_t\delta_t=\int(h_t-g_t)=0$,
\[
D_{\mathrm{KL}}(h_t\|g_t)
=\int_{\RR}g_t\bigl((1+\delta_t)\log(1+\delta_t)-\delta_t\bigr)\dd x
\le 2\|\delta_t\|_\infty^2,
\]
because $(1+s)\log(1+s)-s\le2s^2$ for $|s|\le1/2$.  The right-hand side
tends to zero.  This is the \(H(T)\to0\) input used below to convert the
finite-interval dissipation balance into the infinite tail identity.
\end{proof}

\begin{theorem}[KL quotient domain theorem]%
\label{thm:kl-local-doob-bregman-domain-closure}
\label{prop:kl-local-doob-bregman-domain-closure}
Let $\nu$ be a probability measure on~$\RR$ with mean~$\bar\gamma$ and finite
centred second moment
\[
\mu_2:=\int_{\RR}(\gamma-\bar\gamma)^2\dd\nu(\gamma)<\infty .
\]
Let $[a,b]\subset(0,\infty)$, set
\[
h_t=P_t*\nu,\qquad g_t=P_t(\cdot-\bar\gamma),\qquad u_t=h_t/g_t,
\]
and let $\chi\in C_c^\infty(\RR)$ satisfy \(0\le\chi\le1\),
\(\chi\equiv1\) on $[-1,1]$,
$\chi_R(x)=\chi(x/R)$.  Then the following facts hold uniformly for
$t\in[a,b]$.
\begin{enumerate}[label=\textup{(\roman*)},leftmargin=*]
\item The quotient range is compact, with the explicit bound
\begin{equation}\label{eq:kl-domain-compact-range}
\frac{1}{2(1+\mu_2/a^2)}
\le u_t(x)\le
2+\frac{2\mu_2}{a^2}
\qquad(x\in\RR).
\end{equation}
In particular, with
\[
m_{a,\nu}:=\frac{1}{2(1+\mu_2/a^2)},\qquad
M_{a,\nu}:=2+\frac{2\mu_2}{a^2},
\]
all values of $u_t$ lie in
$J_{a,\nu}:=[m_{a,\nu},M_{a,\nu}]\Subset(0,\infty)$.

\item The spatial derivative is uniformly bounded.  In particular, after
enlarging the constant if necessary,
\begin{equation}\label{eq:kl-domain-derivative-bound}
\|\partial_xu_t\|_\infty
\le
L_{a,\nu}:=\frac{5}{a}\left(1+\frac{\mu_2}{a^2}\right).
\end{equation}
Consequently $u_t$ is uniformly Lipschitz on
$\RR\times[a,b]$.

\item For $\Phi(s)=s\log s$ and
\[
\Lambda(\alpha,\beta):=\alpha\log\frac{\alpha}{\beta}-\alpha+\beta,
\]
there is a constant $C_{a,\nu}<\infty$ such that
\begin{equation}\label{eq:kl-domain-bregman-majorant}
0\le \Lambda(u_t(x),u_t(y))
\le C_{a,\nu}\min\{|x-y|^2,1\}
\qquad(x,y\in\RR).
\end{equation}
Hence
\begin{equation}\label{eq:kl-domain-pointwise-bregman-majorant}
\frac{g_t(x)\Lambda(u_t(x),u_t(y))}{(x-y)^2}
\le C_{a,\nu}g_t(x)\min\{1,|x-y|^{-2}\}
\qquad(x\ne y),
\end{equation}
and
\begin{equation}\label{eq:kl-domain-absolute-bregman}
\sup_{t\in[a,b]}
\iint_{\RR^2}
\frac{g_t(x)\Lambda(u_t(x),u_t(y))}{(x-y)^2}\dd x\dd y<\infty .
\end{equation}

\item Define the localized entropies
\[
H_R(t):=\int_{\RR}\chi_R(x)g_t(x)\Phi(u_t(x))\dd x,
\qquad
H(t):=\int_{\RR}g_t(x)\Phi(u_t(x))\dd x .
\]
Then \(H_R\in AC([a,b])\) for every \(R\ge1\), \(H\in AC([a,b])\),
\(H_R\to H\) uniformly on \([a,b]\), and there are functions
\(e_R\in L^1([a,b])\), with
\[
\|e_R\|_{L^1([a,b])}\longrightarrow0,
\]
such that, for a.e. \(t\in[a,b]\),
\begin{align}
-\frac{\dd}{\dd t}H_R(t)
&=
\frac{1}{2\pi}\iint_{\RR^2}
\chi_R(x)\chi_R(y)
\frac{g_t(x)\Lambda(u_t(x),u_t(y))
      +g_t(y)\Lambda(u_t(y),u_t(x))}
{(x-y)^2}\dd x\dd y
 + e_R(t).
\label{eq:kl-domain-localized-entropy-balance}
\end{align}
Consequently the cutoff identity converges in \(L^1([a,b])\) to the
unlocalized dissipation identity
\begin{equation}\label{eq:kl-domain-unlocalized-dissipation}
\frac{1}{\pi}\iint_{\RR^2}
\frac{g_t(x)\Lambda(u_t(x),u_t(y))}{(x-y)^2}\dd x\dd y .
\end{equation}

\item Equivalently, the map
\[
H(t):=D_{\mathrm{KL}}(h_t\|g_t)
=\int_{\RR}g_t(x)u_t(x)\log u_t(x)\dd x
\]
is absolutely continuous on $[a,b]$.  For a.e. $t\in[a,b]$ its derivative is
the one-variable cutoff limit
\begin{align}
-H'(t)
&=
\lim_{R\to\infty}
\left[
-\int_{\RR}\chi_R(x)\bigl(1+\log u_t(x)\bigr)Lh_t(x)\dd x
+\int_{\RR}\chi_R(x)u_t(x)Lg_t(x)\dd x
\right],
\label{eq:kl-main-cutoff-derivative}
\end{align}
and this same limit is exactly the two-variable cutoff limit
\begin{align}
\lim_{R\to\infty}
\frac{1}{2\pi}\iint_{\RR^2}
\chi_R(x)\chi_R(y)
\frac{g_t(x)\Lambda(u_t(x),u_t(y))
      +g_t(y)\Lambda(u_t(y),u_t(x))}
{(x-y)^2}\dd x\dd y .
\label{eq:kl-main-cutoff-bregman-limit}
\end{align}
The convergence to this limit holds in \(L^1([a,b])\), and by
\eqref{eq:kl-domain-absolute-bregman} the cutoff can then be removed in the
last display.  Thus, for a.e. \(t\in[a,b]\),
\begin{equation}\label{eq:kl-domain-final-derivative}
-H'(t)=
\frac{1}{\pi}\iint_{\RR^2}
\frac{g_t(x)\Lambda(u_t(x),u_t(y))}{(x-y)^2}\dd x\dd y .
\end{equation}
\end{enumerate}
\end{theorem}

\begin{remark}
The theorem supplies the domain hypotheses used in the tail theorem below:
compact quotient range, a uniform Lipschitz bound, an explicit near/far
Bregman majorant, \(L^1([a,b])\) cutoff convergence from the noncompact tests
\(1+\log u_t\) and \(u_t\), and then the unlocalized derivative identity.
\end{remark}

\begin{proof}
The explicit quotient bounds and the Lipschitz estimate follow from
Lemma~\ref{lem:appendix-poisson-quotient-kernel-calculus} and
Lemma~\ref{lem:doob-range-bregman-integrability}; the lower bound uses the
Jensen step recorded in the proof of that lemma.  With
\(\Phi(s)=s\log s\), Proposition~\ref{prop:doob-bregman-domain-closure}
applies on \([a,b]\).  It gives \(H_R\in AC([a,b])\), uniform convergence
\(H_R\to H\), the \(L^1([a,b])\)-vanishing cutoff error in
\eqref{eq:kl-domain-localized-entropy-balance}, and the unlocalized
Doob--Bregman derivative identity.  The cutoff error is not a formal
remainder: in the notation of
Lemma~\ref{lem:doob-bregman-cutoff-commutator}, specialized to
\(\Phi(s)=s\log s\), it is the explicit near/far commutator
\eqref{eq:phi-cutoff-commutator-error-formula}, and
\eqref{eq:phi-cutoff-commutator-l1-vanishing} gives its
\(L^1([a,b])\)-decay.  The displayed one-variable cutoff formula
\eqref{eq:kl-main-cutoff-derivative} is the same cutoff convention fixed in
Lemma~\ref{lem:cauchy-doob-cutoff-calculus}, applied to the bounded Lipschitz
tests \(1+\log u_t\) and \(u_t\).  The Bregman cutoff limit
\eqref{eq:kl-main-cutoff-bregman-limit} is the corresponding specialization of
\eqref{eq:phi-domain-localized-cutoff-balance}; the near/far majorant and
absolute convergence are the specialization of
\eqref{eq:phi-domain-core-majorant}.  Since the symmetric two terms are equal
after exchanging \(x\) and \(y\), the limit is the one-sided expression
\eqref{eq:kl-domain-unlocalized-dissipation}, which proves
\eqref{eq:kl-domain-final-derivative}.
\end{proof}

\begin{proposition}[Domain inputs for the KL tail identity]%
\label{prop:kl-tail-cutoff-inputs}
Let $\nu$ have mean $\bar\gamma$ and finite centred second moment, and use
\[
h_t=P_t*\nu,\qquad g_t=P_t(\cdot-\bar\gamma),\qquad u_t=h_t/g_t .
\]
Fix \(0<a<b<\infty\), and let \(\chi_R(x)=\chi(x/R)\), where
\(\chi\in C_c^\infty(\RR)\), \(0\le\chi\le1\), and
\(\chi\equiv1\) on \([-1,1]\).  Put
\[
H(t):=D_{\mathrm{KL}}(h_t\|g_t)
=\int_{\RR}g_t(x)u_t(x)\log u_t(x)\dd x
\]
and
\[
\Lambda(\alpha,\beta):=
\alpha\log\frac{\alpha}{\beta}-\alpha+\beta .
\]
Then there are constants
\[
0<m_{a,\nu}\le M_{a,\nu}<\infty,\qquad L_{a,\nu}<\infty,
\]
depending only on \(a\) and the centred second moment of \(\nu\), such that
for all \(t\in[a,b]\) and \(x\in\RR\),
\begin{equation}\label{eq:kl-tail-input-range-lipschitz}
m_{a,\nu}\le u_t(x)\le M_{a,\nu},
\qquad
|\partial_xu_t(x)|\le L_{a,\nu}.
\end{equation}
Moreover, there is a constant \(C_{a,\nu}<\infty\), depending only on the
same data, such that for \(t\in[a,b]\) and \(x\ne y\),
\begin{equation}\label{eq:kl-tail-input-bregman-majorant}
0\le
\frac{g_t(x)\Lambda(u_t(x),u_t(y))}{(x-y)^2}
\le
C_{a,\nu}g_t(x)\min\{1,|x-y|^{-2}\},
\end{equation}
with the same estimate after exchanging \(x\) and \(y\).  The right-hand side
is an \(L^1([a,b]\times\RR^2)\) majorant.

Finally, the localized identities with product cutoffs
\(\chi_R(x)\chi_R(y)\) converge in \(L^1([a,b])\) to the unlocalized Bregman
integral:
\begin{align}
&\int_a^b
\left|
\frac{1}{2\pi}\iint_{\RR^2}\chi_R(x)\chi_R(y)
\frac{
g_t(x)\Lambda(u_t(x),u_t(y))
+g_t(y)\Lambda(u_t(y),u_t(x))}
{(x-y)^2}\dd x\dd y \right.                       \notag\\
&\hspace{35mm}\left.
-\frac{1}{\pi}\iint_{\RR^2}
\frac{g_t(x)\Lambda(u_t(x),u_t(y))}
{(x-y)^2}\dd x\dd y
\right|\dd t
\longrightarrow0 .
\label{eq:kl-tail-input-l1-cutoff}
\end{align}
Equivalently, if
\[
H_R(t):=\int_{\RR}\chi_R(x)g_t(x)u_t(x)\log u_t(x)\dd x,
\]
then the localized balances have errors \(e_R\in L^1([a,b])\) with
\(\|e_R\|_{L^1([a,b])}\to0\), and \(H_R\to H\) uniformly on \([a,b]\).
Consequently \(H\in AC([a,b])\), and for a.e. \(t\in[a,b]\)
\begin{equation}\label{eq:kl-tail-input-unlocalized}
-H'(t)=
\frac{1}{\pi}\iint_{\RR^2}
\frac{g_t(x)\Lambda(u_t(x),u_t(y))}
{(x-y)^2}\dd x\dd y .
\end{equation}
In the principal-value normalization
\[
(Lf)(x)=\frac1\pi\operatorname{p.v.}\int_{\RR}
\frac{f(y)-f(x)}{(x-y)^2}\dd y,
\]
this identity is the cutoff limit
\begin{align}
-H'(t)
&=
\lim_{R\to\infty}
\left[
-\int_{\RR}\chi_R(x)(1+\log u_t(x))Lh_t(x)\dd x
+\int_{\RR}\chi_R(x)u_t(x)Lg_t(x)\dd x
\right]                                        \notag\\
&=
\lim_{R\to\infty}
\frac{1}{2\pi}\iint_{\RR^2}\chi_R(x)\chi_R(y)
\frac{
g_t(x)\Lambda(u_t(x),u_t(y))
+g_t(y)\Lambda(u_t(y),u_t(x))}
{(x-y)^2}\dd x\dd y .
\label{eq:kl-tail-input-pv-identity}
\end{align}
The displayed occurrences of \(Lh_t\) and \(Lg_t\) are the
principal-value Cauchy-generator limits fixed by
Lemma~\ref{lem:cauchy-doob-cutoff-calculus}.
\end{proposition}

\begin{proof}
These are exactly the KL specialization of
Theorem~\ref{thm:kl-local-doob-bregman-domain-closure}.  The compact range
and Lipschitz bound are \eqref{eq:kl-domain-compact-range} and
\eqref{eq:kl-domain-derivative-bound}.  The majorant is
\eqref{eq:kl-domain-pointwise-bregman-majorant} and its exchanged-variable
counterpart; its \(L^1([a,b]\times\RR^2)\) integrability is the
near-diagonal/far-field estimate used in
\eqref{eq:kl-domain-absolute-bregman}.  The \(L^1([a,b])\) convergence of the
localized Bregman integrals is \eqref{eq:kl-domain-localized-entropy-balance}
and \eqref{eq:kl-main-cutoff-bregman-limit}; the derivative identity is
\eqref{eq:kl-domain-final-derivative}, with the one-variable
principal-value cutoff convention from \eqref{eq:kl-main-cutoff-derivative}.
\end{proof}

\begin{theorem}[KL Doob tail identity]%
\label{thm:poisson-kl-doob-tail}
\label{cor:poisson-kl-doob-bregman-tail}
Let $\nu$ be a probability measure on~$\RR$ with mean~$\bar\gamma$ and finite
centred second moment
\[
\mu_2:=\int_{\RR}(\gamma-\bar\gamma)^2\dd\nu(\gamma)<\infty .
\]
The domain assertions needed to differentiate the entropy are precisely those
of Proposition~\ref{prop:kl-tail-cutoff-inputs}; the tail endpoint
\(H(T)\to0\) is supplied separately by
Lemma~\ref{lem:poisson-second-moment-kl-decay}.  Define
\[
I_\nu(s):=
\frac{1}{\pi}\iint_{\RR^2}
\frac{
g_s(x)\left(
u_s(x)\log\frac{u_s(x)}{u_s(y)}-u_s(x)+u_s(y)
\right)}
{(x-y)^2}\dd x\dd y .
\]
Then \(I_\nu\in L^1_{\mathrm{loc}}(0,\infty)\), and for every \(t>0\),
\begin{equation}\label{eq:relative-entropy-integrated-dissipation}
D_{\mathrm{KL}}(h_t\|g_t)=\int_t^\infty I_\nu(s)\dd s.
\end{equation}
\end{theorem}

\begin{remark}
Proposition~\ref{prop:kl-tail-cutoff-inputs} is the domain input used below:
on every compact time interval it supplies the principal-value entropy
identity, the near/far Bregman majorant, the \(L^1\)-convergent cutoff
identities, and the unlocalized derivative formula.
Thus the identity is a cutoff-closed domain statement, not a formal nonlocal
product-rule computation.
\end{remark}

\begin{proof}
Set
\[
H(t):=D_{\mathrm{KL}}(h_t\|g_t).
\]
For \(s>0\) the integrand defining \(I_\nu(s)\) is nonnegative and Borel.
Proposition~\ref{prop:kl-tail-cutoff-inputs}, applied on an arbitrary compact
interval \([a,b]\subset(0,\infty)\), gives \(H\in AC([a,b])\),
\(I_\nu\in L^1([a,b])\), and
\[
H'(s)=-I_\nu(s)
\]
for a.e. \(s\in[a,b]\).  Hence the fundamental theorem for absolutely
continuous functions gives
\[
H(b)-H(a)=\int_a^b H'(s)\dd s
=-\int_a^b I_\nu(s)\dd s .
\]
Thus \(I_\nu\in L^1_{\mathrm{loc}}(0,\infty)\).  Taking \(a=t\) and \(b=T\)
gives
\[
\int_t^T I_\nu(s)\dd s=H(t)-H(T).
\]
Lemma~\ref{lem:poisson-second-moment-kl-decay} gives \(H(T)\to0\) as
\(T\to\infty\).  Since \(I_\nu\ge0\), monotone convergence yields
\[
\int_t^\infty I_\nu(s)\dd s=H(t),
\]
which is \eqref{eq:relative-entropy-integrated-dissipation}.
\end{proof}

Combining \eqref{eq:relative-entropy-integrated-dissipation} with the
retained-polynomial form in
\S\ref{subsubsec:finite-moment-constants-variants} gives the coefficient tail
formula used later.  This use is downstream only: neither the KL Doob-tail
identity nor the domain closure in Appendix~\ref{app:doob-bregman-domain} is
an input to the proof of the coefficient expansion
Theorem~\ref{thm:poisson-kl-even-orders}.  If \(N\ge2\) and \(\nu\) has finite centred
\((2N-2)\)th absolute moment, then, for every \(2\le M\le N\),
\begin{equation}\label{eq:bregman-tail-coefficient-recovery}
A_{2M}(\nu)
=
\lim_{t\to\infty}t^{2M}
\left(
\int_t^\infty I_\nu(s)\dd s
-\sum_{m=2}^{M-1}A_{2m}(\nu)t^{-2m}
\right),
\end{equation}
with the sum interpreted as zero when \(M=2\).  This is exactly part
\textup{(E)} of Theorem~\ref{thm:poisson-kl-even-orders}, in the
retained-polynomial form of
\S\ref{subsubsec:finite-moment-constants-variants}, rewritten through the KL
tail identity; it is not a second proof of those coefficients.
