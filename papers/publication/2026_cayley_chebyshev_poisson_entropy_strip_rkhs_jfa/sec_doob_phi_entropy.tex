\section{KL Tail Identity for the Poisson Doob Quotient}%
\label{sec:doob-phi-entropy}

Let \(\nu\) be a probability measure on~\(\RR\) with mean \(\bar\gamma\).
For \(t>0\) put
\[
h_t:=P_t*\nu,\qquad
g_t(x):=P_t(x-\bar\gamma),\qquad
u_t:=h_t/g_t .
\]
The Cauchy generator is normalized by
\[
(Lf)(x)=\frac1\pi\operatorname{p.v.}\int_{\RR}
\frac{f(y)-f(x)}{(x-y)^2}\dd y,
\]
so that \(\partial_tP_t=LP_t\).  This section records only the
KL-specialized consequence of the Doob transform.  The range-local
\(C^{1,1}\) convex-entropy closure used to justify the differentiated identity
is in Appendix~\ref{app:doob-bregman-domain}.

\begin{lemma}[Second-moment KL decay to the translated Poisson profile]%
\label{lem:poisson-second-moment-kl-decay}
Let $\nu$ be a probability measure on~$\RR$ with mean $\bar\gamma$ and
finite centred second moment.  Put
\[
h_t:=P_t*\nu,
\qquad
g_t(x):=P_t(x-\bar\gamma).
\]
Then
\[
D_{\mathrm{KL}}(h_t\|g_t)\longrightarrow0
\qquad(t\to\infty).
\]
\end{lemma}

\begin{proof}
Write $\Delta=\gamma-\bar\gamma$ and $x=\bar\gamma+ty$.  Since
$g_t(\bar\gamma+ty)=(\pi t)^{-1}(1+y^2)^{-1}$,
\[
u_t(\bar\gamma+ty):=\frac{h_t(\bar\gamma+ty)}{g_t(\bar\gamma+ty)}
=\int_{\RR}\frac{1+y^2}{1+(y-\Delta/t)^2}\dd\nu(\gamma).
\]
For $A>0$ decompose $\nu=\nu_A+\nu_A^c$ according to
$|\Delta|\le A$ and $|\Delta|>A$.  The elementary kernel bound
\begin{equation}\label{eq:second-moment-kl-kernel-bound}
\frac{1+y^2}{1+(y-\varepsilon)^2}
\le 2(1+\varepsilon^2)
\qquad(y,\varepsilon\in\RR)
\end{equation}
implies
\[
0\le u_{t,A}^c(y):=\int_{|\Delta|>A}
\frac{1+y^2}{1+(y-\Delta/t)^2}\dd\nu(\gamma)
\le 2\int_{|\Delta|>A}\left(1+\frac{\Delta^2}{t^2}\right)\dd\nu(\gamma).
\]
Hence, for $t\ge1$,
\begin{equation}\label{eq:second-moment-kl-tail-sup}
\sup_y u_{t,A}^c(y)
\le 2\nu(|\Delta|>A)+2\int_{|\Delta|>A}\Delta^2\dd\nu.
\end{equation}

The bounded part has total mass $m_A:=\nu(|\Delta|\le A)$ and first centred
moment $b_A:=\int_{|\Delta|\le A}\Delta\dd\nu$.  The exact expansion
\[
\frac{1+y^2}{1+(y-\varepsilon)^2}
=1+\frac{2y}{1+y^2}\varepsilon
+\frac{(3y^2-1)\varepsilon^2-2y\varepsilon^3}
{(1+y^2)(1+(y-\varepsilon)^2)}
\]
shows that, for each fixed $A$,
\begin{equation}\label{eq:bounded-truncated-quotient-limit}
u_{t,A}(y):=\int_{|\Delta|\le A}
\frac{1+y^2}{1+(y-\Delta/t)^2}\dd\nu(\gamma)
=m_A+\frac{2yb_A}{t(1+y^2)}+O_A(t^{-2})
\end{equation}
uniformly in~$y$.  Since
$b_A=-\int_{|\Delta|>A}\Delta\dd\nu$, Cauchy--Schwarz gives
\[
|b_A|\le \left(\int_{|\Delta|>A}\Delta^2\dd\nu\right)^{1/2}.
\]
Combining this with
\eqref{eq:second-moment-kl-tail-sup} and
\eqref{eq:bounded-truncated-quotient-limit} yields
\[
\limsup_{t\to\infty}\|u_t-1\|_{L^\infty(\RR)}
\le 1-m_A+2\nu(|\Delta|>A)+2\int_{|\Delta|>A}\Delta^2\dd\nu.
\]
Letting $A\to\infty$ gives $\|u_t-1\|_\infty\to0$.

For all sufficiently large $t$ we therefore have
$|u_t-1|\le1/2$.  With $\delta_t=u_t-1$ and using
$\int g_t\delta_t=\int(h_t-g_t)=0$,
\[
D_{\mathrm{KL}}(h_t\|g_t)
=\int_{\RR}g_t\bigl((1+\delta_t)\log(1+\delta_t)-\delta_t\bigr)\dd x
\le 2\|\delta_t\|_\infty^2,
\]
because $(1+s)\log(1+s)-s\le2s^2$ for $|s|\le1/2$.  The right-hand side
tends to zero.
\end{proof}

\begin{domainlemma}
Let $\nu$ be a probability measure on~$\RR$ with mean~$\bar\gamma$ and finite
centred second moment
\[
\mu_2:=\int_{\RR}(\gamma-\bar\gamma)^2\dd\nu(\gamma)<\infty .
\]
Let $[a,b]\subset(0,\infty)$, set
\[
h_t=P_t*\nu,\qquad g_t=P_t(\cdot-\bar\gamma),\qquad u_t=h_t/g_t,
\]
and let $\chi\in C_c^\infty(\RR)$ satisfy $\chi\equiv1$ on $[-1,1]$,
$\chi_R(x)=\chi(x/R)$.  Then the following facts hold uniformly for
$t\in[a,b]$.
\begin{enumerate}[label=\textup{(\roman*)},leftmargin=*]
\item The quotient range is the explicit compact interval
\begin{equation}\label{eq:kl-domain-compact-range}
\frac{1}{2(1+\mu_2/a^2)}
\le u_t(x)\le
2+\frac{2\mu_2}{a^2}
\qquad(x\in\RR).
\end{equation}
In particular, with
\[
m_{a,\nu}:=\frac{1}{2(1+\mu_2/a^2)},\qquad
M_{a,\nu}:=2+\frac{2\mu_2}{a^2},
\]
all values of $u_t$ lie in
$J_{a,\nu}:=[m_{a,\nu},M_{a,\nu}]\Subset(0,\infty)$.

\item The spatial derivative has the explicit bound
\begin{equation}\label{eq:kl-domain-derivative-bound}
\|\partial_xu_t\|_\infty
\le
L_{a,\nu}:=\frac{5}{a}\left(1+\frac{\mu_2}{a^2}\right).
\end{equation}
Consequently $u_t$ is uniformly Lipschitz on
$\RR\times[a,b]$.

\item For $\Phi(s)=s\log s$ and
\[
\Lambda(\alpha,\beta):=\alpha\log\frac{\alpha}{\beta}-\alpha+\beta,
\]
there is a constant $C_{a,\nu}<\infty$ such that
\begin{equation}\label{eq:kl-domain-bregman-majorant}
0\le \Lambda(u_t(x),u_t(y))
\le C_{a,\nu}\min\{|x-y|^2,1\}
\qquad(x,y\in\RR).
\end{equation}
Hence
\begin{equation}\label{eq:kl-domain-absolute-bregman}
\sup_{t\in[a,b]}
\iint_{\RR^2}
\frac{g_t(x)\Lambda(u_t(x),u_t(y))}{(x-y)^2}\dd x\dd y<\infty .
\end{equation}

\item The map
\[
H(t):=D_{\mathrm{KL}}(h_t\|g_t)
=\int_{\RR}g_t(x)u_t(x)\log u_t(x)\dd x
\]
is absolutely continuous on $[a,b]$.  For a.e. $t\in[a,b]$ its derivative is
the one-variable cutoff limit
\begin{align}
-H'(t)
&=
\lim_{R\to\infty}
\left[
-\int_{\RR}\chi_R(x)\bigl(1+\log u_t(x)\bigr)Lh_t(x)\dd x
+\int_{\RR}\chi_R(x)u_t(x)Lg_t(x)\dd x
\right],
\label{eq:kl-main-cutoff-derivative}
\end{align}
and this same limit is exactly the two-variable cutoff limit
\begin{align}
\lim_{R\to\infty}
\frac{1}{\pi}\iint_{\RR^2}
\chi_R(x)\chi_R(y)
\frac{g_t(x)\Lambda(u_t(x),u_t(y))}{(x-y)^2}\dd x\dd y .
\label{eq:kl-main-cutoff-bregman-limit}
\end{align}
By \eqref{eq:kl-domain-absolute-bregman}, the cutoff can then be removed in
the last display.
\end{enumerate}
\end{domainlemma}

\begin{proof}
Write $\Delta=\gamma-\bar\gamma$, $y=(x-\bar\gamma)/t$, and
\[
R_\varepsilon(y):=\frac{1+y^2}{1+(y-\varepsilon)^2}.
\]
Then
\[
u_t(x)=\int_{\RR}R_{\Delta/t}(y)\dd\nu(\gamma).
\]
The inequalities
\[
1+y^2\le2\bigl(1+(y-\varepsilon)^2\bigr)+2\varepsilon^2
\]
and
\[
1+(y-\varepsilon)^2
\le2(1+\varepsilon^2)(1+y^2)
\]
give
\[
R_\varepsilon(y)\le2+2\varepsilon^2,
\qquad
R_\varepsilon(y)\ge\frac{1}{2(1+\varepsilon^2)}.
\]
After integration, the upper bound in
\eqref{eq:kl-domain-compact-range} follows from $t\ge a$.  The lower bound
uses Jensen's inequality for the convex function $s\mapsto(1+s)^{-1}$:
\[
\int_{\RR}\frac{1}{1+(\Delta/t)^2}\dd\nu(\gamma)
\ge
\frac{1}{1+\mu_2/t^2}.
\]

For the derivative bound, differentiating the quotient kernel gives
\[
\partial_yR_\varepsilon(y)
=
\frac{2y}{1+(y-\varepsilon)^2}
-\frac{2(1+y^2)(y-\varepsilon)}
       {(1+(y-\varepsilon)^2)^2}.
\]
Putting $w=y-\varepsilon$ and using $|y|\le |w|+|\varepsilon|$ and
$1+y^2\le2(1+w^2)+2\varepsilon^2$ yields
\[
|\partial_yR_\varepsilon(y)|
\le 5(1+\varepsilon^2)
\qquad(y,\varepsilon\in\RR).
\]
Since $\partial_x=t^{-1}\partial_y$ and $t\ge a$, integration against~$\nu$
proves \eqref{eq:kl-domain-derivative-bound}.

On the compact interval $J_{a,\nu}$ the function
$s\mapsto1+\log s$ is Lipschitz, so the integral form of the Bregman remainder
gives
\[
0\le\Lambda(\alpha,\beta)\le C_{a,\nu}'|\alpha-\beta|^2
\qquad(\alpha,\beta\in J_{a,\nu}).
\]
Combining this with \eqref{eq:kl-domain-derivative-bound} and with the
trivial bound $|u_t(x)-u_t(y)|\le M_{a,\nu}-m_{a,\nu}$ proves
\eqref{eq:kl-domain-bregman-majorant}.  Dividing by $(x-y)^2$ leaves the
integrable kernel $\min\{1,|x-y|^{-2}\}$ in the $y$ variable, and
$\int g_t=1$ proves \eqref{eq:kl-domain-absolute-bregman}.

It remains only to identify the derivative as a domain statement.  Lemma
\ref{lem:cauchy-doob-cutoff-calculus} gives
$\partial_th_t=Lh_t$, $\partial_tg_t=Lg_t$, uniform $L^1$ bounds for these
time derivatives, and the cutoff bilinear identity for bounded Lipschitz
tests.  Since $u_t$ stays in $J_{a,\nu}$, the $L^1$ chain rule for
$g_t\Phi(h_t/g_t)$ gives $H\in AC([a,b])$ and, for a.e. $t\in[a,b]$,
\[
H'(t)=
\int_{\RR}\bigl(1+\log u_t\bigr)Lh_t\dd x
-\int_{\RR}u_tLg_t\dd x,
\]
with the two pairings interpreted through the cutoff limit in
\eqref{eq:kl-main-cutoff-derivative}.  Applying
\eqref{eq:cutoff-cauchy-bilinear} first to
$p=h_t=g_tu_t$, $q=1+\log u_t$, and then to $p=g_t$, $q=-u_t$, gives the
same cutoff limit as
\[
\frac{1}{2\pi}\lim_{R\to\infty}\iint_{\RR^2}\chi_R(x)\chi_R(y)
\frac{g_t(x)\Lambda(u_t(x),u_t(y))
      +g_t(y)\Lambda(u_t(y),u_t(x))}
{(x-y)^2}\dd x\dd y .
\]
Lemma~\ref{lem:doob-bregman-algebra} is the algebraic identity behind this
last display.  The two summands are equal for every finite $R$ after
interchanging $x$ and~$y$, so the symmetric cutoff is precisely
\eqref{eq:kl-main-cutoff-bregman-limit}.  The absolute convergence already
proved in \eqref{eq:kl-domain-absolute-bregman} justifies removing the cutoff.
\end{proof}

\begin{theorem}[KL Doob tail identity with cutoff domain closure]%
\label{thm:poisson-kl-doob-tail}
\label{cor:poisson-kl-doob-bregman-tail}
Let $\nu$ be a probability measure on~$\RR$ with mean~$\bar\gamma$ and finite
centred second moment
\[
\mu_2:=\int_{\RR}(\gamma-\bar\gamma)^2\dd\nu(\gamma)<\infty .
\]
Let
\[
\Phi(s):=s\log s,\qquad
D_\Phi(a,b):=\Phi(a)-\Phi(b)-\Phi'(b)(a-b)
=a\log\frac{a}{b}-a+b .
\]
Then the quotient satisfies the Doob equation
\begin{equation}\label{eq:quotient-flow-equation-main-kl}
\partial_tu_t=L_tu_t,
\qquad
(L_tf)(x):=\frac1{\pi g_t(x)}
\operatorname{p.v.}\int_{\RR}
\frac{g_t(y)(f(y)-f(x))}{(x-y)^2}\dd y .
\end{equation}
The principal values are understood in the Cauchy-generator normalization
fixed above.
For every compact time interval $[a,b]\subset(0,\infty)$ there are
constants
\[
0<m_{a,\nu}\le M_{a,\nu}<\infty,\qquad L_{a,b,\nu}<\infty ,
\]
such that, for all $t\in[a,b]$,
\begin{equation}\label{eq:kl-main-compact-quotient-range}
0<m_{a,\nu}\le u_t(x)\le M_{a,\nu}<\infty,
\qquad
\|\partial_xu_t\|_\infty\le L_{a,b,\nu}.
\end{equation}
One may take
\[
m_{a,\nu}=\frac{1}{2(1+\mu_2/a^2)},\qquad
M_{a,\nu}=2+\frac{2\mu_2}{a^2},\qquad
L_{a,b,\nu}=\frac{5}{a}\left(1+\frac{\mu_2}{a^2}\right).
\]
Consequently $\Phi(u_t)$, $\Phi'(u_t)$, and
$D_\Phi(u_t(x),u_t(y))$ are evaluated only on the compact subinterval
$[m_{a,\nu},M_{a,\nu}]\Subset(0,\infty)$ for $t\in[a,b]$.
Moreover
\[
t\longmapsto D_{\mathrm{KL}}(h_t\|g_t)
=\int_{\RR}g_t(x)u_t(x)\log u_t(x)\dd x
\]
is absolutely continuous on $[a,b]$.
For a.e. $t>0$, the derivative of
$D_{\mathrm{KL}}(h_t\|g_t)$ is obtained as the cutoff limit
\eqref{eq:kl-main-cutoff-derivative}; the exact Bregman cutoff limit is
\eqref{eq:kl-main-cutoff-bregman-limit}.  Equivalently, for the same a.e.
values of \(t\),
\begin{equation}\label{eq:relative-entropy-dissipation-rate}
-\frac{\dd}{\dd t}D_{\mathrm{KL}}(h_t\|g_t)
=
\frac{1}{\pi}\iint_{\RR^2}
\frac{g_t(x)\Lambda(u_t(x),u_t(y))}{(x-y)^2}\dd x\dd y,
\end{equation}
where
\[
\Lambda(a,b):=a\log\frac{a}{b}-a+b=D_\Phi(a,b).
\]
The double integral in
\eqref{eq:relative-entropy-dissipation-rate} is the cutoff limit
\eqref{eq:kl-main-cutoff-bregman-limit}, established by the domain-closure
calculus in Appendix~\ref{app:doob-bregman-domain}; the identity is not a
formal product-rule computation on a nonlocal operator.
Consequently, if
\[
I_\nu(s):=
\frac{1}{\pi}\iint_{\RR^2}
\frac{g_s(x)\Lambda(u_s(x),u_s(y))}{(x-y)^2}\dd x\dd y,
\]
then, for every $t>0$,
\begin{equation}\label{eq:relative-entropy-integrated-dissipation}
D_{\mathrm{KL}}(h_t\|g_t)=\int_t^\infty I_\nu(s)\dd s.
\end{equation}
\end{theorem}

\begin{proof}
The quotient equation is Theorem~\ref{thm:poisson-doob-phi-entropy}
specialized to the Poisson pair $(h_t,g_t)$; its kernel form is
\eqref{eq:doob-generator-kernel}.  The domain lemma gives
\eqref{eq:kl-main-compact-quotient-range}, absolute continuity on every
compact time interval, the derivative cutoff formula
\eqref{eq:kl-main-cutoff-derivative}, and the Bregman cutoff limit
\eqref{eq:kl-main-cutoff-bregman-limit}.  Since the interval $[a,b]$ in the
lemma is arbitrary, these a.e. identities hold locally on $(0,\infty)$, and
\eqref{eq:kl-domain-absolute-bregman} removes the cutoff in
\eqref{eq:relative-entropy-dissipation-rate}.

Finally
Lemma~\ref{lem:poisson-second-moment-kl-decay} gives
$D_{\mathrm{KL}}(h_T\|g_T)\to0$.  Integrating the a.e. identity over $[t,T]$
and letting $T\to\infty$ gives
\eqref{eq:relative-entropy-integrated-dissipation}.
\end{proof}

Combining \eqref{eq:relative-entropy-integrated-dissipation} with
Theorem~\ref{thm:finite-moment-entropy-remainder} gives the coefficient tail
formula used later.  If $N\ge2$, $0<\eta\le1$, and \(\nu\) has finite centred
\((2N+\eta)\)th absolute moment, then, for every \(2\le M\le N\),
\begin{equation}\label{eq:bregman-tail-coefficient-recovery}
A_{2M}(\nu)
=
\lim_{t\to\infty}t^{2M}
\left(
\int_t^\infty I_\nu(s)\dd s
-\sum_{m=2}^{M-1}A_{2m}(\nu)t^{-2m}
\right),
\end{equation}
with the sum interpreted as zero when \(M=2\).  This is exactly the expansion
of Theorem~\ref{thm:finite-moment-entropy-remainder} rewritten through the KL
tail identity.
