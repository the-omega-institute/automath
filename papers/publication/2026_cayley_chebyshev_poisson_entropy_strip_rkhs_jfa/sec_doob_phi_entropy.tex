\section{Non-Autonomous Doob Transform and Entropy Dissipation}%
\label{sec:doob-phi-entropy}

The Poisson comparison quotient
\[
u_t(x):=\frac{h_t(x)}{g_t(x)}
\qquad(t>0,\ x\in\RR)
\]
admits a closed evolution equation with respect to the Cauchy generator.
This identifies the entropy decay as a nonlocal Bregman dissipation.  Throughout
this section the Cauchy generator is normalized by
\[
(Lf)(x)=\frac1\pi\operatorname{p.v.}\int_{\RR}
\frac{f(y)-f(x)}{(x-y)^2}\dd y,
\]
so that \(\partial_tP_t=LP_t\).  All principal-value identities below are
proved first with the displayed cutoff calculus and then passed to the stated
domains.

The dissipation proof is split into its algebraic and analytic components.
Lemma~\ref{lem:doob-bregman-algebra} is purely algebraic and records the
two-point Bregman rearrangement used in the entropy derivative.  Only after
that identity is isolated do we address domains:
Lemma~\ref{lem:cauchy-doob-cutoff-calculus} proves the generator and bilinear
identities with spatial cutoffs and supplies the integrable majorants needed as
the cutoff radius tends to infinity,
Proposition~\ref{prop:smooth-compact-doob-bregman} proves the identity in the
smooth compactly supported quotient class, and
Proposition~\ref{prop:doob-bregman-domain-closure} is the separate
domain-closure step for the Poisson quotient and the range-local $C^{1,1}$
entropy class.  The KL tail identity is then stated on its own in
Corollary~\ref{cor:poisson-kl-doob-bregman-tail}.

This section is auxiliary to the finite-moment entropy theorem.  Its role is
to apply the Doob transform to the same Poisson comparison quotient used in
Section~\ref{sec:entropy-asymptotics}: for the pair $(h_t,g_t)$, the entropy
derivative is expressed by a symmetric Bregman form whose tail coefficients
match the already defined Cayley--Chebyshev coefficients.  The
principal-value generator, cutoff integration by parts, and convex-entropy
approximation are standard analytic tools; they are included only to make the
domain passage for the nonlocal identity explicit.  Thus this section supplies
a dissipation interpretation of the coefficient calculus, not a fourth main
theorem package and not an independent entropy-asymptotic route.

\begin{lemma}[Two-point Bregman algebra for the Doob derivative]%
\label{lem:doob-bregman-algebra}
Let $g_x,g_y>0$ and $u_x,u_y>0$.  Let
$\Phi\in C^1((0,\infty))$, set
\[
G(u):=\Phi(u)-u\Phi'(u),
\qquad
D_\Phi(a,b):=\Phi(a)-\Phi(b)-\Phi'(b)(a-b),
\]
and abbreviate $\Phi'(u_x)-\Phi'(u_y)$ by the literal difference of the two
one-sided entries.  Then
\begin{equation}\label{eq:doob-bregman-algebra}
\begin{split}
&(g_xu_x-g_yu_y)\bigl(\Phi'(u_x)-\Phi'(u_y)\bigr)
 +(g_x-g_y)\bigl(G(u_x)-G(u_y)\bigr)
\\
&\qquad
=g_xD_\Phi(u_x,u_y)+g_yD_\Phi(u_y,u_x).
\end{split}
\end{equation}
\end{lemma}

\begin{proof}
Collect the $g_x$ and $g_y$ coefficients separately.  The coefficient of
$g_x$ on the left of \eqref{eq:doob-bregman-algebra} is
\[
u_x\bigl(\Phi'(u_x)-\Phi'(u_y)\bigr)+G(u_x)-G(u_y).
\]
Substituting $G(u)=\Phi(u)-u\Phi'(u)$ gives
\[
\Phi(u_x)-\Phi(u_y)-\Phi'(u_y)(u_x-u_y)
=D_\Phi(u_x,u_y).
\]
The coefficient of $g_y$ is
\[
-u_y\bigl(\Phi'(u_x)-\Phi'(u_y)\bigr)-G(u_x)+G(u_y)
=\Phi(u_y)-\Phi(u_x)-\Phi'(u_x)(u_y-u_x)
=D_\Phi(u_y,u_x).
\]
This proves the identity.  No analytic domain assertion enters this step.
\end{proof}

\begin{lemma}[Cutoff justification for the Cauchy Doob calculus]%
\label{lem:cauchy-doob-cutoff-calculus}
Let $[a,b]\subset(0,\infty)$ and let $\nu$ have finite centred second
moment.  For $t\in[a,b]$ put $h_t=P_t*\nu$, $g_t=P_t(\cdot-
\bar\gamma)$, and $u_t=h_t/g_t$.  Then the following assertions hold
uniformly for $t\in[a,b]$.
\begin{enumerate}[label=(\roman*)]
\item
      The functions $u_t$ and $\partial_xu_t$ are bounded.  Moreover
      $h_t,g_t,\partial_xh_t,\partial_xg_t,\partial_th_t,\partial_tg_t$
      belong to $L^1(\RR)$, and
      \[
      \partial_th_t=Lh_t,
      \qquad
      \partial_tg_t=Lg_t
      \]
      pointwise, where $L$ is the principal-value Cauchy generator.
\item
      If $q$ is a bounded Lipschitz function and
      $p\in\{h_t,g_t\}$, then the bilinear Cauchy form
      \[
      \mathcal E(p,q)
      :=\frac{1}{2\pi}\iint_{\RR^2}
      \frac{(p(x)-p(y))(q(x)-q(y))}{(x-y)^2}\dd x\dd y
      \]
      is absolutely convergent and
      \begin{equation}\label{eq:cutoff-cauchy-bilinear}
      \int_{\RR}q(x)Lp(x)\dd x=-\mathcal E(p,q).
      \end{equation}
      The identity is obtained as the limit of the compactly supported
      identities with $q$ replaced by $\chi_Rq$, where
      $\chi_R(x)=\chi(x/R)$, $\chi\in C_c^\infty(\RR)$,
      $\chi\equiv1$ on $[-1,1]$.  This fixes the domain convention used in
      the theorem: the left side is the $L^1$ pairing with the
      principal-value generator, and the right side is the absolutely
      convergent cutoff limit.
\item
      If $f$ is bounded $C^2$ with bounded first and second derivatives,
      in particular if $f=u_t$, then, with the same principal-value
      normalization,
      \begin{equation}\label{eq:cutoff-doob-product-rule}
      L(g_tf)(x)-f(x)Lg_t(x)
      =\frac1\pi\operatorname{p.v.}\int_{\RR}
      \frac{g_t(y)(f(y)-f(x))}{(x-y)^2}\dd y .
      \end{equation}
\end{enumerate}
\end{lemma}

\begin{proof}
Write $x=\bar\gamma+ty$ and $\varepsilon=(\gamma-\bar\gamma)/t$.  The
quotient kernel
\[
R_\varepsilon(y)=\frac{1+y^2}{1+(y-\varepsilon)^2}
\]
satisfies, for $j=0,1,2$, bounds of the form
\[
\sup_{y\in\RR}|\partial_y^jR_\varepsilon(y)|\le C_j(1+\varepsilon^2).
\]
Indeed, after differentiating, the numerator has degree at most two in
$(y-\varepsilon)$ and coefficients bounded by a constant multiple of
$1+\varepsilon^2$, while the denominator has degree at least the numerator
degree.  Integrating these estimates against~$\nu$ gives the asserted
uniform bounds for $u_t$ and $\partial_xu_t$ on $[a,b]$.

The $L^1$ assertions follow from convolution and scaling:
\[
\|\partial_xP_t\|_1\le C_a,
\qquad
\|\partial_tP_t\|_1\le C_a
\qquad(a\le t\le b).
\]
The identity $\partial_tP_t=LP_t$ is the one-dimensional Cauchy-semigroup
generator identity in the principal-value normalization used here
\cite[Definition~2.2 and Proposition~2.4]{Kwasnicki2017}.  Convolution with
the finite measure~$\nu$ and dominated convergence give
$\partial_th_t=Lh_t$; the same argument with the point mass at
$\bar\gamma$ gives $\partial_tg_t=Lg_t$.

For the bilinear identity, first take $q\in C_c^\infty(\RR)$.  The usual
principal-value formula for the Cauchy generator and symmetrization give
\eqref{eq:cutoff-cauchy-bilinear}.  The same compactly supported identity
holds for compactly supported Lipschitz tests by mollifying them and using the
estimates below.  For bounded Lipschitz $q$, replace it by
$q_R=\chi_Rq$, $R\ge1$.  Then
\[
\|q_R\|_\infty\le \|q\|_\infty,
\qquad
\|q_R\|_{\operatorname{Lip}}
\le \|q\|_{\operatorname{Lip}}+\|\chi'\|_\infty\|q\|_\infty,
\]
so the cutoff family has bounds independent of~$R$.  Since
$p,\partial_xp\in L^1(\RR)$ and $q$ is bounded and Lipschitz, the near
diagonal part satisfies, uniformly in~$R$,
\[
\iint_{|x-y|<1}
\frac{|p(x)-p(y)|\,|q_R(x)-q_R(y)|}{|x-y|^2}\dd x\dd y
\le 2\|q_R\|_{\operatorname{Lip}}\|\partial_xp\|_1,
\]
because
$p(x)-p(y)=\int_0^1p'(y+s(x-y))(x-y)\dd s$.  The far part satisfies,
again uniformly in~$R$,
\[
\iint_{|x-y|\ge1}
\frac{|p(x)-p(y)|\,|q_R(x)-q_R(y)|}{|x-y|^2}\dd x\dd y
\le 8\|q\|_\infty\|p\|_1,
\]
since
$\int_{|x-y|\ge1}|x-y|^{-2}\dd y=2$ and the same estimate holds after
interchanging $x$ and~$y$.  It remains to show that the cutoff error tends to
zero.  Put $e_R:=q_R-q$.  Then $e_R=0$ on $[-R,R]$,
$\|e_R\|_\infty\le2\|q\|_\infty$, and
$\|e_R\|_{\operatorname{Lip}}\le
\|q\|_{\operatorname{Lip}}+\|\chi'\|_\infty\|q\|_\infty$.  For the near
diagonal contribution, $e_R(x)-e_R(y)\ne0$ and $|x-y|<1$ imply that the line
segment between $x$ and~$y$ lies in $\{|z|>R-1\}$.  Therefore
\[
\iint_{|x-y|<1}
\frac{|p(x)-p(y)|\,|e_R(x)-e_R(y)|}{|x-y|^2}\dd x\dd y
\le C_q\int_{|z|>R-1}|p'(z)|\dd z\longrightarrow0.
\]
For the far contribution, using $|e_R(z)|\le2\|q\|_\infty\mathbf 1_{|z|>R}$
gives a sum of four terms.  The two terms in which the same variable carries
both $p$ and $e_R$ are bounded by
$4\|q\|_\infty\int_{|z|>R}|p(z)|\dd z$.  The cross terms are bounded by
\[
2\|q\|_\infty\int_{\RR}|p(x)|
\int_{\substack{|y|>R\\ |x-y|\ge1}}\frac{\dd y}{|x-y|^2}\dd x,
\]
and the analogous expression with $x$ and~$y$ interchanged; the inner integral
is bounded by~$2$ and tends to zero for each fixed outer variable, so dominated
convergence and $p\in L^1$ make the cross terms vanish.  Hence the bilinear
forms with $q_R$ converge to the form with~$q$.  On the one-variable side,
\[
|e_R(x)Lp(x)|\le 2\|q\|_\infty |Lp(x)|,
\qquad Lp=\partial_tp\in L^1(\RR),
\]
and $e_RLp\to0$ pointwise.  Hence the cutoff identities pass to the limit and
prove \eqref{eq:cutoff-cauchy-bilinear} for bounded Lipschitz~$q$.

Finally apply the principal-value formula to $g_tf$ and to $g_t$.  In the
truncated integrals over $|x-y|>\rho$ the difference is exactly
\[
\frac1\pi\int_{|x-y|>\rho}
\frac{g_t(y)(f(y)-f(x))}{(x-y)^2}\dd y .
\]
The integrand has a finite principal-value limit because near $y=x$ the
linear part contributes an odd kernel and the remaining term is bounded,
while away from $x$ it is integrable.  Letting $\rho\downarrow0$ gives
\eqref{eq:cutoff-doob-product-rule}.
\end{proof}


\begin{lemma}[Compact quotient range and Bregman integrability]%
\label{lem:doob-range-bregman-integrability}
Let $[a,b]\subset(0,\infty)$ and let $\nu$ be a probability measure
on~$\RR$ with mean~$\bar\gamma$ and finite centred second moment
\[
\mu_2:=\int_{\RR}(\gamma-\bar\gamma)^2\dd\nu(\gamma).
\]
Put $h_t=P_t*\nu$, $g_t=P_t(\cdot-\bar\gamma)$, and
$u_t=h_t/g_t$.  Then, for every $t\in[a,b]$ and every $x\in\RR$,
\begin{equation}\label{eq:doob-compact-range-bound}
\frac{1}{2(1+\mu_2/a^2)}
\le u_t(x)\le
2+\frac{2\mu_2}{a^2}.
\end{equation}
Moreover $u_t$ is uniformly Lipschitz in $x$ on
$\RR\times[a,b]$.  Consequently, if
$\Phi:(0,\infty)\to\RR$ is convex, of class $C^1$, and $\Phi'$ is
Lipschitz on compact subintervals of $(0,\infty)$, then there is a
constant $C_{\Phi,a,b}<\infty$ such that, for all $t\in[a,b]$ and
$x,y\in\RR$,
\begin{equation}\label{eq:doob-bregman-pointwise-bound}
0\le D_\Phi(u_t(x),u_t(y))
\le C_{\Phi,a,b}\min\{|x-y|^2,1\}.
\end{equation}
In particular
\begin{equation}\label{eq:doob-bregman-integrability-bound}
\sup_{t\in[a,b]}
\iint_{\RR^2}
\frac{
g_t(x)D_\Phi(u_t(x),u_t(y))
+g_t(y)D_\Phi(u_t(y),u_t(x))}
{(x-y)^2}\dd x\dd y<\infty,
\end{equation}
and $\int_{\RR}g_t(x)|\Phi(u_t(x))|\dd x<\infty$ for every
$t\in[a,b]$.
\end{lemma}

\begin{proof}
Write $\Delta=\gamma-\bar\gamma$ and $y=(x-\bar\gamma)/t$.  Since
\[
\frac{P_t(x-\gamma)}{P_t(x-\bar\gamma)}
=R_{\Delta/t}(y),
\qquad
R_\varepsilon(y)=\frac{1+y^2}{1+(y-\varepsilon)^2},
\]
we have $u_t(x)=\int R_{\Delta/t}(y)\dd\nu(\gamma)$.  The elementary
inequality
\[
1+y^2\le 2\bigl(1+(y-\varepsilon)^2\bigr)+2\varepsilon^2
\]
gives $R_\varepsilon(y)\le2+2\varepsilon^2$, hence
\[
u_t(x)\le 2+2\mu_2/t^2\le 2+2\mu_2/a^2 .
\]
Conversely,
\[
1+(y-\varepsilon)^2
\le2(1+y^2)+2\varepsilon^2
\le2(1+\varepsilon^2)(1+y^2),
\]
so $R_\varepsilon(y)\ge[2(1+\varepsilon^2)]^{-1}$.  Therefore
\[
u_t(x)
\ge \frac12\int_{\RR}\frac{1}{1+(\Delta/t)^2}\dd\nu(\gamma)
\ge \frac{1}{2(1+\mu_2/t^2)}
\ge \frac{1}{2(1+\mu_2/a^2)},
\]
where the middle inequality is Jensen's inequality for the convex
function $s\mapsto(1+s)^{-1}$ on $[0,\infty)$.  This proves
\eqref{eq:doob-compact-range-bound}.

Lemma~\ref{lem:cauchy-doob-cutoff-calculus} gives a constant
$L_{a,b}$ such that $|\partial_xu_t(x)|\le L_{a,b}$ for
$t\in[a,b]$.  Let
\[
m_{a,b}:=\frac{1}{2(1+\mu_2/a^2)},
\qquad
M_{a,b}:=2+\frac{2\mu_2}{a^2}.
\]
If $\Phi'$ has Lipschitz constant $L_\Phi$ on
$[m_{a,b},M_{a,b}]$, then the integral form of the Bregman remainder gives
\[
0\le D_\Phi(\alpha,\beta)\le \frac{L_\Phi}{2}|\alpha-\beta|^2
\qquad(m_{a,b}\le\alpha,\beta\le M_{a,b}).
\]
Combining this with the Lipschitz bound for $u_t$ and with the trivial
bound $|u_t(x)-u_t(y)|\le M_{a,b}-m_{a,b}$ proves
\eqref{eq:doob-bregman-pointwise-bound}.

Dividing \eqref{eq:doob-bregman-pointwise-bound} by $(x-y)^2$ gives the
majorant $C_{\Phi,a,b}\min\{1,|x-y|^{-2}\}$.  Since
\[
\int_{\RR}\min\{1,|x-y|^{-2}\}\dd y<\infty
\]
with a value independent of~$x$, and since $\int g_t=1$, the first term
in \eqref{eq:doob-bregman-integrability-bound} is bounded uniformly in
$t\in[a,b]$; the second term is identical after exchanging $x$ and~$y$.
Finally, \eqref{eq:doob-compact-range-bound} puts $u_t$ in a compact
subinterval of $(0,\infty)$, so $\Phi$ is bounded on the range and
$\int g_t|\Phi(u_t)|\le\sup_{[m_{a,b},M_{a,b}]}|\Phi|$.
\end{proof}

\begin{lemma}[Compact-range approximation for range-local $C^{1,1}$ entropies]%
\label{lem:compact-range-phi-approximation}
Let $[a,b]\subset(0,\infty)$, and let $h_t$, $g_t$, and $u_t=h_t/g_t$ be as
in Lemma~\ref{lem:doob-range-bregman-integrability}.  Let
$J=[m_{a,b},M_{a,b}]$ be the compact interval from its proof.  If
$\Phi:(0,\infty)\to\RR$ is convex, $C^1$, and $\Phi'$ is Lipschitz on compact
subintervals of $(0,\infty)$, then there are convex functions
$\Phi_\varepsilon\in C^\infty((0,\infty))$, defined for all sufficiently small
$\varepsilon>0$, such that
\[
\sup_{s\in J}|\Phi_\varepsilon(s)-\Phi(s)|
+\sup_{s\in J}|\Phi_\varepsilon'(s)-\Phi'(s)|\longrightarrow0,
\]
the functions $\Phi_\varepsilon'$ have a common Lipschitz constant on~$J$, and
\[
\sup_{\alpha,\beta\in J}
|D_{\Phi_\varepsilon}(\alpha,\beta)-D_\Phi(\alpha,\beta)|\longrightarrow0.
\]
Consequently the entropy functionals and the Bregman dissipation integrands in
Proposition~\ref{prop:doob-bregman-domain-closure} may be obtained as limits
from the same formulas for the smooth convex functions $\Phi_\varepsilon$,
with the same compact-range majorants on every time interval $[a,b]$.
\end{lemma}

\begin{proof}
Choose an open interval $J_*$ with compact closure in $(0,\infty)$ and
$J\subset J_*$.  Since $\Phi$ is convex and $C^1$, $\Phi'$ is nondecreasing;
by hypothesis it has some Lipschitz constant $L_*$ on~$J_*$.  Extend
$\Phi'|_{J_*}$ to a globally defined nondecreasing Lipschitz function
$\Psi$ on~$\RR$ with Lipschitz constant $L_*$, for instance by continuing it
linearly with slope~$0$ outside $J_*$.  Let $\rho_\varepsilon$ be a standard
nonnegative even mollifier and set
$\Psi_\varepsilon=\rho_\varepsilon*\Psi$.  For $\varepsilon$ smaller than the
distance from $J$ to $\RR\setminus J_*$, the functions
$\Psi_\varepsilon$ are smooth, nondecreasing, have Lipschitz constant at most
$L_*$ on~$J$, and converge uniformly to $\Phi'$ on~$J$.
Define
\[
\Phi_\varepsilon(s):=\Phi(1)+\int_1^s\Psi_\varepsilon(r)\dd r,
\]
after any smooth convex extension away from $J_*$ if necessary.  Then
$\Phi_\varepsilon$ is smooth and convex on $(0,\infty)$, and the displayed
uniform convergence of $\Phi_\varepsilon$ and $\Phi_\varepsilon'$ on~$J$
follows by integrating the uniform convergence of the derivatives on the
compact interval.  The Bregman convergence is immediate from
\[
|D_{\Phi_\varepsilon}(\alpha,\beta)-D_\Phi(\alpha,\beta)|
\le 2\sup_J|\Phi_\varepsilon-\Phi|
 +(M_{a,b}-m_{a,b})\sup_J|\Phi_\varepsilon'-\Phi'|.
\]
Lemma~\ref{lem:doob-range-bregman-integrability} gives the common bound
$0\le D_{\Phi_\varepsilon}(u_t(x),u_t(y))
\le C\min\{|x-y|^2,1\}$ with $C$ independent of~$\varepsilon$, and it also
puts $u_t$ in~$J$.  Since $\int g_t=1$, dominated convergence applies to the
one-variable entropy integrals and to the two-variable Bregman dissipation
integrals.  This proves the limiting assertion.
\end{proof}

\begin{proposition}[Smooth compactly supported Doob--Bregman identity]%
\label{prop:smooth-compact-doob-bregman}
Let $t\mapsto g_t$ be a positive $C^1$ curve of smooth densities on an open
time interval, and let $t\mapsto u_t$ be a positive $C^1$ curve of smooth
functions such that $u_t-1\in C_c^\infty(\RR)$ for each $t$, with one compact
support and one compact quotient range on compact time intervals.  Put
$h_t=g_tu_t$ and assume
\[
\partial_t h_t=Lh_t,\qquad \partial_tg_t=Lg_t,
\]
with all integrations below absolutely convergent; for instance this holds
after replacing the noncompact tests in Lemma~\ref{lem:cauchy-doob-cutoff-calculus}
by fixed spatial cutoffs.  Let $\Phi\in C^2((0,\infty))$ with $\Phi(1)=0$.
Then
\[
\mathcal H_\Phi(t):=\int_{\RR}g_t(x)\Phi(u_t(x))\dd x
\]
is $C^1$, and
\begin{equation}\label{eq:smooth-compact-phi-entropy-dissipation}
-\frac{\dd}{\dd t}\mathcal H_\Phi(t)
=
\frac{1}{2\pi}\iint_{\RR^2}
\frac{
g_t(x)D_\Phi(u_t(x),u_t(y))
+g_t(y)D_\Phi(u_t(y),u_t(x))}
{(x-y)^2}\dd x\dd y .
\end{equation}
\end{proposition}

\begin{proof}
Because $u_t-1$ has compact support and $\Phi(1)=0$, the entropy integrand is
supported where $u_t\ne1$.  Differentiating under the integral and using
$h_t=g_tu_t$ gives
\[
\frac{\dd}{\dd t}\mathcal H_\Phi(t)
=
\int_{\RR}\Phi'(u_t)\,Lh_t\,\dd x
+
\int_{\RR}\bigl(\Phi(u_t)-u_t\Phi'(u_t)\bigr)Lg_t\,\dd x .
\]
The functions $\Phi'(u_t)-\Phi'(1)$ and
$Q_\Phi(u_t)-Q_\Phi(1)$ are smooth and compactly supported, and constants do
not change the bilinear form.  The constant pairings with $Lh_t$ and $Lg_t$
vanish because the generator preserves total mass.  Applying the compactly
supported Cauchy bilinear identity
\[
\int_{\RR}qLp\,\dd x
=-\frac{1}{2\pi}\iint_{\RR^2}
\frac{(p(x)-p(y))(q(x)-q(y))}{(x-y)^2}\dd x\dd y
\]
first with $p=h_t=g_tu_t$, $q=\Phi'(u_t)$, and then with
$p=g_t$, $q=Q_\Phi(u_t)$, where
$Q_\Phi(s)=\Phi(s)-s\Phi'(s)$.  Thus
\[
\frac{\dd}{\dd t}\mathcal H_\Phi(t)
=-\frac{1}{2\pi}\iint_{\RR^2}\frac{E_t(x,y)}{(x-y)^2}\dd x\dd y,
\]
where, with $u_x=u_t(x)$, $u_y=u_t(y)$, $g_x=g_t(x)$, and $g_y=g_t(y)$,
\[
E_t(x,y)
=(g_xu_x-g_yu_y)\bigl(\Phi'(u_x)-\Phi'(u_y)\bigr)
 +(g_x-g_y)\bigl(Q_\Phi(u_x)-Q_\Phi(u_y)\bigr).
\]
Lemma~\ref{lem:doob-bregman-algebra} rewrites this exactly as
\[
E_t(x,y)=g_t(x)D_\Phi(u_t(x),u_t(y))
+g_t(y)D_\Phi(u_t(y),u_t(x)).
\]
Substitution gives \eqref{eq:smooth-compact-phi-entropy-dissipation}.
\end{proof}

\begin{lemma}[Doob--Bregman domain bounds]%
\label{lem:doob-bregman-domain-bounds}
Let $[a,b]\subset(0,\infty)$, let $\nu$ have finite centred second moment, and
use the notation
\[
h_t=P_t*\nu,
\qquad
g_t=P_t(\cdot-\bar\gamma),
\qquad
u_t=h_t/g_t .
\]
Let $\Phi$ be convex on $(0,\infty)$ and assume that, on every compact
subinterval of $(0,\infty)$, $\Phi$ is $C^1$ and $\Phi'$ is Lipschitz.  Then,
uniformly for $t\in[a,b]$, the following analytic bounds hold.
\begin{enumerate}[label=\textup{(\roman*)},leftmargin=*]
\item The principal-value identities for $Lh_t$, $Lg_t$, and
$L(g_tu_t)-u_tLg_t$ are the limits of the cutoff identities in
Lemma~\ref{lem:cauchy-doob-cutoff-calculus}; the associated bilinear Cauchy
forms are absolutely convergent.
\item The quotient range $u_t(\RR)$ is contained in one compact interval
$J_{a,b}\Subset(0,\infty)$, and the Bregman integrand
\[
\frac{g_t(x)D_\Phi(u_t(x),u_t(y))
      +g_t(y)D_\Phi(u_t(y),u_t(x))}{(x-y)^2}
\]
is absolutely integrable on $\RR^2$, with a bound independent of
$t\in[a,b]$.
\item The map
\[
t\longmapsto \mathcal H_\Phi(t):=
\int_{\RR}g_t(x)\Phi(u_t(x))\dd x
\]
is absolutely continuous on $[a,b]$, and its derivative is obtained by
first proving the identity for the cutoff bilinear forms and then letting the
cutoff radius tend to infinity.
\item The passage from smooth convex entropies to the above range-local
$C^{1,1}$ class is justified by the approximants
$\Phi_\varepsilon$ of Lemma~\ref{lem:compact-range-phi-approximation}; on
$J_{a,b}$ the entropies, first derivatives, and Bregman divergences converge
under the same integrable majorants as in~\textup{(ii)}.
\end{enumerate}
Consequently the dissipation formula
\eqref{eq:phi-entropy-dissipation} is a domain statement in the stated
principal-value normalization, not a formal application of a nonlocal product
rule.
\end{lemma}

\begin{proof}
Part (i) is exactly Lemma~\ref{lem:cauchy-doob-cutoff-calculus}.  That lemma
first proves the compactly supported Cauchy-generator identities, then replaces
the test function by $\chi_Rq$ and passes to the limit using the near-diagonal
estimate controlled by $\partial_xp\in L^1$ and the far-field estimate
controlled by $p\in L^1$.  Thus the limiting forms are absolutely convergent
and fix the principal-value convention used here.

Lemma~\ref{lem:doob-range-bregman-integrability} proves the compact quotient
range and the bound \eqref{eq:doob-bregman-integrability-bound}.  Its proof
uses the integral form of the Bregman remainder on the compact range, so
$D_\Phi(a,b)\le C_{a,b}|a-b|^2$ there, and the cutoff energy estimate for
$u_t$ supplies the two-variable majorant.  This gives (ii).

For (iii), Lemma~\ref{lem:cauchy-doob-cutoff-calculus} gives uniform
$L^1$ bounds for $\partial_th_t$ and $\partial_tg_t$ on $[a,b]$; hence
$t\mapsto h_t$ and $t\mapsto g_t$ are absolutely continuous as $L^1$-valued
maps.  On the compact quotient range, $\Phi$ and $\Phi'$ are bounded and
$\Phi'$ is Lipschitz, so the Nemytskii chain rule in $L^1$ gives absolute
continuity of $\mathcal H_\Phi$.  Applying the cutoff identity from part (i) to
the bounded Lipschitz tests $\Phi'(u_t)$ and
$\Phi(u_t)-u_t\Phi'(u_t)$ gives the displayed derivative at the cutoff level;
the absolute majorants from parts (i) and (ii) justify passage to the limit in
the cutoff radius.

Finally, Lemma~\ref{lem:compact-range-phi-approximation} constructs smooth
convex $\Phi_\varepsilon$ that agree with $\Phi$ to first order on the compact
range in the limit and have uniformly bounded second derivatives there.  The
one-variable entropy terms are dominated by $g_t$ times a constant, and the
two-variable Bregman terms are dominated by the majorant from (ii).  Dominated
convergence therefore passes the smooth identity to the range-local
$C^{1,1}$ entropy.  This proves all four assertions.
\end{proof}

\begin{lemma}[Second-moment KL decay to the translated Poisson profile]%
\label{lem:poisson-second-moment-kl-decay}
Let $\nu$ be a probability measure on~$\RR$ with mean $\bar\gamma$ and
finite centred second moment.  Put
\[
h_t:=P_t*\nu,
\qquad
g_t(x):=P_t(x-\bar\gamma).
\]
Then
\[
D_{\mathrm{KL}}(h_t\|g_t)\longrightarrow0
\qquad(t\to\infty).
\]
\end{lemma}

\begin{proof}
Write $\Delta=\gamma-\bar\gamma$ and $x=\bar\gamma+ty$.  Since
$g_t(\bar\gamma+ty)=(\pi t)^{-1}(1+y^2)^{-1}$,
\[
u_t(\bar\gamma+ty):=\frac{h_t(\bar\gamma+ty)}{g_t(\bar\gamma+ty)}
=\int_{\RR}\frac{1+y^2}{1+(y-\Delta/t)^2}\dd\nu(\gamma).
\]
For $A>0$ decompose $\nu=\nu_A+\nu_A^c$ according to
$|\Delta|\le A$ and $|\Delta|>A$.  The elementary kernel bound
\begin{equation}\label{eq:second-moment-kl-kernel-bound}
\frac{1+y^2}{1+(y-\varepsilon)^2}
\le 2(1+\varepsilon^2)
\qquad(y,\varepsilon\in\RR)
\end{equation}
implies
\[
0\le u_{t,A}^c(y):=\int_{|\Delta|>A}
\frac{1+y^2}{1+(y-\Delta/t)^2}\dd\nu(\gamma)
\le 2\int_{|\Delta|>A}\left(1+\frac{\Delta^2}{t^2}\right)\dd\nu(\gamma).
\]
Hence, for $t\ge1$,
\begin{equation}\label{eq:second-moment-kl-tail-sup}
\sup_y u_{t,A}^c(y)
\le 2\nu(|\Delta|>A)+2\int_{|\Delta|>A}\Delta^2\dd\nu.
\end{equation}

The bounded part has total mass $m_A:=\nu(|\Delta|\le A)$ and first centred
moment $b_A:=\int_{|\Delta|\le A}\Delta\dd\nu$.  The exact expansion
\[
\frac{1+y^2}{1+(y-\varepsilon)^2}
=1+\frac{2y}{1+y^2}\varepsilon
+\frac{(3y^2-1)\varepsilon^2-2y\varepsilon^3}
{(1+y^2)(1+(y-\varepsilon)^2)}
\]
shows that, for each fixed $A$,
\begin{equation}\label{eq:bounded-truncated-quotient-limit}
u_{t,A}(y):=\int_{|\Delta|\le A}
\frac{1+y^2}{1+(y-\Delta/t)^2}\dd\nu(
\gamma)
=m_A+\frac{2yb_A}{t(1+y^2)}+O_A(t^{-2})
\end{equation}
uniformly in~$y$.  Since $b_A=-\int_{|\Delta|>A}\Delta\dd\nu$, Cauchy--Schwarz gives
\[
|b_A|\le \left(\int_{|\Delta|>A}\Delta^2\dd\nu\right)^{1/2}.
\]
Combining this with
\eqref{eq:second-moment-kl-tail-sup} and
\eqref{eq:bounded-truncated-quotient-limit} yields
\[
\limsup_{t\to\infty}\|u_t-1\|_{L^\infty(\RR)}
\le 1-m_A+2\nu(|\Delta|>A)+2\int_{|\Delta|>A}\Delta^2\dd\nu.
\]
Letting $A\to\infty$ gives $\|u_t-1\|_\infty\to0$.

For all sufficiently large $t$ we therefore have
$|u_t-1|\le1/2$.  With $\delta_t=u_t-1$ and using
$\int g_t\delta_t=\int(h_t-g_t)=0$,
\[
D_{\mathrm{KL}}(h_t\|g_t)
=\int_{\RR}g_t\bigl((1+\delta_t)\log(1+\delta_t)-\delta_t\bigr)\dd x
\le 2\|\delta_t\|_\infty^2,
\]
because $(1+s)\log(1+s)-s\le2s^2$ for $|s|\le1/2$.  The right-hand side
tends to zero.
\end{proof}


The next theorem gives the Poisson-quotient mechanism for the comparison pair
$h_t=P_t*\nu$ and $g_t=P_t(\cdot-\bar\gamma)$.  It puts the quotient
$u_t=h_t/g_t$ under the non-autonomous Doob generator.  The entropy
dissipation is then stated separately as a domain-closure proposition, and the
KL tail identity as a final corollary.

\begin{theorem}[Non-autonomous Doob transform]\label{thm:poisson-doob-phi-entropy}
Let $\nu$ be a probability measure on~$\RR$ with mean~$\bar\gamma$ and
finite centred second moment.  For $t>0$ define
\[
h_t:=P_t*\nu,
\qquad
g_t(x):=P_t(x-\bar\gamma),
\qquad
u_t(x):=\frac{h_t(x)}{g_t(x)}.
\]
Let
\[
(Lf)(x):=\frac{1}{\pi}\operatorname{p.v.}\int_{\RR}
\frac{f(y)-f(x)}{(x-y)^2}\dd y
\]
be the one-dimensional Cauchy generator, and set
\begin{equation}\label{eq:doob-generator-definition}
(\mathcal L_t f)(x)
:=\frac{1}{g_t(x)}\bigl(L(g_tf)(x)-f(x)Lg_t(x)\bigr).
\end{equation}
For the displayed formulae below the domain of \(\mathcal L_t\) is the class
of bounded \(C^2\) functions with bounded first and second derivatives; the
Green identity is asserted on \(C_c^\infty(\RR)\).  The cutoff interpretation
is the one fixed in Lemma~\ref{lem:cauchy-doob-cutoff-calculus}.
Then the following assertions hold.
\begin{enumerate}[label=(\roman*)]
\item
      The function $u_t$ is bounded and belongs to $C^\infty(\RR)$ for
      every $t>0$.  More precisely, for every $a>0$ and every $t\ge a$
      the compact range bound \eqref{eq:doob-compact-range-bound} holds.
      Moreover,
      \begin{equation}\label{eq:quotient-flow-equation}
      \partial_tu_t=\mathcal L_tu_t
      \qquad\text{on }\RR\times(0,\infty),
      \end{equation}
      and \eqref{eq:doob-generator-definition} may be written explicitly
      as
      \begin{equation}\label{eq:doob-generator-kernel}
      (\mathcal L_tf)(x)
      =\frac{1}{\pi g_t(x)}\operatorname{p.v.}\int_{\RR}
      \frac{g_t(y)\bigl(f(y)-f(x)\bigr)}{(x-y)^2}\dd y.
      \end{equation}
\item
      For every $f,\psi\in C_c^\infty(\RR)$ one has the weighted Green
      identity
      \begin{equation}\label{eq:doob-green-identity}
      \begin{split}
      \int_{\RR}f(x)(\mathcal L_t\psi)(x)\,g_t(x)^2\dd x
      &=
      -\frac{1}{2\pi}
      \iint_{\RR^2}
      \frac{g_t(x)g_t(y)}{(x-y)^2}
      \\
      &\qquad\qquad\times
      \bigl(f(x)-f(y)\bigr)\bigl(\psi(x)-\psi(y)\bigr)\dd x\dd y.
      \end{split}
      \end{equation}
\end{enumerate}
\end{theorem}

\begin{proof}
For $t>0$ the functions $h_t$ and $g_t$ are positive and belong to
$C^\infty(\RR)$, because they are obtained by convolving the smooth
Poisson kernel with a probability measure.  Writing
$x=\bar\gamma+ty$ and $\varepsilon=\Delta/t$, one has
\[
u_t(x)=\int_{\RR}R_{\Delta/t}(y)\dd\nu(\gamma).
\]
The elementary estimate
\[
1+y^2=1+\bigl((y-\varepsilon)+\varepsilon\bigr)^2
\le 2\bigl(1+(y-\varepsilon)^2\bigr)+2\varepsilon^2
\]
gives
\[
0<R_\varepsilon(y)\le 2+2\varepsilon^2
\qquad(y,\varepsilon\in\RR),
\]
and therefore
\[
0<u_t(x)\le 2+\frac{2\mu_2}{t^2},
\qquad
\mu_2:=\int_{\RR}(\gamma-\bar\gamma)^2\dd\nu(\gamma).
\]
Likewise,
\[
1+(y-\varepsilon)^2
\le 2\bigl(1+y^2\bigr)+2\varepsilon^2
\le 2(1+\varepsilon^2)(1+y^2),
\]
so
\[
R_\varepsilon(y)\ge \frac{1}{2(1+\varepsilon^2)}
\qquad(y,\varepsilon\in\RR).
\]
Hence
\[
u_t(x)\ge \frac12\int_{\RR}\frac{1}{1+(\Delta/t)^2}\dd\nu(\gamma)
\ge \frac{1}{2(1+\mu_2/t^2)},
\]
where the last inequality follows from Jensen's inequality applied to the
convex function $s\mapsto(1+s)^{-1}$ on $[0,\infty)$.
Thus $u_t$ is bounded for every fixed $t>0$.

By Proposition~\ref{prop:poisson-semigroup-law},
$\partial_t h_t=-|D|h_t$ and $\partial_t g_t=-|D|g_t$.  The cutoff
calculus of Lemma~\ref{lem:cauchy-doob-cutoff-calculus} identifies these
derivatives with $Lh_t$ and $Lg_t$ in the principal-value sense and permits
the product-rule passage below.  Since $h_t=u_tg_t$ and $g_t(x)>0$,
\[
\partial_tu_t(x)
=\frac{Lh_t(x)-u_t(x)Lg_t(x)}{g_t(x)}.
\]
Substituting $h_t=u_tg_t$ in
\eqref{eq:cutoff-doob-product-rule} yields
\[
\partial_tu_t(x)
=\frac{1}{\pi g_t(x)}\operatorname{p.v.}\int_{\RR}
\frac{g_t(y)\bigl(u_t(y)-u_t(x)\bigr)}{(x-y)^2}\dd y.
\]
This is \eqref{eq:quotient-flow-equation} and proves
\eqref{eq:doob-generator-kernel}; part~(i) follows.

For part~(ii), let $f,\psi\in C_c^\infty(\RR)$.  The truncated integrals in
Lemma~\ref{lem:cauchy-doob-cutoff-calculus} justify using
\eqref{eq:doob-generator-kernel} under the outer integral, and hence
\[
\int_{\RR}f(x)(\mathcal L_t\psi)(x)\,g_t(x)^2\dd x
=
\frac{1}{\pi}\iint_{\RR^2}
\frac{f(x)g_t(x)g_t(y)\bigl(\psi(y)-\psi(x)\bigr)}
{(x-y)^2}\dd x\dd y.
\]
Interchanging $x$ and $y$ in the same integral and averaging the two
representations gives \eqref{eq:doob-green-identity}.
\end{proof}

\begin{proposition}[Domain closure of the Doob--Bregman identity]%
\label{prop:doob-bregman-domain-closure}
Let $\nu$ be a probability measure on~$\RR$ with mean~$\bar\gamma$ and
finite centred second moment, and use the notation of
Theorem~\ref{thm:poisson-doob-phi-entropy}.  Let
$\Phi:(0,\infty)\to\RR$ be convex, of class $C^1$, with range-local
$C^{1,1}$ control in the sense that $\Phi'$ is Lipschitz on every compact
subinterval of $(0,\infty)$, and assume $\Phi(1)=0$.  Define
\[
\mathcal H_\Phi(t):=\int_{\RR}g_t(x)\Phi(u_t(x))\dd x .
\]
Then $\mathcal H_\Phi(t)<\infty$ for every $t>0$,
$t\mapsto\mathcal H_\Phi(t)$ is absolutely continuous on compact
subintervals of $(0,\infty)$, the double integral below is finite for every
$t>0$, and for a.e. $t>0$,
\begin{equation}\label{eq:phi-entropy-dissipation}
-\frac{\dd}{\dd t}\mathcal H_\Phi(t)
=
\frac{1}{2\pi}\iint_{\RR^2}
\frac{
g_t(x)D_\Phi(u_t(x),u_t(y))
+g_t(y)D_\Phi(u_t(y),u_t(x))}
{(x-y)^2}\dd x\dd y,
\end{equation}
where
\[
D_\Phi(a,b):=\Phi(a)-\Phi(b)-\Phi'(b)(a-b)
\qquad(a,b>0).
\]
The right side is the finite cutoff-limit integral fixed by
Lemmas~\ref{lem:cauchy-doob-cutoff-calculus},
\ref{lem:doob-range-bregman-integrability}, and
\ref{lem:doob-bregman-domain-bounds}.  In particular
$t\mapsto\mathcal H_\Phi(t)$ is nonincreasing.
\end{proposition}

\begin{proof}
Fix $[a,b]\subset(0,\infty)$.  Lemma~\ref{lem:doob-range-bregman-integrability}
places $u_t(\RR)$, $t\in[a,b]$, in a compact subinterval of $(0,\infty)$,
proves $\mathcal H_\Phi(t)<\infty$, and gives the uniform Bregman
double-integral bound \eqref{eq:doob-bregman-integrability-bound}.
The closure argument below deliberately separates the proof into three
steps.  First,
Proposition~\ref{prop:smooth-compact-doob-bregman} proves the formula for
smooth compactly supported quotients, with
Lemma~\ref{lem:doob-bregman-algebra} as the only nontrivial algebraic
identity.  Second, the cutoff estimates of
Lemma~\ref{lem:cauchy-doob-cutoff-calculus} pass the bilinear forms from
compactly supported tests to the Poisson quotient.  Third, the compact-range
approximation of Lemma~\ref{lem:compact-range-phi-approximation} passes from
smooth convex entropies to the stated range-local $C^{1,1}$ class.

Lemma~\ref{lem:cauchy-doob-cutoff-calculus} gives uniform $L^1$ bounds for
$\partial_th_t$ and $\partial_tg_t$, so $t\mapsto h_t$ and $t\mapsto g_t$ are
absolutely continuous as $L^1$-valued maps on $[a,b]$.  Since the quotient
range is compact and $\Phi'$ is Lipschitz there, the chain rule for
$g\Phi(h/g)$ gives $\mathcal H_\Phi\in AC([a,b])$ and, for a.e. $t$,
\[
\frac{\dd}{\dd t}\mathcal H_\Phi(t)
=
\int_{\RR}\Phi'(u_t)\,\partial_th_t\dd x
+
\int_{\RR}\bigl(\Phi(u_t)-u_t\Phi'(u_t)\bigr)\partial_tg_t\dd x.
\]
Substituting $\partial_th_t=L(u_tg_t)$ and $\partial_tg_t=Lg_t$, and applying
the cutoff bilinear identity \eqref{eq:cutoff-cauchy-bilinear} with the
bounded Lipschitz tests $\Phi'(u_t)$ and
$\Phi(u_t)-u_t\Phi'(u_t)$, gives the cutoff-limit version of the smooth
identity in Proposition~\ref{prop:smooth-compact-doob-bregman}.  The algebraic
step in that passage is precisely Lemma~\ref{lem:doob-bregman-algebra}, with
\[
u_x=u_t(x),\qquad u_y=u_t(y),\qquad g_x=g_t(x),\qquad g_y=g_t(y).
\]
Thus the cutoff limit is the Bregman integral in
\eqref{eq:phi-entropy-dissipation}.

The remaining passages are exactly the closure statements in
Lemma~\ref{lem:doob-bregman-domain-bounds}.  The cutoff convergence is
controlled by the near-diagonal and far-field majorants of
Lemma~\ref{lem:cauchy-doob-cutoff-calculus}; absolute convergence is the
bound \eqref{eq:doob-bregman-integrability-bound}; and the passage from smooth
convex entropies to the present range-local $C^{1,1}$ class uses the
approximants $\Phi_\varepsilon$ of
Lemma~\ref{lem:compact-range-phi-approximation}.  Their first derivatives have
a common Lipschitz constant on the compact quotient range, so the same
one-variable and Bregman majorants dominate both sides and dominated
convergence lets $\varepsilon\downarrow0$.  Convexity gives
$D_\Phi\ge0$, hence monotonicity.  Since $[a,b]$ was arbitrary, the proposition
follows.
\end{proof}

\begin{corollary}[KL Doob--Bregman tail identity]%
\label{cor:poisson-kl-doob-bregman-tail}
Let $\nu$ be a probability measure on~$\RR$ with mean~$\bar\gamma$ and finite
centred second moment, and use the notation of
Theorem~\ref{thm:poisson-doob-phi-entropy}.  Define
\begin{equation}\label{eq:relative-entropy-dissipation-rate}
\mathcal I_\nu(s)
:=
\frac{1}{\pi}\iint_{\RR^2}
\frac{g_s(x)\Lambda(u_s(x),u_s(y))}{(x-y)^2}\dd x\dd y,
\qquad
\Lambda(a,b):=a\log\frac{a}{b}-a+b.
\end{equation}
Then, for every $t>0$,
\begin{equation}\label{eq:relative-entropy-integrated-dissipation}
D_{\mathrm{KL}}(h_t\|g_t)
=
\int_t^\infty \mathcal I_\nu(s)\dd s.
\end{equation}
\end{corollary}

\begin{proof}
For $\Phi(s)=s\log s$,
\[
\mathcal H_\Phi(t)
=\int_{\RR}g_t(x)u_t(x)\log u_t(x)\dd x
=D_{\mathrm{KL}}(h_t\|g_t).
\]
The corresponding Bregman divergence is exactly
$D_\Phi(a,b)=\Lambda(a,b)$.  Therefore
\eqref{eq:phi-entropy-dissipation} gives, for a.e. $s>0$,
\[
-\frac{\dd}{\dd s}D_{\mathrm{KL}}(h_s\|g_s)
=
\frac{1}{2\pi}\iint_{\RR^2}
\frac{g_s(x)\Lambda(u_s(x),u_s(y))
      +g_s(y)\Lambda(u_s(y),u_s(x))}
{(x-y)^2}\dd x\dd y .
\]
The second summand equals the first after interchanging $x$ and $y$; the
kernel is symmetric and Lemma~\ref{lem:doob-bregman-domain-bounds} supplies the
absolute integrability needed for this change of variables.  Thus the
right-hand side is precisely $\mathcal I_\nu(s)$ as defined in
\eqref{eq:relative-entropy-dissipation-rate}.  Lemma~\ref{lem:poisson-second-moment-kl-decay}
gives $D_{\mathrm{KL}}(h_T\|g_T)\to0$ under the same finite-second-moment
hypothesis as this corollary.  Since the entropy is absolutely continuous on
compact time intervals, integrating the a.e. identity over $[t,T]$ gives
\[
D_{\mathrm{KL}}(h_t\|g_t)-D_{\mathrm{KL}}(h_T\|g_T)
=\int_t^T\mathcal I_\nu(s)\dd s .
\]
Letting $T\to\infty$ proves
\eqref{eq:relative-entropy-integrated-dissipation}.
\end{proof}

\begin{corollary}[Rigidity of the Poisson comparison quotient]%
\label{cor:poisson-doob-rigidity}
Let $\nu$ have mean $\bar\gamma$ and finite centred second moment, and use
the notation of Theorem~\ref{thm:poisson-doob-phi-entropy} and
Corollary~\ref{cor:poisson-kl-doob-bregman-tail}.  The following assertions
are equivalent:
\begin{enumerate}[label=(\roman*)]
\item $\nu=\delta_{\bar\gamma}$;
\item $D_{\mathrm{KL}}(h_t\|g_t)=0$ for some $t>0$;
\item the function $t\mapsto D_{\mathrm{KL}}(h_t\|g_t)$ is constant on a
      non-degenerate interval in $(0,\infty)$;
\item $\mathcal I_\nu(t)=0$ for almost every $t$ in a non-degenerate
      interval in $(0,\infty)$.
\end{enumerate}
Thus the Doob--Bregman identity is used here, through its KL specialization, as
a rigidity and coefficient comparison device for the Poisson quotient.  The
broader $\Phi$ domain-closure proposition supplies the cutoff and domain
justification for that specialization; the paper does not use it as a new sharp
log-Sobolev or log-HLS inequality.
\end{corollary}

\begin{proof}
If $\nu=\delta_{\bar\gamma}$, then $h_t=g_t$ for every $t>0$, so the
relative entropy and the dissipation both vanish.  This proves
(i)$\Rightarrow$(ii)--(iv).

Assume (ii).  Since relative entropy is non-negative and vanishes only when
the two probability densities agree almost everywhere, $h_t=g_t$ a.e. for
that value of~$t$.  Taking Fourier transforms gives
\[
e^{-t|\xi|}\widehat\nu(\xi)
=e^{-t|\xi|}e^{-\ii\bar\gamma\xi}
\qquad(\xi\in\RR).
\]
Hence $\widehat\nu(\xi)=e^{-\ii\bar\gamma\xi}$ for every~$\xi$, and the
uniqueness theorem for characteristic functions gives
$\nu=\delta_{\bar\gamma}$.  Thus (ii)$\Rightarrow$(i).

If (iii) holds on an interval $J$, then the absolutely continuous entropy
function has derivative zero a.e. on~$J$.  By
\eqref{eq:relative-entropy-dissipation-rate}, this is exactly (iv).  Now
assume (iv).  For a.e. $t\in J$, the integral defining $\mathcal I_\nu(t)$
vanishes.  The kernel $(x-y)^{-2}$ is positive off the diagonal,
$g_t(x)>0$, and $\Lambda(a,b)\ge0$ with equality only when $a=b$; hence
$u_t(x)=u_t(y)$ for almost every pair $(x,y)$.  Therefore $u_t$ is a.e.
constant.  Since $\int h_t=\int g_t=1$, the constant is~$1$, so
$h_t=g_t$ a.e. for such a~$t$.  The implication (ii)$\Rightarrow$(i)
applies, and (iv)$\Rightarrow$(i) follows.
\end{proof}

\begin{corollary}[Bregman tail representation of the Poisson entropy coefficients]%
\label{cor:poisson-bregman-tail-coefficients}
Let $N\ge2$, let $0<\eta\le1$, and let $\nu$ be a probability measure
on~$\RR$ with mean $\bar\gamma$ and finite centred $(2N+\eta)$th absolute
moment.  Put
\[
h_t=P_t*\nu,
\qquad
g_t(x)=P_t(x-\bar\gamma),
\qquad
u_t=\frac{h_t}{g_t},
\]
and let $\mathcal I_\nu(t)$ be the nonlocal Bregman dissipation in
\eqref{eq:relative-entropy-dissipation-rate}.  Then, for every
$2\le M\le N$,
\begin{equation}\label{eq:bregman-tail-coefficient-recovery}
A_{2M}(\nu)
=
\lim_{t\to\infty}t^{2M}
\left(
\int_t^\infty\mathcal I_\nu(s)\dd s
-\sum_{m=2}^{M-1}A_{2m}(\nu)t^{-2m}
\right),
\end{equation}
with the sum interpreted as zero when $M=2$.  Equivalently, each even
entropy coefficient is recovered as a successive tail coefficient of
the $\Phi(s)=s\log s$ Bregman dissipation for the Cauchy Doob transform.
This is a representation theorem for the already-defined Poisson entropy
coefficients $A_{2M}(\nu)$ from
Theorem~\ref{thm:universal-entropy-coefficients} and
Theorem~\ref{thm:finite-moment-entropy-remainder}: if numbers
$B_4,\ldots,B_{2N}$ satisfy, for every $2\le M\le N$,
\[
B_{2M}
=
\lim_{t\to\infty}t^{2M}
\left(
\int_t^\infty\mathcal I_\nu(s)\dd s
-\sum_{m=2}^{M-1}B_{2m}t^{-2m}
\right),
\]
then $B_{2M}=A_{2M}(\nu)$ for all $2\le M\le N$.  No coefficient theory for
other nonlocal semigroups is asserted here.
\end{corollary}

\begin{proof}
Corollary~\ref{cor:poisson-kl-doob-bregman-tail} gives
the exact tail identity
\[
D_{\mathrm{KL}}(h_t\|g_t)=\int_t^\infty\mathcal I_\nu(s)\dd s.
\]
This identity is the only use of the Doob--Bregman calculus in the coefficient
recovery below; all coefficient asymptotics enter through the Poisson entropy
expansion already proved in Section~\ref{sec:entropy-asymptotics}.
The finite-moment expansion in
Theorem~\ref{thm:finite-moment-entropy-remainder} gives, for each fixed
$M\le N$,
\[
D_{\mathrm{KL}}(h_t\|g_t)
=\sum_{m=2}^{M}A_{2m}(\nu)t^{-2m}+o(t^{-2M}),
\qquad t\to\infty.
\]
Substituting the tail identity into this expansion, subtracting the terms
with $2\le m<M$, and multiplying by $t^{2M}$ yields
\eqref{eq:bregman-tail-coefficient-recovery}.

The uniqueness assertion is triangular.  For $M=2$ the displayed limit with
the empty sum equals the leading coefficient of the same entropy tail, hence
$B_4=A_4(\nu)$.  Assume inductively that $B_{2m}=A_{2m}(\nu)$ for
$2\le m<M$.  The two recursive limits for $B_{2M}$ and $A_{2M}(\nu)$ then
have identical subtracted lower-order sums, so both equal the coefficient of
$t^{-2M}$ in the expansion above.  Thus $B_{2M}=A_{2M}(\nu)$.  This proves
that the Bregman formula only re-expresses the Poisson entropy coefficients
already defined by the Cayley--Chebyshev coefficient calculus.
\end{proof}
