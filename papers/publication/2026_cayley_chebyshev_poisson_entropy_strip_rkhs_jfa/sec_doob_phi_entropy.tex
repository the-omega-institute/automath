\section{KL Tail Identity for the Poisson Doob Quotient}%
\label{sec:doob-phi-entropy}

Let \(\nu\) be a probability measure on~\(\RR\) with mean \(\bar\gamma\).
For \(t>0\) put
\[
h_t:=P_t*\nu,\qquad
g_t(x):=P_t(x-\bar\gamma),\qquad
u_t:=h_t/g_t .
\]
The Cauchy generator is normalized by
\[
(Lf)(x)=\frac1\pi\operatorname{p.v.}\int_{\RR}
\frac{f(y)-f(x)}{(x-y)^2}\dd y,
\]
so that \(\partial_tP_t=LP_t\).  This section records only the
KL-specialized consequence of the Doob transform.  The range-local
\(C^{1,1}\) convex-entropy closure used to justify the differentiated identity
is in Appendix~\ref{app:doob-bregman-domain}.  The surrounding literature on
nonlocal and discrete entropy-gradient structures includes the finite-chain
entropy convexity theory of Erbar--Maas~\cite{ErbarMaas2012RicciMarkov},
Maas~\cite{Maas2011EntropyMarkov}, Mielke~\cite{Mielke2013EntropyMarkov},
and Erbar's jump-process entropy gradient flows
\cite{Erbar2014JumpEntropy}.  The identity below is not a curvature statement
in those metric structures; it is the Cauchy Doob quotient's cutoff-closed
Bregman dissipation formula.

\begin{lemma}[Second-moment KL decay to the translated Poisson profile]%
\label{lem:poisson-second-moment-kl-decay}
Let $\nu$ be a probability measure on~$\RR$ with mean $\bar\gamma$ and
finite centred second moment.  Put
\[
h_t:=P_t*\nu,
\qquad
g_t(x):=P_t(x-\bar\gamma).
\]
Then
\[
D_{\mathrm{KL}}(h_t\|g_t)\longrightarrow0
\qquad(t\to\infty).
\]
\end{lemma}

\begin{proof}
Write $\Delta=\gamma-\bar\gamma$ and $x=\bar\gamma+ty$.  Since
$g_t(\bar\gamma+ty)=(\pi t)^{-1}(1+y^2)^{-1}$,
\[
u_t(\bar\gamma+ty):=\frac{h_t(\bar\gamma+ty)}{g_t(\bar\gamma+ty)}
=\int_{\RR}\frac{1+y^2}{1+(y-\Delta/t)^2}\dd\nu(\gamma).
\]
For $A>0$ decompose $\nu=\nu_A+\nu_A^c$ according to
$|\Delta|\le A$ and $|\Delta|>A$.  The elementary kernel bound
\begin{equation}\label{eq:second-moment-kl-kernel-bound}
\frac{1+y^2}{1+(y-\varepsilon)^2}
\le 2(1+\varepsilon^2)
\qquad(y,\varepsilon\in\RR)
\end{equation}
implies
\[
0\le u_{t,A}^c(y):=\int_{|\Delta|>A}
\frac{1+y^2}{1+(y-\Delta/t)^2}\dd\nu(\gamma)
\le 2\int_{|\Delta|>A}\left(1+\frac{\Delta^2}{t^2}\right)\dd\nu(\gamma).
\]
Hence, for $t\ge1$,
\begin{equation}\label{eq:second-moment-kl-tail-sup}
\sup_y u_{t,A}^c(y)
\le 2\nu(|\Delta|>A)+2\int_{|\Delta|>A}\Delta^2\dd\nu.
\end{equation}

The bounded part has total mass $m_A:=\nu(|\Delta|\le A)$ and first centred
moment $b_A:=\int_{|\Delta|\le A}\Delta\dd\nu$.  The exact expansion
\[
\frac{1+y^2}{1+(y-\varepsilon)^2}
=1+\frac{2y}{1+y^2}\varepsilon
+\frac{(3y^2-1)\varepsilon^2-2y\varepsilon^3}
{(1+y^2)(1+(y-\varepsilon)^2)}
\]
shows that, for each fixed $A$,
\begin{equation}\label{eq:bounded-truncated-quotient-limit}
u_{t,A}(y):=\int_{|\Delta|\le A}
\frac{1+y^2}{1+(y-\Delta/t)^2}\dd\nu(\gamma)
=m_A+\frac{2yb_A}{t(1+y^2)}+O_A(t^{-2})
\end{equation}
uniformly in~$y$.  Since
$b_A=-\int_{|\Delta|>A}\Delta\dd\nu$, Cauchy--Schwarz gives
\[
|b_A|\le \left(\int_{|\Delta|>A}\Delta^2\dd\nu\right)^{1/2}.
\]
Combining this with
\eqref{eq:second-moment-kl-tail-sup} and
\eqref{eq:bounded-truncated-quotient-limit} yields
\[
\limsup_{t\to\infty}\|u_t-1\|_{L^\infty(\RR)}
\le 1-m_A+2\nu(|\Delta|>A)+2\int_{|\Delta|>A}\Delta^2\dd\nu.
\]
Letting $A\to\infty$ gives $\|u_t-1\|_\infty\to0$.

For all sufficiently large $t$ we therefore have
$|u_t-1|\le1/2$.  With $\delta_t=u_t-1$ and using
$\int g_t\delta_t=\int(h_t-g_t)=0$,
\[
D_{\mathrm{KL}}(h_t\|g_t)
=\int_{\RR}g_t\bigl((1+\delta_t)\log(1+\delta_t)-\delta_t\bigr)\dd x
\le 2\|\delta_t\|_\infty^2,
\]
because $(1+s)\log(1+s)-s\le2s^2$ for $|s|\le1/2$.  The right-hand side
tends to zero.  This is the \(H(T)\to0\) input used below to convert the
finite-interval dissipation balance into the infinite tail identity.
\end{proof}

\begin{lemma}[Explicit Poisson quotient-kernel calculus]%
\label{lem:explicit-poisson-quotient-kernel-calculus}
For
\[
R_\varepsilon(y):=\frac{1+y^2}{1+(y-\varepsilon)^2}
\qquad(y,\varepsilon\in\RR)
\]
put \(w=y-\varepsilon\).  Then
\begin{align}
R_\varepsilon(y)
&=1+\frac{2\varepsilon w+\varepsilon^2}{1+w^2},
\label{eq:explicit-R-eps-zero}\\
\partial_yR_\varepsilon(y)
&=
\frac{2\varepsilon(1-w^2)-2\varepsilon^2w}{(1+w^2)^2},
\label{eq:explicit-R-eps-first}\\
\partial_y^2R_\varepsilon(y)
&=
\frac{4\varepsilon w(w^2-3)+2\varepsilon^2(3w^2-1)}
     {(1+w^2)^3}.
\label{eq:explicit-R-eps-second}
\end{align}
Moreover
\begin{equation}\label{eq:explicit-R-eps-size-bounds}
0<R_\varepsilon(y)\le 2(1+\varepsilon^2),
\qquad
R_\varepsilon(y)\ge\frac{1}{2(1+\varepsilon^2)},
\end{equation}
and the derivative bounds
\begin{equation}\label{eq:explicit-R-eps-derivative-bounds}
|\partial_yR_\varepsilon(y)|\le 3(1+\varepsilon^2),
\qquad
|\partial_y^2R_\varepsilon(y)|\le 16(1+\varepsilon^2)
\end{equation}
hold uniformly in \(y,\varepsilon\).
\end{lemma}

\begin{proof}
The identity \(y=w+\varepsilon\) gives
\[
1+y^2=1+(w+\varepsilon)^2
=(1+w^2)+2\varepsilon w+\varepsilon^2,
\]
which proves \eqref{eq:explicit-R-eps-zero}.  Differentiating with respect
to \(y\), equivalently with respect to \(w\), gives
\[
\partial_yR_\varepsilon(y)
=\frac{2\varepsilon(1+w^2)-(2\varepsilon w+\varepsilon^2)2w}
       {(1+w^2)^2}
=\frac{2\varepsilon(1-w^2)-2\varepsilon^2w}{(1+w^2)^2},
\]
and one further differentiation gives
\eqref{eq:explicit-R-eps-second}.

The upper bound in \eqref{eq:explicit-R-eps-size-bounds} follows from
\[
1+y^2\le 2\bigl(1+(y-\varepsilon)^2\bigr)+2\varepsilon^2,
\]
and the lower bound from
\[
1+(y-\varepsilon)^2
\le2(1+\varepsilon^2)(1+y^2).
\]
For the first derivative, use
\[
\frac{|1-w^2|}{(1+w^2)^2}\le1,
\qquad
\frac{|w|}{(1+w^2)^2}\le1,
\qquad
2|\varepsilon|\le1+\varepsilon^2,
\]
in \eqref{eq:explicit-R-eps-first}; this gives
\(|\partial_yR_\varepsilon|\le3(1+\varepsilon^2)\).  For the second
derivative, \eqref{eq:explicit-R-eps-second} and the elementary bounds
\[
\frac{|w|^3}{(1+w^2)^3}\le1,\quad
\frac{|w|}{(1+w^2)^3}\le1,\quad
\frac{w^2}{(1+w^2)^3}\le1,\quad
\frac{1}{(1+w^2)^3}\le1
\]
give
\[
|\partial_y^2R_\varepsilon(y)|
\le16|\varepsilon|+8\varepsilon^2
\le16(1+\varepsilon^2).
\]
\end{proof}

\begin{theorem}[KL quotient domain theorem]%
\label{thm:kl-local-doob-bregman-domain-closure}
\label{prop:kl-local-doob-bregman-domain-closure}
Let $\nu$ be a probability measure on~$\RR$ with mean~$\bar\gamma$ and finite
centred second moment
\[
\mu_2:=\int_{\RR}(\gamma-\bar\gamma)^2\dd\nu(\gamma)<\infty .
\]
Let $[a,b]\subset(0,\infty)$, set
\[
h_t=P_t*\nu,\qquad g_t=P_t(\cdot-\bar\gamma),\qquad u_t=h_t/g_t,
\]
and let $\chi\in C_c^\infty(\RR)$ satisfy \(0\le\chi\le1\),
\(\chi\equiv1\) on $[-1,1]$,
$\chi_R(x)=\chi(x/R)$.  Then the following facts hold uniformly for
$t\in[a,b]$.
\begin{enumerate}[label=\textup{(\roman*)},leftmargin=*]
\item The quotient range is the explicit compact interval
\begin{equation}\label{eq:kl-domain-compact-range}
\frac{1}{2(1+\mu_2/a^2)}
\le u_t(x)\le
2+\frac{2\mu_2}{a^2}
\qquad(x\in\RR).
\end{equation}
In particular, with
\[
m_{a,\nu}:=\frac{1}{2(1+\mu_2/a^2)},\qquad
M_{a,\nu}:=2+\frac{2\mu_2}{a^2},
\]
all values of $u_t$ lie in
$J_{a,\nu}:=[m_{a,\nu},M_{a,\nu}]\Subset(0,\infty)$.

\item The spatial derivative has the explicit bound
\begin{equation}\label{eq:kl-domain-derivative-bound}
\|\partial_xu_t\|_\infty
\le
L_{a,\nu}:=\frac{5}{a}\left(1+\frac{\mu_2}{a^2}\right).
\end{equation}
Consequently $u_t$ is uniformly Lipschitz on
$\RR\times[a,b]$.

\item For $\Phi(s)=s\log s$ and
\[
\Lambda(\alpha,\beta):=\alpha\log\frac{\alpha}{\beta}-\alpha+\beta,
\]
there is a constant $C_{a,\nu}<\infty$ such that
\begin{equation}\label{eq:kl-domain-bregman-majorant}
0\le \Lambda(u_t(x),u_t(y))
\le C_{a,\nu}\min\{|x-y|^2,1\}
\qquad(x,y\in\RR).
\end{equation}
Hence
\begin{equation}\label{eq:kl-domain-pointwise-bregman-majorant}
\frac{g_t(x)\Lambda(u_t(x),u_t(y))}{(x-y)^2}
\le C_{a,\nu}g_t(x)\min\{1,|x-y|^{-2}\}
\qquad(x\ne y),
\end{equation}
and
\begin{equation}\label{eq:kl-domain-absolute-bregman}
\sup_{t\in[a,b]}
\iint_{\RR^2}
\frac{g_t(x)\Lambda(u_t(x),u_t(y))}{(x-y)^2}\dd x\dd y<\infty .
\end{equation}

\item Define the localized entropies
\[
H_R(t):=\int_{\RR}\chi_R(x)g_t(x)\Phi(u_t(x))\dd x,
\qquad
H(t):=\int_{\RR}g_t(x)\Phi(u_t(x))\dd x .
\]
Then \(H_R\in AC([a,b])\) for every \(R\ge1\), \(H\in AC([a,b])\),
\(H_R\to H\) uniformly on \([a,b]\), and there are functions
\(e_R\in L^1([a,b])\), with
\[
\|e_R\|_{L^1([a,b])}\longrightarrow0,
\]
such that, for a.e. \(t\in[a,b]\),
\begin{align}
-\frac{\dd}{\dd t}H_R(t)
&=
\frac{1}{2\pi}\iint_{\RR^2}
\chi_R(x)\chi_R(y)
\frac{g_t(x)\Lambda(u_t(x),u_t(y))
      +g_t(y)\Lambda(u_t(y),u_t(x))}
{(x-y)^2}\dd x\dd y
 + e_R(t).
\label{eq:kl-domain-localized-entropy-balance}
\end{align}
Consequently the cutoff identity converges in \(L^1([a,b])\) to the
unlocalized dissipation identity
\begin{equation}\label{eq:kl-domain-unlocalized-dissipation}
\frac{1}{\pi}\iint_{\RR^2}
\frac{g_t(x)\Lambda(u_t(x),u_t(y))}{(x-y)^2}\dd x\dd y .
\end{equation}

\item Equivalently, the map
\[
H(t):=D_{\mathrm{KL}}(h_t\|g_t)
=\int_{\RR}g_t(x)u_t(x)\log u_t(x)\dd x
\]
is absolutely continuous on $[a,b]$.  For a.e. $t\in[a,b]$ its derivative is
the one-variable cutoff limit
\begin{align}
-H'(t)
&=
\lim_{R\to\infty}
\left[
-\int_{\RR}\chi_R(x)\bigl(1+\log u_t(x)\bigr)Lh_t(x)\dd x
+\int_{\RR}\chi_R(x)u_t(x)Lg_t(x)\dd x
\right],
\label{eq:kl-main-cutoff-derivative}
\end{align}
and this same limit is exactly the two-variable cutoff limit
\begin{align}
\lim_{R\to\infty}
\frac{1}{2\pi}\iint_{\RR^2}
\chi_R(x)\chi_R(y)
\frac{g_t(x)\Lambda(u_t(x),u_t(y))
      +g_t(y)\Lambda(u_t(y),u_t(x))}
{(x-y)^2}\dd x\dd y .
\label{eq:kl-main-cutoff-bregman-limit}
\end{align}
The convergence to this limit holds in \(L^1([a,b])\), and by
\eqref{eq:kl-domain-absolute-bregman} the cutoff can then be removed in the
last display.  Thus, for a.e. \(t\in[a,b]\),
\begin{equation}\label{eq:kl-domain-final-derivative}
-H'(t)=
\frac{1}{\pi}\iint_{\RR^2}
\frac{g_t(x)\Lambda(u_t(x),u_t(y))}{(x-y)^2}\dd x\dd y .
\end{equation}
\end{enumerate}
\end{theorem}

\begin{remark}
The theorem supplies the domain hypotheses used in the tail theorem below:
compact quotient range, a uniform Lipschitz bound, an explicit near/far
Bregman majorant, \(L^1([a,b])\) cutoff convergence from the noncompact tests
\(1+\log u_t\) and \(u_t\), and then the unlocalized derivative identity.
\end{remark}

\begin{proof}
Write $\Delta=\gamma-\bar\gamma$, $y=(x-\bar\gamma)/t$, and
\[
R_\varepsilon(y):=\frac{1+y^2}{1+(y-\varepsilon)^2}.
\]
Then
\[
u_t(x)=\int_{\RR}R_{\Delta/t}(y)\dd\nu(\gamma).
\]
Lemma~\ref{lem:explicit-poisson-quotient-kernel-calculus} gives
\[
R_\varepsilon(y)\le2(1+\varepsilon^2),
\qquad
R_\varepsilon(y)\ge\frac{1}{2(1+\varepsilon^2)}.
\]
After integration, the upper bound in
\eqref{eq:kl-domain-compact-range} follows from $t\ge a$.  The lower bound
uses Jensen's inequality for the convex function $s\mapsto(1+s)^{-1}$:
\[
\int_{\RR}\frac{1}{1+(\Delta/t)^2}\dd\nu(\gamma)
\ge
\frac{1}{1+\mu_2/t^2}.
\]

For the derivative bound, the explicit first derivative estimate
\eqref{eq:explicit-R-eps-derivative-bounds} gives
\[
|\partial_yR_\varepsilon(y)|
\le3(1+\varepsilon^2)
\qquad(y,\varepsilon\in\RR).
\]
Since $\partial_x=t^{-1}\partial_y$ and $t\ge a$, integration against~$\nu$
proves the slightly sharper bound
\[
\|\partial_xu_t\|_\infty
\le\frac{3}{a}\left(1+\frac{\mu_2}{a^2}\right),
\]
which implies \eqref{eq:kl-domain-derivative-bound}.

On the compact interval $J_{a,\nu}$ the function
$s\mapsto1+\log s$ is Lipschitz, so the integral form of the Bregman remainder
gives
\[
0\le\Lambda(\alpha,\beta)\le C_{a,\nu}'|\alpha-\beta|^2
\qquad(\alpha,\beta\in J_{a,\nu}).
\]
Combining this with \eqref{eq:kl-domain-derivative-bound} and with the
trivial bound $|u_t(x)-u_t(y)|\le M_{a,\nu}-m_{a,\nu}$ proves
\eqref{eq:kl-domain-bregman-majorant}.  Dividing by $(x-y)^2$ gives
\eqref{eq:kl-domain-pointwise-bregman-majorant} and leaves the
integrable kernel $\min\{1,|x-y|^{-2}\}$ in the $y$ variable, and
$\int g_t=1$ proves \eqref{eq:kl-domain-absolute-bregman}.

It remains only to identify the derivative as a domain statement.  Lemma
\ref{lem:cauchy-doob-cutoff-calculus} gives
$\partial_th_t=Lh_t$, $\partial_tg_t=Lg_t$, uniform $L^1$ bounds for these
time derivatives, and the cutoff bilinear identity for bounded Lipschitz
tests.  Since $u_t$ stays in $J_{a,\nu}$, the function $\Phi(u_t)$ is
bounded uniformly on $\RR\times[a,b]$.  Hence
\[
\sup_{t\in[a,b]}|H_R(t)-H(t)|
\le
\sup_{J_{a,\nu}}|\Phi|
\sup_{t\in[a,b]}\int_{\RR}|1-\chi_R(x)|g_t(x)\dd x
\longrightarrow0,
\]
because the translated Poisson tails are tight uniformly for \(t\in[a,b]\).
The $L^1$ chain rule for $g_t\Phi(h_t/g_t)$ gives $H\in AC([a,b])$ and, for
a.e. $t\in[a,b]$,
\[
H'(t)=
\int_{\RR}\bigl(1+\log u_t\bigr)Lh_t\dd x
-\int_{\RR}u_tLg_t\dd x,
\]
with the two pairings interpreted through the cutoff limit in
\eqref{eq:kl-main-cutoff-derivative}.  Applying
\eqref{eq:cutoff-cauchy-bilinear} first to
$p=h_t=g_tu_t$, $q=1+\log u_t$, and then to $p=g_t$, $q=-u_t$, gives the
same cutoff limit as
\[
\frac{1}{2\pi}\lim_{R\to\infty}\iint_{\RR^2}\chi_R(x)\chi_R(y)
\frac{g_t(x)\Lambda(u_t(x),u_t(y))
      +g_t(y)\Lambda(u_t(y),u_t(x))}
{(x-y)^2}\dd x\dd y .
\]
Lemma~\ref{lem:doob-bregman-algebra} is the algebraic identity behind this
last display.  The two summands are equal for every finite $R$ after
interchanging $x$ and~$y$, so the symmetric cutoff is precisely
\eqref{eq:kl-main-cutoff-bregman-limit}.  The absolute convergence already
proved in \eqref{eq:kl-domain-absolute-bregman} justifies removing the cutoff
for each fixed \(t\), and the cutoff convergence is explicit.  For the first
Bregman term set
\[
K_t(x,y):=
\frac{g_t(x)\Lambda(u_t(x),u_t(y))}{(x-y)^2}
\qquad(x\ne y).
\]
On the near diagonal \(|x-y|<1\),
\eqref{eq:kl-domain-bregman-majorant} gives
\begin{equation}\label{eq:kl-cutoff-near-majorant}
0\le K_t(x,y)\le C_{a,\nu}g_t(x)\mathbf{1}_{\{|x-y|<1\}},
\end{equation}
and on the far field \(|x-y|\ge1\),
\eqref{eq:kl-domain-pointwise-bregman-majorant} gives
\begin{equation}\label{eq:kl-cutoff-far-majorant}
0\le K_t(x,y)\le
C_{a,\nu}g_t(x)|x-y|^{-2}\mathbf{1}_{\{|x-y|\ge1\}} .
\end{equation}
Both majorants are in \(L^1([a,b]\times\RR^2)\), since
\[
\int_a^b\int_{\RR}g_t(x)\dd x\dd t=b-a,\qquad
\int_{|z|<1}\dd z=2,\qquad
\int_{|z|\ge1}|z|^{-2}\dd z=2 .
\]
The second Bregman term has the same two estimates after interchanging
\(x\) and \(y\).  Because \(0\le\chi_R\le1\) and
\(\chi_R(x)\chi_R(y)\to1\) pointwise, dominated convergence with the
near-diagonal majorant \eqref{eq:kl-cutoff-near-majorant} and the far-field
majorant \eqref{eq:kl-cutoff-far-majorant} removes the symmetric cutoff in
\(L^1([a,b])\).  After interchanging variables in the second symmetric term,
the limit is exactly the one-sided expression
\eqref{eq:kl-domain-unlocalized-dissipation}, which proves
\eqref{eq:kl-domain-final-derivative}.

It remains to record the localized identity with its error term.  For each
\(R\), the compactly supported factor \(\chi_R\) puts the differentiated
entropy pairing in the cutoff domain, so \(H_R\in AC([a,b])\) and, for a.e.
\(t\),
\[
H_R'(t)=
\int_{\RR}\chi_R(x)(1+\log u_t(x))Lh_t(x)\dd x
-\int_{\RR}\chi_R(x)u_t(x)Lg_t(x)\dd x .
\]
Since \(1+\log u_t\) and \(u_t\) are bounded on
\(\RR\times[a,b]\),
\[
|H_R'(t)-H'(t)|
\le C_{a,\nu}\int_{\RR}|1-\chi_R(x)|
\bigl(|Lh_t(x)|+|Lg_t(x)|\bigr)\dd x .
\]
The right side tends to zero in \(L^1([a,b])\).  Indeed,
\(Lh_t=\partial_th_t\), \(Lg_t=\partial_tg_t\), and Fubini gives, for
\(p_t=h_t\) or \(p_t=g_t\),
\[
\int_a^b\int_{\RR}|1-\chi_R(x)|\,|Lp_t(x)|\dd x\dd t\longrightarrow0,
\]
because \(|1-\chi_R(x)|\to0\) pointwise and the dominating functions are
\(\int_a^b|\partial_tP_t(x-\gamma)|\dd t\) integrated against the finite
measure \(\nu\), or the corresponding point mass at \(\bar\gamma\); their
total \(x\)-integral is bounded by
\((b-a)\sup_{a\le t\le b}\|\partial_tP_t\|_1<\infty\).

Define \(e_R(t)\) by moving the symmetric cutoff integral to the other side
of \eqref{eq:kl-domain-localized-entropy-balance}.  Then
\[
\begin{split}
\|e_R\|_{L^1([a,b])}
&\le
\|H_R'-H'\|_{L^1([a,b])}
\\
&\quad+
\left\|
\frac{1}{2\pi}\iint_{\RR^2}(1-\chi_R(x)\chi_R(y))
\mathcal K_t(x,y)\,\dd x\dd y
\right\|_{L^1_t},
\end{split}
\]
where
\[
\mathcal K_t(x,y):=
\frac{g_t(x)\Lambda(u_t(x),u_t(y))
      +g_t(y)\Lambda(u_t(y),u_t(x))}
{(x-y)^2}.
\]
The first term has just been shown to vanish.  The second term vanishes by
dominated convergence using exactly the near-diagonal majorant
\eqref{eq:kl-cutoff-near-majorant} and the far-field majorant
\eqref{eq:kl-cutoff-far-majorant}, together with the identical estimates for
the \(g_t(y)\Lambda(u_t(y),u_t(x))\) term after exchanging variables.  This
proves \(\|e_R\|_{L^1([a,b])}\to0\), the localized balance
\eqref{eq:kl-domain-localized-entropy-balance}, and the \(L^1\) convergence
of the displayed cutoff integral to the unlocalized dissipation.
\end{proof}

\begin{theorem}[KL Doob tail identity]%
\label{thm:poisson-kl-doob-tail}
\label{cor:poisson-kl-doob-bregman-tail}
Let $\nu$ be a probability measure on~$\RR$ with mean~$\bar\gamma$ and finite
centred second moment
\[
\mu_2:=\int_{\RR}(\gamma-\bar\gamma)^2\dd\nu(\gamma)<\infty .
\]
Let
\[
\Phi(s):=s\log s,\qquad
D_\Phi(a,b):=\Phi(a)-\Phi(b)-\Phi'(b)(a-b)
=a\log\frac{a}{b}-a+b .
\]
On every compact interval \([\alpha,\beta]\subset(0,\infty)\),
Theorem~\ref{thm:kl-local-doob-bregman-domain-closure} supplies the compact
quotient range for \(u_t\), the uniform Lipschitz bound, the near/far Bregman
majorant, the \(L^1([\alpha,\beta])\) cutoff convergence, and the
unlocalized derivative identity.  The following tail identity is asserted
only with those domain closures in force.
Then the quotient satisfies the Doob equation
\begin{equation}\label{eq:quotient-flow-equation-main-kl}
\partial_tu_t=L_tu_t,
\qquad
(L_tf)(x):=\frac1{\pi g_t(x)}
\operatorname{p.v.}\int_{\RR}
\frac{g_t(y)(f(y)-f(x))}{(x-y)^2}\dd y .
\end{equation}
The principal values are understood in the Cauchy-generator normalization
fixed above.  Moreover
\[
t\longmapsto D_{\mathrm{KL}}(h_t\|g_t)
=\int_{\RR}g_t(x)u_t(x)\log u_t(x)\dd x
\]
is absolutely continuous on every compact subinterval of \((0,\infty)\).
For a.e. \(t>0\),
\begin{equation}\label{eq:relative-entropy-dissipation-rate}
-\frac{\dd}{\dd t}D_{\mathrm{KL}}(h_t\|g_t)
=
\frac{1}{\pi}\iint_{\RR^2}
\frac{g_t(x)\Lambda(u_t(x),u_t(y))}{(x-y)^2}\dd x\dd y,
\end{equation}
where
\[
\Lambda(a,b):=a\log\frac{a}{b}-a+b=D_\Phi(a,b).
\]
The double integral in
\eqref{eq:relative-entropy-dissipation-rate} is the cutoff limit
\eqref{eq:kl-main-cutoff-bregman-limit}; equivalently the one-variable
derivative is the cutoff limit \eqref{eq:kl-main-cutoff-derivative}.  The
identity is understood in this cutoff sense.  If
\[
I_\nu(s):=
\frac{1}{\pi}\iint_{\RR^2}
\frac{g_s(x)\Lambda(u_s(x),u_s(y))}{(x-y)^2}\dd x\dd y,
\]
then \(I_\nu\) is a measurable finite function on \((0,\infty)\) and
\(I_\nu\in L^1_{\mathrm{loc}}(0,\infty)\).  More precisely, for every
\(0<a<b<\infty\),
\begin{equation}\label{eq:relative-entropy-local-integrability}
I_\nu\in L^1(a,b),
\end{equation}
and, with
\[
H(t):=D_{\mathrm{KL}}(h_t\|g_t),
\]
\begin{equation}\label{eq:relative-entropy-finite-interval-balance}
D_{\mathrm{KL}}(h_b\|g_b)-D_{\mathrm{KL}}(h_a\|g_a)
=-\int_a^b I_\nu(s)\dd s .
\end{equation}
Since Lemma~\ref{lem:poisson-second-moment-kl-decay} gives
\(D_{\mathrm{KL}}(h_T\|g_T)\to0\), this finite-interval identity implies
for every \(t>0\)
\begin{equation}\label{eq:relative-entropy-tail-integrability}
\int_t^\infty I_\nu(s)\dd s
=D_{\mathrm{KL}}(h_t\|g_t)<\infty .
\end{equation}
Equivalently,
\begin{equation}\label{eq:relative-entropy-integrated-dissipation}
D_{\mathrm{KL}}(h_t\|g_t)=\int_t^\infty I_\nu(s)\dd s.
\end{equation}
\end{theorem}

\begin{remark}
Theorem~\ref{thm:kl-local-doob-bregman-domain-closure} is the domain input:
on every compact time interval it gives the compact quotient range, the
\(\partial_xu_t\) bound, the near/far Bregman majorants, the
\(L^1\)-convergent cutoff identities, and the unlocalized derivative formula.
Thus the identity is a cutoff-closed domain statement, not a formal nonlocal
product-rule computation.
\end{remark}

\begin{proof}
The quotient equation is Theorem~\ref{thm:poisson-doob-phi-entropy}
specialized to the Poisson pair $(h_t,g_t)$; its kernel form is
\eqref{eq:doob-generator-kernel}.
Theorem~\ref{thm:kl-local-doob-bregman-domain-closure} gives absolute
continuity on every compact time interval, the derivative cutoff formula
\eqref{eq:kl-main-cutoff-derivative}, and the Bregman cutoff limit
\eqref{eq:kl-main-cutoff-bregman-limit}.  Since the interval $[a,b]$ in the
domain theorem is arbitrary, these a.e. identities hold locally on
$(0,\infty)$.  The same theorem removes the cutoff and gives the
derivative formula \eqref{eq:relative-entropy-dissipation-rate}.

It remains to record the measure-theoretic content of the tail identity.
For \(s>0\) the functions \(g_s\) and \(u_s\) are continuous in the spatial
variables and continuous in \(s\), while \(\Lambda\) is continuous on
\((0,\infty)^2\).  Hence the nonnegative integrand defining \(I_\nu(s)\) is
Borel on \((0,\infty)\times\RR^2\).  Tonelli's theorem gives measurability of
\(s\mapsto I_\nu(s)\), and the local absolute-convergence estimate
\eqref{eq:kl-domain-absolute-bregman} makes it finite for every \(s>0\).
Set
\[
H(t):=D_{\mathrm{KL}}(h_t\|g_t).
\]
On each compact interval \([a,b]\subset(0,\infty)\), the preceding paragraph
gives \(H\in AC([a,b])\) and
\[
H'(s)=-I_\nu(s)
\]
for a.e. \(s\in[a,b]\).  Therefore \(I_\nu\in L^1([a,b])\) and the
fundamental theorem for absolutely continuous functions gives
\[
H(b)-H(a)=\int_a^b H'(s)\dd s
=-\int_a^b I_\nu(s)\dd s .
\]
Thus \(I_\nu\in L^1_{\mathrm{loc}}(0,\infty)\).  Taking \(a=t\) and
\(b=T\) gives
\[
\int_t^T I_\nu(s)\dd s=H(t)-H(T).
\]
Lemma~\ref{lem:poisson-second-moment-kl-decay} gives \(H(T)\to0\) as
\(T\to\infty\).  Since \(I_\nu\ge0\), monotone convergence yields
\[
\int_t^\infty I_\nu(s)\dd s=H(t),
\]
which is \eqref{eq:relative-entropy-integrated-dissipation}.  In particular,
\(I_\nu\) is measurable and locally integrable, the finite-interval identity is
\(H(b)-H(a)=-\int_a^b I_\nu(s)\dd s\), and
\(\int_t^\infty I_\nu(s)\dd s<\infty\) follows from \(H(T)\to0\).
\end{proof}

Combining \eqref{eq:relative-entropy-integrated-dissipation} with
Theorem~\ref{thm:finite-moment-entropy-remainder} gives the coefficient tail
formula used later.  If \(N\ge2\) and \(\nu\) has finite centred
\((2N-2)\)th absolute moment, then, for every \(2\le M\le N\),
\begin{equation}\label{eq:bregman-tail-coefficient-recovery}
A_{2M}(\nu)
=
\lim_{t\to\infty}t^{2M}
\left(
\int_t^\infty I_\nu(s)\dd s
-\sum_{m=2}^{M-1}A_{2m}(\nu)t^{-2m}
\right),
\end{equation}
with the sum interpreted as zero when \(M=2\).  This is exactly the expansion
of Theorem~\ref{thm:finite-moment-entropy-remainder} rewritten through the KL
tail identity.
