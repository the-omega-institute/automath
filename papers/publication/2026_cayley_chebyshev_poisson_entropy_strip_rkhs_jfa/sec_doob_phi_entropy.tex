\section{KL Tail Identity for the Poisson Doob Quotient}%
\label{sec:doob-phi-entropy}

Let \(\nu\) be a probability measure on~\(\RR\) with mean \(\bar\gamma\).
For \(t>0\) put
\[
h_t:=P_t*\nu,\qquad
g_t(x):=P_t(x-\bar\gamma),\qquad
u_t:=h_t/g_t .
\]
The Cauchy generator is normalized by
\[
(Lf)(x)=\frac1\pi\operatorname{p.v.}\int_{\RR}
\frac{f(y)-f(x)}{(x-y)^2}\dd y,
\]
so that \(\partial_tP_t=LP_t\).  This section records the KL-specialized
Doob--Bregman dissipation identity and its coefficient consequence.  The
coefficient expansion itself is proved in
Theorem~\ref{thm:poisson-kl-even-orders} by the Cayley quotient-mode Taylor
calculus; the Doob identity below supplies an equivalent dissipation-tail
representation after that coefficient theorem is known.

\begin{lemma}[Second-moment KL decay to the translated Poisson profile]%
\label{lem:poisson-second-moment-kl-decay}
Let $\nu$ be a probability measure on~$\RR$ with mean $\bar\gamma$ and
finite centred second moment.  Put
\[
h_t:=P_t*\nu,
\qquad
g_t(x):=P_t(x-\bar\gamma).
\]
Then
\[
D_{\mathrm{KL}}(h_t\|g_t)\longrightarrow0
\qquad(t\to\infty).
\]
\end{lemma}

\begin{proof}
Write $X_c=\gamma-\bar\gamma$ and $x=\bar\gamma+ty$.  Since
$g_t(\bar\gamma+ty)=(\pi t)^{-1}(1+y^2)^{-1}$,
\[
u_t(\bar\gamma+ty):=\frac{h_t(\bar\gamma+ty)}{g_t(\bar\gamma+ty)}
=\int_{\RR}\frac{1+y^2}{1+(y-X_c/t)^2}\dd\nu(\gamma).
\]
For $A>0$ decompose $\nu=\nu_A+\nu_A^c$ according to
$|X_c|\le A$ and $|X_c|>A$.  The elementary kernel bound
\begin{equation}\label{eq:second-moment-kl-kernel-bound}
\frac{1+y^2}{1+(y-\varepsilon)^2}
\le 2(1+\varepsilon^2)
\qquad(y,\varepsilon\in\RR)
\end{equation}
implies
\[
0\le u_{t,A}^c(y):=\int_{|X_c|>A}
\frac{1+y^2}{1+(y-X_c/t)^2}\dd\nu(\gamma)
\le 2\int_{|X_c|>A}\left(1+\frac{X_c^2}{t^2}\right)\dd\nu(\gamma).
\]
Hence, for $t\ge1$,
\begin{equation}\label{eq:second-moment-kl-tail-sup}
\sup_y u_{t,A}^c(y)
\le 2\nu(|X_c|>A)+2\int_{|X_c|>A}X_c^2\dd\nu .
\end{equation}

The bounded part has total mass $m_A:=\nu(|X_c|\le A)$ and first centred
moment $b_A:=\int_{|X_c|\le A}X_c\dd\nu$.  The exact expansion
\[
\frac{1+y^2}{1+(y-\varepsilon)^2}
=1+\frac{2y}{1+y^2}\varepsilon
+\frac{(3y^2-1)\varepsilon^2-2y\varepsilon^3}
{(1+y^2)(1+(y-\varepsilon)^2)}
\]
shows that, for each fixed $A$,
\begin{equation}\label{eq:bounded-truncated-quotient-limit}
u_{t,A}(y):=\int_{|X_c|\le A}
\frac{1+y^2}{1+(y-X_c/t)^2}\dd\nu(\gamma)
=m_A+\frac{2yb_A}{t(1+y^2)}+O_A(t^{-2})
\end{equation}
uniformly in~$y$.  Since
$b_A=-\int_{|X_c|>A}X_c\dd\nu$, Cauchy--Schwarz gives
\[
|b_A|\le \left(\int_{|X_c|>A}X_c^2\dd\nu\right)^{1/2}.
\]
Combining this with
\eqref{eq:second-moment-kl-tail-sup} and
\eqref{eq:bounded-truncated-quotient-limit} yields
\[
\limsup_{t\to\infty}\|u_t-1\|_{L^\infty(\RR)}
\le 1-m_A+2\nu(|X_c|>A)+2\int_{|X_c|>A}X_c^2\dd\nu .
\]
Letting $A\to\infty$ gives $\|u_t-1\|_\infty\to0$.

For all sufficiently large $t$ we therefore have
$|u_t-1|\le1/2$.  With $\delta_t=u_t-1$ and using
$\int g_t\delta_t=\int(h_t-g_t)=0$,
\[
D_{\mathrm{KL}}(h_t\|g_t)
=\int_{\RR}g_t\bigl((1+\delta_t)\log(1+\delta_t)-\delta_t\bigr)\dd x
\le 2\|\delta_t\|_\infty^2,
\]
because $(1+s)\log(1+s)-s\le2s^2$ for $|s|\le1/2$.  The right-hand side
tends to zero.
\end{proof}

\begin{theorem}[KL quotient domain theorem]%
\label{thm:kl-local-doob-bregman-domain-closure}
\label{prop:kl-local-doob-bregman-domain-closure}
Let $\nu$ be a probability measure on~$\RR$ with mean~$\bar\gamma$ and finite
centred second moment.  Let \(0<a<b<\infty\), and use
\[
h_t=P_t*\nu,\qquad g_t=P_t(\cdot-\bar\gamma),\qquad u_t=h_t/g_t .
\]
Set
\[
\Lambda(\alpha,\beta):=\alpha\log\frac{\alpha}{\beta}-\alpha+\beta
\qquad(\alpha,\beta>0)
\]
and
\[
H(t):=D_{\mathrm{KL}}(h_t\|g_t).
\]
Then \(H\in AC([a,b])\), the Bregman integral
\[
\iint_{\RR^2}
\frac{g_t(x)\Lambda(u_t(x),u_t(y))}{(x-y)^2}\dd x\dd y
\]
is finite for a.e. \(t\in[a,b]\) and belongs to \(L^1([a,b])\), and
\begin{equation}\label{eq:kl-domain-final-derivative}
-H'(t)=
\frac{1}{\pi}\iint_{\RR^2}
\frac{g_t(x)\Lambda(u_t(x),u_t(y))}{(x-y)^2}\dd x\dd y
\qquad\text{for a.e. }t\in[a,b].
\end{equation}
More explicitly, the quotient is bounded away from \(0\) and infinity and is
uniformly Lipschitz on \(\RR\times[a,b]\).  Hence there is
\(C_{a,b,\nu}<\infty\) such that, for \(x\ne y\),
\begin{equation}\label{eq:kl-domain-pointwise-bregman-majorant}
0\le
\frac{g_t(x)\Lambda(u_t(x),u_t(y))}{(x-y)^2}
\le C_{a,b,\nu}g_t(x)\min\{1,|x-y|^{-2}\},
\end{equation}
and the same bound holds after exchanging \(x\) and \(y\).  The identity
\eqref{eq:kl-domain-final-derivative} is the \(L^1([a,b])\) limit of the
standard product-cutoff identities; with
\(\chi_R(x)=\chi(x/R)\), \(0\le\chi\le1\), \(\chi\equiv1\) on \([-1,1]\),
\begin{align}
-H'(t)
&=
\lim_{R\to\infty}
\frac{1}{2\pi}\iint_{\RR^2}\chi_R(x)\chi_R(y)
\frac{
g_t(x)\Lambda(u_t(x),u_t(y))
+g_t(y)\Lambda(u_t(y),u_t(x))}
{(x-y)^2}\dd x\dd y .
\label{eq:kl-main-cutoff-bregman-limit}
\end{align}
\end{theorem}

\begin{proof}
This is the KL specialization of
Proposition~\ref{prop:doob-bregman-domain-closure} with
\(\Phi(s)=s\log s\).  The Bregman divergence of this \(\Phi\) is precisely
\[
D_\Phi(\alpha,\beta)
=\alpha\log(\alpha/\beta)-\alpha+\beta=\Lambda(\alpha,\beta).
\]
Proposition~\ref{prop:doob-bregman-domain-closure} gives absolute
continuity of \(H=\mathcal H_\Phi\), the compact quotient range, the
Lipschitz bound, the majorant \eqref{eq:kl-domain-pointwise-bregman-majorant},
and \(L^1([a,b])\) convergence of the product-cutoff Bregman forms.  Since
the two terms in the symmetric Bregman form are equal after exchanging
\(x\) and \(y\), the factor \(1/(2\pi)\) in
\eqref{eq:phi-entropy-dissipation} becomes the one-sided factor \(1/\pi\),
which is \eqref{eq:kl-domain-final-derivative}.
\end{proof}

\begin{proposition}[Domain inputs for the KL tail identity]%
\label{prop:kl-tail-cutoff-inputs}
Under the hypotheses and notation of
Theorem~\ref{thm:kl-local-doob-bregman-domain-closure}, define
\[
H(t):=D_{\mathrm{KL}}(h_t\|g_t),
\qquad
I_\nu(t):=
\frac{1}{\pi}\iint_{\RR^2}
\frac{g_t(x)\Lambda(u_t(x),u_t(y))}{(x-y)^2}\dd x\dd y .
\]
Then \(I_\nu\in L^1_{\mathrm{loc}}(0,\infty)\), \(H\in AC_{\mathrm{loc}}(0,\infty)\),
and
\begin{equation}\label{eq:kl-tail-input-unlocalized}
H'(t)=-I_\nu(t)
\qquad\text{for a.e. }t>0.
\end{equation}
The same identity is obtained as the principal-value, product-cutoff limit
fixed in Appendix~\ref{app:doob-bregman-domain}; no formal nonlocal product
rule is used.
\end{proposition}

\begin{proof}
Apply Theorem~\ref{thm:kl-local-doob-bregman-domain-closure} on each compact
interval \([a,b]\subset(0,\infty)\).
\end{proof}

\begin{theorem}[KL Doob tail identity]%
\label{thm:poisson-kl-doob-tail}
\label{cor:poisson-kl-doob-bregman-tail}
Let $\nu$ be a probability measure on~$\RR$ with mean~$\bar\gamma$ and finite
centred second moment.  Define
\[
I_\nu(s):=
\frac{1}{\pi}\iint_{\RR^2}
\frac{
g_s(x)\left(
u_s(x)\log\frac{u_s(x)}{u_s(y)}-u_s(x)+u_s(y)
\right)}
{(x-y)^2}\dd x\dd y .
\]
Then \(I_\nu\in L^1_{\mathrm{loc}}(0,\infty)\), and for every \(t>0\),
\begin{equation}\label{eq:relative-entropy-integrated-dissipation}
D_{\mathrm{KL}}(h_t\|g_t)=\int_t^\infty I_\nu(s)\dd s.
\end{equation}
\end{theorem}

\begin{proof}
Set \(H(t):=D_{\mathrm{KL}}(h_t\|g_t)\).  Proposition~\ref{prop:kl-tail-cutoff-inputs}
gives \(H'(s)=-I_\nu(s)\) for a.e. \(s>0\) and
\(I_\nu\in L^1_{\mathrm{loc}}(0,\infty)\).  Thus, for \(0<t<T\),
\[
\int_t^T I_\nu(s)\dd s=H(t)-H(T).
\]
Lemma~\ref{lem:poisson-second-moment-kl-decay} gives \(H(T)\to0\) as
\(T\to\infty\).  Since \(I_\nu\ge0\), monotone convergence yields
\eqref{eq:relative-entropy-integrated-dissipation}.
\end{proof}

\begin{proposition}[Dissipation-tail coefficient expansion]%
\label{prop:dissipation-tail-coefficient-expansion}
Let \(N\ge2\), and assume that \(\nu\) has finite centred
\((2N-2)\)th absolute moment.  Then the KL dissipation tail has the same
universal coefficient polynomial layer as the entropy expansion:
\begin{equation}\label{eq:bregman-tail-coefficient-expansion}
\int_t^\infty I_\nu(s)\dd s
=\sum_{m=2}^{N}A_{2m}(\nu)t^{-2m}+o(t^{-2N})
\qquad(t\to\infty).
\end{equation}
Equivalently, for every \(2\le M\le N\),
\begin{equation}\label{eq:bregman-tail-coefficient-recovery}
A_{2M}(\nu)
=
\lim_{t\to\infty}t^{2M}
\left(
\int_t^\infty I_\nu(s)\dd s
-\sum_{m=2}^{M-1}A_{2m}(\nu)t^{-2m}
\right),
\end{equation}
with the sum interpreted as zero when \(M=2\).

More locally, for each fixed \(\lambda>1\),
\begin{equation}\label{eq:bregman-finite-window-coefficients}
\int_t^{\lambda t} I_\nu(s)\dd s
=
\sum_{m=2}^{N}A_{2m}(\nu)(1-\lambda^{-2m})t^{-2m}
+o(t^{-2N}).
\end{equation}
Thus the averaged dissipation on a multiplicative window satisfies
\begin{equation}\label{eq:bregman-window-average-coefficients}
\frac{1}{(\lambda-1)t}\int_t^{\lambda t} I_\nu(s)\dd s
=
\sum_{m=2}^{N}
\frac{1-\lambda^{-2m}}{\lambda-1}\,A_{2m}(\nu)t^{-(2m+1)}
+o(t^{-(2N+1)}).
\end{equation}
If, in addition, \(0<\eta\le1\) and the centred
\((2N-2+\eta)\)th absolute moment is finite, then the error terms above have
the corresponding positive-margin bounds obtained by applying part
\textup{(PM)} of Theorem~\ref{thm:poisson-kl-even-orders} at \(t\) and
\(\lambda t\).
\end{proposition}

\begin{proof}
The tail identity \eqref{eq:relative-entropy-integrated-dissipation} identifies
\(\int_t^\infty I_\nu\) with \(D_{\mathrm{KL}}(h_t\|g_t)\).  Applying part
\textup{(E)} of Theorem~\ref{thm:poisson-kl-even-orders} gives
\eqref{eq:bregman-tail-coefficient-expansion} and hence the coefficient
recovery formula \eqref{eq:bregman-tail-coefficient-recovery}.

For the finite window, use the already proved tail identity twice:
\[
H(t):=D_{\mathrm{KL}}(h_t\|g_t),
\qquad
\int_t^{\lambda t}I_\nu(s)\dd s=H(t)-H(\lambda t).
\]
Substitute the expansion of \(H\) at \(t\) and at \(\lambda t\).  The
coefficient of \(t^{-2m}\) is
\(A_{2m}(\nu)-A_{2m}(\nu)\lambda^{-2m}\), which gives
\eqref{eq:bregman-finite-window-coefficients}; division by
\((\lambda-1)t\) gives \eqref{eq:bregman-window-average-coefficients}.
The positive-margin assertion follows from the explicit remainder estimate in
Theorem~\ref{thm:poisson-kl-even-orders}\textup{(PM)} for \(H(t)\) and
\(H(\lambda t)\).  No pointwise asymptotic for \(I_\nu(t)\) is asserted here:
the coefficient theorem controls the entropy tail and therefore its exact
tail-window averages, while differentiating an asymptotic remainder would
require additional regularity not used in the finite-moment hierarchy.
\end{proof}
