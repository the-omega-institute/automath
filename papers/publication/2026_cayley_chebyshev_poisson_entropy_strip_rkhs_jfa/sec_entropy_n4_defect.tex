\subsection{Explicit sixth/eighth layer and defect consequence}

\begin{lemma}[Normalized Laurent representatives through order six]%
\label{lem:n4-normalized-laurent-representatives}
For \(2\le n\le6\), write \(Q_n(z)=\sum q_{n,k}z^k\).  Then \(q_{n,0}=0\)
and, for \(0<|k|\le n\),
\[
q_{n,k}=
\lambda_n\operatorname{sgn}(k)^{\,n\bmod2}
\binom{n-1}{|k|-1},
\]
where
\[
\lambda_2=-\frac14,\quad
\lambda_3=\frac{\ii}{8},\quad
\lambda_4=\frac1{16},\quad
\lambda_5=-\frac{\ii}{32},\quad
\lambda_6=-\frac1{64}.
\]
\end{lemma}

\begin{proof}
This is Theorem~\ref{thm:finite-fourier-mode-integral-algorithm} written out
for \(n=2,\ldots,6\), including the Laurent factors \(1/2\) and \(1/(2\ii)\).
\end{proof}

\begin{lemma}[Laurent constant-term certificate through order eight]%
\label{lem:eighth-order-coefficient-arithmetic}
\label{eq:eighth-order-constant-term-recurrence}
For the mode products entering \(A_4,A_6,A_8\), the required integrals are
\[
\begin{array}{c|c}
\mathbf n&I(\mathbf n)\\
\hline
(2,2)&1/4\\
(2,4)&-1/8\\
(3,3)&3/16\\
(2,2,2)&-3/32\\
(2,6)&3/64\\
(3,5)&-15/128\\
(4,4)&5/32\\
(2,2,4)&3/32\\
(2,3,3)&-9/128\\
(2,2,2,2)&9/64 .
\end{array}
\]
Odd total-order entries through order eight vanish.
\end{lemma}

\begin{proof}
Apply the constant-term rule
\eqref{eq:mode-integral-constant-term} to the Laurent representatives in
Lemma~\ref{lem:n4-normalized-laurent-representatives}.  The displayed table
is the finite zero-sum convolution of the listed coefficient vectors.
\end{proof}

\begin{lemma}[Substitution into \(A_4,A_6,A_8\)]%
\label{lem:n4-entropy-coefficient-substitution}
For \(\mu_2=\sigma^2\),
\[
A_4=\frac{\sigma^4}{8},
\qquad
A_6=\frac{\sigma^6+6\mu_3^2-8\sigma^2\mu_4}{64},
\]
and
\[
A_8=
\frac{3\sigma^8-12\sigma^4\mu_4+9\sigma^2\mu_3^2
+12\sigma^2\mu_6-30\mu_3\mu_5+20\mu_4^2}{256}.
\]
\end{lemma}

\begin{proof}
Put \(d_t=t^{-2}a_2+\cdots+t^{-6}a_6\), \(a_j=\mu_ju_j\), and use
\[
\Phi(s)=\frac12s^2-\frac16s^3+\frac1{12}s^4+O(s^5).
\]
The \(t^{-4},t^{-6},t^{-8}\) polynomial layers are
\[
\frac12a_2^2,\qquad
a_2a_4+\frac12a_3^2-\frac16a_2^3,
\]
and
\[
a_2a_6+a_3a_5+\frac12a_4^2
-\frac12a_2^2a_4-\frac12a_2a_3^2+\frac1{12}a_2^4.
\]
Substitute the constant terms from
Lemma~\ref{lem:eighth-order-coefficient-arithmetic}.
\end{proof}

\begin{theorem}[Self-contained sixth/eighth coefficient identification]%
\label{thm:poisson-kl-eighth}
Let \(\nu\) have mean \(\bar\gamma\) and finite variance
\(\sigma^2\ge0\).
If \(\nu\) has finite centred fourth absolute moment, then
\[
D_{\rm KL}(P_t*\nu\|P_t(\cdot-\bar\gamma))
=\frac{\sigma^4}{8t^4}
+\frac{\sigma^6+6\mu_3^2-8\sigma^2\mu_4}{64t^6}
+o(t^{-6}).
\]
If \(\nu\) has finite centred sixth absolute moment, then
\begin{equation}\label{eq:poisson-kl-eighth}
D_{\rm KL}(P_t*\nu\|P_t(\cdot-\bar\gamma))
=\frac{\sigma^4}{8t^4}+\frac{A_6(\nu)}{t^6}
+\frac{A_8(\nu)}{t^8}+o(t^{-8}),
\end{equation}
where \(A_6,A_8\) are the polynomials in
Lemma~\ref{lem:n4-entropy-coefficient-substitution}.
When \(\sigma^2=0\), these are the trivial exact identities for the point
mass \(\delta_{\bar\gamma}\).  Positive variance is not a hypothesis for raw
coefficient identification; it is imposed only in normalized defect
statements, such as Corollary~\ref{cor:coefficient-defect-coupling}, where
\(\rho_\sigma\) and powers of \(\sigma^{-1}\) are used.
\end{theorem}

\begin{proof}
If \(\sigma^2=0\), then \(\nu=\delta_{\bar\gamma}\),
\(P_t*\nu=P_t(\cdot-\bar\gamma)\), and every centred moment in the displayed
polynomials is zero, so both formulas are exact.  Assume now
\(\sigma^2>0\).
Use Theorem~\ref{thm:poisson-kl-even-orders} with \(N=3\) for the sixth
coefficient and \(N=4\) for the eighth coefficient.  The finite Laurent
substitution is Lemma~\ref{lem:n4-entropy-coefficient-substitution}.
\end{proof}

\begin{theorem}[Quantitative \(N=4\) remainder under the seventh moment]%
\label{thm:poisson-kl-eighth-seventh-remainder}
If, in addition, the centred seventh absolute moment is finite, then
\[
D_{\rm KL}(P_t*\nu\|P_t(\cdot-\bar\gamma))
=\frac{\sigma^4}{8t^4}+\frac{A_6(\nu)}{t^6}
+\frac{A_8(\nu)}{t^8}+O_\nu(t^{-9}).
\]
\end{theorem}

\begin{proof}
Apply Lemma~\ref{lem:positive-margin-quotient-remainder} with
\(N=4\), \(L=6\), and \(\eta=1\).  The seventh absolute moment gives the
quotient expansion
\[
\delta_t(y)=\sum_{n=2}^{6}\mu_nu_n(y)t^{-n}+\rho_{6,t}(y),
\qquad
\|\rho_{6,t}\|_\infty+\|\rho_{6,t}\|_{L^1(\omega)}=O_\nu(t^{-7}),
\]
and \(\|\delta_t\|_\infty=O_\nu(t^{-2})\).  In the entropy expansion
\(\Phi(s)=(1+s)\log(1+s)-s\), every term containing \(\rho_{6,t}\) also
contains at least one further density factor at the quadratic entropy scale;
therefore its \(L^1(\omega)\) contribution is \(O_\nu(t^{-2}t^{-7})\).
The Taylor tail after degree \(4\) is \(O_\nu(t^{-10})\).  Hence the retained
pure-mode coefficients \(A_4,A_6,A_8\), identified in
Theorem~\ref{thm:poisson-kl-eighth}, have a total remainder
\(O_\nu(t^{-9})\).
\end{proof}

\begin{unnumberedlemma}[Normalized sign--Rademacher coupling]
Let \(X\) be a real random variable with \(\EE X=0\), \(\EE X^2=1\), and
\(\EE X^4<\infty\).  Put
\[
\beta_3=\EE X^3,\qquad
Q_2=X^2-1-\beta_3X,\qquad
\kappa=\EE Q_2^2,\qquad
\eta=\frac{\beta_3^2}{32}+\frac{\kappa}{8}.
\]
For \(\rho_1=\frac12(\delta_{-1}+\delta_1)\),
\begin{equation}\label{eq:normalized-sign-rademacher-coupling}
W_2(\mathcal L(X),\rho_1)
\le 4\eta^{1/4}+8\eta^{1/2}.
\end{equation}
\end{unnumberedlemma}

\begin{proof}
First record the defect size which is seen by the sign map.  Expanding
\(Q_2^2\) and using \(\EE X=0\), \(\EE X^2=1\), and
\(\EE X^3=\beta_3\) gives
\[
\kappa
=\EE[(X^2-1-\beta_3X)^2]
=\EE[(X^2-1)^2]-\beta_3^2.
\]
Thus, with
\[
M:=\EE[(X^2-1)^2]=\kappa+\beta_3^2,
\]
one has
\begin{equation}\label{eq:sign-defect-size-bound}
M\le 32\eta .
\end{equation}

Let \(S=1\) on \(\{X\ge0\}\) and \(S=-1\) on \(\{X<0\}\), and let
\(r_p=\mathcal L(S)=p\delta_1+(1-p)\delta_{-1}\).  For every real \(x\),
with this convention at \(x=0\),
\[
|x-\operatorname{sgn}x|\le |x^2-1|;
\]
indeed \(|x^2-1|=|x-1|(x+1)\) for \(x\ge0\), and
\(|x^2-1|=|x+1|(1-x)\) for \(x<0\).  Coupling \(X\) and \(S\) on the same
probability space therefore gives
\begin{equation}\label{eq:couple-x-to-sign}
W_2(\mathcal L(X),r_p)
\le \bigl(\EE|X-S|^2\bigr)^{1/2}
\le M^{1/2}.
\end{equation}

It remains to couple the biased sign law to the symmetric Rademacher law.
On the two-point set \(\{-1,1\}\) the optimal quadratic transport moves the
excess mass \(|p-\frac12|\) across distance \(2\), hence
\[
W_2(r_p,\rho_1)^2=4\left|p-\frac12\right|.
\]
Since \(\EE X=0\),
\[
2p-1=\EE S=\EE(S-X),
\]
and Cauchy--Schwarz together with \eqref{eq:couple-x-to-sign} gives
\[
\left|p-\frac12\right|
\le \frac12\EE|S-X|
\le \frac12 M^{1/2}.
\]
Consequently
\begin{equation}\label{eq:couple-biased-sign-to-symmetric}
W_2(r_p,\rho_1)\le \sqrt2\,M^{1/4}.
\end{equation}
The triangle inequality, followed by \eqref{eq:sign-defect-size-bound}, now
yields
\[
W_2(\mathcal L(X),\rho_1)
\le M^{1/2}+\sqrt2\,M^{1/4}
\le \sqrt{32}\,\eta^{1/2}+\sqrt2\,32^{1/4}\eta^{1/4}.
\]
Finally \(\sqrt{32}<8\) and \(\sqrt2\,32^{1/4}=2^{7/4}<4\), proving
\eqref{eq:normalized-sign-rademacher-coupling}.
\end{proof}

For the coefficient-defect statement below, assume \(\nu\) has mean
\(\bar\gamma\), variance \(\sigma^2>0\), and finite centred fourth absolute
moment, and put \(X=(\gamma-\bar\gamma)/\sigma\).  With
\[
\beta_3=\EE X^3,\qquad
Q_2=X^2-1-\beta_3X,
\]
the sixth-order coefficient defect attached to \(A_6(\nu)\) is
\begin{equation}\label{eq:coefficient-defect-normalized-algebra}
\Delta_6(\nu)
:=-\frac{7}{64}\sigma^6
-\frac{\sigma^6+6\mu_3^2-8\sigma^2\mu_4}{64}
=\sigma^6\left(
\frac{\beta_3^2}{32}+\frac{\|Q_2\|_2^2}{8}
\right).
\end{equation}
Here \(\|Q_2\|_2^2=\EE_\nu Q_2^2\), with \(Q_2\) evaluated on the normalized
centred variable \(X\).

\begin{corollary}[Coefficient-defect coupling]%
\label{cor:coefficient-defect-coupling}
Let \(\nu\) have mean \(\bar\gamma\), variance \(\sigma^2>0\), and finite
centred fourth absolute moment.  Let \(\Delta_6(\nu)\) be the coefficient
defect in \eqref{eq:coefficient-defect-normalized-algebra}.
For the centred law \(\nu_c\) and
\(\rho_\sigma=\frac12(\delta_{-\sigma}+\delta_\sigma)\),
\[
W_2(\nu_c,\rho_\sigma)
\le
4\,\sigma^{-1/2}\Delta_6(\nu)^{1/4}
+8\,\sigma^{-2}\Delta_6(\nu)^{1/2}.
\]
Moreover \(\Delta_6(\nu)\ge0\), and equality holds if and only if
\(\nu_c=\rho_\sigma\).  The assertion is a moment-coupling consequence of the
identified \(A_6\) coefficient; it uses no sixth moment, eighth-order
coefficient, finite-time KL value, or post-processing constant.
\end{corollary}

\begin{proof}
With \(X=(\gamma-\bar\gamma)/\sigma\), \(\beta_3=\EE X^3\),
\(\beta_4=\EE X^4\), and \(Q_2=X^2-1-\beta_3X\),
\[
\|Q_2\|_2^2
=\EE[(X^2-1-\beta_3X)^2]
=\beta_4-1-\beta_3^2 .
\]
Together with \eqref{eq:coefficient-defect-normalized-algebra}, this gives
\(\Delta_6(\nu)=\sigma^6\eta\) for the normalized defect \(\eta\) above.
Because quadratic Wasserstein distance is homogeneous under dilation,
\[
W_2(\nu_c,\rho_\sigma)
=\sigma\,W_2(\mathcal L(X),\rho_1)
\]
and the preceding normalized sign--Rademacher coupling lemma gives
\[
\sigma\,W_2(\mathcal L(X),\rho_1)
\le \sigma\{4\eta^{1/4}+8\eta^{1/2}\},
\]
which is exactly
\[
W_2(\nu_c,\rho_\sigma)
\le
4\,\sigma^{-1/2}\Delta_6(\nu)^{1/4}
+8\,\sigma^{-2}\Delta_6(\nu)^{1/2}.
\]
The same identity gives \(\Delta_6\ge0\).  If \(\Delta_6=0\), then
\(\beta_3=0\) and \(Q_2=0\) almost surely, so \(X^2=1\) almost surely; with
\(\EE X=0\), this forces \(X=\pm1\) with equal probabilities.  Hence
\(\nu_c=\rho_\sigma\).  The converse follows by direct substitution.
\end{proof}

\begin{remark}[Relation with functional-inequality stability]
Corollary~\ref{cor:coefficient-defect-coupling} is not an optimizer-stability
theorem for a sharp functional inequality.  Its input is the finite
fourth-moment algebra which identifies \(A_6(\nu)\), and its output is a
coupling estimate between the normalized law and the symmetric two-point law.
The comparison with logarithmic Sobolev, Moser--Onofri, or log-HLS stability
results is therefore only contextual: those results study deficits for sharp
functional inequalities under their own normalization and proximity
hypotheses, whereas \eqref{eq:coefficient-defect-normalized-algebra} is a
coefficient identity for the matched Poisson KL expansion.
\end{remark}
