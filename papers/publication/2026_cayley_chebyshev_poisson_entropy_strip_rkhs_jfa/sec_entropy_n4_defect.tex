\subsection{Explicit coefficients through order six}
\label{subsec:explicit-sixth-order-coefficients}

This main-text coefficient subsection stops at the order used by the defect
theorem.  The Cayley--Chebyshev constant-term rule is the one already proved
in Theorem~\ref{thm:finite-fourier-mode-integral-algorithm}: with
\(Q_n(z)=\sum_kq_{n,k}z^k\),
\[
J(n_1,\ldots,n_r)=[z^0]\prod_{j=1}^rQ_{n_j}(z).
\]
The optional eighth-order coefficient, its seventh-moment quantitative
remainder, and the longer row certificate are separated into
Appendix~\ref{app:order-eight-laurent-certificate}.  They are not used in the
sixth-order defect coupling below.

\begin{lemma}[Laurent constants for the sixth-order coefficient]%
\label{lem:sixth-order-coefficient-arithmetic}
For the Laurent representatives \(Q_n\) of
\eqref{eq:mode-laurent-polynomial-definition},
\[
J(2,2)=\frac14,\qquad
J(2,4)=-\frac18,\qquad
J(3,3)=\frac3{16},\qquad
J(2,2,2)=-\frac3{32}.
\]
Consequently, for \(\mu_2=\sigma^2\),
\[
A_4=\frac{\sigma^4}{8},\qquad
A_6=\frac{\sigma^6+6\mu_3^2-8\sigma^2\mu_4}{64}.
\]
\end{lemma}

\begin{proof}
The required rows are the first four rows of the deterministic finite
recurrence in Lemma~\ref{lem:appendix-d-deterministic-laurent-recurrence};
they are also listed in
Appendix~\ref{app:order-eight-laurent-certificate}.  Substituting these
constant terms into the universal coefficient formula
\eqref{eq:universal-entropy-coefficient} gives
\[
A_4=\frac12\mu_2^2J(2,2)=\frac{\sigma^4}{8},
\]
and
\[
\begin{aligned}
A_6
&=\mu_2\mu_4J(2,4)+\frac12\mu_3^2J(3,3)
-\frac16\mu_2^3J(2,2,2)\\
&=\sigma^2\mu_4\Bigl(-\frac18\Bigr)
+\frac12\mu_3^2\Bigl(\frac3{16}\Bigr)
-\frac16\sigma^6\Bigl(-\frac3{32}\Bigr)\\
&=\frac{\sigma^6+6\mu_3^2-8\sigma^2\mu_4}{64}.
\end{aligned}
\]
\end{proof}

\begin{theorem}[Coefficient identification through order six]%
\label{thm:poisson-kl-sixth-coefficient}
Let \(\nu\) have mean \(\bar{\gamma}\), finite variance
\(\sigma^2\ge0\), and finite centred fourth absolute moment.  Then
\[
D_{\rm KL}(P_t*\nu\|P_t(\cdot-\bar{\gamma}))
=\frac{\sigma^4}{8t^4}
+\frac{\sigma^6+6\mu_3^2-8\sigma^2\mu_4}{64t^6}
+o(t^{-6}).
\]
When \(\sigma^2=0\), this is the trivial exact identity for the point mass
\(\delta_{\bar{\gamma}}\).  Positive variance is not needed for the raw
coefficient identity; it enters only in the normalized defect statement
below, where \(\rho_\sigma\) and powers of \(\sigma^{-1}\) are used.
\end{theorem}

\begin{proof}
If \(\sigma^2=0\), then \(\nu=\delta_{\bar{\gamma}}\) and both sides vanish.
Otherwise apply Theorem~\ref{thm:poisson-kl-even-orders} with \(N=3\), and
use the finite Laurent substitution in
Lemma~\ref{lem:sixth-order-coefficient-arithmetic}.
\end{proof}

\subsection{Fourth-moment algebra behind the sixth-order defect}
\label{subsec:coefficient-defect-coupling}
\label{subsec:fourth-moment-stability}

The natural statement is first a normalized fourth-moment algebra lemma,
independent of Poisson kernels, entropy, and finite-time KL values.  This
keeps the terminology close to the fourth-moment-theorem literature, for
example Nourdin--Peccati--Swan
\cite{NourdinPeccatiSwan2014EntropyFourthMoment}, while separating the
present result from that theory: no Wiener-chaos limit theorem, entropy CLT,
or functional-inequality stability theorem is used here.  Only after this
normalized algebra is recorded do we insert the identified \(A_6\) coefficient
from Theorem~\ref{thm:poisson-kl-sixth-coefficient}.  The order-eight
coefficient \(A_8\), the seventh-moment remainder, and any finite-time KL
post-processing are outside the statement below.

\begin{lemma}[Normalized fourth-moment defect algebra]%
\label{lem:normalized-fourth-moment-defect-algebra}
Let \(X\) be a real random variable with \(\EE X=0\), \(\EE X^2=1\), and
\(\EE X^4<\infty\).  Put
\[
\beta_3=\EE X^3,\qquad
Q_2=X^2-1-\beta_3X,\qquad
\kappa=\EE Q_2^2,\qquad
\eta=\frac{\beta_3^2}{32}+\frac{\kappa}{8}.
\]
Then
\begin{equation}\label{eq:normalized-fourth-moment-algebra}
\eta
=\frac{\beta_3^2}{32}+\frac{\EE(X^2-1-\beta_3X)^2}{8}
=\frac{\EE X^4-1}{8}-\frac{3\beta_3^2}{32}.
\end{equation}
In particular \(\eta\ge0\).  Moreover \(\eta=0\) if and only if
\(\mathcal L(X)=\rho_1:=\frac12(\delta_{-1}+\delta_1)\).
\end{lemma}

\begin{proof}
Expanding \(Q_2^2\) and using \(\EE X=0\), \(\EE X^2=1\), and
\(\EE X^3=\beta_3\) gives
\[
\kappa
=\EE[(X^2-1-\beta_3X)^2]
=\EE[(X^2-1)^2]-\beta_3^2
=\EE X^4-1-\beta_3^2.
\]
Substitution gives \eqref{eq:normalized-fourth-moment-algebra}.  The first
displayed form of \(\eta\) proves non-negativity.  If \(\eta=0\), then
\(\beta_3=0\) and \(Q_2=0\) almost surely, so \(X^2=1\) almost surely; since
\(\EE X=0\), the two signs have equal mass.  The converse is immediate.
\end{proof}

\begin{lemma}[Elementary normalized \(W_2\) coupling]%
\label{lem:normalized-fourth-moment-w2-coupling}
Let \(X\) be a real random variable with \(\EE X=0\), \(\EE X^2=1\), and
\(\EE X^4<\infty\).  Put
\[
\beta_3=\EE X^3,\qquad
Q_2=X^2-1-\beta_3X,\qquad
\kappa=\EE Q_2^2,\qquad
\eta=\frac{\beta_3^2}{32}+\frac{\kappa}{8}.
\]
For \(\rho_1=\frac12(\delta_{-1}+\delta_1)\),
\begin{equation}\label{eq:normalized-sign-rademacher-coupling}
W_2(\mathcal L(X),\rho_1)
\le 4\eta^{1/4}+8\eta^{1/2}.
\end{equation}
\end{lemma}

\begin{proof}
First record the defect size which is seen by the sign map.  Expanding
\(Q_2^2\) and using \(\EE X=0\), \(\EE X^2=1\), and
\(\EE X^3=\beta_3\) gives
\[
\kappa
=\EE[(X^2-1-\beta_3X)^2]
=\EE[(X^2-1)^2]-\beta_3^2.
\]
Thus, with
\[
M:=\EE[(X^2-1)^2]=\kappa+\beta_3^2,
\]
one has
\begin{equation}\label{eq:sign-defect-size-bound}
M\le 32\eta .
\end{equation}

Let \(S=1\) on \(\{X\ge0\}\) and \(S=-1\) on \(\{X<0\}\), and let
\(r_p=\mathcal L(S)=p\delta_1+(1-p)\delta_{-1}\).  For every real \(x\),
with this convention at \(x=0\),
\[
|x-\operatorname{sgn}x|\le |x^2-1|;
\]
indeed \(|x^2-1|=|x-1|(x+1)\) for \(x\ge0\), and
\(|x^2-1|=|x+1|(1-x)\) for \(x<0\).  Coupling \(X\) and \(S\) on the same
probability space therefore gives
\begin{equation}\label{eq:couple-x-to-sign}
W_2(\mathcal L(X),r_p)
\le \bigl(\EE|X-S|^2\bigr)^{1/2}
\le M^{1/2}.
\end{equation}

It remains to couple the biased sign law to the symmetric Rademacher law.
On the two-point set \(\{-1,1\}\) the optimal quadratic transport moves the
excess mass \(|p-\frac12|\) across distance \(2\), hence
\[
W_2(r_p,\rho_1)^2=4\left|p-\frac12\right|.
\]
Since \(\EE X=0\),
\[
2p-1=\EE S=\EE(S-X),
\]
and Cauchy--Schwarz together with \eqref{eq:couple-x-to-sign} gives
\[
\left|p-\frac12\right|
\le \frac12\EE|S-X|
\le \frac12 M^{1/2}.
\]
Consequently
\begin{equation}\label{eq:couple-biased-sign-to-symmetric}
W_2(r_p,\rho_1)\le \sqrt2\,M^{1/4}.
\end{equation}
The triangle inequality, followed by \eqref{eq:sign-defect-size-bound}, now
yields
\[
W_2(\mathcal L(X),\rho_1)
\le M^{1/2}+\sqrt2\,M^{1/4}
\le \sqrt{32}\,\eta^{1/2}+\sqrt2\,32^{1/4}\eta^{1/4}.
\]
Finally \(\sqrt{32}<8\) and \(\sqrt2\,32^{1/4}=2^{7/4}<4\), proving
\eqref{eq:normalized-sign-rademacher-coupling}.
\end{proof}

We use the following transport notation in the scaled corollary.  The centred
law is
\[
\nu_c:=(\gamma-\bar{\gamma})_\#\nu .
\]
For probability measures \(\mu,\rho\) on \(\RR\) with finite second moments,
\[
W_2^2(\mu,\rho)
:=\inf_{\pi\in\Pi(\mu,\rho)}
\int_{\RR\times\RR}|x-y|^2\,\dd\pi(x,y),
\]
where \(\Pi(\mu,\rho)\) is the set of couplings of \(\mu\) and \(\rho\).  If
\(S_a(x)=ax\), then pushing a coupling forward by \(S_a\times S_a\), and then
applying the same argument to \(a^{-1}\) when \(a\ne0\), gives the dilation
identity
\[
W_2\bigl((S_a)_\#\mu,(S_a)_\#\rho\bigr)=|a|\,W_2(\mu,\rho).
\]
In particular, with \(X=(\gamma-\bar{\gamma})/\sigma\),
\(\nu_c=(S_\sigma)_\#\mathcal L(X)\) and
\(\rho_\sigma=(S_\sigma)_\#\rho_1\).

\begin{corollary}[Scaled sixth-order coefficient consequence]%
\label{cor:coefficient-defect-coupling}
Let \(\nu\) have mean \(\bar{\gamma}\), variance \(\sigma^2>0\), and finite
centred fourth absolute moment, and write
\(\mu_k=\EE(\gamma-\bar{\gamma})^k\) for \(k=3,4\).  Let
\[
A_6(\nu)=\frac{\sigma^6+6\mu_3^2-8\sigma^2\mu_4}{64}
\]
be the \(t^{-6}\) entropy coefficient identified in
Theorem~\ref{thm:poisson-kl-sixth-coefficient}, and define the scaled defect
from the symmetric two-point coefficient by
\begin{equation}\label{eq:coefficient-defect-normalized-algebra}
\Delta_6(\nu)
:=-\frac{7}{64}\sigma^6-A_6(\nu).
\end{equation}
For \(X=(\gamma-\bar{\gamma})/\sigma\), \(\beta_3=\EE X^3\), and
\(Q_2=X^2-1-\beta_3X\), the normalized fourth-moment lemma gives the
algebraic identity
\begin{equation}\label{eq:scaled-fourth-moment-defect}
\Delta_6(\nu)
=\sigma^6\left(
\frac{\beta_3^2}{32}+\frac{\EE Q_2^2}{8}
\right).
\end{equation}
Consequently, for the centred law \(\nu_c\) and
\(\rho_\sigma=\frac12(\delta_{-\sigma}+\delta_\sigma)\),
\[
W_2(\nu_c,\rho_\sigma)
\le
4\,\sigma^{-1/2}\Delta_6(\nu)^{1/4}
+8\,\sigma^{-2}\Delta_6(\nu)^{1/2}.
\]
Moreover \(\Delta_6(\nu)\ge0\), and equality holds if and only if
\(\nu_c=\rho_\sigma\).

This is a scaled corollary of the normalized fourth-moment algebra and the
elementary \(W_2\) coupling above.  It is not a finite-time KL stability
statement: a numerical value of
\(D_{\rm KL}(P_t*\nu\|P_t(\cdot-\bar{\gamma}))\) at any finite \(t\) does not
by itself provide the quotient remainder, the \(A_6\) coefficient, or
\(\Delta_6(\nu)\).  The constants are non-sharp, and the assertion uses no
sixth moment, eighth-order coefficient, finite-time KL value, or
post-processing constant.
\end{corollary}

\begin{proof}
With \(X=(\gamma-\bar{\gamma})/\sigma\), \(\beta_3=\EE X^3\),
\(\beta_4=\EE X^4\), and \(Q_2=X^2-1-\beta_3X\), the displayed coefficient
formula gives
\[
\begin{aligned}
\sigma^{-6}\Delta_6(\nu)
&=-\frac7{64}
-\frac{1+6\beta_3^2-8\beta_4}{64}  \\
&=\frac{\beta_4-1}{8}-\frac{3\beta_3^2}{32}
=\frac{\beta_3^2}{32}+\frac{\EE Q_2^2}{8},
\end{aligned}
\]
where the last equality is
Lemma~\ref{lem:normalized-fourth-moment-defect-algebra}.  Thus
\(\Delta_6(\nu)=\sigma^6\eta\).
Because quadratic Wasserstein distance is homogeneous under dilation,
\[
W_2(\nu_c,\rho_\sigma)
=\sigma\,W_2(\mathcal L(X),\rho_1)
\]
and Lemma~\ref{lem:normalized-fourth-moment-w2-coupling} gives
\[
\sigma\,W_2(\mathcal L(X),\rho_1)
\le \sigma\{4\eta^{1/4}+8\eta^{1/2}\},
\]
which is exactly
\[
W_2(\nu_c,\rho_\sigma)
\le
4\,\sigma^{-1/2}\Delta_6(\nu)^{1/4}
+8\,\sigma^{-2}\Delta_6(\nu)^{1/2}.
\]
The constants in this displayed estimate are the deliberately non-sharp
constants from the elementary coupling above.  Non-negativity and the
equality case are exactly the equality statement in
Lemma~\ref{lem:normalized-fourth-moment-defect-algebra}, followed by the
dilation \(x\mapsto\sigma x\).
\end{proof}

\begin{remark}[Relation with functional-inequality stability]
As emphasized in the introduction and in the statement of
Corollary~\ref{cor:coefficient-defect-coupling},
Corollary~\ref{cor:coefficient-defect-coupling} is only a scaled
finite-moment consequence of the normalized algebra
\eqref{eq:normalized-fourth-moment-algebra} and the coefficient
identification of \(A_6\).  It is not an optimizer-stability theorem for a
sharp functional inequality, and no finite-time KL stability estimate follows
from it without separate verified coefficient and remainder information.
\end{remark}
