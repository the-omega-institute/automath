\subsection{Explicit coefficient layer through order eight and coefficient-defect coupling}
\label{subsec:coefficient-defect-coupling}
\label{subsec:fourth-moment-stability}

The Laurent arithmetic below is reproducible from the coefficient-extraction
recipe in Theorem~\ref{thm:finite-fourier-mode-integral-algorithm}.  With
\(q_{n,k}\) defined by the uniform coefficient formula
\eqref{eq:mode-laurent-coefficients},
\[
J(n_1,\ldots,n_r)=[z^0]\prod_{j=1}^rQ_{n_j}(z).
\]

The full order-eight Laurent certificate is kept in one place, Appendix
\ref{app:order-eight-laurent-certificate}.  It displays
\(Q_2,\ldots,Q_6\), the deterministic zero-sum recurrence, the complete
constant-term table, and hand checks for the sign-sensitive rows.  The only
main-text arithmetic needed for orientation is the leading coefficient:
\[
Q_2(z)=-\frac14\bigl(z+z^{-1}+z^2+z^{-2}\bigr),
\qquad
J(2,2)=[z^0]Q_2^2=4\cdot\frac1{16}=\frac14.
\]
Hence \(A_4=\frac12\mu_2^2J(2,2)=\mu_2^2/8=\sigma^4/8\), matching the
coefficient used below.

\begin{lemma}[Laurent recurrence and coefficients through order eight]%
\label{lem:eighth-order-coefficient-arithmetic}
If \(Q_n(z)=\sum_k q_{n,k}z^k\), the constant term
\(J(\mathbf n)=[z^0]\prod_{j=1}^r Q_{n_j}(z)\) is obtained by the finite
recurrence
\begin{equation}\label{eq:eighth-order-constant-term-recurrence}
C_0(s)=\mathbf 1_{\{s=0\}},\qquad
C_j(s)=\sum_{0<|k|\le n_j}q_{n_j,k}\,C_{j-1}(s-k),
\quad J(\mathbf n)=C_r(0).
\end{equation}
This is the specialization, through total order eight, of the deterministic
exact recurrence in
Lemma~\ref{lem:appendix-d-deterministic-laurent-recurrence}; it returns
rational constant terms for arbitrary requested rows.
Applying this recurrence to the rows entering \(A_4,A_6,A_8\) gives
\[
A_4=\frac{\sigma^4}{8},\qquad
A_6=\frac{\sigma^6+6\mu_3^2-8\sigma^2\mu_4}{64},
\]
and
\[
A_8=
\frac{3\sigma^8-12\sigma^4\mu_4+9\sigma^2\mu_3^2
+12\sigma^2\mu_6-30\mu_3\mu_5+20\mu_4^2}{256}.
\]
The row-by-row finite certificate used for these substitutions is displayed
in Appendix~\ref{app:order-eight-laurent-certificate}.
\end{lemma}

\begin{proof}
Apply the constant-term rule
\eqref{eq:mode-integral-constant-term} to the Laurent representatives
defined by \eqref{eq:mode-laurent-polynomial-definition} and
\eqref{eq:mode-laurent-coefficients}.  Multiplying the Laurent polynomials
one factor at a time gives exactly
\eqref{eq:eighth-order-constant-term-recurrence}; this is finite because
\(q_{n,k}=0\) unless \(0<|k|\le n\).  Combining the recurrence values with
the universal coefficient formula
\eqref{eq:universal-entropy-coefficient} gives the coefficient rows
\[
\begin{aligned}
A_4&=\frac12\mu_2^2J(2,2)=\frac{\sigma^4}{8},\\
A_6&=\mu_2\mu_4J(2,4)+\frac12\mu_3^2J(3,3)
-\frac16\mu_2^3J(2,2,2)
=\frac{\sigma^6+6\mu_3^2-8\sigma^2\mu_4}{64},\\
A_8&=\mu_2\mu_6J(2,6)+\mu_3\mu_5J(3,5)
+\frac12\mu_4^2J(4,4)
-\frac12\mu_2^2\mu_4J(2,2,4)\\
&\hspace{2.5em}
-\frac12\mu_2\mu_3^2J(2,3,3)
+\frac1{12}\mu_2^4J(2,2,2,2)\\
&=\frac{3\sigma^8-12\sigma^4\mu_4+9\sigma^2\mu_3^2
+12\sigma^2\mu_6-30\mu_3\mu_5+20\mu_4^2}{256}.
\end{aligned}
\]
Substituting the constant terms displayed in
Appendix~\ref{app:order-eight-laurent-certificate} gives the rational
polynomials directly:
\[
\frac12\sigma^4\cdot\frac14=\frac{\sigma^4}{8},
\]
\[
\sigma^2\mu_4\Bigl(-\frac18\Bigr)
+\frac12\mu_3^2\Bigl(\frac3{16}\Bigr)
-\frac16\sigma^6\Bigl(-\frac3{32}\Bigr)
=\frac{\sigma^6+6\mu_3^2-8\sigma^2\mu_4}{64},
\]
and
\[
\begin{aligned}
&\sigma^2\mu_6\Bigl(\frac3{64}\Bigr)
+\mu_3\mu_5\Bigl(-\frac{15}{128}\Bigr)
+\frac12\mu_4^2\Bigl(\frac5{32}\Bigr)\\
&\quad
-\frac12\sigma^4\mu_4\Bigl(\frac3{32}\Bigr)
-\frac12\sigma^2\mu_3^2\Bigl(-\frac9{128}\Bigr)
+\frac1{12}\sigma^8\Bigl(\frac9{64}\Bigr)\\
&=\frac{3\sigma^8-12\sigma^4\mu_4+9\sigma^2\mu_3^2
+12\sigma^2\mu_6-30\mu_3\mu_5+20\mu_4^2}{256}.
\end{aligned}
\]
The parity identity \(u_n(-y)=(-1)^nu_n(y)\) separately eliminates the odd
total-order products through order eight.
\end{proof}

\begin{lemma}[Substitution into \(A_4,A_6,A_8\)]%
\label{lem:n4-entropy-coefficient-substitution}
For \(\mu_2=\sigma^2\),
\[
A_4=\frac{\sigma^4}{8},
\qquad
A_6=\frac{\sigma^6+6\mu_3^2-8\sigma^2\mu_4}{64},
\]
and
\[
A_8=
\frac{3\sigma^8-12\sigma^4\mu_4+9\sigma^2\mu_3^2
+12\sigma^2\mu_6-30\mu_3\mu_5+20\mu_4^2}{256}.
\]
\end{lemma}

\begin{proof}
Put \(d_t=t^{-2}a_2+\cdots+t^{-6}a_6\), \(a_j=\mu_ju_j\), and use
\[
\Phi(s)=\frac12s^2-\frac16s^3+\frac1{12}s^4+O(s^5).
\]
The \(t^{-4},t^{-6},t^{-8}\) polynomial layers are
\[
\frac12a_2^2,\qquad
a_2a_4+\frac12a_3^2-\frac16a_2^3,
\]
and
\[
a_2a_6+a_3a_5+\frac12a_4^2
-\frac12a_2^2a_4-\frac12a_2a_3^2+\frac1{12}a_2^4.
\]
Substitute the constant terms from
Lemma~\ref{lem:eighth-order-coefficient-arithmetic}.
\end{proof}

\noindent
The exact constant terms entering the coefficient-identification theorem are
not repeated here.  Appendix~\ref{app:order-eight-laurent-certificate}
contains the single row table for \(A_4,A_6,A_8\), including
\[
J(2,6)=\frac3{64},\qquad
J(3,5)=-\frac{15}{128},\qquad
J(2,3,3)=-\frac9{128},
\]
with hand verification of one even row and the two sign-sensitive odd rows.

\begin{theorem}[Self-contained coefficient identification through order eight]%
\label{thm:poisson-kl-eighth}
Let \(\nu\) have mean \(\bar{\gamma}\) and finite variance
\(\sigma^2\ge0\).
If \(\nu\) has finite centred fourth absolute moment, then
\[
D_{\rm KL}(P_t*\nu\|P_t(\cdot-\bar{\gamma}))
=\frac{\sigma^4}{8t^4}
+\frac{\sigma^6+6\mu_3^2-8\sigma^2\mu_4}{64t^6}
+o(t^{-6}).
\]
If \(\nu\) has finite centred sixth absolute moment, then
\begin{equation}\label{eq:poisson-kl-eighth}
D_{\rm KL}(P_t*\nu\|P_t(\cdot-\bar{\gamma}))
=\frac{\sigma^4}{8t^4}+\frac{A_6(\nu)}{t^6}
+\frac{A_8(\nu)}{t^8}+o(t^{-8}),
\end{equation}
where \(A_6,A_8\) are the polynomials in
Lemma~\ref{lem:n4-entropy-coefficient-substitution}.
When \(\sigma^2=0\), these are the trivial exact identities for the point
mass \(\delta_{\bar{\gamma}}\).  Positive variance is not a hypothesis for raw
coefficient identification; it is imposed only in normalized defect
statements, such as Corollary~\ref{cor:coefficient-defect-coupling}, where
\(\rho_\sigma\) and powers of \(\sigma^{-1}\) are used.
No eighth-order defect variable is defined or used in this theorem: \(A_8\)
is identified only as the \(t^{-8}\) entropy coefficient.  The sole
coefficient-defect consequence in this subsection is the sixth-order
\(\Delta_6\) coupling in Corollary~\ref{cor:coefficient-defect-coupling}.
\end{theorem}

\begin{proof}
If \(\sigma^2=0\), then \(\nu=\delta_{\bar{\gamma}}\),
\(P_t*\nu=P_t(\cdot-\bar{\gamma})\), and every centred moment in the displayed
polynomials is zero, so both formulas are exact.  Assume now
\(\sigma^2>0\).
Use Theorem~\ref{thm:poisson-kl-even-orders} with \(N=3\) for the sixth
coefficient and \(N=4\) for the eighth coefficient.  The finite Laurent
substitution is Lemma~\ref{lem:n4-entropy-coefficient-substitution}.
\end{proof}

\begin{theorem}[Quantitative \(N=4\) remainder under the seventh moment]%
\label{thm:poisson-kl-eighth-seventh-remainder}
If, in addition, the centred seventh absolute moment is finite, then
\[
D_{\rm KL}(P_t*\nu\|P_t(\cdot-\bar{\gamma}))
=\frac{\sigma^4}{8t^4}+\frac{A_6(\nu)}{t^6}
+\frac{A_8(\nu)}{t^8}+O_\nu(t^{-9}).
\]
\end{theorem}

\begin{proof}
Apply Lemma~\ref{lem:positive-margin-quotient-remainder} with
\(N=4\), \(L=6\), and \(\eta=1\).  The seventh absolute moment gives the
quotient expansion
\[
\delta_t(y)=\sum_{n=2}^{6}\mu_nu_n(y)t^{-n}+\rho_{6,t}(y),
\qquad
\|\rho_{6,t}\|_\infty+\|\rho_{6,t}\|_{L^1(\omega)}=O_\nu(t^{-7}),
\]
and \(\|\delta_t\|_\infty=O_\nu(t^{-2})\).  In the entropy expansion
\(\Phi(s)=(1+s)\log(1+s)-s\), every term containing \(\rho_{6,t}\) also
contains at least one further density factor at the quadratic entropy scale;
therefore its \(L^1(\omega)\) contribution is \(O_\nu(t^{-2}t^{-7})\).
The Taylor tail after degree \(4\) is \(O_\nu(t^{-10})\).  Hence the retained
pure-mode coefficients \(A_4,A_6,A_8\), identified in
Theorem~\ref{thm:poisson-kl-eighth}, have a total remainder
\(O_\nu(t^{-9})\).
\end{proof}

\begin{unnumberedlemma}[Normalized sign--Rademacher coupling]
Let \(X\) be a real random variable with \(\EE X=0\), \(\EE X^2=1\), and
\(\EE X^4<\infty\).  Put
\[
\beta_3=\EE X^3,\qquad
Q_2=X^2-1-\beta_3X,\qquad
\kappa=\EE Q_2^2,\qquad
\eta=\frac{\beta_3^2}{32}+\frac{\kappa}{8}.
\]
For \(\rho_1=\frac12(\delta_{-1}+\delta_1)\),
\begin{equation}\label{eq:normalized-sign-rademacher-coupling}
W_2(\mathcal L(X),\rho_1)
\le 4\eta^{1/4}+8\eta^{1/2}.
\end{equation}
\end{unnumberedlemma}

\begin{proof}
First record the defect size which is seen by the sign map.  Expanding
\(Q_2^2\) and using \(\EE X=0\), \(\EE X^2=1\), and
\(\EE X^3=\beta_3\) gives
\[
\kappa
=\EE[(X^2-1-\beta_3X)^2]
=\EE[(X^2-1)^2]-\beta_3^2.
\]
Thus, with
\[
M:=\EE[(X^2-1)^2]=\kappa+\beta_3^2,
\]
one has
\begin{equation}\label{eq:sign-defect-size-bound}
M\le 32\eta .
\end{equation}

Let \(S=1\) on \(\{X\ge0\}\) and \(S=-1\) on \(\{X<0\}\), and let
\(r_p=\mathcal L(S)=p\delta_1+(1-p)\delta_{-1}\).  For every real \(x\),
with this convention at \(x=0\),
\[
|x-\operatorname{sgn}x|\le |x^2-1|;
\]
indeed \(|x^2-1|=|x-1|(x+1)\) for \(x\ge0\), and
\(|x^2-1|=|x+1|(1-x)\) for \(x<0\).  Coupling \(X\) and \(S\) on the same
probability space therefore gives
\begin{equation}\label{eq:couple-x-to-sign}
W_2(\mathcal L(X),r_p)
\le \bigl(\EE|X-S|^2\bigr)^{1/2}
\le M^{1/2}.
\end{equation}

It remains to couple the biased sign law to the symmetric Rademacher law.
On the two-point set \(\{-1,1\}\) the optimal quadratic transport moves the
excess mass \(|p-\frac12|\) across distance \(2\), hence
\[
W_2(r_p,\rho_1)^2=4\left|p-\frac12\right|.
\]
Since \(\EE X=0\),
\[
2p-1=\EE S=\EE(S-X),
\]
and Cauchy--Schwarz together with \eqref{eq:couple-x-to-sign} gives
\[
\left|p-\frac12\right|
\le \frac12\EE|S-X|
\le \frac12 M^{1/2}.
\]
Consequently
\begin{equation}\label{eq:couple-biased-sign-to-symmetric}
W_2(r_p,\rho_1)\le \sqrt2\,M^{1/4}.
\end{equation}
The triangle inequality, followed by \eqref{eq:sign-defect-size-bound}, now
yields
\[
W_2(\mathcal L(X),\rho_1)
\le M^{1/2}+\sqrt2\,M^{1/4}
\le \sqrt{32}\,\eta^{1/2}+\sqrt2\,32^{1/4}\eta^{1/4}.
\]
Finally \(\sqrt{32}<8\) and \(\sqrt2\,32^{1/4}=2^{7/4}<4\), proving
\eqref{eq:normalized-sign-rademacher-coupling}.
\end{proof}

For the coefficient-defect statement below, assume \(\nu\) has mean
\(\bar{\gamma}\), variance \(\sigma^2>0\), and finite centred fourth absolute
moment, and put \(X=(\gamma-\bar{\gamma})/\sigma\).  With
\[
\beta_3=\EE X^3,\qquad
Q_2=X^2-1-\beta_3X,
\]
the sixth-order coefficient defect attached to \(A_6(\nu)\) is
\begin{equation}\label{eq:coefficient-defect-normalized-algebra}
\Delta_6(\nu)
:=-\frac{7}{64}\sigma^6
-\frac{\sigma^6+6\mu_3^2-8\sigma^2\mu_4}{64}
=\sigma^6\left(
\frac{\beta_3^2}{32}+\frac{\|Q_2\|_2^2}{8}
\right).
\end{equation}
Here \(\|Q_2\|_2^2=\EE_\nu Q_2^2\), with \(Q_2\) evaluated on the normalized
centred variable \(X\).
This normalized identity is the algebraic input for the following
moment-coupling statement.

We use the following transport notation in the corollary.  The centred law is
\[
\nu_c:=(\gamma-\bar{\gamma})_\#\nu .
\]
For probability measures \(\mu,\rho\) on \(\RR\) with finite second moments,
\[
W_2^2(\mu,\rho)
:=\inf_{\pi\in\Pi(\mu,\rho)}
\int_{\RR\times\RR}|x-y|^2\,\dd\pi(x,y),
\]
where \(\Pi(\mu,\rho)\) is the set of couplings of \(\mu\) and \(\rho\).  If
\(S_a(x)=ax\), then pushing a coupling forward by \(S_a\times S_a\), and then
applying the same argument to \(a^{-1}\) when \(a\ne0\), gives the dilation
identity
\[
W_2\bigl((S_a)_\#\mu,(S_a)_\#\rho\bigr)=|a|\,W_2(\mu,\rho).
\]
In particular, with \(X=(\gamma-\bar{\gamma})/\sigma\),
\(\nu_c=(S_\sigma)_\#\mathcal L(X)\) and
\(\rho_\sigma=(S_\sigma)_\#\rho_1\).

The following corollary is the section-level proof of
Theorem~\ref{thm:intro-main-coefficient-defect}.

\begin{corollary}[Coefficient-defect coupling]%
\label{cor:coefficient-defect-coupling}
Let \(\nu\) have mean \(\bar{\gamma}\), variance \(\sigma^2>0\), and finite
centred fourth absolute moment.  Let \(\Delta_6(\nu)\) be the coefficient
defect in \eqref{eq:coefficient-defect-normalized-algebra}.
For the centred law \(\nu_c\) and
\(\rho_\sigma=\frac12(\delta_{-\sigma}+\delta_\sigma)\),
\[
W_2(\nu_c,\rho_\sigma)
\le
4\,\sigma^{-1/2}\Delta_6(\nu)^{1/4}
+8\,\sigma^{-2}\Delta_6(\nu)^{1/2}.
\]
Moreover \(\Delta_6(\nu)\ge0\), and equality holds if and only if
\(\nu_c=\rho_\sigma\).  The assertion is a moment-coupling consequence of the
identified \(A_6\) coefficient.  The constants are non-sharp and chosen for a
one-line moment coupling.  The assertion uses no sixth moment, eighth-order
coefficient, finite-time KL value, or post-processing constant.
\end{corollary}

\begin{proof}
With \(X=(\gamma-\bar{\gamma})/\sigma\), \(\beta_3=\EE X^3\),
\(\beta_4=\EE X^4\), and \(Q_2=X^2-1-\beta_3X\),
\[
\|Q_2\|_2^2
=\EE[(X^2-1-\beta_3X)^2]
=\beta_4-1-\beta_3^2 .
\]
Together with \eqref{eq:coefficient-defect-normalized-algebra}, this gives
\(\Delta_6(\nu)=\sigma^6\eta\) for the normalized defect \(\eta\) above.
Because quadratic Wasserstein distance is homogeneous under dilation,
\[
W_2(\nu_c,\rho_\sigma)
=\sigma\,W_2(\mathcal L(X),\rho_1)
\]
and the preceding normalized sign--Rademacher coupling lemma gives
\[
\sigma\,W_2(\mathcal L(X),\rho_1)
\le \sigma\{4\eta^{1/4}+8\eta^{1/2}\},
\]
which is exactly
\[
W_2(\nu_c,\rho_\sigma)
\le
4\,\sigma^{-1/2}\Delta_6(\nu)^{1/4}
+8\,\sigma^{-2}\Delta_6(\nu)^{1/2}.
\]
The constants in this displayed estimate are the deliberately non-sharp
constants from the elementary one-line moment coupling above.
The same identity gives \(\Delta_6\ge0\).  If \(\Delta_6=0\), then
\(\beta_3=0\) and \(Q_2=0\) almost surely, so \(X^2=1\) almost surely; with
\(\EE X=0\), this forces \(X=\pm1\) with equal probabilities.  Hence
\(\nu_c=\rho_\sigma\).  The converse follows by direct substitution.
\end{proof}

\begin{remark}[Relation with functional-inequality stability]
As emphasized in the introduction and in the statement of
Corollary~\ref{cor:coefficient-defect-coupling},
Corollary~\ref{cor:coefficient-defect-coupling} is only a finite-moment
coupling consequence of the coefficient identity
\eqref{eq:coefficient-defect-normalized-algebra}.  It is not an
optimizer-stability theorem for a sharp functional inequality.
\end{remark}
