\begin{unnumberedlemma}[Two-background KL perturbation]
Let \((E,\Omega)\) be a probability space and let
\(\epsilon_s\downarrow0\).  For \(i=1,2\), let \(A_{i,s}\) be probability
densities such that, for one constant \(c>0\) and all sufficiently large
\(s\),
\begin{equation}\label{eq:abstract-two-background-hypotheses}
 A_{i,s}\ge c,\qquad
 \|A_{i,s}-1\|_1=o(1),\qquad
 \|A_{1,s}-A_{2,s}\|_1=o(\epsilon_s).
\end{equation}
Let \(U_{i,s}=\epsilon_sB_{i,s}\), where
\begin{equation}\label{eq:abstract-two-background-modes}
 \sup_s\bigl(\|B_{1,s}\|_\infty+\|B_{2,s}\|_\infty\bigr)<\infty,
 \qquad \int B_{i,s}\dd\Omega=0.
\end{equation}
Suppose
\[
 q_{i,s}=A_{i,s}+U_{i,s}+R_{i,s}
\]
are probability densities, uniformly bounded below by a positive constant,
and
\begin{equation}\label{eq:abstract-two-background-remainders}
 \|R_{i,s}\|_\infty+\|R_{i,s}\|_1=o(\epsilon_s).
\end{equation}
Assume also that
\(\int A_{1,s}\log^+A_{1,s}\dd\Omega<\infty\) for every sufficiently
large \(s\).  Then all relative entropies below are finite and
\begin{equation}\label{eq:abstract-two-background-conclusion}
 D_{\rm KL}(q_{1,s}\Omega\|q_{2,s}\Omega)
 =D_{\rm KL}(A_{1,s}\Omega\|A_{2,s}\Omega)
 +\frac{\epsilon_s^2}{2}
   \int(B_{1,s}-B_{2,s})^2\dd\Omega+o(\epsilon_s^2).
\end{equation}
\end{unnumberedlemma}

\begin{proof}
Suppress \(s\), write \(P_i=A_i+U_i\), and put
\[
 f(a,b)=a\log(a/b)-a+b.
\]
For large \(s\), \(P_i\ge c/2\).  Since both \(P_i\) and \(q_i\) are
bounded below, the segments \(P_i+tR_i\), \(0\le t\le1\), have a common
positive lower bound.  On any such region,
\begin{equation}\label{eq:abstract-two-background-first-partials}
 f_a(a,b)=\log(a/b),\qquad f_b(a,b)=1-a/b,
 \qquad |f_a(a,b)|+|f_b(a,b)|\le C|a-b|.
\end{equation}
The mean-value theorem and
\[
 |(P_1+tR_1)-(P_2+tR_2)|
 \le |A_1-A_2|+|U_1-U_2|+|R_1|+|R_2|
\]
therefore give the explicit remainder-removal bound
\begin{align}
 &\left|\int f(q_1,q_2)\dd\Omega-
              \int f(P_1,P_2)\dd\Omega\right|\notag\\
 &\quad\le
 C(\|R_1\|_\infty+\|R_2\|_\infty)\|A_1-A_2\|_1\notag\\
 &\qquad+C(\|U_1\|_\infty+\|U_2\|_\infty)
          (\|R_1\|_1+\|R_2\|_1)\notag\\
 &\qquad+C(\|R_1\|_\infty+\|R_2\|_\infty)
          (\|R_1\|_1+\|R_2\|_1)
 =o(\epsilon_s^2).
 \label{eq:abstract-two-background-remove-remainders}
\end{align}

It remains to expand the two positive backgrounds.  Set
\[
 a_t=A_1+tU_1,\qquad b_t=A_2+tU_2,
 \qquad H(t)=\int f(a_t,b_t)\dd\Omega.
\]
The zero-mean condition in \eqref{eq:abstract-two-background-modes} makes
\(a_t\) and \(b_t\) probability densities.  The four quantities used in
Taylor's formula are, explicitly,
\begin{align}
 H(0)&=D_{\rm KL}(A_1\Omega\|A_2\Omega),
 \label{eq:abstract-two-background-H-zero}\\
 H'(t)&=\int\left\{U_1\log\frac{a_t}{b_t}
            +U_2\left(1-\frac{a_t}{b_t}\right)\right\}\dd\Omega,
 \label{eq:abstract-two-background-H-first}\\
 H''(t)&=\int\left\{\frac{U_1^2}{a_t}
       -\frac{2U_1U_2}{b_t}+\frac{a_tU_2^2}{b_t^2}\right\}\dd\Omega,
 \label{eq:abstract-two-background-H-second}\\
 H'''(t)&=\int\left\{-\frac{U_1^3}{a_t^2}
       +\frac{3U_1U_2^2}{b_t^2}
       -\frac{2a_tU_2^3}{b_t^3}\right\}\dd\Omega.
 \label{eq:abstract-two-background-H-third}
\end{align}
These formulas follow by direct differentiation; the estimates below also
provide an integrable majorant and hence justify differentiation under the
integral.

At \(t=0\), the two coefficients in \eqref{eq:abstract-two-background-H-first}
are bounded by \(C|A_1-A_2|\).  Thus
\begin{equation}\label{eq:abstract-two-background-first-variation}
 |H'(0)|\le
 C(\|U_1\|_\infty+\|U_2\|_\infty)\|A_1-A_2\|_1
 =o(\epsilon_s^2).
\end{equation}
For the Hessian, the lower bound in
\eqref{eq:abstract-two-background-hypotheses} gives
\begin{align}
 \left\|\frac1{A_i}-1\right\|_1
 &\le c^{-1}\|A_i-1\|_1=o(1),
 \label{eq:abstract-two-background-inverse-coefficient}\\
 \left\|\frac{A_1}{A_2^2}-1\right\|_1
 &\le c^{-2}\|A_1-A_2\|_1
      +c^{-1}\|A_2-1\|_1=o(1).
 \label{eq:abstract-two-background-third-coefficient}
\end{align}
Indeed, the second line uses
\(A_1/A_2^2-1=(A_1-A_2)/A_2^2+(1/A_2-1)\).
Since \(\|U_i\|_\infty=O(\epsilon_s)\), these coefficient estimates in
\eqref{eq:abstract-two-background-H-second} yield
\begin{equation}\label{eq:abstract-two-background-hessian-limit}
 H''(0)=\int(U_1-U_2)^2\dd\Omega+o(\epsilon_s^2)
 =\epsilon_s^2\int(B_{1,s}-B_{2,s})^2\dd\Omega
  +o(\epsilon_s^2).
\end{equation}

Finally, \(a_t,b_t\) are uniformly bounded below and
\(|U_1|+|U_2|\le C\epsilon_s\).  Formula
\eqref{eq:abstract-two-background-H-third} consequently gives
\[
 \left|\frac{\dd^3}{\dd t^3}f(a_t,b_t)\right|
 \le C\epsilon_s^3(1+a_t).
\]
Because \(\int a_t\dd\Omega=1\),
\begin{equation}\label{eq:abstract-two-background-cubic-bound}
 \sup_{0\le t\le1}|H'''(t)|\le C\epsilon_s^3
 =o(\epsilon_s^2).
\end{equation}
Taylor's theorem, \eqref{eq:abstract-two-background-first-variation},
\eqref{eq:abstract-two-background-hessian-limit}, and
\eqref{eq:abstract-two-background-cubic-bound} expand \(H(1)\).  Combining
that expansion with
\eqref{eq:abstract-two-background-remove-remainders} proves
\eqref{eq:abstract-two-background-conclusion}.  Finiteness follows from the
assumed positive lower bounds, the finite positive entropy of \(A_1\), and
the fact that \(U_i,R_i\) are bounded perturbations; the negative part of
\(a\log(a/b)\) is bounded in absolute integral by
\(e^{-1}\int b\dd\Omega=e^{-1}\).
\end{proof}
