\section*{Supplementary Note: Finite Phase Sampling Checks for Cayley Coefficients}%
\label{sec:finite-phase-sampling}

This supplementary note records the finite-grid interface used only as a
verification and optional data-input layer.  It is not included in the main
paper.  The entropy asymptotics in
Section~\ref{sec:entropy-asymptotics} do not depend on finite grids: they use
the quotient expansion, finite-moment transfer, and Laurent constant-term
extraction in
Theorem~\ref{thm:finite-fourier-mode-integral-algorithm}.  The statements
below say that the same Laurent constants can be recovered exactly from
root-of-unity grids whenever the Fourier support lies below the aliasing
barrier.  All routine eighth-order arithmetic is therefore kept in the main
coefficient lemmas,
Lemmas~\ref{lem:n4-normalized-laurent-representatives} and
\ref{lem:eighth-order-coefficient-arithmetic}, rather than repeated here.

\begin{definition}[Finite phase grids]\label{def:finite-phase-grid}
For an integer \(N\ge1\), set
\[
\mathcal G_N:=\{z\in\TT:\ z^N=1\}.
\]
\end{definition}

\begin{proposition}[Phase-coordinate interface and character identification]%
\label{prop:phase-coordinate-identification}
Let
\[
z(\theta):=\iota(\tan\theta)=e^{2\ii\theta}
\qquad\bigl(-\pi/2<\theta<\pi/2\bigr).
\]
Then
\[
\cos(2k\theta)=\operatorname{Re} z(\theta)^k,
\qquad
\sin(2k\theta)=\operatorname{Im} z(\theta)^k
\qquad(k\ge1).
\]
If \(K<N\), then normalized averaging over \(\mathcal G_N\) agrees with Haar
averaging on every character \(z^\ell\) with \(|\ell|\le K\):
\[
\frac1N\sum_{\zeta\in\mathcal G_N}\zeta^\ell
=\int_{\TT} z^\ell\,\dd m_{\TT}(z)
=
\begin{cases}
1,& \ell=0,\\
0,& 0<|\ell|\le K.
\end{cases}
\]
\end{proposition}

\begin{proof}
The identity \(z(\theta)=e^{2\ii\theta}\) is the Cayley formula
\eqref{eq:cayley-embed} with \(x=\tan\theta\).  The trigonometric identities
are its real and imaginary parts.  The grid average is the finite geometric
sum; below the scale \(N\), no nonzero exponent is divisible by \(N\).  Haar
averaging has the same values on characters.
\end{proof}

\begin{proposition}[Finite-grid coefficient recovery and aliasing]%
\label{prop:finite-grid-coefficient-recovery}
Let \(f\) be continuous on \(\TT\) with absolutely convergent Fourier series
\[
f(z)=\sum_{m\in\ZZ}\widehat f(m)z^m,
\qquad
\sum_{m\in\ZZ}|\widehat f(m)|<\infty,
\]
where
\[
\widehat f(m):=\int_{\TT}f(z)z^{-m}\,\dd m_{\TT}(z).
\]
For \(N\ge1\) and \(k\in\ZZ\) define
\[
\widehat f_N(k):=\frac1N\sum_{\zeta\in\mathcal G_N}f(\zeta)\zeta^{-k}.
\]
Then
\begin{equation}\label{eq:finite-grid-aliasing}
\widehat f_N(k)=\sum_{q\in\ZZ}\widehat f(k+qN),
\end{equation}
and hence
\begin{equation}\label{eq:finite-grid-aliasing-bound}
\bigl|\widehat f_N(k)-\widehat f(k)\bigr|
\le
\sum_{\substack{q\in\ZZ\\q\ne0}}\bigl|\widehat f(k+qN)\bigr|.
\end{equation}
In particular, if the Fourier support of \(z^{-k}f\) is contained in
\(\{-K,\ldots,K\}\) with \(K<N\), then \(\widehat f_N(k)=\widehat f(k)\).
For a real phase profile \(F(\theta)=f(e^{2\ii\theta})\), the corresponding
cosine and sine coefficient recoveries satisfy
\[
|\mathcal C_{k,N}(F)-\mathcal C_k(F)|
+|\mathcal S_{k,N}(F)-\mathcal S_k(F)|
\le
4\sum_{\substack{q\in\ZZ\\q\ne0}}\bigl|\widehat f(k+qN)\bigr|.
\]
\end{proposition}

\begin{proof}
Termwise averaging gives
\[
\widehat f_N(k)
=\sum_{m\in\ZZ}\widehat f(m)
\left(\frac1N\sum_{\zeta\in\mathcal G_N}\zeta^{m-k}\right).
\]
The inner average is one exactly when \(m-k\in N\ZZ\), proving
\eqref{eq:finite-grid-aliasing}; the bound follows by removing the
\(q=0\) term.  The exact-recovery statement is the case where the remaining
aliased coefficients vanish.  The last estimate is the same bound applied to
the real and imaginary parts of the phase Fourier coefficient.
\end{proof}

\begin{theorem}[Finite-grid finite-time moment recovery]%
\label{thm:finite-grid-poisson-moment-recovery}
Let \(\nu\) satisfy the hypotheses of
Proposition~\ref{prop:poisson-cauchy-moment-inversion-by-phase-modes}, and
let \(F_t\) be the corresponding phase profile.  Let \(L\ge9\) be odd.  For
each \(\zeta\in\mathcal G_L\), let
\(\theta_\zeta\in(-\pi/2,\pi/2)\) be the unique angle with
\(\zeta=e^{2\ii\theta_\zeta}\), and set
\[
\widehat F_{t,L}(k):=\frac1L\sum_{\zeta\in\mathcal G_L}
F_t(\theta_\zeta)\zeta^{-k},
\]
\[
\mathcal C_{k,L}(t):=2\operatorname{Re}\widehat F_{t,L}(k),
\qquad
\mathcal S_{k,L}(t):=-2\operatorname{Im}\widehat F_{t,L}(k).
\]
Then there are constants \(C_\nu,T_\nu>0\) such that, for all
\(t\ge T_\nu\),
\begin{align}
\left|
\mathcal C_{1,L}(t)-\mathcal C_{4,L}(t)
+\frac{\sigma^2}{2t^2}
\right|
&\le C_{\nu}t^{-5},\label{eq:finite-grid-sigma-recovery}\\
\left|
\mathcal S_{3,L}(t)+\frac{\mu_3}{4t^3}
\right|
&\le C_{\nu}t^{-5},\label{eq:finite-grid-mu-three-recovery}\\
\left|
\mathcal C_{4,L}(t)-\frac{\mu_4}{8t^4}
\right|
&\le C_{\nu}t^{-5}.\label{eq:finite-grid-mu-four-recovery}
\end{align}
Consequently
\[
\sigma^2=\lim_{t\to\infty}
-2t^2\bigl(\mathcal C_{1,L}(t)-\mathcal C_{4,L}(t)\bigr),
\quad
\mu_3=\lim_{t\to\infty}-4t^3\mathcal S_{3,L}(t),
\quad
\mu_4=\lim_{t\to\infty}8t^4\mathcal C_{4,L}(t).
\]
\end{theorem}

\begin{proof}
The proof of
Proposition~\ref{prop:poisson-cauchy-moment-inversion-by-phase-modes} gives
\[
F_t(\theta)=
\frac{\sigma^2}{t^2}u_2(\tan\theta)
+\frac{\mu_3}{t^3}u_3(\tan\theta)
+\frac{\mu_4}{t^4}u_4(\tan\theta)+O_\nu(t^{-5})
\]
uniformly in \(\theta\).  The polynomial part has phase Fourier support in
\(\{-4,\ldots,4\}\).  After multiplication by \(z^{-k}\) for
\(k=1,3,4\), the support lies in \(\{-8,\ldots,8\}\), so
Proposition~\ref{prop:finite-grid-coefficient-recovery} is exact for
\(L\ge9\).  Applying the coefficient identities already computed in
Proposition~\ref{prop:poisson-cauchy-moment-inversion-by-phase-modes} and
bounding the grid average of the uniform remainder gives the three displayed
estimates.
\end{proof}

\begin{corollary}[Alias-free Cayley mode-product averaging]%
\label{cor:alias-free-cayley-mode-product-quadrature}
Let \(Q(z)=\sum_{|m|\le K}a_mz^m\) be a Laurent polynomial on \(\TT\).  If
\(K<N\), then
\[
\int_{\TT}Q(z)\,\dd m_{\TT}(z)
=\frac1N\sum_{\zeta\in\mathcal G_N}Q(\zeta)=a_0.
\]
Consequently every Cauchy-weighted integral of a finite product of
Cayley--Chebyshev modes is recovered exactly from \(\mathcal G_N\) once its
phase Laurent support is contained in \(\{-K,\ldots,K\}\) with \(K<N\).
\end{corollary}

\begin{proof}
Apply Proposition~\ref{prop:finite-grid-coefficient-recovery} with \(k=0\).
The last sentence follows from the Haar-pullback identity
\eqref{eq:weighted-integral} and the phase identification above.
\end{proof}

\begin{proposition}[Finite-phase check of the normalized eighth-order coefficient computation]%
\label{prop:eighth-order-finite-phase-verification}
Let \(n_1,\ldots,n_k\ge2\), put \(m=n_1+\cdots+n_k\), and define
\[
P_{n_1,\ldots,n_k}(e^{2\ii\theta})
:=\prod_{j=1}^{k}u_{n_j}(\tan\theta).
\]
Then \(P_{n_1,\ldots,n_k}\) is a Laurent polynomial with Fourier support in
\(\{-m,\ldots,m\}\).  Hence, for every integer \(L>m\),
\begin{equation}\label{eq:finite-grid-mode-product-verification}
\int_{\RR}\prod_{j=1}^{k}u_{n_j}(y)\,
\frac{\dd y}{\pi(1+y^2)}
=
\frac1L\sum_{\zeta\in\mathcal G_L}P_{n_1,\ldots,n_k}(\zeta).
\end{equation}
In particular, every grid \(\mathcal G_L\) with \(L\ge9\) exactly checks all
mode-product integrals entering the \(t^{-4}\), \(t^{-6}\), and \(t^{-8}\)
coefficients in Theorem~\ref{thm:poisson-kl-eighth}, including the
odd-order cancellations through order \(t^{-8}\).  With the Laurent
normalization
\[
Q_2(z)=-\frac14(z+z^{-1}+z^2+z^{-2}),
\]
one has \(Q_2(1)=u_2(0)=-1\); this fixes the phase normalization used by the
coefficient table.  The resulting grid averages are exactly the constant
terms listed in Lemma~\ref{lem:eighth-order-coefficient-arithmetic}, and
therefore yield
\[
\mathcal A_{4,L}=\frac{\sigma^4}{8},\qquad
\mathcal A_{6,L}=A_6(\nu),\qquad
\mathcal A_{8,L}=A_8(\nu)
\]
for the formal eighth-order coefficient expressions of
Theorem~\ref{thm:poisson-kl-eighth}.
\end{proposition}

\begin{proof}
The finite Fourier formulas
\eqref{eq:cayley-mode-even-fourier} and
\eqref{eq:cayley-mode-odd-fourier} show that \(u_n(\tan\theta)\) has phase
support in \(\{-n,\ldots,n\}\), so products have the displayed support.
The change of variables \(y=\tan\theta\), followed by
\(z=e^{2\ii\theta}\), identifies the Cauchy-weighted line integral with the
Haar constant coefficient on \(\TT\).  Corollary
\ref{cor:alias-free-cayley-mode-product-quadrature} then proves
\eqref{eq:finite-grid-mode-product-verification}.  For the eighth-order
products the total index is at most \(8\), so \(L\ge9\) is alias-free.
Lemma~\ref{lem:eighth-order-coefficient-arithmetic} already evaluates the
same constant terms in Laurent form, and
Lemma~\ref{lem:n4-entropy-coefficient-substitution} substitutes them into the
entropy coefficients.  This gives the displayed coefficient identities
without repeating the finite arithmetic.
\end{proof}
