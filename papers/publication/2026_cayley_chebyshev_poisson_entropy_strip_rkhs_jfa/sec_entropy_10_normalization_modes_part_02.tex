\[
S_{\nu,6,1}\le7\sum_{n=2}^{6}|\mu_n|+110M_7(\nu),
\qquad
T_{\nu,6,1}\le
\max\!\left\{1,\left(14\sum_{n=2}^{6}|\mu_n|+220M_7(\nu)\right)^{1/2}\right\}.
\]
  In particular, on any centred $(L+\eta)$-moment ball
  $\EE|X_c|^{L+\eta}\le M$, Lyapunov's inequality gives the fully closed
  choices
  \begin{equation}\label{eq:closed-moment-ball-density-envelope}
  C_{\nu,L,\eta}\le 2\widehat C_LM,\qquad
  S_{\nu,L,\eta}\le
  \widehat B_L\sum_{n=2}^{L}M^{n/(L+\eta)}+2\widehat C_LM,
  \end{equation}
  and
  \begin{equation}\label{eq:closed-moment-ball-threshold-envelope}
  T_{\nu,L,\eta}\le
  \max\!\left\{1,\left(2\widehat B_L\sum_{n=2}^{L}M^{n/(L+\eta)}
  +4\widehat C_LM\right)^{1/2}\right\}.
  \end{equation}
\end{lemma}

\begin{proof}
The Chebyshev representation
\eqref{eq:cayley-chebyshev-mode} gives
\[
|u_n(\tan\theta)|
\le |\cos\theta|^n\sup_{|x|\le1}|U_n(x)|
\le n+1,
\]
because $\sup_{|x|\le1}|U_n(x)|=n+1$.  This proves
\eqref{eq:closed-mode-sup-bound}.

For the remainder write, as in the proof of
Proposition~\ref{prop:cayley-mode-remainder},
$s=\sin\theta$, $c=\cos\theta$, and $d=\tan\theta-\varepsilon$.  Then
\[
v_L(\tan\theta,\varepsilon)
=\frac{c^{L-1}\bigl(U_{L+1}(s)-sU_L(s)+dc\,U_L(s)\bigr)}
       {1+d^2}.
\]
Since $|c|\le1$, $|s|\le1$,
$\sup_{|x|\le1}|U_L(x)|=L+1$, and
$\sup_{|x|\le1}|U_{L+1}(x)|=L+2$,
\[
|v_L(\tan\theta,\varepsilon)|
\le (L+2)+(L+1)
 +(L+1)\frac{|d|\,|c|^L}{1+d^2}
\le 2L+3+\frac{L+1}{2}
=\frac{5L+7}{2}.
\]
This is uniform in $\theta$ and $\varepsilon$, proving
\eqref{eq:closed-vL-sup-bound}.  The bounds for $B_L,V_L,C_L$ follow
directly from their definitions in
Theorem~\ref{thm:finite-moment-density-remainder}; substituting those
dominants into the displayed definitions of
$C_{\nu,L,\eta}$, $S_{\nu,L,\eta}$, and $T_{\nu,L,\eta}$ gives
\eqref{eq:closed-law-density-constant-envelope} and
\eqref{eq:closed-law-smallness-envelope}.
If $\EE|X_c|^{L+\eta}\le M$, then
$|\mu_n|\le M^{n/(L+\eta)}$ for $2\le n\le L$, so the two moment-ball
displays follow from the same substitutions.
\end{proof}

\begin{lemma}[Tail-uniform Cayley mode remainder]%
\label{lem:tail-uniform-cayley-mode-remainder}
Let $N\ge1$, let $X$ be a real random variable with
$\EE|X|^{N+1}<\infty$, and let
\[
\mathcal R_{N,t}(y):=
\EE\!\left[
R_{X/t}(y)-\sum_{n=0}^{N}u_n(y)\frac{X^n}{t^n}
\right]
\qquad(t>0).
\]
Then there is a constant $M_N<\infty$, depending only on~$N$, such that
\begin{equation}\label{eq:tail-uniform-cayley-mode-remainder}
\sup_{y\in\RR}|\mathcal R_{N,t}(y)|
\;+\;\|\mathcal R_{N,t}\|_{L^1(\omega)}
\le
M_N\,\EE|X|^{N+1}\,t^{-(N+1)},
\end{equation}
where $\omega(\dd y)=\pi^{-1}(1+y^2)^{-1}\dd y$.  Thus the remainder is
uniform in the phase variable after integration over the full, unbounded
tail of~$X$.
\end{lemma}

\begin{proof}
By Proposition~\ref{prop:cayley-mode-remainder},
\[
R_{X/t}(y)-\sum_{n=0}^{N}u_n(y)\frac{X^n}{t^n}
=\frac{X^{N+1}}{t^{N+1}}v_N(y,X/t)
\]
for every $y$ and every value of~$X$.  Let
$M_N:=\|v_N\|_{L^\infty(\RR^2)}$.  Taking expectations gives
\[
|\mathcal R_{N,t}(y)|
\le M_N\,\EE|X|^{N+1}\,t^{-(N+1)}
\qquad(y\in\RR).
\]
This is the asserted uniform estimate.  Since $\omega$ is a probability
measure, the same bound also gives the displayed $L^1(\omega)$ estimate.
\end{proof}

\begin{theorem}[Density quotient expansion with finite-moment remainder]%
\label{thm:finite-moment-density-remainder}
Let $L\ge2$, let $0<\eta\le1$, and let $\nu$ be a probability measure
on~$\RR$ with mean $\bar{\gamma}$ and finite centred $(L+\eta)$th absolute
moment.  Put
$X_c=\gamma-\bar{\gamma}$,
$\mu_n=\int X_c^n\dd\nu(\gamma)$, and
\[
\delta_t(\bar{\gamma}+ty):=\frac{(P_t*\nu)(\bar{\gamma}+ty)}
{P_t(ty)}-1.
\]
Let
\[
B_L:=\max_{0\le n\le L}\|u_n\|_\infty,
\qquad
V_L:=\|v_L\|_\infty,
\qquad
C_L:=\max\{V_L,\,6+(L+1)B_L\}.
\]
Then \eqref{eq:finite-moment-density-remainder} holds with
$C_{\nu,L,\eta}=2C_L\EE|X_c|^{L+\eta}$.  Moreover, with
\[
S_{\nu,L,\eta}:=
B_L\sum_{n=2}^{L}|\mu_n|+C_{\nu,L,\eta},
\qquad
T_{\nu,L,\eta}:=\max\{1,(2S_{\nu,L,\eta})^{1/2}\},
\]
one has
\begin{equation}\label{eq:finite-moment-density-remainder}
\delta_t(\bar{\gamma}+ty)
=\sum_{n=2}^{L}\mu_n u_n(y)t^{-n}+\rho_{L,t}(y),
\qquad
\sup_y|\rho_{L,t}(y)|+\|\rho_{L,t}\|_{L^1(\omega)}
\le C_{\nu,L,\eta}t^{-(L+\eta)},
\end{equation}
and
\begin{equation}\label{eq:finite-moment-density-smallness}
\sup_y|\delta_t(\bar{\gamma}+ty)|\le S_{\nu,L,\eta}t^{-2},
\qquad t\ge1,
\end{equation}
with $\sup_y|\delta_t(\bar{\gamma}+ty)|\le1/2$ for
$t\ge T_{\nu,L,\eta}$.
\end{theorem}

\begin{proof}
With $x=\bar{\gamma}+ty$ and $g_t(x)=P_t(x-\bar{\gamma})=P_t(ty)$,
\[
\frac{(P_t*\nu)(x)}{g_t(x)}
=\int_{\RR}R_{X_c/t}(y)\dd\nu(\gamma).
\]
For $|\varepsilon|\le1$, Proposition~\ref{prop:cayley-mode-remainder} and
the boundedness of $v_L$ give
\[
\left|R_\varepsilon(y)-\sum_{n=0}^{L}u_n(y)\varepsilon^n\right|
    \le V_L |\varepsilon|^{L+1}
    \le V_L |\varepsilon|^{L+\eta}.
\]
For $|\varepsilon|>1$, write $y=z+\varepsilon$.  Then
\[
R_\varepsilon(y)=\frac{1+(z+\varepsilon)^2}{1+z^2}
\le 3(1+\varepsilon^2)
\le 6|\varepsilon|^{L+\eta},
\]
uniformly in~$z$, because $L+\eta\ge2$.  Also
\[
\left|\sum_{n=0}^{L}u_n(y)\varepsilon^n\right|
\le (L+1)B_L|\varepsilon|^L
\le (L+1)B_L|\varepsilon|^{L+\eta}.
\]
Thus the pointwise remainder is bounded by
$C_L|\varepsilon|^{L+\eta}$ uniformly in~$y$.
Taking $\varepsilon=X_c/t$ and expectations yields the displayed
$t^{-(L+\eta)}$ bound in both $L^\infty$ and $L^1(\omega)$, since $\omega$
has total mass one.  Since $\mu_0=1$ and $\mu_1=0$, subtracting the
constant term gives \eqref{eq:finite-moment-density-remainder}.  Each
$u_n$ is bounded by Theorem~\ref{thm:cayley-chebyshev-mode}; hence the
finite sum begins at order~$t^{-2}$.  The displayed definition of
$S_{\nu,L,\eta}$ therefore gives
\eqref{eq:finite-moment-density-smallness}, because $L+\eta\ge2$ and
$t\ge1$.  The displayed choice of $T_{\nu,L,\eta}$ gives the final
smallness assertion.
\end{proof}

For later use we record the first modes explicitly:
\begin{equation}\label{eq:cayley-first-modes}
u_2(y)=\frac{3y^2-1}{(1+y^2)^2},
\qquad
u_3(y)=\frac{4y(y^2-1)}{(1+y^2)^3},
\qquad
u_4(y)=\frac{5y^4-10y^2+1}{(1+y^2)^4},
\end{equation}
\begin{equation}\label{eq:cayley-next-modes}
u_5(y)=\frac{6y^5-20y^3+6y}{(1+y^2)^5},
\qquad
u_6(y)=\frac{7y^6-35y^4+21y^2-1}{(1+y^2)^6}.
\end{equation}

The trigonometric coefficients below are ordinary Haar-Fourier coefficients of
the circle characters in the Cayley phase coordinate.  The eighth-order
entropy arithmetic itself is not repeated here; its single finite Laurent
certificate is Lemma~\ref{lem:eighth-order-coefficient-arithmetic}.

\begin{proposition}[Poisson--Cauchy moment inversion by phase modes]%
\label{prop:poisson-cauchy-moment-inversion-by-phase-modes}
Let $\nu$ be a probability measure on~$\RR$ with mean~$\bar{\gamma}$ and
finite centred fifth absolute moment.  Put
\[
X_c:=\gamma-\bar{\gamma},\qquad
\mu_n:=\int_{\RR}X_c^n\,\dd\nu(\gamma),\qquad
\sigma^2:=\mu_2,
\]
and let $h_t:=P_t*\nu$ and $g_t(x):=P_t(x-\bar{\gamma})$.  For
$\theta\in(-\pi/2,\pi/2)$ define the phase profile
\[
F_t(\theta):=
\frac{h_t(\bar{\gamma}+t\tan\theta)}
     {g_t(\bar{\gamma}+t\tan\theta)}-1.
\]
For $k\ge1$ set
\[
\mathcal C_k(t):=
2\int_{-\pi/2}^{\pi/2}F_t(\theta)\cos(2k\theta)\,\frac{\dd\theta}{\pi},
\qquad
\mathcal S_k(t):=
2\int_{-\pi/2}^{\pi/2}F_t(\theta)\sin(2k\theta)\,\frac{\dd\theta}{\pi}.
\]
Then
\[
\sigma^2=\lim_{t\to\infty}-2t^2\bigl(\mathcal C_1(t)-\mathcal C_4(t)\bigr),
\qquad
\mu_3=\lim_{t\to\infty}-4t^3\mathcal S_3(t),
\qquad
\mu_4=\lim_{t\to\infty}8t^4\mathcal C_4(t).
\]
More precisely,
\[
\mathcal C_1(t)-\mathcal C_4(t)
=-\frac{\sigma^2}{2t^2}+O(t^{-5}),\qquad
\mathcal S_3(t)=-\frac{\mu_3}{4t^3}+O(t^{-5}),
\qquad
\mathcal C_4(t)=\frac{\mu_4}{8t^4}+O(t^{-5}).
\]
The $O(t^{-5})$ term in the middle estimate is an actual fifth-order
remainder: when the fifth centred moment is finite it contains the
$\sin(6\theta)$ coefficient of $u_5$, and no coefficient-free cancellation at
order $t^{-5}$ is asserted.
\end{proposition}

\begin{proof}
Write $y=\tan\theta$.  For $x=\bar{\gamma}+ty$ one has
\[
\frac{P_t(x-\gamma)}{P_t(x-\bar{\gamma})}
=\frac{1+y^2}{1+(y-X_c/t)^2}
=R_{X_c/t}(y).
\]
Therefore
\[
F_t(\theta)=\int_{\RR}\bigl(R_{X_c/t}(\tan\theta)-1\bigr)
\,\dd\nu(\gamma).
\]
Proposition~\ref{prop:cayley-mode-remainder} gives the pointwise
order-four identity
\[
R_{X_c/t}(y)-1
=\sum_{n=1}^{4}u_n(y)\frac{X_c^n}{t^n}
+\frac{X_c^5}{t^5}v_4(y,X_c/t).
\]
Taking $\nu$-expectations and using $\mu_1=0$ gives, uniformly in
$\theta$,
\[
F_t(\theta)
=\frac{\sigma^2}{t^2}u_2(\tan\theta)
+\frac{\mu_3}{t^3}u_3(\tan\theta)
+\frac{\mu_4}{t^4}u_4(\tan\theta)
+O(t^{-5}).
\]
The finite fifth absolute moment is used only through
Lemma~\ref{lem:tail-uniform-cayley-mode-remainder}, with $X=X_c$ and
truncation order~$4$, to bound the last expectation uniformly.
Thus the $O(t^{-5})$ term has not been discarded coefficientwise; it is the
full order-five tail produced by $v_4$.  If one instead expands through
$u_5$, the formula
\[
u_5(\tan\theta)=\frac1{16}\bigl(\sin2\theta+4\sin4\theta
+6\sin6\theta+4\sin8\theta+\sin10\theta\bigr)
\]
shows explicitly that the $\mathcal S_3$ channel has a fifth-order
contribution $3\mu_5/(8t^5)$ whenever $\mu_5$ is finite; the proposition uses
only the leading coefficient needed for the moment limit.

The first three nonconstant modes have the finite phase-Fourier
expansions
\[
u_2(\tan\theta)=-\frac12\bigl(\cos2\theta+\cos4\theta\bigr),
\]
\[
u_3(\tan\theta)
=-\frac14\bigl(\sin2\theta+2\sin4\theta+\sin6\theta\bigr),
\]
\[
u_4(\tan\theta)
=\frac18\bigl(\cos2\theta+3\cos4\theta+3\cos6\theta+\cos8\theta\bigr),
\]
which are the cases $n=2,3,4$ of
Theorem~\ref{thm:cayley-chebyshev-mode}.  The functions
$\cos(2k\theta)$ and $\sin(2k\theta)$ are orthogonal on
$(-\pi/2,\pi/2)$, and the prefactor $2/\pi$ in the definitions of
$\mathcal C_k$ and $\mathcal S_k$ extracts the corresponding coefficient.
Thus the preceding uniform expansion gives
\[
\mathcal C_1(t)
=-\frac{\sigma^2}{2t^2}+\frac{\mu_4}{8t^4}+O(t^{-5}),
\qquad
\mathcal C_4(t)=\frac{\mu_4}{8t^4}+O(t^{-5}),
\]
and
\[
\mathcal S_3(t)=-\frac{\mu_3}{4t^3}+O(t^{-5}).
\]
The last displayed remainder includes the possible fifth-order sine-mode
contribution just identified.
Subtracting the two cosine coefficients and multiplying by the displayed
powers of~$t$ proves the three inversion formulas.
\end{proof}

\begin{lemma}[Mode parity cancellation]\label{lem:mode-parity-vanish}
For integers $n_1,\dots,n_k\ge 1$ let
\[
J_{n_1,\dots,n_k}:=
\int_{\RR}
\frac{u_{n_1}(y)\cdots u_{n_k}(y)}{\pi(1+y^2)}\,\dd y.
\]
Then $J_{n_1,\dots,n_k}=0$ whenever $n_1+\cdots+n_k$ is odd.
\end{lemma}

\begin{proof}
From \eqref{eq:cayley-mode-parity}, $u_n(-y)=(-1)^n u_n(y)$ for every
$n\ge1$, while $(1+y^2)^{-1}$ is even.  Hence the integrand is odd if
$n_1+\cdots+n_k$ is odd, so the integral vanishes.
\end{proof}
