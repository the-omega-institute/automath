\subsection{Eighth-order defect layer extracted from the all-order calculus}%
\label{subsec:eighth-order-defects}

The following theorem states the seventh-moment eighth-order entropy jet and
the quantitative $O(t^{-9})$ remainder directly.  The finite-phase, defect,
and coefficient-level \(W_2\) statements that follow use this jet.  We first
record the full finite coefficient audit used by the theorem.

\begin{lemma}[Laurent audit for the fourth, sixth, and eighth entropy coefficients]%
\label{lem:eighth-order-coefficient-arithmetic}
Let \(\omega(\dd y)=\pi^{-1}(1+y^2)^{-1}\dd y\), let
\(I(n_1,\ldots,n_r)\) be the Cauchy mode integral
\eqref{eq:mode-integral-definition}, and define \(Q_n\) by
\eqref{eq:mode-laurent-polynomial-definition}.  For the modes needed through
order eight,
\begin{align*}
Q_2(z)&=-\frac12(z+z^{-1})-\frac12(z^2+z^{-2}),\\
Q_3(z)&=-\frac1{4\ii}\bigl((z-z^{-1})+2(z^2-z^{-2})+(z^3-z^{-3})\bigr),\\
Q_4(z)&=\frac18\bigl((z+z^{-1})+3(z^2+z^{-2})
+3(z^3+z^{-3})+(z^4+z^{-4})\bigr),\\
Q_5(z)&=\frac1{16\ii}\bigl((z-z^{-1})+4(z^2-z^{-2})
+6(z^3-z^{-3})+4(z^4-z^{-4})+(z^5-z^{-5})\bigr),\\
Q_6(z)&=-\frac1{32}\bigl((z+z^{-1})+5(z^2+z^{-2})
+10(z^3+z^{-3})\\
&\hspace{6.5em}
+10(z^4+z^{-4})+5(z^5+z^{-5})+(z^6+z^{-6})\bigr).
\end{align*}
The required constant Laurent coefficients are
\[
\begin{array}{c|c@{\qquad}c|c@{\qquad}c}
\text{product} & [z^0] & & \text{product} & [z^0] \\
\hline
Q_2^2 & \frac14 &&
Q_2Q_6 & \frac{3}{64} \\
Q_3^2 & \frac{3}{16} &&
Q_3Q_5 & -\frac{15}{128} \\
Q_2Q_4 & -\frac18 &&
Q_4^2 & \frac{5}{32} \\
Q_2^3 & -\frac{3}{32} &&
Q_2^2Q_4 & \frac{3}{32} \\
&&&
Q_2Q_3^2 & -\frac{9}{128} \\
&&&
Q_2^4 & \frac{9}{64}
\end{array}
\]
Equivalently,
\[
\begin{array}{c|c|c}
\text{coefficient} & \text{polynomial before integration} &
\text{mode integrals used} \\
\hline
B_4 &
\frac12a_2^2 &
I(2,2)=\frac14 \\
\hline
B_6 &
a_2a_4+\frac12a_3^2-\frac16a_2^3 &
\begin{gathered}
I(2,4)=-\frac18,\quad I(3,3)=\frac{3}{16},\\
I(2,2,2)=-\frac{3}{32}
\end{gathered}
\\
\hline
B_8 &
\begin{gathered}
a_2a_6+a_3a_5+\frac12a_4^2\\
-\frac12a_2^2a_4-\frac12a_2a_3^2+\frac1{12}a_2^4
\end{gathered}
&
\begin{gathered}
I(2,6)=\frac{3}{64},\quad I(3,5)=-\frac{15}{128},\\
I(4,4)=\frac{5}{32},\quad I(2,2,4)=\frac{3}{32},\\
I(2,3,3)=-\frac{9}{128},\quad I(2,2,2,2)=\frac{9}{64}
\end{gathered}
\end{array}
\]
where \(a_j=\mu_j u_j\) for \(2\le j\le6\) and \(\mu_2=\sigma^2\).  Hence
\[
\int_{\RR}B_4\,\dd\omega=\frac{\sigma^4}{8},\qquad
\int_{\RR}B_6\,\dd\omega
=\frac{\sigma^6+6\mu_3^2-8\sigma^2\mu_4}{64},
\]
and
\[
\int_{\RR}B_8\,\dd\omega
=\frac{3\sigma^8-12\sigma^4\mu_4+9\sigma^2\mu_3^2
+12\sigma^2\mu_6-30\mu_3\mu_5+20\mu_4^2}{256}.
\]
\end{lemma}

\begin{proof}
The formulas for \(Q_2,\ldots,Q_6\) are the special cases
\(n=2,\ldots,6\) of
\eqref{eq:mode-laurent-even}--\eqref{eq:mode-laurent-odd}.  By
Theorem~\ref{thm:finite-fourier-mode-integral-algorithm},
\[
I(n_1,\ldots,n_r)=[z^0]Q_{n_1}\cdots Q_{n_r}.
\]
Multiplying the displayed Laurent polynomials gives the constant-term table.
Substituting those constants into \(B_4\), \(B_6\), and \(B_8\) gives
\[
\int_{\RR}B_4\,\dd\omega
=\frac{\sigma^4}{2}I(2,2)=\frac{\sigma^4}{8},
\]
\[
\begin{aligned}
\int_{\RR}B_6\,\dd\omega
&=\sigma^2\mu_4 I(2,4)
+\frac{\mu_3^2}{2}I(3,3)
-\frac{\sigma^6}{6}I(2,2,2)  \\
&=-\frac{\sigma^2\mu_4}{8}
+\frac{3\mu_3^2}{32}
+\frac{\sigma^6}{64}
=\frac{\sigma^6+6\mu_3^2-8\sigma^2\mu_4}{64},
\end{aligned}
\]
and
\[
\begin{aligned}
\int_{\RR}B_8\,\dd\omega
&=\sigma^2\mu_6 I(2,6)+\mu_3\mu_5 I(3,5)
+\frac{\mu_4^2}{2}I(4,4)\\
&\quad-\frac{\sigma^4\mu_4}{2}I(2,2,4)
-\frac{\sigma^2\mu_3^2}{2}I(2,3,3)
+\frac{\sigma^8}{12}I(2,2,2,2)\\
&=\frac{3\sigma^8-12\sigma^4\mu_4+9\sigma^2\mu_3^2
+12\sigma^2\mu_6-30\mu_3\mu_5+20\mu_4^2}{256}.
\end{aligned}
\]
\end{proof}

\begin{lemma}[Formal and seventh-moment eighth coefficient]%
\label{lem:formal-seventh-eighth-coefficient}
The formal degree-eight coefficient \(A^{\rm form}_8\) involves only
\(\mathfrak m_2,\ldots,\mathfrak m_6\).  Its evaluated polynomial is
\begin{equation}\label{eq:formal-eighth-evaluated-polynomial}
{\rm ev}_{\nu,8}(A^{\rm form}_8)
=
\frac{3\mu_2^4-12\mu_2^2\mu_4+9\mu_2\mu_3^2
+12\mu_2\mu_6-30\mu_3\mu_5+20\mu_4^2}{256}.
\end{equation}
If \(\nu\) has a finite centred seventh absolute moment, then the polynomial
in \eqref{eq:formal-eighth-evaluated-polynomial} is the actual \(t^{-8}\)
coefficient in the Poisson entropy expansion, with an \(O_\nu(t^{-9})\)
remainder through order eight.  We denote this seventh-moment entropy
coefficient by
\begin{equation}\label{eq:c8-seventh-defined-from-formal}
C_8^{(7)}(\nu):={\rm ev}_{\nu,8}(A^{\rm form}_8).
\end{equation}
If, in addition, \(\nu\) satisfies the stronger finite centred
\((8+\eta)\)th absolute moment hypothesis of
Theorem~\ref{thm:poisson-kl-even-orders}, then
\[
C_8^{(7)}(\nu)=A_8(\nu).
\]
\end{lemma}

\begin{proof}
In the formal definition \eqref{eq:universal-entropy-coefficient}, a
degree-eight monomial has the form
\(\mathfrak m_{n_1}\cdots\mathfrak m_{n_r}\) with \(r\ge2\),
\(n_j\ge2\), and \(n_1+\cdots+n_r=8\).  Hence every index satisfies
\(2\le n_j\le6\); no \(\mathfrak m_7\) or \(\mathfrak m_8\) can occur.
The degree-eight product before integration is
\[
a_2a_6+a_3a_5+\frac12a_4^2
-\frac12a_2^2a_4-\frac12a_2a_3^2+\frac1{12}a_2^4
\]
with \(a_j=\mu_j u_j\).  The audit table above gives its integral
\[
\begin{aligned}
&\mu_2\mu_6 I(2,6)+\mu_3\mu_5 I(3,5)
+\frac12\mu_4^2 I(4,4)
-\frac12\mu_2^2\mu_4 I(2,2,4)\\
&\qquad
-\frac12\mu_2\mu_3^2 I(2,3,3)
+\frac1{12}\mu_2^4 I(2,2,2,2),
\end{aligned}
\]
which is exactly \eqref{eq:formal-eighth-evaluated-polynomial} after
substitution of the listed mode integrals.

It remains to justify why the same formal expression is an actual entropy
coefficient under only a seventh moment.  Let \(\Delta=\gamma-\bar\gamma\) and
write \(x=\bar\gamma+ty\), with
\(\omega(\dd y)=\pi^{-1}(1+y^2)^{-1}\dd y\).
Lemma~\ref{lem:tail-uniform-cayley-mode-remainder}
with truncation order \(6\) gives
\[
\delta_t(\bar\gamma+ty)
=\sum_{n=2}^{6}\mu_nu_n(y)t^{-n}+\theta_t(y),
\qquad
\|\theta_t\|_\infty+\|\theta_t\|_{L^1(\omega)}
=O_\nu(t^{-7}).
\]
Since the displayed finite sum is \(O_\nu(t^{-2})\) uniformly, the Taylor
expansion
\(\Phi(s)=(1+s)\log(1+s)-s
=s^2/2-s^3/6+s^4/12+O(s^5)\) gives, uniformly in \(y\),
\[
\Phi(\delta_t)
=\frac12d_t^2-\frac16d_t^3+\frac1{12}d_t^4+O_\nu(t^{-9}),
\qquad
d_t:=\sum_{n=2}^{6}\mu_nu_n(y)t^{-n}.
\]
Indeed \(d_t\theta_t=O_\nu(t^{-9})\), while every term containing
\(\theta_t\) in degree at least three is \(O_\nu(t^{-11})\), and
\(\Phi^{(5)}\)-terms are \(O_\nu(t^{-10})\).  Therefore no occurrence of the
seventh-moment remainder can contribute to the \(t^{-8}\) coefficient.  After
integrating against \(\omega\), the \(t^{-8}\) coefficient is precisely the
finite formal coefficient already computed, and the discarded terms give an
\(O_\nu(t^{-9})\) remainder.  Under the additional finite-\((8+\eta)\)
hypothesis, Theorem~\ref{thm:poisson-kl-even-orders} identifies
\(A_8(\nu)={\rm ev}_{\nu,8}(A^{\rm form}_8)\) as the actual finite-order
entropy coefficient at order \(t^{-8}\), so \(C_8^{(7)}(\nu)=A_8(\nu)\).
\end{proof}

\begin{theorem}[Seventh-moment eighth-order Poisson entropy jet]%
\label{thm:poisson-kl-eighth}
Let $\nu$ be a probability measure on~$\RR$ with mean~$\bar\gamma$,
variance $\sigma^2>0$, and finite centred seventh absolute moment.  Then,
with
\[
\Delta:=\gamma-\bar\gamma,
\qquad
\mu_n:=\int_{\RR}\Delta^n\dd\nu(\gamma),
\]
and
\[
h_t:=P_t*\nu,
\qquad
g_t(x):=P_t(x-\bar\gamma),
\]
the seventh-moment coefficient jet through order eight is
\[
C_4=\frac{\sigma^4}{8},\qquad
C_6=\frac{\sigma^6+6\mu_3^2-8\sigma^2\mu_4}{64},
\]
\[
C_8^{(7)}(\nu)=
\frac{3\sigma^8-12\sigma^4\mu_4+9\sigma^2\mu_3^2
+12\sigma^2\mu_6-30\mu_3\mu_5+20\mu_4^2}{256}.
\]
Here \(C_8^{(7)}(\nu)\) is the actual seventh-moment \(t^{-8}\) coefficient
from Lemma~\ref{lem:formal-seventh-eighth-coefficient}.  It is evaluated by
the same formal polynomial \(A^{\rm form}_8\), but it is not denoted
\(A_8(\nu)\) unless the stronger finite centred \((8+\eta)\)th absolute
moment hypothesis of Theorem~\ref{thm:poisson-kl-even-orders} also holds.
With these coefficients one has, as $t\to\infty$,
\begin{equation}\label{eq:poisson-kl-eighth}
\begin{split}
D_{\mathrm{KL}}(h_t\|g_t)
=\frac{\sigma^4}{8t^4}
+\frac{\sigma^6+6\mu_3^2-8\sigma^2\mu_4}{64\,t^6}
\\
\quad
+\frac{3\sigma^8-12\sigma^4\mu_4+9\sigma^2\mu_3^2}{256\,t^8}
+\frac{12\sigma^2\mu_6-30\mu_3\mu_5+20\mu_4^2}{256\,t^8}
+O_\nu(t^{-9}).
\end{split}
\end{equation}
More precisely, there are constants $B_{\nu,8},T_{\nu,8}<\infty$ such
that, for all $t\ge T_{\nu,8}$,
\begin{equation}\label{eq:poisson-kl-eighth-remainder}
\begin{split}
\Bigg|D_{\mathrm{KL}}(h_t\|g_t)
&-\frac{\sigma^4}{8t^4}
-\frac{\sigma^6+6\mu_3^2-8\sigma^2\mu_4}{64\,t^6}
\\
&-\frac{3\sigma^8-12\sigma^4\mu_4+9\sigma^2\mu_3^2
+12\sigma^2\mu_6-30\mu_3\mu_5+20\mu_4^2}{256\,t^8}
\Bigg|
\le B_{\nu,8}t^{-9}.
\end{split}
\end{equation}
\end{theorem}

\begin{proof}
The products and parity cancellations are those of
Theorem~\ref{thm:universal-entropy-coefficients}, and the actual mode integrals
are evaluated by Theorem~\ref{thm:finite-fourier-mode-integral-algorithm}.
What is special here is the lower-moment asymptotic bridge isolated in
Lemma~\ref{lem:formal-seventh-eighth-coefficient}.  Thus the eighth-order
coefficient below is denoted \(C_8^{(7)}(\nu)\), not \(A_8(\nu)\), unless the
stronger finite centred \((8+\eta)\)th absolute moment hypothesis is also
available.

Let $\omega(\dd y)=\pi^{-1}(1+y^2)^{-1}\dd y$.  With
$x=\bar\gamma+ty$, Lemma~\ref{lem:tail-uniform-cayley-mode-remainder}
with $X=\Delta$ and truncation order~$6$ gives
\[
\delta_t(\bar\gamma+ty)
=\frac{\sigma^2}{t^2}u_2(y)
+\frac{\mu_3}{t^3}u_3(y)
+\frac{\mu_4}{t^4}u_4(y)
+\frac{\mu_5}{t^5}u_5(y)
+\frac{\mu_6}{t^6}u_6(y)
+\theta_t(y),
\]
where, if $v_6$ is the bounded remainder from
Proposition~\ref{prop:cayley-mode-remainder}, then
\[
\|\theta_t\|_{L^\infty(\RR)}
+\|\theta_t\|_{L^1(\omega)}
\le
\|v_6\|_{L^\infty(\RR^2)}\,\EE|\Delta|^7\,t^{-7}.
\]
Also
$g_t(\bar\gamma+ty)\dd(\bar\gamma+ty)=\omega(\dd y)$, hence
\[
D_{\mathrm{KL}}(h_t\|g_t)
=\int_{\RR}\Phi(\delta_t(\bar\gamma+ty))\,\omega(\dd y).
\]
Set
\[
a_2:=\sigma^2\,u_2,\qquad a_3:=\mu_3\,u_3,\qquad a_4:=\mu_4\,u_4,
\qquad a_5:=\mu_5\,u_5,\qquad a_6:=\mu_6\,u_6.
\]
Write
\[
d_t:=t^{-2}a_2+t^{-3}a_3+t^{-4}a_4+t^{-5}a_5+t^{-6}a_6,
\qquad
\delta_t=d_t+\theta_t,
\]
with the same $\theta_t$ as above.  The finite seventh absolute moment is
used exactly here: it makes the omitted mode $\theta_t$ small enough that
any occurrence of it lies past order~$t^{-8}$.  For the quantitative
remainder, set
\[
U_n:=\|u_n\|_{L^\infty(\RR)}\quad(2\le n\le6),\qquad
V_6:=\|v_6\|_{L^\infty(\RR^2)},
\]
\[
A_\nu:=\sum_{n=2}^{6}|\mu_n|U_n,\qquad
M_\nu:=V_6\,\EE|\Delta|^7,\qquad H_\nu:=A_\nu+M_\nu .
\]
Then, for $t\ge1$,
\[
\|d_t\|_\infty\le A_\nu t^{-2},\qquad
\|\theta_t\|_\infty+\|\theta_t\|_{L^1(\omega)}
\le M_\nu t^{-7},\qquad
\|\delta_t\|_\infty\le H_\nu t^{-2}.
\]
Let
\[
K_\Phi:=\sum_{j=5}^{\infty}\frac{2^{-(j-5)}}{j(j-1)},
\qquad
T_{\nu,8}:=\max\{1,(2H_\nu)^{1/2}\}.
\]
For $t\ge T_{\nu,8}$,
\[
\left|\Phi(\delta_t)
-\left(\frac12\delta_t^2-\frac16\delta_t^3
+\frac1{12}\delta_t^4\right)\right|
\le K_\Phi H_\nu^5t^{-10}.
\]
For $k=2,3,4$ put
\[
S_{k,\nu}:=
\sum_{\substack{2\le n_1,\dots,n_k\le6\\ n_1+\cdots+n_k\ge9}}
\prod_{j=1}^{k}|\mu_{n_j}|U_{n_j}.
\]
The discarded terms in the exact finite product $d_t^k$ after total
$t$-degree eight are bounded by $S_{k,\nu}t^{-9}$ in $L^\infty$.  Moreover
\[
E_{2,\nu}:=2A_\nu M_\nu+M_\nu^2,\qquad
E_{3,\nu}:=3A_\nu^2M_\nu+3A_\nu M_\nu^2+M_\nu^3,
\]
\[
E_{4,\nu}:=4A_\nu^3M_\nu+6A_\nu^2M_\nu^2
+4A_\nu M_\nu^3+M_\nu^4
\]
give, for $t\ge1$, the following finite computable bounds:
\begin{align*}
\|\delta_t^2-d_t^2\|_\infty
&\le 2A_\nu M_\nu t^{-9}+M_\nu^2t^{-14}
\le E_{2,\nu}t^{-9},\\
\|\delta_t^3-d_t^3\|_\infty
&\le 3A_\nu^2M_\nu t^{-11}+3A_\nu M_\nu^2t^{-16}
+M_\nu^3t^{-21}
\le E_{3,\nu}t^{-9},\\
\|\delta_t^4-d_t^4\|_\infty
&\le 4A_\nu^3M_\nu t^{-13}+6A_\nu^2M_\nu^2t^{-18}
+4A_\nu M_\nu^3t^{-23}+M_\nu^4t^{-28}
\le E_{4,\nu}t^{-9}.
\end{align*}
Thus the sole order-$t^{-9}$ contribution involving the seventh-moment
remainder is the quadratic cross term $2d_t\theta_t$; all higher powers that
contain $\theta_t$ are strictly smaller.
Consequently one may take
\begin{equation}\label{eq:eighth-remainder-constant}
B_{\nu,8}:=
\frac12(S_{2,\nu}+E_{2,\nu})
+\frac16(S_{3,\nu}+E_{3,\nu})
+\frac1{12}(S_{4,\nu}+E_{4,\nu})
+K_\Phi H_\nu^5 .
\end{equation}
This bound is used in the quantitative large-time corollary below.  It remains
to identify the coefficients produced by the finite product pattern.
Since
the preceding estimates imply $\delta_t=O(t^{-2})$, $d_t=O(t^{-2})$, and
$\theta_t=O(t^{-7})$ uniformly, we have
\[
\delta_t^2=d_t^2+2d_t\theta_t+\theta_t^2=d_t^2+O(t^{-9}),
\]
while
\[
\delta_t^3=d_t^3+O(t^{-11}),\qquad
\delta_t^4=d_t^4+O(t^{-13}).
\]
Combining these estimates with the Taylor remainder for~$\Phi$ gives the
following uniform rule for coefficients:
\[
\begin{aligned}
[t^{-m}]\Phi(\delta_t)
&=[t^{-m}]\bigl(\tfrac12d_t^2-\tfrac16d_t^3+\tfrac1{12}d_t^4\bigr),
\\
&\hspace{5em}4\le m\le8.
\end{aligned}
\]
Here coefficient extraction is understood modulo a uniform $O(t^{-9})$
remainder.  This is the only point in the proof where the seventh absolute
moment is needed beyond the lower moments appearing in the displayed
coefficients.  Equivalently,
\[
\Phi(\delta_t)
=\frac12d_t^2-\frac16d_t^3+\frac1{12}d_t^4+O(t^{-9})
=\sum_{m=4}^{8}t^{-m}B_m(y)+O(t^{-9})
\]
uniformly in $y$.  The coefficient functions are therefore obtained by
expanding only the following finite products in powers of~$t^{-1}$.  No
term containing $\theta_t$ can contribute to any displayed coefficient, and
all terms of total order larger than eight are absorbed in the uniform
$O(t^{-9})$ remainder:
\begin{align*}
d_t^2
&=t^{-4}a_2^2+2t^{-5}a_2a_3
+t^{-6}(2a_2a_4+a_3^2) \\
&\quad
+t^{-7}(2a_2a_5+2a_3a_4)
+t^{-8}(2a_2a_6+2a_3a_5+a_4^2)+O(t^{-9}),\\
d_t^3
&=t^{-6}a_2^3+3t^{-7}a_2^2a_3
+t^{-8}(3a_2^2a_4+3a_2a_3^2)+O(t^{-9}),\\
d_t^4
&=t^{-8}a_2^4+O(t^{-9}).
\end{align*}
Thus
\begin{align*}
B_4&=\frac12a_2^2,\\
B_5&=a_2a_3,\\
B_6&=a_2a_4+\frac12a_3^2-\frac16a_2^3,\\
B_7&=a_2a_5+a_3a_4-\frac12a_2^2a_3,\\
B_8&=a_2a_6+a_3a_5+\frac12a_4^2
-\frac12a_2^2a_4-\frac12a_2a_3^2+\frac1{12}a_2^4.
\end{align*}
The definitions of $S_{k,\nu}$, $E_{k,\nu}$, and $K_\Phi$ above give,
for $t\ge T_{\nu,8}$,
\[
\left\|
\Phi(\delta_t)-\sum_{m=4}^{8}t^{-m}B_m
\right\|_{L^\infty(\RR)}
\le B_{\nu,8}t^{-9}.
\]
Thus the extraction rule gives $[t^{-m}]\Phi(\delta_t)=B_m$ for
$4\le m\le8$.  In particular, the $t^{-8}$ coefficient is exactly the sum
of three contributions:
\[
\frac12[t^{-8}]d_t^2
=a_2a_6+a_3a_5+\frac12a_4^2,
\]
\[
-\frac16[t^{-8}]d_t^3
=-\frac12a_2^2a_4-\frac12a_2a_3^2,
\qquad
\frac1{12}[t^{-8}]d_t^4=\frac1{12}a_2^4.
\]
The functions $B_5$ and $B_7$ are sums of products
$u_{n_1}\cdots u_{n_k}$ with odd total index, so
Lemma~\ref{lem:mode-parity-vanish} gives
\[
\int_{\RR}B_5\,\dd\omega=\int_{\RR}B_7\,\dd\omega=0.
\]
The remaining coefficients are obtained by the finite Fourier algorithm,
Theorem~\ref{thm:finite-fourier-mode-integral-algorithm}.  The explicit
constant-term substitutions for the products in $B_4$, $B_6$, and $B_8$ are
collected in Lemma~\ref{lem:eighth-order-coefficient-arithmetic}, which gives
\[
\int_{\RR}B_4\,\dd\omega=\frac{\sigma^4}{8},
\qquad
\int_{\RR}B_6\,\dd\omega
=\frac{\sigma^6+6\mu_3^2-8\sigma^2\mu_4}{64},
\]
\[
\int_{\RR}B_8\,\dd\omega
=\frac{3\sigma^8-12\sigma^4\mu_4+9\sigma^2\mu_3^2}{256}
+\frac{12\sigma^2\mu_6-30\mu_3\mu_5+20\mu_4^2}{256}.
\]
Since $\omega(\RR)=1$, the finite computable uniform bound
\eqref{eq:eighth-remainder-constant} gives
\eqref{eq:poisson-kl-eighth-remainder}.  Substituting the three displayed
coefficients proves \eqref{eq:poisson-kl-eighth}.
\end{proof}

\begin{lemma}[Compatibility of the seventh-moment eighth coefficient with the finite-moment coefficient]%
\label{lem:seventh-eighth-formal-compatibility}
Let $\nu$ have mean~$\bar\gamma$, positive variance, and finite centred
seventh absolute moment.  Let \(C_8^{(7)}(\nu)\) be the coefficient defined
in Lemma~\ref{lem:formal-seventh-eighth-coefficient}.  Then
\[
C_8^{(7)}(\nu)
=
{\rm ev}_{\nu,8}(A_8^{\rm form})
=
\frac{3\sigma^8-12\sigma^4\mu_4+9\sigma^2\mu_3^2
+12\sigma^2\mu_6-30\mu_3\mu_5+20\mu_4^2}{256}.
\]
If, in addition, $\nu$ has a finite centred $(8+\eta)$th absolute moment for
some $0<\eta\le1$, then the coefficient $A_8(\nu)$ in
Theorem~\ref{thm:poisson-kl-even-orders} is the actual finite-order entropy
coefficient at order \(t^{-8}\), and
\[
A_8(\nu)=C_8^{(7)}(\nu).
\]
\end{lemma}

\begin{proof}
This is exactly Lemma~\ref{lem:formal-seventh-eighth-coefficient}, restated
for later references from the defect and stability estimates.
\end{proof}

\begin{corollary}[Moment-jet factorization of the eighth-order consequence]%
\label{cor:eighth-order-jet-factorization}
Let $\nu$ and $\widetilde\nu$ be probability measures on~$\RR$ with means
$\bar\gamma$ and $\widetilde{\bar\gamma}$, positive variances, and finite
centred seventh absolute moments.  Write
\[
\mu_n:=\int(\gamma-\bar\gamma)^n\dd\nu(\gamma),
\qquad
\widetilde\mu_n:=\int(\gamma-\widetilde{\bar\gamma})^n
\dd\widetilde\nu(\gamma).
\]
If
\[
\mu_n=\widetilde\mu_n,
\qquad 2\le n\le6,
\]
then the two matched-mean Poisson entropy expansions have the same
eighth-order jet:
\[
D_{\mathrm{KL}}(P_t*\nu\|P_t(\cdot-\bar\gamma))
-D_{\mathrm{KL}}(P_t*\widetilde\nu\|P_t(\cdot-\widetilde{\bar\gamma}))
=O_{\nu,\widetilde\nu}(t^{-9}).
\]
Equivalently, the seventh-moment coefficients $C_4$, $C_6$, and
$C_8^{(7)}$ in Theorem~\ref{thm:poisson-kl-eighth}, and hence the defect
coordinates $C_6$, $C_8^{(7)}$, $\Delta_6$, and $\Delta_8$ used in
Corollaries~\ref{thm:poisson-defect-ladder}
and~\ref{thm:poisson-defect-amplification}, factor through the finite
centred moment jet $(\mu_2,\ldots,\mu_6)$.  Higher moments enter this
eighth-order consequence only through the validity and size of the
order-nine remainder.
\end{corollary}

\begin{proof}
Apply Theorem~\ref{thm:poisson-kl-eighth} to both laws.  The displayed
formula \eqref{eq:poisson-kl-eighth} expresses the $t^{-4}$ coefficient as
\(\mu_2^2/8\), the $t^{-6}$ coefficient as
\((\mu_2^3+6\mu_3^2-8\mu_2\mu_4)/64\), and the $t^{-8}$ coefficient as
\[
\frac{3\mu_2^4-12\mu_2^2\mu_4+9\mu_2\mu_3^2
+12\mu_2\mu_6-30\mu_3\mu_5+20\mu_4^2}{256}.
\]
These three polynomials involve only the centred moments
\(\mu_2,\ldots,\mu_6\).  Under the hypothesis
\(\mu_n=\widetilde\mu_n\) for $2\le n\le6$, the retained coefficients in the
two expansions are therefore equal.  Subtracting the two estimates
\eqref{eq:poisson-kl-eighth-remainder}, and using
$\max\{B_{\nu,8},B_{\widetilde\nu,8}\}$ after increasing the lower time
threshold if necessary, gives the stated
$O_{\nu,\widetilde\nu}(t^{-9})$ difference.

The expressions defining $C_6$ and $C_8^{(7)}$ are precisely the $t^{-6}$ and
$t^{-8}$ coefficients just identified, and, in the seventh-moment regime,
\(\Delta_6=-7\mu_2^3/64-C_6\),
\(\Delta_8=C_8^{(7)}-23\mu_2^4/256\).  Hence these defect variables also depend
only on $(\mu_2,\ldots,\mu_6)$.  The seventh absolute moments are used by
Theorem~\ref{thm:poisson-kl-eighth} to produce the remainders, so they
affect the constants and threshold but not the printed eighth-order jet.
\end{proof}

The finite phase verification of the printed eighth-order coefficient
arithmetic is recorded in
Proposition~\ref{prop:eighth-order-finite-phase-verification}.  It is an
appendix certificate for the constant-coefficient calculations above, not an
additional ingredient in the entropy expansion.

The next two defect corollaries use $C_6$ and the actual seventh-moment
coefficient $C_8^{(7)}(\nu)$ under the finite centred seventh absolute moment
hypothesis of Theorem~\ref{thm:poisson-kl-eighth}.  For readability inside
these corollaries we define a local abbreviation $C_8:=C_8^{(7)}(\nu)$ after
the hypothesis is in force.  The fourth-moment algebra in
Subsection~\ref{subsec:fourth-moment-stability} keeps separate superscripted
notation until the corresponding entropy hypothesis is imposed.
They are downstream defect consequences, not theorem-level replacements for the
eighth-order jet.

\begin{corollary}[A two-level defect ladder]%
\label{thm:poisson-defect-ladder}
Let $\nu$ be a probability measure on~$\RR$ with mean~$\bar\gamma$,
variance $\sigma^2>0$, and finite centred seventh absolute moment.  Define
\begin{align*}
C_6&:=\frac{\sigma^6+6\mu_3^2-8\sigma^2\mu_4}{64}, \\
C_8&:=C_8^{(7)}(\nu):=
\frac{3\sigma^8-12\sigma^4\mu_4+9\sigma^2\mu_3^2+12\sigma^2\mu_6}{256}
+\frac{-30\mu_3\mu_5+20\mu_4^2}{256}.
\end{align*}
Let $X:=(\gamma-\bar\gamma)/\sigma$ on the probability space
$(\RR,\nu)$, set
\[
\beta_n:=\EE[X^n],\qquad n\in\NN,\ n\ge1,\qquad
\beta_3:=\EE[X^3],
\qquad
Q_2:=X^2-1-\beta_3X,
\]
and define
\[
Q_3:=XQ_2-\|Q_2\|_{L^2(\nu)}^2X-\frac{\beta_3}{4}Q_2.
\]
Then
\begin{equation}\label{eq:defect-c6}
\frac{64}{\sigma^6}C_6
=-7-2\beta_3^2-8\|Q_2\|_{L^2(\nu)}^2,
\end{equation}
and
\begin{equation}\label{eq:defect-c8}
\frac{256}{\sigma^8}C_8
=23
+12\|Q_3\|_{L^2(\nu)}^2
+32\|Q_2\|_{L^2(\nu)}^4
+\frac{61}{4}\beta_3^2\|Q_2\|_{L^2(\nu)}^2
+52\|Q_2\|_{L^2(\nu)}^2
+2\beta_3^4
+13\beta_3^2.
\end{equation}
In particular
\begin{equation}\label{eq:defect-c8-lower}
C_8\ge \frac{23}{256}\sigma^8,
\end{equation}
and equality holds if and only if
\[
\nu=\frac12\delta_{\bar\gamma+\sigma}
+\frac12\delta_{\bar\gamma-\sigma}.
\]
The identity \eqref{eq:defect-c6} recovers the classical Pearson
moment relation, while \eqref{eq:defect-c8} gives the next higher-order
component in the same defect coordinates.
\end{corollary}

\begin{proof}
Write
\[
\kappa:=\|Q_2\|_{L^2(\nu)}^2=\EE[Q_2^2],
\qquad
L:=\EE[XQ_2^2],
\qquad
M:=\EE[Q_2^3].
\]
Since $Q_2=X^2-1-\beta_3X$, one has
\[
\kappa=\beta_4-1-\beta_3^2,
\]
and therefore
\[
\frac{64}{\sigma^6}C_6
=1+6\beta_3^2-8\beta_4
=-7-2\beta_3^2-8\kappa,
\]
which is exactly \eqref{eq:defect-c6}.

The identity $X^2=1+\beta_3X+Q_2$ yields
\[
\beta_5=\beta_3(\beta_3^2+2+2\kappa)+L,
\]
\[
\beta_6=1+3\beta_3^2+3\kappa+\beta_3^4+3\beta_3^2\kappa+3\beta_3L+M.
\]
Hence, if
\[
P_8:=3-12\beta_4+9\beta_3^2+12\beta_6-30\beta_3\beta_5+20\beta_4^2,
\]
then a direct substitution gives
\[
P_8
=23+20\kappa^2+16\beta_3^2\kappa+64\kappa+2\beta_3^4+13\beta_3^2+6\beta_3L+12M.
\]
On the other hand,
using $\EE[XQ_2]=0$, $\EE[Q_2]=0$, $\EE[X^2Q_2^2]=\kappa+\beta_3L+M$, and
$\EE[XQ_2^2]=L$, the definition
$Q_3=XQ_2-\kappa X-(\beta_3/4)Q_2$ gives
\[
\|Q_3\|_{L^2(\nu)}^2
=M+\frac{\beta_3}{2}L+\kappa-\kappa^2+\frac{\beta_3^2}{16}\kappa.
\]
Equivalently,
\[
12\|Q_3\|_{L^2(\nu)}^2
=12M+6\beta_3L+12\kappa-12\kappa^2+\frac34\beta_3^2\kappa.
\]
Substituting this last line into the preceding expression for $P_8$ gives
the exact regrouping
\[
P_8=23+12\|Q_3\|_2^2+32\kappa^2
+\frac{61}{4}\beta_3^2\kappa+52\kappa+2\beta_3^4+13\beta_3^2,
\]
which is \eqref{eq:defect-c8}.  Since
$P_8=256C_8/\sigma^8$, inequality \eqref{eq:defect-c8-lower} follows.

If equality holds in \eqref{eq:defect-c8-lower}, then every non-negative
term on the right-hand side of \eqref{eq:defect-c8} vanishes.  In
particular $\beta_3=0$ and $\|Q_2\|_{L^2(\nu)}=0$, so $X^2=1$
$\nu$-almost everywhere.  Since $\EE[X]=0$ and $\EE[X^2]=1$, it follows
that $X=\pm 1$ with equal weights.  Thus
\[
\nu=\frac12\delta_{\bar\gamma+\sigma}
+\frac12\delta_{\bar\gamma-\sigma}.
\]
The converse is immediate.
\end{proof}

\begin{corollary}[Defect amplification]\label{thm:poisson-defect-amplification}
Let $\nu$ be a probability measure on~$\RR$ with mean~$\bar\gamma$,
variance $\sigma^2>0$, and finite centred seventh absolute moment.  Define
\[
\Delta_6(\nu):=-\frac{7}{64}\sigma^6-C_6,
\qquad
\Delta_8(\nu):=C_8-\frac{23}{256}\sigma^8,
\]
where $C_6$ is the $t^{-6}$ entropy coefficient and
$C_8:=C_8^{(7)}(\nu)$ is the actual seventh-moment $t^{-8}$ entropy
coefficient from Theorem~\ref{thm:poisson-kl-eighth}.  Set
\[
X:=\frac{\gamma-\bar\gamma}{\sigma},
\qquad
\beta_3:=\EE[X^3],
\qquad
Q_2:=X^2-1-\beta_3X,
\]
\[
\kappa:=\|Q_2\|_{L^2(\nu)}^2,
\qquad
Q_3:=XQ_2-\kappa X-\frac{\beta_3}{4}Q_2.
\]
Then
\[
\Delta_6(\nu)
=\frac{\sigma^6}{32}\beta_3^2+\frac{\sigma^6}{8}\kappa,
\]
and
\begin{align}
\Delta_8(\nu)
&=
\frac{13\sigma^2}{8}\Delta_6(\nu)
\notag\\
&\quad
+\frac{\sigma^8}{256}\left(
12\|Q_3\|_{L^2(\nu)}^2
+32\kappa^2
+\frac{61}{4}\beta_3^2\kappa
+2\beta_3^4
\right). \label{eq:defect-amplification}
\end{align}
In particular,
\[
\Delta_8(\nu)\ge \frac{13\sigma^2}{8}\Delta_6(\nu),
\]
and equality in this proved sixth/eighth-order defect layer holds if and only if
\[
\nu=\frac12\delta_{\bar\gamma+\sigma}
+\frac12\delta_{\bar\gamma-\sigma}.
\]
\end{corollary}

\begin{proof}
Equation~\eqref{eq:defect-c6} gives
\[
\Delta_6(\nu)
=-\frac{\sigma^6}{64}\left(7+\frac{64}{\sigma^6}C_6\right)
=\frac{\sigma^6}{32}\beta_3^2+\frac{\sigma^6}{8}\kappa.
\]
Likewise, subtracting $23$ from \eqref{eq:defect-c8} yields
\[
\Delta_8(\nu)
=\frac{\sigma^8}{256}\Bigl(
12\|Q_3\|_{L^2(\nu)}^2
+32\kappa^2
+\frac{61}{4}\beta_3^2\kappa
+52\kappa
+2\beta_3^4
+13\beta_3^2
\Bigr).
\]
By the preceding formula for $\Delta_6(\nu)$,
\[
\frac{13\sigma^2}{8}\Delta_6(\nu)
=\frac{\sigma^8}{256}\bigl(52\kappa+13\beta_3^2\bigr),
\]
and substituting this identity proves \eqref{eq:defect-amplification}.

If equality holds in the lower bound, then every non-negative term in
\eqref{eq:defect-amplification} vanishes.  In particular,
$\beta_3=0$ and $\kappa=0$, hence $Q_2=X^2-1$ vanishes
$\nu$-almost everywhere.  Since $\EE[X]=0$ and $\EE[X^2]=1$, it follows
that $X=\pm 1$ with equal weights, which is exactly the stated
two-point law.  The converse is immediate.
\end{proof}

\begin{corollary}[Symmetric two-point entropy-defect rigidity]%
\label{cor:poisson-two-point-defect-rigidity}
Let $\nu$ be a probability measure on~$\RR$ with mean~$\bar\gamma$,
variance $\sigma^2>0$, and finite centred seventh absolute moment.  Let
$C_6$ be the $t^{-6}$ entropy coefficient and put
$C_8:=C_8^{(7)}(\nu)$ for the actual seventh-moment $t^{-8}$ coefficient
from Theorem~\ref{thm:poisson-kl-eighth}.  Define
\[
\Delta_6(\nu):=-\frac{7}{64}\sigma^6-C_6,
\qquad
\Delta_8(\nu):=C_8-\frac{23}{256}\sigma^8.
\]
Let $\nu_c$ be the centred law of $\gamma-\bar\gamma$ and set
\[
\rho_\sigma:=\frac12(\delta_{-\sigma}+\delta_\sigma).
\]
Then, among all laws with the same mean and variance in this moment class,
\[
C_6\le -\frac{7}{64}\sigma^6,
\qquad
C_8\ge \frac{23}{256}\sigma^8.
\]
Moreover the following conditions are equivalent:
\begin{enumerate}[label=\textup{(\roman*)},leftmargin=*]
\item $\nu_c=\rho_\sigma$, equivalently
$\nu=\frac12\delta_{\bar\gamma-\sigma}
+\frac12\delta_{\bar\gamma+\sigma}$;
\item $C_6=-\frac{7}{64}\sigma^6$;
\item $C_8=\frac{23}{256}\sigma^8$;
\item $\Delta_6(\nu)=\Delta_8(\nu)=0$;
\item the amplification bound is sharp,
\[
\Delta_8(\nu)=\frac{13\sigma^2}{8}\Delta_6(\nu).
\]
\end{enumerate}
Thus, at the proved sixth/eighth-order layer, and with no assertion here about
an unproved all-order defect hierarchy, the symmetric two-point law is
the unique entropy-defect extremizer and the unique rigidity class for the
amplification chain.
\end{corollary}

\begin{proof}
Use the notation of Corollaries~\ref{thm:poisson-defect-ladder}
and~\ref{thm:poisson-defect-amplification}.  From
\eqref{eq:defect-c6},
\[
C_6+\frac{7}{64}\sigma^6
=-\frac{\sigma^6}{32}\beta_3^2-\frac{\sigma^6}{8}\kappa
\le0,
\qquad
\kappa:=\|Q_2\|_{L^2(\nu)}^2.
\]
Hence $C_6$ is maximized by, and only by, laws for which
$\beta_3=0$ and $\kappa=0$.  In that case
$Q_2=X^2-1$ vanishes almost surely, so $X=\pm1$ almost surely; since
$\EE X=0$ and $\EE X^2=1$, the two signs have equal mass.  This is exactly
$\nu_c=\rho_\sigma$.  Conversely, for $\nu_c=\rho_\sigma$ one has
$\beta_3=0$ and $Q_2=0$, so equality holds in the displayed bound for
$C_6$.

The lower bound for $C_8$ and the characterization of equality in that
bound are precisely \eqref{eq:defect-c8-lower} and the equality statement
in Corollary~\ref{thm:poisson-defect-ladder}.  The definitions of
$\Delta_6$ and $\Delta_8$ convert equality in the two coefficient bounds
into $\Delta_6(\nu)=0$ and $\Delta_8(\nu)=0$, respectively.  Finally,
Corollary~\ref{thm:poisson-defect-amplification} states that equality in
\[
\Delta_8(\nu)\ge\frac{13\sigma^2}{8}\Delta_6(\nu)
\]
holds if and only if $\nu_c=\rho_\sigma$.  Combining these equivalences
proves all the listed conditions equivalent.
\end{proof}
