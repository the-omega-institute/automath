\subsection{Eighth-order defect layer extracted from the all-order calculus}%
\label{subsec:eighth-order-defects}

The preceding two subsections separate the formal coefficient polynomial from
the finite-moment remainder theorem.  This block specializes the same formal
definitions through order $t^{-8}$, verifies the required mode integrals by
finite phase averaging, and rewrites the sixth- and eighth-order coefficients in
non-negative defect coordinates.  The standing hypothesis here is the finite
centred seventh absolute moment used in the direct mode-remainder proof of
Corollary~\ref{cor:poisson-kl-eighth}.  This is weaker than the $N=4$
hypothesis of Theorem~\ref{thm:poisson-kl-even-orders}, which asks for a finite
centred $(8+\eta)$th absolute moment; compact-support analytic interchange is
not used in this subsection.  The finite phase calculations certify the
constants, and the direct seventh-moment estimate supplies their entropy
meaning.


\begin{corollary}[Eighth-order coefficient corollary under seventh moments]%
\label{cor:poisson-kl-eighth}
Let $\nu$ be a probability measure on~$\RR$ with mean~$\bar\gamma$,
variance $\sigma^2>0$, and finite centred seventh absolute moment.  Then,
with
\[
\Delta:=\gamma-\bar\gamma,
\qquad
\mu_n:=\int_{\RR}\Delta^n\dd\nu(\gamma),
\]
and
\[
h_t:=P_t*\nu,
\qquad
g_t(x):=P_t(x-\bar\gamma),
\]
one has, as $t\to\infty$,
\begin{equation}\label{eq:poisson-kl-eighth}
\begin{split}
D_{\mathrm{KL}}(h_t\|g_t)
=\frac{\sigma^4}{8t^4}
+\frac{\sigma^6+6\mu_3^2-8\sigma^2\mu_4}{64\,t^6}
\\
\quad
+\frac{3\sigma^8-12\sigma^4\mu_4+9\sigma^2\mu_3^2}{256\,t^8}
+\frac{12\sigma^2\mu_6-30\mu_3\mu_5+20\mu_4^2}{256\,t^8}
+O_\nu(t^{-9}).
\end{split}
\end{equation}
More precisely, there are constants $B_{\nu,8},T_{\nu,8}<\infty$ such
that, for all $t\ge T_{\nu,8}$,
\begin{equation}\label{eq:poisson-kl-eighth-remainder}
\begin{split}
\Bigg|D_{\mathrm{KL}}(h_t\|g_t)
&-\frac{\sigma^4}{8t^4}
-\frac{\sigma^6+6\mu_3^2-8\sigma^2\mu_4}{64\,t^6}
\\
&-\frac{3\sigma^8-12\sigma^4\mu_4+9\sigma^2\mu_3^2
+12\sigma^2\mu_6-30\mu_3\mu_5+20\mu_4^2}{256\,t^8}
\Bigg|
\le B_{\nu,8}t^{-9}.
\end{split}
\end{equation}
\end{corollary}

\begin{proof}
The products and parity cancellations are those of
Theorem~\ref{thm:universal-entropy-coefficients}, and the actual mode integrals
are evaluated by Theorem~\ref{thm:finite-fourier-mode-integral-algorithm}.
What is special here is the lower-moment asymptotic bridge.  The $N=4$ case of
Theorem~\ref{thm:poisson-kl-even-orders} gives the same coefficient notation
only under the stronger finite centred $(8+\eta)$th absolute moment hypothesis;
the present remainder estimate is the direct seventh-moment estimate below.

Let $\omega(\dd y)=\pi^{-1}(1+y^2)^{-1}\dd y$.  With
$x=\bar\gamma+ty$, Lemma~\ref{lem:tail-uniform-cayley-mode-remainder}
with $X=\Delta$ and truncation order~$6$ gives
\[
\delta_t(\bar\gamma+ty)
=\frac{\sigma^2}{t^2}u_2(y)
+\frac{\mu_3}{t^3}u_3(y)
+\frac{\mu_4}{t^4}u_4(y)
+\frac{\mu_5}{t^5}u_5(y)
+\frac{\mu_6}{t^6}u_6(y)
+\theta_t(y),
\]
where, if $v_6$ is the bounded remainder from
Proposition~\ref{prop:cayley-mode-remainder}, then
\[
\|\theta_t\|_{L^\infty(\RR)}
+\|\theta_t\|_{L^1(\omega)}
\le
\|v_6\|_{L^\infty(\RR^2)}\,\EE|\Delta|^7\,t^{-7}.
\]
Also
$g_t(\bar\gamma+ty)\dd(\bar\gamma+ty)=\omega(\dd y)$, hence
\[
D_{\mathrm{KL}}(h_t\|g_t)
=\int_{\RR}\Phi(\delta_t(\bar\gamma+ty))\,\omega(\dd y).
\]
Set
\[
a_2:=\sigma^2\,u_2,\qquad a_3:=\mu_3\,u_3,\qquad a_4:=\mu_4\,u_4,
\qquad a_5:=\mu_5\,u_5,\qquad a_6:=\mu_6\,u_6.
\]
Write
\[
d_t:=t^{-2}a_2+t^{-3}a_3+t^{-4}a_4+t^{-5}a_5+t^{-6}a_6,
\qquad
\delta_t=d_t+\theta_t,
\]
with the same $\theta_t$ as above.  The finite seventh absolute moment is
used exactly here: it makes the omitted mode $\theta_t$ small enough that
any occurrence of it lies past order~$t^{-8}$.  For the quantitative
remainder, set
\[
U_n:=\|u_n\|_{L^\infty(\RR)}\quad(2\le n\le6),\qquad
V_6:=\|v_6\|_{L^\infty(\RR^2)},
\]
\[
A_\nu:=\sum_{n=2}^{6}|\mu_n|U_n,\qquad
M_\nu:=V_6\,\EE|\Delta|^7,\qquad H_\nu:=A_\nu+M_\nu .
\]
Then, for $t\ge1$,
\[
\|d_t\|_\infty\le A_\nu t^{-2},\qquad
\|\theta_t\|_\infty+\|\theta_t\|_{L^1(\omega)}
\le M_\nu t^{-7},\qquad
\|\delta_t\|_\infty\le H_\nu t^{-2}.
\]
Let
\[
K_\Phi:=\sum_{j=5}^{\infty}\frac{2^{-(j-5)}}{j(j-1)},
\qquad
T_{\nu,8}:=\max\{1,(2H_\nu)^{1/2}\}.
\]
For $t\ge T_{\nu,8}$,
\[
\left|\Phi(\delta_t)
-\left(\frac12\delta_t^2-\frac16\delta_t^3
+\frac1{12}\delta_t^4\right)\right|
\le K_\Phi H_\nu^5t^{-10}.
\]
For $k=2,3,4$ put
\[
S_{k,\nu}:=
\sum_{\substack{2\le n_1,\dots,n_k\le6\\ n_1+\cdots+n_k\ge9}}
\prod_{j=1}^{k}|\mu_{n_j}|U_{n_j}.
\]
The discarded terms in the exact finite product $d_t^k$ after total
$t$-degree eight are bounded by $S_{k,\nu}t^{-9}$ in $L^\infty$.  Moreover
\[
E_{2,\nu}:=2A_\nu M_\nu+M_\nu^2,\qquad
E_{3,\nu}:=3A_\nu^2M_\nu+3A_\nu M_\nu^2+M_\nu^3,
\]
\[
E_{4,\nu}:=4A_\nu^3M_\nu+6A_\nu^2M_\nu^2
+4A_\nu M_\nu^3+M_\nu^4
\]
give, for $t\ge1$, the following explicit bounds:
\begin{align*}
\|\delta_t^2-d_t^2\|_\infty
&\le 2A_\nu M_\nu t^{-9}+M_\nu^2t^{-14}
\le E_{2,\nu}t^{-9},\\
\|\delta_t^3-d_t^3\|_\infty
&\le 3A_\nu^2M_\nu t^{-11}+3A_\nu M_\nu^2t^{-16}
+M_\nu^3t^{-21}
\le E_{3,\nu}t^{-9},\\
\|\delta_t^4-d_t^4\|_\infty
&\le 4A_\nu^3M_\nu t^{-13}+6A_\nu^2M_\nu^2t^{-18}
+4A_\nu M_\nu^3t^{-23}+M_\nu^4t^{-28}
\le E_{4,\nu}t^{-9}.
\end{align*}
Thus the sole order-$t^{-9}$ contribution involving the seventh-moment
remainder is the quadratic cross term $2d_t\theta_t$; all higher powers that
contain $\theta_t$ are strictly smaller.
Consequently one may take
\begin{equation}\label{eq:eighth-remainder-constant}
B_{\nu,8}:=
\frac12(S_{2,\nu}+E_{2,\nu})
+\frac16(S_{3,\nu}+E_{3,\nu})
+\frac1{12}(S_{4,\nu}+E_{4,\nu})
+K_\Phi H_\nu^5 .
\end{equation}
This bound is used in the quantitative large-time corollary below.  It remains
to identify the coefficients produced by the finite product pattern.
Since
the preceding estimates imply $\delta_t=O(t^{-2})$, $d_t=O(t^{-2})$, and
$\theta_t=O(t^{-7})$ uniformly, we have
\[
\delta_t^2=d_t^2+2d_t\theta_t+\theta_t^2=d_t^2+O(t^{-9}),
\]
while
\[
\delta_t^3=d_t^3+O(t^{-11}),\qquad
\delta_t^4=d_t^4+O(t^{-13}).
\]
Combining these estimates with the Taylor remainder for~$\Phi$ gives the
following uniform rule for coefficients:
\[
\begin{aligned}
[t^{-m}]\Phi(\delta_t)
&=[t^{-m}]\bigl(\tfrac12d_t^2-\tfrac16d_t^3+\tfrac1{12}d_t^4\bigr),
\\
&\hspace{5em}4\le m\le8.
\end{aligned}
\]
Here coefficient extraction is understood modulo a uniform $O(t^{-9})$
remainder.  This is the only point in the proof where the seventh absolute
moment is needed beyond the lower moments appearing in the displayed
coefficients.  Equivalently,
\[
\Phi(\delta_t)
=\frac12d_t^2-\frac16d_t^3+\frac1{12}d_t^4+O(t^{-9})
=\sum_{m=4}^{8}t^{-m}B_m(y)+O(t^{-9})
\]
uniformly in $y$.  The coefficient functions are therefore obtained by
expanding only the following finite products in powers of~$t^{-1}$.  No
term containing $\theta_t$ can contribute to any displayed coefficient, and
all terms of total order larger than eight are absorbed in the uniform
$O(t^{-9})$ remainder:
\begin{align*}
d_t^2
&=t^{-4}a_2^2+2t^{-5}a_2a_3
+t^{-6}(2a_2a_4+a_3^2) \\
&\quad
+t^{-7}(2a_2a_5+2a_3a_4)
+t^{-8}(2a_2a_6+2a_3a_5+a_4^2)+O(t^{-9}),\\
d_t^3
&=t^{-6}a_2^3+3t^{-7}a_2^2a_3
+t^{-8}(3a_2^2a_4+3a_2a_3^2)+O(t^{-9}),\\
d_t^4
&=t^{-8}a_2^4+O(t^{-9}).
\end{align*}
Thus
\begin{align*}
B_4&=\frac12a_2^2,\\
B_5&=a_2a_3,\\
B_6&=a_2a_4+\frac12a_3^2-\frac16a_2^3,\\
B_7&=a_2a_5+a_3a_4-\frac12a_2^2a_3,\\
B_8&=a_2a_6+a_3a_5+\frac12a_4^2
-\frac12a_2^2a_4-\frac12a_2a_3^2+\frac1{12}a_2^4.
\end{align*}
The definitions of $S_{k,\nu}$, $E_{k,\nu}$, and $K_\Phi$ above give,
for $t\ge T_{\nu,8}$,
\[
\left\|
\Phi(\delta_t)-\sum_{m=4}^{8}t^{-m}B_m
\right\|_{L^\infty(\RR)}
\le B_{\nu,8}t^{-9}.
\]
Thus the extraction rule gives $[t^{-m}]\Phi(\delta_t)=B_m$ for
$4\le m\le8$.  In particular, the $t^{-8}$ coefficient is exactly the sum
of three contributions:
\[
\frac12[t^{-8}]d_t^2
=a_2a_6+a_3a_5+\frac12a_4^2,
\]
\[
-\frac16[t^{-8}]d_t^3
=-\frac12a_2^2a_4-\frac12a_2a_3^2,
\qquad
\frac1{12}[t^{-8}]d_t^4=\frac1{12}a_2^4.
\]
The functions $B_5$ and $B_7$ are sums of products
$u_{n_1}\cdots u_{n_k}$ with odd total index, so
Lemma~\ref{lem:mode-parity-vanish} gives
\[
\int_{\RR}B_5\,\dd\omega=\int_{\RR}B_7\,\dd\omega=0.
\]
The remaining coefficients are obtained by the finite Fourier algorithm,
Theorem~\ref{thm:finite-fourier-mode-integral-algorithm}.  The explicit
constant-term substitutions for the products in $B_4$, $B_6$, and $B_8$ are
collected in Appendix~\ref{app:entropy-integrals}; equivalently,
Lemma~\ref{lem:eighth-order-coefficient-arithmetic} gives
\[
\int_{\RR}B_4\,\dd\omega=\frac{\sigma^4}{8},
\qquad
\int_{\RR}B_6\,\dd\omega
=\frac{\sigma^6+6\mu_3^2-8\sigma^2\mu_4}{64},
\]
\[
\int_{\RR}B_8\,\dd\omega
=\frac{3\sigma^8-12\sigma^4\mu_4+9\sigma^2\mu_3^2}{256}
+\frac{12\sigma^2\mu_6-30\mu_3\mu_5+20\mu_4^2}{256}.
\]
Since $\omega(\RR)=1$, the explicit uniform bound
\eqref{eq:eighth-remainder-constant} gives
\eqref{eq:poisson-kl-eighth-remainder}.  Substituting the three displayed
coefficients proves \eqref{eq:poisson-kl-eighth}.
\end{proof}

\begin{corollary}[Seventh-moment bridge to the all-order coefficient chain]%
\label{cor:seventh-moment-eighth-bridge}
Under the hypotheses of Corollary~\ref{cor:poisson-kl-eighth}, the coefficients
of $t^{-4}$, $t^{-6}$, and $t^{-8}$ in
\eqref{eq:poisson-kl-eighth} are precisely the universal evaluations
$A_4(\nu)$, $A_6(\nu)$, and $A_8(\nu)$ from
\eqref{eq:universal-entropy-coefficient-evaluation}, while the odd
evaluations $A_5(\nu)$ and $A_7(\nu)$ vanish.  Consequently the defect
coordinates used in Theorems~\ref{thm:poisson-defect-ladder}
and~\ref{thm:poisson-defect-amplification} lie on the same coefficient chain as
Theorem~\ref{thm:poisson-kl-even-orders}, but their eighth-order finite-moment
asymptotic justification is the direct seventh-moment remainder
\eqref{eq:poisson-kl-eighth-remainder}.  If, in addition,
$\int|\gamma-\bar\gamma|^{8+\eta}\dd\nu(\gamma)<\infty$ for some
$0<\eta\le1$, then the same eighth-order jet is the $N=4$ truncation of
Theorem~\ref{thm:poisson-kl-even-orders}.
\end{corollary}

\begin{proof}
The proof of Corollary~\ref{cor:poisson-kl-eighth} gives, with an
$L^\infty(\RR)$ error bounded by $B_{\nu,8}t^{-9}$,
\[
\Phi(\delta_t(\bar\gamma+ty))
=\sum_{m=4}^{8}t^{-m}B_m(y)+O_\nu(t^{-9}),
\]
where the functions $B_m$ are the finite products displayed in that proof.
After integration against $\omega$, the definitions of $B_4$, $B_6$, and $B_8$
are exactly the degree-$4$, degree-$6$, and degree-$8$ terms obtained by
substituting the finite mode expansion $\sum_{n=2}^{6}\mu_n\,u_n\,t^{-n}$ into
\[
\Phi(z)=\sum_{r\ge2}\frac{(-1)^r}{r(r-1)}z^r
\]
and retaining total degree at most eight.  This is precisely the finite
evaluation formula \eqref{eq:universal-entropy-coefficient-evaluation}; the
equality with the displayed polynomials follows from the finite Fourier
algorithm, Theorem~\ref{thm:finite-fourier-mode-integral-algorithm}, and the
low-order arithmetic recorded in Appendix~\ref{app:entropy-integrals}.  The
same comparison shows that the degree-five and degree-seven finite evaluations
are $\int B_5\,\dd\omega$ and $\int B_7\,\dd\omega$, and these vanish by
Lemma~\ref{lem:mode-parity-vanish}.  Hence the defect coordinates built from
the displayed $t^{-6}$ and $t^{-8}$ coefficients use the universal coefficient
polynomials while relying on the seventh-moment bound
\eqref{eq:poisson-kl-eighth-remainder} for the asymptotic remainder.  Under
the stronger finite $(8+\eta)$th absolute moment hypothesis,
Theorem~\ref{thm:poisson-kl-even-orders} applies with $N=4$ and gives the same
coefficient evaluations through order $t^{-8}$, proving the final assertion.
\end{proof}

\begin{proposition}[Subordinate eighth-order coefficient jet]%
\label{prop:eighth-order-jet-subordinate}
Let
\[
\mathcal J_8(\nu):=(A_4(\nu),A_6(\nu),A_8(\nu))
\]
denote the even universal coefficient jet through order eight whenever the
displayed evaluations exist.  Under the finite centred seventh absolute moment
hypotheses of Corollary~\ref{cor:poisson-kl-eighth}, the printed coefficients
in \eqref{eq:poisson-kl-eighth} are exactly $\mathcal J_8(\nu)$ and the
remainder is justified by the seventh-moment estimate
\eqref{eq:poisson-kl-eighth-remainder}.  Under the stronger finite centred
$(8+\eta)$th absolute moment hypothesis, the same jet is the $N=4$
specialization of the all-order finite-range expansion
Theorem~\ref{thm:poisson-kl-even-orders}.  Thus the eighth-order formula is a
low-order finite-moment consequence of
Theorem~\ref{thm:universal-entropy-coefficients} and
Theorem~\ref{thm:poisson-kl-even-orders}; it supplies the first defect layer
for the later variational results but does not replace the all-order
coefficient theorem.
\end{proposition}

\begin{proof}
The first assertion is exactly
Corollary~\ref{cor:seventh-moment-eighth-bridge}: after the seventh-moment mode
remainder has removed all terms beyond order eight, the surviving degree-four,
degree-six, and degree-eight products are the universal coefficient products
from \eqref{eq:universal-entropy-coefficient}.  The finite Fourier algorithm
then computes their Cauchy mode integrals, producing the three displayed
coefficients in \eqref{eq:poisson-kl-eighth}; the direct bound
\eqref{eq:poisson-kl-eighth-remainder} is the asymptotic input at this lower
moment level.  If $\int|\gamma-\bar\gamma|^{8+\eta}\dd\nu<\infty$, then
Theorem~\ref{thm:poisson-kl-even-orders} applies with $N=4$ and gives
\[
D_{\mathrm{KL}}(P_t*\nu\|P_t(\cdot-\bar\gamma))
=A_4(\nu)t^{-4}+A_6(\nu)t^{-6}+A_8(\nu)t^{-8}+o(t^{-8}),
\]
with all odd coefficients absent by
Theorem~\ref{thm:universal-entropy-coefficients}.  Hence the same ordered
triple $\mathcal J_8(\nu)$ is obtained either as the $N=4$ truncation of the
all-order finite-moment theorem or, under the weaker seventh-moment hypothesis,
as the direct low-order bridge.  The final sentence is therefore a restatement
of the proved implications: the universal coefficient theorem supplies the
coefficient algebra, the all-order theorem supplies the general asymptotic
statement, and Corollary~\ref{cor:poisson-kl-eighth} supplies only the first
explicit lower-moment jet used by the defect theorems.
\end{proof}

\begin{lemma}[Separation of entropy coefficient regimes]%
\label{lem:entropy-coefficient-regime-separation}
The entropy coefficients used in this section involve the following distinct
hypotheses.
\begin{enumerate}[label=\textup{(\roman*)},leftmargin=*]
\item The formal coefficients $A_k^{\rm form}$ and the parity cancellation are
      determined before any law is chosen by
      Theorem~\ref{thm:universal-entropy-coefficients} and the finite Fourier
      algorithm of Theorem~\ref{thm:finite-fourier-mode-integral-algorithm}.
\item If $\nu$ has a finite centred $(2N+\eta)$th absolute moment, then the
      evaluated coefficients $A_{2m}(\nu)$, $2\le m\le N$, are the actual
      entropy-asymptotic coefficients through order $t^{-2N}$, with the
      fixed-law finite-moment remainder of
      Theorem~\ref{thm:finite-moment-entropy-remainder}.  If the centred law is
      compactly supported, the same coefficients are the analytic Taylor
      coefficients of the convergent entropy series in
      Theorem~\ref{thm:analytic-compact-support-interchange}; this is a
      stronger analytic argument, not an extra hypothesis in the finite-moment
      theorem.
\item The eighth-order corollary and the defect ladder use the finite centred
      seventh absolute moment hypothesis of Corollary~\ref{cor:poisson-kl-eighth}.
      Under this hypothesis, the displayed $t^{-4}$, $t^{-6}$, and $t^{-8}$
      coefficients are the universal evaluations $A_4(\nu)$, $A_6(\nu)$, and
      $A_8(\nu)$, although this assertion need not be an instance of
      Theorem~\ref{thm:poisson-kl-even-orders} with $N=4$ and need not come
      from compact-support analytic interchange.
\item The seventh-moment route is an independent eighth-order theorem route:
      the implication
      \[
      \EE|\gamma-\bar\gamma|^7<\infty
      \quad\Longrightarrow\quad
      \eqref{eq:poisson-kl-eighth}\text{ and }\eqref{eq:poisson-kl-eighth-remainder}
      \]
      does not factor through the $N=4$ finite-moment theorem.  In particular
      the hypotheses in Corollary~\ref{cor:poisson-kl-eighth} are neither
      shorthand for, nor a consequence of, the finite centred
      $(8+\eta)$th absolute moment assumption used by
      Theorem~\ref{thm:poisson-kl-even-orders}.
\end{enumerate}
The separation in \textup{(iii)} is sharp: for every $0<\eta\le1$ there are
centred probability laws with positive variance and finite seventh absolute
moment for which Corollary~\ref{cor:poisson-kl-eighth},
Theorem~\ref{thm:poisson-defect-ladder}, and
Theorem~\ref{thm:poisson-defect-amplification} apply, while the $N=4$ case of
Theorem~\ref{thm:poisson-kl-even-orders} with exponent $8+\eta$ and the compact
support theorem both do not apply.
\end{lemma}

\begin{proof}
Part~\textup{(i)} is exactly the law-free construction of
\eqref{eq:universal-entropy-coefficient}: the mode integrals are finite
Cauchy-weighted integrals of Cayley--Chebyshev modes, the finite Fourier
algorithm computes them by Laurent coefficient extraction, and
Theorem~\ref{thm:universal-entropy-coefficients} proves that the odd formal
coefficients vanish by parity.

For~\textup{(ii)}, Proposition~\ref{prop:universal-entropy-coefficient-evaluation}
evaluates the formal polynomials whenever the required moments exist.
Theorem~\ref{thm:poisson-kl-even-orders} then identifies these evaluations
with the actual finite-moment entropy coefficients through order $t^{-2N}$,
and Theorem~\ref{thm:finite-moment-entropy-remainder} records the same
identification together with the explicit fixed-law remainder constants.  If
the centred law is compactly supported, the dominated-convergence and absolute
summability argument in Theorem~\ref{thm:analytic-compact-support-interchange}
gives a convergent analytic entropy series with the same coefficient formula.
Thus the compact-support assertion is a separate analytic strengthening; the
finite-moment asymptotic theorem has already been proved without it.

For~\textup{(iii)}, Corollary~\ref{cor:poisson-kl-eighth} proves the
eighth-order expansion directly from the seventh-moment mode remainder, and
Corollary~\ref{cor:seventh-moment-eighth-bridge} identifies the three displayed
coefficients with $A_4(\nu)$, $A_6(\nu)$, and $A_8(\nu)$.  The defect ladder
and amplification theorems use only these coefficients and the same
seventh-moment hypothesis, so their coefficient input is supplied by this
direct route.

Part~\textup{(iv)} follows from the same direct proof.  The proof of
Corollary~\ref{cor:poisson-kl-eighth} invokes
Lemma~\ref{lem:tail-uniform-cayley-mode-remainder} with truncation order~$6$,
then estimates the entropy Taylor remainder through order~$t^{-9}$ using the
finite centred seventh absolute moment.  At no step does it use the
finite-moment density remainder
Theorem~\ref{thm:finite-moment-density-remainder} with $L=8$, and hence it
does not require a finite centred $(8+\eta)$th absolute moment.  Therefore the
seventh-moment statement cannot be logically reduced to the $N=4$ case of
Theorem~\ref{thm:poisson-kl-even-orders} unless the stronger moment hypothesis
is added separately.

It remains only to prove that the separation is strict.  Fix
$0<\eta\le1$.  Let $X$ be a symmetric real random variable with a bounded
density near the origin and with two-sided tails proportional to
$(1+|x|)^{-9}$ for large $|x|$, normalized to total mass one.  The integral
test gives
\[
\EE |X|^r<\infty \quad\text{for } r<8,
\qquad
\EE |X|^r=\infty \quad\text{for } r\ge8.
\]
In particular $\EE X=0$, $0<\EE X^2<\infty$, and $\EE|X|^7<\infty$, while
\(\EE |X|^{8+\eta}=\infty\).  The law of $X$ is not compactly supported.
Therefore Corollary~\ref{cor:poisson-kl-eighth} applies to this law, and so do
Theorems~\ref{thm:poisson-defect-ladder} and
\ref{thm:poisson-defect-amplification}; however the $N=4$ hypothesis of
Theorem~\ref{thm:poisson-kl-even-orders} with exponent $8+\eta$ fails, and
Theorem~\ref{thm:analytic-compact-support-interchange} is unavailable.  This
proves the announced sharp separation of the three routes.
\end{proof}

\begin{corollary}[Moment-jet factorization of the eighth-order consequence]%
\label{cor:eighth-order-jet-factorization}
Let $\nu$ and $\widetilde\nu$ be probability measures on~$\RR$ with means
$\bar\gamma$ and $\widetilde{\bar\gamma}$, positive variances, and finite
centred seventh absolute moments.  Write
\[
\mu_n:=\int(\gamma-\bar\gamma)^n\dd\nu(\gamma),
\qquad
\widetilde\mu_n:=\int(\gamma-\widetilde{\bar\gamma})^n
\dd\widetilde\nu(\gamma).
\]
If
\[
\mu_n=\widetilde\mu_n,
\qquad 2\le n\le6,
\]
then the two matched-mean Poisson entropy expansions have the same
eighth-order jet:
\[
D_{\mathrm{KL}}(P_t*\nu\|P_t(\cdot-\bar\gamma))
-D_{\mathrm{KL}}(P_t*\widetilde\nu\|P_t(\cdot-\widetilde{\bar\gamma}))
=O_{\nu,\widetilde\nu}(t^{-9}).
\]
Equivalently, the coefficients $A_4$, $A_6$, and $A_8$ in
Corollary~\ref{cor:poisson-kl-eighth}, and hence the defect coordinates
$C_6$, $C_8$, $\Delta_6$, and $\Delta_8$ used in
Theorems~\ref{thm:poisson-defect-ladder}
and~\ref{thm:poisson-defect-amplification}, factor through the finite
centred moment jet $(\mu_2,\ldots,\mu_6)$.  Higher moments enter this
eighth-order consequence only through the validity and size of the
order-nine remainder.
\end{corollary}

\begin{proof}
Apply Corollary~\ref{cor:poisson-kl-eighth} to both laws.  The displayed
formula \eqref{eq:poisson-kl-eighth} expresses the $t^{-4}$ coefficient as
\(\mu_2^2/8\), the $t^{-6}$ coefficient as
\((\mu_2^3+6\mu_3^2-8\mu_2\mu_4)/64\), and the $t^{-8}$ coefficient as
\[
\frac{3\mu_2^4-12\mu_2^2\mu_4+9\mu_2\mu_3^2
+12\mu_2\mu_6-30\mu_3\mu_5+20\mu_4^2}{256}.
\]
These three polynomials involve only the centred moments
\(\mu_2,\ldots,\mu_6\).  Under the hypothesis
\(\mu_n=\widetilde\mu_n\) for $2\le n\le6$, the retained coefficients in the
two expansions are therefore equal.  Subtracting the two estimates
\eqref{eq:poisson-kl-eighth-remainder}, and using
$\max\{B_{\nu,8},B_{\widetilde\nu,8}\}$ after increasing the lower time
threshold if necessary, gives the stated
$O_{\nu,\widetilde\nu}(t^{-9})$ difference.

The expressions defining $C_6$ and $C_8$ are precisely the $t^{-6}$ and
$t^{-8}$ coefficients just identified, and
\(\Delta_6=-7\mu_2^3/64-C_6\),
\(\Delta_8=C_8-23\mu_2^4/256\).  Hence these defect variables also depend
only on $(\mu_2,\ldots,\mu_6)$.  The seventh absolute moments are used by
Corollary~\ref{cor:poisson-kl-eighth} to produce the remainders, so they
affect the constants and threshold but not the printed eighth-order jet.
\end{proof}

\begin{proposition}[Exact finite-phase eighth-order verification]%
\label{prop:eighth-order-finite-phase-verification}
Let $n_1,\dots,n_k\ge2$ and put
\[
m:=n_1+\cdots+n_k.
\]
Define the circle function $P_{n_1,\dots,n_k}$ by
\[
P_{n_1,\dots,n_k}(e^{2\ii\theta})
:=\prod_{j=1}^{k}u_{n_j}(\tan\theta),
\qquad -\pi/2<\theta<\pi/2.
\]
Then $P_{n_1,\dots,n_k}$ is a Laurent polynomial in the circle variable
$z=e^{2\ii\theta}$ with Fourier support contained in
$\{-m,\dots,m\}$.  Hence, for every integer $L>m$,
\begin{equation}\label{eq:finite-grid-mode-product-verification}
\int_{\RR}
\prod_{j=1}^{k}u_{n_j}(y)\,
\frac{\dd y}{\pi(1+y^2)}
=
\frac1L\sum_{\zeta\in\mu_L}P_{n_1,\dots,n_k}(\zeta).
\end{equation}
In particular, any phase grid $\mu_L$ with $L\ge9$ exactly computes all
mode-product integrals entering the coefficients of
Corollary~\ref{cor:poisson-kl-eighth} through order~$t^{-8}$, including the
odd-order cancellations at orders $t^{-5}$ and $t^{-7}$.

More explicitly, for $L\ge9$ define the exact phase functional
\[
\mathfrak Q_L[F]:=\frac1L\sum_{\zeta\in\mu_L}F(\zeta)
\]
on Laurent polynomials of Fourier degree at most~$8$.  With
$P_{\mathbf n}:=P_{n_1,\dots,n_k}$ and with
\[
\sigma^2=\mu_2,\qquad
[\mathbf n]=P_{\mathbf n},
\]
put
\begin{align*}
\mathcal A_{4,L}
&:=\frac12\mu_2^2\,\mathfrak Q_L([2,2]),\\
\mathcal A_{5,L}
&:=\mu_2\mu_3\,\mathfrak Q_L([2,3]),\\
\mathcal A_{6,L}
&:=\mu_2\mu_4\,\mathfrak Q_L([2,4])
  +\frac12\mu_3^2\,\mathfrak Q_L([3,3])
  -\frac16\mu_2^3\,\mathfrak Q_L([2,2,2]),\\
\mathcal A_{7,L}
&:=\mu_2\mu_5\,\mathfrak Q_L([2,5])
  +\mu_3\mu_4\,\mathfrak Q_L([3,4])
  -\frac12\mu_2^2\mu_3\,\mathfrak Q_L([2,2,3]),\\
\mathcal A_{8,L}
&:=\mu_2\mu_6\,\mathfrak Q_L([2,6])
  +\mu_3\mu_5\,\mathfrak Q_L([3,5])
  +\frac12\mu_4^2\,\mathfrak Q_L([4,4])\\
&\qquad
  -\frac12\mu_2^2\mu_4\,\mathfrak Q_L([2,2,4])
  -\frac12\mu_2\mu_3^2\,\mathfrak Q_L([2,3,3])
  +\frac1{12}\mu_2^4\,\mathfrak Q_L([2,2,2,2]).
\end{align*}
Then
\[
\mathcal A_{5,L}=\mathcal A_{7,L}=0,
\]
and
\[
\mathcal A_{4,L}=\frac{\sigma^4}{8},\qquad
\mathcal A_{6,L}=C_6,\qquad
\mathcal A_{8,L}=C_8,
\]
where $C_6$ and $C_8$ are the displayed $t^{-6}$ and $t^{-8}$ entropy
coefficients in Corollary~\ref{cor:poisson-kl-eighth}.  Hence the finite
phase grid verifies the entire printed eighth-order entropy jet, not merely
individual example integrals.  The defect coordinates used in
Theorems~\ref{thm:poisson-defect-ladder}
and~\ref{thm:poisson-defect-amplification} are therefore finite phase
quantities once the moment coordinates appearing in
Corollary~\ref{cor:poisson-kl-eighth} are fixed.
\end{proposition}

\begin{proof}
By the finite Fourier formulas
\eqref{eq:cayley-mode-even-fourier} and
\eqref{eq:cayley-mode-odd-fourier}, each function
$u_n(\tan\theta)$ is a finite linear combination of
$z^r$ and $z^{-r}$ with $1\le r\le n$.  Multiplying the $k$ expansions
shows that $P_{n_1,\dots,n_k}$ is a Laurent polynomial whose exponents lie
between $-m$ and $m$.

The change of variables $y=\tan\theta$ gives
\[
\frac{\dd y}{\pi(1+y^2)}=\frac{\dd\theta}{\pi},
\]
and $z=e^{2\ii\theta}$ pushes $\dd\theta/\pi$ on
$(-\pi/2,\pi/2)$ to normalized Haar measure on~$\TT$.  Therefore the
left-hand side of \eqref{eq:finite-grid-mode-product-verification} is the
constant Fourier coefficient of $P_{n_1,\dots,n_k}$.  If $L>m$, no
nonzero exponent in the support interval $\{-m,\dots,m\}$ is divisible
by~$L$.  The character average over $\mu_L$ is thus $1$ on the constant
term and $0$ on every other term, which proves
\eqref{eq:finite-grid-mode-product-verification}.

This is the specialization of
Theorem~\ref{thm:finite-fourier-mode-integral-algorithm} to the products
needed by Corollary~\ref{cor:poisson-kl-eighth}.  That corollary uses only
mode products
whose total index is at most~$8$:
\[
u_2^2,\quad
u_2u_4,\ u_3^2,\ u_2^3,\quad
u_2u_6,\ u_3u_5,\ u_4^2,\ u_2^2u_4,\ u_2u_3^2,\ u_2^4,
\]
together with the odd-total-index products that form the discarded
$t^{-5}$ and $t^{-7}$ coefficients.  Taking $L\ge9$ puts all these
products below the aliasing barrier.  The same exact grid identity
therefore recovers the displayed even coefficients and also computes the
zero odd coefficients, since their Haar integrals vanish by
Lemma~\ref{lem:mode-parity-vanish}.

It remains only to match the displayed grid formulas with the entropy
coefficients.  In the proof of Corollary~\ref{cor:poisson-kl-eighth} the
finite Taylor polynomial
\[
\frac12d_t^2-\frac16d_t^3+\frac1{12}d_t^4
\]
has coefficient functions
\begin{align*}
B_4&=\frac12a_2^2,&
B_5&=a_2a_3,\\
B_6&=a_2a_4+\frac12a_3^2-\frac16a_2^3,&
B_7&=a_2a_5+a_3a_4-\frac12a_2^2a_3,\\
B_8&=a_2a_6+a_3a_5+\frac12a_4^2
-\frac12a_2^2a_4-\frac12a_2a_3^2+\frac1{12}a_2^4,
\end{align*}
where $a_j=\mu_j u_j$ for $2\le j\le6$ and $\mu_2=\sigma^2$.  Replacing each
Cauchy integral of a mode product in these five expressions by the exact
root-of-unity average \eqref{eq:finite-grid-mode-product-verification} gives
precisely the five quantities
$\mathcal A_{4,L},\ldots,\mathcal A_{8,L}$.  The odd quantities vanish by the
same parity lemma, and the even quantities are the integrals
\[
\int_{\RR}B_4\,\dd\omega,\qquad
\int_{\RR}B_6\,\dd\omega,\qquad
\int_{\RR}B_8\,\dd\omega.
\]
Corollary~\ref{cor:poisson-kl-eighth}, whose coefficient arithmetic is
recorded in Appendix~\ref{app:entropy-integrals}, identifies these three
integrals with $\sigma^4/8$, $C_6$, and $C_8$.  The defect coordinates are
defined algebraically from these same coefficients and~$\sigma^2$, so they
are verified by the same finite phase data.
\end{proof}

The next two defect theorems use $C_6$ and $C_8$ as entropy-asymptotic
coefficients under the finite centred seventh absolute moment hypothesis of
Corollary~\ref{cor:poisson-kl-eighth}.  The fourth-moment algebra in
Subsection~\ref{subsec:fourth-moment-stability} keeps separate superscripted
notation until that same hypothesis is imposed.

\begin{theorem}[A two-level defect ladder]%
\label{thm:poisson-defect-ladder}
Let $\nu$ be a probability measure on~$\RR$ with mean~$\bar\gamma$,
variance $\sigma^2>0$, and finite centred seventh absolute moment.  Define
\begin{align*}
C_6&:=\frac{\sigma^6+6\mu_3^2-8\sigma^2\mu_4}{64}, \\
C_8&:=
\frac{3\sigma^8-12\sigma^4\mu_4+9\sigma^2\mu_3^2+12\sigma^2\mu_6}{256}
+\frac{-30\mu_3\mu_5+20\mu_4^2}{256}.
\end{align*}
Let $X:=(\gamma-\bar\gamma)/\sigma$ on the probability space
$(\RR,\nu)$, set
\[
\beta_n:=\EE[X^n],\qquad n\in\NN,\ n\ge1,\qquad
\beta_3:=\EE[X^3],
\qquad
Q_2:=X^2-1-\beta_3X,
\]
and define
\[
Q_3:=XQ_2-\|Q_2\|_{L^2(\nu)}^2X-\frac{\beta_3}{4}Q_2.
\]
Then
\begin{equation}\label{eq:defect-c6}
\frac{64}{\sigma^6}C_6
=-7-2\beta_3^2-8\|Q_2\|_{L^2(\nu)}^2,
\end{equation}
and
\begin{equation}\label{eq:defect-c8}
\frac{256}{\sigma^8}C_8
=23
+12\|Q_3\|_{L^2(\nu)}^2
+32\|Q_2\|_{L^2(\nu)}^4
+\frac{61}{4}\beta_3^2\|Q_2\|_{L^2(\nu)}^2
+52\|Q_2\|_{L^2(\nu)}^2
+2\beta_3^4
+13\beta_3^2.
\end{equation}
In particular
\begin{equation}\label{eq:defect-c8-lower}
C_8\ge \frac{23}{256}\sigma^8,
\end{equation}
and equality holds if and only if
\[
\nu=\frac12\delta_{\bar\gamma+\sigma}
+\frac12\delta_{\bar\gamma-\sigma}.
\]
The identity \eqref{eq:defect-c6} recovers the classical Pearson
moment relation, while \eqref{eq:defect-c8} gives the next higher-order
component in the same defect coordinates.
\end{theorem}

\begin{proof}
Write
\[
\kappa:=\|Q_2\|_{L^2(\nu)}^2=\EE[Q_2^2],
\qquad
L:=\EE[XQ_2^2],
\qquad
M:=\EE[Q_2^3].
\]
Since $Q_2=X^2-1-\beta_3X$, one has
\[
\kappa=\beta_4-1-\beta_3^2,
\]
and therefore
\[
\frac{64}{\sigma^6}C_6
=1+6\beta_3^2-8\beta_4
=-7-2\beta_3^2-8\kappa,
\]
which is exactly \eqref{eq:defect-c6}.

The identity $X^2=1+\beta_3X+Q_2$ yields
\[
\beta_5=\beta_3(\beta_3^2+2+2\kappa)+L,
\]
\[
\beta_6=1+3\beta_3^2+3\kappa+\beta_3^4+3\beta_3^2\kappa+3\beta_3L+M.
\]
Hence, if
\[
P_8:=3-12\beta_4+9\beta_3^2+12\beta_6-30\beta_3\beta_5+20\beta_4^2,
\]
then a direct substitution gives
\[
P_8
=23+20\kappa^2+16\beta_3^2\kappa+64\kappa+2\beta_3^4+13\beta_3^2+6\beta_3L+12M.
\]
On the other hand,
using $\EE[XQ_2]=0$, $\EE[Q_2]=0$, $\EE[X^2Q_2^2]=\kappa+\beta_3L+M$, and
$\EE[XQ_2^2]=L$, the definition
$Q_3=XQ_2-\kappa X-(\beta_3/4)Q_2$ gives
\[
\|Q_3\|_{L^2(\nu)}^2
=M+\frac{\beta_3}{2}L+\kappa-\kappa^2+\frac{\beta_3^2}{16}\kappa.
\]
Equivalently,
\[
12\|Q_3\|_{L^2(\nu)}^2
=12M+6\beta_3L+12\kappa-12\kappa^2+\frac34\beta_3^2\kappa.
\]
Substituting this last line into the preceding expression for $P_8$ gives
the exact regrouping
\[
P_8=23+12\|Q_3\|_2^2+32\kappa^2
+\frac{61}{4}\beta_3^2\kappa+52\kappa+2\beta_3^4+13\beta_3^2,
\]
which is \eqref{eq:defect-c8}.  Since
$P_8=256C_8/\sigma^8$, inequality \eqref{eq:defect-c8-lower} follows.

If equality holds in \eqref{eq:defect-c8-lower}, then every non-negative
term on the right-hand side of \eqref{eq:defect-c8} vanishes.  In
particular $\beta_3=0$ and $\|Q_2\|_{L^2(\nu)}=0$, so $X^2=1$
$\nu$-almost everywhere.  Since $\EE[X]=0$ and $\EE[X^2]=1$, it follows
that $X=\pm 1$ with equal weights.  Thus
\[
\nu=\frac12\delta_{\bar\gamma+\sigma}
+\frac12\delta_{\bar\gamma-\sigma}.
\]
The converse is immediate.
\end{proof}

\begin{theorem}[Defect amplification]\label{thm:poisson-defect-amplification}
Let $\nu$ be a probability measure on~$\RR$ with mean~$\bar\gamma$,
variance $\sigma^2>0$, and finite centred seventh absolute moment.  Define
\[
\Delta_6(\nu):=-\frac{7}{64}\sigma^6-C_6,
\qquad
\Delta_8(\nu):=C_8-\frac{23}{256}\sigma^8,
\]
where $C_6$ and $C_8$ are the entropy coefficients from
Corollary~\ref{cor:poisson-kl-eighth}.  Set
\[
X:=\frac{\gamma-\bar\gamma}{\sigma},
\qquad
\beta_3:=\EE[X^3],
\qquad
Q_2:=X^2-1-\beta_3X,
\]
\[
\kappa:=\|Q_2\|_{L^2(\nu)}^2,
\qquad
Q_3:=XQ_2-\kappa X-\frac{\beta_3}{4}Q_2.
\]
Then
\[
\Delta_6(\nu)
=\frac{\sigma^6}{32}\beta_3^2+\frac{\sigma^6}{8}\kappa,
\]
and
\begin{align}
\Delta_8(\nu)
&=
\frac{13\sigma^2}{8}\Delta_6(\nu)
\notag\\
&\quad
+\frac{\sigma^8}{256}\left(
12\|Q_3\|_{L^2(\nu)}^2
+32\kappa^2
+\frac{61}{4}\beta_3^2\kappa
+2\beta_3^4
\right). \label{eq:defect-amplification}
\end{align}
In particular,
\[
\Delta_8(\nu)\ge \frac{13\sigma^2}{8}\Delta_6(\nu),
\]
and equality in this proved sixth/eighth-order defect layer holds if and only if
\[
\nu=\frac12\delta_{\bar\gamma+\sigma}
+\frac12\delta_{\bar\gamma-\sigma}.
\]
\end{theorem}

\begin{proof}
Equation~\eqref{eq:defect-c6} gives
\[
\Delta_6(\nu)
=-\frac{\sigma^6}{64}\left(7+\frac{64}{\sigma^6}C_6\right)
=\frac{\sigma^6}{32}\beta_3^2+\frac{\sigma^6}{8}\kappa.
\]
Likewise, subtracting $23$ from \eqref{eq:defect-c8} yields
\[
\Delta_8(\nu)
=\frac{\sigma^8}{256}\Bigl(
12\|Q_3\|_{L^2(\nu)}^2
+32\kappa^2
+\frac{61}{4}\beta_3^2\kappa
+52\kappa
+2\beta_3^4
+13\beta_3^2
\Bigr).
\]
By the preceding formula for $\Delta_6(\nu)$,
\[
\frac{13\sigma^2}{8}\Delta_6(\nu)
=\frac{\sigma^8}{256}\bigl(52\kappa+13\beta_3^2\bigr),
\]
and substituting this identity proves \eqref{eq:defect-amplification}.

If equality holds in the lower bound, then every non-negative term in
\eqref{eq:defect-amplification} vanishes.  In particular,
$\beta_3=0$ and $\kappa=0$, hence $Q_2=X^2-1$ vanishes
$\nu$-almost everywhere.  Since $\EE[X]=0$ and $\EE[X^2]=1$, it follows
that $X=\pm 1$ with equal weights, which is exactly the stated
two-point law.  The converse is immediate.
\end{proof}

\begin{corollary}[Symmetric two-point entropy-defect rigidity]%
\label{cor:poisson-two-point-defect-rigidity}
Let $\nu$ be a probability measure on~$\RR$ with mean~$\bar\gamma$,
variance $\sigma^2>0$, and finite centred seventh absolute moment.  Let
$C_6$ and $C_8$ be the entropy coefficients from
Corollary~\ref{cor:poisson-kl-eighth}, and define
\[
\Delta_6(\nu):=-\frac{7}{64}\sigma^6-C_6,
\qquad
\Delta_8(\nu):=C_8-\frac{23}{256}\sigma^8.
\]
Let $\nu_c$ be the centred law of $\gamma-\bar\gamma$ and set
\[
\rho_\sigma:=\frac12(\delta_{-\sigma}+\delta_\sigma).
\]
Then, among all laws with the same mean and variance in this moment class,
\[
C_6\le -\frac{7}{64}\sigma^6,
\qquad
C_8\ge \frac{23}{256}\sigma^8.
\]
Moreover the following conditions are equivalent:
\begin{enumerate}[label=\textup{(\roman*)},leftmargin=*]
\item $\nu_c=\rho_\sigma$, equivalently
$\nu=\frac12\delta_{\bar\gamma-\sigma}
+\frac12\delta_{\bar\gamma+\sigma}$;
\item $C_6=-\frac{7}{64}\sigma^6$;
\item $C_8=\frac{23}{256}\sigma^8$;
\item $\Delta_6(\nu)=\Delta_8(\nu)=0$;
\item the amplification bound is sharp,
\[
\Delta_8(\nu)=\frac{13\sigma^2}{8}\Delta_6(\nu).
\]
\end{enumerate}
Thus, at the proved sixth/eighth-order layer, and with no assertion here about
an unproved all-order defect hierarchy, the symmetric two-point law is
the unique entropy-defect extremizer and the unique rigidity class for the
amplification chain.
\end{corollary}

\begin{proof}
Use the notation of Theorems~\ref{thm:poisson-defect-ladder}
and~\ref{thm:poisson-defect-amplification}.  From
\eqref{eq:defect-c6},
\[
C_6+\frac{7}{64}\sigma^6
=-\frac{\sigma^6}{32}\beta_3^2-\frac{\sigma^6}{8}\kappa
\le0,
\qquad
\kappa:=\|Q_2\|_{L^2(\nu)}^2.
\]
Hence $C_6$ is maximized by, and only by, laws for which
$\beta_3=0$ and $\kappa=0$.  In that case
$Q_2=X^2-1$ vanishes almost surely, so $X=\pm1$ almost surely; since
$\EE X=0$ and $\EE X^2=1$, the two signs have equal mass.  This is exactly
$\nu_c=\rho_\sigma$.  Conversely, for $\nu_c=\rho_\sigma$ one has
$\beta_3=0$ and $Q_2=0$, so equality holds in the displayed bound for
$C_6$.

The lower bound for $C_8$ and the characterization of equality in that
bound are precisely \eqref{eq:defect-c8-lower} and the equality statement
in Theorem~\ref{thm:poisson-defect-ladder}.  The definitions of
$\Delta_6$ and $\Delta_8$ convert equality in the two coefficient bounds
into $\Delta_6(\nu)=0$ and $\Delta_8(\nu)=0$, respectively.  Finally,
Theorem~\ref{thm:poisson-defect-amplification} states that equality in
\[
\Delta_8(\nu)\ge\frac{13\sigma^2}{8}\Delta_6(\nu)
\]
holds if and only if $\nu_c=\rho_\sigma$.  Combining these equivalences
proves all the listed conditions equivalent.
\end{proof}

\begin{example}[Gaussian input]\label{ex:gaussian-defects}
Let $\nu$ be the centred Gaussian law with variance~$\sigma^2$.  For
$X=(\gamma-\bar\gamma)/\sigma$ one has
\[
\beta_3=\beta_5=0,\qquad \beta_4=3,\qquad \beta_6=15,
\qquad \kappa=\beta_4-1=2.
\]
Hence the defect formulas give
\[
\Delta_6(\nu)=\frac{\sigma^6}{4},
\qquad
\Delta_8(\nu)=\frac{19}{16}\sigma^8.
\]
Equivalently, inserting the Gaussian moments into the coefficient
formula for $C_8$ gives
\[
\frac{256}{\sigma^8}C_8
=3-12\cdot3+12\cdot15+20\cdot9=327,
\]
and subtracting the extremal value~$23$ yields the same
$\Delta_8=(327-23)\sigma^8/256$.
\end{example}

\begin{example}[Zero-inflated symmetric two-point family]%
\label{ex:zero-inflated-defects}
Fix $0<p\le1$ and let $X_p$ be the centred variance-one random variable
defined by
\[
\mathbb P(X_p=0)=1-p,
\qquad
\mathbb P(X_p=p^{-1/2})=\mathbb P(X_p=-p^{-1/2})=\frac p2.
\]
For the law of $\bar\gamma+\sigma X_p$ one has
\[
\beta_3=\beta_5=0,\qquad
\beta_4=\frac1p,\qquad
\beta_6=\frac1{p^2},\qquad
\kappa=\frac1p-1.
\]
Therefore
\[
\Delta_6(\nu)
=\frac{\sigma^6}{8}\left(\frac1p-1\right),
\]
and
\[
\Delta_8(\nu)
=\frac{\sigma^8}{256}
\left(\frac{32}{p^2}-\frac{12}{p}-20\right)
=\frac{\sigma^8}{64}\,
\frac{(1-p)(5p+8)}{p^2}.
\]
At $p=1$ the law is the symmetric two-point extremizer and both defects
vanish; for $0<p<1$ the extra atom at the mean produces a positive
sixth- and eighth-order defect.
\end{example}
