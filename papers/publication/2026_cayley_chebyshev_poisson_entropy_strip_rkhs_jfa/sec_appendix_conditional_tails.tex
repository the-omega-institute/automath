\section{Conditional missing-top-moment tail transfer}%
\label{app:conditional-missing-top-moment-tails}

This appendix contains the conditional missing-top-moment transfer and the
three concrete quotient-verification classes used as optional consequences in
the one-dimensional theory.  The main coefficient hierarchy does not depend
on these statements.  The input for the entropy conclusion is always the
matched quotient expansion with its \(L^\infty\cap L^1(\omega)\) error bound;
the divergent truncated moment \(M_p(t)\) alone is not an entropy hypothesis.
\subsubsection*{Conditional quotient theorem}\leavevmode\par\smallskip

\medskip
\noindent\fbox{%
\begin{minipage}{0.94\linewidth}
\textbf{Conditional scope.}
The next theorem is the borderline-tail result used in this paper.  Its
hypothesis is the displayed quotient expansion with the stated growth bounds.
A divergent truncated moment \(M_p(t)\) by itself is not an entropy
hypothesis.  The exact-power, perturbative-power, and monotone slowly varying
statements following it are only verification corollaries which prove this
quotient input in three concrete subclasses.
\end{minipage}}
\medskip

\begin{theorem}[Conditional quotient theorem for a truncated top mode]%
\label{thm:tail-replacements-proved-here}
\label{prop:conditional-truncated-top-moment-replacement}
Let \(N\ge3\), put \(p=2N-2\ge4\), and let \(\nu\) be a symmetric
probability law on \(\RR\) with finite absolute moments of every order
\(q<p\).  Let
\[
  M_p(t):=\EE\!\left[X^p\mathbf1_{\{|X|\le t\}}\right],
  \qquad
  \Delta_t(y):=\frac{(P_t*\nu)(ty)}{P_t(ty)}-1 .
\]
Assume first that the quotient admits the \(L^\infty(\RR)\cap L^1(\omega)\)
expansion
\begin{equation}\label{eq:conditional-truncated-top-mode-quotient}
  \Delta_t(y)=
  A_t(y)
  +u_p(y)t^{-p}M_p(t)+e_t(y),
  \qquad
  A_t(y):=\sum_{n=2}^{p-1}\mu_nu_n(y)t^{-n},
\end{equation}
where
\begin{equation}\label{eq:conditional-truncated-top-mode-error}
  \|e_t\|_\infty+\|e_t\|_{L^1(\omega)}
  =o(t^{-p}M_p(t)).
\end{equation}
Assume also the growth bounds
\begin{equation}\label{eq:conditional-truncated-moment-growth}
  M_p(t)\to\infty,\qquad M_p(t)=o(t^{p-2}).
\end{equation}
The growth \(M_p(t)\to\infty\) has no standalone consequence here; the entropy
conclusion below uses the quotient expansion and error estimate above.
Then, with \(H(t)=D_{\mathrm{KL}}(P_t*\nu\|P_t)\),
\begin{equation}\label{eq:conditional-truncated-top-moment-entropy}
H(t)=
\sum_{m=2}^{N-1}A_{2m}(\nu)t^{-2m}
+J(2,p)\mu_2\,t^{-2N}M_p(t)
+o(t^{-2N}M_p(t)),
\end{equation}
where
\[
  \kappa_N=J(2,p)=(-1)^N(N-1)2^{-2N+2}.
\]
\end{theorem}

\begin{proof}
Set
\[
  A_t(y)=\sum_{n=2}^{p-1}\mu_nu_n(y)t^{-n},
  \qquad
  B_t(y)=u_p(y)t^{-p}M_p(t).
\]
The Cayley modes are bounded, and
\eqref{eq:conditional-truncated-moment-growth}--\eqref{eq:conditional-truncated-top-mode-error}
put the quotient expansion exactly in the form of
Lemma~\ref{lem:moving-scale-taylor-transfer} on
\((\RR,\omega)\), with \(a_n=\mu_nu_n\) and \(b=u_p\).  Since
\(\|\Delta_t\|_\infty\to0\), the KL identity gives
\[
  H(t)=\int_{\RR}\Phi(\Delta_t(y))\,\omega(\dd y),
  \qquad
  \Phi(s)=(1+s)\log(1+s)-s .
\]
Set
\[
  s_t:=t^{-2N}M_p(t)=t^{-(p+2)}M_p(t).
\]
The moving-scale transfer gives
\begin{equation}\label{eq:conditional-transfer-applied}
  H(t)
  =
  \int_{\RR}\Phi(A_t(y))\,\omega(\dd y)
  +\mu_2J(2,p)s_t+o(s_t).
\end{equation}

It remains only to identify the pure finite-jet part
\(\int\Phi(A_t)\).  Since \(\|A_t\|_\infty=O(t^{-2})\), Taylor expansion
through degree \(N\) gives
\[
\Phi(A_t)=
\sum_{r=2}^{N}\frac{(-1)^r}{r(r-1)}A_t^r
+O(|A_t|^{N+1}).
\]
Here the displayed remainder is \(O(t^{-2N-2})=o(s_t)\).  Expanding the
finite polynomial \(A_t^r\), the coefficient of \(t^{-k}\) is exactly the
universal coefficient \(A_k(\nu)\) from
\eqref{eq:coefficient-identification-transfer-expansion}, with no factor
involving the missing \(p\)th moment.  The odd coefficients vanish by the
mode-parity argument in
Theorem~\ref{thm:universal-entropy-coefficients}.  Therefore the pure part is
\[
\int_{\RR}\Phi(A_t)\,\dd\omega
=
\sum_{m=2}^{N-1}A_{2m}(\nu)t^{-2m}+o(s_t).
\]
Indeed, the next pure even layer has size \(O(t^{-2N})=o(s_t)\), because
\(M_p(t)\to\infty\), and all higher pure finite-jet layers are smaller.
Substituting this in \eqref{eq:conditional-transfer-applied} leaves only the
moving-scale cross coefficient \(\mu_2J(2,p)\).  Finally, the computation in
Proposition~\ref{prop:poisson-kl-moment-threshold-sharpness} gives
\(J(2,p)=\kappa_N\).  This proves
\eqref{eq:conditional-truncated-top-moment-entropy}.
\end{proof}

\subsubsection*{Borderline \(p=2\) infinite-variance layer}\leavevmode\par\smallskip

The conditional missing-top-moment theorem above starts at \(p=2N-2\ge4\).
The infinite-variance borderline \(p=2\) is different: the leading KL scale
is quadratic in the divergent truncated variance, not linear.  The following
appendix calculation records the separate tail behavior used only as a
consistency check against the sharp variance threshold in the main text.

\begin{theorem}[Borderline infinite-variance power tail]%
\label{prop:n2-borderline-infinite-variance-layer}
Let \(c>0\) satisfy \(c\mathrm e^{-2}<1\), and let \(\nu\) be the symmetric
probability law with density
\[
f(x)=\frac{1-c\mathrm e^{-2}}{2}\mathbf1_{\{|x|\le1\}}
      +c|x|^{-3}\mathbf1_{\{|x|\ge \mathrm e\}} .
\]
Then \(\nu\) has finite mean and infinite variance, and
\[
\EE\!\left[X^2\mathbf1_{\{|X|\le t\}}\right]\sim 2c\log t,
\]
while
\[
D_{\mathrm{KL}}(P_t*\nu\|P_t)
=\frac{c^2}{2}\,t^{-4}(\log t)^2+O(t^{-4}\log t).
\]
Thus the borderline \(N=2\) law is not a \(t^{-4}\log t\) replacement; its
leading layer is \(t^{-4}(\log t)^2\).
\end{theorem}

\begin{lemma}[Symmetric \(p=2\) quotient tail split]
\label{lem:symmetric-p2-far-tail-quotient}
Let
\[
  S_a(y):=R_a(y)+R_{-a}(y)-2 .
\]
There is a universal constant \(C<\infty\) such that, for all
\(t\ge \mathrm e\),
\begin{equation}\label{eq:symmetric-p2-far-tail-quotient}
  \left\|
  \int_t^\infty S_{x/t}(y)\,x^{-3}\,\dd x
  \right\|_{L^\infty(\RR)}
  +
  \left\|
  \int_t^\infty S_{x/t}(y)\,x^{-3}\,\dd x
  \right\|_{L^1(\omega)}
  \le Ct^{-2}.
\end{equation}
Equivalently, for each fixed \(c_0\),
\begin{align*}
&\sup_y\left|
c_0t^{-2}\int_1^\infty
\{R_\varepsilon(y)+R_{-\varepsilon}(y)-2\}\varepsilon^{-3}\,\dd\varepsilon
\right| \\
&\qquad
+\int_{\RR}\left|
c_0t^{-2}\int_1^\infty
\{R_\varepsilon(y)+R_{-\varepsilon}(y)-2\}\varepsilon^{-3}\,\dd\varepsilon
\right|\omega(\dd y)
\le C(c_0)t^{-2}.
\end{align*}
Moreover, if \(R\ge1\), then as \(t\to\infty\)
\begin{equation}\label{eq:symmetric-p2-full-tail-split}
\int_R^\infty S_{x/t}(y)x^{-3}\,\dd x
=2u_2(y)t^{-2}\log t+O(t^{-2})
\end{equation}
in \(L^\infty(\RR)\cap L^1(\omega)\), with the constant depending on \(R\)
but not on \(y\) or \(t\).
\end{lemma}

\begin{proof}
After the change of variables \(a=x/t\), the integral over \(x>t\) is
\[
  t^{-2}\int_1^\infty S_a(y)a^{-3}\,\dd a .
\]
It is therefore enough to prove a uniform bound in \(y\) for the last
integral.  Pairing the two translates before estimating is essential: the
unpaired estimate with \((1+x^2/t^2)x^{-3}\) would produce the divergent
\(\int_t^\infty x^{-1}\dd x\).  With \(A=1+y^2+a^2\),
\[
\begin{aligned}
R_a(y)+R_{-a}(y)-2
&=(1+y^2)\left(\frac1{A-2ay}+\frac1{A+2ay}\right)-2  \\
&=
\frac{2a^2(3y^2-1-a^2)}
     {\{1+(y-a)^2\}\{1+(y+a)^2\}} .
\end{aligned}
\]
Thus, writing \(Y=|y|\),
\[
 |S_a(y)|a^{-3}
 \le
 \frac{2(a^2+3Y^2+1)}
      {a\{1+(a-Y)^2\}\{1+(a+Y)^2\}}
 \le
 \frac{C}{a\{1+(a-Y)^2\}},
\]
because \(a^2+3Y^2+1\le C\{1+(a+Y)^2\}\).  If \(Y\le2\), then
\[
  \int_1^\infty \frac{\dd a}{a\{1+(a-Y)^2\}}
  \le
  \int_1^\infty \frac{\dd a}{1+(a-2)^2}<\infty .
\]
If \(Y>2\), split the integral over
\[
[1,Y/2],\qquad
[Y/2,2Y]\cap\{|a-Y|>1\},\qquad
\{|a-Y|\le1\},\qquad
[2Y,\infty).
\]
The four pieces are respectively
\[
O(Y^{-2}\log Y),\qquad O(Y^{-1}),\qquad O(Y^{-1}),\qquad O(Y^{-2}).
\]
The local \(O(Y^{-1})\) term is \(L^1(\omega)\), since
\(\omega(\dd y)\asymp Y^{-2}\dd Y\) for \(Y>2\).  Hence
\[
  \sup_{y\in\RR}\int_1^\infty |S_a(y)|a^{-3}\,\dd a<\infty .
\]
This gives the \(L^\infty\) estimate in
\eqref{eq:symmetric-p2-far-tail-quotient}; the \(L^1(\omega)\) estimate
follows because \(\omega\) is a probability measure.

For \(R\le x\le t\), Taylor expansion through order \(3\), paired at \(x\)
and \(-x\), cancels the odd modes and yields
\[
S_{x/t}(y)=2u_2(y)(x/t)^2+O((x/t)^4)
\qquad(0\le x/t\le1),
\]
uniformly in \(y\).  Therefore
\[
\int_R^t S_{x/t}(y)x^{-3}\,\dd x
=2u_2(y)t^{-2}\int_R^t x^{-1}\,\dd x
+O\!\left(t^{-4}\int_R^t x\,\dd x\right)
=2u_2(y)t^{-2}\log t+O(t^{-2}).
\]
Adding \eqref{eq:symmetric-p2-far-tail-quotient} gives
\eqref{eq:symmetric-p2-full-tail-split}.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{prop:n2-borderline-infinite-variance-layer}]
The law is symmetric.  Its absolute first moment is finite, while its second
moment is infinite, since
\[
  2c\int_{\mathrm e}^{\infty}x^{-2}\,\dd x<\infty,
  \qquad
  2c\int_{\mathrm e}^{\infty}x^{-1}\,\dd x=\infty .
\]
The truncated second moment is
\[
  \EE[X^2\mathbf1_{\{|X|\le t\}}]
  =O(1)+2c\int_{\mathrm e}^{t}x^{-1}\,\dd x
  \sim 2c\log t .
\]

For \(t\ge \mathrm e\), put
\[
  \Delta_t(y)=\frac{(P_t*\nu)(ty)}{P_t(ty)}-1 .
\]
We prove the quotient expansion
\begin{equation}\label{eq:n2-borderline-quotient-expansion}
\Delta_t(y)=2c\,u_2(y)t^{-2}\log t+E_t(y),
\qquad
\|E_t\|_\infty+\|E_t\|_{L^1(\omega)}=O(t^{-2})
\end{equation}
in \(L^\infty\cap L^1(\omega)\).

The compact core \(|x|\le1\) has bounded support and is symmetric.  Expanding
the paired expression \(R_{x/t}(y)+R_{-x/t}(y)-2\) through order \(3\) gives a
contribution
\[
  t^{-2}u_2(y)
  \int_{|x|\le1}x^2\,\frac{1-c\mathrm e^{-2}}{2}\,\dd x
  +O(t^{-4}),
\]
uniformly in \(y\), which is part of \(E_t\).  For the tail, symmetry gives
\[
  c\int_{\mathrm e}^{\infty}
  \{R_{x/t}(y)+R_{-x/t}(y)-2\}x^{-3}\,\dd x .
\]
Lemma~\ref{lem:symmetric-p2-far-tail-quotient}, with \(R=\mathrm e\), gives
the full tail contribution
\[
  2c\,u_2(y)t^{-2}\log t+O(t^{-2})
\]
in both \(L^\infty\) and \(L^1(\omega)\).  The pairing in that lemma is the
separate \(p=2\) tail argument: the unpaired far-tail quotient bound would be
too crude for the infinite-variance layer.  This proves
\eqref{eq:n2-borderline-quotient-expansion}.

Since \(\|\Delta_t\|_\infty=O(t^{-2}\log t)\),
\[
D_{\mathrm{KL}}(P_t*\nu\|P_t)
=\frac12\int_{\RR}\Delta_t(y)^2\,\omega(\dd y)+O(t^{-6}(\log t)^3).
\]
Using \(J(2,2)=\int u_2^2\,\dd\omega=1/4\), and using
\(\|E_t\|_\infty+\|E_t\|_{L^1(\omega)}=O(t^{-2})\) for the cross and error
terms, the quadratic term is
\[
\frac12(2c)^2J(2,2)t^{-4}(\log t)^2+O(t^{-4}\log t)
=\frac{c^2}{2}t^{-4}(\log t)^2+O(t^{-4}\log t).
\]
\end{proof}

\subsubsection*{Consequences: verified borderline-tail classes}\leavevmode\par\smallskip

\medskip
\noindent\fbox{%
\begin{minipage}{0.94\linewidth}
\begin{definition}[Admissible borderline tail classes]%
\label{def:admissible-borderline-tail-classes}
Fix an even integer \(p\ge4\), \(R>0\), and \(c>0\).  Every class in this
definition uses the same disjoint core-plus-tail convention: \(\nu\) has a
nonnegative normalized density
\[
  f(x)=g(x)+\{c|x|^{-p-1}L_f(|x|)+r_f(x)\}
       \mathbf1_{\{|x|\ge R\}},
\]
where \(g\ge0\) is symmetric, compactly supported, and
\(\operatorname{supp}g\subset(-R,R)\), \(L_f\ge0\) is measurable on
\([R,\infty)\), and \(r_f\) is an even measurable perturbation on
\(\{|x|\ge R\}\).  Put
\[
  M_p(t):=\int_{|x|\le t}x^pf(x)\,\dd x,\qquad
  \mathcal R_t^{(p)}(y):=
  \int_{|x|\le t}
  \left\{R_{x/t}(y)-\sum_{n=0}^{p}u_n(y)(x/t)^n\right\}f(x)\,\dd x .
\]

\(\nu\) belongs to the admissible class \(\mathcal A_{p,R,c}\) if it is
symmetric, has finite absolute moments of every order \(q<p\), and satisfies
the four quotient controls
\begin{align}
t^{-n}\int_{|x|>t}|x|^nf(x)\,\dd x
  &=o(t^{-p}M_p(t))\qquad(0\le n<p), \label{eq:admissible-lower-tail-control}\\
t^{-p}\int_{|x|\le t}x^pf(x)\,\dd x
  &=t^{-p}M_p(t), \label{eq:admissible-top-mode-control}\\
\|\mathcal R_t^{(p)}\|_\infty+\|\mathcal R_t^{(p)}\|_{L^1(\omega)}
  &=o(t^{-p}M_p(t)), \label{eq:admissible-local-remainder-control}\\
\int_{|x|>t}\left(1+\frac{x^2}{t^2}\right)f(x)\,\dd x
  &=o(t^{-p}M_p(t)). \label{eq:admissible-far-tail-control}
\end{align}
together with the growth bounds
\begin{equation}\label{eq:admissible-top-moment-growth}
  M_p(t)\to\infty,\qquad M_p(t)=o(t^{p-2}).
\end{equation}
The statement \(M_p(t)\to\infty\) alone is not an admissible quotient
hypothesis and gives no entropy conclusion in this subsection; the displayed
quotient controls are the hypotheses that make the truncated top moment
visible to KL.
The three named verification subclasses used below are the following.

\emph{Exact power tail.}
\(\mathcal A_{p,R,c}^{\rm ex}\) is the class with \(L_f\equiv1\) and
\(r_f=0\).  Equivalently, after the compact core,
\[
  f(x)=g(x)+c|x|^{-p-1}\mathbf1_{\{|x|\ge R\}} .
\]

\emph{Perturbative power tail.}
\(\mathcal A_{p,R,c}^{\rm pert}\) is the class with \(L_f\equiv1\) and
\(r_f=r\), where \(r\) is even and satisfies
\[
\int_{|x|\ge R}|x|^q|r(x)|\,\dd x<\infty\quad(0\le q<p),
\qquad
\int_{R\le |x|\le t}|x|^pr(x)\,\dd x=O(1),
\]
\[
t^{-(p+1)}\int_{R\le |x|\le t}|x|^{p+1}|r(x)|\,\dd x=O(t^{-p}),
\qquad
\int_{|x|>t}\left(1+\frac{x^2}{t^2}\right)|r(x)|\,\dd x=O(t^{-p}).
\]

\emph{Monotone slowly varying tail.}
\(\mathcal A_{p,R,c}^{\rm sv}\) is the class with \(r_f=0\) and
\(L_f=L\), where \(L\) is nonnegative on \([R,\infty)\), eventually positive,
eventually monotone, slowly varying, and \(M_p(t)\to\infty\).
\end{definition}
\end{minipage}}
\medskip

Thus membership in \(\mathcal A_{p,R,c}\) records, in one place, the
symmetry, lower moments, moving truncated top moment, and exact estimates
needed for the quotient Taylor calculation.  In particular, for
\[
  \Delta_t(y):=\frac{(P_t*\nu)(ty)}{P_t(ty)}-1
\]
and \(\mu_n=\int x^nf(x)\,\dd x\), \(2\le n<p\), Taylor expansion on
\(|x|\le t\), the lower-tail replacement
\eqref{eq:admissible-lower-tail-control}, the local remainder
\eqref{eq:admissible-local-remainder-control}, and the far-tail control
\eqref{eq:admissible-far-tail-control} give the admissible quotient expansion
\begin{equation}\label{eq:admissible-borderline-quotient-expansion}
  \Delta_t(y)=
  \sum_{n=2}^{p-1}\mu_nu_n(y)t^{-n}
  +u_p(y)t^{-p}M_p(t)+e_t(y),
  \qquad
  \|e_t\|_\infty+\|e_t\|_{L^1(\omega)}
  =o(t^{-p}M_p(t)).
\end{equation}

\begin{corollary}[Admissible quotient verification consequence]%
\label{thm:admissible-borderline-tail-entropy}
Let \(N\ge3\), put \(p=2N-2\), and let
\(\nu\in\mathcal A_{p,R,c}\) in the sense of
Definition~\ref{def:admissible-borderline-tail-classes}.  Let
\[
  M_p(t)=\EE[X^p{\bf1}_{\{|X|\le t\}}],
  \qquad
  H(t)=D_{\mathrm{KL}}(P_t*\nu\|P_t).
\]
Then
\[
H(t)=
\sum_{m=2}^{N-1}A_{2m}(\nu)t^{-2m}
+\kappa_N\mu_2\,t^{-2N}M_p(t)
+o(t^{-2N}M_p(t)),
\]
where
\[
  \kappa_N=J(2,p)=(-1)^N(N-1)2^{-2N+2}.
\]
\end{corollary}

\begin{proof}
The admissible controls give the quotient expansion
\eqref{eq:admissible-borderline-quotient-expansion}, and
\eqref{eq:admissible-top-moment-growth} gives the growth hypotheses
\eqref{eq:conditional-truncated-moment-growth}.  Thus the conditional
top-mode transfer Theorem~\ref{thm:tail-replacements-proved-here} applies
directly.
\end{proof}

The next three corollaries are verification statements only.  They establish
the quotient controls defining \(\mathcal A_{p,R,c}\) and then invoke the
conditional theorem; they do not replace it by a general regular-variation
principle.

\subsubsection*{Exact power-tail verification}

The exact power law is a verification of
Corollary~\ref{thm:admissible-borderline-tail-entropy}.  The quotient expansion
is displayed because it is the admissibility input.

\begin{corollary}[Exact-power verification corollary]%
\label{prop:log-tail-counterexample-top-moment}
Let \(N\ge3\), put \(p=2N-2\), and choose \(R>0\) and \(c>0\) such that
\[
  \frac{2c}{pR^p}<1 .
\]
Let \(g\in L^1(\RR)\) be any symmetric nonnegative compactly supported
function with \(\operatorname{supp}g\subset(-R,R)\) and total mass
\(1-2c/(pR^p)\), and define the symmetric probability density
\begin{equation}\label{eq:normalized-borderline-tail-density}
  f(x)=g(x)+c|x|^{-(p+1)}\mathbf1_{\{|x|\ge R\}} .
\end{equation}
Let \(\nu\) be the law with density \(f\).  Then
\(\nu\in\mathcal A_{p,R,c}^{\rm ex}\subset\mathcal A_{p,R,c}\), where the
classes are those of
Definition~\ref{def:admissible-borderline-tail-classes}.  In particular
\(\nu\) is centred,
\[
\EE|X|^q<\infty\quad(q<p),\qquad \EE|X|^p=\infty ,
\]
and
\[
\EE\!\left[X^p\mathbf1_{\{|X|\le t\}}\right]=2c\log t+O(1).
\]
For
\[
  \Delta_t(y):=\frac{(P_t*\nu)(ty)}{P_t(ty)}-1
\]
the exact quotient expansion
\begin{equation}\label{eq:exact-power-tail-theorem-quotient}
  \Delta_t(y)=
  \sum_{n=2}^{p-1}\mu_nu_n(y)t^{-n}
  +2c\,u_p(y)t^{-p}\log t+E_t(y)
\end{equation}
holds in \(L^\infty(\RR)\cap L^1(\omega)\), with
\begin{equation}\label{eq:exact-power-tail-theorem-quotient-error}
  \|E_t\|_\infty+\|E_t\|_{L^1(\omega)}=O(t^{-p}).
\end{equation}
Since \(M_p(t)=2c\log t+O(1)\), this verifies the admissible input of
Corollary~\ref{thm:admissible-borderline-tail-entropy}.
With \(H(t)=D_{\mathrm{KL}}(P_t*\nu\|P_t)\),
\begin{equation}\label{eq:core-log-tail-counterexample-asymptotic}
H(t)=\sum_{m=2}^{N-1}A_{2m}(\nu)t^{-2m}
    +2c\,\kappa_N\,\mu_2\,t^{-2N}\log t
    +o(t^{-2N}\log t),
\end{equation}
where
\[
\kappa_N=J(2,p)=(-1)^N(N-1)2^{-2N+2}\ne0
\]
is the quadratic cross coefficient, equivalently the coefficient of the
monomial
\(\mathfrak m_2\mathfrak m_{2N-2}\) in
Proposition~\ref{prop:poisson-kl-moment-threshold-sharpness}.  Thus, for this
explicit power-tail law, the missing \(A_{2N}t^{-2N}\) polynomial layer is
replaced by the displayed tail-dependent logarithmic layer.  The case \(N=2\)
is excluded from this statement.
\end{corollary}

\subsubsection*{Perturbative power-tail verification}

The perturbative verification
\ref{thm:exact-perturbative-power-tail-log-replacement} is separated
from the exact power law.  For later reference, when \(p\ge4\), \(R>0\), and
\(r\) is an even integrable function on \(\{|x|\ge R\}\), the four
perturbative quotient controls are
\begin{equation}\label{eq:perturbative-power-tail-lower-moments}
  \int_{|x|\ge R}|x|^q|r(x)|\,\dd x<\infty
  \qquad(0\le q<p),
\end{equation}
\begin{equation}\label{eq:perturbative-power-tail-top-moment}
  \int_{R\le |x|\le t}|x|^p r(x)\,\dd x=O(1),
\end{equation}
\begin{equation}\label{eq:perturbative-power-tail-local-remainder}
  t^{-(p+1)}\int_{R\le |x|\le t}|x|^{p+1}|r(x)|\,\dd x=O(t^{-p}),
\end{equation}
and
\begin{equation}\label{eq:perturbative-power-tail-far-remainder}
  \int_{|x|>t}\left(1+\frac{x^2}{t^2}\right)|r(x)|\,\dd x=O(t^{-p}).
\end{equation}
These are precisely the perturbative subclass conditions in
Definition~\ref{def:admissible-borderline-tail-classes}; they verify
membership in \(\mathcal A_{p,R,c}^{\rm pert}\) after the perturbation has
been absorbed into the compact-plus-tail decomposition.
Lemma~\ref{lem:perturbative-power-tail-quotient-expansion}
records the resulting quotient expansion.  The theorem below repeats the
controls and the quotient expansion before transferring it to entropy.  No
conclusion is claimed for a perturbation that fails these controls.

\begin{lemma}[Pointwise sufficient condition for the perturbative tail]%
\label{lem:pointwise-perturbative-power-tail-sufficient-condition}
Let \(p\ge4\), \(R>0\), and let \(r\) be a measurable function on
\(\{|x|\ge R\}\).  Suppose that, for some \(C<\infty\) and \(\eta>0\),
\begin{equation}\label{eq:pointwise-perturbative-tail-condition}
  |r(x)|\le C|x|^{-p-1-\eta}\qquad(|x|\ge R).
\end{equation}
Then \(r\) satisfies
\eqref{eq:perturbative-power-tail-lower-moments}--\eqref{eq:perturbative-power-tail-far-remainder}.
\end{lemma}

\begin{proof}
For \(0\le q<p\),
\[
  \int_{|x|\ge R}|x|^q|r(x)|\,\dd x
  \le 2C\int_R^\infty x^{q-p-1-\eta}\,\dd x<\infty,
\]
which proves \eqref{eq:perturbative-power-tail-lower-moments}.  Also
\[
  \left|\int_{R\le |x|\le t}|x|^pr(x)\,\dd x\right|
  \le 2C\int_R^\infty x^{-1-\eta}\,\dd x=O(1),
\]
so \eqref{eq:perturbative-power-tail-top-moment} holds.  For the local
Taylor-remainder condition,
\[
  t^{-(p+1)}\int_{R\le |x|\le t}|x|^{p+1}|r(x)|\,\dd x
  \le
  2Ct^{-(p+1)}\int_R^t x^{-\eta}\,\dd x
  =
  \begin{cases}
    O(t^{-p-\eta}),&0<\eta<1,\\
    O(t^{-p-1}\log t),&\eta=1,\\
    O(t^{-p-1}),&\eta>1,
  \end{cases}
\]
and each alternative is \(O(t^{-p})\).  Finally,
\[
\begin{aligned}
  \int_{|x|>t}\left(1+\frac{x^2}{t^2}\right)|r(x)|\,\dd x
  &\le
  2C\int_t^\infty x^{-p-1-\eta}\,\dd x
  +2Ct^{-2}\int_t^\infty x^{1-p-\eta}\,\dd x  \\
  &=O(t^{-p-\eta})=O(t^{-p}),
\end{aligned}
\]
because \(p\ge4\).  This proves
\eqref{eq:perturbative-power-tail-far-remainder}.
\end{proof}

\begin{lemma}[Borderline exact power-tail quotient expansion]%
\label{lem:borderline-exact-power-tail-quotient-expansion}
Let \(p\ge4\) be even.  Let
\[
  f(x)=g(x)+c|x|^{-(p+1)}\mathbf1_{\{|x|\ge R\}},
\]
where \(c>0\), \(R>0\), and \(g\) is a symmetric nonnegative integrable
compactly supported function.  Assume that \(f\) is normalized to have mass
one, and let \(\nu\) be the corresponding symmetric law.  For
\[
  \Delta_t(y):=\frac{(P_t*\nu)(ty)}{P_t(ty)}-1
  =\int_{\RR}\{R_{x/t}(y)-1\}f(x)\,\dd x ,
\]
and for \(2\le n\le p-1\), let \(\mu_n=\int x^nf(x)\,\dd x\).  Then, in
\(L^\infty(\RR)\cap L^1(\omega)\),
\begin{equation}\label{eq:borderline-exact-power-tail-quotient-expansion}
  \Delta_t(y)=
  \sum_{n=2}^{p-1}\mu_nu_n(y)t^{-n}
  +2c\,u_p(y)t^{-p}\log t+E_t(y),
\end{equation}
where
\begin{equation}\label{eq:borderline-exact-power-tail-quotient-error}
  \|E_t\|_\infty+\|E_t\|_{L^1(\omega)}=O(t^{-p}).
\end{equation}
The implicit constant may depend on \(p,R,c\), and \(g\), but not on \(t\).
\end{lemma}

\begin{proof}
Choose \(B\ge R\) with \(\operatorname{supp}g\subset[-B,B]\).  It is enough
to consider \(t\ge \max\{B,1\}\).  Split the compact core as
\[
  g=g_{\rm in}+g_{\rm out},\qquad
  g_{\rm in}=g\,\mathbf1_{\{|x|<R\}},\qquad
  \operatorname{supp}g_{\rm out}\subset\{R\le |x|\le B\}.
\]
We decompose \(\Delta_t=C_t^{\rm in}+C_t^{\rm out}+T_t+F_t\), where
\(C_t^{\rm in}\) and \(C_t^{\rm out}\) are the contributions of
\(g_{\rm in}\) and \(g_{\rm out}\), \(T_t\) is the explicit power-tail
contribution on \(R\le |x|\le t\), and \(F_t\) is the explicit far tail
\(|x|>t\).  All estimates below are uniform in \(y\); since \(\omega\) is a
probability measure, the same bounds also give the \(L^1(\omega)\) estimates.

\emph{Compact core.}  On the full compact set \(|x|\le B\), the local Cayley
Taylor estimate gives
\[
  R_{x/t}(y)-1
  =\sum_{n=1}^{p}u_n(y)(x/t)^n
   +O_{p,B}(|x/t|^{p+1})
\]
in \(L^\infty_y\).  Hence
\[
  C_t^{\rm in}(y)+C_t^{\rm out}(y)
  =\sum_{n=1}^{p}u_n(y)t^{-n}\int_{\RR}x^ng(x)\,\dd x
          +O(t^{-(p+1)}).
\]
The odd core terms vanish by symmetry.  The \(n=p\) core term is \(O(t^{-p})\)
and will be absorbed into \(E_t\); the terms \(2\le n\le p-1\) contribute the
core parts of the moments \(\mu_n\).

This explicitly includes the possible compact outside piece.  Equivalently,
for \(t>B\),
\[
\int g_{\rm out}(x)\{R_{x/t}(y)-1\}\,\dd x
=\sum_{n=1}^{p}u_n(y)t^{-n}
  \int x^ng_{\rm out}(x)\,\dd x+O(t^{-(p+1)}).
\]
Evenness removes odd \(n\).  The terms \(2\le n<p\) are part of the lower
moments \(\mu_n\).  If
\[
  C_p:=\int x^pg_{\rm out}(x)\,\dd x ,
\]
then the compact outside top contribution is \(C_pu_p(y)t^{-p}\).  In the
later entropy transfer with \(p=2N-2\), its only possible logarithmic-scale
interaction with the leading \(\mu_2u_2t^{-2}\)-mode has size
\[
  \mu_2C_pJ(2,p)t^{-(p+2)}=O(t^{-2N})=o(t^{-2N}\log t),
\]
so this compact \(p\)-mode cannot change the logarithmic coefficient
\(2c\kappa_N\mu_2\) in
Theorem~\ref{prop:log-tail-counterexample-top-moment}.

\emph{Truncated tail moments.}  On \(R\le |x|\le t\), the same expansion gives
\[
  T_t(y)=
  \sum_{n=1}^{p}u_n(y)t^{-n}
  \int_{R\le |x|\le t}x^nc|x|^{-p-1}\,\dd x
  +\mathcal R_t(y).
\]
Again the odd terms vanish.  For \(2\le n<p\), replacing the truncated tail
moment by the full tail moment costs
\[
  t^{-n}\int_{|x|>t}|x|^nc|x|^{-p-1}\,\dd x
  \le Ct^{-n}\int_t^\infty x^{n-p-1}\,\dd x
  =O(t^{-p}).
\]
Thus the core and truncated-tail terms together give
\(\sum_{n=2}^{p-1}\mu_nu_n(y)t^{-n}\), up to an \(O(t^{-p})\) error.  The
borderline top tail mode is
\[
  t^{-p}u_p(y)\int_{R\le |x|\le t}x^pc|x|^{-p-1}\,\dd x
  =
  2c\,u_p(y)t^{-p}\int_R^t\frac{\dd x}{x}
  =2c\,u_p(y)t^{-p}\log t+O(t^{-p}).
\]

\emph{Taylor remainder on the truncated tail.}  The remainder satisfies
\[
  \|\mathcal R_t\|_\infty
  \le
  Ct^{-(p+1)}
  \int_{R\le |x|\le t}|x|^{p+1}c|x|^{-p-1}\,\dd x
  \le Ct^{-p}.
\]
This is the only place where the nonintegrable \((p+1)\)-moment would have
entered; truncation at \(t\) leaves exactly the \(O(t^{-p})\) remainder needed
for \eqref{eq:borderline-exact-power-tail-quotient-error}.

\emph{Far tail.}  For all \(\varepsilon\in\RR\),
\[
  |R_\varepsilon(y)-1|\le C(1+\varepsilon^2)
\]
uniformly in \(y\).  Since \(p\ge4\),
\[
  \|F_t\|_\infty
  \le
  C\int_{|x|>t}\left(1+\frac{x^2}{t^2}\right)c|x|^{-p-1}\,\dd x
  \le
  C\left\{\int_t^\infty x^{-p-1}\,\dd x
  +t^{-2}\int_t^\infty x^{1-p}\,\dd x\right\}
  =O(t^{-p}).
\]
The already-assigned polynomial moments omitted from \(|x|>t\) are also
\(O(t^{-p})\) by the estimate in the truncated-tail paragraph.  Combining the
compact-core, truncated-tail, Taylor-remainder, and far-tail estimates proves
\eqref{eq:borderline-exact-power-tail-quotient-expansion} and
\eqref{eq:borderline-exact-power-tail-quotient-error}.
\end{proof}

\begin{proof}[Proof of Corollary~\ref{prop:log-tail-counterexample-top-moment}]
The mass condition makes \eqref{eq:normalized-borderline-tail-density} a
probability density:
\[
  \int_{\RR}f(x)\,\dd x
  =\left(1-\frac{2c}{pR^p}\right)
    +2c\int_R^\infty x^{-p-1}\,\dd x
  =1 .
\]
The compact core is symmetric and nonnegative by construction, the tail is
symmetric, and therefore \(\nu\) is centred.  The moment assertions are
immediate from compact support of \(g\) and
\[
2c\int_R^\infty x^q x^{-p-1}\,\dd x<\infty
\quad\Longleftrightarrow\quad q<p.
\]
The truncated top moment is
\[
\EE\!\left[X^p\mathbf1_{\{|X|\le t\}}\right]
=O(1)+2c\int_R^t x^{-1}\,\dd x
=2c\log t+O(1).
\]

For \(x=ty\), write
\[
\Delta_t(y)=\frac{(P_t*\nu)(ty)}{P_t(ty)}-1
=\int_{\RR}\{R_{x/t}(y)-1\}f(x)\dd x .
\]
Lemma~\ref{lem:borderline-exact-power-tail-quotient-expansion} gives the
quotient expansion \eqref{eq:exact-power-tail-theorem-quotient} with
\eqref{eq:exact-power-tail-theorem-quotient-error}, namely
\[
\Delta_t(y)=\sum_{n=2}^{p-1}\mu_nu_n(y)t^{-n}
            +2c\,u_p(y)t^{-p}\log t+\varepsilon_t(y),
\]
with
\(\|\varepsilon_t\|_\infty+\|\varepsilon_t\|_{L^1(\omega)}\le Ct^{-p}\).
The compact-core, truncated-tail, local-remainder, and far-tail estimates in
Lemma~\ref{lem:borderline-exact-power-tail-quotient-expansion} are precisely
the four controls
\eqref{eq:admissible-lower-tail-control}--\eqref{eq:admissible-far-tail-control};
together with \(M_p(t)=2c\log t+O(1)\), they prove
\(\nu\in\mathcal A_{p,R,c}\).  The admissible quotient consequence,
Corollary~\ref{thm:admissible-borderline-tail-entropy}, therefore gives
\[
H(t)=
\sum_{m=2}^{N-1}A_{2m}(\nu)t^{-2m}
+\kappa_N\mu_2\,t^{-2N}M_p(t)
+o(t^{-2N}M_p(t)).
\]
Substituting \(M_p(t)=2c\log t+O(1)\) proves
\eqref{eq:core-log-tail-counterexample-asymptotic}.
\end{proof}

\begin{lemma}[Perturbative power-tail quotient expansion]%
\label{lem:perturbative-power-tail-quotient-expansion}
Let \(p\ge4\) be even.  Let \(R>0\), \(c>0\), let
\(g\in L^1(\RR)\) be symmetric and nonnegative with
\(\operatorname{supp}g\subset(-R,R)\), and let \(r\) be an even integrable
function on \(\{|x|\ge R\}\).  Assume that
\[
  f(x)=g(x)+\bigl(c|x|^{-p-1}+r(x)\bigr)\mathbf1_{\{|x|\ge R\}}
\]
is nonnegative and normalized to have mass one, and assume
\eqref{eq:perturbative-power-tail-lower-moments}--\eqref{eq:perturbative-power-tail-far-remainder}.
Let \(\nu\) be the law with density \(f\), put
\[
  M_p(t):=\int_{|x|\le t}x^pf(x)\,\dd x,
  \qquad
  \Delta_t(y):=\frac{(P_t*\nu)(ty)}{P_t(ty)}-1 ,
\]
and, for \(2\le n\le p-1\), let
\(\mu_n=\int_{\RR}x^nf(x)\,\dd x\).  Then \(\nu\) is centred, has finite
absolute moments of every order \(q<p\), and
\begin{equation}\label{eq:perturbative-power-tail-top-truncated-moment}
  M_p(t)=2c\log t+O(1).
\end{equation}
Moreover, in \(L^\infty(\RR)\cap L^1(\omega)\),
\begin{equation}\label{eq:perturbative-power-tail-quotient-expansion}
  \Delta_t(y)=
  \sum_{n=2}^{p-1}\mu_nu_n(y)t^{-n}
  +u_p(y)t^{-p}M_p(t)+E_t(y),
\end{equation}
where
\begin{equation}\label{eq:perturbative-power-tail-quotient-error}
  \|E_t\|_\infty+\|E_t\|_{L^1(\omega)}=O(t^{-p}).
\end{equation}
\end{lemma}

\begin{proof}
It is enough to consider \(t\ge\max\{R,1\}\).  The symmetry of \(g\), of
\(|x|^{-p-1}\), and of \(r\) makes \(f\) symmetric; hence \(\nu\) is centred
and every odd truncated or full moment below vanishes.  The compact core has
all moments, the explicit tail has finite \(q\)th absolute moment exactly for
\(q<p\), and
\eqref{eq:perturbative-power-tail-lower-moments} gives the corresponding
moments of \(r\).  Thus the lower moments \(\mu_n\), \(2\le n<p\), are
well-defined.

We first record the four perturbative estimates used in the quotient
calculation.  From \eqref{eq:perturbative-power-tail-local-remainder} there is
a constant \(C\) such that
\[
  A(T):=\int_{R\le |x|\le T}|x|^{p+1}|r(x)|\,\dd x\le CT
  \qquad(T\ge R).
\]
For \(0\le n<p\), a dyadic decomposition gives the lower-moment tail
replacement
\begin{equation}\label{eq:perturbative-lower-moment-tail-replacement}
\begin{aligned}
  \int_{|x|>t}|x|^n|r(x)|\,\dd x
  &\le
  \sum_{k=0}^{\infty}(2^kt)^{\,n-p-1}
  \int_{2^kt<|x|\le2^{k+1}t}|x|^{p+1}|r(x)|\,\dd x  \\
  &\le C_n t^{\,n-p}.
\end{aligned}
\end{equation}
Consequently
\[
  t^{-n}\int_{R\le |x|\le t}x^nr(x)\,\dd x
  =
  t^{-n}\int_{|x|\ge R}x^nr(x)\,\dd x+O(t^{-p})
  \qquad(0\le n<p).
\]
The top signed moment of the perturbation is controlled separately by
\eqref{eq:perturbative-power-tail-top-moment}:
\begin{equation}\label{eq:perturbative-top-signed-moment-bound}
  \int_{R\le |x|\le t}x^pr(x)\,\dd x=O(1).
\end{equation}
Hence
\[
  M_p(t)
  =
  \int_{|x|<R}x^pg(x)\,\dd x
  +2c\int_R^t x^{-1}\,\dd x
  +\int_{R\le |x|\le t}x^pr(x)\,\dd x
  =2c\log t+O(1),
\]
which proves \eqref{eq:perturbative-power-tail-top-truncated-moment}.

The local Taylor remainder for the perturbation is also explicit.  On
\(|x|\le t\), \eqref{eq:one-dimensional-two-region-taylor-remainder} gives
\[
  \left|R_{x/t}(y)-\sum_{n=0}^{p}u_n(y)(x/t)^n\right|
  \le C_p |x/t|^{p+1}
  \qquad(y\in\RR),
\]
and therefore
\begin{equation}\label{eq:perturbative-local-taylor-remainder-bound}
\left\|
\int_{R\le |x|\le t}
\left\{R_{x/t}-\sum_{n=0}^{p}u_n(x/t)^n\right\}r(x)\,\dd x
\right\|_{\infty}
\le
C_pt^{-(p+1)}
\int_{R\le |x|\le t}|x|^{p+1}|r(x)|\,\dd x
=O(t^{-p}).
\end{equation}
Since \(\omega\) is a probability measure, the same estimate holds in
\(L^1(\omega)\).  Finally, the elementary bound
\(|R_\varepsilon(y)-1|\le C(1+\varepsilon^2)\), uniform in \(y\), and
\eqref{eq:perturbative-power-tail-far-remainder} give the far-tail quotient
bound
\begin{equation}\label{eq:perturbative-far-tail-quotient-bound}
\left\|
\int_{|x|>t}\{R_{x/t}-1\}r(x)\,\dd x
\right\|_\infty
+
\left\|
\int_{|x|>t}\{R_{x/t}-1\}r(x)\,\dd x
\right\|_{L^1(\omega)}
=O(t^{-p}).
\end{equation}

We now assemble the quotient.  Split the integral defining \(\Delta_t\) into
\(|x|\le t\) and \(|x|>t\).  On \(|x|\le t\), Taylor expansion through order
\(p\) gives
\[
  \int_{|x|\le t}\{R_{x/t}(y)-1\}f(x)\,\dd x
  =
  \sum_{n=1}^{p}u_n(y)t^{-n}\int_{|x|\le t}x^nf(x)\,\dd x
  +\mathcal R_t(y).
\]
The compact core and the explicit \(c|x|^{-p-1}\) tail have local remainders
\(O(t^{-p})\), by the same estimates displayed in
Lemma~\ref{lem:borderline-exact-power-tail-quotient-expansion}; the
perturbative part is controlled by
\eqref{eq:perturbative-local-taylor-remainder-bound}.  Thus
\(\|\mathcal R_t\|_\infty+\|\mathcal R_t\|_{L^1(\omega)}=O(t^{-p})\).
For \(2\le n<p\), replacing the truncated \(n\)th moment by the full moment
costs \(O(t^{-p})\): for the explicit power tail this is
\[
  t^{-n}\int_{|x|>t}|x|^n c|x|^{-p-1}\,\dd x=O(t^{-p}),
\]
and for \(r\) it is
\eqref{eq:perturbative-lower-moment-tail-replacement}.  The odd terms vanish
by symmetry, and the \(n=p\) term is exactly \(u_p(y)t^{-p}M_p(t)\).

It remains only to estimate the far tail.  For the explicit power tail,
\[
  \int_{|x|>t}(1+x^2/t^2)c|x|^{-p-1}\,\dd x=O(t^{-p}),
\]
and the perturbative far tail is
\eqref{eq:perturbative-far-tail-quotient-bound}.  Combining the local
expansion, the lower-moment replacements, the top truncated moment, and the
far-tail bound gives \eqref{eq:perturbative-power-tail-quotient-expansion} and
\eqref{eq:perturbative-power-tail-quotient-error}.
\end{proof}

\begin{corollary}[Perturbative-power verification corollary]%
\label{thm:exact-perturbative-power-tail-log-replacement}
\label{cor:perturbative-power-tail-log-replacement}
Let \(N\ge3\), put \(p=2N-2\), and let \(R>0\), \(c>0\).  Let \(f\) be an
even probability density of the form
\[
f(x)=g(x)+\{c|x|^{-p-1}+r(x)\}\mathbf1_{\{|x|\ge R\}},
\]
where \(g\in L^1(\RR)\) is symmetric and nonnegative with
\(\operatorname{supp}g\subset(-R,R)\), while \(r\) is even and satisfies
\[
\int_{|x|\ge R}|x|^q|r(x)|\,\dd x<\infty\quad(0\le q<p),
\qquad
\int_{R\le |x|\le t}|x|^pr(x)\,\dd x=O(1),
\]
\[
t^{-(p+1)}\int_{R\le |x|\le t}|x|^{p+1}|r(x)|\,\dd x=O(t^{-p}),
\qquad
\int_{|x|>t}\left(1+\frac{x^2}{t^2}\right)|r(x)|\,\dd x=O(t^{-p}).
\]
Let \(\nu\) be the law with density \(f\), set
\[
H(t):=D_{\mathrm{KL}}(P_t*\nu\|P_t),
\qquad
\mu_2:=\int_{\RR}x^2f(x)\,\dd x,
\]
and let
\[
\kappa_N:=J(2,p)=(-1)^N(N-1)2^{-2N+2}.
\]
Then \(\nu\) is centred, has finite absolute moments of every order \(q<p\),
belongs to
\(\mathcal A_{p,R,c}^{\rm pert}\subset\mathcal A_{p,R,c}\) in the sense of
Definition~\ref{def:admissible-borderline-tail-classes}, and
\[
  M_p(t):=\EE[X^p\mathbf1_{\{|X|\le t\}}]=2c\log t+O(1).
\]
Moreover, for
\[
  \Delta_t(y):=\frac{(P_t*\nu)(ty)}{P_t(ty)}-1,
\]
the quotient expansion
\begin{equation}\label{eq:perturbative-power-tail-theorem-quotient}
\Delta_t(y)=
\sum_{n=2}^{p-1}\mu_nu_n(y)t^{-n}
+u_p(y)t^{-p}M_p(t)+E_t(y)
\end{equation}
holds in \(L^\infty(\RR)\cap L^1(\omega)\), with
\begin{equation}\label{eq:perturbative-power-tail-theorem-quotient-error}
\|E_t\|_\infty+\|E_t\|_{L^1(\omega)}=O(t^{-p}).
\end{equation}
Since \(M_p(t)=2c\log t+O(1)\), this verifies the admissible input of
Corollary~\ref{thm:admissible-borderline-tail-entropy}.  Consequently
\[
D_{\mathrm{KL}}(P_t*\nu\|P_t)
=\sum_{m=2}^{N-1}A_{2m}(\nu)t^{-2m}
+2c\,\kappa_N\mu_2\,t^{-2N}\log t
+o(t^{-2N}\log t).
\]
\end{corollary}

\begin{proof}
The support condition on \(g\) makes the displayed formula a disjoint
core-plus-tail decomposition: \(g\) contributes only on \((-R,R)\), while the
explicit power tail and the perturbation \(r\) contribute only on
\(\{|x|\ge R\}\).  Therefore the hypotheses are exactly those of
Lemma~\ref{lem:perturbative-power-tail-quotient-expansion} with this same
cutoff \(R\).  That lemma proves the centering, the lower moment assertions,
and
\[
  M_p(t)=\EE[X^p\mathbf1_{\{|X|\le t\}}]=2c\log t+O(1).
\]
It also gives the quotient expansion
\eqref{eq:perturbative-power-tail-theorem-quotient} with
\eqref{eq:perturbative-power-tail-theorem-quotient-error}, namely
\[
\Delta_t(y)=
\sum_{n=2}^{p-1}\mu_nu_n(y)t^{-n}
+u_p(y)t^{-p}M_p(t)+E_t(y),
\qquad
\|E_t\|_\infty+\|E_t\|_{L^1(\omega)}=O(t^{-p}).
\]
The same lemma proves the lower-moment replacement, the top truncated-moment
identity, the local Taylor-remainder bound, and the far-tail bound that form
\eqref{eq:admissible-lower-tail-control}--\eqref{eq:admissible-far-tail-control}.
Since \(M_p(t)\sim2c\log t\), these estimates verify
\(\nu\in\mathcal A_{p,R,c}^{\rm pert}\subset\mathcal A_{p,R,c}\) from
Definition~\ref{def:admissible-borderline-tail-classes}.
Since \(M_p(t)\sim2c\log t\), the error is
\(o(t^{-p}M_p(t))\), while
\[
  M_p(t)\to\infty,\qquad M_p(t)=o(t^{p-2})
\]
because \(p\ge4\).  Thus \(\nu\in\mathcal A_{p,R,c}\), and the admissible
quotient consequence, Corollary~\ref{thm:admissible-borderline-tail-entropy}, gives
\[
H(t)=
\sum_{m=2}^{N-1}A_{2m}(\nu)t^{-2m}
+\kappa_N\mu_2\,t^{-2N}M_p(t)
+o(t^{-2N}M_p(t)).
\]
Substituting \(M_p(t)=2c\log t+O(1)\) proves the displayed asymptotic.
\end{proof}

\subsubsection*{Monotone slowly varying verification}

The next lemma is where slow variation is used to prove membership in the
monotone slowly varying subclass \(\mathcal A_{p,R,c}^{\rm sv}\) of
Definition~\ref{def:admissible-borderline-tail-classes}.

\begin{lemma}[Slow variation supplies the quotient controls]%
\label{lem:slowly-varying-quotient-controls}
Let \(p\ge4\), \(R>0\), and \(c>0\).  Let \(L\) be eventually positive,
eventually monotone, and slowly varying on \([R,\infty)\).  Put
\[
  \mathcal M_L(t):=\int_R^t\frac{L(s)}s\,\dd s,
\]
and assume \(\mathcal M_L(t)\to\infty\).  Then, as \(t\to\infty\),
\begin{equation}\label{eq:slowly-varying-top-moment-growth}
  L(t)=o(\mathcal M_L(t)),\qquad
  \mathcal M_L(t)=o(t^\alpha)\quad\text{for every }\alpha>0 .
\end{equation}
Moreover, for every \(0\le n<p\), Karamata's theorem gives the sharper
tail equivalent
\begin{equation}\label{eq:slowly-varying-karamata-tail-equivalent}
t^{-n}\int_t^\infty x^{n-p-1}L(x)\,\dd x
\sim \frac{1}{p-n}t^{-p}L(t)
=o(t^{-p}\mathcal M_L(t)),
\end{equation}
and hence
\begin{equation}\label{eq:slowly-varying-lower-potter-control}
t^{-n}\int_t^\infty x^n c x^{-p-1}L(x)\,\dd x
=o(t^{-p}\mathcal M_L(t)),
\end{equation}
and, with \(r_t(x):=c x^{-p-1}\{L(x)-L(t)\}\) on \(R\le x\le t\),
\begin{equation}\label{eq:slowly-varying-local-potter-control}
t^{-(p+1)}\int_R^t x^{p+1}|r_t(x)|\,\dd x
=o(t^{-p}\mathcal M_L(t)),
\end{equation}
while
\begin{equation}\label{eq:slowly-varying-far-potter-control}
\int_t^\infty\left(1+\frac{x^2}{t^2}\right)
c x^{-p-1}L(x)\,\dd x
=O(t^{-p}L(t))=o(t^{-p}\mathcal M_L(t)).
\end{equation}
Finally, eventual monotonicity and Karamata's theorem imply
\begin{equation}\label{eq:slowly-varying-local-karamata-control}
t^{-(p+1)}\int_R^t |L(x)-L(t)|\,\dd x
\le
t^{-(p+1)}\left(\int_R^t L(x)\,\dd x+tL(t)\right)
=O(t^{-p}L(t))
=o(t^{-p}\mathcal M_L(t)).
\end{equation}
For \(0\le n<p\), the lower tail moment
\(\int_R^\infty x^{n-p-1}L(x)\,\dd x\) is finite.
\end{lemma}

\begin{proof}
We use only standard consequences of slow variation, recorded here at the
scales needed in the quotient argument.  Finite initial intervals do not affect
any of the asymptotic estimates, so we may increase \(R\) once so that \(L\) is
positive and monotone on \([R,\infty)\).  Potter's estimate says that, for
each \(\delta>0\), after this enlargement there is a constant \(C_\delta\) such
that, for \(x,t\ge R\),
\begin{equation}\label{eq:potter-bound-used-locally}
  \frac{L(x)}{L(t)}
  \le C_\delta\max\{(x/t)^\delta,(t/x)^\delta\}.
\end{equation}
This estimate is a consequence of the uniform convergence theorem for slowly
varying functions and is not an additional hypothesis.

First we compare \(L(t)\) with \(\mathcal M_L(t)\).  Fix \(A>1\).  By the
uniform convergence theorem, \(L(tu)/L(t)\to1\) uniformly for
\(u\in[1/A,1]\).  Hence, for \(t\) large,
\[
  \mathcal M_L(t)\ge\int_{t/A}^{t}\frac{L(s)}s\,\dd s
  \ge \frac12L(t)\log A .
\]
Since \(A\) is arbitrary and \(\mathcal M_L(t)\to\infty\), this gives
\(L(t)=o(\mathcal M_L(t))\).  Taking
\(\delta=\alpha/2\) in \eqref{eq:potter-bound-used-locally} with \(t=R\)
gives \(L(s)\le C_\alpha s^{\alpha/2}\) for large \(s\), and therefore
\[
  \mathcal M_L(t)\le C_\alpha\int_R^t s^{\alpha/2-1}\,\dd s
  =O(t^{\alpha/2})=o(t^\alpha).
\]
This proves \eqref{eq:slowly-varying-top-moment-growth}.

Let \(0\le n<p\) and set \(\beta=p-n>0\).  Choose
\(0<\delta<\beta\).  For \(x\ge t\), Potter's estimate gives
\[
  L(x)\le C_\delta L(t)(x/t)^\delta .
\]
Thus
\[
\begin{aligned}
t^{-n}\int_t^\infty x^n c x^{-p-1}L(x)\,\dd x
&\le C t^{-n}L(t)t^{-\delta}
   \int_t^\infty x^{-\beta-1+\delta}\,\dd x  \\
&\le C_{\beta,\delta}t^{-p}L(t)
 =o(t^{-p}\mathcal M_L(t)).
\end{aligned}
\]
The exact equivalent in
\eqref{eq:slowly-varying-karamata-tail-equivalent} is the standard Karamata
tail theorem applied to the regularly varying function
\(x\mapsto x^{n-p-1}L(x)\), whose index is less than \(-1\).
The same estimate with \(t=R\), or equivalently Potter with
\(\delta<\beta\), gives
\(\int_R^\infty x^{n-p-1}L(x)\,\dd x<\infty\).

For the local Taylor remainder, eventual monotonicity and Karamata's theorem
give \(\int_R^tL(x)\,\dd x=O(tL(t))\).  Consequently
\[
\begin{aligned}
t^{-(p+1)}\int_R^t x^{p+1}|r_t(x)|\,\dd x
&=ct^{-(p+1)}\int_R^t |L(x)-L(t)|\,\dd x  \\
&\le Ct^{-(p+1)}\left(\int_R^t L(x)\,\dd x+tL(t)\right)  \\
&=O(t^{-p}L(t))
 =o(t^{-p}\mathcal M_L(t)).
\end{aligned}
\]
This is also the displayed control
\eqref{eq:slowly-varying-local-karamata-control}.

Finally, applying
\eqref{eq:slowly-varying-karamata-tail-equivalent} with \(n=0\) and \(n=2\)
gives
\[
  \int_t^\infty x^{-p-1}L(x)\,\dd x
  \sim \frac1p t^{-p}L(t),
  \qquad
  t^{-2}\int_t^\infty x^{1-p}L(x)\,\dd x
  \sim \frac1{p-2}t^{-p}L(t).
\]
Together with \(L(t)=o(\mathcal M_L(t))\), this proves
\eqref{eq:slowly-varying-far-potter-control}.
\end{proof}

\begin{corollary}[Monotone slowly varying verification corollary]%
\label{cor:conditional-slowly-varying-borderline-tail}
Let \(N\ge3\), put \(p=2N-2\), and let \(R>0\), \(c>0\).  Let \(L\) be
slowly varying, eventually positive, eventually monotone, and nonnegative on
\([R,\infty)\), and let \(g\in L^1(\RR)\) be symmetric and nonnegative with
\(\operatorname{supp}g\subset(-R,R)\).  Suppose
\[
  f(x)=g(x)+c|x|^{-p-1}L(|x|)\mathbf1_{\{|x|\ge R\}}
\]
is normalized to have mass one, and let \(\nu\) be the corresponding
symmetric probability law.  Put
\[
M_p(t):=\int_{|x|\le t}x^pf(x)\,\dd x
       =\int_{|x|<R}x^pg(x)\,\dd x
         +2c\int_R^t\frac{L(s)}s\,\dd s .
\]
Within this monotone slowly varying subclass, assume that \(M_p(t)\to\infty\).
This divergence is used only to prove the quotient estimates displayed below;
it is not, by itself, an entropy hypothesis.
Then \(\nu\in\mathcal A_{p,R,c}^{\rm sv}\subset\mathcal A_{p,R,c}\), where
the classes are those of
Definition~\ref{def:admissible-borderline-tail-classes}.  With
\[
  \Delta_t(y):=\frac{(P_t*\nu)(ty)}{P_t(ty)}-1,
  \qquad H(t):=D_{\mathrm{KL}}(P_t*\nu\|P_t),
\]
one has, in \(L^\infty(\RR)\cap L^1(\omega)\),
\begin{equation}\label{eq:slowly-varying-quotient-expansion}
  \Delta_t(y)=
  \sum_{n=2}^{p-1}\mu_nu_n(y)t^{-n}
  +u_p(y)t^{-p}M_p(t)
  +o(t^{-p}M_p(t)),
\end{equation}
where \(\mu_n=\int x^nf(x)\,\dd x\) for \(2\le n<p\).  Consequently
\[
  M_p(t)\to\infty,\qquad M_p(t)=o(t^{p-2}),
\]
and the quotient expansion is exactly of the conditional-transfer form
\[
  \Delta_t(y)=
  \sum_{n=2}^{p-1}\mu_nu_n(y)t^{-n}
  +u_p(y)t^{-p}M_p(t)+e_t(y),
  \qquad
  \|e_t\|_\infty+\|e_t\|_{L^1(\omega)}=o(t^{-p}M_p(t)).
\]
Applying Corollary~\ref{thm:admissible-borderline-tail-entropy} therefore gives
\begin{equation}\label{eq:slowly-varying-entropy-expansion}
H(t)=
\sum_{m=2}^{N-1}A_{2m}(\nu)t^{-2m}
+\kappa_N\mu_2\,t^{-2N}M_p(t)
+o(t^{-2N}M_p(t)),
\end{equation}
where
\[
  \kappa_N=J(2,p)=(-1)^N(N-1)2^{-2N+2}.
\]
\end{corollary}

\begin{proof}
The law is symmetric, hence centred, and all odd moments below vanish.  The
compact core has all moments.  Since
\[
  M_p(t)=O(1)+2c\,\mathcal M_L(t),
  \qquad
  \mathcal M_L(t):=\int_R^t\frac{L(s)}s\,\dd s,
\]
the assumption \(M_p(t)\to\infty\) is equivalent to
\(\mathcal M_L(t)\to\infty\).  Lemma~\ref{lem:slowly-varying-quotient-controls}
therefore gives
\[
  L(t)=o(M_p(t)),\qquad M_p(t)=o(t^{p-2}),
\]
as well as the three quotient-scale estimates below with \(M_p(t)\) in place
of \(\mathcal M_L(t)\).  It also gives the finiteness of the lower tail
moments, so \(\mu_n=\int x^nf(x)\,\dd x\) exists for \(0\le n<p\).
The lower-moment tail replacement is
\begin{equation}\label{eq:slowly-varying-lower-moment-tail-replacement}
  t^{-n}\int_t^\infty x^n c x^{-p-1}L(x)\,\dd x=o(t^{-p}M_p(t)),
  \qquad 0\le n<p.
\end{equation}

For \(R\le x\le t\), write
\[
  r_t(x):=c x^{-p-1}\{L(x)-L(t)\}.
\]
The local Taylor-remainder control supplied by
Lemma~\ref{lem:slowly-varying-quotient-controls} is
\begin{equation}\label{eq:slowly-varying-local-taylor-bound}
t^{-(p+1)}\int_R^t x^{p+1}|r_t(x)|\,\dd x
=o(t^{-p}M_p(t)).
\end{equation}
The top truncated moment is the displayed \(M_p(t)\); equivalently
\begin{equation}\label{eq:slowly-varying-top-mode-identity}
  \int_{R\le |x|\le t}x^p c|x|^{-p-1}L(|x|)\,\dd x
  =M_p(t)+O(1),
\end{equation}
where the \(O(1)\) term is the compact core contribution.  The far-tail
quotient bound is exactly
\begin{equation}\label{eq:slowly-varying-far-tail-bound}
\left\|
\int_{|x|>t}\{R_{x/t}-1\}c|x|^{-p-1}L(|x|)\,\dd x
\right\|_{L^\infty\cap L^1(\omega)}
=o(t^{-p}M_p(t)),
\end{equation}
because \(|R_\varepsilon(y)-1|\le C(1+\varepsilon^2)\) uniformly in \(y\)
and \(\omega\) is a probability measure.

It remains to assemble the quotient.  On \(|x|<R\), the compact-core Taylor
expansion contributes the moments \(2\le n<p\), while its \(p\)th mode and
remainder are \(O(t^{-p})=o(t^{-p}M_p(t))\).  On \(R\le |x|\le t\), Taylor
expansion through degree \(p\), paired by symmetry, gives
\[
\sum_{n=2}^{p}u_n(y)t^{-n}
\int_{R\le |x|\le t}x^n c|x|^{-p-1}L(|x|)\,\dd x
+\mathcal R_t(y).
\]
The remainder \(\mathcal R_t\) is \(o(t^{-p}M_p(t))\) in
\(L^\infty\cap L^1(\omega)\): replacing \(L(x)\) by \(L(t)\) gives
\[
  t^{-(p+1)}L(t)\int_R^t \dd x
  =O(t^{-p}L(t))=o(t^{-p}M_p(t)),
\]
and the difference is controlled by
\eqref{eq:slowly-varying-local-taylor-bound}.  For \(2\le n<p\), replacing
the truncated lower moments by the full lower moments costs
\eqref{eq:slowly-varying-lower-moment-tail-replacement}.  The \(n=p\) term
is \(u_p(y)t^{-p}M_p(t)+O(t^{-p})\), hence has the asserted leading form.
Finally the far tail is \eqref{eq:slowly-varying-far-tail-bound}.  These
estimates prove \eqref{eq:slowly-varying-quotient-expansion}.  Together with
the growth conclusions supplied by
Lemma~\ref{lem:slowly-varying-quotient-controls}, they prove
\(\nu\in\mathcal A_{p,R,c}^{\rm sv}\subset\mathcal A_{p,R,c}\).  The
admissible quotient consequence,
Corollary~\ref{thm:admissible-borderline-tail-entropy}, gives
\eqref{eq:slowly-varying-entropy-expansion}.
\end{proof}
