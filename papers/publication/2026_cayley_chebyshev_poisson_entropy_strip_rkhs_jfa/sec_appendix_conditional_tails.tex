\section{Optional missing-top-moment tail transfer}%
\label{app:conditional-missing-top-moment-tails}

This optional appendix records a conditional tail replacement outside the
three main theorems.  A divergent truncated moment is not an entropy
hypothesis by itself: the input is a matched quotient expansion with an
\(L^\infty\cap L^1(\omega)\) remainder.  The table lists three tail classes;
the following lemmas verify the required quotient controls.

\subsubsection*{Conditional quotient theorem}\leavevmode\par\smallskip

\begin{theorem}[Truncated-top-mode quotient theorem]%
\label{thm:tail-replacements-proved-here}
\label{prop:conditional-truncated-top-moment-replacement}
Let \(N\ge3\), put \(p=2N-2\ge4\), and let \(\nu\) be a symmetric
probability law on \(\RR\) with finite absolute moments of every order
\(q<p\).  Let
\[
  M_p(t):=\EE\!\left[X^p\mathbf1_{\{|X|\le t\}}\right],
  \qquad
  \Delta_t(y):=\frac{(P_t*\nu)(ty)}{P_t(ty)}-1 .
\]
Assume that the quotient admits the \(L^\infty(\RR)\cap L^1(\omega)\)
expansion
\begin{equation}\label{eq:conditional-truncated-top-mode-quotient}
  \Delta_t(y)=
  A_t(y)+u_p(y)t^{-p}M_p(t)+e_t(y),
  \qquad
  A_t(y):=\sum_{n=2}^{p-1}\mu_nu_n(y)t^{-n},
\end{equation}
where
\begin{equation}\label{eq:conditional-truncated-top-mode-error}
  \|e_t\|_\infty+\|e_t\|_{L^1(\omega)}
  =o(t^{-p}M_p(t)).
\end{equation}
Assume also that
\begin{equation}\label{eq:conditional-truncated-moment-growth}
  M_p(t)\to\infty,\qquad M_p(t)=o(t^{p-2}).
\end{equation}
Then, with \(H(t)=D_{\mathrm{KL}}(P_t*\nu\|P_t)\),
\begin{equation}\label{eq:conditional-truncated-top-moment-entropy}
H(t)=
\sum_{m=2}^{N-1}A_{2m}(\nu)t^{-2m}
+J(2,p)\mu_2\,t^{-2N}M_p(t)
+o(t^{-2N}M_p(t)),
\end{equation}
where
\[
  J(2,p)=(-1)^N(N-1)2^{-2N+2}.
\]
\end{theorem}

\begin{proof}
Write \(B_t(y)=u_p(y)t^{-p}M_p(t)\) and
\(s_t=t^{-2N}M_p(t)=t^{-(p+2)}M_p(t)\).  The growth assumptions and
\eqref{eq:conditional-truncated-top-mode-error} put
\(\Delta_t=A_t+B_t+e_t\) in the moving-scale Taylor transfer
Lemma~\ref{lem:moving-scale-taylor-transfer}.  Hence
\[
  H(t)=\int_{\RR}\Phi(A_t)\,\dd\omega+\mu_2J(2,p)s_t+o(s_t),
  \qquad
  \Phi(s)=(1+s)\log(1+s)-s .
\]
Since \(\|A_t\|_\infty=O(t^{-2})\), Taylor expansion through degree \(N\)
gives
\[
  \int_{\RR}\Phi(A_t)\,\dd\omega
  =
  \sum_{m=2}^{N-1}A_{2m}(\nu)t^{-2m}+o(s_t).
\]
The next pure finite-jet layer is \(O(t^{-2N})=o(s_t)\), because
\(M_p(t)\to\infty\).  Finally,
Proposition~\ref{prop:poisson-kl-moment-threshold-sharpness} computes
\[
J(2,p)=(-1)^N(N-1)2^{-2N+2}.
\]
Substitution proves \eqref{eq:conditional-truncated-top-moment-entropy}.
\end{proof}

\subsubsection*{Borderline \(p=2\) infinite-variance layer}\leavevmode\par\smallskip

The case \(p=2\) is separate from the optional tail replacement above: the
leading KL scale is quadratic in the divergent truncated variance.  The next
calculation is only a consistency check for the sharp variance threshold.

\begin{theorem}[Borderline infinite-variance power tail]%
\label{prop:n2-borderline-infinite-variance-layer}
Let \(c>0\) satisfy \(c\mathrm e^{-2}<1\), and let \(\nu\) be the symmetric
probability law with density
\[
f(x)=\frac{1-c\mathrm e^{-2}}{2}\mathbf1_{\{|x|\le1\}}
      +c|x|^{-3}\mathbf1_{\{|x|\ge \mathrm e\}} .
\]
Then \(\nu\) has finite mean and infinite variance, and
\[
\EE\!\left[X^2\mathbf1_{\{|X|\le t\}}\right]\sim 2c\log t,
\]
while
\[
D_{\mathrm{KL}}(P_t*\nu\|P_t)
=\frac{c^2}{2}\,t^{-4}(\log t)^2+O(t^{-4}\log t).
\]
\end{theorem}

\begin{lemma}[Symmetric \(p=2\) quotient tail split]
\label{lem:symmetric-p2-far-tail-quotient}
For \(S_a(y):=R_a(y)+R_{-a}(y)-2\) there is a universal \(C<\infty\) such
that, for \(t\ge\mathrm e\),
\begin{equation}\label{eq:symmetric-p2-far-tail-quotient}
  \left\|
  \int_t^\infty S_{x/t}(y)x^{-3}\,\dd x
  \right\|_{L^\infty}
  +
  \left\|
  \int_t^\infty S_{x/t}(y)x^{-3}\,\dd x
  \right\|_{L^1(\omega)}
  \le Ct^{-2}.
\end{equation}
Moreover, for each fixed \(R\ge1\),
\begin{equation}\label{eq:symmetric-p2-full-tail-split}
\int_R^\infty S_{x/t}(y)x^{-3}\,\dd x
=2u_2(y)t^{-2}\log t+O(t^{-2})
\end{equation}
in \(L^\infty(\RR)\cap L^1(\omega)\).
\end{lemma}

\begin{proof}
After the change of variables \(a=x/t\), the far-tail part is
\[
  t^{-2}\int_1^\infty S_a(y)a^{-3}\,\dd a .
\]
With \(A=1+y^2+a^2\),
\[
S_a(y)
=\frac{2a^2(3y^2-1-a^2)}
       {\{1+(y-a)^2\}\{1+(y+a)^2\}} .
\]
Consequently
\[
 |S_a(y)|a^{-3}
 \le \frac{C}{a\{1+(a-|y|)^2\}}.
\]
Splitting \(a\) into the four ranges
\([1,|y|/2]\), \([|y|/2,2|y|]\cap\{|a-|y||>1\}\),
\(\{|a-|y||\le1\}\), and \([2|y|,\infty)\) gives a uniform integrable bound
in \(y\) with respect to \(\omega\), proving
\eqref{eq:symmetric-p2-far-tail-quotient}.  On \(R\le x\le t\), Taylor
expansion of the paired expression cancels odd modes and gives
\[
S_{x/t}(y)=2u_2(y)(x/t)^2+O((x/t)^4)
\]
uniformly in \(y\).  Integrating \(x^{-3}\) over \([R,t]\) and adding the
far-tail estimate proves \eqref{eq:symmetric-p2-full-tail-split}.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{prop:n2-borderline-infinite-variance-layer}]
The moment assertions follow from
\[
  2c\int_{\mathrm e}^{\infty}x^{-2}\,\dd x<\infty,\qquad
  2c\int_{\mathrm e}^{\infty}x^{-1}\,\dd x=\infty,
\]
and
\[
  \EE[X^2\mathbf1_{\{|X|\le t\}}]
  =O(1)+2c\int_{\mathrm e}^{t}x^{-1}\,\dd x
  \sim2c\log t .
\]
The compact core contributes \(O(t^{-2})\) to the quotient.  Lemma
\ref{lem:symmetric-p2-far-tail-quotient}, with \(R=\mathrm e\), gives
\[
\Delta_t(y)=2c\,u_2(y)t^{-2}\log t+E_t(y),
\qquad
\|E_t\|_\infty+\|E_t\|_{L^1(\omega)}=O(t^{-2}).
\]
Since \(\|\Delta_t\|_\infty=O(t^{-2}\log t)\),
\[
D_{\mathrm{KL}}(P_t*\nu\|P_t)
=\frac12\int\Delta_t^2\,\dd\omega+O(t^{-6}(\log t)^3).
\]
Using \(J(2,2)=\int u_2^2\,\dd\omega=1/4\) gives
\[
\frac12(2c)^2J(2,2)t^{-4}(\log t)^2+O(t^{-4}\log t)
=\frac{c^2}{2}t^{-4}(\log t)^2+O(t^{-4}\log t).
\]
\end{proof}

\subsubsection*{Verified borderline-tail classes}\leavevmode\par\smallskip

\begin{definition}[Admissible borderline tail classes]%
\label{def:admissible-borderline-tail-classes}
Fix an even integer \(p\ge4\), \(R>0\), and \(c>0\).  A symmetric density
\[
  f(x)=g(x)+\{c|x|^{-p-1}L_f(|x|)+r_f(x)\}
       \mathbf1_{\{|x|\ge R\}},
\]
with \(g\ge0\) symmetric, compactly supported in \((-R,R)\), belongs to
\(\mathcal A_{p,R,c}\) if all moments of orders \(q<p\) are finite, if
\[
  M_p(t):=\int_{|x|\le t}x^pf(x)\,\dd x
\]
satisfies \(M_p(t)\to\infty\) and \(M_p(t)=o(t^{p-2})\), and if the four
quotient controls
\begin{align}
t^{-n}\int_{|x|>t}|x|^nf(x)\,\dd x
  &=o(t^{-p}M_p(t))\quad(0\le n<p), \label{eq:admissible-lower-tail-control}\\
t^{-p}\int_{|x|\le t}x^pf(x)\,\dd x
  &=t^{-p}M_p(t), \label{eq:admissible-top-mode-control}\\
\|\mathcal R_t^{(p)}\|_\infty+\|\mathcal R_t^{(p)}\|_{L^1(\omega)}
  &=o(t^{-p}M_p(t)), \label{eq:admissible-local-remainder-control}\\
\int_{|x|>t}\left(1+\frac{x^2}{t^2}\right)f(x)\,\dd x
  &=o(t^{-p}M_p(t)) \label{eq:admissible-far-tail-control}
\end{align}
hold, where
\[
\mathcal R_t^{(p)}(y):=
  \int_{|x|\le t}
  \left\{R_{x/t}(y)-\sum_{n=0}^{p}u_n(y)(x/t)^n\right\}f(x)\,\dd x .
\]
We write \(\mathcal A_{p,R,c}^{\rm ex}\), \(\mathcal A_{p,R,c}^{\rm pert}\),
and \(\mathcal A_{p,R,c}^{\rm sv}\) for the exact-power, perturbative-power,
and monotone slowly varying subclasses displayed in
Table~\ref{tab:missing-top-tail-verification}.
\end{definition}

The controls in Definition~\ref{def:admissible-borderline-tail-classes} give
the admissible quotient expansion
\begin{equation}\label{eq:admissible-borderline-quotient-expansion}
  \Delta_t(y)=
  \sum_{n=2}^{p-1}\mu_nu_n(y)t^{-n}
  +u_p(y)t^{-p}M_p(t)+e_t(y),
  \qquad
  \|e_t\|_\infty+\|e_t\|_{L^1(\omega)}
  =o(t^{-p}M_p(t)).
\end{equation}

\begin{corollary}[Admissible quotient verification consequence]%
\label{thm:admissible-borderline-tail-entropy}
Let \(N\ge3\), put \(p=2N-2\), and let
\(\nu\in\mathcal A_{p,R,c}\).  With
\[
  M_p(t)=\EE[X^p{\bf1}_{\{|X|\le t\}}],
  \qquad
  H(t)=D_{\mathrm{KL}}(P_t*\nu\|P_t),
\]
one has
\[
H(t)=
\sum_{m=2}^{N-1}A_{2m}(\nu)t^{-2m}
+\kappa_N\mu_2\,t^{-2N}M_p(t)
+o(t^{-2N}M_p(t)),
\]
where
\[
  \kappa_N=J(2,p)=(-1)^N(N-1)2^{-2N+2}.
\]
\end{corollary}

\begin{proof}
Definition~\ref{def:admissible-borderline-tail-classes} gives
\eqref{eq:admissible-borderline-quotient-expansion} and the growth
assumptions of \eqref{eq:conditional-truncated-moment-growth}.  Apply
Theorem~\ref{thm:tail-replacements-proved-here}.
\end{proof}

\begin{lemma}[Local quotient remainder estimate]
\label{lem:borderline-local-quotient-remainder-estimate}
Let \(p\ge1\).  There is a constant \(C_p<\infty\) such that, for every
locally integrable \(h\) and every \(t>0\),
\[
\mathcal E_{p,t,h}(y):=
  \int_{|x|\le t}
  \left\{R_{x/t}(y)-\sum_{n=0}^{p}u_n(y)(x/t)^n\right\}h(x)\,\dd x
\]
satisfies
\begin{equation}\label{eq:borderline-local-quotient-remainder-estimate}
\|\mathcal E_{p,t,h}\|_\infty+
\|\mathcal E_{p,t,h}\|_{L^1(\omega)}
\le
C_p t^{-(p+1)}\int_{|x|\le t}|x|^{p+1}|h(x)|\,\dd x .
\end{equation}
\end{lemma}

\begin{proof}
Lemma~\ref{lem:uniform-one-dimensional-cayley-taylor-remainder} gives
\[
R_{x/t}(y)-\sum_{n=0}^{p}u_n(y)(x/t)^n
=(x/t)^{p+1}v_p(y,x/t),
\]
where \(v_p\) is bounded uniformly in \(y\) and \(x/t\).  Taking absolute
values and integrating gives the \(L^\infty\) bound.  Since \(\omega\) is a
probability measure, the same pointwise bound gives the \(L^1(\omega)\) bound.
\end{proof}

\begin{lemma}[Exact-power borderline controls]
\label{lem:exact-power-borderline-tail-controls}
Let \(p\ge4\) be even and
\[
f(x)=g(x)+c|x|^{-p-1}\mathbf1_{\{|x|\ge R\}},
\]
where \(c>0\), \(R>0\), \(g\ge0\) is symmetric and compactly supported in
\((-R,R)\), and \(f\) is normalized.  Then \(f\in\mathcal A_{p,R,c}^{\rm ex}\).
More precisely,
\[
M_p(t)=2c\log t+O(1),
\]
and the quotient expansion in Table~\ref{tab:missing-top-tail-verification}
holds with an \(O(t^{-p})\) remainder in
\(L^\infty(\RR)\cap L^1(\omega)\).
\end{lemma}

\begin{proof}
It is enough to consider \(t>R\).  The compact part contributes only bounded
moments, while
\[
\int_{R\le |x|\le t}x^p c|x|^{-p-1}\,\dd x
=2c\int_R^t\frac{\dd x}{x}
=2c\log t+O(1).
\]
Hence \(M_p(t)=2c\log t+O(1)\), so \(M_p(t)\to\infty\) and
\(M_p(t)=o(t^{p-2})\).

For \(0\le n<p\),
\[
t^{-n}\int_{|x|>t}|x|^nf(x)\,\dd x
=2c\,t^{-n}\int_t^\infty x^{n-p-1}\,\dd x
=O(t^{-p})
=o(t^{-p}M_p(t)).
\]
Lemma~\ref{lem:borderline-local-quotient-remainder-estimate} gives
\[
\|\mathcal R_t^{(p)}\|_\infty+\|\mathcal R_t^{(p)}\|_{L^1(\omega)}
\le
C_pt^{-(p+1)}
\left\{\int_{|x|<R}|x|^{p+1}g(x)\,\dd x
+2c\int_R^t\dd x\right\}
=O(t^{-p}),
\]
which is \(o(t^{-p}M_p(t))\).  Finally,
\[
\int_{|x|>t}\left(1+\frac{x^2}{t^2}\right)f(x)\,\dd x
=2c\int_t^\infty\left(1+\frac{x^2}{t^2}\right)x^{-p-1}\,\dd x
=O(t^{-p})
=o(t^{-p}M_p(t)).
\]
Thus all controls in
Definition~\ref{def:admissible-borderline-tail-classes} hold.

The quotient expansion follows by expanding \(R_{x/t}\) on \(|x|\le t\).  The
lower modes \(2\le n<p\) may be completed from truncated to full moments at
cost \(O(t^{-p})\) by the first estimate above.  The top mode is
\[
u_p(y)t^{-p}\int_{|x|\le t}x^pf(x)\,\dd x
=2c\,u_p(y)t^{-p}\log t+O(t^{-p}),
\]
and the local remainder, omitted polynomial tail, and far quotient tail are
all \(O(t^{-p})\) in \(L^\infty\cap L^1(\omega)\).
\end{proof}

\begin{lemma}[Perturbative-power borderline controls]
\label{lem:perturbative-power-borderline-tail-controls}
Let \(p\ge4\) be even and
\[
f(x)=g(x)+\{c|x|^{-p-1}+r(x)\}\mathbf1_{\{|x|\ge R\}},
\]
where \(c>0\), \(R>0\), \(g\ge0\) is symmetric and compactly supported in
\((-R,R)\), \(r\) is even, and \(f\) is nonnegative and normalized.  Assume
\begin{align*}
\int_{R\le |x|\le t}|x|^pr(x)\,\dd x&=O(1),\\
t^{-n}\int_{|x|>t}|x|^n|r(x)|\,\dd x&=O(t^{-p})
      \qquad(0\le n<p),\\
t^{-(p+1)}\int_{R\le |x|\le t}|x|^{p+1}|r(x)|\,\dd x&=O(t^{-p}),\\
\int_{|x|>t}\left(1+\frac{x^2}{t^2}\right)|r(x)|\,\dd x&=O(t^{-p}).
\end{align*}
Then \(f\in\mathcal A_{p,R,c}^{\rm pert}\).  Moreover,
\[
M_p(t)=2c\log t+O(1),
\]
and the quotient expansion in Table~\ref{tab:missing-top-tail-verification}
holds with \(2c\log t\) equivalently replaced by \(M_p(t)\).
The pointwise bound \(|r(x)|\le C|x|^{-p-1-\eta}\), \(\eta>0\), implies the
four displayed assumptions.
\end{lemma}

\begin{proof}
The exact-power part is handled by
Lemma~\ref{lem:exact-power-borderline-tail-controls}.  It remains to show
that the perturbation contributes only \(O(t^{-p})\) to each admissibility
control.  The first assumption gives
\[
M_p(t)=2c\log t+O(1).
\]
The second assumption is exactly the \(O(t^{-p})\) perturbative contribution
to the lower-tail controls.  The third assumption and
Lemma~\ref{lem:borderline-local-quotient-remainder-estimate} give an
\(O(t^{-p})\) perturbative local Taylor remainder.  The fourth assumption is
the \(O(t^{-p})\) perturbative far-tail control.  Since
\(M_p(t)\sim2c\log t\), each of these \(O(t^{-p})\) terms is
\(o(t^{-p}M_p(t))\), and the admissible controls follow.

The quotient expansion is obtained from the same split as in the exact-power
case.  The \(r\)-contribution to the top mode is \(O(t^{-p})\), so
\[
u_p(y)t^{-p}M_p(t)
=2c\,u_p(y)t^{-p}\log t+O(t^{-p}),
\]
while all lower-mode completions and quotient remainders caused by \(r\) are
\(O(t^{-p})\) in \(L^\infty\cap L^1(\omega)\).

If \(|r(x)|\le C|x|^{-p-1-\eta}\), then
\[
\int_{R\le |x|\le t}|x|^p|r(x)|\,\dd x=O(1),
\]
\[
t^{-n}\int_{|x|>t}|x|^n|r(x)|\,\dd x
\le C t^{-n}\int_t^\infty x^{n-p-1-\eta}\,\dd x
=O(t^{-p-\eta}),
\]
and
\[
t^{-(p+1)}\int_{R\le |x|\le t}|x|^{p+1}|r(x)|\,\dd x=O(t^{-p}),
\qquad
\int_{|x|>t}\left(1+\frac{x^2}{t^2}\right)|r(x)|\,\dd x
=O(t^{-p-\eta}).
\]
Thus the pointwise condition implies the four assumptions.
\end{proof}

\begin{lemma}[Slowly varying integral controls]
\label{lem:slowly-varying-integral-controls}
Let \(p\ge4\) be an integer.  Let \(L_0\) be eventually positive, eventually
monotone, slowly varying, and let
\[
  M_0(t)=\int_R^t\frac{L_0(s)}{s}\,\dd s\to\infty .
\]
Then
\[
  L_0(t)=o(M_0(t)).
\]
For every integer \(0\le n<p\),
\begin{equation}\label{eq:slowly-varying-polynomial-tail-control}
  \int_t^\infty x^{n-p-1}L_0(x)\,\dd x
  =O(t^{n-p}L_0(t))
  =o(t^{n-p}M_0(t)).
\end{equation}
Moreover
\begin{equation}\label{eq:slowly-varying-local-integral-control}
  t^{-(p+1)}\int_R^tL_0(x)\,\dd x
  =O(t^{-p}L_0(t))
  =o(t^{-p}M_0(t)).
\end{equation}
Consequently, for the symmetric density tail
\[
  c|x|^{-p-1}L_0(|x|)\mathbf1_{\{|x|\ge R\}},
\]
the omitted polynomial tails are \(o(t^{-p}M_0(t))\), the local Taylor
remainder in
Lemma~\ref{lem:borderline-local-quotient-remainder-estimate} is
\(o(t^{-p}M_0(t))\), and the far-tail quotient contribution is
\(o(t^{-p}M_0(t))\) in \(L^\infty(\RR)\cap L^1(\omega)\).
\end{lemma}

\begin{proof}
Increasing \(R\) changes \(M_0\) by a bounded amount, so we may suppose that
\(L_0\) is positive and monotone on \([R,\infty)\).

We first prove \(L_0(t)=o(M_0(t))\).  By the uniform convergence theorem for
slowly varying functions, for each fixed \(A>1\),
\[
  \int_{t/A}^{t}\frac{L_0(s)}s\,\dd s
  =L_0(t)\log A+o(L_0(t)).
\]
This integral is bounded by \(M_0(t)\) up to a fixed additive constant.
Hence
\[
  \limsup_{t\to\infty}\frac{L_0(t)}{M_0(t)}\le \frac1{\log A}.
\]
Letting \(A\to\infty\) gives \(L_0(t)=o(M_0(t))\).

For \(0\le n<p\), the function \(x^{n-p-1}L_0(x)\) is regularly varying
with index \(n-p-1<-1\).  Karamata's tail theorem gives
\[
  \int_t^\infty x^{n-p-1}L_0(x)\,\dd x
  \sim \frac{t^{n-p}L_0(t)}{p-n},
\]
which proves \eqref{eq:slowly-varying-polynomial-tail-control} after using
\(L_0(t)=o(M_0(t))\).  Karamata's theorem at index \(0>-1\) also gives
\[
  \int_R^tL_0(x)\,\dd x=O(tL_0(t)),
\]
and hence \eqref{eq:slowly-varying-local-integral-control}.
For completeness, these two Karamata applications may be replaced by the
standard dyadic estimates that follow from monotonicity and Potter bounds:
for any \(\varepsilon\in(0,\min\{p-n,1\})\),
\[
\int_{2^jt}^{2^{j+1}t}x^{n-p-1}L_0(x)\,\dd x
\le C_\varepsilon
t^{n-p}L_0(t)\,2^{-j(p-n-\varepsilon)},
\]
and summing over \(j\ge0\) gives the tail bound.  Similarly,
\[
\int_R^tL_0(x)\,\dd x
\le C_\varepsilon L_0(t)t^\varepsilon
\int_R^t x^{-\varepsilon}\,\dd x
\le C'_\varepsilon tL_0(t),
\]
where Potter's bound is used uniformly for \(R\le x\le t\), after increasing
\(R\) if necessary.

The three quotient estimates are now direct.  For \(0\le n<p\),
\[
  t^{-n}\int_{|x|>t}|x|^n c|x|^{-p-1}L_0(|x|)\,\dd x
  =2c\,t^{-n}\int_t^\infty x^{n-p-1}L_0(x)\,\dd x
  =o(t^{-p}M_0(t)).
\]
Lemma~\ref{lem:borderline-local-quotient-remainder-estimate} and
\eqref{eq:slowly-varying-local-integral-control} give the local Taylor
remainder bound.  Finally, since \(p\ge4\),
\begin{align*}
  \int_{|x|>t}\left(1+\frac{x^2}{t^2}\right)
  c|x|^{-p-1}L_0(|x|)\,\dd x
  &\le C\int_t^\infty x^{-p-1}L_0(x)\,\dd x \\
  &\quad
  +Ct^{-2}\int_t^\infty x^{1-p}L_0(x)\,\dd x
  =O(t^{-p}L_0(t)),
\end{align*}
where the last estimate is another application of Karamata's tail theorem.
This is \(o(t^{-p}M_0(t))\).
\end{proof}

\begin{lemma}[Monotone slowly varying borderline controls]
\label{lem:slowly-varying-borderline-tail-controls}
Let \(p\ge4\) be even and
\[
f(x)=g(x)+c|x|^{-p-1}L(|x|)\mathbf1_{\{|x|\ge R\}},
\]
where \(c>0\), \(R>0\), \(g\ge0\) is symmetric and compactly supported in
\((-R,R)\), and \(f\) is normalized.  Assume that \(L\) is eventually
positive, eventually monotone, slowly varying, and
\[
\mathcal M(t):=\int_R^t\frac{L(s)}{s}\,\dd s\to\infty .
\]
Then \(f\in\mathcal A_{p,R,c}^{\rm sv}\), with
\[
M_p(t)=2c\mathcal M(t)+O(1),
\]
and the slowly varying quotient expansion in
Table~\ref{tab:missing-top-tail-verification} holds.
\end{lemma}

\begin{proof}
Increasing \(R\) changes only the compactly supported part, so we may suppose
that \(L\) is positive and monotone on \([R,\infty)\).  First,
\[
M_p(t)=O(1)+2c\int_R^t\frac{L(s)}{s}\,\dd s
=2c\mathcal M(t)+O(1).
\]
The growth \(M_p(t)=o(t^{p-2})\) follows from Potter's bound: for any
\(\alpha>0\), \(L(s)\le C_\alpha s^\alpha\) for large \(s\), hence
\(\mathcal M(t)=O(t^\alpha)\); take \(\alpha<p-2\).

Lemma~\ref{lem:slowly-varying-integral-controls}, applied with
\(L_0=L\) and \(M_0=\mathcal M\), gives the polynomial tail bounds, the
local Taylor remainder, and the far-tail quotient bound.  The compact part
contributes only \(O(t^{-p})\) to these three estimates, which is
\(o(t^{-p}M_p(t))\) because \(\mathcal M(t)\to\infty\).

These estimates verify all controls in
Definition~\ref{def:admissible-borderline-tail-classes}.  Expanding
\(R_{x/t}\) on \(|x|\le t\), completing lower modes with the lower-tail
estimate, and absorbing the local and far-tail terms gives
\[
\Delta_t(y)=\sum_{n=2}^{p-1}\mu_nu_n(y)t^{-n}
+u_p(y)t^{-p}M_p(t)
+o_{L^\infty\cap L^1(\omega)}(t^{-p}M_p(t)).
\]
\end{proof}

\begin{proposition}[Optional verification of the three subclasses]%
\label{prop:log-tail-counterexample-top-moment}
\label{lem:pointwise-perturbative-power-tail-sufficient-condition}
\label{lem:borderline-exact-power-tail-quotient-expansion}
\label{thm:exact-perturbative-power-tail-log-replacement}
\label{cor:perturbative-power-tail-log-replacement}
\label{lem:perturbative-power-tail-quotient-expansion}
\label{lem:slowly-varying-quotient-controls}
\label{cor:conditional-slowly-varying-borderline-tail}
Let \(N\ge3\), put \(p=2N-2\), and let \(\nu\) have a symmetric normalized
density belonging to one of the three subclasses in
Table~\ref{tab:missing-top-tail-verification}.  Then
\(\nu\in\mathcal A_{p,R,c}\), the quotient expansion
\eqref{eq:admissible-borderline-quotient-expansion} holds, and the entropy
asymptotic of
Corollary~\ref{thm:admissible-borderline-tail-entropy} follows.  In
particular, for the exact-power and perturbative-power rows,
\[
D_{\mathrm{KL}}(P_t*\nu\|P_t)
=\sum_{m=2}^{N-1}A_{2m}(\nu)t^{-2m}
+2c\,\kappa_N\mu_2\,t^{-2N}\log t
+o(t^{-2N}\log t),
\]
where \(\kappa_N=(-1)^N(N-1)2^{-2N+2}\).  For the monotone slowly varying
row,
\begin{equation}\label{eq:slowly-varying-entropy-expansion}
D_{\mathrm{KL}}(P_t*\nu\|P_t)
=\sum_{m=2}^{N-1}A_{2m}(\nu)t^{-2m}
+\kappa_N\mu_2\,t^{-2N}M_p(t)
+o(t^{-2N}M_p(t)).
\end{equation}
\end{proposition}

\begin{proof}
Lemma~\ref{lem:exact-power-borderline-tail-controls},
Lemma~\ref{lem:perturbative-power-borderline-tail-controls}, and
Lemma~\ref{lem:slowly-varying-borderline-tail-controls} verify the four
controls in Definition~\ref{def:admissible-borderline-tail-classes} for the
three rows of Table~\ref{tab:missing-top-tail-verification}.  They also prove
the corresponding \(L^\infty\cap L^1(\omega)\) quotient expansions displayed
in the output column.  Hence
\eqref{eq:admissible-borderline-quotient-expansion} holds in each subclass,
and Corollary~\ref{thm:admissible-borderline-tail-entropy} gives the stated
entropy expansions.
\end{proof}
