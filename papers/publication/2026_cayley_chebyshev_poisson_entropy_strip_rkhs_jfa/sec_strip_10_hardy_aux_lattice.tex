\subsection{Hardy splitting and the auxiliary lattice projection}

\begin{proposition}[Classical Hardy splitting of the Poisson strip model]%
\label{prop:poisson-hardy-splitting}
Let
\[
\mathcal H_K^+
:=
\left\{
f\in\mathcal H_K:\ \widehat f(\xi)=0 \text{ for a.e. }\xi<0
\right\},
\]
\[
\mathcal H_K^-
:=
\left\{
f\in\mathcal H_K:\ \widehat f(\xi)=0 \text{ for a.e. }\xi>0
\right\}.
\]
Then the classical positive/negative Hardy decomposition transported through
the Poisson image model gives
\[
\mathcal H_K=\mathcal H_K^+\oplus \mathcal H_K^-
\]
orthogonally.  Moreover, each $f_\pm\in\mathcal H_K^\pm$ extends
holomorphically to $\mathbb S$, and $\mathcal H_K^\pm$ is a
reproducing-kernel Hilbert space on~$\mathbb S$ with kernel
\begin{equation}\label{eq:hardy-splitting-kernels}
K_+(z,w)=\frac{1}{2(2-\ii(z-\overline w))},
\qquad
K_-(z,w)=\frac{1}{2(2+\ii(z-\overline w))}.
\end{equation}
For $a,b\in\RR$ one has
\begin{equation}\label{eq:hardy-splitting-real-kernel}
K(a,b)=K_+(a,b)+K_-(a,b).
\end{equation}
Thus the proposition is a background identification of the two Hardy
half-strip components of the already constructed Poisson image space; the
Poisson-specific data retained for the lattice theorem are the normalization
$K(a,b)=\pi P_2(a-b)$ and the explicit kernels in
\eqref{eq:hardy-splitting-kernels}.
\end{proposition}

\begin{proof}
This is the standard Paley--Wiener/Hardy support splitting for a strip,
written in the normalization of Proposition~\ref{prop:gram-space-poisson-image};
see the classical Hardy-space references in
Remark~\ref{rem:poisson-image-framework}.
The norm formula from Proposition~\ref{prop:gram-space-poisson-image}
shows that $\mathcal H_K^+$ and $\mathcal H_K^-$ are closed subspaces of
$\mathcal H_K$, and they are orthogonal because their Fourier supports are
disjoint.  The decomposition
\[
\widehat f
=\widehat f\,\mathbf 1_{[0,\infty)}
+\widehat f\,\mathbf 1_{(-\infty,0]}
\]
therefore yields the asserted orthogonal sum.

For $w\in\mathbb S$, define $K_\pm(\cdot,w)$ by
\[
\widehat{K_+(\cdot,w)}(\xi)
:=
\pi e^{-2\xi}e^{-\ii\overline w\,\xi}\mathbf 1_{[0,\infty)}(\xi),
\]
\[
\widehat{K_-(\cdot,w)}(\xi)
:=
\pi e^{2\xi}e^{-\ii\overline w\,\xi}\mathbf 1_{(-\infty,0]}(\xi).
\]
Since $|\Im w|<1$, these functions belong to $\mathcal H_K^\pm$.  If
$f_+\in\mathcal H_K^+$, then
\begin{align*}
\langle f_+,K_+(\cdot,w)\rangle_{\mathcal H_K}
&=
\frac{1}{2\pi^2}\int_0^\infty
e^{2\xi}\widehat f_+(\xi)\,
\overline{\pi e^{-2\xi}e^{-\ii\overline w\,\xi}}\dd\xi \\
&=
\frac{1}{2\pi}\int_0^\infty \widehat f_+(\xi)e^{\ii w\xi}\dd\xi
=f_+(w).
\end{align*}
The same argument gives the reproducing identity for $\mathcal H_K^-$.  By
Fourier inversion,
\[
K_+(z,w)
=\frac{1}{2\pi}\int_0^\infty
\pi e^{-2\xi}e^{\ii(z-\overline w)\xi}\dd\xi
=\frac{1}{2(2-\ii(z-\overline w))},
\]
\[
K_-(z,w)
=\frac{1}{2\pi}\int_{-\infty}^0
\pi e^{2\xi}e^{\ii(z-\overline w)\xi}\dd\xi
=\frac{1}{2(2+\ii(z-\overline w))},
\]
which proves \eqref{eq:hardy-splitting-kernels}.  Adding the two kernels
on the real axis yields \eqref{eq:hardy-splitting-real-kernel}.
\end{proof}

We next record the only auxiliary scaled-lattice calculation used later.  Its
purpose is not to introduce a separate sampling theory, but to derive the
closed Poisson weight and orthogonal projection formula that will be specialized
immediately to the integer orbit; the reconstruction result used in the paper is
the integer specialization in
Corollary~\ref{cor:integer-orbit-projection}, the quotient
Proposition~\ref{prop:integer-orbit-scoped-quotient}, and the Poisson
observation theorem which follows them.  The general principal
shift-invariant-space mechanism is the background framework
\cite{deBoorDeVoreRon1994,RonShen1995,AldroubiGrochenig2001}; in the present
model both the periodized weight and the projector can be written in closed
form.

\begin{lemma}[Auxiliary \(h\ZZ\) calculation for the integer orbit]%
\label{lem:auxiliary-uniform-lattice-projection}
Let $h>0$.  This auxiliary statement is used in the sequel only through its
integer specialization $h=1$, which gives the projection and quotient package
for the Poisson observation channel.  Set
\[
I_h:=\Bigl[-\frac{\pi}{h},\frac{\pi}{h}\Bigr],
\qquad
\Sigma_h(\xi):=\sum_{m\in\ZZ}e^{-2\abs{\xi+2\pi m/h}}
\qquad(\xi\in I_h).
\]
Then
\begin{equation}\label{eq:lattice-periodized-weight}
\Sigma_h(\xi)
=\frac{\cosh(2(\pi/h-\abs{\xi}))}{\sinh(2\pi/h)}
\qquad(\xi\in I_h).
\end{equation}
Define
\[
\mathcal H_K(h\ZZ)
:=
\overline{\operatorname{span}}\{K(\cdot,hn):n\in\ZZ\}
\subset\mathcal H_K
\]
and
\[
\mathcal N_h
:=
\{f\in\mathcal H_K:\ f(hn)=0\ \text{for every }n\in\ZZ\}.
\]
For $f\in\mathcal H_K$ write
\[
q_f(\xi):=e^{2\abs{\xi}}\widehat f(\xi).
\]
Then the following hold.
\begin{enumerate}[label=(\roman*)]
\item
      One has $f\in\mathcal H_K(h\ZZ)$ if and only if $q_f$ is
      $2\pi/h$-periodic and
      \[
      \int_{I_h}\Sigma_h(\xi)\abs{q_f(\xi)}^2\dd\xi<\infty.
      \]
      In that case,
      \begin{equation}\label{eq:lattice-subspace-norm}
      \|f\|_{\mathcal H_K}^2
      =\frac{1}{2\pi^2}\int_{I_h}\Sigma_h(\xi)\abs{q_f(\xi)}^2\dd\xi.
      \end{equation}
\item
      The orthogonal complement of~$\mathcal H_K(h\ZZ)$ is exactly
      $\mathcal N_h$, and
      \begin{equation}\label{eq:lattice-alias-cancellation}
      f\in\mathcal N_h
      \iff
      \sum_{m\in\ZZ}\widehat f\!\left(\xi+\frac{2\pi m}{h}\right)=0
      \quad\text{for a.e. }\xi\in I_h.
      \end{equation}
      Consequently,
      \[
      \mathcal H_K=\mathcal H_K(h\ZZ)\oplus \mathcal N_h
      \]
      orthogonally.
\item
      Let $\Pi_h:\mathcal H_K\to\mathcal H_K(h\ZZ)$ denote the orthogonal
      projection.  Then, for almost every $\xi\in I_h$, the series
      \[
      \sum_{m\in\ZZ}\widehat f\!\left(\xi+\frac{2\pi m}{h}\right)
      \]
      converges absolutely and
      \begin{equation}\label{eq:lattice-alias-projection}
      q_{\Pi_hf}(\xi)
      =\frac{\displaystyle\sum_{m\in\ZZ}
      \widehat f\!\left(\xi+\frac{2\pi m}{h}\right)}
      {\Sigma_h(\xi)},
      \end{equation}
      with $q_{\Pi_hf}$ extended $2\pi/h$-periodically to~$\RR$.
      Equivalently,
      \[
      \widehat{\Pi_hf}(\xi)=e^{-2\abs{\xi}}q_{\Pi_hf}(\xi)
      \qquad\text{for a.e. }\xi\in\RR.
      \]
      Moreover,
      \begin{equation}\label{eq:lattice-projection-norm}
      \|\Pi_hf\|_{\mathcal H_K}^2
      =\frac{1}{2\pi^2}\int_{I_h}
      \frac{\left|\sum_{m\in\ZZ}
      \widehat f\!\left(\xi+\frac{2\pi m}{h}\right)\right|^2}
      {\Sigma_h(\xi)}\,\dd\xi,
      \end{equation}
      and $\Pi_hf$ is the unique element of~$\mathcal H_K(h\ZZ)$
      satisfying
      \[
      (\Pi_hf)(hn)=f(hn)
      \qquad(n\in\ZZ).
      \]
\item
      The $h\ZZ$ samples determine exactly the projection $\Pi_hf$ and no
      finer ambient datum: for $f,g\in\mathcal H_K$,
      \begin{equation}\label{eq:uniform-lattice-quotient-rigidity}
      f(hn)=g(hn)\quad(n\in\ZZ)
      \qquad\Longleftrightarrow\qquad
      \Pi_hf=\Pi_hg .
      \end{equation}
      Equivalently,
      \[
      \mathcal H_K/\mathcal N_h\longrightarrow\mathcal H_K(h\ZZ),
      \qquad
      f+\mathcal N_h\longmapsto\Pi_hf,
      \]
      is a well-defined isometric Hilbert-space isomorphism.  Thus full
      reconstruction from the $h\ZZ$ orbit holds for a given
      $f\in\mathcal H_K$ if and only if $f\in\mathcal H_K(h\ZZ)$.
\item
      At the only specialization used below, namely $h=1$, put
      $\Pi_{\ZZ}:=\Pi_1$.  Any reconstruction rule
      \[
      \mathcal R:\mathcal H_K\longrightarrow \mathcal H_K(\ZZ)
      \]
      whose value depends only on the integer samples and which is exact on
      $\mathcal H_K(\ZZ)$ must equal the integer projection:
      \[
      \mathcal Rf=\Pi_{\ZZ}f\qquad(f\in\mathcal H_K).
      \]
      Thus the auxiliary $h\ZZ$ fiberization contributes to the paper only
      through the integer projection and its quotient; the Poisson-specific
      main-chain inputs are the integer symbol in
      Proposition~\ref{prop:poisson-lattice-symbol} and the observation-channel
      theorem, Theorem~\ref{thm:poisson-observation-lattice-sampling}.
\item
      The observation-channel consequence used below is exactly the
      integer-orbit projection obtained from the specialization $h=1$; no
      shifted-lattice, noninteger-lattice, or scaled-channel reconstruction
      theorem is asserted from the auxiliary parameter~$h$.
\end{enumerate}
\end{lemma}

Only the specialization $h=1$ is used in the Poisson observation-channel
theorems below.  The parameter $h$ is retained in
Lemma~\ref{lem:auxiliary-uniform-lattice-projection} as a compact derivation of
the unshifted integer-orbit projection theorem.  The reconstruction result
used in the sequel is the integer-orbit projection and quotient theorem,
sharpened by the Poisson-specific symbol and observation-channel results below.

\begin{proof}
Let
\[
a:=\frac{2\pi}{h}.
\]
For $\xi\in I_h$ one has $\abs{\xi}\le a/2$, so
\begin{align*}
\Sigma_h(\xi)
&=e^{-2\abs{\xi}}
+\sum_{m=1}^{\infty}e^{-2(\abs{\xi}+ma)}
+\sum_{m=1}^{\infty}e^{-2(ma-\abs{\xi})} \\
&=\frac{e^{-2\abs{\xi}}+e^{-2a}e^{2\abs{\xi}}}{1-e^{-2a}}
=\frac{\cosh(a-2\abs{\xi})}{\sinh a},
\end{align*}
which is \eqref{eq:lattice-periodized-weight}.

If
\[
F=\sum_{n=-N}^{N}c_nK(\cdot,hn),
\]
then
\[
\widehat F(\xi)=\pi e^{-2\abs{\xi}}\sum_{n=-N}^{N}c_ne^{-\ii hn\xi},
\]
hence
\[
q_F(\xi)=\pi\sum_{n=-N}^{N}c_ne^{-\ii hn\xi},
\]
which is $a$-periodic.  Using the norm formula from
Proposition~\ref{prop:gram-space-poisson-image} and partitioning~$\RR$
into the translates $I_h+ma$ gives
\[
\|F\|_{\mathcal H_K}^2
=\frac{1}{2\pi^2}\int_{I_h}
\sum_{m\in\ZZ}e^{-2\abs{\xi+ma}}\abs{q_F(\xi)}^2\dd\xi
=\frac{1}{2\pi^2}\int_{I_h}\Sigma_h(\xi)\abs{q_F(\xi)}^2\dd\xi.
\]
Passing to the closure proves one direction of~(i).

Conversely, let $q$ be $a$-periodic with
\[
\int_{I_h}\Sigma_h(\xi)\abs{q(\xi)}^2\dd\xi<\infty.
\]
Since $\Sigma_h$ is continuous and strictly positive on the compact
interval~$I_h$, the spaces
$L^2(I_h,\Sigma_h(\xi)\dd\xi)$ and $L^2(I_h,\dd\xi)$ coincide with
equivalent norms.  The trigonometric polynomials
\[
\sum_{n=-N}^{N}d_ne^{-\ii hn\xi}
\]
are therefore dense in $L^2(I_h,\Sigma_h(\xi)\dd\xi)$.  Choose such a
sequence $p_N$ with $p_N\to \pi^{-1}q$ in that weighted norm, and define
$F_N\in\mathcal H_K(h\ZZ)$ by
\[
F_N:=\sum_{n=-N}^{N}d_nK(\cdot,hn).
\]
Then $q_{F_N}=\pi p_N$, so the preceding norm identity gives
\[
\|F_N-F_M\|_{\mathcal H_K}^2
=\frac{1}{2\pi^2}\int_{I_h}\Sigma_h(\xi)
\abs{q_{F_N}(\xi)-q_{F_M}(\xi)}^2\dd\xi.
\]
Hence $(F_N)$ is Cauchy in~$\mathcal H_K$ and converges to some
$F\in\mathcal H_K(h\ZZ)$ with $q_F=q$.  This proves~(i).

For every $f\in\mathcal H_K$ and almost every $\xi\in I_h$,
Cauchy--Schwarz gives
\[
\sum_{m\in\ZZ}\abs{\widehat f(\xi+ma)}
\le
\Sigma_h(\xi)^{1/2}
\left(
\sum_{m\in\ZZ}e^{2\abs{\xi+ma}}
\abs{\widehat f(\xi+ma)}^2
\right)^{1/2},
\]
so the alias sum is absolutely convergent for almost every~$\xi$.

If $g\in\mathcal H_K(h\ZZ)$, then by~(i) its profile $q_g$ is
$a$-periodic.  Using the Fourier norm formula again,
\begin{align*}
\langle f,g\rangle_{\mathcal H_K}
&=
\frac{1}{2\pi^2}\int_{I_h}
\sum_{m\in\ZZ}e^{-2\abs{\xi+ma}}q_f(\xi+ma)\,
\overline{q_g(\xi)}\dd\xi \\
&=
\frac{1}{2\pi^2}\int_{I_h}
\left(
\sum_{m\in\ZZ}\widehat f(\xi+ma)
\right)\overline{q_g(\xi)}\dd\xi.
\end{align*}
Since the periodic profiles $q_g$ form a dense subspace of
$L^2(I_h,\Sigma_h(\xi)\dd\xi)$ and $\Sigma_h$ is bounded above and below
on~$I_h$, the same set is dense in $L^2(I_h,\dd\xi)$.  Therefore
$f\perp\mathcal H_K(h\ZZ)$ if and only if the alias sum in
\eqref{eq:lattice-alias-cancellation} vanishes almost everywhere on~$I_h$.

By the reproducing property,
\[
f(hn)=\langle f,K(\cdot,hn)\rangle_{\mathcal H_K}
\qquad(n\in\ZZ),
\]
and the closed linear span of the kernel sections $K(\cdot,hn)$ is
exactly $\mathcal H_K(h\ZZ)$.  Consequently,
\[
f\in\mathcal N_h
\iff
f\perp\mathcal H_K(h\ZZ).
\]
This proves~(ii).

Let $g\in\mathcal H_K(h\ZZ)$ and write $q_g$ for its periodic profile.
Then
\[
\|f-g\|_{\mathcal H_K}^2
=\frac{1}{2\pi^2}\int_{I_h}
\sum_{m\in\ZZ}e^{-2\abs{\xi+ma}}
\abs{q_f(\xi+ma)-q_g(\xi)}^2\dd\xi.
\]
For each $\xi\in I_h$, the integrand is minimized when $q_g(\xi)$ is the
weighted average of the fiber values $q_f(\xi+ma)$ with weights
$e^{-2\abs{\xi+ma}}$.  Since
\[
e^{-2\abs{\xi+ma}}q_f(\xi+ma)=\widehat f(\xi+ma),
\]
the minimizing profile is exactly the one in
\eqref{eq:lattice-alias-projection}.  This proves the projection formula.
Substituting the minimizing value into the norm identity gives
\eqref{eq:lattice-projection-norm}.  Finally,
\[
f-\Pi_hf\in\mathcal N_h
\]
by~(ii), so $(\Pi_hf)(hn)=f(hn)$ for every $n\in\ZZ$.  If
$G\in\mathcal H_K(h\ZZ)$ satisfies $G(hn)=f(hn)$ for all $n$, then
$G-\Pi_hf\in\mathcal H_K(h\ZZ)\cap\mathcal N_h=\{0\}$, hence
$G=\Pi_hf$.  This proves~(iii).

It remains to identify the quotient determined by the samples.  The forward
implication in \eqref{eq:uniform-lattice-quotient-rigidity} follows because
equal $h\ZZ$ samples give $f-g\in\mathcal N_h$, and by~(ii) this is the
orthogonal complement of $\mathcal H_K(h\ZZ)$, so the two orthogonal
projections coincide.  The reverse implication follows from~(iii), since
$\Pi_hf$ and $\Pi_hg$ interpolate respectively the samples of $f$ and~$g$.
The displayed quotient map is well defined because adding an element of
$\mathcal N_h=\mathcal H_K(h\ZZ)^\perp$ does not change the projection.  It is
surjective by applying it to $F+\mathcal N_h$ with $F\in\mathcal H_K(h\ZZ)$,
injective by \eqref{eq:uniform-lattice-quotient-rigidity}, and isometric by
the orthogonal decomposition
\[
\|f+\mathcal N_h\|_{\mathcal H_K/\mathcal N_h}
=\inf_{u\in\mathcal N_h}\|\Pi_hf+(f-\Pi_hf)+u\|_{\mathcal H_K}
=\|\Pi_hf\|_{\mathcal H_K}.
\]
  Finally, full reconstruction means $f$ equals the unique lattice-subspace
  interpolant $\Pi_hf$, and this is equivalent to
  $f\in\mathcal H_K(h\ZZ)$.  This proves~(iv).

For~(v), set $h=1$ in~(iv).  If $\mathcal R$ depends only on the integer
samples, then $f$ and $\Pi_{\ZZ}f$ have the same samples by~(iii), so
\[
\mathcal Rf=\mathcal R(\Pi_{\ZZ}f).
\]
Since $\Pi_{\ZZ}f\in\mathcal H_K(\ZZ)$ and $\mathcal R$ is exact on that
subspace, the right-hand side is $\Pi_{\ZZ}f$.  Hence
$\mathcal R=\Pi_{\ZZ}$.  The final sentence records precisely this dependence
of the later integer-orbit theorems on the present auxiliary calculation.
This proves~(v).

For~(vi), the assertions above have been stated and proved for the closed
unshifted family $h\ZZ$ only, and the later Poisson channel uses only $h=1$.
\end{proof}

\begin{corollary}[Integer-orbit projection for the Poisson symbol]%
\label{cor:integer-orbit-projection}
For the integer lattice $h=1$, let
\[
\Pi_{\ZZ}:=\Pi_1,
\qquad
\mathcal N_{\ZZ}:=\{f\in\mathcal H_K:\ f(n)=0\text{ for every }n\in\ZZ\}.
\]
Then
\[
\mathcal H_K=\mathcal H_K(\ZZ)\oplus\mathcal N_{\ZZ}
\]
orthogonally, and $\Pi_{\ZZ}f$ is the unique element of
$\mathcal H_K(\ZZ)$ satisfying
\[
(\Pi_{\ZZ}f)(n)=f(n)\qquad(n\in\ZZ).
\]
If $q_f(\xi):=e^{2|\xi|}\widehat f(\xi)$, then for almost every
$\theta\in[-\pi,\pi]$,
\begin{equation}\label{eq:integer-orbit-projection-symbol}
q_{\Pi_{\ZZ}f}(\theta)
=
\frac{\displaystyle\sum_{m\in\ZZ}\widehat f(\theta+2\pi m)}
{\displaystyle\sum_{m\in\ZZ}e^{-2|\theta+2\pi m|}}
=
\frac{\pi\displaystyle\sum_{m\in\ZZ}\widehat f(\theta+2\pi m)}
{\sigma(\theta)},
\end{equation}
extended $2\pi$-periodically, where the Poisson symbol is exactly
\[
\sigma(\theta)=\frac{\pi\cosh(2(\pi-|\theta|))}{\sinh(2\pi)}
\qquad(-\pi\le \theta\le \pi).
\]
Thus the only lattice specialization used in
Theorem~\ref{thm:poisson-observation-lattice-sampling} is the integer-orbit
projection with this explicit Poisson symbol.
Moreover, for every $f\in\mathcal H_K$, the integer sample sequence
$(f(n))_{n\in\ZZ}$ determines exactly the projection $\Pi_{\ZZ}f$: it is
the unique element of $\mathcal H_K(\ZZ)$ with those integer samples, and
\[
f-\Pi_{\ZZ}f\in\mathcal N_{\ZZ}.
\]
Consequently the integer orbit determines the full function $f$ if and only
if $f\in\mathcal H_K(\ZZ)$.
\end{corollary}

\begin{proof}
The first two assertions are Lemma~\ref{lem:auxiliary-uniform-lattice-projection}
with $h=1$.  In the notation of that proposition,
\[
\Sigma_1(\theta)=\sum_{m\in\ZZ}e^{-2|\theta+2\pi m|}
\qquad(-\pi\le\theta\le\pi).
\]
Multiplying the closed form \eqref{eq:lattice-periodized-weight} for
$\Sigma_1$ by~$\pi$ gives the stated Poisson symbol
\[
\sigma(\theta):=\pi\Sigma_1(\theta)
=\frac{\pi\cosh(2(\pi-|\theta|))}{\sinh(2\pi)}.
\]
Substituting $h=1$ in the projection formula
\eqref{eq:lattice-alias-projection} and replacing $\Sigma_1$ by
$\sigma/\pi$ gives \eqref{eq:integer-orbit-projection-symbol}.  The final
sentence records the dependence of
Theorem~\ref{thm:poisson-observation-lattice-sampling} on this integer
case.  The uniqueness assertion and the residual inclusion are the last
part of Lemma~\ref{lem:auxiliary-uniform-lattice-projection} with $h=1$.  If
$f\in\mathcal H_K(\ZZ)$, then $\Pi_{\ZZ}f=f$, so the integer orbit recovers
the full function.  Conversely, if the integer orbit recovers $f$ as an
element of the reconstruction space $\mathcal H_K(\ZZ)$, then
$f=\Pi_{\ZZ}f\in\mathcal H_K(\ZZ)$.
\end{proof}

\begin{proposition}[Integer orbit is the lattice interface used below]%
\label{prop:integer-orbit-lattice-interface}
The lattice input to the observation-channel theorem is exactly the following
integer-orbit package:
\[
\Pi_{\ZZ}:\mathcal H_K\to\mathcal H_K(\ZZ),
\qquad
\sigma(\theta)=\frac{\pi\cosh(2(\pi-|\theta|))}{\sinh(2\pi)},
\qquad
-\pi\le\theta\le\pi,
\]
together with the quotient identity
\[
f(n)=g(n)\ (n\in\ZZ)
\quad\Longleftrightarrow\quad
\Pi_{\ZZ}f=\Pi_{\ZZ}g .
\]
Equivalently,
\[
\mathcal H_K/\mathcal N_{\ZZ}\longrightarrow\mathcal H_K(\ZZ),
\qquad
f+\mathcal N_{\ZZ}\longmapsto\Pi_{\ZZ}f,
\]
is the Hilbert-space quotient determined by the integer evaluations.  Hence
integer samples determine exactly the lattice projection, and they determine
the full ambient function $f\in\mathcal H_K$ if and only if
$f\in\mathcal H_K(\ZZ)$.
The general $h\ZZ$ auxiliary lemma,
Lemma~\ref{lem:auxiliary-uniform-lattice-projection}, supplies only this
integer package.  The Poisson-specific symbol
Proposition~\ref{prop:poisson-lattice-symbol}, the standard imported
sampling consequences recorded immediately after that proposition, and the observation-channel
theorem, Theorem~\ref{thm:poisson-observation-lattice-sampling}, are the
main-chain lattice inputs.
\end{proposition}

\begin{proof}
Setting $h=1$ in Lemma~\ref{lem:auxiliary-uniform-lattice-projection} gives the
orthogonal projection $\Pi_{\ZZ}$, the null space of functions vanishing on
the integer orbit, and the quotient identity.  The same specialization gives
\[
\Sigma_1(\theta)=\sum_{m\in\ZZ}e^{-2|\theta+2\pi m|}
=\frac{\cosh(2(\pi-|\theta|))}{\sinh(2\pi)}.
\]
Because the RKHS kernel satisfies
\[
K(a,b)=\frac{2}{4+(a-b)^2}=\pi P_2(a-b),
\]
the Toeplitz Gram symbol for the integer kernel sections is
\(\sigma(\theta)=\pi\Sigma_1(\theta)\), which is exactly the closed symbol
computed independently in Proposition~\ref{prop:poisson-lattice-symbol}.
The quotient map is well defined and isometric by the orthogonal
decomposition
\[
\mathcal H_K=\mathcal H_K(\ZZ)\oplus\mathcal N_{\ZZ}
\]
from the same specialization.  The quotient identity shows that two functions
have the same integer data exactly when they have the same image under this
map.  Thus integer samples determine no finer datum in the ambient space than
$\Pi_{\ZZ}f$, and $\Pi_{\ZZ}f=f$ is equivalent to
$f\in\mathcal H_K(\ZZ)$.
The sampling and cardinal statements below use only this positive symbol and
the integer closed span $\mathcal H_K(\ZZ)$, through the standard imported
criterion and Corollary~\ref{cor:poisson-cardinal-reconstruction}.  Finally,
Theorem~\ref{thm:poisson-observation-lattice-sampling} applies the same
projection to the RKHS observation profile
  \(F_t=\mathcal U\widetilde\delta_t\).  Thus the auxiliary
  $h\ZZ$ calculation enters only through the integer package displayed above.
\end{proof}

\begin{corollary}[Integer-orbit reconstruction rigidity]%
\label{cor:integer-orbit-reconstruction-rigidity}
Let $Y$ be a set, and let $\mathcal R:\mathcal H_K\to Y$ be any rule whose
value depends only on the integer orbit, in the sense that
\[
f(n)=g(n)\quad(n\in\ZZ)\qquad\Longrightarrow\qquad
\mathcal R(f)=\mathcal R(g).
\]
Then $\mathcal R$ factors uniquely through the integer projection
$\Pi_{\ZZ}$:
\[
\mathcal R=\widetilde{\mathcal R}\circ\Pi_{\ZZ}
\qquad\text{for a unique }\widetilde{\mathcal R}:\mathcal H_K(\ZZ)\to Y.
\]
In particular, if a reconstruction map
$\mathcal R:\mathcal H_K\to\mathcal H_K(\ZZ)$ depends only on integer samples
and is exact on the lattice subspace, meaning
$\mathcal R(F)=F$ for every $F\in\mathcal H_K(\ZZ)$, then
$\mathcal R=\Pi_{\ZZ}$.  Thus the integer-orbit analysis in this paper has a
single possible ambient reconstruction: the lattice projection.
\end{corollary}

\begin{proof}
Define
$\widetilde{\mathcal R}(F):=\mathcal R(F)$ for
$F\in\mathcal H_K(\ZZ)$.  For any $f\in\mathcal H_K$,
Corollary~\ref{cor:integer-orbit-projection} gives
$(\Pi_{\ZZ}f)(n)=f(n)$ for every $n\in\ZZ$.  Since $\mathcal R$ depends only
on these samples, $\mathcal R(f)=\mathcal R(\Pi_{\ZZ}f)$, proving the stated
factorization.  Uniqueness follows because $\Pi_{\ZZ}$ is onto
$\mathcal H_K(\ZZ)$.

If $\mathcal R$ takes values in $\mathcal H_K(\ZZ)$ and is exact on
$\mathcal H_K(\ZZ)$, then
\[
\mathcal R(f)=\mathcal R(\Pi_{\ZZ}f)=\Pi_{\ZZ}f
\]
for every $f\in\mathcal H_K$.  Hence $\mathcal R=\Pi_{\ZZ}$.
\end{proof}

\begin{proposition}[Integer orbit and reconstruction quotient]%
\label{prop:integer-orbit-scoped-quotient}
The integer samples define exactly the quotient of $\mathcal H_K$ by the
nullspace $\mathcal N_{\ZZ}$ and no finer reconstruction on the ambient RKHS.
More precisely, for $f,g\in\mathcal H_K$,
\[
f(n)=g(n)\quad(n\in\ZZ)
\qquad\Longleftrightarrow\qquad
\Pi_{\ZZ}f=\Pi_{\ZZ}g .
\]
Equivalently, the map
\[
\mathcal H_K/\mathcal N_{\ZZ}\longrightarrow \mathcal H_K(\ZZ),
\qquad
f+\mathcal N_{\ZZ}\longmapsto \Pi_{\ZZ}f,
\]
is a well-defined isometric Hilbert-space isomorphism.  Therefore any
integer-orbit reconstruction from these samples factors uniquely through
$\Pi_{\ZZ}$ as in
Corollary~\ref{cor:integer-orbit-reconstruction-rigidity}; it reconstructs
the full ambient function if and only if that function already lies in
$\mathcal H_K(\ZZ)$.
\end{proposition}

\begin{proof}
By Corollary~\ref{cor:integer-orbit-projection},
$f-\Pi_{\ZZ}f\in\mathcal N_{\ZZ}$ and $\Pi_{\ZZ}f$ has the same integer
samples as~$f$.  Hence if $\Pi_{\ZZ}f=\Pi_{\ZZ}g$, then $f$ and $g$ have the
same integer samples.  Conversely, if $f(n)=g(n)$ for every $n\in\ZZ$, then
$f-g\in\mathcal N_{\ZZ}$.  Since
\[
\mathcal H_K=\mathcal H_K(\ZZ)\oplus\mathcal N_{\ZZ}
\]
orthogonally, the projection of $f-g$ onto $\mathcal H_K(\ZZ)$ is zero, so
$\Pi_{\ZZ}f=\Pi_{\ZZ}g$.

The displayed map is well defined because changing $f$ by an element of
$\mathcal N_{\ZZ}=\mathcal H_K(\ZZ)^\perp$ does not change its orthogonal
projection.  It is injective by the first paragraph, surjective because every
$F\in\mathcal H_K(\ZZ)$ is the image of its own coset, and isometric for the
quotient norm because the orthogonal decomposition gives
\[
\|f+\mathcal N_{\ZZ}\|_{\mathcal H_K/\mathcal N_{\ZZ}}
=\inf_{u\in\mathcal N_{\ZZ}}\|\Pi_{\ZZ}f+(f-\Pi_{\ZZ}f)+u\|_{\mathcal H_K}
=\|\Pi_{\ZZ}f\|_{\mathcal H_K}.
\]
Thus the only data-dependent object determined by integer samples in the
ambient RKHS is the projection $\Pi_{\ZZ}f$.  Full reconstruction is therefore
equivalent to $f=\Pi_{\ZZ}f$, namely to $f\in\mathcal H_K(\ZZ)$.
\end{proof}
