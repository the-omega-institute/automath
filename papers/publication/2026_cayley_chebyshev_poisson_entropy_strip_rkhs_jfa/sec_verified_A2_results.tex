\section{Mode Closure, Gaussian Compression, and Dimensional Thresholds}
\label{sec:verified-A2-results}

This section records extensions which do not follow by merely evaluating
further members of the coefficient hierarchy.  The comparison law is allowed
to move, and the multidimensional argument is carried out in the topology
actually needed for the leading entropy coefficient.  Throughout this
section, Gaussian quadrature refers to quadrature for a probability measure
on the real line, not to Gaussian smoothing.

\subsection{The first Cayley channel}

Let \(X=\gamma-\bar\gamma\), and, for \(t>0\), define
\begin{equation}\label{eq:first-cayley-channel-definition}
 a_t(X):=\frac{\ii X}{2t-\ii X},
 \qquad c_k(t):=\EE[a_t(X)^k],\quad k\ge1.
\end{equation}
The following statement concerns the centred law of \(X\).  In particular,
it is invariant under a simultaneous translation of the original law and
its matched reference.

\begin{theorem}[First-channel closure and the even-moment hierarchy]
\label{thm:first-cayley-mode-closure}
Let \(\nu\) be a probability law on \(\RR\) with finite mean, and use the
Cayley coordinate
\[
 z=\frac{1+\ii y}{1-\ii y}\in\TT\setminus\{-1\}.
\]
Then, for every \(t>0\),
\begin{equation}\label{eq:first-channel-exact-fourier-series}
 \frac{(P_t*\nu)(\bar\gamma+ty)}{P_t(ty)}
 =1+\sum_{k\ge1}
 \bigl(\overline{c_k(t)}z^k+c_k(t)z^{-k}\bigr).
\end{equation}
The series converges pointwise for every \(y\in\RR\), locally uniformly in
\(y\), and in \(L^2(\omega)\) whenever \(\EE X^2<\infty\).  Its coefficients
satisfy the exact differential closure
\begin{equation}\label{eq:first-channel-differential-closure}
 c_{k+1}(t)=-c_k(t)-\frac{t}{k}c_k'(t),
 \qquad k\ge1.
\end{equation}
Consequently, the scale trajectory \(c_1\), with its successive derivatives,
determines the matched quotient of the centred law at every scale.  Under
finite variance one also has
\begin{equation}\label{eq:first-channel-chi-square-parseval}
 \chi^2\!\left(P_t*\nu\,\middle\|\,
 P_t(\mathord\cdot-\bar\gamma)\right)
 =2\sum_{k\ge1}|c_k(t)|^2.
\end{equation}
If \(\varphi_X(s)=\EE e^{\ii sX}\), then
\begin{equation}\label{eq:first-channel-laplace-resolvent}
 \frac{1+c_1(t)}{2t}
 =\EE\frac1{2t-\ii X}
 =\int_0^\infty e^{-2ts}\varphi_X(s)\,\dd s.
\end{equation}
Thus the full scale trajectory identifies the centred law by the classical
uniqueness of the Laplace and characteristic transforms.  It identifies the
untranslated law \(\nu\) only when \(\bar\gamma\) is supplied separately.

Moreover, if \(n\ge1\) and \(\mu_{2j}=\EE X^{2j}<\infty\) for
\(1\le j<n\), then
\begin{equation}\label{eq:first-channel-even-moment-remainder}
\begin{aligned}
 &(-1)^{n-1}4^nt^{2n}
 \left[-\Re c_1(t)-\sum_{j=1}^{n-1}
 (-1)^{j-1}\frac{\mu_{2j}}{4^jt^{2j}}\right]\\
 &\hspace{32mm}
 =\EE\!\left[\frac{X^{2n}}{1+X^2/(4t^2)}\right]
 \uparrow\EE|X|^{2n}
 \quad(t\to\infty),
\end{aligned}
\end{equation}
where the limit is permitted to be \(+\infty\).
\end{theorem}

\begin{proof}
Fix a deterministic translation \(\varepsilon=X/t\).
The pole calculation in
Lemma~\ref{lem:second-mode-cauchy-translate-transform} gives
\[
 R_\varepsilon(y)
 =1+\sum_{k\ge1}\bigl(\overline{a_t(X)}^{\,k}z^k
 +a_t(X)^kz^{-k}\bigr).
\]
Indeed, the interior pole in that calculation is
\(-\varepsilon/(\varepsilon+2\ii)=\ii X/(2t-\ii X)\), while the inverse of
the exterior pole is its complex conjugate.  For finite \(X\),
\(|a_t(X)|<1\).  On a compact subset of
\(\TT\setminus\{-1\}\), the denominators in the two geometric remainders
are bounded away from zero uniformly over the compactified parameter line
\(X\in\RR\cup\{\infty\}\); the only boundary value of \(a_t(X)\) is
\(-1\).  Dominated convergence therefore permits expectation of the finite
partial sums and passage to the limit, proving
\eqref{eq:first-channel-exact-fourier-series} and the asserted local
uniformity.

Pointwise differentiation gives
\[
 a_t(X)^{k+1}=-a_t(X)^k-\frac{t}{k}
 \frac{\dd}{\dd t}a_t(X)^k.
\]
The derivative is integrable locally uniformly in \(t\), because the same
identity bounds its absolute value by \(2k/t\).  Differentiation under the
expectation proves \eqref{eq:first-channel-differential-closure}.

Also \(1+a_t(X)=2t/(2t-\ii X)\).  The identity
\[
 \frac1{2t-\ii X}=\int_0^\infty e^{-(2t-\ii X)s}\,\dd s
\]
and Fubini's theorem prove \eqref{eq:first-channel-laplace-resolvent}.

If \(\EE X^2<\infty\), Jensen's inequality and the geometric series give
\[
 \sum_{k\ge1}|c_k(t)|^2
 \le\EE\sum_{k\ge1}|a_t(X)|^{2k}
 =\frac{\EE X^2}{4t^2}<\infty.
\]
Thus the Fourier series converges in \(L^2(\omega)\), where it agrees with
the quotient already identified pointwise.  Parseval's identity gives
\eqref{eq:first-channel-chi-square-parseval}.

Finally,
\begin{equation}\label{eq:first-channel-real-part}
 -\Re c_1(t)=\EE\frac{X^2}{4t^2+X^2}.
\end{equation}
For \(q\ge0\), finite geometric division yields
\[
 \frac q{1+q}=\sum_{j=1}^{n-1}(-1)^{j-1}q^j
 +(-1)^{n-1}\frac{q^n}{1+q}.
\]
Apply this identity with \(q=X^2/(4t^2)\), take expectations, and multiply
by \(4^nt^{2n}\).  The right side of
\eqref{eq:first-channel-even-moment-remainder} increases pointwise with
\(t\), so monotone convergence proves the final assertion.
\end{proof}

\subsection{The infinite critical-moment boundary is not spectrally identifiable}

The finite critical-moment theorem below cannot be extended by replacing its
moment hypothesis with vague multivariate regular variation at the same
index.  We use the standard measure formulation of multivariate regular
variation on \(\RR^d\setminus\{0\}\), as in
Hult--Lindskog~\cite{HultLindskog2006RegularVariation}.  For a probability
measure \(\sigma\) on \(S^{d-1}\) and \(p>0\), write
\begin{equation}\label{eq:critical-rv-tail-measure}
 \mu_\sigma(\dd\rho,\dd\theta)
 =p\rho^{-p-1}\dd\rho\,\sigma(\dd\theta).
\end{equation}

We first record a lower bound which requires no regular-variation
hypothesis.

\begin{lemma}[Finite-covariance Fatou lower bound]
\label{lem:finite-covariance-fatou-lower-bound}
Let \(d\ge1\), and let \(\nu\) be a centred probability law on \(\RR^d\)
with finite covariance matrix \(\Sigma\).  Then, with the relative entropy
understood in the extended sense,
\begin{equation}\label{eq:finite-covariance-fatou-lower-bound}
 \liminf_{t\to\infty}t^4H_d(t)\ge \mathcal Q_d(\Sigma).
\end{equation}
\end{lemma}

\begin{proof}
Fix \(y\in\RR^d\).  Taylor's formula near the origin, followed by the
elementary large-\(z\) bound, gives
\[
 \left|F_y(z)-1-D_zF_y(0)z\right|\le C_y|z|^2,
 \qquad z\in\RR^d.
\]
Indeed, the assertion is the usual second-order Taylor bound for
\(|z|\le1\); for \(|z|>1\), \(F_y(z)\le(1+|y|^2)^{(d+1)/2}\), while the
subtracted affine function is \(O_y(1+|z|)\).  Centering and dominated
convergence therefore imply
\[
 t^2\delta_t^{(d)}(y)\longrightarrow b_\Sigma(y).
\]
Since \(\Phi\ge0\) on \([-1,\infty)\) and
\(t^4\Phi(\delta_t^{(d)}(y))\to b_\Sigma(y)^2/2\), Fatou's lemma yields
\[
 \liminf_{t\to\infty}t^4H_d(t)
 \ge \frac12\int_{\RR^d}b_\Sigma^2\dd\Omega_d
 =\mathcal Q_d(\Sigma),
\]
where the last identity is Theorem~\ref{thm:multidim-covariance-algebra}.
\end{proof}

\begin{theorem}[Critical vague tails do not identify the leading entropy]
\label{thm:critical-vague-tail-nonidentifiability}
Fix \(d\ge4\), and set
\[
 p=\frac{4(d+1)}{d+5},\qquad
 q=\frac{d+5}{d+1}=1+\frac4{d+1};
\]
thus \(pq=4\).  Let \(\sigma\) be normalized surface measure on
\(S^{d-1}\), and let \(L(r)=\log r\).  For every positive-definite matrix
\(\Sigma\), there are two centred probability laws
\(\nu_{\rm ac}\) and \(\nu_{\rm cl}\) with covariance \(\Sigma\) such that
\begin{equation}\label{eq:critical-vague-tail-common-limit}
 \frac{r^p}{\log r}\,\nu_\bullet(r\,\mathord\cdot)
 \xrightarrow[r\to\infty]{v}\mu_\sigma
 \quad\text{on }\RR^d\setminus\{0\},
 \qquad \bullet\in\{\mathrm{ac},\mathrm{cl}\},
\end{equation}
but
\begin{equation}\label{eq:critical-vague-tail-contrasting-entropy}
 t^4H_{d,{\rm ac}}(t)\longrightarrow\mathcal Q_d(\Sigma),
 \qquad
 \limsup_{t\to\infty}t^4H_{d,{\rm cl}}(t)=+\infty.
\end{equation}
Consequently, the spectral measure and the slowly varying factor do not
determine whether the covariance lower bound is an asymptotic equality,
even when the spectral measure is smooth, everywhere nondegenerate, and
antipodally symmetric.
\end{theorem}

\begin{proof}
We construct the tails first and then complete them on a bounded set.  Choose
\(s_1\) large and put \(R_0=e^{s_1}\).  The diffuse tail is
\begin{equation}\label{eq:critical-diffuse-tail-definition}
 \tau_{\rm ac}(\dd r,\dd\theta)
 =p r^{-p-1}\log r\,\mathbf1_{\{r\ge R_0\}}
   \dd r\,\sigma(\dd\theta).
\end{equation}
Changing variables \(r=R\rho\) and applying dominated convergence against
each compactly supported continuous test function on
\(\RR^d\setminus\{0\}\) proves
\(R^p(\log R)^{-1}\tau_{\rm ac}(R\,\mathord\cdot)\to\mu_\sigma\).

For the clustered tail, define recursively
\[
 s_{n+1}=s_n+s_n^{-1/2},\qquad R_n=e^{s_n},\qquad
 W_n=\int_{s_n}^{s_{n+1}}pu e^{-pu}\dd u.
\]
Choose an antipodally symmetric equal-weight empirical probability
\(\pi_n\) on \(S^{d-1}\), supported on
\(N_n\asymp s_n^{1/4}\) points, so that \(\pi_n\Rightarrow\sigma\), and set
\begin{equation}\label{eq:critical-clustered-tail-definition}
 \tau_{\rm cl}=\sum_{n\ge1}W_n(\theta\mapsto R_n\theta)_\#\pi_n.
\end{equation}
This tail has the same vague limit.  To verify the assertion without a
moving-set argument, let \(f\in C_c(\RR^d\setminus\{0\})\) and write
\(R=e^s\).  Then
\[
 \frac{e^{ps}}s\int f(x/e^s)\tau_{\rm cl}(\dd x)
 =\frac{e^{ps}}s\sum_nW_n
   \int f(e^{s_n-s}\theta)\pi_n(\dd\theta).
\]
Only indices with \(s_n-s\) in a fixed compact interval occur.  On this
interval the maximal mesh \(s_n^{-1/2}\) tends to zero, \(u/s\to1\)
uniformly, and \(\pi_n\Rightarrow\sigma\) uniformly over the indices in
question.  The right side is therefore a Riemann sum converging to
\[
 \int_\RR pe^{-pv}\int_{S^{d-1}}f(e^v\theta)
 \sigma(\dd\theta)\dd v
 =\int f\dd\mu_\sigma.
\]
This proves \eqref{eq:critical-vague-tail-common-limit} for the two tails.

Both tails have arbitrarily small total mass and covariance when \(s_1\) is
large, because \(p>2\).  They are symmetric.  After increasing \(s_1\),
their covariance matrices \(C_\bullet\) satisfy
\(C_\bullet<\Sigma\).  A bounded symmetric measure of the remaining mass
can realize the covariance \(\Sigma-C_\bullet\): diagonalize that matrix
and place equal pairs of atoms on its eigenvectors with the corresponding
radii.  Thus each tail can be completed to a centred probability law with
covariance exactly \(\Sigma\), without changing its vague limit.

We next prove the diffuse asymptotic.  Let
\(m=2\) when \(p\le3\) and \(m=3\) when \(p>3\).  The local Taylor estimate
\eqref{eq:local-multidim-taylor-remainder}, the boundedness of the retained
modes, and symmetry give
\begin{equation}\label{eq:critical-diffuse-quotient-remainder}
 \|r_t\|_\infty+\|r_t\|_{L^1(\Omega_d)}
 \le C\left\{
 t^{-(m+1)}\EE[|X|^{m+1};|X|\le t]+t^{-p}\log t
 \right\},
\end{equation}
where
\(\delta_t^{(d)}=t^{-2}b_\Sigma+r_t\).  For completeness, the only
estimate in \eqref{eq:critical-diffuse-quotient-remainder} not already
contained in the local Taylor bound is the uniform control of the diffuse
tail.  In Lebesgue coordinates it follows from
\begin{equation}\label{eq:critical-diffuse-convolution-bound}
 \sup_{y\in\RR^d}\int_{|z|>1}F_y(z)|z|^{-d-p}
 (1+\log|z|)\dd z<\infty.
\end{equation}
For \(|y|\le2\) the bound is immediate.  If \(|y|=R>2\), split the
integral into \(|z-y|\le R/2\) and its complement.  In the first region
\(|z|\asymp R\), and translation by \(y\) bounds the integral by
\(CR^{1-p}(1+\log R)\); in the second region \(F_y(z)\le C_d\), and the
radial weight is integrable.  Since \(p>1\), this proves
\eqref{eq:critical-diffuse-convolution-bound}.  Integration in \(y\) uses
\(\int F_y(z)\Omega_d(\dd y)=1\), and gives the corresponding \(L^1\)
bound.  The polynomial tail terms are bounded by the same
\(O(t^{-p}\log t)\) quantity because \(m<p\), proving
\eqref{eq:critical-diffuse-quotient-remainder}.

When \(p\ne3\), the right side of
\eqref{eq:critical-diffuse-quotient-remainder} is
\(O(t^{-p}\log t)=o(t^{-2})\).  At the sole integer endpoint \(p=3\),
equivalently \(d=11\), its first term is
\(O(t^{-3}(\log t)^2)=o(t^{-2})\).  The second-order quotient criterion,
Theorem~\ref{thm:second-order-quotient-kl-coefficient}, now gives the first
limit in \eqref{eq:critical-vague-tail-contrasting-entropy}.  Notice also
that the whole construction is antipodally symmetric; at \(d=11\) every
truncated third-moment tensor is therefore identically zero.

It remains to obtain the clustered lower bound.  Each atom in the \(n\)th
shell has mass
\[
 a_n=\frac{W_n}{N_n}\sim p s_n^{1/4}R_n^{-p},
 \qquad R_n^pa_n\longrightarrow\infty.
\]
Fix one such atom, choose a sufficiently small constant \(\kappa>0\), and
put
\[
 t_n=\kappa R_na_n^{1/(d+1)}.
\]
Then \(t_n\to\infty\) and \(t_n/R_n\to0\).  If \(A_n\) is the ball of
radius \(t_n\) centred at that atom, the corresponding mixture component
and the centred reference satisfy
\[
 (P_{t_n}^{(d)}*\nu_{\rm cl})(A_n)\ge c_da_n,
 \qquad
 P_{t_n}^{(d)}(A_n)\le C_d(t_n/R_n)^{d+1}
 =C_d\kappa^{d+1}a_n.
\]
Choose \(\kappa\) so that the second constant is smaller by a fixed factor.
Data processing to \(\{A_n,A_n^c\}\), followed by
\(D(\operatorname{Ber}(u)\|\operatorname{Ber}(v))
\ge u\log(u/v)-u+v\), gives \(H_{d,{\rm cl}}(t_n)\ge c'_da_n\).
Consequently,
\[
 t_n^4H_{d,{\rm cl}}(t_n)
 \ge c_dR_n^4a_n^{1+4/(d+1)}
 =c_d(R_n^pa_n)^q
 \asymp s_n^{q/4}\longrightarrow\infty,
\]
where \(pq=4\).  This proves the second limit and the theorem.
\end{proof}

\subsection{Moment-matched comparisons and Gaussian compression}

For a probability law \(\rho\) with a finite \(j\)th absolute moment, write
\(m_j(\rho)=\int x^j\rho(\dd x)\).  The comparison theorem below does not
require either law to be centred.

\begin{theorem}[First unmatched moment for Poisson relative entropy]
\label{thm:moment-matched-poisson-kl}
Let \(r\ge2\), and let \(\nu,\eta\) be probability laws on \(\RR\) with
finite \(r\)th absolute moments.  Suppose
\[
 m_j(\nu)=m_j(\eta)\quad(0\le j<r),
 \qquad \Delta_r:=m_r(\nu)-m_r(\eta)\ne0.
\]
Then
\begin{equation}\label{eq:moment-matched-poisson-kl}
 D_{\rm KL}(P_t*\nu\|P_t*\eta)
 =C_r\Delta_r^2t^{-2r}+o(t^{-2r}),
 \qquad
 C_r=4^{-r}\binom{2r-2}{r-1}.
\end{equation}
\end{theorem}

\begin{proof}
Divide both smoothed densities by \(P_t(ty)\), and denote the resulting
quotients by \(q_t^\nu\) and \(q_t^\eta\).  The global two-region Taylor
remainder \eqref{eq:one-dimensional-two-region-taylor-remainder}, applied
at order \(r\), gives
\begin{equation}\label{eq:moment-matched-quotient-difference}
 q_t^\nu-q_t^\eta
 =\Delta_r u_r t^{-r}+o(t^{-r})
 \quad\text{in }L^\infty(\omega)\cap L^1(\omega).
\end{equation}
The same argument at order one shows \(q_t^\eta\to1\) uniformly.  Since the
two quotient differences have zero \(\omega\)-mean, the exact Bregman form is
\[
 D_{\rm KL}(P_t*\nu\|P_t*\eta)
 =\int q_t^\eta
 \Phi\!\left(\frac{q_t^\nu-q_t^\eta}{q_t^\eta}\right)\dd\omega.
\]
Uniform Taylor expansion of \(\Phi(s)=s^2/2+O(s^3)\), together with
\eqref{eq:moment-matched-quotient-difference}, yields
\[
 D_{\rm KL}(P_t*\nu\|P_t*\eta)
 =\frac{\Delta_r^2}{2}t^{-2r}\int u_r^2\dd\omega+o(t^{-2r}).
\]
By the Laurent coefficient formula
\eqref{eq:mode-laurent-coefficients} and Parseval,
\[
 \frac12\int u_r^2\dd\omega
 =4^{-r}\sum_{j=0}^{r-1}\binom{r-1}{j}^2
 =4^{-r}\binom{2r-2}{r-1},
\]
where the last equality is Vandermonde's identity.
\end{proof}

We now state the Gaussian quadrature consequence with the nondegeneracy
hypothesis made explicit.  This qualification is necessary: an \(n\)-node
quadrature rule is not defined for an arbitrary law supported on fewer than
\(n\) points.

\begin{theorem}[Gaussian-comparison entropy threshold]
\label{thm:gauss-poisson-kl-threshold}
Let \(n\ge1\), and let \(\nu\) be a probability law on \(\RR\) satisfying
\[
 \EE|X|^{2n-1}<\infty,
 \qquad |\operatorname{supp}\nu|\ge n.
\]
Let \(\pi_n\) be the monic degree-\(n\) polynomial orthogonal to
\(1,x,\ldots,x^{n-1}\) with respect to \(\nu\), and let \(G_n\nu\) be its
\(n\)-node Gaussian quadrature law.  Thus
\begin{equation}\label{eq:gauss-moment-matching}
 m_j(G_n\nu)=m_j(\nu),\qquad 0\le j\le2n-1.
\end{equation}
Then
\begin{equation}\label{eq:gauss-poisson-finite-limit}
 \EE|X|^{2n}<\infty
 \quad\Longrightarrow\quad
 \lim_{t\to\infty}t^{4n}
 D_{\rm KL}(P_t*\nu\|P_t*G_n\nu)
 =4^{-2n}\binom{4n-2}{2n-1}
 \|\pi_n\|_{L^2(\nu)}^4,
\end{equation}
whereas
\begin{equation}\label{eq:gauss-poisson-infinite-limit}
 \EE|X|^{2n}=\infty
 \quad\Longrightarrow\quad
 t^{4n}D_{\rm KL}(P_t*\nu\|P_t*G_n\nu)\longrightarrow+\infty.
\end{equation}
Consequently,
\begin{equation}\label{eq:gauss-poisson-moment-equivalence}
 \EE|X|^{2n}<\infty
 \quad\Longleftrightarrow\quad
 \limsup_{t\to\infty}t^{4n}
 D_{\rm KL}(P_t*\nu\|P_t*G_n\nu)<\infty.
\end{equation}
\end{theorem}

\begin{proof}
The existence, positivity, and exactness
\eqref{eq:gauss-moment-matching} of the Gaussian rule are classical; see
\cite{GolubWelsch1969GaussQuadrature}.  If the \(2n\)th moment is finite,
then, because \(\pi_n\) vanishes at every quadrature node and
\(\pi_n^2-x^{2n}\) has degree at most \(2n-1\),
\begin{equation}\label{eq:gauss-missing-moment-norm}
 m_{2n}(\nu)-m_{2n}(G_n\nu)
 =\int\pi_n^2\dd\nu=\|\pi_n\|_{L^2(\nu)}^2.
\end{equation}
Theorem~\ref{thm:moment-matched-poisson-kl}, with \(r=2n\), now proves
\eqref{eq:gauss-poisson-finite-limit}.  If the norm in
\eqref{eq:gauss-missing-moment-norm} vanishes, then \(\nu\) is itself
supported on the Gaussian nodes and both sides of
\eqref{eq:gauss-poisson-finite-limit} are zero.

Suppose instead that \(\EE|X|^{2n}=\infty\).  Apply
\eqref{eq:first-channel-even-moment-remainder} to \(\nu\) and to the
finite-support law \(G_n\nu\).  Their even moments through order \(2n-2\)
agree, and therefore
\[
 t^{2n}\left|\Re\{c_1^\nu(t)-c_1^{G_n\nu}(t)\}\right|
 \longrightarrow+\infty.
\]
The real part of \(z\) is a bounded test of norm one on the Cayley circle.
Pinsker's inequality and the Fourier coefficient identity therefore give
\[
 2D_{\rm KL}(P_t*\nu\|P_t*G_n\nu)
 \ge\left|\Re\{c_1^\nu(t)-c_1^{G_n\nu}(t)\}\right|^2.
\]
Multiplication by \(t^{4n}\) proves
\eqref{eq:gauss-poisson-infinite-limit}, and the equivalence follows.
\end{proof}

\begin{remark}[Classical optimal degree]
The assertion that an \(n\)-node positive quadrature rule has maximal exact
degree \(2n-1\), and that Gaussian quadrature attains this degree, is a
classical quadrature theorem, not a new result of this article.  The node
polynomial gives the elementary obstruction at degree \(2n\); see
\cite{GolubWelsch1969GaussQuadrature}.
\end{remark}

\begin{corollary}[Two-node fourth-moment coefficient]
\label{cor:two-node-gauss-fourth-moment}
Assume
\[
 \EE X=0,\qquad \EE X^2=1,\qquad \EE|X|^3<\infty,
\]
and put
\[
 \beta_3=\EE X^3,\qquad
 \pi_2(x)=x^2-\beta_3x-1.
\]
Let \(G_2\nu\) be the two-node Gaussian quadrature law.  If
\(\EE X^4<\infty\), then, with
\[
 \kappa=\|\pi_2\|_{L^2(\nu)}^2
 =\EE X^4-1-\beta_3^2,
\]
one has
\begin{equation}\label{eq:two-node-fourth-moment-coefficient}
 \lim_{t\to\infty}t^8D_{\rm KL}(P_t*\nu\|P_t*G_2\nu)
 =\frac5{64}\kappa^2.
\end{equation}
If \(\EE X^4=\infty\), the left side of
\eqref{eq:two-node-fourth-moment-coefficient} tends to \(+\infty\).  For a
symmetric law,
\(G_2\nu=(\delta_{-1}+\delta_1)/2\) and
\(\kappa=\EE X^4-1\).
\end{corollary}

\begin{proof}
The polynomial \(\pi_2\) is monic and orthogonal to \(1,x\).  Its squared
norm is the identity proved in \path{sec_appendix.tex}, Lemma ``Normalized
fourth-moment defect algebra.''  Apply
Theorem~\ref{thm:gauss-poisson-kl-threshold} and use
\(4^{-4}\binom63=5/64\).  In the symmetric case, exactness through degree
three and unit variance force the nodes and weights to be
\(\{-1,1\}\) and \(\{1/2,1/2\}\).
\end{proof}

\begin{theorem}[Square boundary at a missing Gaussian moment]
\label{thm:gauss-regular-variation-square-law}
Let \(n\ge2\), put \(p=2n\), and let \(\nu\) satisfy the hypotheses of
Theorem~\ref{thm:gauss-poisson-kl-threshold}.  Assume, for an eventually
positive slowly varying function \(L\) and constants \(c_+,c_-\ge0\) with
\(c_++c_->0\),
\[
 \mathbb P(X>x)\sim c_+x^{-p}L(x),\qquad
 \mathbb P(X<-x)\sim c_-x^{-p}L(x),
\]
and assume
\[
 \ell_L(t)=\int_{x_0}^t\frac{L(s)}s\dd s\longrightarrow\infty.
\]
Then, with
\(M_p(t)=\EE[|X|^p\mathbf1_{\{|X|\le t\}}]\),
\begin{equation}\label{eq:gauss-regular-variation-square-law}
 D_{\rm KL}(P_t*\nu\|P_t*G_n\nu)
 \sim C_p t^{-2p}M_p(t)^2,
 \qquad C_p=4^{-p}\binom{2p-2}{p-1}.
\end{equation}
In particular,
\[
 M_p(t)\sim p(c_++c_-)\ell_L(t),
\]
so the boundary normalization is the square of the integrated slowly
varying factor and the leading constant is positive.
\end{theorem}

\begin{proof}
The quotient estimate proved for
Theorem~\ref{cor:main-body-conditional-missing-top-moment-tail}, before the
entropy transfer is applied, gives
\[
 q_t^\nu
 =\sum_{j=0}^{p-1}m_j(\nu)u_jt^{-j}
 +u_pt^{-p}M_p(t)+e_t,
 \qquad
 \|e_t\|_\infty+\|e_t\|_{L^1(\omega)}
 =o(t^{-p}M_p(t)).
\]
The finite-support law \(G_n\nu\) has an ordinary expansion through order
\(p\), and its moments through order \(p-1\) agree with those of \(\nu\).
Since \(M_p(t)\to\infty\), subtraction yields
\begin{equation}\label{eq:gauss-boundary-quotient-difference}
 q_t^\nu-q_t^{G_n\nu}
 =u_pt^{-p}M_p(t)+o_{L^\infty\cap L^1(\omega)}
 (t^{-p}M_p(t)).
\end{equation}
Interpolation also gives the same relative error in \(L^2(\omega)\).
Furthermore \(q_t^{G_n\nu}\to1\) uniformly.  Applying the Bregman expansion
from the proof of Theorem~\ref{thm:moment-matched-poisson-kl} to
\eqref{eq:gauss-boundary-quotient-difference} gives
\[
 D_{\rm KL}(P_t*\nu\|P_t*G_n\nu)
 =\frac12t^{-2p}M_p(t)^2\int u_p^2\dd\omega
 +o(t^{-2p}M_p(t)^2).
\]
The Parseval calculation in that theorem identifies the constant as \(C_p\).
The asymptotic for \(M_p\) is Karamata's boundary integral theorem, exactly
as in Theorem~\ref{cor:main-body-conditional-missing-top-moment-tail}.
\end{proof}

\subsection{The multidimensional \texorpdfstring{\(L^2\)}{L2} and entropy
thresholds}

Write \(a=(d+1)/2\) and
\[
 F_y(z)=\left(\frac{1+|y|^2}{1+|y-z|^2}\right)^a.
\]
Let \(T_2(y,z)\) denote the Taylor polynomial of \(F_y(z)\) at \(z=0\)
through homogeneous degree two.

\begin{lemma}[Large translations in the Cauchy quotient space]
\label{lem:multidim-large-translation-l2-growth}
For every \(d\ge1\),
\begin{equation}\label{eq:translated-quotient-l2-asymptotic}
 \|F_\cdot(z)\|_{L^2(\Omega_d)}^2
 \sim2^{-d}|z|^{d+1}\qquad(|z|\to\infty).
\end{equation}
If
\[
 q_d:=\max\left\{2,\frac{d+1}{2}\right\},
\]
then a constant \(C_d\) exists such that
\begin{equation}\label{eq:second-order-l2-global-remainder}
 \|F_\cdot(z)-T_2(\cdot,z)\|_{L^2(\Omega_d)}
 \le C_d
 \begin{cases}|z|^3,&|z|\le1,\\ |z|^{q_d},&|z|>1.
 \end{cases}
\end{equation}
\end{lemma}

\begin{proof}
With \(c_d=\Gamma((d+1)/2)/\pi^{(d+1)/2}\) and \(y=z+u\),
\[
 \|F_\cdot(z)\|_2^2
 =c_d\int_{\RR^d}
 \frac{(1+|z+u|^2)^a}{(1+|u|^2)^{d+1}}\dd u.
\]
After division by \(|z|^{d+1}\), dominated convergence applies and gives
\[
 c_d\int_{\RR^d}(1+|u|^2)^{-(d+1)}\dd u
 =\frac{\Gamma((d+1)/2)\Gamma(d/2+1)}
 {\sqrt\pi\Gamma(d+1)}=2^{-d}.
\]
For \(|z|\le1\), differentiation under the \(L^2(\Omega_d)\) norm through
third order is uniform on the unit ball, and Taylor's formula gives the
first line of \eqref{eq:second-order-l2-global-remainder}.  For \(|z|>1\),
\eqref{eq:translated-quotient-l2-asymptotic}, enlarged to a global upper
bound, and the fact that \(T_2\) is a degree-two polynomial in \(z\) with
\(L^2(\Omega_d)\) coefficients give
\[
 \|F_\cdot(z)-T_2(\cdot,z)\|_2
 \le C_d(|z|^{(d+1)/2}+|z|^2),
\]
which is the second line.
\end{proof}

\begin{theorem}[Sharp absolute-moment condition for the \(L^2\) quotient]
\label{thm:multidim-l2-moment-threshold}
Let \(\nu\) be a centred probability law on \(\RR^d\), with covariance
matrix \(\Sigma\).  If
\begin{equation}\label{eq:multidim-qd-moment}
 \EE|X|^{q_d}<\infty,
 \qquad q_d=\max\left\{2,\frac{d+1}{2}\right\},
\end{equation}
then
\begin{equation}\label{eq:multidim-l2-second-order-convergence}
 t^2\delta_t^{(d)}\longrightarrow b_\Sigma
 \quad\text{in }L^2(\Omega_d).
\end{equation}
For \(d\ge4\), this exponent is sharp among hypotheses stated solely as a
finite absolute moment: for every \(2\le p<q_d\), there is a centred law
with positive-definite covariance and finite \(p\)th absolute moment for
which \eqref{eq:multidim-l2-second-order-convergence} fails.
\end{theorem}

\begin{proof}
After centering removes the degree-one term and covariance averages the
degree-two term, Minkowski's inequality gives
\[
 \|t^2\delta_t^{(d)}-b_\Sigma\|_2
 \le\EE\left[t^2
 \|F_\cdot(X/t)-T_2(\cdot,X/t)\|_2\right].
\]
The integrand converges pointwise to zero.  On \(|X|\le t\), the first line
of \eqref{eq:second-order-l2-global-remainder} bounds it by \(C|X|^2\);
on \(|X|>t\), the second line bounds it by
\(C|X|^{q_d}t^{2-q_d}\le C|X|^{q_d}\).  Dominated convergence proves
\eqref{eq:multidim-l2-second-order-convergence}.

For sharpness, let \(a=(d+1)/2>2\), choose \(p<a\), and take a symmetric
spike law with masses \(w_j/2\) at \(\pm R_je_1\), where
\(w_j=\alpha_jR_j^{-p}\), \(\sum_j\alpha_j<\infty\), and \(R_j\) grows
recursively.  Add a sufficiently small bounded symmetric component spanning
\(\RR^d\).  The resulting law has positive-definite finite covariance and
finite \(p\)th moment.  Choose
\[
 0<s<\min\left\{\frac{a-p}{a-2},\,1-\frac{p}{d+1}\right\}
\]
and put \(t_j=R_j^s\).
On a fixed ball about \(R_je_1/t_j\), the \(j\)th translated quotient has
\(L^2(\Omega_d)\) size comparable to
\((R_j/t_j)^a\), by the localized proof of
\eqref{eq:translated-quotient-l2-asymptotic}.  Consequently its contribution
to \(t_j^2\delta_{t_j}^{(d)}\) has size at least
\[
 c\alpha_jR_j^{a-p-s(a-2)}.
\]
The first restriction on \(s\) makes the displayed exponent positive; the
second gives \(w_j(R_j/t_j)^{d+1}\to\infty\), so the localized positive spike
also dominates the constant background.  Choosing \(R_j\) fast enough makes
the displayed quantity diverge and dominates the bounded quadratic mode and
all previously chosen terms.  Hence the \(L^2\)
convergence fails.
\end{proof}

\begin{theorem}[Low-dimensional finite-covariance entropy threshold]
\label{thm:low-dimensional-finite-covariance-threshold}
Let \(1\le d\le3\), and let \(\nu\) be a probability law on \(\RR^d\) with
finite mean.  With
\[
 H_d(t)=D_{\rm KL}\!\left(P_t^{(d)}*\nu\,\middle\|\,
 P_t^{(d)}(\mathord\cdot-\bar\gamma)\right),
\]
one has
\begin{equation}\label{eq:low-dimensional-covariance-iff}
 \EE|X|^2<\infty
 \quad\Longleftrightarrow\quad
 \limsup_{t\to\infty}t^4H_d(t)<\infty.
\end{equation}
If the covariance is finite, then
\begin{equation}\label{eq:low-dimensional-covariance-limit}
 H_d(t)=\mathcal Q_d(\Sigma)t^{-4}+o(t^{-4}).
\end{equation}
If it is infinite, then \(t^4H_d(t)\to+\infty\) in the extended sense.
\end{theorem}

\begin{proof}
For \(d\le3\), \(q_d=2\).  Thus Theorem
\ref{thm:multidim-l2-moment-threshold} and the \(L^2\) alternative in
Theorem~\ref{thm:second-order-quotient-kl-coefficient} prove
\eqref{eq:low-dimensional-covariance-limit}.  The entropy representation is
valid in the extended sense by change of variables; for large \(t\), the
\(L^2\) convergence makes \(\Phi(\delta_t^{(d)})\) integrable because
\(\Phi(s)\le C s^2\) on \([-1,\infty)\).

Conversely, if \(\EE|X|^2=\infty\), some unit vector \(e\) satisfies
\(\EE(e\cdot X)^2=\infty\).  Projection onto \(e\) sends the radial
\(d\)-dimensional Poisson law to the one-dimensional Cauchy law at the same
scale.  The data-processing inequality and
Theorem~\ref{thm:sharp-leading-kl-variance-threshold} therefore give
\[
 H_d(t)\ge
 D_{\rm KL}(P_t*\nu_e\|P_t(\mathord\cdot-\bar\gamma_e)),
\]
whose product with \(t^4\) tends to \(+\infty\).
\end{proof}

\begin{proposition}[Finite covariance does not control high-dimensional KL]
\label{prop:high-dimensional-finite-covariance-obstruction}
For every \(d\ge4\), there exists a centred discrete probability law on
\(\RR^d\) with positive-definite finite covariance such that
\begin{equation}\label{eq:high-dimensional-finite-covariance-obstruction}
 \limsup_{t\to\infty}t^4H_d(t)=+\infty.
\end{equation}
\end{proposition}

\begin{proof}
Choose positive \(\alpha_j\) with \(\sum_j\alpha_j<\infty\), radii
\(R_j\uparrow\infty\), and weights
\(w_j=\alpha_jR_j^{-2}\).  After rescaling the sequence if necessary,
\(W=\sum_jw_j<1\).  Begin with
\[
 (1-W)\delta_0+\sum_{j\ge1}\frac{w_j}{2}
 (\delta_{R_je_1}+\delta_{-R_je_1}),
\]
and transfer an arbitrarily small amount of the mass at zero to a bounded
centred symmetric law spanning \(\RR^d\).  The covariance is then positive
definite and finite because \(\sum_jw_jR_j^2=\sum_j\alpha_j<\infty\).

Fix a sufficiently small constant \(\kappa>0\), and set
\[
 t_j=\kappa R_jw_j^{1/(d+1)}.
\]
The radii may be chosen recursively so that \(t_j\to\infty\).  Let \(A_j\)
be the ball of radius \(t_j\) centred at \(R_je_1\).  The \(j\)th mixture
component gives
\[
 (P_{t_j}^{(d)}*\nu)(A_j)\ge c_dw_j,
\]
whereas, since \(t_j/R_j\to0\), the centred reference satisfies
\[
 P_{t_j}^{(d)}(A_j)
 \le C_d(t_j/R_j)^{d+1}
 =C_d\kappa^{d+1}w_j.
\]
Choose \(\kappa\) so that the second constant is strictly smaller than the
first.  Data processing to the partition \(\{A_j,A_j^c\}\) and the
elementary Bernoulli relative-entropy bound then give
\(H_d(t_j)\ge c'_dw_j\).  Hence
\[
 t_j^4H_d(t_j)
 \ge c''_dR_j^4w_j^{1+4/(d+1)}
 =c''_d\alpha_j^{1+4/(d+1)}
 R_j^{2(d-3)/(d+1)}.
\]
The exponent of \(R_j\) is positive exactly when \(d>3\).  Choosing the
radii recursively makes the final expression tend to infinity.
\end{proof}

For the high-dimensional entropy topology, define
\begin{equation}\label{eq:critical-kl-exponents}
 q_d:=\frac{d+5}{d+1},
 \qquad
 p_d:=\frac{4(d+1)}{d+5}.
\end{equation}
Thus \(1<q_d<2\), \(2<p_d<4\), and \(p_dq_d=4\) when \(d\ge4\).

\begin{lemma}[Critical translation remainder]
\label{lem:critical-lq-translation-remainder}
Let \(d\ge4\), let \(p_d,q_d\) be as in
\eqref{eq:critical-kl-exponents}, and put \(m=\lfloor p_d\rfloor\).
If \(T_m(y,z)\) is the Taylor polynomial of \(F_y(z)\) through degree
\(m\), then
\begin{equation}\label{eq:critical-lq-exact-translation-norm}
 \|F_\cdot(z)\|_{L^{q_d}(\Omega_d)}^{q_d}
 =1+\frac{2(d+5)}{(d+1)(d+3)}|z|^2
 +\frac{3}{(d+1)(d+3)}|z|^4.
\end{equation}
Consequently, for a constant depending only on \(d\),
\begin{equation}\label{eq:critical-lq-global-remainder}
 \|F_\cdot(z)-T_m(\cdot,z)\|_{L^{q_d}(\Omega_d)}
 \le C_d
 \begin{cases}
 |z|^{m+1},& |z|\le1,\\
 |z|^{p_d},& |z|>1,
 \end{cases}
 \le C_d|z|^{p_d}.
\end{equation}
Here \(m=2\) for \(4\le d\le10\), while \(m=3\) for \(d\ge11\),
including the integer endpoint \(p_{11}=3\).
\end{lemma}

\begin{proof}
Since \(a(q_d-1)=2\), translation by \(z\) gives
\[
 \|F_\cdot(z)\|_{q_d}^{q_d}
 =c_d\int_{\RR^d}
 \frac{(1+|u+z|^2)^2}{(1+|u|^2)^{(d+5)/2}}\dd u.
\]
Write \(A=1+|u|^2\).  The odd term in the expansion of
\((A+|z|^2+2u\cdot z)^2\) integrates to zero, and rotational invariance
gives
\[
 \int(u\cdot z)^2(1+|u|^2)^{-(d+5)/2}\dd u
 =\frac{|z|^2}{d}\int |u|^2(1+|u|^2)^{-(d+5)/2}\dd u.
\]
The three normalized beta integrals needed here are
\[
 \begin{aligned}
 c_d\int(1+|u|^2)^{-(d+1)/2}\dd u&=1,\\
 c_d\int(1+|u|^2)^{-(d+3)/2}\dd u&=\frac1{d+1},\\
 c_d\int(1+|u|^2)^{-(d+5)/2}\dd u
 &=\frac3{(d+1)(d+3)}.
 \end{aligned}
\]
Substitution proves \eqref{eq:critical-lq-exact-translation-norm}.

For \(|z|\le1\), the local quotient--Taylor estimate
\eqref{eq:local-multidim-taylor-remainder} implies the first line of
\eqref{eq:critical-lq-global-remainder}.  For \(|z|>1\), the exact identity
gives \(\|F_\cdot(z)\|_{q_d}\le C_d|z|^{4/q_d}=C_d|z|^{p_d}\).
Every coefficient of \(T_m\) is bounded in \(y\), and \(m\le p_d\), so
\(\|T_m(\cdot,z)\|_{q_d}\le C_d|z|^{p_d}\).  Finally,
\(m+1>p_d\), which proves the last inequality for \(|z|\le1\).
\end{proof}

For \(0<\alpha<2\), let \(p_1^{(\alpha,d)}\) be the density of any
centred isotropic strictly \(\alpha\)-stable law on \(\RR^d\), and use the
spatial-scale normalization
\[
 p_s^{(\alpha,d)}(x)=s^{-d}p_1^{(\alpha,d)}(x/s),\qquad s>0.
\]
Changing the scale parameter in the characteristic exponent only changes
the choice of \(p_1^{(\alpha,d)}\), so this notation covers every isotropic
strictly stable kernel.  Put
\begin{equation}\label{eq:stable-critical-exponents}
 \beta=d+\alpha,\qquad
 q_{\alpha,d}:=\min\left\{2,1+\frac4\beta\right\},\qquad
 p_{\alpha,d}:=\frac4{q_{\alpha,d}}
 =\max\left\{2,\frac{4(d+\alpha)}{d+\alpha+4}\right\}.
\end{equation}
Write \(\Omega_{\alpha,d}(\dd y)=p_1^{(\alpha,d)}(y)\dd y\) and
\[
 F_y^{(\alpha,d)}(z)
 :=\frac{p_1^{(\alpha,d)}(y-z)}{p_1^{(\alpha,d)}(y)}.
\]

\begin{lemma}[Stable critical translation remainder]
\label{lem:stable-critical-translation-remainder}
Let \(d\ge1\), \(0<\alpha<2\), \(p=p_{\alpha,d}\),
\(q=q_{\alpha,d}\), and \(m=\lfloor p\rfloor\).  If
\(T_m^{(\alpha,d)}(y,z)\) is the Taylor polynomial of
\(F_y^{(\alpha,d)}(z)\) at \(z=0\) through degree \(m\), then
\begin{equation}\label{eq:stable-translation-lq-growth}
 c_{\alpha,d,q}(1+|z|)^{\beta(q-1)}
 \le \|F_\cdot^{(\alpha,d)}(z)\|_{L^q(\Omega_{\alpha,d})}^q
 \le C_{\alpha,d,q}(1+|z|)^{\beta(q-1)}.
\end{equation}
Moreover,
\begin{equation}\label{eq:stable-critical-global-remainder}
 \|F_\cdot^{(\alpha,d)}(z)-T_m^{(\alpha,d)}(\cdot,z)\|_{L^q(\Omega_{\alpha,d})}
 \le C_{\alpha,d}
 \begin{cases}
  |z|^{m+1},&|z|\le1,\\
  |z|^p,&|z|>1,
 \end{cases}
 \le C_{\alpha,d}|z|^p.
\end{equation}
\end{lemma}

\begin{proof}
The standard stable-kernel estimates in properties (G)(ii)--(iii) of
\cite{IshigeKawakamiMichihisa2017} give, with constants depending on the
chosen stable scale,
\begin{equation}\label{eq:stable-density-derivative-bounds}
 c(1+|y|)^{-\beta}\le p_1^{(\alpha,d)}(y)
 \le C(1+|y|)^{-\beta},\qquad
 |\partial^\eta p_1^{(\alpha,d)}(y)|
 \le C_\eta(1+|y|)^{-\beta-|\eta|}.
\end{equation}
In particular every quotient
\(\partial^\eta p_1^{(\alpha,d)}/p_1^{(\alpha,d)}\) is bounded.  The upper
bound in \eqref{eq:stable-translation-lq-growth} follows from
\[
 \begin{aligned}
 \int p_1(y-z)^q p_1(y)^{1-q}\dd y
 &\le C\int(1+|y-z|)^{-\beta q}
              (1+|y|)^{\beta(q-1)}\dd y\\
 &\le C(1+|z|)^{\beta(q-1)}
       \int(1+|y-z|)^{-\beta}\dd y,
 \end{aligned}
\]
where \(p_1=p_1^{(\alpha,d)}\) and
\(1+|y|\le(1+|z|)(1+|y-z|)\).  The final integral is finite because
\(\beta>d\).  For the lower bound, integrate over \(|y-z|\le1\) and use
\eqref{eq:stable-density-derivative-bounds}; this gives the claimed power
for \(|z|\ge2\).  Positivity and continuity on bounded \(z\)-sets complete
the estimate.

For \(|z|\le1\), \eqref{eq:stable-density-derivative-bounds} also gives
\(p_1(y-\theta z)\le C p_1(y)\), uniformly in \(0\le\theta\le1\).
Taylor's formula along the segment from \(y\) to \(y-z\), followed by the
derivative bound in \eqref{eq:stable-density-derivative-bounds}, therefore
gives the pointwise bound
\(|F_y^{(\alpha,d)}(z)-T_m^{(\alpha,d)}(y,z)|\le C|z|^{m+1}\), uniformly
in \(y\).  For \(|z|>1\), \eqref{eq:stable-translation-lq-growth} gives
\(\|F_\cdot^{(\alpha,d)}(z)\|_q\le C|z|^{\beta(1-1/q)}\).  The Taylor
coefficients are bounded, so
\(\|T_m^{(\alpha,d)}(\cdot,z)\|_q\le C|z|^m\).  Finally,
\[
 \beta(1-1/q)\le p,\qquad m\le p<m+1,
\]
with equality in the first relation when \(\beta>4\).  These observations
prove \eqref{eq:stable-critical-global-remainder}.
\end{proof}

\begin{lemma}[Critical Bregman perturbation bound]
\label{lem:critical-lq-kl-perturbation}
Let \(1<q\le2\) and
\(\Phi(s)=(1+s)\log(1+s)-s\) on \([-1,\infty)\), with
\(0\log0=0\).  There is a constant \(C_q\) such that
\begin{equation}\label{eq:critical-bregman-pointwise-bound}
 0\le \Phi(u+v)-\Phi(u)-\log(1+u)v\le C_q|v|^q
\end{equation}
whenever \(|u|\le1/4\) and \(u+v\ge-1\).
\end{lemma}

\begin{proof}
The lower bound is convexity.  If \(|v|\le1/2\), the segment between
\(u\) and \(u+v\) is contained in \([-3/4,3/4]\).  Since
\(\Phi''(s)=(1+s)^{-1}\) there, Taylor's theorem gives an upper bound
\(C v^2\le C|v|^q\).  If \(v<-1/2\), the admissible pairs
\((u,v)\) range over a compact set and \(|v|^q\ge2^{-q}\); continuity at
\(u+v=-1\) gives the result.  If \(v>1/2\), direct expansion of the left
side shows that it is at most \(C(1+v)\log(1+v)+Cv\), which is bounded by
\(C_qv^q\).  These three regions prove
\eqref{eq:critical-bregman-pointwise-bound}.
\end{proof}

\begin{theorem}[Optimal uniform sufficient moment exponent for stable KL asymptotics]
\label{thm:high-dimensional-kl-moment-threshold}
Let \(d\ge1\), \(0<\alpha<2\), and let \(\nu\) be a probability law on
\(\RR^d\) with mean \(\bar\gamma\).  Put
\(X=\gamma-\bar\gamma\), and assume
\begin{equation}\label{eq:critical-kl-moment-hypothesis}
 \EE|X|^{p_{\alpha,d}}<\infty,
 \qquad
 p_{\alpha,d}=\max\left\{2,
 \frac{4(d+\alpha)}{d+\alpha+4}\right\},
\end{equation}
then the covariance \(\Sigma=\EE[XX^{\mathsf T}]\) is finite and
\begin{equation}\label{eq:critical-kl-leading-asymptotic}
 H_{\alpha,d}(s):=
 D_{\rm KL}\!\left(p_s^{(\alpha,d)}*\nu\,\middle\|\,
 p_s^{(\alpha,d)}(\mathord\cdot-\bar\gamma)\right)
 =\mathcal Q_{\alpha,d}(\Sigma)s^{-4}+o(s^{-4}),
\end{equation}
where
\begin{equation}\label{eq:stable-leading-covariance-functional}
 b_{\alpha,d,\Sigma}(y)
 :=\frac1{2p_1^{(\alpha,d)}(y)}
   \sum_{i,j=1}^d\Sigma_{ij}\partial_{ij}p_1^{(\alpha,d)}(y),
 \qquad
 \mathcal Q_{\alpha,d}(\Sigma)
 :=\frac12\int b_{\alpha,d,\Sigma}^2\dd\Omega_{\alpha,d}.
\end{equation}
The exponent is optimal only in the following uniform sense.  For every
\(0<r<2\), there is a mean-zero law with finite \(r\)th absolute moment
and \(\limsup_{s\to\infty}s^4H_{\alpha,d}(s)=+\infty\).  If
\(d+\alpha>4\), then for every \(2\le r<p_{\alpha,d}\) the counterexample
may additionally be chosen with positive-definite finite covariance.
Thus no smaller moment exponent implies
\eqref{eq:critical-kl-leading-asymptotic} uniformly over its entire moment
class, and the endpoint belongs to the sufficient side.

This is not a law-by-law moment criterion.  When \(d+\alpha>4\), finite
\(p_{\alpha,d}\)-moment is not asserted to be necessary for an individual
law; the exact finite-covariance criterion is the nonlinear raw-tail energy
in Theorem~\ref{thm:stable-law-by-law-decomposition} below.  When
\(d+\alpha\le4\), the exponent is two and boundedness of
\(s^4H_{\alpha,d}(s)\) does force finite covariance, as the proof below
shows.  For \(\alpha=1\) with the standard normalization
\(p_1^{(1,d)}=P_1^{(d)}\), one has
\(\mathcal Q_{1,d}=\mathcal Q_d\), so the theorem contains the radial
Poisson result above.
\end{theorem}

\begin{proof}
Translation permits us to assume \(\bar\gamma=0\).  Fix
\(q=q_{\alpha,d}\), \(p=p_{\alpha,d}\), and \(m=\lfloor p\rfloor\),
and write
\[
 R_m(y,z)=F_y^{(\alpha,d)}(z)-T_m^{(\alpha,d)}(y,z),
 \qquad r_s(y)=\EE R_m(y,X/s).
\]
Minkowski's inequality and
\eqref{eq:stable-critical-global-remainder} give
\[
 s^p\|R_m(\cdot,X/s)\|_q\le C_{\alpha,d}|X|^p.
\]
For each fixed \(X\), the local part of the same estimate and \(m+1>p\)
give
\[
 s^p\|R_m(\cdot,X/s)\|_q
 \le C_{\alpha,d}|X|^{m+1}s^{p-m-1}\longrightarrow0.
\]
Dominated convergence therefore yields
\begin{equation}\label{eq:critical-lq-integrated-remainder}
 \|r_s\|_{L^q(\Omega_{\alpha,d})}=o(s^{-p}).
\end{equation}

Centering removes the linear Taylor layer.  Averaging the remaining layers
gives
\begin{equation}\label{eq:critical-kl-quotient-decomposition}
 \delta_s^{(\alpha,d)}=u_s+r_s,
 \qquad
 u_s=s^{-2}b_{\alpha,d,\Sigma}
       +\mathbf 1_{\{m=3\}}s^{-3}B_3,
\end{equation}
where \(B_3\) is a bounded linear combination of third-order modes.  In the
second case the required third moments exist because \(p\ge3\).  Hence
\(\|u_s\|_\infty=O(s^{-2})\).  The boundedness of all these modes follows
from \eqref{eq:stable-density-derivative-bounds}.

The identity
\(F_y^{(\alpha,d)}(z)\Omega_{\alpha,d}(\dd y)
=p_1^{(\alpha,d)}(y-z)\dd y\) shows that every translated quotient has
integral one.  Moreover,
\(\int|F_y^{(\alpha,d)}(z)-1|\Omega_{\alpha,d}(\dd y)\le2\), so Tonelli's
theorem gives \(\delta_s^{(\alpha,d)}\in L^1(\Omega_{\alpha,d})\) with
zero integral.  The KL change of variables is therefore
\[
 H_{\alpha,d}(s)=\int\Phi(\delta_s^{(\alpha,d)})\dd\Omega_{\alpha,d}
\]
in the extended sense.  For all sufficiently large \(s\), apply
Lemma~\ref{lem:critical-lq-kl-perturbation} to \(u=u_s\), \(v=r_s\).
Hölder's inequality and \eqref{eq:critical-lq-integrated-remainder} give
\begin{align*}
 \left|H_{\alpha,d}(s)-\int\Phi(u_s)\dd\Omega_{\alpha,d}\right|
 &\le \|\log(1+u_s)\|_{q/(q-1)}\|r_s\|_q
       +C_q\|r_s\|_q^q\\
 &\le O(s^{-2})o(s^{-p})+o(s^{-pq})=o(s^{-4}),
\end{align*}
because \(p\ge2\) and \(pq=4\).
This estimate also proves that the extended KL integral is finite for all
sufficiently large \(s\).

Finally, the boundedness of the retained modes permits a uniform Taylor
expansion of \(\Phi(u_s)\).  From
\eqref{eq:critical-kl-quotient-decomposition},
\[
 \int\Phi(u_s)\dd\Omega_{\alpha,d}
 =\frac{s^{-4}}2\int b_{\alpha,d,\Sigma}^2\dd\Omega_{\alpha,d}
   +o(s^{-4})
 =\mathcal Q_{\alpha,d}(\Sigma)s^{-4}+o(s^{-4}),
\]
which proves \eqref{eq:critical-kl-leading-asymptotic}.

We first prove the obstruction below two moments.  If
\(\limsup_{s\to\infty}s^4H_{\alpha,d}(s)<\infty\), Pinsker's inequality
and scaling give, for every fixed \(\xi\in\RR^d\),
\[
 |\widehat p_1^{(\alpha,d)}(\xi)|
 |\varphi_X(\xi/s)-1|
 \le \|p_1^{(\alpha,d)}*\nu_s-p_1^{(\alpha,d)}\|_1
 \le \sqrt{2H_{\alpha,d}(s)}=O(s^{-2}),
\]
where \(\nu_s\) is the law of \(X/s\).  Since the stable characteristic
function never vanishes, take \(\xi=e\) for a unit vector \(e\), multiply
the real part by \(s^2\), and use Fatou's lemma to obtain
\[
 \frac12\EE(e\cdot X)^2
 \le \liminf_{s\to\infty}s^2
       \EE[1-\cos((e\cdot X)/s)]<\infty.
\]
Thus bounded scaled entropy forces finite covariance.  Given \(0<r<2\),
choose \(a\in(\max\{1,r\},2)\) and put symmetric masses proportional to
\(n^{-a-1}\) at \(\pm ne_1\), with the remaining mass at zero.  This law
has mean zero and finite \(r\)th moment but infinite variance, so its scaled
entropy has infinite limsup.

It remains to prove optimality between two and \(p\) when \(\beta>4\).
Fix \(2\le r<p\), choose positive \(a_j\) with \(\sum_ja_j<\infty\), and
put symmetric masses \(w_j/2\) at \(\pm R_je_1\), where
\(w_j=a_jR_j^{-r}\).  Add a sufficiently small bounded symmetric spanning
component.  The resulting law has positive-definite finite covariance and
finite \(r\)th moment.  Set
\[
 s_j=\kappa R_jw_j^{1/\beta},\qquad
 A_j=B(R_je_1,s_j).
\]
The component at \(R_je_1\) gives
\((p_{s_j}^{(\alpha,d)}*\nu)(A_j)\ge c_{\alpha,d}w_j\).  On the other hand,
\eqref{eq:stable-density-derivative-bounds} and \(s_j/R_j\to0\) give
\[
 p_{s_j}^{(\alpha,d)}(A_j)
 \le C_{\alpha,d}(s_j/R_j)^\beta
 =C_{\alpha,d}\kappa^\beta w_j.
\]
Choose \(\kappa\) so that the second Bernoulli mass is a fixed small
fraction of the first.  Data processing to
\(\{A_j,A_j^c\}\) then yields \(H_{\alpha,d}(s_j)\ge c w_j\), and hence
\[
 s_j^4H_{\alpha,d}(s_j)
 \ge cR_j^4w_j^{1+4/\beta}
 =ca_j^{q_{\alpha,d}}R_j^{4-rq_{\alpha,d}}.
\]
Here \(q_{\alpha,d}=1+4/\beta\) and
\(r<p=4/q_{\alpha,d}\), so the exponent of \(R_j\) is positive.
Recursive choice of the radii makes \(s_j\to\infty\) and the last display
unbounded while preserving all moment assertions.
\end{proof}

\subsection{The law-by-law finite-covariance decomposition}

The preceding theorem optimizes over moment classes.  The next theorem gives
the exact law-by-law interface under finite covariance alone.  It separates
the universal quadratic covariance energy from the full nonlinear energy of
the raw tail beyond the smoothing scale.

\begin{theorem}[Stable law-by-law tail decomposition]
\label{thm:stable-law-by-law-decomposition}
Let \(d\ge1\), \(0<\alpha<2\), and let \(\nu\) be a centred probability law
on \(\RR^d\) with finite covariance matrix \(\Sigma\).  Put
\[
 H_{\alpha,d}(s):=
 D_{\rm KL}\!\left(p_s^{(\alpha,d)}*\nu\,\middle\|\,
 p_s^{(\alpha,d)}\right)
\]
and define
\begin{equation}\label{eq:stable-raw-tail-potential}
 V_s^\nu(y):=\int_{\{|x|>s\}}
 \frac{p_1^{(\alpha,d)}(y-x/s)}
      {p_1^{(\alpha,d)}(y)}\,\nu(\dd x).
\end{equation}
Then
\begin{equation}\label{eq:stable-law-by-law-decomposition}
 H_{\alpha,d}(s)=\mathcal Q_{\alpha,d}(\Sigma)s^{-4}
 +\int_{\RR^d}\Phi(V_s^\nu(y))\,\Omega_{\alpha,d}(\dd y)
 +o(s^{-4}).
\end{equation}
Consequently,
\begin{equation}\label{eq:stable-law-by-law-criterion}
 s^4H_{\alpha,d}(s)\longrightarrow\mathcal Q_{\alpha,d}(\Sigma)
 \quad\Longleftrightarrow\quad
 s^4\int_{\RR^d}\Phi(V_s^\nu)\dd\Omega_{\alpha,d}\longrightarrow0.
\end{equation}
No regular-variation, absolute-continuity, or moment hypothesis above order two
is imposed.
\end{theorem}

\begin{proof}
For brevity write
\[
 p_1=p_1^{(\alpha,d)},\qquad
 F_y(z)=F_y^{(\alpha,d)}(z)
 =\frac{p_1(y-z)}{p_1(y)}.
\]
Let \(\ell_y(z)\) and \(B_2(y,z)\) denote the homogeneous first- and
second-order Taylor terms of \(F_y(z)\) at zero.  The estimates
\eqref{eq:stable-density-derivative-bounds} and Taylor's formula give, for
\(|z|\le1\),
\begin{equation}\label{eq:stable-local-taylor-modulus}
 \|F_\cdot(z)-1-\ell_\cdot(z)-B_2(\cdot,z)\|_\infty
 +\|F_\cdot(z)-1-\ell_\cdot(z)-B_2(\cdot,z)\|_{L^1(\Omega_{\alpha,d})}
 \le\varepsilon_{\alpha,d}(|z|)|z|^2,
\end{equation}
where \(\varepsilon_{\alpha,d}\) is bounded on \([0,1]\) and tends to zero
at zero.  The uniform term is in fact \(O(|z|^3)\).  For the \(L^1\)
term, multiplication by \(p_1(y)\) turns it into the second-order
translation remainder of \(p_1\).  Since \(D^2p_1\in L^1\), the integral
form of Taylor's theorem and translation continuity in \(L^1\) give
\[
 \int_0^1(1-\theta)|z|^2
 \|D^2p_1(\mathord\cdot-\theta z)-D^2p_1\|_1\dd\theta=o(|z|^2).
\]
The Taylor coefficients also satisfy
\begin{equation}\label{eq:stable-taylor-coefficient-norms}
 \|B_j(\cdot,z)\|_\infty
 +\|B_j(\cdot,z)\|_{L^1(\Omega_{\alpha,d})}
 \le C|z|^j,\qquad j=1,2,\quad B_1=\ell.
\end{equation}

Write
\[
 \delta_s(y)=\int\{F_y(x/s)-1\}\nu(\dd x),\qquad
 u_s=s^{-2}b_{\alpha,d,\Sigma},\qquad
 \tau_s=\nu(|x|>s).
\]
Finite covariance implies
\begin{equation}\label{eq:stable-tail-mass-smallness}
 \tau_s\le s^{-2}\int_{|x|>s}|x|^2\nu(\dd x)=o(s^{-2}).
\end{equation}
Centering and the identity
\(\int B_2(y,x)\nu(\dd x)=b_{\alpha,d,\Sigma}(y)\) give the exact split
\begin{equation}\label{eq:stable-raw-tail-split}
 \delta_s=u_s+V_s^\nu-\tau_s+R_s,
\end{equation}
where
\[
 R_s=\int_{|x|\le s}\!
 \{F_\cdot(x/s)-1-\ell_\cdot(x/s)-B_2(\cdot,x/s)\}\nu(\dd x)
 -\int_{|x|>s}\!\{\ell_\cdot(x/s)+B_2(\cdot,x/s)\}\nu(\dd x).
\]
Equations \eqref{eq:stable-local-taylor-modulus} and
\eqref{eq:stable-taylor-coefficient-norms}, followed by dominated
convergence after multiplication by \(s^2\), show that
\begin{equation}\label{eq:stable-raw-tail-remainder-smallness}
 \|R_s\|_\infty+\|R_s\|_{L^1(\Omega_{\alpha,d})}=o(s^{-2}).
\end{equation}
For the omitted tail modes one uses
\[
 s^{-1}\int_{|x|>s}|x|\nu(\dd x)
 \le s^{-2}\int_{|x|>s}|x|^2\nu(\dd x)
\]
and the analogous quadratic bound.

The translation identity for \(F_y\) gives the exact normalization
\begin{equation}\label{eq:stable-tail-potential-normalization}
 V_s^\nu\ge0,\qquad
 \int V_s^\nu\dd\Omega_{\alpha,d}=\tau_s.
\end{equation}
All entropy integrals used below are finite.  Indeed,
\eqref{eq:stable-density-derivative-bounds} and
\(1+|w+z|\le(1+|w|)(1+|z|)\) give
\[
 D_{\rm KL}\!\left(p_1(\mathord\cdot-z)\,\middle\|\,p_1\right)
 \le C_{\alpha,d}+(d+\alpha)\log(1+|z|).
\]
Convexity of relative entropy and \(\EE|X|^2<\infty\) therefore make
\(H_{\alpha,d}(s)\) finite.  If \(\tau_s>0\), write
\(V_s^\nu=\tau_s q_s\), where \(q_s\) is the quotient of the normalized
tail mixture by \(p_1\).  Convexity gives
\[
 \int q_s\log q_s\dd\Omega_{\alpha,d}
 \le \frac1{\tau_s}\int_{|x|>s}
 \left\{C_{\alpha,d}+(d+\alpha)\log(1+|x|/s)\right\}\nu(\dd x)<\infty.
\]
Since \(\Phi(\tau q)\le\tau q\log(1+q)\) for \(0\le\tau\le1\), and the
negative part of \(q\log q\) is bounded by \(e^{-1}\), this also proves
\(\int\Phi(V_s^\nu)\dd\Omega_{\alpha,d}<\infty\).  The case \(\tau_s=0\)
is immediate.

For \(|u|\le1/4\) and \(v\ge0\), two integrations of
\(\Phi''(w)=(1+w)^{-1}\) yield
\begin{equation}\label{eq:stable-phi-interaction}
 |\Phi(u+v)-\Phi(u)-\Phi(v)|\le C|u|v.
\end{equation}
All negative terms in \eqref{eq:stable-raw-tail-split} are uniformly larger
than \(-1/2\) for large \(s\).  The mean-value theorem,
\(\log(1+v)\le v\), and
\eqref{eq:stable-tail-potential-normalization} therefore give
\begin{align}
 \int|\Phi(\delta_s)-\Phi(u_s+V_s^\nu-\tau_s)|
       \dd\Omega_{\alpha,d}&=o(s^{-4}),\label{eq:stable-remove-remainder}\\
 \int|\Phi(u_s+V_s^\nu-\tau_s)-\Phi(u_s+V_s^\nu)|
       \dd\Omega_{\alpha,d}&=o(s^{-4}),\label{eq:stable-remove-mass}\\
 \int|\Phi(u_s+V_s^\nu)-\Phi(u_s)-\Phi(V_s^\nu)|
       \dd\Omega_{\alpha,d}&=o(s^{-4}).\label{eq:stable-remove-interaction}
\end{align}
Indeed, the three left sides are bounded, respectively, by
\[
 C\|R_s\|_\infty\tau_s+Cs^{-2}\|R_s\|_1
 +C\|R_s\|_\infty\|R_s\|_1,
 \quad C\tau_s(\tau_s+s^{-2}),
 \quad C\|u_s\|_\infty\tau_s,
\]
and each is \(o(s^{-4})\).

Finally, integration of the second derivatives in
\eqref{eq:stable-leading-covariance-functional} gives
\(\int b_{\alpha,d,\Sigma}\dd\Omega_{\alpha,d}=0\).  Boundedness of the
covariance mode then implies
\begin{equation}\label{eq:stable-covariance-phi-expansion}
 \int\Phi(u_s)\dd\Omega_{\alpha,d}
 =\frac{s^{-4}}2\int b_{\alpha,d,\Sigma}^2\dd\Omega_{\alpha,d}
 +O(s^{-6})
 =\mathcal Q_{\alpha,d}(\Sigma)s^{-4}+O(s^{-6}).
\end{equation}
Since \(H_{\alpha,d}(s)=\int\Phi(\delta_s)\dd\Omega_{\alpha,d}\),
equations \eqref{eq:stable-remove-remainder}--
\eqref{eq:stable-covariance-phi-expansion} prove
\eqref{eq:stable-law-by-law-decomposition}.  The equivalence
\eqref{eq:stable-law-by-law-criterion} follows because the tail energy is
nonnegative.
\end{proof}

\begin{remark}[What the decomposition resolves]
When \(d+\alpha\le4\), one has \(p_{\alpha,d}=2\), so every
finite-covariance law makes the tail energy \(o(s^{-4})\) by
Theorem~\ref{thm:high-dimensional-kl-moment-threshold}.  When
\(d+\alpha>4\), finite covariance alone need not do so.  In that regime
\eqref{eq:stable-law-by-law-criterion} is the exact missing law-by-law
condition: it neither replaces the nonlinear tail energy by a moment nor
assumes regular variation.
\end{remark}
