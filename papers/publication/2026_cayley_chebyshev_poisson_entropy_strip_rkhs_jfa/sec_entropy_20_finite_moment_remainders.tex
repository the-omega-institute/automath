\subsection{Finite-moment entropy asymptotics and remainders}%
\label{subsec:finite-moment-entropy}

\subsubsection{Density-to-entropy passage}

This subsection passes from formal coefficient algebra to genuine entropy
asymptotics.  The density-to-entropy passage is a finite operation at fixed
order $2N$.
First the density ratio is written as a finite mode polynomial plus an
$L^\infty\cap L^1(\omega)$ error.  Then
$\Phi(z)=(1+z)\log(1+z)-z$ is expanded only through degree~$N$, because the
density perturbation begins at order $t^{-2}$.  Thus a product containing one
density error also contains the entropy quadratic order and is two powers
smaller than the density error itself; products with more density-error
factors are smaller still.  The displayed two-scale constants below are
convenience envelopes for this refined product hierarchy and for the
high-degree finite-mode tail.  After integration, the remaining finite
polynomial is exactly the universal coefficient polynomial.

\begin{lemma}[Product hierarchy for the density remainder]%
\label{lem:density-remainder-product-hierarchy}
Let $p>2$, $2\le r$, and let $A,\rho$ be measurable functions satisfying
\[
\|A\|_{L^\infty(\RR)}\le M t^{-2},
\qquad
\|\rho\|_{L^\infty(\RR)}+\|\rho\|_{L^1(\omega)}
\le C t^{-p}
\]
for some $t\ge1$.  Then
\begin{equation}\label{eq:density-remainder-product-hierarchy}
\|(A+\rho)^r-A^r\|_{L^1(\omega)}
\le
\sum_{\ell=1}^{r}\binom r\ell
M^{r-\ell}C^\ell
t^{-\{\ell p+2(r-\ell)\}} .
\end{equation}
In particular, since $p>2$,
\begin{equation}\label{eq:density-remainder-product-envelope}
\|(A+\rho)^r-A^r\|_{L^1(\omega)}
\le
\sum_{\ell=1}^{r}\binom r\ell
M^{r-\ell}C^\ell
t^{-(p+2)} .
\end{equation}
\end{lemma}

\begin{proof}
Expand
\[
(A+\rho)^r-A^r
=\sum_{\ell=1}^r\binom r\ell A^{r-\ell}\rho^\ell .
\]
For a fixed $\ell$, put one factor of $\rho$ in $L^1(\omega)$ and all other
factors in $L^\infty$.  This gives
\[
\|A^{r-\ell}\rho^\ell\|_{L^1(\omega)}
\le
M^{r-\ell}C^\ell
t^{-2(r-\ell)}t^{-\ell p},
\]
which is \eqref{eq:density-remainder-product-hierarchy}.  The exponent
$\ell p+2(r-\ell)$ is minimized at $\ell=1$ and $r=2$ when $p>2$ and
$r\ge2$, giving the envelope \eqref{eq:density-remainder-product-envelope}.
\end{proof}

\begin{lemma}[Exact-moment density truncation with little-oh remainder]%
\label{lem:exact-moment-density-truncation}
Let \(L\ge2\), and let \(\nu\) be a probability measure on \(\RR\) with
mean \(\bar\gamma\) and finite centred \(L\)th absolute moment.  Put
\(\Delta=\gamma-\bar\gamma\) and
\[
\delta_t(\bar\gamma+ty)
:=
\frac{(P_t*\nu)(\bar\gamma+ty)}{P_t(ty)}-1 .
\]
Define
\[
\rho_{L,t}(y):=
\int_{\RR}\left(
R_{\Delta/t}(y)-\sum_{n=0}^{L}u_n(y)(\Delta/t)^n
\right)\dd\nu(\gamma).
\]
Then, with \(\mu_n=\int\Delta^n\dd\nu(\gamma)\) for \(0\le n\le L\),
\begin{equation}\label{eq:exact-moment-density-truncation}
\delta_t(\bar\gamma+ty)
=
\sum_{n=2}^{L}\mu_nu_n(y)t^{-n}+\rho_{L,t}(y),
\qquad
\|\rho_{L,t}\|_{\infty}+\|\rho_{L,t}\|_{L^1(\omega)}=o(t^{-L}).
\end{equation}
Moreover \(\sup_y|\delta_t(\bar\gamma+ty)|=O(t^{-2})\).
In particular, when \(L=2N-2\), this is the exact little-oh density input
used for the coefficient theorem through order \(t^{-2N}\); no
positive-margin moment is being assumed in this lemma.
\end{lemma}

\begin{proof}
The quotient normalization is the one used throughout this section:
\[
\frac{(P_t*\nu)(\bar\gamma+ty)}{P_t(ty)}
=\int_{\RR}R_{\Delta/t}(y)\dd\nu(\gamma).
\]
We prove the \(L^\infty\) and \(L^1(\omega)\) remainders by the same split.
On \(|\Delta|\le t\), Proposition~\ref{prop:cayley-mode-remainder} gives the
exact finite Taylor remainder
\[
R_{\Delta/t}(y)-\sum_{n=0}^{L}u_n(y)(\Delta/t)^n
=\left(\frac{\Delta}{t}\right)^{L+1}v_L(y,\Delta/t).
\]
With \(V_L=\|v_L\|_\infty<\infty\), uniformly in \(y\),
\[
\left|
\int_{|\Delta|\le t}\left(
R_{\Delta/t}(y)-\sum_{n=0}^{L}u_n(y)(\Delta/t)^n
\right)\dd\nu
\right|
\le
V_Lt^{-L}\EE\!\left[
|\Delta|^L\frac{|\Delta|}{t}\mathbf1_{\{|\Delta|\le t\}}
\right].
\]
The expectation tends to zero by dominated convergence, since the multiplier
\((|\Delta|/t)\mathbf1_{\{|\Delta|\le t\}}\) is bounded by \(1\) and converges
pointwise to \(0\).

On \(|\Delta|>t\), put \(\varepsilon=\Delta/t\).  The large-\(\varepsilon\)
estimate used in Theorem~\ref{thm:finite-moment-density-remainder} gives,
for \(L\ge2\),
\[
R_{\varepsilon}(y)\le 3(1+\varepsilon^2)\le6|\varepsilon|^L,
\qquad
\left|\sum_{n=0}^{L}u_n(y)\varepsilon^n\right|
\le (L+1)B_L|\varepsilon|^L
\]
whenever \(|\varepsilon|>1\), where
\(B_L=\max_{0\le n\le L}\|u_n\|_\infty\).  Hence, uniformly in \(y\),
\[
\left|
\int_{|\Delta|>t}\left(
R_{\Delta/t}(y)-\sum_{n=0}^{L}u_n(y)(\Delta/t)^n
\right)\dd\nu
\right|
\le
\{6+(L+1)B_L\}t^{-L}\EE[
|\Delta|^L\mathbf1_{\{|\Delta|>t\}}],
\]
which is \(o(t^{-L})\).  Combining the two pieces gives
\(\|\rho_{L,t}\|_\infty=o(t^{-L})\).  Since \(\omega\) is a probability
measure and the same pointwise majorants dominate the \(y\)-integral, the
identical estimates give \(\|\rho_{L,t}\|_{L^1(\omega)}=o(t^{-L})\).

The constant term cancels and the \(n=1\) term integrates to zero because
\(\mu_1=0\), giving \eqref{eq:exact-moment-density-truncation}.

Finally, Lemma~\ref{lem:tail-uniform-cayley-mode-remainder} with \(N=1\)
applies because \(L\ge2\) implies \(\EE|\Delta|^2<\infty\).  Since
\(\mu_1=0\), it gives
\(\sup_y|\delta_t(\bar\gamma+ty)|=O(t^{-2})\).
\end{proof}

\begin{proposition}[Finite-moment density-to-entropy transfer]%
\label{prop:density-to-entropy-descent}%
\label{lem:density-to-entropy-domination}%
\label{lem:explicit-entropy-constant-bound}
Let $N\ge2$, $0<\eta\le1$, and let
\[
\omega(\dd y)=\frac{\dd y}{\pi(1+y^2)}.
\]
Let $u_2,\ldots,u_{2N}$ be bounded measurable functions on~$\RR$, and let
$\mu_2,\ldots,\mu_{2N}$ be real numbers.  For
$2\le r\le N$ and $2\le n_j\le2N$, put
\[
I(n_1,\ldots,n_r)
:=\int_{\RR}u_{n_1}(y)\cdots u_{n_r}(y)\,\omega(\dd y),
\]
which is absolutely convergent because $\omega(\RR)=1$ and the modes are
bounded.  For $4\le m\le2N$ define the finite polynomial coefficient
\begin{equation}\label{eq:density-to-entropy-descent-coefficient}
\mathcal A_m(\mu,u)
:=
\sum_{r=2}^{\min\{N,\lfloor m/2\rfloor\}}
\frac{(-1)^r}{r(r-1)}
\sum_{\substack{2\le n_1,\ldots,n_r\le2N\\
n_1+\cdots+n_r=m}}
\mu_{n_1}\cdots\mu_{n_r}\,I(n_1,\ldots,n_r).
\end{equation}
Assume that for every $t\ge T$ there are measurable functions
$\delta_t,A_t,\rho_t$ such that
\begin{equation}\label{eq:density-to-entropy-descent-density}
\delta_t=A_t+\rho_t,\qquad
A_t(y)=\sum_{n=2}^{2N}\mu_nu_n(y)t^{-n},
\end{equation}
and that the following are the only analytic hypotheses:
\begin{equation}\label{eq:density-to-entropy-descent-hypotheses}
\|\rho_t\|_{L^\infty(\RR)}+\|\rho_t\|_{L^1(\omega)}
\le C_\rho t^{-(2N+\eta)},
\end{equation}
\[
\|\delta_t\|_{L^\infty(\RR)}\le\min\{1/2,S t^{-2}\},
\]
for fixed finite constants $C_\rho,S$.  Put
\[
M_A:=\sum_{n=2}^{2N}|\mu_n|\,\|u_n\|_{L^\infty(\RR)} .
\]
Then $\|A_t\|_{L^\infty(\RR)}\le M_A t^{-2}$ for $t\ge1$,
$\Phi(\delta_t)\in L^1(\omega)$ for
\[
\Phi(z)=(1+z)\log(1+z)-z,
\]
and, for every $t\ge\max\{1,T\}$,
\begin{equation}\label{eq:density-to-entropy-descent-expansion}
\int_{\RR}\Phi(\delta_t(y))\,\omega(\dd y)
=\sum_{m=4}^{2N}\mathcal A_m(\mu,u)t^{-m}
+\mathcal E_{N,t}.
\end{equation}
The remainder is controlled by the finite computable bound
\begin{equation}\label{eq:density-to-entropy-descent-remainder}
|\mathcal E_{N,t}|
\le
\mathfrak P_N(C_\rho,M_A)t^{-(2N+\eta+2)}
+\mathfrak Q_N(\mu,u)t^{-(2N+1)}
+2S^{N+1}t^{-(2N+2)},
\end{equation}
where
\begin{equation}\label{eq:entropy-constant-product-error}
\mathfrak P_N(C_\rho,M_A)
:=
\sum_{r=2}^{N}\frac{1}{r(r-1)}
\sum_{\ell=1}^{r}\binom r\ell M_A^{r-\ell}C_\rho^\ell
\end{equation}
and
\begin{equation}\label{eq:entropy-constant-truncation-error}
\mathfrak Q_N(\mu,u)
:=
\sum_{r=2}^{N}\frac{1}{r(r-1)}
\sum_{m=2N+1}^{2Nr}
\sum_{\substack{2\le n_1,\ldots,n_r\le2N\\
n_1+\cdots+n_r=m}}
|\mu_{n_1}\cdots\mu_{n_r}|\,
\|u_{n_1}\cdots u_{n_r}\|_{L^1(\omega)} .
\end{equation}
Thus the descent from an $L^\infty\cap L^1(\omega)$ density-ratio remainder to
the entropy integral is a finite calculation.  The finite
$(2N+\eta)$-moment input is used only before this proposition, to produce
\eqref{eq:density-to-entropy-descent-density} and the
$L^\infty\cap L^1(\omega)$ bound
\eqref{eq:density-to-entropy-descent-hypotheses}.  Inside the transfer,
Lemma~\ref{lem:density-remainder-product-hierarchy} shows that every product
containing at least one $\rho_t$ is in fact $O(t^{-(2N+\eta+2)})$ after the
entropy quadratic order is used.  The term
$\mathfrak Q_N(\mu,u)t^{-(2N+1)}$ is not a density-remainder loss: it is the
finite high-degree tail of the retained mode polynomial.  Since
$0<\eta\le1$, the coarser envelope
\begin{equation}\label{eq:density-to-entropy-convenience-envelope}
|\mathcal E_{N,t}|
\le
\bigl(\mathfrak P_N(C_\rho,M_A)+\mathfrak Q_N(\mu,u)\bigr)t^{-(2N+\eta)}
+2S^{N+1}t^{-(2N+2)}
\end{equation}
also follows for $t\ge1$.  Later displayed constants using
$C_{N,\eta}^{\rm dens}$ are this convenience envelope, not sharp product
constants.

In the Poisson finite-moment application of
Theorem~\ref{thm:finite-moment-density-remainder} with $L=2N$, where
$C_\rho=C_{\nu,2N,\eta}$ and $S=S_{\nu,2N,\eta}$, the fixed-law constants may
therefore be chosen as
\[
T_{N,\eta}^{\rm ent}(\nu):=\max\{1,T_{\nu,2N,\eta}\},
\]
and
\begin{equation}\label{eq:explicit-entropy-constant-bound}
C_{N,\eta}^{\rm dens}(\nu)
=\mathfrak P_N(C_{\nu,2N,\eta},M_A)+\mathfrak Q_N(\mu,u),
\qquad
C_{N,\eta}^{\rm Tay}(\nu)=2S_{\nu,2N,\eta}^{N+1}.
\end{equation}
The refined fixed-law bound supplied by
\eqref{eq:density-to-entropy-descent-remainder} is
\begin{equation}\label{eq:explicit-entropy-constant-refined-bound}
\mathfrak P_N(C_{\nu,2N,\eta},M_A)t^{-(2N+\eta+2)}
+\mathfrak Q_N(\mu,u)t^{-(2N+1)}
+C_{N,\eta}^{\rm Tay}(\nu)t^{-(2N+2)}.
\end{equation}
These are finite expressions in the density-remainder constant from
Theorem~\ref{thm:finite-moment-density-remainder}, the moments through order
$2N$, and the finite mode-product integrals.  If one wants a purely numerical
envelope, or the coarser two-term form
\eqref{eq:density-to-entropy-convenience-envelope},
Lemma~\ref{lem:closed-cayley-mode-constant-envelopes} gives
$\|u_n\|_\infty\le n+1$ and hence
\[
\|u_{n_1}\cdots u_{n_r}\|_{L^1(\omega)}
\le\prod_{j=1}^{r}(n_j+1).
\]
The proposition records usable bounds for each specified law; it does not
assert sharpness or uniform control over a moment class.
\end{proposition}

\begin{proof}
Since $\omega$ has total mass one and the functions $u_n$ are bounded, every
finite product in \eqref{eq:density-to-entropy-descent-coefficient} belongs to
$L^1(\omega)$.  Thus the displayed coefficient polynomial consists of
absolutely convergent integrals and finite sums.
For $t\ge1$ the finite-mode part satisfies
\[
\|A_t\|_\infty
\le\sum_{n=2}^{2N}|\mu_n|\,\|u_n\|_\infty t^{-n}
\le M_A t^{-2}.
\]

For $|z|\le1/2$,
\[
\Phi(z)=\sum_{r=2}^{N}\frac{(-1)^r}{r(r-1)}z^r+R_{N+1}(z),
\qquad |R_{N+1}(z)|\le2|z|^{N+1}.
\]
The smallness assumption on $\delta_t$ therefore implies
$\Phi(\delta_t)\in L^1(\omega)$ and
\[
\|R_{N+1}(\delta_t)\|_{L^1(\omega)}
\le2S^{N+1}t^{-(2N+2)}.
\]
For each $2\le r\le N$, apply
Lemma~\ref{lem:density-remainder-product-hierarchy} with
$p=2N+\eta$, $M=M_A$, and $C=C_\rho$.  Since $2N+\eta>2$, this gives
\[
\|\delta_t^r-A_t^r\|_{L^1(\omega)}
\le
\sum_{\ell=1}^{r}\binom r\ell
M_A^{r-\ell}C_\rho^\ell t^{-(2N+\eta+2)}.
\]
After multiplication by $1/(r(r-1))$ and summation in $r$, this gives the
$\mathfrak P_N(C_\rho,M_A)t^{-(2N+\eta+2)}$ part of the refined remainder.

It remains only to expand the finite polynomial $A_t^r$.  The exact identity
\[
\int_{\RR}A_t(y)^r\,\omega(\dd y)
=
\sum_{m=2r}^{2Nr}t^{-m}
\sum_{\substack{2\le n_1,\ldots,n_r\le2N\\
n_1+\cdots+n_r=m}}
\mu_{n_1}\cdots\mu_{n_r}\,I(n_1,\ldots,n_r)
\]
splits into the retained part $m\le2N$ and the discarded part $m>2N$.  The
retained part, summed over $2\le r\le N$ with Taylor coefficients, is exactly
$\sum_{m=4}^{2N}\mathcal A_m(\mu,u)t^{-m}$.  The discarded part is bounded in
absolute value by $\mathfrak Q_N(\mu,u)t^{-(2N+1)}$.  Combining this
high-degree polynomial tail with the product error and the Taylor tail proves
\eqref{eq:density-to-entropy-descent-expansion} and
\eqref{eq:density-to-entropy-descent-remainder}.
\end{proof}

\begin{lemma}[Closed moment-ball domination of density and entropy constants]%
\label{lem:moment-ball-uniform-constant-domination}
Fix $N\ge2$, $0<\eta\le1$, and $M<\infty$, and put $p=2N+\eta$.  There are
finite constants
\[
\overline C_{N,\eta,M}^{\rho},\qquad
\overline S_{N,\eta,M},
\qquad
\overline C_{N,\eta,M}^{\rm dens},
\qquad
\overline C_{N,\eta,M}^{\rm Tay},
\qquad
\overline T_{N,\eta,M}^{\rm ent}
\]
given by closed finite formulas in $N,\eta,M$ below, such that every
probability measure $\nu$ with finite mean $\bar\gamma$ and
\[
\int_{\RR}|\gamma-\bar\gamma|^p\dd\nu(\gamma)\le M
\]
satisfies
\begin{align*}
C_{\nu,2N,\eta}&\le \overline C_{N,\eta,M}^{\rho},
&
S_{\nu,2N,\eta}&\le \overline S_{N,\eta,M},
\\
C_{N,\eta}^{\rm dens}(\nu)&\le
\overline C_{N,\eta,M}^{\rm dens},
&
C_{N,\eta}^{\rm Tay}(\nu)&\le
\overline C_{N,\eta,M}^{\rm Tay},
&
T_{N,\eta}^{\rm ent}(\nu)&\le
\overline T_{N,\eta,M}^{\rm ent}.
\end{align*}
Consequently the density remainder
\eqref{eq:finite-moment-density-remainder}, the density smallness threshold
\eqref{eq:finite-moment-density-smallness}, and the entropy constants in
\eqref{eq:explicit-entropy-constant-bound} are uniformly dominated on the
centred $p$-moment ball by constants involving no unspecified mode norms.
\end{lemma}

\begin{proof}
Let
\[
\widehat B:=2N+1,\qquad
\widehat C:=\max\!\left\{\frac{10N+7}{2},\,6+(2N+1)^2\right\}.
\]
These are the closed envelopes for $B_{2N}$ and $C_{2N}$ from
Lemma~\ref{lem:closed-cayley-mode-constant-envelopes}.  Define
\[
\overline C_{N,\eta,M}^{\rho}:=2\widehat C M,
\qquad
\overline S_{N,\eta,M}:=
\widehat B\sum_{n=2}^{2N}M^{n/p}+2\widehat C M,
\]
\[
\overline M_{A,N,\eta,M}:=\sum_{n=2}^{2N}(n+1)M^{n/p}.
\]
For the finite high-degree mode tail set
\[
\overline{\mathfrak Q}_{N,\eta,M}:=
\sum_{r=2}^{N}\frac{1}{r(r-1)}
\sum_{m=2N+1}^{2Nr}
\sum_{\substack{2\le n_1,\dots,n_r\le2N\\
n_1+\cdots+n_r=m}}
\left(\prod_{j=1}^{r}M^{n_j/p}(n_j+1)\right),
\]
and set
\[
\overline{\mathfrak P}_{N,\eta,M}:=
\sum_{r=2}^{N}\frac{1}{r(r-1)}
\sum_{\ell=1}^{r}\binom r\ell
(\overline M_{A,N,\eta,M})^{r-\ell}
(\overline C_{N,\eta,M}^{\rho})^\ell .
\]
Finally define
\[
\overline C_{N,\eta,M}^{\rm dens}:=
\overline{\mathfrak P}_{N,\eta,M}
+\overline{\mathfrak Q}_{N,\eta,M},
\qquad
\overline C_{N,\eta,M}^{\rm Tay}:=
2(\overline S_{N,\eta,M})^{N+1},
\]
\[
\overline T_{N,\eta,M}^{\rm ent}:=
\max\{1,(2\overline S_{N,\eta,M})^{1/2}\}.
\]

For $2\le n\le2N$, Lyapunov's inequality gives
$|\mu_n|\le M^{n/p}$.  The density theorem with $L=2N$ and
Lemma~\ref{lem:closed-cayley-mode-constant-envelopes} give the closed
constant bound
\[
C_{\nu,2N,\eta}=2C_{2N}\EE|\Delta|^p\le2\widehat C M
=\overline C_{N,\eta,M}^{\rho},
\]
and since
\[
S_{\nu,2N,\eta}=B_{2N}\sum_{n=2}^{2N}|\mu_n|+C_{\nu,2N,\eta},
\]
the same moment bounds give
\[
\overline S_{N,\eta,M}:=
\widehat B\sum_{n=2}^{2N}M^{n/p}+2\widehat C M .
\]
The threshold definition then yields
\[
T_{N,\eta}^{\rm ent}(\nu)=\max\{1,T_{\nu,2N,\eta}\}
\le \max\{1,(2\overline S_{N,\eta,M})^{1/2}\}
=:\overline T_{N,\eta,M}^{\rm ent}.
\]

It remains only to dominate the finite entropy sums in
Proposition~\ref{prop:density-to-entropy-descent}.  The closed mode bound
$\|u_n\|_\infty\le n+1$ gives
$M_A\le\overline M_{A,N,\eta,M}$.  Substituting this and
$C_\rho\le\overline C_{N,\eta,M}^{\rho}$ into the finite sum
\eqref{eq:entropy-constant-product-error} gives
$\mathfrak P_N(C_{\nu,2N,\eta},M_A)
\le\overline{\mathfrak P}_{N,\eta,M}$.
Substituting the bounds
$|\mu_{n_j}|\le M^{n_j/p}$ and
\[
\|u_{n_1}\cdots u_{n_r}\|_{L^1(\omega)}
\le \prod_{j=1}^{r}\|u_{n_j}\|_\infty
\le \prod_{j=1}^{r}(n_j+1),
\]
because $\omega$ has total mass one, gives
$\mathfrak Q_N(\mu,u)\le\overline{\mathfrak Q}_{N,\eta,M}$.  Hence
$C_{N,\eta}^{\rm dens}(\nu)\le
\overline C_{N,\eta,M}^{\rm dens}$.  Finally
$C_{N,\eta}^{\rm Tay}(\nu)=2S_{\nu,2N,\eta}^{N+1}$ is bounded by
$\overline C_{N,\eta,M}^{\rm Tay}$.  All constants are therefore controlled
by the displayed finite formulas in $N,\eta,M$ alone.
\end{proof}

\begin{theorem}[Exact-threshold coefficient theorem for Poisson entropy]%
\label{thm:poisson-kl-even-orders}
Let $N\ge2$ and let $\nu$ be a probability measure on~$\RR$ with mean
$\bar\gamma$ and finite centred \((2N-2)\)th absolute moment.  Set
\[
\Delta:=\gamma-\bar\gamma,
\qquad
\mu_n:=\int_{\RR}\Delta^n\dd\nu(\gamma)\quad\text{whenever it exists},
\]
and define
\[
h_t:=P_t*\nu,
\qquad
g_t(x):=P_t(x-\bar\gamma).
\]
No positive-variance hypothesis is imposed in this finite-order finite-moment
statement; when the variance is zero the displayed coefficients vanish and
$h_t=g_t$.
Write \(A_k^{\rm alg}(\nu)\) for the finite-moment evaluations defined in
\eqref{eq:universal-entropy-coefficient-evaluation}.  The theorem identifies
these algebraic evaluations with the genuine large-time entropy coefficients
through order \(t^{-2N}\):
\begin{equation}\label{eq:poisson-kl-even-orders}
D_{\mathrm{KL}}(h_t\|g_t)
=\sum_{k=4}^{2N}A_k^{\rm alg}(\nu)t^{-k}+o(t^{-2N})
=\sum_{m=2}^{N}A_{2m}^{\rm alg}(\nu)t^{-2m}+o(t^{-2N})
\qquad(t\to\infty).
\end{equation}
After this identification we also write \(A_k^{\rm ent}(\nu)=A_k^{\rm alg}(\nu)\)
for these retained coefficients.
This is the sharp coefficient theorem at the fixed-law moment threshold:
the \(+\eta\) moment hypothesis is reserved below for quantitative
remainders, not for the existence of the coefficients in
\eqref{eq:poisson-kl-even-orders}.
Every entropy monomial contributing to \(A_{2N}^{\rm form}\) has at least
two density factors and therefore involves centred moments only through
order \(2N-2\).  This is the exact fixed-law moment threshold for the
universal coefficient through order \(t^{-2N}\): with \(L=2N-2\), the
finite \(L\)th absolute moment gives an
\(L^\infty\cap L^1(\omega)\) density remainder \(o(t^{-L})\), and every
entropy term containing that remainder also contains the quadratic entropy
scale.  Thus the first discarded density-remainder contribution is
\(o(t^{-(L+2)})=o(t^{-2N})\), and no coefficient through order \(t^{-2N}\)
is changed.  The next theorem supplies explicit
rates and constants when a finite \((2N-2+\eta)\)-moment is available; the
stronger \((2N+\eta)\)-moment estimates in
Corollaries~\ref{cor:finite-moment-entropy-constant-remainder} and
\ref{cor:uniform-moment-ball-entropy-remainder} are convenient sufficient
forms.
Equivalently, no odd power occurs in the expansion through order~$t^{-2N}$.
\end{theorem}


\begin{proof}
Put
\[
L:=2N-2 .
\]
We first fix the normalization in the rescaled variable.  For
$x=\bar\gamma+ty$,
\[
\frac{P_t(x-\gamma)}{g_t(x)}
=
\frac{t^2+(x-\bar\gamma)^2}{t^2+(x-\gamma)^2}
=
\frac{1+y^2}{1+(y-\Delta/t)^2}
=R_{\Delta/t}(y),
\]
where $\Delta=\gamma-\bar\gamma$.  Hence the density quotient in the
Cauchy scale is the function
\[
\widetilde\delta_t(y)
:=\frac{h_t(\bar\gamma+ty)}{g_t(\bar\gamma+ty)}-1
=\int_{\RR}\bigl(R_{\Delta/t}(y)-1\bigr)\dd\nu(\gamma).
\]
This is the same normalization as
\eqref{eq:cayley-comparison-kernel} and as the Laurent polynomials
\eqref{eq:mode-laurent-even}--\eqref{eq:mode-laurent-odd}; no reciprocal
kernel or extra factor of \(1+y^2\) is being introduced.

Apply Lemma~\ref{lem:exact-moment-density-truncation} with this value of
\(L\).  The constant \(n=0\) term cancels in \(R_{\Delta/t}-1\), and the
\(n=1\) term integrates to zero because \(\mu_1=\int\Delta\,\dd\nu=0\).
Thus the finite-mode part begins at order \(t^{-2}\).  Put
\[
d_t(y):=\sum_{n=2}^{L}\mu_n\,u_n(y)t^{-n}.
\]
The exact-threshold density remainder gives
\[
\widetilde\delta_t(y)=d_t(y)+\rho_t(y),\qquad
\|\rho_t\|_{\infty}+\|\rho_t\|_{L^1(\omega)}
=o(t^{-L}),
\]
and \(\|\widetilde\delta_t\|_\infty=O(t^{-2})\).  The finite sum \(d_t\) is
also \(O(t^{-2})\) uniformly.

The change of variables is normalized by the same Cauchy law:
\[
g_t(\bar\gamma+ty)\,\dd(\bar\gamma+ty)
=P_t(ty)\,t\,\dd y
=\frac{\dd y}{\pi(1+y^2)}
=\omega(\dd y).
\]
Since \(\int g_t(\widetilde\delta_t\circ((\,\cdot-\bar\gamma)/t))\,\dd x=0\),
the KL entropy is
\[
D_{\mathrm{KL}}(h_t\|g_t)
=\int_{\RR}\Phi(\widetilde\delta_t(y))\,\omega(\dd y),
\qquad
\Phi(s)=(1+s)\log(1+s)-s.
\]

Choose \(T\) so large that
\(\|\widetilde\delta_t\|_\infty\le1/2\) for \(t\ge T\).  For such \(t\),
\[
\Phi(s)=\sum_{r=2}^{N}\frac{(-1)^r}{r(r-1)}s^r+O(s^{N+1})
\qquad(|s|\le1/2).
\]
The Taylor tail contributes \(O(t^{-2N-2})\) after integration.  For each
\(2\le r\le N\), the binomial expansion of
\((d_t+\rho_t)^r-d_t^r\), with one \(\rho_t\) factor placed in
\(L^1(\omega)\) and all other factors in \(L^\infty\), gives
\[
\|\widetilde\delta_t^r-d_t^r\|_{L^1(\omega)}
\le
\sum_{\ell=1}^{r}\binom r\ell
O(t^{-2(r-\ell)})\,o(t^{-L\ell}).
\]
For \(\ell=1\) this is
\(o(t^{-(L+2(r-1))})\), whose largest possible order occurs at \(r=2\) and
is \(o(t^{-(L+2)})=o(t^{-2N})\).  For \(\ell\ge2\), it is
\(o(t^{-2L})\), and since \(L=2N-2\ge2\) one has \(2L\ge L+2=2N\).
Thus all density-remainder products are \(o(t^{-2N})\).
Expanding the
finite powers of \(d_t\) gives
\begin{align*}
D_{\mathrm{KL}}(h_t\|g_t)
&=\sum_{k=2}^{N}\frac{(-1)^k}{k(k-1)}
\sum_{m=2k}^{2N}\frac{1}{t^m}
\sum_{\substack{n_1,\dots,n_k\ge 2\\n_1+\cdots+n_k=m}}
\mu_{n_1}\cdots\mu_{n_k}\,I_{n_1,\dots,n_k} \\
&\qquad +o(t^{-2N}),
\end{align*}
where $I_{n_1,\dots,n_k}=I(n_1,\dots,n_k)$ is the mode integral from
\eqref{eq:mode-integral-definition}.  No omitted density mode can contribute
to a retained term: if \(n_1+\cdots+n_k=m\le2N\), \(k\ge2\), and all
\(n_j\ge2\), then every \(n_j\le2N-2\), which is the highest mode retained in
\(d_t\).  Terms in \(d_t^k\) of total degree \(m>2N\) start at
\(t^{-(2N+1)}\) and are included in the \(o(t^{-2N})\) remainder.

Comparing the coefficient of $t^{-m}$ with the evaluated formal polynomial
\eqref{eq:universal-entropy-coefficient-evaluation} gives $A_m(\nu)$.
The parity part of Theorem~\ref{thm:universal-entropy-coefficients} gives
$A_m(\nu)=0$ for odd~$m$, so only even powers remain.  Reindexing $m=2r$
gives
\[
D_{\mathrm{KL}}(h_t\|g_t)
=\sum_{r=2}^{N} A_{2r}(\nu)\,t^{-2r}+o(t^{-2N}),
\]
which proves \eqref{eq:poisson-kl-even-orders}.
\end{proof}

\begin{theorem}[Positive-margin quantitative retained-order theorem]%
\label{thm:positive-margin-retained-order-entropy}
Let \(N\ge2\), let \(0<\eta\le1\), and let \(\nu\) be a probability
measure on \(\RR\) with mean \(\bar\gamma\) and finite centred
\((2N-2+\eta)\)th absolute moment.  Put
\[
L:=2N-2,\qquad p:=L+\eta,\qquad
\Delta:=\gamma-\bar\gamma,
\]
and truncate the density expansion at the retained order \(L\):
\[
d_t(y):=\sum_{n=2}^{L}\mu_nu_n(y)t^{-n}.
\]
Let \(C_L\) and \(B_L\) be the finite constants from
Theorem~\ref{thm:finite-moment-density-remainder}, and set
\[
C_\rho:=2C_L\EE|\Delta|^p,\qquad
M_A:=\sum_{n=2}^{L}|\mu_n|\,\|u_n\|_\infty,\qquad
S:=B_L\sum_{n=2}^{L}|\mu_n|+C_\rho .
\]
Define the density-remainder product constant
\[
C_{\nu,N,\eta}:=
\sum_{r=2}^{N}\frac{1}{r(r-1)}
\sum_{\ell=1}^{r}\binom r\ell
M_A^{r-\ell}C_\rho^\ell ,
\]
the finite retained-mode tail
\[
Q_{\nu,N}:=
\sum_{r=2}^{N}\frac{1}{r(r-1)}
\sum_{m=2N+1}^{Lr}
\sum_{\substack{2\le n_1,\ldots,n_r\le L\\
n_1+\cdots+n_r=m}}
|\mu_{n_1}\cdots\mu_{n_r}|\,
\|u_{n_1}\cdots u_{n_r}\|_{L^1(\omega)} ,
\]
with the convention that an empty sum is zero, and
\[
C'_{\nu,N}:=2S^{N+1},\qquad T_{\nu,N,\eta}:=\max\{1,(2S)^{1/2}\}.
\]
Then, for every \(t\ge T_{\nu,N,\eta}\),
\begin{equation}\label{eq:positive-margin-retained-order-theorem}
\left|
D_{\mathrm{KL}}(P_t*\nu\|P_t(\cdot-\bar\gamma))
-\sum_{m=2}^{N}A_{2m}(\nu)t^{-2m}
\right|
\le
C_{\nu,N,\eta}t^{-(2N+\eta)}
+Q_{\nu,N}t^{-(2N+1)}
+C'_{\nu,N}t^{-(2N+2)} .
\end{equation}
Thus the coefficient theorem uses only the exact
\((2N-2)\)-moment threshold, while the positive margin supplies a
quantitative remainder for the same retained density modes
\(u_2,\ldots,u_{2N-2}\).
\end{theorem}

\begin{proof}
Theorem~\ref{thm:finite-moment-density-remainder}, applied with
truncation \(L=2N-2\), gives in the Cauchy scale
\[
\widetilde\delta_t(y)=d_t(y)+\rho_t(y),\qquad
\|\rho_t\|_\infty+\|\rho_t\|_{L^1(\omega)}
\le C_\rho t^{-p},
\]
and \(\|\widetilde\delta_t\|_\infty\le S t^{-2}\).  Increasing the lower
threshold to \(T_{\nu,N,\eta}\) gives
\(\|\widetilde\delta_t\|_\infty\le1/2\), so
\[
\Phi(z)=\sum_{r=2}^{N}\frac{(-1)^r}{r(r-1)}z^r+O(z^{N+1})
\qquad(|z|\le1/2).
\]
The Taylor tail is bounded by \(C'_{\nu,N}t^{-(2N+2)}\).

For each \(2\le r\le N\), Lemma~\ref{lem:density-remainder-product-hierarchy}
with \(p=L+\eta\), \(A=d_t\), and \(\rho=\rho_t\) shows that all products
containing at least one density remainder are bounded by
\[
\sum_{\ell=1}^{r}\binom r\ell M_A^{r-\ell}C_\rho^\ell
t^{-(p+2)}
=
\sum_{\ell=1}^{r}\binom r\ell M_A^{r-\ell}C_\rho^\ell
t^{-(2N+\eta)}.
\]
After summing the Taylor coefficients in \(r\), this is the
\(C_{\nu,N,\eta}t^{-(2N+\eta)}\) term.

The purely finite part \(d_t^r\) has retained total degrees \(m\le2N\) and a
finite high-degree tail \(m\ge2N+1\).  The latter is exactly bounded by
\(Q_{\nu,N}t^{-(2N+1)}\).  If an entropy monomial has total degree
\(m\le2N\), at least two factors occur and every index is at least \(2\);
therefore no individual index can exceed \(2N-2=L\).  Hence truncating the
density expansion at \(L\) removes no coefficient \(A_m(\nu)\) with
\(m\le2N\).  The parity theorem removes the odd retained coefficients, and
\eqref{eq:positive-margin-retained-order-theorem} follows.
\end{proof}

\begin{proposition}[Sharpness of the moment threshold for this universal moment-polynomial coefficient]%
\label{prop:poisson-kl-moment-threshold-sharpness}
Fix \(N\ge2\) and put \(L=2N-2\).  The moment hypothesis in
Theorem~\ref{thm:poisson-kl-even-orders} cannot be lowered for a universal
coefficient theorem with the same evaluated algebraic polynomial
\(A_{2N}^{\rm alg}\), for this universal moment-polynomial coefficient.

For \(N=2\), the leading coefficient is
\[
A_4(\nu)=\frac18\mu_2^2,
\]
so a universal statement through order \(t^{-4}\) must at least make the
variance finite.  For \(N\ge3\), the degree-\(2N\) formal coefficient has the
nonzero top-moment term
\begin{equation}\label{eq:top-moment-coefficient-sharpness}
A_{2N}^{\rm form}
=\kappa_N\,\mathfrak m_2\mathfrak m_{2N-2}
 +P_N(\mathfrak m_2,\ldots,\mathfrak m_{2N-3}),
\qquad
\kappa_N=(-1)^N(N-1)2^{-2N+2}\ne0,
\end{equation}
where \(P_N\) involves no \(\mathfrak m_{2N-2}\).  Hence the coefficient
through order \(t^{-2N}\) is not determined by moments below order
\(2N-2\).

Moreover, for every \(q<L\) there is a centred symmetric probability law
with finite \(q\)th absolute moment and infinite \(L\)th absolute moment.
For such a law the top universal moment-polynomial coefficient in
\eqref{eq:top-moment-coefficient-sharpness} is not a finite number.  Thus no
fixed-law theorem that keeps the same universal coefficient \(A_{2N}\) can
replace the finite \(L\)th absolute moment by a strictly weaker moment
assumption.  This statement is only a sharpness statement for that universal
moment-polynomial coefficient; it does not rule out alternative truncated,
renormalized, or distribution-dependent expansions below the threshold.
\end{proposition}

\begin{proof}
The case \(N=2\) is the first line of the coefficient computation:
\[
A_4(\nu)=\frac12\mu_2^2 I(2,2)
=\frac12\mu_2^2\cdot\frac14
=\frac18\mu_2^2 .
\]
Assume \(N\ge3\).  In the formal coefficient
\eqref{eq:universal-entropy-coefficient}, a monomial containing
\(\mathfrak m_L\) and contributing to total degree \(L+2=2N\) must have
exactly two density factors, with indices \(2\) and \(L\).  All other
entropy monomials either have total degree different from \(2N\) or involve
only moments of order at most \(L-1\).  Since the quadratic Taylor
coefficient is \(1/2\) and the ordered pairs \((2,L)\) and \((L,2)\) both
occur, the coefficient of \(\mathfrak m_2\mathfrak m_L\) is \(I(2,L)\).

Write \(L=2m\), so \(m=N-1\).  From
\eqref{eq:mode-laurent-even},
\[
Q_{2m}(z)=(-1)^m2^{-2m}
\sum_{j=0}^{2m-1}\binom{2m-1}{j}
\bigl(z^{j+1}+z^{-(j+1)}\bigr),
\]
while \(Q_2(z)=-(z+z^{-1}+z^2+z^{-2})/4\).  Therefore only the
exponents \(\pm1\) and \(\pm2\) of \(Q_{2m}\) enter \(I(2,2m)\), and
\[
\begin{aligned}
I(2,2m)
&=[z^0]Q_2Q_{2m}  \\
&=-\frac12\,(-1)^m2^{-2m}
\left\{\binom{2m-1}{0}+\binom{2m-1}{1}\right\}  \\
&=(-1)^{m+1}m\,2^{-2m}
=(-1)^N(N-1)2^{-2N+2}.
\end{aligned}
\]
This proves \eqref{eq:top-moment-coefficient-sharpness}.

For the counterexample assertion, choose a symmetric probability density on
\(|x|\ge e\) proportional to
\[
\frac{1}{|x|^{L+1}\log |x|}
\]
and put any remaining mass symmetrically on a bounded interval so that the
total mass is one.  The law is centred by symmetry.  For every \(q<L\),
\(\int |x|^q\,\dd\nu(x)<\infty\), because the tail integral is dominated by
\(\int_e^\infty x^{q-L-1}\dd x\).  At order \(L\), however,
\[
\int |x|^L\,\dd\nu(x)
\asymp \int_e^\infty \frac{\dd x}{x\log x}
=\infty .
\]
Thus the moment required by the nonzero top term of \(A_{2N}\) is genuinely
absent although every lower absolute moment is finite.
\end{proof}

\begin{corollary}[Consolidated constants for the finite-moment entropy expansion]%
\label{cor:finite-moment-entropy-constant-remainder}
Let \(N\ge2\), let \(0<\eta\le1\), and let \(\nu\) be a probability measure
on \(\RR\) with mean \(\bar\gamma\) and finite centred
\((2N+\eta)\)th absolute moment.  This stronger hypothesis is not needed for
the coefficient identity in Theorem~\ref{thm:poisson-kl-even-orders}; it is a
convenient sufficient assumption for a completely explicit fixed-law
remainder estimate at the theorem statement.  Put
\[
p:=2N+\eta,\qquad \Delta:=\gamma-\bar\gamma .
\]
Let
\[
C_\rho:=2C_{2N}\EE|\Delta|^p,
\qquad
M_A:=\sum_{n=2}^{2N}|\mu_n|\,\|u_n\|_\infty,
\]
where \(C_{2N}\) is the finite density constant from
Theorem~\ref{thm:finite-moment-density-remainder}.  Define
\[
P_N(C_\rho,M_A)
:=
\sum_{r=2}^{N}\frac1{r(r-1)}
\sum_{\ell=1}^{r}\binom r\ell
M_A^{r-\ell}C_\rho^\ell ,
\]
and
\[
Q_N(\mu,u)
:=
\sum_{r=2}^{N}\frac1{r(r-1)}
\sum_{m=2N+1}^{2Nr}
\sum_{\substack{2\le n_1,\ldots,n_r\le2N\\
n_1+\cdots+n_r=m}}
|\mu_{n_1}\cdots\mu_{n_r}|\,
\|u_{n_1}\cdots u_{n_r}\|_{L^1(\omega)} .
\]
Finally set
\[
S:=B_{2N}\sum_{n=2}^{2N}|\mu_n|+C_\rho,
\qquad
T:=\max\{1,(2S)^{1/2}\},
\]
where \(B_{2N}\) is the finite mode-size constant from
Theorem~\ref{thm:finite-moment-density-remainder}.  Then, for every
\(t\ge T\),
\begin{align}
\label{eq:finite-moment-entropy-remainder-estimate}
\left|
D_{\mathrm{KL}}(P_t*\nu\|P_t(\cdot-\bar\gamma))
-\sum_{m=2}^{N}A_{2m}(\nu)t^{-2m}
\right|
&\le
\{P_N(C_\rho,M_A)+Q_N(\mu,u)\}t^{-(2N+\eta)}
\notag\\
&\qquad
+2S^{N+1}t^{-(2N+2)} .
\end{align}
The refined three-term estimate behind this convenience envelope is
\[
P_N(C_\rho,M_A)t^{-(2N+\eta+2)}
+Q_N(\mu,u)t^{-(2N+1)}
+2S^{N+1}t^{-(2N+2)} .
\]
For a centred \(p\)-moment ball \(\EE|\Delta|^p\le M\), the same estimate is
made uniform by replacing the law-dependent quantities with the closed finite
dominating constants of
Lemma~\ref{lem:moment-ball-uniform-constant-domination}.  In particular
\(\|u_n\|_\infty\le n+1\) and
\(\|u_{n_1}\cdots u_{n_r}\|_{L^1(\omega)}
\le\prod_j(n_j+1)\) give a version involving no unspecified mode norms.
\end{corollary}

\begin{proof}
Apply Theorem~\ref{thm:finite-moment-density-remainder} with \(L=2N\).
In the Cauchy scale \(x=\bar\gamma+ty\) this gives
\[
\delta_t(\bar\gamma+ty)
=A_t(y)+\rho_t(y),
\qquad
A_t(y):=\sum_{n=2}^{2N}\mu_nu_n(y)t^{-n},
\]
with
\[
\|\rho_t\|_\infty+\|\rho_t\|_{L^1(\omega)}
\le C_\rho t^{-(2N+\eta)},
\qquad
\|\delta_t\|_\infty\le S t^{-2}
\]
for \(t\ge1\), and \(\|\delta_t\|_\infty\le1/2\) for \(t\ge T\).  Also
\(\|A_t\|_\infty\le M_At^{-2}\).

Proposition~\ref{prop:density-to-entropy-descent} applies with this density
input.  Its product-remainder term is exactly
\(P_N(C_\rho,M_A)t^{-(2N+\eta+2)}\), its finite high-degree mode tail is
\(Q_N(\mu,u)t^{-(2N+1)}\), and its Taylor remainder is
\(2S^{N+1}t^{-(2N+2)}\).  The retained coefficients are the same
\(A_m(\nu)\) as in Theorem~\ref{thm:poisson-kl-even-orders}, and parity
removes the odd retained powers.  Since \(0<\eta\le1\) and \(t\ge1\), the
first two terms are bounded by
\(\{P_N(C_\rho,M_A)+Q_N(\mu,u)\}t^{-(2N+\eta)}\), proving
\eqref{eq:finite-moment-entropy-remainder-estimate}.  The moment-ball
replacement is Lemma~\ref{lem:moment-ball-uniform-constant-domination}
together with the closed mode bound from
Lemma~\ref{lem:closed-cayley-mode-constant-envelopes}.
\end{proof}

\begin{corollary}[Positive-margin retained-order entropy bridge for explicit constants]%
\label{cor:optimized-retained-order-entropy-bridge}
Let \(N\ge2\), let \(0<\eta\le1\), and let \(\nu\) be a probability
measure on \(\RR\) with mean \(\bar\gamma\) and finite centred
\((2N-2+\eta)\)th absolute moment.  Put
\[
p:=2N-2+\eta,\qquad L:=2N-2,\qquad
\Delta:=\gamma-\bar\gamma .
\]
Let
\[
C_\rho^{\rm opt}:=2C_L\EE|\Delta|^p,
\qquad
M_A^{\rm opt}:=\sum_{n=2}^{L}|\mu_n|\,\|u_n\|_\infty,
\]
where \(C_L\) is the finite density constant from
Theorem~\ref{thm:finite-moment-density-remainder}.  Define
\[
P_N^{\rm opt}(C_\rho^{\rm opt},M_A^{\rm opt})
:=
\sum_{r=2}^{N}\frac{1}{r(r-1)}
\sum_{\ell=1}^{r}\binom r\ell
(M_A^{\rm opt})^{r-\ell}(C_\rho^{\rm opt})^\ell
\]
and
\[
Q_N^{\rm opt}(\mu,u)
:=
\sum_{r=2}^{N}\frac{1}{r(r-1)}
\sum_{m=2N+1}^{Lr}
\sum_{\substack{2\le n_1,\ldots,n_r\le L\\
n_1+\cdots+n_r=m}}
|\mu_{n_1}\cdots\mu_{n_r}|\,
\|u_{n_1}\cdots u_{n_r}\|_{L^1(\omega)} ,
\]
with the convention that an empty sum is zero.  Finally set
\[
S^{\rm opt}:=B_L\sum_{n=2}^{L}|\mu_n|+C_\rho^{\rm opt},
\qquad
T^{\rm opt}:=\max\{1,(2S^{\rm opt})^{1/2}\},
\]
where \(B_L\) is the finite mode-size constant in
Theorem~\ref{thm:finite-moment-density-remainder}.  Then, for every
\(t\ge T^{\rm opt}\),
\begin{align}
\label{eq:optimized-retained-order-entropy-bridge}
\left|
D_{\mathrm{KL}}(P_t*\nu\|P_t(\cdot-\bar\gamma))
-\sum_{m=2}^{N}A_{2m}(\nu)t^{-2m}
\right|
&\le
P_N^{\rm opt}(C_\rho^{\rm opt},M_A^{\rm opt})\,t^{-(2N+\eta)}
\notag\\
&\quad
{}+Q_N^{\rm opt}(\mu,u)\,t^{-(2N+1)}
+2(S^{\rm opt})^{N+1}t^{-(2N+2)} .
\end{align}
This is exactly Theorem~\ref{thm:positive-margin-retained-order-entropy}
with explicit names for the three finite constants used later in the
\(N=4\) remainder theorem:
\[
C_{\nu,N,\eta}=P_N^{\rm opt},\qquad
Q_{\nu,N}=Q_N^{\rm opt},\qquad
C'_{\nu,N}=2(S^{\rm opt})^{N+1}.
\]
Consequently,
\begin{equation}\label{eq:optimized-retained-order-entropy-envelope}
\left|
D_{\mathrm{KL}}(P_t*\nu\|P_t(\cdot-\bar\gamma))
-\sum_{m=2}^{N}A_{2m}(\nu)t^{-2m}
\right|
\le
\bigl\{
P_N^{\rm opt}+Q_N^{\rm opt}+2(S^{\rm opt})^{N+1}
\bigr\}t^{-(2N+\eta)}
\end{equation}
  for \(t\ge T^{\rm opt}\).

  In the specialization \(N=4\), \(\eta=1\), assume
  \(\EE|\Delta|^7<\infty\), set \(L=6\), and write
  \[
  B_{\nu,8}:=
  P_4^{\rm opt}(C_\rho^{\rm opt},M_A^{\rm opt})
  +Q_4^{\rm opt}(\mu,u)
  +2(S^{\rm opt})^5,
  \qquad
  T_{\nu,8}:=T^{\rm opt}.
  \]
  Here \(P_4^{\rm opt}\), \(Q_4^{\rm opt}\), and \(S^{\rm opt}\) are the
  preceding optimized quantities with \(N=4\), \(L=6\), and \(\eta=1\).
  Then, for every \(t\ge T_{\nu,8}\),
  \begin{equation}\label{eq:optimized-n4-seventh-envelope}
  \left|
  D_{\mathrm{KL}}(P_t*\nu\|P_t(\cdot-\bar\gamma))
  -A_4(\nu)t^{-4}-A_6(\nu)t^{-6}-A_8(\nu)t^{-8}
  \right|
  \le B_{\nu,8}t^{-9}.
  \end{equation}
  Thus the order-nine constant used below is not an additional analytic
  input: it is exactly the optimized envelope obtained from the three explicit
  quantities \(P_4^{\rm opt}\), \(Q_4^{\rm opt}\), and \(S^{\rm opt}\) above.
  The fixed-law coefficient identity is the \(N=4\) specialization of
  Theorem~\ref{thm:poisson-kl-even-orders}; after the explicit coefficient computation in
  Theorem~\ref{thm:poisson-kl-eighth}, the displayed \(A_8(\nu)\) is the
  actual \(t^{-8}\) entropy coefficient.  The displayed rate is the
  \(N=4,\eta=1\) specialization of the positive-margin envelope
  above.
\end{corollary}

\begin{proof}
Apply Theorem~\ref{thm:finite-moment-density-remainder} with
\(L=2N-2\).  In the Cauchy scale
\(x=\bar\gamma+ty\) this gives
\[
\delta_t(\bar\gamma+ty)
=A_t(y)+\rho_t(y),
\qquad
A_t(y):=\sum_{n=2}^{L}\mu_nu_n(y)t^{-n},
\]
with
\[
\|\rho_t\|_\infty+\|\rho_t\|_{L^1(\omega)}
\le C_\rho^{\rm opt}t^{-p},
\qquad
\|\delta_t\|_\infty\le S^{\rm opt}t^{-2}
\]
for \(t\ge1\), after increasing the lower threshold to \(T^{\rm opt}\).
Also \(\|A_t\|_\infty\le M_A^{\rm opt}t^{-2}\).

The proof of Proposition~\ref{prop:density-to-entropy-descent} now applies
with the density-remainder exponent \(p=2N-2+\eta\) and the finite mode range
\(2\le n\le L\).  Lemma~\ref{lem:density-remainder-product-hierarchy}
places every product containing at least one \(\rho_t\) at order
\(p+2=2N+\eta\), producing the \(P_N^{\rm opt}\)-term.  The finite mode
polynomial terms whose total degree exceeds \(2N\) start at \(2N+1\) and are
bounded by \(Q_N^{\rm opt}t^{-(2N+1)}\).  The Taylor remainder for
\(\Phi(z)=(1+z)\log(1+z)-z\) contributes
\(2(S^{\rm opt})^{N+1}t^{-(2N+2)}\).

It remains only to check that truncating the density modes at
\(L=2N-2\) does not remove a retained entropy coefficient.  Every entropy
monomial contributing to total degree \(m\le2N\) has at least two factors
and all indices at least \(2\).  If one index were \(2N-1\) or larger, then
the total degree would be at least \((2N-1)+2=2N+1\).  Hence all retained
monomials use only the modes \(u_2,\ldots,u_{2N-2}\), and their finite
polynomial coefficients are the same evaluations \(A_m(\nu)\) as in
Theorem~\ref{thm:poisson-kl-even-orders}.  Parity removes the odd
coefficients, giving \eqref{eq:optimized-retained-order-entropy-bridge}.
Since \(0<\eta\le1\) and \(t\ge1\), the last two powers in
\eqref{eq:optimized-retained-order-entropy-bridge} are bounded by
\(t^{-(2N+\eta)}\), which proves
\eqref{eq:optimized-retained-order-entropy-envelope}.
\end{proof}

\begin{corollary}[Uniform finite-moment entropy remainder on moment balls]%
\label{cor:uniform-moment-ball-entropy-remainder}
Fix $N\ge2$, $0<\eta\le1$, and $M<\infty$.  There are finite constants
\[
C_{N,\eta,M}^{\rm dens},\qquad C_{N,\eta,M}^{\rm Tay},
\qquad T_{N,\eta,M}^{\rm ent}
\]
given by the closed finite formulas of
Lemma~\ref{lem:moment-ball-uniform-constant-domination}, depending only on
$N$, $\eta$, and $M$, such that the following holds uniformly for every
probability measure $\nu$ on
$\RR$ with finite mean $\bar\gamma$ and centred moment bound
\[
\int_{\RR}|\gamma-\bar\gamma|^{2N+\eta}\dd\nu(\gamma)
\le M .
\]
With $h_t:=P_t*\nu$, $g_t(x):=P_t(x-\bar\gamma)$, and the law-dependent
coefficients $A_{2m}(\nu)$ from
\eqref{eq:universal-entropy-coefficient-evaluation}, one has, for every
$t\ge T_{N,\eta,M}^{\rm ent}$,
\begin{equation}\label{eq:uniform-moment-ball-entropy-remainder}
\left|
D_{\mathrm{KL}}(h_t\|g_t)-\sum_{m=2}^{N}A_{2m}(\nu)t^{-2m}
\right|
\le
C_{N,\eta,M}^{\rm dens}t^{-(2N+\eta)}
+C_{N,\eta,M}^{\rm Tay}t^{-(2N+2)} .
\end{equation}
Consequently the expansion in Theorem~\ref{thm:poisson-kl-even-orders} is
uniform on every centred $(2N+\eta)$-moment ball, although the coefficients
remain the moment polynomials of the individual law.
\end{corollary}

\begin{proof}
Apply Lemma~\ref{lem:moment-ball-uniform-constant-domination}.  It gives
uniform dominating constants for the density remainder constant
$C_{\nu,2N,\eta}$, the smallness quantity $S_{\nu,2N,\eta}$, the entropy
constants $C_{N,\eta}^{\rm dens}(\nu)$ and $C_{N,\eta}^{\rm Tay}(\nu)$ from
\eqref{eq:explicit-entropy-constant-bound}, and the threshold
$T_{N,\eta}^{\rm ent}(\nu)$, all given by closed formulas in $N,\eta,M$.
Choose
$C_{N,\eta,M}^{\rm dens}$, $C_{N,\eta,M}^{\rm Tay}$, and
$T_{N,\eta,M}^{\rm ent}$ to be those dominating constants.  The finite computable
remainder estimate in Corollary~\ref{cor:finite-moment-entropy-constant-remainder},
with these larger constants substituted, gives
\eqref{eq:uniform-moment-ball-entropy-remainder} for every law in the moment
ball.  The coefficients are still the evaluated moment polynomials of the
individual law, so only the remainder constants and threshold are made uniform.
\end{proof}

\begin{theorem}[Exact-threshold finite-moment entropy remainder]%
\label{thm:finite-moment-entropy-remainder}
Let \(N\ge2\), and let \(\nu\) be a probability measure on \(\RR\) with mean
\(\bar\gamma\) and finite centred \((2N-2)\)th absolute moment.  Put
\[
\mathcal R_N(t;\nu)
:=D_{\mathrm{KL}}(P_t*\nu\|P_t(\cdot-\bar\gamma))
-\sum_{m=2}^{N}A_{2m}(\nu)t^{-2m}.
\]
Then
\[
\mathcal R_N(t;\nu)=o(t^{-2N})
\qquad(t\to\infty).
\]
Equivalently, after the universal even coefficient polynomial is subtracted,
the exact centred \((2N-2)\)th absolute moment hypothesis leaves no further
contribution at any order \(t^{-k}\) with \(k\le2N\).

If, in addition, \(0<\eta\le1\) and the centred
\((2N-2+\eta)\)th absolute moment is finite, the same retained expansion has
the theorem-level positive-margin quantitative remainder estimate
\eqref{eq:positive-margin-retained-order-theorem} of
Theorem~\ref{thm:positive-margin-retained-order-entropy}.  The optimized
notation used in the \(N=4\) coefficient computation is recorded in
Corollary~\ref{cor:optimized-retained-order-entropy-bridge}.  Under the still
stronger centred \((2N+\eta)\)th absolute moment, the convenience constants and
uniform moment-ball version are those of
Corollaries~\ref{cor:finite-moment-entropy-constant-remainder} and
\ref{cor:uniform-moment-ball-entropy-remainder}.  These \(+\eta\) assumptions
are used only to turn the little-oh remainder into explicit fixed-law or
uniform \(O\)-bounds; they are not part of the coefficient theorem.
\end{theorem}

\begin{proof}
This is precisely Theorem~\ref{thm:poisson-kl-even-orders} with the odd
coefficients suppressed by parity.  That theorem is proved from
Lemma~\ref{lem:exact-moment-density-truncation} with
\(L=2N-2\).  The density remainder is only
\(o(t^{-(2N-2)})\), but every entropy contribution containing it has at least
one further density factor, and the retained density perturbation starts at
order \(t^{-2}\).  Hence all such products are \(o(t^{-2N})\).  No omitted
density mode can enter a retained entropy monomial, because an entropy
monomial of total degree at most \(2N\) has at least two mode indices, each at
least \(2\), so each individual retained index is at most \(2N-2\).
Consequently
\[
D_{\mathrm{KL}}(P_t*\nu\|P_t(\cdot-\bar\gamma))
=\sum_{k=4}^{2N}A_k(\nu)t^{-k}+o(t^{-2N}),
\]
and parity gives
\(\sum_{k=4}^{2N}A_k(\nu)t^{-k}
=\sum_{m=2}^{N}A_{2m}(\nu)t^{-2m}\).
Subtracting this polynomial gives the stated exact-threshold remainder.

The quantitative assertion with the same retained modes is
Theorem~\ref{thm:positive-margin-retained-order-entropy}.  There the extra
\(\eta\) moment improves the density remainder, truncated at \(L=2N-2\), to an
explicit
\(O(t^{-(2N-2+\eta)})\) bound, so products containing the density remainder
are bounded at order \(t^{-(2N+\eta)}\).  Corollary
~\ref{cor:optimized-retained-order-entropy-bridge} only rewrites those
constants in the notation used later.  The stronger
\((2N+\eta)\)-moment estimate instead retains density modes through \(2N\) and
records the convenience constants in
\eqref{eq:finite-moment-entropy-remainder-estimate}; the moment-ball corollary
dominates those constants uniformly.  These estimates refine the exact
little-oh conclusion but are not needed for it.
\end{proof}

\begin{theorem}[Analytic compact-support interchange]%
\label{thm:analytic-compact-support-interchange}
Let $\nu$ be a probability measure with mean $\bar\gamma$ and compactly
supported centred variable $\Delta=\gamma-\bar\gamma$.  Then there is
$T_\nu\ge1$ such that, for $t\ge T_\nu$, the series
\begin{equation}\label{eq:analytic-density-series}
\delta_t(\bar\gamma+ty)
=\sum_{n=2}^{\infty}\mu_n\,u_n(y)t^{-n}
\end{equation}
converges absolutely and uniformly in $y\in\RR$.  The entropy Taylor series
\begin{equation}\label{eq:analytic-entropy-series}
D_{\mathrm{KL}}(P_t*\nu\|P_t(\cdot-\bar\gamma))
=\sum_{k=4}^{\infty}A_k(\nu)t^{-k}
\end{equation}
also converges absolutely for $t\ge T_\nu$, where $A_k(\nu)$ is the
finite-moment evaluation of $A_k^{\rm form}$ from
\eqref{eq:universal-entropy-coefficient-evaluation}.  In particular the
all-order coefficients are the genuine analytic coefficients under this
strong compact-support hypothesis, while
Theorem~\ref{thm:finite-moment-entropy-remainder} gives the separate
finite-moment asymptotic justification.
\end{theorem}

\begin{proof}
Let $|\Delta|\le M$ almost surely.  The generating function for the
Chebyshev polynomials gives
\[
R_\varepsilon(\tan\theta)
=\sum_{n=0}^{\infty}\cos^n\theta\,U_n(\sin\theta)\varepsilon^n.
\]
Since $|U_n(s)|\le n+1$ for $|s|\le1$ and $|\cos\theta|\le1$, the series is
dominated uniformly by $\sum_{n\ge0}(n+1)|\varepsilon|^n$ whenever
$|\varepsilon|<1$.  Choose $T_\nu\ge2M$.  Then
$|\Delta|/t\le1/2$, so dominated convergence permits termwise integration
against~$\nu$ and gives the absolutely and uniformly convergent density
series.  The $n=0$ term is one and the $n=1$ term is zero because
$\EE\Delta=0$, which proves \eqref{eq:analytic-density-series}.

The uniform convergence gives $\delta_t=O(t^{-2})$ uniformly.  Enlarging
$T_\nu$ if needed, $|\delta_t|\le1/2$, and the Taylor series for
$\Phi(s)=(1+s)\log(1+s)-s$ converges absolutely and uniformly after the
substitution $s=\delta_t(\bar\gamma+ty)$.  To make the interchange explicit,
write
\[
b_n(t):=|\mu_n|\,\|u_n\|_\infty t^{-n}
\qquad(n\ge2),
\qquad
B(t):=\sum_{n\ge2}b_n(t).
\]
The compact support bound and $\|u_n\|_\infty\le n+1$ give
$B(t)\le\sum_{n\ge2}(n+1)(M/t)^n<\infty$, and after increasing $T_\nu$ we
may assume $B(t)\le1/2$.  Hence every product occurring in the entropy Taylor
series has the summable absolute majorant
\[
\sum_{r=2}^{\infty}\frac1{r(r-1)}
\sum_{n_1,\dots,n_r\ge2}
b_{n_1}(t)\cdots b_{n_r}(t)
=\sum_{r=2}^{\infty}\frac{B(t)^r}{r(r-1)}<\infty.
\]
Since $\omega$ has total mass one and
\[
\int\bigl|u_{n_1}(y)\cdots u_{n_r}(y)\bigr|\,\omega(\dd y)
\le \prod_{j=1}^r\|u_{n_j}\|_\infty,
\]
this single majorant justifies both Fubini/Tonelli for
$\Phi(\delta_t)=\sum_{r\ge2}(-1)^r\delta_t^r/(r(r-1))$ and termwise
multiplication of the absolutely convergent density series.  For a fixed
power $t^{-k}$, only finitely many tuples $r\ge2$, $n_j\ge2$, and
$n_1+\cdots+n_r=k$ occur.  The degree-$k$ coefficient obtained by multiplying
the density series is therefore the finite-moment evaluation of the formal
expression in
\eqref{eq:universal-entropy-coefficient}; hence
\eqref{eq:analytic-entropy-series} follows.
\end{proof}
