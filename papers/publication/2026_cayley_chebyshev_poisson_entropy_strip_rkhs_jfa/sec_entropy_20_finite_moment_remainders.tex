\subsection{Finite-moment entropy asymptotics and remainders}%
\label{subsec:finite-moment-entropy}

\subsubsection{Density-to-entropy passage}

This subsection passes from formal coefficient algebra to genuine entropy
asymptotics.  The density-to-entropy passage is a finite operation at fixed
order $2N$.
First the density ratio is written as a finite mode polynomial plus an
$L^\infty\cap L^1(\omega)$ error.  Then
$\Phi(z)=(1+z)\log(1+z)-z$ is expanded only through degree~$N$, because the
density perturbation begins at order $t^{-2}$.  Thus a product containing one
density error also contains the entropy quadratic order and is two powers
smaller than the density error itself; products with more density-error
factors are smaller still.  The displayed two-scale constants below are
convenience envelopes for this refined product hierarchy and for the
high-degree finite-mode tail.  After integration, the remaining finite
polynomial is exactly the universal coefficient polynomial.

\begin{lemma}[Product hierarchy for the density remainder]%
\label{lem:density-remainder-product-hierarchy}
Let $p>2$, $2\le r$, and let $A,\rho$ be measurable functions satisfying
\[
\|A\|_{L^\infty(\RR)}\le M t^{-2},
\qquad
\|\rho\|_{L^\infty(\RR)}+\|\rho\|_{L^1(\omega)}
\le C t^{-p}
\]
for some $t\ge1$.  Then
\begin{equation}\label{eq:density-remainder-product-hierarchy}
\|(A+\rho)^r-A^r\|_{L^1(\omega)}
\le
\sum_{\ell=1}^{r}\binom r\ell
M^{r-\ell}C^\ell
t^{-\{\ell p+2(r-\ell)\}} .
\end{equation}
In particular, since $p>2$,
\begin{equation}\label{eq:density-remainder-product-envelope}
\|(A+\rho)^r-A^r\|_{L^1(\omega)}
\le
\sum_{\ell=1}^{r}\binom r\ell
M^{r-\ell}C^\ell
t^{-(p+2)} .
\end{equation}
\end{lemma}

\begin{proof}
Expand
\[
(A+\rho)^r-A^r
=\sum_{\ell=1}^r\binom r\ell A^{r-\ell}\rho^\ell .
\]
For a fixed $\ell$, put one factor of $\rho$ in $L^1(\omega)$ and all other
factors in $L^\infty$.  This gives
\[
\|A^{r-\ell}\rho^\ell\|_{L^1(\omega)}
\le
\|A\|_\infty^{r-\ell}\|\rho\|_\infty^{\ell-1}
\|\rho\|_{L^1(\omega)}
\le
M^{r-\ell}C^\ell
t^{-2(r-\ell)}t^{-\ell p},
\]
which is \eqref{eq:density-remainder-product-hierarchy}.  The exponent
$\ell p+2(r-\ell)$ is minimized at $\ell=1$ and $r=2$ when $p>2$ and
$r\ge2$, giving the envelope \eqref{eq:density-remainder-product-envelope}.
\end{proof}

\begin{lemma}[Exact-moment density truncation with little-oh remainder]%
\label{lem:exact-moment-density-truncation}
Let \(L\ge2\), and let \(\nu\) be a probability measure on \(\RR\) with
mean \(\bar{\gamma}\) and finite centred \(L\)th absolute moment.  Put
\(X_c=\gamma-\bar{\gamma}\) and
\[
\delta_t(\bar{\gamma}+ty)
:=
\frac{(P_t*\nu)(\bar{\gamma}+ty)}{P_t(ty)}-1 .
\]
Equivalently, for \(x=\bar{\gamma}+ty\),
\[
\frac{P_t(x-\gamma)}{P_t(x-\bar{\gamma})}
=\frac{t^2+(x-\bar{\gamma})^2}{t^2+(x-\gamma)^2}
=\frac{1+y^2}{1+(y-X_c/t)^2}
=R_{X_c/t}(y),
\]
and therefore
\[
\delta_t(\bar{\gamma}+ty)
=\int_{\RR}\bigl(R_{X_c/t}(y)-1\bigr)\dd\nu(\gamma).
\]
Define
\[
\rho_{L,t}(y):=
\int_{\RR}\left(
R_{X_c/t}(y)-\sum_{n=0}^{L}u_n(y)(X_c/t)^n
\right)\dd\nu(\gamma).
\]
Then, with \(\mu_n=\int X_c^n\dd\nu(\gamma)\) for \(0\le n\le L\),
\begin{equation}\label{eq:exact-moment-density-truncation}
\delta_t(\bar{\gamma}+ty)
=
\sum_{n=2}^{L}\mu_nu_n(y)t^{-n}+\rho_{L,t}(y),
\qquad
\|\rho_{L,t}\|_{\infty}+\|\rho_{L,t}\|_{L^1(\omega)}=o(t^{-L}).
\end{equation}
The \(n=0\) term is absent from the displayed sum because the quotient is
compared with \(1\), and the \(n=1\) term is absent because
\(\int X_c\,\dd\nu=0\).
Moreover \(\sup_y|\delta_t(\bar{\gamma}+ty)|=O(t^{-2})\).
In particular, when \(L=2N-2\), this is exactly the little-oh density input
used in Theorem~\ref{thm:poisson-kl-even-orders} for the displayed universal
polynomial coefficient \(A_{2N}\); the same \(O(t^{-2})\) bound gives the
smallness input for the entropy Taylor expansion.  The statement is the
finite-moment evaluability input for the displayed universal polynomial:
the tail split below replaces any \((L+1)\)-moment Taylor-remainder
hypothesis and keeps the evaluability hypothesis at \(L\) for that coefficient
polynomial.
Thus the later coefficient-evaluation
entropy hierarchy uses this lemma only through the displayed quotient
normalization, the cancellation of the \(0\) and \(1\) modes, and the
\(L^\infty\cap L^1(\omega)\) little-oh remainder.
\end{lemma}

\begin{proof}
The quotient normalization is the one used throughout this section:
\[
\frac{(P_t*\nu)(\bar{\gamma}+ty)}{P_t(ty)}
=\int_{\RR}R_{X_c/t}(y)\dd\nu(\gamma).
\]
We prove the \(L^\infty\) and \(L^1(\omega)\) remainders by the same split.
On \(|X_c|\le t\), the local part of the Cayley Taylor remainder gives,
uniformly in \(y\),
\[
\left|R_{X_c/t}(y)-\sum_{n=0}^{L}u_n(y)(X_c/t)^n\right|
\le C_L\left|\frac{X_c}{t}\right|^{L+1}.
\]
Consequently
\[
\left|
\int_{|X_c|\le t}\left(
R_{X_c/t}(y)-\sum_{n=0}^{L}u_n(y)(X_c/t)^n
\right)\dd\nu
\right|
\le
C_Lt^{-L}\EE\!\left[
|X_c|^L\frac{|X_c|}{t}\mathbf1_{\{|X_c|\le t\}}
\right].
\]
The expectation tends to zero by dominated convergence, since the multiplier
\((|X_c|/t)\mathbf1_{\{|X_c|\le t\}}\) is bounded by \(1\) and converges
pointwise to \(0\).

On \(|X_c|>t\), put \(\varepsilon=X_c/t\).  The large-\(\varepsilon\)
estimate used in Theorem~\ref{thm:finite-moment-density-remainder} gives,
for \(L\ge2\),
\[
R_{\varepsilon}(y)\le 3(1+\varepsilon^2)\le6|\varepsilon|^L,
\qquad
\left|\sum_{n=0}^{L}u_n(y)\varepsilon^n\right|
\le (L+1)B_L|\varepsilon|^L
\]
whenever \(|\varepsilon|>1\), where
\(B_L=\max_{0\le n\le L}\|u_n\|_\infty\).  Hence, uniformly in \(y\),
\[
\left|
\int_{|X_c|>t}\left(
R_{X_c/t}(y)-\sum_{n=0}^{L}u_n(y)(X_c/t)^n
\right)\dd\nu
\right|
\le
\{6+(L+1)B_L\}t^{-L}\EE[
|X_c|^L\mathbf1_{\{|X_c|>t\}}],
\]
which is \(o(t^{-L})\).  Combining the two pieces gives
\(\|\rho_{L,t}\|_\infty=o(t^{-L})\).  Since \(\omega\) is a probability
measure and the same pointwise majorants dominate the \(y\)-integral, the
identical estimates give \(\|\rho_{L,t}\|_{L^1(\omega)}=o(t^{-L})\).

The constant term cancels and the \(n=1\) term integrates to zero because
\(\mu_1=0\), giving \eqref{eq:exact-moment-density-truncation}.

Finally, Lemma~\ref{lem:tail-uniform-cayley-mode-remainder} with \(N=1\)
applies because \(L\ge2\) implies \(\EE|X_c|^2<\infty\).  Since
\(\mu_1=0\), it gives
\(\sup_y|\delta_t(\bar{\gamma}+ty)|=O(t^{-2})\).
This is the density-level input used by
Theorem~\ref{thm:poisson-kl-even-orders} when it identifies the universal
moment-polynomial coefficient \(A_{2N}\) at the first moment level where the
displayed coefficient polynomial is evaluable.
\end{proof}

\begin{proposition}[Retained-order finite-moment density-to-entropy transfer]%
\label{prop:density-to-entropy-descent}%
\label{lem:density-to-entropy-domination}%
\label{lem:explicit-entropy-constant-bound}
Let $N\ge2$, $0<\eta\le1$, and let
\[
L:=2N-2,\qquad
\omega(\dd y)=\frac{\dd y}{\pi(1+y^2)}.
\]
Let $u_2,\ldots,u_L$ be bounded measurable functions on~$\RR$, and let
$\mu_2,\ldots,\mu_L$ be real numbers.  For
$2\le r\le N$ and $2\le n_j\le L$, put
\[
J(n_1,\ldots,n_r)
:=\int_{\RR}u_{n_1}(y)\cdots u_{n_r}(y)\,\omega(\dd y),
\]
which is absolutely convergent because $\omega(\RR)=1$ and the modes are
bounded.  For $4\le m\le2N$ define the finite polynomial coefficient
\begin{equation}\label{eq:density-to-entropy-descent-coefficient}
\mathcal A_m(\mu,u)
:=
\sum_{r=2}^{\min\{N,\lfloor m/2\rfloor\}}
\frac{(-1)^r}{r(r-1)}
\sum_{\substack{2\le n_1,\ldots,n_r\le L\\
n_1+\cdots+n_r=m}}
\mu_{n_1}\cdots\mu_{n_r}\,J(n_1,\ldots,n_r).
\end{equation}
Equivalently, \(\mu_nu_n=0\) for \(n>L\).  This retained-order convention is
the one needed for coefficients through order \(2N\): every entropy monomial
has at least two density factors, so a density mode with index \(>L=2N-2\)
first contributes beyond order \(2N\).
Assume that for every $t\ge T$ there are measurable functions
$\delta_t,A_t,\rho_t$ such that
\begin{equation}\label{eq:density-to-entropy-descent-density}
\delta_t=A_t+\rho_t,\qquad
A_t(y)=\sum_{n=2}^{L}\mu_nu_n(y)t^{-n},
\end{equation}
and that the following are the only analytic hypotheses:
\begin{equation}\label{eq:density-to-entropy-descent-hypotheses}
\|\rho_t\|_{L^\infty(\RR)}+\|\rho_t\|_{L^1(\omega)}
\le C_\rho t^{-(L+\eta)},
\end{equation}
\[
\|\delta_t\|_{L^\infty(\RR)}\le\min\{1/2,S t^{-2}\},
\]
for fixed finite constants $C_\rho,S$.  Put
\[
M_A:=\sum_{n=2}^{L}|\mu_n|\,\|u_n\|_{L^\infty(\RR)} .
\]
Then $\|A_t\|_{L^\infty(\RR)}\le M_A t^{-2}$ for $t\ge1$,
$\Phi(\delta_t)\in L^1(\omega)$ for
\[
\Phi(z)=(1+z)\log(1+z)-z,
\]
and, for every $t\ge\max\{1,T\}$,
\begin{equation}\label{eq:density-to-entropy-descent-expansion}
\int_{\RR}\Phi(\delta_t(y))\,\omega(\dd y)
=\sum_{m=4}^{2N}\mathcal A_m(\mu,u)t^{-m}
+\mathcal E_{N,t}.
\end{equation}
The remainder is controlled by the finite computable bound
\begin{equation}\label{eq:density-to-entropy-descent-remainder}
|\mathcal E_{N,t}|
\le
\mathfrak P_N(C_\rho,M_A)t^{-(2N+\eta)}
+\mathfrak Q_N(\mu,u)t^{-(2N+1)}
+2S^{N+1}t^{-(2N+2)},
\end{equation}
where
\begin{equation}\label{eq:entropy-constant-product-error}
\mathfrak P_N(C_\rho,M_A)
:=
\sum_{r=2}^{N}\frac{1}{r(r-1)}
\sum_{\ell=1}^{r}\binom r\ell M_A^{r-\ell}C_\rho^\ell
\end{equation}
and
\begin{equation}\label{eq:entropy-constant-truncation-error}
\mathfrak Q_N(\mu,u)
:=
\sum_{r=2}^{N}\frac{1}{r(r-1)}
\sum_{m=2N+1}^{Lr}
\sum_{\substack{2\le n_1,\ldots,n_r\le L\\
n_1+\cdots+n_r=m}}
|\mu_{n_1}\cdots\mu_{n_r}|\,
\|u_{n_1}\cdots u_{n_r}\|_{L^1(\omega)} .
\end{equation}
Empty sums are interpreted as zero.
Thus the descent from an $L^\infty\cap L^1(\omega)$ density-ratio remainder to
the entropy integral is a finite retained-order calculation.  The finite
\((2N-2+\eta)\)-moment input is used only before this proposition, to produce
\eqref{eq:density-to-entropy-descent-density} and the
$L^\infty\cap L^1(\omega)$ bound
\eqref{eq:density-to-entropy-descent-hypotheses}.  Inside the transfer,
Lemma~\ref{lem:density-remainder-product-hierarchy} shows that every product
containing at least one $\rho_t$ is \(O(t^{-(2N+\eta)})\) after the
entropy quadratic order is used.  The term
$\mathfrak Q_N(\mu,u)t^{-(2N+1)}$ is the finite high-degree tail of the
retained mode polynomial.  Since \(0<\eta\le1\), the single power envelope
\begin{equation}\label{eq:density-to-entropy-convenience-envelope}
|\mathcal E_{N,t}|
\le
\bigl(\mathfrak P_N(C_\rho,M_A)+\mathfrak Q_N(\mu,u)
+2S^{N+1}\bigr)t^{-(2N+\eta)}
\end{equation}
also follows for $t\ge1$.

In the Poisson finite-moment application, assume that \(\nu\) has finite
centred \(p\)th absolute moment with
\[
p=L+\eta=2N-2+\eta,\qquad X_c=\gamma-\bar{\gamma} .
\]
Let \(C_L\) and \(B_L\) be the density constants from
Theorem~\ref{thm:finite-moment-density-remainder}, and put
\[
C_\rho:=2C_L\EE|X_c|^p,\qquad
M_A:=\sum_{n=2}^{L}|\mu_n|\,\|u_n\|_\infty,\qquad
S:=B_L\sum_{n=2}^{L}|\mu_n|+C_\rho .
\]
Define
\[
C_{\nu,N,\eta}:=\mathfrak P_N(C_\rho,M_A),\qquad
Q_{\nu,N}:=\mathfrak Q_N(\mu,u),\qquad
C'_{\nu,N}:=2S^{N+1},
\]
and
\[
T_{\nu,N,\eta}:=\max\{1,(2S)^{1/2}\}.
\]
Then, for every \(t\ge T_{\nu,N,\eta}\),
\begin{equation}\label{eq:explicit-entropy-constant-bound}
\left|
D_{\mathrm{KL}}(P_t*\nu\|P_t(\cdot-\bar{\gamma}))
-\sum_{m=2}^{N}A_{2m}(\nu)t^{-2m}
\right|
\le
C_{\nu,N,\eta}t^{-(2N+\eta)}
+Q_{\nu,N}t^{-(2N+1)}
+C'_{\nu,N}t^{-(2N+2)} .
\end{equation}
\begin{equation}\label{eq:positive-margin-retained-order-theorem}
\left|
D_{\mathrm{KL}}(P_t*\nu\|P_t(\cdot-\bar{\gamma}))
-\sum_{m=2}^{N}A_{2m}(\nu)t^{-2m}
\right|
\le
\bigl(C_{\nu,N,\eta}+Q_{\nu,N}+C'_{\nu,N}\bigr)t^{-(2N+\eta)}
\end{equation}
is the corresponding coarse retained-order envelope.  These are finite
expressions in the density-remainder constants from
Theorem~\ref{thm:finite-moment-density-remainder}, the moments through order
\(L=2N-2\), and the finite mode-product integrals.  If one wants a purely
numerical envelope,
Lemma~\ref{lem:closed-cayley-mode-constant-envelopes} gives
$\|u_n\|_\infty\le n+1$ and hence
\[
\|u_{n_1}\cdots u_{n_r}\|_{L^1(\omega)}
\le\prod_{j=1}^{r}(n_j+1).
\]
This proposition is the single quantitative transfer regime used later.  The
corresponding \(\eta=0\), little-oh argument is carried out directly in
Theorem~\ref{thm:poisson-kl-even-orders}.
\end{proposition}

\begin{proof}
Since $\omega$ has total mass one and the functions $u_n$ are bounded, every
finite product in \eqref{eq:density-to-entropy-descent-coefficient} belongs to
$L^1(\omega)$.  Thus the displayed coefficient polynomial consists of
absolutely convergent integrals and finite sums.
For $t\ge1$ the finite-mode part satisfies
\[
\|A_t\|_\infty
\le\sum_{n=2}^{L}|\mu_n|\,\|u_n\|_\infty t^{-n}
\le M_A t^{-2}.
\]

For $|z|\le1/2$,
\[
\Phi(z)=\sum_{r=2}^{N}\frac{(-1)^r}{r(r-1)}z^r+R_{N+1}(z),
\qquad |R_{N+1}(z)|\le2|z|^{N+1}.
\]
The smallness assumption on $\delta_t$ therefore implies
$\Phi(\delta_t)\in L^1(\omega)$ and
\[
\|R_{N+1}(\delta_t)\|_{L^1(\omega)}
\le2S^{N+1}t^{-(2N+2)}.
\]
For each $2\le r\le N$, apply
Lemma~\ref{lem:density-remainder-product-hierarchy} with
$p=L+\eta$, $M=M_A$, and $C=C_\rho$.  Since $L+\eta>2$, this gives
\[
\|\delta_t^r-A_t^r\|_{L^1(\omega)}
\le
\sum_{\ell=1}^{r}\binom r\ell
M_A^{r-\ell}C_\rho^\ell t^{-(2N+\eta)}.
\]
After multiplication by $1/(r(r-1))$ and summation in $r$, this gives the
$\mathfrak P_N(C_\rho,M_A)t^{-(2N+\eta)}$ part of the refined remainder.

It remains only to expand the finite polynomial $A_t^r$.  The exact identity
\[
\int_{\RR}A_t(y)^r\,\omega(\dd y)
=
\sum_{m=2r}^{Lr}t^{-m}
\sum_{\substack{2\le n_1,\ldots,n_r\le L\\
n_1+\cdots+n_r=m}}
\mu_{n_1}\cdots\mu_{n_r}\,J(n_1,\ldots,n_r)
\]
splits into the retained part $m\le2N$ and the discarded part $m>2N$.  The
retained part, summed over $2\le r\le N$ with Taylor coefficients, is exactly
$\sum_{m=4}^{2N}\mathcal A_m(\mu,u)t^{-m}$.  The discarded part is bounded in
absolute value by $\mathfrak Q_N(\mu,u)t^{-(2N+1)}$.  Combining this
high-degree polynomial tail with the product error and the Taylor tail proves
\eqref{eq:density-to-entropy-descent-expansion} and
\eqref{eq:density-to-entropy-descent-remainder}.
\end{proof}

\begin{lemma}[Closed moment-ball domination of density and entropy constants]%
\label{lem:moment-ball-uniform-constant-domination}
Fix $N\ge2$, $0<\eta\le1$, \(M<\infty\), and put
\[
L:=2N-2,\qquad p:=L+\eta.
\]
There are finite constants
\[
\overline C_{N,\eta,M}^{\rho},\qquad
\overline S_{N,\eta,M},
\qquad
\overline C_{\nu,N,\eta},\qquad
\overline Q_{N,\eta,M},\qquad
\overline C'_{N,\eta,M},\qquad
\overline T_{N,\eta,M}^{\rm ent}
\]
given by closed finite formulas in $N,\eta,M$ below, such that every
probability measure $\nu$ with finite mean $\bar{\gamma}$ and
\[
\int_{\RR}|\gamma-\bar{\gamma}|^p\dd\nu(\gamma)\le M
\]
satisfies
\begin{align*}
C_{\nu,L,\eta}&\le \overline C_{N,\eta,M}^{\rho},
&
S_{\nu,L,\eta}&\le \overline S_{N,\eta,M},
\\
C_{\nu,N,\eta}&\le \overline C_{\nu,N,\eta},
&
Q_{\nu,N}&\le \overline Q_{N,\eta,M},
&
C'_{\nu,N}&\le \overline C'_{N,\eta,M},
&
T_{\nu,N,\eta}&\le
\overline T_{N,\eta,M}^{\rm ent}.
\end{align*}
Consequently the density remainder
\eqref{eq:finite-moment-density-remainder}, the density smallness threshold
\eqref{eq:finite-moment-density-smallness}, and the entropy constants in
\eqref{eq:explicit-entropy-constant-bound} are uniformly dominated on the
centred \(p=2N-2+\eta\) moment ball by constants involving no unspecified
mode norms.
\end{lemma}

\begin{proof}
Let
\[
\widehat B:=L+1,\qquad
\widehat C:=\max\!\left\{\frac{5L+7}{2},\,6+(L+1)^2\right\}.
\]
These are the closed envelopes for \(B_L\) and \(C_L\) from
Lemma~\ref{lem:closed-cayley-mode-constant-envelopes}.  Define
\[
\overline C_{N,\eta,M}^{\rho}:=2\widehat C M,
\qquad
\overline S_{N,\eta,M}:=
\widehat B\sum_{n=2}^{L}M^{n/p}+2\widehat C M,
\]
\[
\overline M_{A,N,\eta,M}:=\sum_{n=2}^{L}(n+1)M^{n/p}.
\]
For the finite high-degree mode tail set
\[
\overline Q_{N,\eta,M}:=
\sum_{r=2}^{N}\frac{1}{r(r-1)}
\sum_{m=2N+1}^{Lr}
\sum_{\substack{2\le n_1,\dots,n_r\le L\\
n_1+\cdots+n_r=m}}
\left(\prod_{j=1}^{r}M^{n_j/p}(n_j+1)\right),
\]
where an empty sum is zero,
and set
\[
\overline C_{\nu,N,\eta}:=
\sum_{r=2}^{N}\frac{1}{r(r-1)}
\sum_{\ell=1}^{r}\binom r\ell
(\overline M_{A,N,\eta,M})^{r-\ell}
(\overline C_{N,\eta,M}^{\rho})^\ell .
\]
Finally define
\[
\overline C'_{N,\eta,M}:=
2(\overline S_{N,\eta,M})^{N+1},
\]
\[
\overline T_{N,\eta,M}^{\rm ent}:=
\max\{1,(2\overline S_{N,\eta,M})^{1/2}\}.
\]

For \(2\le n\le L\), Lyapunov's inequality gives
\(|\mu_n|\le M^{n/p}\).  The density theorem with retained order \(L\) and
Lemma~\ref{lem:closed-cayley-mode-constant-envelopes} give the closed
constant bound
\[
C_{\nu,L,\eta}=2C_L\EE|X_c|^p\le2\widehat C M
=\overline C_{N,\eta,M}^{\rho},
\]
and since
\[
S_{\nu,L,\eta}=B_L\sum_{n=2}^{L}|\mu_n|+C_{\nu,L,\eta},
\]
the same moment bounds give
\[
\overline S_{N,\eta,M}:=
\widehat B\sum_{n=2}^{L}M^{n/p}+2\widehat C M .
\]
The threshold definition then yields
\[
T_{\nu,N,\eta}=\max\{1,(2S_{\nu,L,\eta})^{1/2}\}
\le \max\{1,(2\overline S_{N,\eta,M})^{1/2}\}
=:\overline T_{N,\eta,M}^{\rm ent}.
\]

It remains only to dominate the finite entropy sums in
Proposition~\ref{prop:density-to-entropy-descent}.  The closed mode bound
$\|u_n\|_\infty\le n+1$ gives
$M_A\le\overline M_{A,N,\eta,M}$.  Substituting this and
$C_\rho\le\overline C_{N,\eta,M}^{\rho}$ into the finite sum
\eqref{eq:entropy-constant-product-error} gives
\(C_{\nu,N,\eta}\le\overline C_{\nu,N,\eta}\).
Substituting the bounds
$|\mu_{n_j}|\le M^{n_j/p}$ and
\[
\|u_{n_1}\cdots u_{n_r}\|_{L^1(\omega)}
\le \prod_{j=1}^{r}\|u_{n_j}\|_\infty
\le \prod_{j=1}^{r}(n_j+1),
\]
because $\omega$ has total mass one, gives
\(Q_{\nu,N}\le\overline Q_{N,\eta,M}\).  Finally
\(C'_{\nu,N}=2S_{\nu,L,\eta}^{N+1}\) is bounded by
\(\overline C'_{N,\eta,M}\).  All constants are therefore controlled by the
displayed finite formulas in \(N,\eta,M\) alone.
\end{proof}

The coefficient theorem uses the fixed-law Poisson comparison from the
introduction: the law \(\nu\) is fixed, and the Poisson time \(t\) tends to
infinity in the scalar functional
\[
D_{\mathrm{KL}}(P_t*\nu\|P_t(\cdot-\bar{\gamma})).
\]
Large-time moment expansions for heat, fractional heat, Poisson-kernel
limits, and geometric heat-type flows give profile corrections for the
evolved density or operator kernel
\cite{DuoandikoetxeaZuazua1992,EscobedoZuazua1991,KarchSchonbek2001,
Yamamoto2012FractionalExpansion,IshigeKawakami2012,
IshigeKawakamiMichihisa2017,IshigeKawakamiMichihisa2022RefinedFractional,
FilaIshigeKawakami2012PoissonKernel,Iwabuchi2015CriticalBurgersPoisson,
AnkerPapageorgiouZhang2023HeatSymmetricSpaces,
Papageorgiou2024FractionalLargeTime}.
Extension-method papers for the fractional Laplacian
\cite{CaffarelliSilvestre2007,StingaTorrea2010} fix a standard route from
nonlocal operators to local degenerate problems and provide normalization
background.
Stable-law entropy expansions and recent JFA entropic CLT work concern
convolution powers, normalized sums, R\'enyi-divergence CLT comparison, or
Esscher-transformed sums
\cite{BobkovChistyakovGotze2013,BobkovChistyakovGotzeEdgeworth2013,
BobkovChistyakovGotzeBerryEsseen2014,
BobkovGotzeRenyiSurvey2025,BobkovGotzeRenyi2025,
BobkovGotzeEsscher2025}.
The theorems below treat fixed-law Poisson comparisons
\[
D_{\mathrm{KL}}(P_t*\nu\|P_t(\cdot-\bar{\gamma})),
\]
and its radial \(d\)-dimensional analogue, after the mean-matched Poisson
normalization.  The first theorem records the finite-moment entropy jet in
dimension \(d\) for the radial Poisson kernel under
\(L\ge\max\{2,d+1\}\) and \(\EE|X_c|^L<\infty\).  The following scalar theorem
then evaluates the one-dimensional coefficients explicitly.  Within that
scalar functional, the Cayley--Chebyshev quotient
\((1+y^2)/(1+(y-X_c/t)^2)\) makes every retained entropy coefficient a
finite polynomial in centred moments.  The scalar theorem is a finite-moment
Poisson entropy-jet theorem, with a one-dimensional Laurent extraction layer
that makes the \(N=4\) coefficients explicit; its scope is stated once after
that theorem.

\begin{lemma}[Local multidimensional quotient--Taylor remainder]%
\label{lem:local-multidim-quotient-taylor-remainder}
Let \(d\ge1\), put \(a=(d+1)/2\), and, for \(y,z\in\RR^d\), set
\[
F_y(z):=
\left(\frac{1+|y|^2}{1+|y-z|^2}\right)^a .
\]
For a multiindex \(\alpha\in\NN_0^d\), define
\[
U_\alpha^{(d)}(y):=\frac{1}{\alpha!}\,\partial_z^\alpha F_y(0).
\]
For each integer \(L\ge2\) there is a constant \(C_{d,L}<\infty\) such that
\begin{equation}\label{eq:local-multidim-coefficient-bound}
\sup_{y\in\RR^d}|U_\alpha^{(d)}(y)|\le C_{d,L}
\qquad (|\alpha|\le L),
\end{equation}
and
\begin{equation}\label{eq:local-multidim-taylor-remainder}
\sup_{y\in\RR^d}
\left|
F_y(z)-\sum_{|\alpha|\le L}U_\alpha^{(d)}(y)z^\alpha
\right|
\le C_{d,L}|z|^{L+1}
\qquad (|z|\le1).
\end{equation}
\end{lemma}

\begin{proof}
For each multiindex \(\beta\), repeated differentiation gives the sharper
structure
\begin{equation}\label{eq:multidim-poisson-derivative-structure}
D_z^\beta F_y(w)
=(1+|y|^2)^a
\frac{P_\beta(y-w)}{(1+|y-w|^2)^{a+|\beta|}},
\end{equation}
where \(P_\beta\) is a polynomial, depending only on \(d\) and \(\beta\),
of degree at most \(|\beta|\).  Indeed, \(D_{z_j}=-D_{\xi_j}\) for
\(\xi=y-w\), and differentiating \((1+|\xi|^2)^{-b}\) once raises the
denominator exponent by one and multiplies by a polynomial of degree at most
one more.  If \(|w|\le1\) and \(|y|>2\), then \(|y-w|\ge |y|/2\), and
\[
|D_z^\beta F_y(w)|
\le
C_{d,\beta}
(1+|y|^2)^a
\frac{(1+|y|)^{|\beta|}}{(1+|y|^2)^{a+|\beta|}}
\le C_{d,\beta}.
\]
On \(|y|\le2\), the same derivative is bounded by compactness because
\(1+|y-w|^2\ge1\).  Hence, for every fixed \(L\),
\[
\sup_{\substack{y\in\RR^d\\ |w|\le1\\ |\beta|\le L+1}}
|\partial_z^\beta F_y(w)|<\infty .
\]
Taking \(w=0\) in the preceding derivative bound, with
\(|\beta|\le L\), also gives the asserted boundedness of
\(U_\alpha^{(d)}\), namely \eqref{eq:local-multidim-coefficient-bound}.

The multivariable Taylor formula with integral remainder at \(z=0\), using
the same derivative bound with \(|\beta|=L+1\), gives
\eqref{eq:local-multidim-taylor-remainder}.
\end{proof}

\begin{lemma}[Uniform radial Poisson Taylor remainder]%
\label{lem:multidim-uniform-radial-poisson-taylor}
Let \(d\ge1\), put \(a=(d+1)/2\), and, for \(y,z\in\RR^d\), set
\[
F_y(z):=
\left(\frac{1+|y|^2}{1+|y-z|^2}\right)^a .
\]
For a multiindex \(\alpha\in\NN_0^d\), define
\[
U_\alpha^{(d)}(y):=\frac{1}{\alpha!}\,\partial_z^\alpha F_y(0).
\]
If \(L\ge d+1\), then there is a constant \(C_{d,L}<\infty\) such that
\begin{equation}\label{eq:multidim-uniform-taylor-remainder-bound}
\left|
F_y(z)-\sum_{|\alpha|\le L}U_\alpha^{(d)}(y)z^\alpha
\right|
\le C_{d,L}\Bigl(
|z|^{L+1}\mathbf1_{\{|z|\le1\}}
+|z|^L\mathbf1_{\{|z|>1\}}
\Bigr)
\end{equation}
for all \(y,z\in\RR^d\).  Moreover
\[
\sup_{y\in\RR^d}|U_\alpha^{(d)}(y)|<\infty
\qquad (|\alpha|\le L).
\]
Consequently, if \(X\) is an \(\RR^d\)-valued random vector with
\(\EE|X|^L<\infty\), \(\Omega_d\) is any probability measure on \(\RR^d\),
and
\[
r_t(y):=\EE\left[
F_y(X/t)-\sum_{|\alpha|\le L}U_\alpha^{(d)}(y)(X/t)^\alpha
\right],
\]
then
\begin{equation}\label{eq:multidim-uniform-density-remainder-consequence}
\|r_t\|_{L^\infty(\RR^d)}
+\|r_t\|_{L^1(\Omega_d)}
=o(t^{-L}).
\end{equation}
\end{lemma}

\begin{proof}
Lemma~\ref{lem:local-multidim-quotient-taylor-remainder} gives the
coefficient bound and the \(|z|\le1\) part of
\eqref{eq:multidim-uniform-taylor-remainder-bound}.

For the large-\(z\) bound, fix \(|z|>1\) and split according to whether
\(|y-z|\le |z|/2\).  If \(|y-z|\le |z|/2\), then \(|y|\le3|z|/2\), and
\[
F_y(z)\le (1+|y|^2)^a\le C_d|z|^{d+1}.
\]
On the complementary region \(|y-z|>|z|/2\), the elementary inequality
\[
1+|y|^2
\le 2(1+|y-z|^2+|z|^2)
\le 2(1+|z|^2)(1+|y-z|^2)
\]
gives
\[
F_y(z)\le 2^a(1+|z|^2)^a\le C_d|z|^{d+1}.
\]
Thus
\[
\sup_y F_y(z)\le C_d|z|^{d+1}\le C_{d,L}|z|^L
\qquad(|z|>1),
\]
where \(L\ge d+1\) is used only in the final comparison.  The boundedness of
the Taylor coefficients gives, for \(|z|>1\),
\[
\sup_y\sum_{|\alpha|\le L}|U_\alpha^{(d)}(y)|\,|z|^{|\alpha|}
\le C_{d,L}\sum_{k=0}^L |z|^k
\le C_{d,L}'|z|^L.
\]
Consequently
\[
\sup_y\left(
|F_y(z)|
+\sum_{|\alpha|\le L}|U_\alpha^{(d)}(y)|\,|z|^{|\alpha|}
\right)
\le C_{d,L}|z|^L
\qquad(|z|>1),
\]
and this proves the \(|z|>1\) part of
\eqref{eq:multidim-uniform-taylor-remainder-bound}.

With \(z=X/t\), the bound just proved gives, uniformly in \(y\),
\[
|r_t(y)|
\le
C_{d,L}\left[
t^{-(L+1)}\EE\!\left(|X|^{L+1}\mathbf1_{\{|X|\le t\}}\right)
+t^{-L}\EE\!\left(|X|^L\mathbf1_{\{|X|>t\}}\right)
\right].
\]
The two terms have the required order because
\[
t^{-(L+1)}\EE\!\left(|X|^{L+1}\mathbf1_{\{|X|\le t\}}\right)
=
t^{-L}\EE\!\left[
|X|^L\frac{|X|}{t}\mathbf1_{\{|X|\le t\}}
\right]
=o(t^{-L})
\]
by dominated convergence, and
\[
t^{-L}\EE\!\left(|X|^L\mathbf1_{\{|X|>t\}}\right)=o(t^{-L})
\]
by integrability of \(|X|^L\).  These estimates are uniform in \(y\), so they
prove the \(L^\infty\)-part of
\eqref{eq:multidim-uniform-density-remainder-consequence}.  The
\(L^1(\Omega_d)\)-part follows from the same uniform bound because
\(\Omega_d\) is a probability measure.
\end{proof}

\medskip
\noindent\textbf{Multidimensional radial Poisson jet hypothesis
\((\mathrm{MPJ}_{d,L})\).}
For \(d\ge1\) and an integer \(L\), say that \((\nu,d,L)\) satisfies
\((\mathrm{MPJ}_{d,L})\) if
\[
L\ge\max\{2,d+1\},\qquad
\bar{\gamma}=\int_{\RR^d}\gamma\,\dd\nu(\gamma)\ \text{exists},\qquad
\EE|\gamma-\bar{\gamma}|^L<\infty .
\]
The condition \(L\ge d+1\) is part of the analytic hypothesis, not a
notational convention: it is exactly what makes the large-translation bound
in \eqref{eq:multidim-uniform-taylor-remainder-bound} integrable after
averaging over \(X_c/t\).
\medskip

\begin{lemma}[Where the multidimensional finite-\(L\) hypothesis enters]%
\label{lem:multidim-threshold-proof-interface}
In the proof of
Theorems~\ref{thm:multidimensional-density-quotient} and
\ref{thm:multidimensional-poisson-entropy-jets}, the restriction
\(L\ge d+1\) is used only to close the global
\(L^\infty_y\) quotient-remainder estimate in
Lemma~\ref{lem:multidim-uniform-radial-poisson-taylor}.  More precisely, the
large-translation part of the deterministic Taylor remainder is bounded by
\[
\sup_{y\in\RR^d}
\left|
F_y(z)-\sum_{|\alpha|\le L}U_\alpha^{(d)}(y)z^\alpha
\right|
\le C_{d,L}|z|^L
\qquad(|z|>1),
\]
only because \(F_y(z)\) itself can have size \(C_d|z|^{d+1}\) near the
translated pole \(y=z\).  After setting \(z=X_c/t\), the proof therefore
requires
\[
t^{-L}\EE\!\left[
|X_c|^L\mathbf1_{\{|X_c|>t\}}
+|X_c|^L\frac{|X_c|}{t}\mathbf1_{\{|X_c|\le t\}}
\right]=o(t^{-L}),
\]
which is exactly the finite \(L\)th absolute-moment input used in
\eqref{eq:multidim-uniform-density-remainder-consequence}.

Thus the multidimensional entropy statements below are packaged as
finite-\(L\) jet theorems.  The coefficient calculation in
Proposition~\ref{prop:multidim-formal-covariance-layer} is separate from this
remainder step: it is a finite-covariance identity for the formal quadratic
layer.
When \((\mathrm{MPJ}_{d,L})\) is also available, that identity becomes the
leading \(t^{-4}\) coefficient of the finite multidimensional entropy jet by
Corollary~\ref{cor:multidim-leading-kl-coefficient}.
\end{lemma}

\begin{proposition}[Large-translation obstruction for the quotient method]%
\label{prop:multidim-large-translation-obstruction}
Fix \(d\ge1\), put \(a=(d+1)/2\), and let
\[
F_y(z)=
\left(\frac{1+|y|^2}{1+|y-z|^2}\right)^a .
\]
For an integer \(M\ge0\), set
\[
T_M(y,z):=\sum_{|\alpha|\le M}U_\alpha^{(d)}(y)z^\alpha,
\qquad
U_\alpha^{(d)}(y)=\frac1{\alpha!}\partial_z^\alpha F_y(0).
\]
There are constants \(c_{d,M},C_{d,M}>0\) and \(R_{d,M}<\infty\) such that,
for all \(|z|\ge R_{d,M}\),
\begin{equation}\label{eq:multidim-obstruction-linfty-lower}
\|F_\cdot(z)-T_M(\cdot,z)\|_{L^\infty(\RR^d)}
\ge c_{d,M}|z|^{d+1},
\end{equation}
and
\begin{equation}\label{eq:multidim-obstruction-l2-lower}
\|F_\cdot(z)-T_M(\cdot,z)\|_{L^2(\Omega_d)}
\ge c_{d,M}|z|^{(d+1)/2}.
\end{equation}
Consequently the \(L^\infty\)-quotient proof cannot be lowered below the
\(d+1\) tail exponent by a sharper bookkeeping of the same Taylor remainder.
Replacing the uniform norm by an \(L^2(\Omega_d)\) norm improves the
deterministic large-translation exponent only to \((d+1)/2\).  This leaves a
possible low-dimensional refinement in \(d\le3\), but it does not give an
all-dimensional finite-covariance quotient theorem, since in dimensions
\(d\ge4\) the required deterministic exponent is still stronger than
covariance.  Thus Theorems~\ref{thm:multidimensional-density-quotient} and
\ref{thm:multidimensional-poisson-entropy-jets} remain full finite-\(L\)
quotient-jet theorems under \((\mathrm{MPJ}_{d,L})\).  The covariance
calculation in Proposition~\ref{prop:multidim-formal-covariance-layer} does not
use this obstruction; it identifies the formal quadratic covariance layer
itself under the finite second-moment assumption.
\end{proposition}

\begin{proof}
It is enough to work on the ball
\[
B_z:=\{y\in\RR^d:\ |y-z|\le 1/4\}.
\]
For \(y\in B_z\) and \(|z|\ge2\), one has
\(|z|/2\le |y|\le 2|z|\) and \(1+|y-z|^2\le 17/16\).  Hence
\[
F_y(z)
\ge c_d(1+|y|^2)^a
\ge c_d'|z|^{d+1}.
\]
On the same ball the Taylor coefficients at the origin are much smaller than
the translated peak.  Indeed, the derivative structure
\eqref{eq:multidim-poisson-derivative-structure}, with \(w=0\), gives
\[
|U_\alpha^{(d)}(y)|
\le C_{d,\alpha}(1+|y|)^{-|\alpha|}
\qquad(|y|\ge1),
\]
and therefore
\[
|U_\alpha^{(d)}(y)z^\alpha|
\le C_{d,\alpha}
\qquad (y\in B_z,\ |z|\ge2).
\]
For fixed \(M\), \(|T_M(y,z)|\le C_{d,M}\) on \(B_z\).  Increasing
\(R_{d,M}\) if necessary, the translated peak dominates this bounded Taylor
polynomial, and
\[
|F_y(z)-T_M(y,z)|\ge c_{d,M}|z|^{d+1}
\qquad (y\in B_z,\ |z|\ge R_{d,M}).
\]
This proves the \(L^\infty\) lower bound.

For the \(L^2(\Omega_d)\) bound, use the same pointwise lower bound on
\(B_z\) and the radial Poisson weight
\(\Omega_d(\dd y)=c_d(1+|y|^2)^{-(d+1)/2}\dd y\).  Since
\(|z|/2\le |y|\le2|z|\) on \(B_z\),
\[
\Omega_d(B_z)\ge c_d''|z|^{-(d+1)}.
\]
Thus
\[
\|F_\cdot(z)-T_M(\cdot,z)\|_{L^2(\Omega_d)}^2
\ge c_{d,M}|z|^{2(d+1)}\Omega_d(B_z)
\ge c_{d,M}'|z|^{d+1},
\]
which is \eqref{eq:multidim-obstruction-l2-lower} after taking square roots.
\end{proof}

\begin{theorem}[Finite-\(L\) multidimensional density quotient expansion, \(L\ge d+1\)]%
\label{thm:multidimensional-density-quotient}
Let \(d\ge1\), let
\[
P_t^{(d)}(x):=c_d\,\frac{t}{(t^2+|x|^2)^{(d+1)/2}},
\qquad x\in\RR^d,\ t>0,
\]
be the radial Poisson kernel on \(\RR^d\), normalized to integrate to one.
Let \(\nu\) be a probability measure on \(\RR^d\).  Fix an integer \(L\)
such that \((\nu,d,L)\) satisfies
\((\mathrm{MPJ}_{d,L})\), and put
\(X_c:=\gamma-\bar{\gamma}\).  Set
\[
h_t:=P_t^{(d)}*\nu,\qquad
g_t(x):=P_t^{(d)}(x-\bar{\gamma}).
\]
For multiindices \(\alpha\in\NN_0^d\), write
\[
\mu_\alpha:=\EE[X_c^\alpha]\quad(|\alpha|\le L).
\]
Let
\[
\Omega_d(\dd y):=P_1^{(d)}(y)\dd y
\]
and define the universal functions \(U_\alpha^{(d)}\) by the Taylor expansion
at \(z=0\)
\begin{equation}\label{eq:multidim-poisson-ratio-taylor}
\left(\frac{1+|y|^2}{1+|y-z|^2}\right)^{(d+1)/2}
=\sum_{\alpha\in\NN_0^d}U_\alpha^{(d)}(y)z^\alpha .
\end{equation}
Then, for \(x=\bar{\gamma}+ty\), the Cauchy-scale density quotient satisfies
\begin{equation}\label{eq:multidim-density-expansion}
\frac{h_t(\bar{\gamma}+ty)}{g_t(\bar{\gamma}+ty)}-1
=\sum_{j=2}^{L}D_j^{(d)}(y;\nu)t^{-j}
+\rho_{L,t}(y),
\qquad
\|\rho_{L,t}\|_\infty+\|\rho_{L,t}\|_{L^1(\Omega_d)}=o(t^{-L}),
\end{equation}
where, for \(2\le j\le L\),
\begin{equation}\label{eq:multidim-density-layers}
D_j^{(d)}(y;\nu):=
\sum_{|\alpha|=j}\mu_\alpha U_\alpha^{(d)}(y).
\end{equation}
Lemma~\ref{lem:multidim-uniform-radial-poisson-taylor} supplies, uniformly
in \(y\), both the finite Taylor remainder bound
\eqref{eq:multidim-uniform-taylor-remainder-bound} and its
\(L^\infty\cap L^1(\Omega_d)\) integrated consequence
\eqref{eq:multidim-uniform-density-remainder-consequence}; these are the only
large-\(z\) and density-remainder estimates used here.
\end{theorem}

\begin{proof}
Write \(a=(d+1)/2\).  For \(x=\bar{\gamma}+ty\),
\[
\frac{P_t^{(d)}(x-\gamma)}{g_t(x)}
=
\left(\frac{t^2+|x-\bar{\gamma}|^2}{t^2+|x-\gamma|^2}\right)^a
=
\left(\frac{1+|y|^2}{1+|y-X_c/t|^2}\right)^a.
\]
Thus
\[
\widetilde\delta_t(y)
:=\frac{h_t(\bar{\gamma}+ty)}{g_t(\bar{\gamma}+ty)}-1
=\EE\!\left[
\left(\frac{1+|y|^2}{1+|y-X_c/t|^2}\right)^a-1
\right].
\]
Apply Lemma~\ref{lem:multidim-uniform-radial-poisson-taylor} with
\(X=X_c\).  Its uniform Taylor-remainder consequence gives the error term
below in both
\(L^\infty(\RR^d)\) and \(L^1(\Omega_d)\).  Since \(U_0^{(d)}=1\) and the
first-order terms integrate to zero against the centred variable \(X_c\), we
obtain \eqref{eq:multidim-density-expansion}.  The proof of the stated
remainder is exactly the tail split in the Taylor lemma, with \(z=X_c/t\):
the region \(|X_c|\le t\) gives the integrable factor
\(|X_c|^L(|X_c|/t)\), while \(|X_c|>t\) gives the finite \(L\)th-moment tail.
\end{proof}

\begin{lemma}[Standalone multidimensional density-to-entropy transfer]%
\label{lem:multidim-density-to-entropy-transfer}
Let \((S,\Omega)\) be a probability space, let \(L\ge2\), and let
\(D_2,\ldots,D_L\in L^\infty(\Omega)\).  Suppose that, for all sufficiently
large \(t\),
\[
\delta_t=\sum_{j=2}^{L}D_jt^{-j}+\rho_t,
\]
where
\[
\|\rho_t\|_\infty+\|\rho_t\|_{L^1(\Omega)}=o(t^{-L})
\]
and
\[
\|\delta_t\|_\infty=O(t^{-2}).
\]
Then, for
\[
\Phi(s)=(1+s)\log(1+s)-s,
\]
\begin{equation}\label{eq:abstract-multidim-transfer-coefficients}
\mathfrak A_m(D_2,\ldots,D_L)
:=
\sum_{r=2}^{\lfloor m/2\rfloor}
\frac{(-1)^r}{r(r-1)}
\sum_{\substack{2\le j_1,\ldots,j_r\le L\\
j_1+\cdots+j_r=m}}
\int_S D_{j_1}\cdots D_{j_r}\,\dd\Omega
\end{equation}
one has
\begin{equation}\label{eq:abstract-multidim-transfer-expansion}
\int_S\Phi(\delta_t)\,\dd\Omega
=
\sum_{m=4}^{L+2}\mathfrak A_m(D_2,\ldots,D_L)t^{-m}
+o(t^{-(L+2)}).
\end{equation}
\end{lemma}

\begin{proof}
Put
\[
B_t:=\sum_{j=2}^{L}D_jt^{-j}.
\]
Since the \(D_j\)'s are bounded and \(\Omega\) has total mass one,
\(\|B_t\|_\infty=O(t^{-2})\), and every finite product appearing in
\eqref{eq:abstract-multidim-transfer-coefficients} belongs to
\(L^1(\Omega)\).  The assumed smallness of \(\delta_t\) gives
\(|\delta_t|\le1/2\) for large \(t\).  Let
\[
R:=\left\lfloor\frac{L+2}{2}\right\rfloor .
\]
For \(|s|\le1/2\),
\[
\Phi(s)=\sum_{r=2}^{R}\frac{(-1)^r}{r(r-1)}s^r+O(|s|^{R+1}).
\]
Because \(\|\delta_t\|_\infty=O(t^{-2})\), the Taylor tail has
\[
\int_SO(|\delta_t|^{R+1})\,\dd\Omega
=O(t^{-2(R+1)})=o(t^{-(L+2)}).
\]

It remains to replace \(\delta_t^r\) by \(B_t^r\), for \(2\le r\le R\).  In
\[
\delta_t^r-B_t^r=(B_t+\rho_t)^r-B_t^r
\]
each summand contains at least one factor \(\rho_t\).  Hence, using one
\(\rho_t\) in \(L^1(\Omega)\) and all other factors in \(L^\infty\),
\[
\|\delta_t^r-B_t^r\|_{L^1(\Omega)}
\le
\sum_{\ell=1}^{r}\binom r\ell
\|B_t\|_\infty^{r-\ell}
\|\rho_t\|_\infty^{\ell-1}
\|\rho_t\|_{L^1(\Omega)}
=o(t^{-L}t^{-2(r-1)}).
\]
Since \(r\ge2\), this is \(o(t^{-(L+2)})\).  Therefore
\[
\int_S\Phi(\delta_t)\,\dd\Omega
=
\sum_{r=2}^{R}\frac{(-1)^r}{r(r-1)}
\int_SB_t^r\,\dd\Omega
+o(t^{-(L+2)}).
\]

Finally expand the finite polynomial:
\[
\int_SB_t^r\,\dd\Omega
=
\sum_{m=2r}^{Lr}t^{-m}
\sum_{\substack{2\le j_1,\ldots,j_r\le L\\
j_1+\cdots+j_r=m}}
\int_SD_{j_1}\cdots D_{j_r}\,\dd\Omega .
\]
The terms with \(m\le L+2\), summed over \(2\le r\le R\), give exactly the
coefficients in \eqref{eq:abstract-multidim-transfer-coefficients}.  The
remaining finite terms have integer exponents \(m\ge L+3\), hence contribute
\(O(t^{-(L+3)})=o(t^{-(L+2)})\).  This proves
\eqref{eq:abstract-multidim-transfer-expansion}.
\end{proof}

\begin{theorem}[Finite-\(L\) multidimensional entropy transfer, \(L\ge d+1\)]%
\label{thm:multidimensional-poisson-entropy-jets}
Assume the hypotheses and notation of
Proposition~\ref{thm:multidimensional-density-quotient}.  Thus
\((\nu,d,L)\) satisfies \((\mathrm{MPJ}_{d,L})\), \(h_t=P_t^{(d)}*\nu\),
\(g_t(x)=P_t^{(d)}(x-\bar{\gamma})\), and the density layers
\(D_j^{(d)}\) are those in \eqref{eq:multidim-density-layers}.  For
\(4\le m\le L+2\), define \(\mathcal A_m^{(d)}(\nu)\) by
\begin{equation}\label{eq:multidim-entropy-coefficient-layers}
\mathcal A_m^{(d)}(\nu)=
\sum_{r=2}^{\lfloor m/2\rfloor}
\frac{(-1)^r}{r(r-1)}
\sum_{\substack{2\le j_1,\ldots,j_r\le L\\
j_1+\cdots+j_r=m}}
\int_{\RR^d}D_{j_1}^{(d)}(y;\nu)\cdots
D_{j_r}^{(d)}(y;\nu)\,\Omega_d(\dd y).
\end{equation}
Then
\begin{equation}\label{eq:multidim-poisson-entropy-jet}
D_{\rm KL}(h_t\|g_t)
=\sum_{m=4}^{L+2}\mathcal A_m^{(d)}(\nu)t^{-m}
+o(t^{-(L+2)})
\qquad(t\to\infty).
\end{equation}
\end{theorem}

\begin{proof}
The change of variables gives
\[
g_t(\bar{\gamma}+ty)t^d\dd y=P_1^{(d)}(y)\dd y=\Omega_d(\dd y),
\]
and therefore, with
\(\widetilde\delta_t(y)=h_t(\bar{\gamma}+ty)/g_t(\bar{\gamma}+ty)-1\),
the density masses cancel in the same variables:
\[
\int_{\RR^d}\widetilde\delta_t(y)\,\Omega_d(\dd y)
=\int_{\RR^d}\bigl(h_t(x)-g_t(x)\bigr)\,\dd x=0.
\]
Hence
\[
D_{\rm KL}(h_t\|g_t)
=\int_{\RR^d}\Phi(\widetilde\delta_t(y))\,\Omega_d(\dd y),
\qquad
\Phi(s)=(1+s)\log(1+s)-s.
\]
By Proposition~\ref{thm:multidimensional-density-quotient},
\[
\widetilde\delta_t(y)
=\sum_{j=2}^{L}D_j^{(d)}(y;\nu)t^{-j}
+\rho_{L,t}(y),
\qquad
\|\rho_{L,t}\|_\infty+\|\rho_{L,t}\|_{L^1(\Omega_d)}=o(t^{-L}).
\]
The finite sum starts at order \(t^{-2}\).  For each \(2\le j\le L\), the
retained layer \(D_j^{(d)}\) is bounded because it is a finite linear
combination of the bounded Taylor modes \(U_\alpha^{(d)}\) from
Lemma~\ref{lem:multidim-uniform-radial-poisson-taylor}, with finite
coefficients \(\mu_\alpha\), \(|\alpha|=j\).  Together with
\(\|\rho_{L,t}\|_\infty=o(t^{-L})\), this gives
\(\|\widetilde\delta_t\|_\infty=O(t^{-2})\).  Therefore the tuple
\[
(S,\Omega,\delta_t,D_2,\ldots,D_L,\rho_t)
=
(\RR^d,\Omega_d,\widetilde\delta_t,
D_2^{(d)}(\cdot;\nu),\ldots,D_L^{(d)}(\cdot;\nu),\rho_{L,t})
\]
satisfies every hypothesis of the standalone transfer lemma
Lemma~\ref{lem:multidim-density-to-entropy-transfer}.  The abstract
coefficients
\(\mathfrak A_m(D_2^{(d)},\ldots,D_L^{(d)})\) in
\eqref{eq:abstract-multidim-transfer-coefficients} are precisely the
\(\mathcal A_m^{(d)}(\nu)\) defined in
\eqref{eq:multidim-entropy-coefficient-layers}.  The lemma therefore proves
\eqref{eq:multidim-poisson-entropy-jet}.
\end{proof}

\begin{corollary}[Multidimensional coefficient formula]%
\label{cor:multidim-coefficient-formula}
Assume the hypotheses and notation of
Proposition~\ref{thm:multidimensional-density-quotient}.  For \(4\le m\le L+2\),
the coefficient in Theorem~\ref{thm:multidimensional-poisson-entropy-jets}
is equivalently
\begin{equation}\label{eq:multidim-entropy-coefficient}
\mathcal A_m^{(d)}(\nu):=
\sum_{r=2}^{\lfloor m/2\rfloor}
\frac{(-1)^r}{r(r-1)}
\sum_{\substack{|\alpha_1|,\ldots,|\alpha_r|\ge2\\
|\alpha_1|+\cdots+|\alpha_r|=m\\
|\alpha_j|\le L}}
\mu_{\alpha_1}\cdots\mu_{\alpha_r}
\int_{\RR^d}\prod_{j=1}^rU_{\alpha_j}^{(d)}(y)\,\Omega_d(\dd y).
\end{equation}
The coefficient is finite and universal: it depends on \(d\), \(m\), and the
centred tensor moments
\(\{\mu_\alpha:2\le|\alpha|\le m-2\}\), but not on any higher moment of
\(\nu\).  In particular a contribution of total entropy degree \(m\) cannot
contain a tensor moment of order above \(m-2\).
\end{corollary}

\begin{proof}
Substitute the density-layer definition
\eqref{eq:multidim-density-layers} into
\eqref{eq:multidim-entropy-coefficient-layers} and collect multiindices by
homogeneous degree.  Since every entropy monomial has at least two density
factors, each individual multiindex in a total degree \(m\) contribution has
length at most \(m-2\).
\end{proof}

\begin{lemma}[Multidimensional parity and low-order display]%
\label{lem:multidim-parity-low-order-display}
Assume the hypotheses and notation of
Theorems~\ref{thm:multidimensional-density-quotient} and
\ref{thm:multidimensional-poisson-entropy-jets}.  Then all odd entropy
coefficients vanish:
\begin{equation}\label{eq:multidim-odd-parity-cancellation}
\mathcal A_{2q+1}^{(d)}(\nu)=0
\qquad(2q+1\le L+2).
\end{equation}
The first potentially nonzero coefficients are
\begin{align}
\mathcal A_4^{(d)}
&=\frac12\int_{\RR^d}D_2^2\,\dd\Omega_d,
\label{eq:multidim-first-coeff-a4}\\
\mathcal A_6^{(d)}
&=\int_{\RR^d}D_2D_4\,\dd\Omega_d
+\frac12\int_{\RR^d}D_3^2\,\dd\Omega_d
-\frac16\int_{\RR^d}D_2^3\,\dd\Omega_d,
\label{eq:multidim-first-coeff-a6}\\
\mathcal A_8^{(d)}
&=\int_{\RR^d}D_2D_6\,\dd\Omega_d
+\int_{\RR^d}D_3D_5\,\dd\Omega_d
+\frac12\int_{\RR^d}D_4^2\,\dd\Omega_d
\notag\\
&\quad
-\frac12\int_{\RR^d}D_2^2D_4\,\dd\Omega_d
-\frac12\int_{\RR^d}D_2D_3^2\,\dd\Omega_d
+\frac1{12}\int_{\RR^d}D_2^4\,\dd\Omega_d,
\label{eq:multidim-first-coeff-a8}
\end{align}
with the convention that a displayed term containing \(D_j\) is omitted when
\(j>L\).  Here \(D_j=D_j^{(d)}(\,\cdot\,;\nu)\).

The leading term is coercive in the covariance:
Proposition~\ref{prop:multidim-formal-covariance-layer} gives an explicit positive
decomposition of \(\mathcal A_4^{(d)}\), so
\(\mathcal A_4^{(d)}\ge0\) and equality holds exactly for zero covariance.
No fixed sign is asserted for the higher even coefficients; the formulas
\eqref{eq:multidim-first-coeff-a6}--\eqref{eq:multidim-first-coeff-a8}
display the alternating Bregman-Taylor contributions that destroy a general
positivity statement beyond the leading quadratic layer.

For even \(m\), the tensor-moment order \(m-2\) enters the universal
coefficient \(\mathcal A_m^{(d)}\) only through the quadratic-layer pairing
\(\int D_2D_{m-2}\,\dd\Omega_d\).  This is the algebraic sharpness statement
for the displayed universal coefficient; the detected part of the top tensor
is the component paired with \(D_2\).
\end{lemma}

\begin{remark}[Notation for the leading multidimensional layer]
The finite-jet coefficient in
\eqref{eq:multidim-first-coeff-a4} is denoted
\(\mathcal A_4^{(d)}\) because it is part of the full
\((\mathrm{MPJ}_{d,L})\) entropy jet.  The same quadratic expression is
denoted \(\mathcal Q_d\) in
Proposition~\ref{prop:multidim-formal-covariance-layer} when only finite
covariance is assumed.  Corollary~\ref{cor:multidim-leading-kl-coefficient}
is the separate step that turns this formal layer into a leading KL coefficient
under \((\mathrm{MPJ}_{d,L})\).
\end{remark}

\begin{proof}[Proof of the parity and low-order display]
The identity \(F_{-y}(z)=F_y(-z)\) in
\eqref{eq:multidim-poisson-ratio-taylor} implies
\[
U_\alpha^{(d)}(-y)=(-1)^{|\alpha|}U_\alpha^{(d)}(y),
\qquad
D_j^{(d)}(-y;\nu)=(-1)^jD_j^{(d)}(y;\nu).
\]
Since \(\Omega_d\) is even, every product in
\eqref{eq:multidim-entropy-coefficient-layers} with odd total degree has zero
integral.  This proves \eqref{eq:multidim-odd-parity-cancellation}.  The
formulas \eqref{eq:multidim-first-coeff-a4}--\eqref{eq:multidim-first-coeff-a8}
are obtained by listing the ordered compositions of \(m\) by parts at least
two and applying the Taylor coefficients \(1/2,-1/6,1/12\) of \(\Phi\).
For even \(m\), the only compositions containing a part \(m-2\) are
\((2,m-2)\) and \((m-2,2)\); their combined contribution is
\(\int D_2D_{m-2}\,\dd\Omega_d\).  All remaining compositions use only
density layers of order at most \(m-3\).
\end{proof}

\begin{proof}[Moment identities used in Proposition~\ref{prop:multidim-formal-covariance-layer}]
Let \(Y\) have density \(\Omega_d(\dd y)=P_1^{(d)}(y)\dd y\), set
\(R=|Y|^2\), and put
\[
D_d:=(d+1)(d+3)(d+5)(d+7).
\]
Then
\[
\EE\frac{1}{(1+R)^4}=\frac{105}{D_d},\qquad
\EE\frac{R}{(1+R)^4}=\frac{15d}{D_d},\qquad
\EE\frac{R^2}{(1+R)^4}=\frac{3d(d+2)}{D_d}.
\]
Also, if \(\Theta\) is uniformly distributed on the unit sphere in \(\RR^d\)
and \(A\) is a real symmetric traceless \(d\times d\) matrix, then
\[
\EE_\Theta\bigl[(\Theta^{\mathsf T}A\Theta)^2\bigr]
=\frac{2}{d(d+2)}\|A\|_{\rm HS}^2 .
\]
Indeed, since \(\Omega_d\) is radial, the polar decomposition
\(Y=\sqrt R\,\Theta\) has \(\Theta\) uniform on the sphere, independent of
\(R\), and \(R\sim{\rm BetaPrime}(d/2,1/2)\).  Hence
\begin{equation}\label{eq:radial-beta-prime-moments}
\EE\frac{R^k}{(1+R)^4}
=
\frac{B\bigl(d/2+k,\,9/2-k\bigr)}
{B\bigl(d/2,\,1/2\bigr)}
\qquad(k=0,1,2).
\end{equation}
Expanding the beta functions gives the three displayed rational moments; in
particular the \(d=1,\ k=0\) value is \(35/128\).
The angular identity follows by contracting
\[
\EE_\Theta[\Theta_i\Theta_j\Theta_k\Theta_l]
=\frac{\delta_{ij}\delta_{kl}
+\delta_{ik}\delta_{jl}
+\delta_{il}\delta_{jk}}{d(d+2)}
\]
with the entries of \(A\), and then using \(\operatorname{tr}A=0\).
\end{proof}


\begin{proposition}[Algebraic covariance quadratic form under finite covariance]%
\label{prop:multidim-formal-covariance-layer}
Let \(d\ge1\), and let \(\nu\) be a probability measure on \(\RR^d\) with
finite centred second moment.  Put
\[
\bar{\gamma}=\int_{\RR^d}\gamma\,\dd\nu(\gamma),
\qquad
X_c=\gamma-\bar{\gamma},
\]
and let
\[
\Sigma:=\EE[X_cX_c^{\mathsf T}]
\]
be the centred covariance matrix.  Put
\[
\Sigma_0:=\Sigma-\frac{\operatorname{tr}\Sigma}{d}I_d .
\]
Let \(b_\Sigma\) denote the degree-two centred density mode obtained from the
radial Poisson quotient, explicitly
\begin{equation}\label{eq:multidim-leading-theorem-density-mode}
b_\Sigma(y)
 =
\frac{d+1}{2}\left(
-\frac{\operatorname{tr}\Sigma}{1+|y|^2}
+\frac{(d+3)y^{\mathsf T}\Sigma y}{(1+|y|^2)^2}
\right).
\end{equation}
The algebraic quadratic entropy-integrand layer determined by this mode is
the finite quantity
\begin{equation}\label{eq:multidim-leading-positive-defect}
\mathcal Q_d(\Sigma)
:=\frac12\int_{\RR^d}b_\Sigma(y)^2\,\Omega_d(\dd y)
=
c_d^{\rm iso}\,(\operatorname{tr}\Sigma)^2
+c_d^{\rm tr}\,\|\Sigma_0\|_{\rm HS}^2,
\end{equation}
where
\begin{equation}\label{eq:multidim-leading-positive-constants}
c_d^{\rm iso}:=
\frac{3(d+1)(7d+9)}
{4d(d+3)(d+5)(d+7)},
\qquad
c_d^{\rm tr}:=
\frac{3(d+1)(d+3)}
{4(d+5)(d+7)} .
\end{equation}
Two immediate checks are
\[
d=1:\qquad \Sigma_0=0,\qquad c_1^{\rm iso}=\frac18,\qquad
\mathcal Q_1(\Sigma)=\frac{(\operatorname{tr}\Sigma)^2}{8},
\]
and, for an isotropic covariance \(\Sigma=\tau I_d\),
\[
\Sigma_0=0,\qquad
\mathcal Q_d(\Sigma)
=c_d^{\rm iso}d^2\tau^2 .
\]
This identity assumes only \(\EE|X_c|^2<\infty\) and is a covariance-layer
calculation.  Finite covariance alone identifies the quadratic form below;
the present proof of the KL asymptotic uses the stronger quotient-jet
hypothesis \((\mathrm{MPJ}_{d,L})\), supplied here by finite centred
\(L\)-moment with \(L\ge d+1\).  The asymptotic statement
\[
D_{\rm KL}\bigl(P_t^{(d)}*\nu\,\|\,P_t^{(d)}(\cdot-\bar{\gamma})\bigr)
=\mathcal Q_d(\Sigma)t^{-4}+o(t^{-4})
\]
is Corollary~\ref{cor:multidim-leading-kl-coefficient}.
Both constants are strictly positive.  Consequently
\(\mathcal Q_d(\Sigma)\ge0\), and
\(\mathcal Q_d(\Sigma)=0\) if and only if
\(\Sigma=0\).  Thus the formal layer is the first potentially nonzero
quadratic matched-entropy layer.  In the degenerate zero-covariance case this
formal layer, and the whole matched comparison, is zero.  In dimension \(d=1\)
this gives
\(\mathcal Q_1(\Sigma)=\mu_2^2/8\), which is the leading coefficient in
Theorem~\ref{thm:poisson-kl-even-orders} under its one-dimensional finite-jet
hypotheses, and also agrees with the separate sharp leading threshold theorem.
The analytic role of the dimension-dependent moment hypothesis is isolated in
Lemma~\ref{lem:multidim-threshold-proof-interface} and
Proposition~\ref{prop:multidim-large-translation-obstruction}: the
quotient-remainder proof must dominate translated peaks before the nonlinear
entropy transfer is applied, and those peaks have \(L^\infty\) size
\(|z|^{d+1}\) and \(L^2(\Omega_d)\) size \(|z|^{(d+1)/2}\).  That role
concerns the construction of the finite jet, not the formal covariance-layer
calculation.  The radial beta-prime integrals and the traceless spherical
fourth moment used to obtain the two displayed constants are recorded
immediately before this proposition.
\end{proposition}

\begin{proof}
First compute the formal quadratic density layer under only a second moment.  The
degree-two Taylor coefficient below is bounded as a function of \(y\), so
averaging it against \(X_c\) uses only
\(\EE|X_c|^2=\operatorname{tr}\Sigma<\infty\).  When the finite quotient jet
exists, Lemma~\ref{lem:multidim-parity-low-order-display} identifies the
\(t^{-4}\) entropy coefficient as
\[
\mathcal Q_d(\Sigma)
=\frac12\int_{\RR^d} b_\Sigma(y)^2\,\Omega_d(\dd y),
\]
where \(b_\Sigma\) is this degree-two density mode.  The same integral is a
well-defined formal quadratic layer of the KL integrand whenever the
covariance is finite.  The integral is finite because \(b_\Sigma\) is bounded.
It remains to compute the coefficient and diagonalize the resulting quadratic
form on symmetric matrices.

Let \(a=(d+1)/2\) and \(A=1+|y|^2\).  For a deterministic vector \(z\), the
exact quotient is
\[
F_y(z):=
\left(\frac{1+|y|^2}{1+|y-z|^2}\right)^a
=
\left(1+\frac{-2y\cdot z+|z|^2}{A}\right)^{-a}.
\]
Its second-order Taylor polynomial at \(z=0\) is
\[
F_y(z)
=1+\frac{2a\,y\cdot z-a|z|^2}{A}
+\frac{a(a+1)}{2}\frac{4(y\cdot z)^2}{A^2}
+O_y(|z|^3).
\]
Equivalently, the homogeneous degree-two part is exactly
\begin{equation}\label{eq:multidim-leading-homogeneous-degree-two}
\operatorname{hom}_2F_y(z)
=
a\left(
-\frac{|z|^2}{A}
+\frac{(d+3)(y\cdot z)^2}{A^2}
\right),
\end{equation}
because \(2(a+1)=d+3\).
Only the homogeneous degree-two part contributes to the quadratic covariance
layer: the degree-zero term is the reference density, and the degree-one term
vanishes after centering.  With \(z=X_c/t\),
\[
\EE |X_c|^2=\operatorname{tr}\Sigma,\qquad
\EE (y\cdot X_c)^2=y^{\mathsf T}\Sigma y,
\]
so the \(t^{-2}\) density coefficient is
\begin{equation}\label{eq:multidim-leading-density-coefficient}
b_\Sigma(y)
=
a\left(
-\frac{\operatorname{tr}\Sigma}{A}
+\frac{(d+3)y^{\mathsf T}\Sigma y}{A^2}
\right).
\end{equation}
Write \(\Sigma=\tau I_d+\Sigma_0\), where
\(\tau=(\operatorname{tr}\Sigma)/d\) and
\(\operatorname{tr}\Sigma_0=0\).  Then
\begin{equation}\label{eq:multidim-leading-isotropic-traceless-split}
b_\Sigma(y)
=a\tau\,\frac{3|y|^2-d}{(1+|y|^2)^2}
+a(d+3)\frac{y^{\mathsf T}\Sigma_0y}{(1+|y|^2)^2}.
\end{equation}
The mixed integral is zero by rotational symmetry.  Indeed, with
\(Y=\sqrt R\,\Theta\) and \(\Theta\) uniform on the sphere,
\[
\EE_\Theta[\Theta^{\mathsf T}\Sigma_0\Theta]
=\frac{\operatorname{tr}\Sigma_0}{d}=0,
\]
and therefore
\[
\int_{\RR^d}
\frac{(3|y|^2-d)y^{\mathsf T}\Sigma_0y}
{(1+|y|^2)^4}\,\Omega_d(\dd y)
=
\EE\!\left[
\frac{(3R-d)R}{(1+R)^4}
\right]\EE_\Theta[\Theta^{\mathsf T}\Sigma_0\Theta]
=0 .
\]

It remains to compute the two constants.  Let \(R=|Y|^2\) for
\(Y\sim\Omega_d\).  The beta-prime moments in
\eqref{eq:radial-beta-prime-moments} give
\[
\begin{split}
\EE\frac{(3R-d)^2}{(1+R)^4}
\ &=
9\EE\frac{R^2}{(1+R)^4}
-6d\EE\frac{R}{(1+R)^4}
+d^2\EE\frac1{(1+R)^4}\\
\ &=
\frac{6d(7d+9)}
{(d+1)(d+3)(d+5)(d+7)}.
\end{split}
\]
The isotropic contribution to \(\frac12\int b_\Sigma^2\dd\Omega_d\) is
\[
\begin{aligned}
\frac12 a^2\tau^2
\EE\frac{(3R-d)^2}{(1+R)^4}
&=
\frac12\frac{(d+1)^2}{4}
\frac{(\operatorname{tr}\Sigma)^2}{d^2}
\frac{6d(7d+9)}
{(d+1)(d+3)(d+5)(d+7)}\\
&=
\frac{3(d+1)(7d+9)}
{4d(d+3)(d+5)(d+7)}
(\operatorname{tr}\Sigma)^2
=c_d^{\rm iso}(\operatorname{tr}\Sigma)^2 .
\end{aligned}
\]

For the traceless part, use the independent polar decomposition
\(Y=\sqrt R\,\Theta\).  The spherical fourth-moment identity recorded above,
applied to the traceless matrix \(\Sigma_0\), and the \(k=2\) radial moment
give
\[
\begin{split}
\int_{\RR^d}
\frac{(y^{\mathsf T}\Sigma_0y)^2}{(1+|y|^2)^4}\,\Omega_d(\dd y)
\ &=
\EE\frac{R^2}{(1+R)^4}\,
\EE_\Theta[(\Theta^{\mathsf T}\Sigma_0\Theta)^2]\\
\ &=
\frac{3d(d+2)}{(d+1)(d+3)(d+5)(d+7)}
\frac{2}{d(d+2)}
\|\Sigma_0\|_{\rm HS}^2\\
\ &=
\frac{6}{(d+1)(d+3)(d+5)(d+7)}
\|\Sigma_0\|_{\rm HS}^2.
\end{split}
\]
Multiplying by \(\frac12a^2(d+3)^2\) gives
\[
\frac12a^2(d+3)^2
\frac{6}{(d+1)(d+3)(d+5)(d+7)}
=
\frac{3(d+1)(d+3)}{4(d+5)(d+7)}
=c_d^{\rm tr}.
\]
This proves \eqref{eq:multidim-leading-positive-defect}.
The constants are positive for every \(d\ge1\).  If the coefficient is zero,
then \(\operatorname{tr}\Sigma=0\) and \(\Sigma_0=0\), hence
\(\Sigma=0\).  Since \(\Sigma\) is positive semidefinite, this is equivalent
to \(X_c=0\) almost surely.  The one-dimensional specialization has
\(\Sigma_0=0\), \(\operatorname{tr}\Sigma=\mu_2\), and
\[
c_1^{\rm iso}
=\frac{3\cdot2\cdot16}{4\cdot1\cdot4\cdot6\cdot8}
=\frac18 .
\]
\end{proof}

\begin{corollary}[Leading KL coefficient under \((\mathrm{MPJ}_{d,L})\)]%
\label{cor:multidim-leading-kl-coefficient}
Let \(d\ge1\), let \(L\) be an integer, and assume that
\((\nu,d,L)\) satisfies \((\mathrm{MPJ}_{d,L})\).  With
\(h_t=P_t^{(d)}*\nu\) and
\(g_t=P_t^{(d)}(\cdot-\bar{\gamma})\),
\[
D_{\rm KL}(h_t\|g_t)
=\mathcal Q_d(\Sigma)t^{-4}+o(t^{-4}),
\]
where \(\mathcal Q_d(\Sigma)\) is the covariance quadratic form in
Proposition~\ref{prop:multidim-formal-covariance-layer}.
\end{corollary}

\begin{proof}
The hypothesis \((\mathrm{MPJ}_{d,L})\) is exactly the hypothesis of
Theorem~\ref{thm:multidimensional-poisson-entropy-jets}: it packages
\(L\ge\max\{2,d+1\}\), existence of the barycentre, and finite centred
\(L\)th absolute moment.  That theorem gives the multidimensional entropy
expansion with first even layer
\(\mathcal A_4^{(d)}\), and
Lemma~\ref{lem:multidim-parity-low-order-display} identifies
\[
\mathcal A_4^{(d)}
=\frac12\int_{\RR^d}D_2(y;\nu)^2\,\Omega_d(\dd y).
\]
For centred data the degree-two density layer \(D_2\) is precisely the bounded
mode \(b_\Sigma\) computed in
Proposition~\ref{prop:multidim-formal-covariance-layer}.  Hence
\(\mathcal A_4^{(d)}=\mathcal Q_d(\Sigma)\), giving the displayed
leading term.  The higher coefficients and the remainder are those supplied by
Theorem~\ref{thm:multidimensional-poisson-entropy-jets}; no part of this
corollary weakens its hypotheses.
\end{proof}

\begin{theorem}[Finite-moment Poisson entropy transfer and coefficient formula]%
\label{thm:poisson-kl-even-orders}
Let \(N\ge2\), put \(L:=2N-2\), and let \(\nu\) be a probability measure
on~\(\RR\) with mean \(\bar{\gamma}\) and finite centred \(L\)th absolute
moment.  Set
\[
X_c:=\gamma-\bar{\gamma},
\qquad
\mu_n:=\int_{\RR}X_c^n\dd\nu(\gamma)\quad\text{whenever it exists},
\]
and define
\[
h_t:=P_t*\nu,
\qquad
g_t(x):=P_t(x-\bar{\gamma}).
\]
For \(x=\bar{\gamma}+ty\), define
\[
\delta_t(y):=\frac{h_t(\bar{\gamma}+ty)}{g_t(\bar{\gamma}+ty)}-1 .
\]
Then the finite-moment mechanism is the following.

\textup{(a) Density quotient.}
The exact Cauchy quotient expansion of
Lemma~\ref{lem:exact-moment-density-truncation} gives
\[
\delta_t(y)
=\sum_{n=2}^{L}\mu_nu_n(y)t^{-n}+\rho_t(y),
\qquad
\|\rho_t\|_\infty+\|\rho_t\|_{L^1(\omega)}=o(t^{-L}),
\]
where \(\omega(\dd y)=\pi^{-1}(1+y^2)^{-1}\dd y\) and the linear mode is
absent because \(X_c\) is centred.

\textup{(b) Finite \(\Phi\)-transfer.}
With \(\Phi(s)=(1+s)\log(1+s)-s\),
\[
D_{\mathrm{KL}}(h_t\|g_t)
=\int_{\RR}\Phi(\delta_t(y))\,\omega(\dd y),
\]
and this identity transfers the density expansion to an entropy expansion
through order \(t^{-2N}\):
\begin{equation}\label{eq:poisson-kl-even-orders}
D_{\mathrm{KL}}(h_t\|g_t)
=\sum_{k=4}^{2N}A_k(\nu)t^{-k}+o(t^{-2N})
=\sum_{m=2}^{N}A_{2m}(\nu)t^{-2m}+o(t^{-2N})
\qquad(t\to\infty).
\end{equation}

\textup{(c) Coefficient formula.}
For \(4\le m\le2N\),
\begin{equation}\label{eq:coefficient-identification-transfer-expansion}
A_m(\nu)=
\sum_{r=2}^{\lfloor m/2\rfloor}
\frac{(-1)^r}{r(r-1)}
\sum_{\substack{n_1+\cdots+n_r=m\\2\le n_j\le 2N-2}}
\mu_{n_1}\cdots\mu_{n_r}
\int_{\RR}u_{n_1}\cdots u_{n_r}\,\dd\omega ,
\end{equation}
where \(\omega(\dd y)=\pi^{-1}(1+y^2)^{-1}\dd y\), and \(u_n\) are the
Cayley quotient coefficients from
\[
\frac{1+y^2}{1+(y-\varepsilon)^2}
=\sum_{n=0}^{\infty}u_n(y)\varepsilon^n .
\]
The formula is finite because every \(n_j\le m-2\le2N-2\).
The parity theorem gives \(A_m(\nu)=0\) for retained odd \(m\).
\end{theorem}

\noindent For \(N\ge3\), the \((2N-2)\)-moment assumption is the first finite
moment level at which the universal polynomial \(A_{2N}\) can be evaluated
for all laws; this is not asserted to be an analytic iff threshold for the
existence of a possibly non-polynomial \(t^{-2N}\)-scale expansion.
Proposition~\ref{prop:analytic-first-evaluability-obstruction} proves the
algebraic obstruction for the retained polynomial:
\[
A^{\mathrm{form}}_{2N}
=\kappa_N\,\mathfrak m_2\mathfrak m_{2N-2}
+P_N(\mathfrak m_2,\ldots,\mathfrak m_{2N-3}),
\qquad
\kappa_N=(-1)^N(N-1)2^{-2N+2}\ne0 .
\]

\begin{proof}
Put \(L:=2N-2\) and
\[
\delta_t(y)=\frac{h_t(\bar{\gamma}+ty)}{g_t(\bar{\gamma}+ty)}-1 .
\]
Lemma~\ref{lem:exact-moment-density-truncation} gives
\begin{equation}\label{eq:coefficient-identification-transfer-density-input}
\delta_t(y)=d_t(y)+\rho_t(y),\qquad
d_t(y)=\sum_{n=2}^{L}\mu_nu_n(y)t^{-n},
\end{equation}
with
\begin{equation}\label{eq:coefficient-identification-transfer-density-remainder}
\|d_t\|_\infty=O(t^{-2}),\qquad
\|\rho_t\|_\infty+\|\rho_t\|_{L^1(\omega)}=o(t^{-L}).
\end{equation}
The quotient identity behind this input is
\[
\frac{P_t(\bar{\gamma}+ty-\gamma)}{P_t(ty)}
=\frac{1+y^2}{1+(y-X_c/t)^2},
\]
with the constant mode removed by subtracting \(1\) and the linear mode
removed by \(\EE X_c=0\).  Also
\[
g_t(\bar{\gamma}+ty)t\,\dd y=P_t(ty)t\,\dd y=\omega(\dd y).
\]
Since \(h_t=g_t(1+\delta_t)\) in this variable and
\(\int\delta_t\,\dd\omega=0\),
\begin{equation}\label{eq:coefficient-identification-transfer-kl}
D_{\mathrm{KL}}(h_t\|g_t)
=\int_{\RR}\Phi(\delta_t(y))\,\omega(\dd y),
\qquad
\Phi(s)=(1+s)\log(1+s)-s .
\end{equation}
For \(t\) large, \(\|\delta_t\|_\infty=O(t^{-2})\) and
\[
\Phi(\delta_t)=
\sum_{r=2}^{N}\frac{(-1)^r}{r(r-1)}\delta_t^r
+O(|\delta_t|^{N+1}).
\]
The Taylor tail is \(O(t^{-2N-2})=o(t^{-2N})=o(t^{-(L+2)})\).

For every \(2\le r\le N\), the binomial expansion and one
\(L^1(\omega)\) placement of the remainder give
\[
\|\delta_t^r-d_t^r\|_{L^1(\omega)}
\le
C_r\sum_{\ell=1}^{r}
\|d_t\|_\infty^{r-\ell}
\|\rho_t\|_\infty^{\ell-1}
\|\rho_t\|_{L^1(\omega)}
=o(t^{-(L+2)}).
\]
Indeed the worst term is \(\ell=1,r=2\), namely
\(O(t^{-2})o(t^{-L})\), and all other mixed terms contain either an additional
\(d_t\)-factor or an additional \(\rho_t\)-factor.  Since \(L+2=2N\), only the
pure powers of \(d_t\) remain at the retained entropy scale.
In particular the worst layer may be read without the binomial notation as
\[
\|\delta_t^2-d_t^2\|_{L^1(\omega)}
\le
2\|d_t\|_\infty\|\rho_t\|_{L^1(\omega)}
+\|\rho_t\|_\infty\|\rho_t\|_{L^1(\omega)}
=O(t^{-2})o(t^{-L})+o(t^{-2L})
=o(t^{-2N}).
\]
Thus no mixed remainder term survives at the retained entropy order.

For the explicit \(N=4\) layer, \(L=6\) and
\[
d_t=t^{-2}a_2+t^{-3}a_3+t^{-4}a_4+t^{-5}a_5+t^{-6}a_6,
\qquad
a_j=\mu_j u_j .
\]
Combining this finite polynomial with
\[
\Phi(s)=\frac12s^2-\frac16s^3+\frac1{12}s^4+O(s^5)
\]
leaves, through order \(t^{-8}\), only the \(d_t^2,d_t^3,d_t^4\)
contributions.  The resulting \(B_4,B_6,B_8\) polynomials and their Laurent
constant terms are computed once in
Lemma~\ref{lem:n4-entropy-coefficient-substitution}; the present theorem is
the analytic justification that those finite Laurent values are the actual
KL coefficients under the finite sixth-moment hypothesis.

Expanding those pure powers gives, for \(4\le m\le2N\), exactly the coefficient
displayed in
\eqref{eq:coefficient-identification-transfer-expansion}.  This is the
finite-moment evaluation of
\eqref{eq:universal-entropy-coefficient}: every retained entropy monomial has
at least two mode indices \(n_j\ge2\), so an index \(>L\) would force total
degree at least \((L+1)+2=L+3>2N\).  Therefore
\[
D_{\mathrm{KL}}(h_t\|g_t)
=\sum_{m=4}^{2N}A_m(\nu)t^{-m}+o(t^{-2N}).
\]
The parity part of Theorem~\ref{thm:universal-entropy-coefficients} gives
\(A_m(\nu)=0\) for odd \(m\), and reindexing \(m=2r\) proves
\eqref{eq:poisson-kl-even-orders}.
\end{proof}

\subsubsection{Coefficient evaluability and analytic scope}%
\label{subsubsec:finite-moment-threshold-scope}

The three propositions below are the scope discussion for
Theorem~\ref{thm:poisson-kl-even-orders}.  They are ordered from obstruction
to exact coefficient mechanism to a concrete analytic lower-moment regime.

\begin{proposition}[Analytic obstruction to lower-moment replacement]%
\label{prop:analytic-first-evaluability-obstruction}
Let \(N\ge3\), set \(L=2N-2\), and suppose \(0<q<L\).  There exists a
centred symmetric probability law \(\nu\) on \(\RR\) such that
\[
\EE_\nu |X|^q<\infty,
\qquad
\EE_\nu |X|^L=\infty .
\]
For this law the universal coefficient \(A_{2N}\) from
\eqref{eq:universal-entropy-coefficient-evaluation} is not finite.  More
precisely, its formal top layer has the form
\begin{equation}\label{eq:analytic-obstruction-top-term}
A_{2N}^{\rm form}
=\kappa_N\,\mathfrak m_2\mathfrak m_L
+P_N(\mathfrak m_2,\ldots,\mathfrak m_{L-1}),
\qquad
\kappa_N=(-1)^N(N-1)2^{-2N+2}\ne0,
\end{equation}
and \(\mathfrak m_L\) is infinite for the constructed law while
\(\mathfrak m_2>0\).  Consequently no law-independent polynomial in centred
moments of orders at most \(L-1\), or in absolute moments below \(L\), can
replace the displayed universal coefficient \(A_{2N}\) on the class of laws
with only lower moment control.
\end{proposition}

\begin{proof}
Let \(a_j=\mathrm e^j\).  Choose \(c>0\) so small that
\[
\sum_{j\ge1}p_j<1,\qquad
p_j:=\frac{c}{j\,a_j^L}.
\]
Define the symmetric atomic probability law
\[
\nu:=\left(1-\sum_{j\ge1}p_j\right)\delta_0
\;+\;
\sum_{j\ge1}\frac{p_j}{2}(\delta_{a_j}+\delta_{-a_j}).
\]
It is centred and non-degenerate.  For the prescribed \(0<q<L\),
\[
\EE_\nu |X|^q=\sum_{j\ge1}p_ja_j^q
=c\sum_{j\ge1}\frac{\mathrm e^{-(L-q)j}}{j}<\infty,
\]
whereas
\[
\EE_\nu |X|^L=\sum_{j\ge1}p_ja_j^L
=c\sum_{j\ge1}\frac1j=\infty.
\]
In fact the same construction has finite absolute moment of every order
\(r<L\), but only the prescribed \(q\)-moment is needed for the obstruction.

The coefficient identity \eqref{eq:analytic-obstruction-top-term} is the top
term computation proved in
Proposition~\ref{prop:poisson-kl-moment-threshold-sharpness}.  Since
\(\kappa_N\ne0\), \(\mathfrak m_2>0\), and \(\mathfrak m_L=\infty\), the
universal coefficient \(A_{2N}\) is not finite for this law.

If a law-independent lower-moment polynomial
\(R(\mathfrak m_2,\ldots,\mathfrak m_{L-1})\) could replace \(A_{2N}\) on
all lower-moment laws, then it would have to give a finite value for the
constructed law because all its inputs are finite.  That contradicts the
non-finiteness forced by the nonzero top term in the universal coefficient.
The same contradiction applies to any polynomial built from absolute moments
of orders below \(L\), which are also finite for this law.
\end{proof}

\begin{proposition}[Top-moment dependence of the universal coefficient]%
\label{prop:poisson-kl-moment-threshold-sharpness}
Let \(N\ge2\) and \(L=2N-2\).
For \(N=2\), \(A_4(\nu)=\mu_2^2/8\).  For \(N\ge3\), the universal
polynomial \(A_{2N}\) in Theorem~\ref{thm:poisson-kl-even-orders} has an
unavoidable top-moment term:
\begin{equation}\label{eq:top-moment-coefficient-sharpness}
A_{2N}^{\rm form}
=\kappa_N\,\mathfrak m_2\mathfrak m_{2N-2}
 +P_N(\mathfrak m_2,\ldots,\mathfrak m_{2N-3}),
\qquad
\kappa_N=(-1)^N(N-1)2^{-2N+2}\ne0,
\end{equation}
where \(P_N\) involves no \(\mathfrak m_{2N-2}\).  Hence the displayed
universal coefficient through order \(t^{-2N}\) cannot be evaluated from
centred moments of orders at most \(2N-3\) alone.  This is an algebraic
non-evaluability statement for the same universal polynomial, not an
analytic if-and-only-if threshold statement.
\end{proposition}

\begin{proof}
The case \(N=2\) is the first line of the coefficient computation:
\[
A_4(\nu)=\frac12\mu_2^2 J(2,2)
=\frac12\mu_2^2\cdot\frac14
=\frac18\mu_2^2 .
\]
Assume \(N\ge3\).  In the formal coefficient
\eqref{eq:universal-entropy-coefficient}, a monomial containing
\(\mathfrak m_L\) and contributing to total degree \(L+2=2N\) must have
exactly two density factors, with indices \(2\) and \(L\).  All other
entropy monomials either have total degree different from \(2N\) or involve
only moments of order at most \(L-1\).  Since the quadratic Taylor
coefficient is \(1/2\) and the ordered pairs \((2,L)\) and \((L,2)\) both
occur, the coefficient of \(\mathfrak m_2\mathfrak m_L\) is \(J(2,L)\).

Write \(L=2m\), so \(m=N-1\).  From
\eqref{eq:mode-laurent-even},
\[
Q_{2m}(z)=(-1)^m2^{-2m}
\sum_{j=0}^{2m-1}\binom{2m-1}{j}
\bigl(z^{j+1}+z^{-(j+1)}\bigr),
\]
while \(Q_2(z)=-(z+z^{-1}+z^2+z^{-2})/4\).  Therefore only the
exponents \(\pm1\) and \(\pm2\) of \(Q_{2m}\) enter \(J(2,2m)\), and
\[
\begin{aligned}
J(2,2m)
&=[z^0]Q_2Q_{2m}  \\
&=-\frac12\,(-1)^m2^{-2m}
\left\{\binom{2m-1}{0}+\binom{2m-1}{1}\right\}  \\
&=(-1)^{m+1}m\,2^{-2m}
=(-1)^N(N-1)2^{-2N+2}.
\end{aligned}
\]
This proves \eqref{eq:top-moment-coefficient-sharpness}, and hence only the
claimed lower-moment non-evaluability of the universal polynomial \(A_{2N}\).

\end{proof}

\begin{lemma}[Borderline power-tail quotient expansion]%
\label{lem:borderline-power-tail-quotient-expansion}
Let \(N\ge3\), put \(L=2N-2\), and let \(\nu\) be a symmetric probability
law on \(\RR\) with density
\[
f(x)=c|x|^{-L-1}\mathbf1_{\{|x|\ge \mathrm e\}}+f_0(x),
\]
where \(c>0\) and \(f_0\) is a bounded symmetric density supported in
\(\{|x|\le \mathrm e\}\).  Put
\[
\delta_t(y):=\frac{(P_t*\nu)(ty)}{P_t(ty)}-1
=\int_{\RR}\{R_{x/t}(y)-1\}f(x)\dd x,
\qquad
R_\varepsilon(y)=\frac{1+y^2}{1+(y-\varepsilon)^2}.
\]
For \(2\le n\le L-1\), write
\(\mu_n=\int x^n f(x)\dd x\).  Then, as \(t\to\infty\),
\begin{equation}\label{eq:borderline-power-tail-quotient-expansion}
\delta_t(y)=
\sum_{n=2}^{L-1}\mu_nu_n(y)t^{-n}
+2c\,u_L(y)t^{-L}\log t+\varepsilon_t(y),
\end{equation}
with
\begin{equation}\label{eq:borderline-power-tail-quotient-remainder}
\|\varepsilon_t\|_\infty+\|\varepsilon_t\|_{L^1(\omega)}
\le C t^{-L}.
\end{equation}
\end{lemma}

\begin{proof}
It is enough to consider \(t\ge \mathrm e\).  All estimates below are
uniform in \(y\); since \(\omega\) is a probability measure and the same
pointwise majorants are used, they also give the stated \(L^1(\omega)\)
bounds.  On \(|x|\le t\), the local Cayley Taylor estimate gives, uniformly
in \(y\),
\[
R_{x/t}(y)-1
=\sum_{n=1}^{L}u_n(y)(x/t)^n+O(|x/t|^{L+1}) .
\]

\emph{The core \(|X|\le\mathrm e\).}
The core contributes its finite moments to the modes \(n<L\).  Its \(n=L\)
mode is
\[
t^{-L}u_L(y)\int_{|x|\le\mathrm e}x^Lf_0(x)\dd x=O(t^{-L}),
\]
and is absorbed into \(\varepsilon_t\).  Odd core modes vanish by symmetry,
and the \(n=0\) and \(n=1\) quotient modes are absent because the quotient is
compared with \(1\) and the law is centred.

\emph{The truncated power tail \(\mathrm e<|X|\le t\).}
For \(2\le n<L\), replacing the truncated tail moment by the full tail moment
costs
\[
t^{-n}\int_{|x|>t}|x|^n c|x|^{-L-1}\dd x
\le C t^{-n}\int_t^\infty x^{n-L-1}\dd x
=O(t^{-L}).
\]
Thus the core plus truncated tail gives the full moments
\(\mu_nu_n(y)t^{-n}\) for \(2\le n\le L-1\), up to an \(O(t^{-L})\) error.
For the missing top mode,
\[
t^{-L}u_L(y)\int_{\mathrm e<|x|\le t}x^L c|x|^{-L-1}\dd x
=2c\,u_L(y)t^{-L}\int_{\mathrm e}^{t}\frac{\dd x}{x}
=2c\,u_L(y)t^{-L}\log t+O(t^{-L}).
\]

\emph{The large tail \(|X|>t\).}
The quotient bound used in
Lemma~\ref{lem:exact-moment-density-truncation},
\[
|R_\varepsilon(y)-1|\le C(1+\varepsilon^2),
\]
and \(L\ge4\) give
\[
\int_{|x|>t}|R_{x/t}(y)-1|f(x)\dd x
\le C\left\{\int_t^\infty x^{-L-1}\dd x
+t^{-2}\int_t^\infty x^{1-L}\dd x\right\}
=O(t^{-L}).
\]
The part of the displayed expansion already assigned to the omitted large
tail is also \(O(t^{-L})\), because for \(2\le n<L\)
\[
t^{-n}\int_{|x|>t}|x|^n f(x)\dd x=O(t^{-L}).
\]

\emph{The Taylor remainder.}
The remainder on \(|x|\le\mathrm e\) is \(O(t^{-(L+1)})\).  On
\(\mathrm e<|x|\le t\),
\[
t^{-(L+1)}\int_{\mathrm e<|x|\le t}|x|^{L+1}c|x|^{-L-1}\dd x
\le C t^{-(L+1)}\int_{\mathrm e}^{t}\dd x
=O(t^{-L}).
\]
Combining the four estimates proves
\eqref{eq:borderline-power-tail-quotient-expansion} and
\eqref{eq:borderline-power-tail-quotient-remainder}.
\end{proof}

\begin{proposition}[Explicit symmetric power-tail logarithmic replacement example]%
\label{prop:log-tail-counterexample-top-moment}
Let \(N\ge3\), put \(L=2N-2\), and fix \(c>0\) so small that
\(2c\,\mathrm e^{-L}/L<1\).  Let \(\nu\) be the symmetric probability law
with density
\[
f(x)=\frac{1-2c\,\mathrm e^{-L}/L}{2}\,\mathbf1_{\{|x|\le1\}}
      +c|x|^{-L-1}\mathbf1_{\{|x|\ge\mathrm e\}} .
\]
Then \(\nu\) is centred, \(\mu_2>0\), and
\[
\EE|X|^q<\infty\quad(q<L),
\qquad
\EE|X|^L=\infty .
\]
Let
\[
H(t):=D_{\mathrm{KL}}\bigl(P_t*\nu\,\|\,P_t\bigr),
\qquad
S_{N-1}(t):=\sum_{m=2}^{N-1}A_{2m}(\nu)t^{-2m},
\]
where the coefficients in \(S_{N-1}\) are the finite universal
moment-polynomial coefficients, which use only moments of order at most
\(2N-4<L\).  Then
\begin{equation}\label{eq:log-tail-counterexample-asymptotic}
H(t)
=S_{N-1}(t)
+2c\,\mu_2\,\kappa_N\,t^{-2N}\log t
+O(t^{-2N}),
\qquad t\to\infty,
\end{equation}
where
\[
\kappa_N=J(2,L)=(-1)^N(N-1)2^{-2N+2}\ne0 .
\]
Equivalently,
\begin{equation}\label{eq:log-tail-counterexample-limit}
\lim_{t\to\infty}
\frac{t^{2N}}{\log t}
\left\{H(t)-\sum_{m=2}^{N-1}A_{2m}(\nu)t^{-2m}\right\}
=2c\,\mu_2\,\kappa_N\ne0 .
\end{equation}
Thus, for this explicit family, the first non-evaluable top moment is not
represented by a lower-moment evaluation of \(A_{2N}\): the verified
power-tail calculation replaces the missing \(A_{2N}t^{-2N}\) layer with a
tail-dependent \(t^{-2N}\log t\) term.  Already when \(N=3\), a symmetric law
with finite variance and infinite fourth moment satisfies
\[
H(t)=\frac{\mu_2^2}{8t^4}
-\frac{c\mu_2}{4}\frac{\log t}{t^6}
+O(t^{-6}),
\]
since \(\kappa_3=-1/8\).
\end{proposition}

\begin{proof}
The moment assertions follow directly from the tail integral
\[
\int_{\mathrm e}^{\infty}x^q\,c x^{-L-1}\,\dd x,
\]
which is finite exactly when \(q<L\), while the \(q=L\) integral is
\(c\int_{\mathrm e}^{\infty}x^{-1}\dd x=\infty\) on each half-line.  The
law is centred by symmetry and has positive variance because it is not
concentrated at the origin.

Set
\[
A_t(y):=\sum_{n=2}^{L-1}\mu_nu_n(y)t^{-n}.
\]
With
\[
B_t(y):=2c\,u_L(y)t^{-L}\log t,
\]
\[
\Delta_t(y):=\frac{(P_t*\nu)(ty)}{P_t(ty)}-1,
\]
Lemma~\ref{lem:borderline-power-tail-quotient-expansion}, applied with
\[
f_0(x)=\frac{1-2c\,\mathrm e^{-L}/L}{2}\mathbf1_{\{|x|\le1\}},
\]
gives
\begin{equation}\label{eq:log-tail-density-expansion}
\Delta_t(y)
=A_t(y)+B_t(y)+\varepsilon_t(y)
\end{equation}
where
\[
\|\varepsilon_t\|_\infty+\|\varepsilon_t\|_{L^1(\omega)}=O(t^{-L}).
\]
Moreover
\[
\|A_t\|_\infty=O(t^{-2}),\qquad
\|B_t\|_\infty+\|B_t\|_{L^1(\omega)}=O(t^{-L}\log t).
\]

Since \(\mu_2<\infty\), Lemma~\ref{lem:exact-moment-density-truncation} with
\(L=2\) gives \(\|\Delta_t\|_\infty=O(t^{-2})\).  Therefore
\[
\Phi(\Delta_t)
=\sum_{r=2}^{N}\frac{(-1)^r}{r(r-1)}\Delta_t^r+O(t^{-2N-2})
\]
in \(L^1(\omega)\).  This is the Taylor-remainder error; after integration it
is \(O(t^{-2N-2})\).

We now separate the polynomial, logarithmic, and nonlinear entropy errors in
the Taylor polynomial.  First, the pure \(A_t\) contribution is
\[
\sum_{r=2}^{N}\frac{(-1)^r}{r(r-1)}
\int_{\RR}A_t(y)^r\,\omega(\dd y)
=S_{N-1}(t)+O(t^{-2N}).
\]
Indeed the law is symmetric, so \(A_t\) contains only even mode degrees below
\(L\); the retained pure products of total degree at most \(L=2N-2\) are
exactly the universal coefficients in \(S_{N-1}\), and every remaining pure
\(A_t\) product has total degree at least \(L+2=2N\).

Next consider all terms containing the density error \(\varepsilon_t\).  For
the quadratic part,
\[
\|A_t\varepsilon_t\|_{L^1(\omega)}
=O(t^{-2})O(t^{-L})=O(t^{-(L+2)}),
\]
\[
\|B_t\varepsilon_t\|_{L^1(\omega)}
\le \|B_t\|_\infty\|\varepsilon_t\|_{L^1(\omega)}
=O(t^{-2L}\log t)=O(t^{-(L+2)}),
\]
and
\[
\|\varepsilon_t^2\|_{L^1(\omega)}
\le \|\varepsilon_t\|_\infty\|\varepsilon_t\|_{L^1(\omega)}
=O(t^{-2L})=O(t^{-(L+2)}).
\]
For every Taylor degree \(r\ge3\), the part of
\(\Delta_t^r-(A_t+B_t)^r\) containing \(\varepsilon_t\) is bounded by
\[
C_r\|\Delta_t\|_\infty^{r-1}\|\varepsilon_t\|_{L^1(\omega)}
=O(t^{-2(r-1)}t^{-L})
=O(t^{-(L+4)})
=O(t^{-(L+2)}).
\]

Now consider terms containing \(B_t\) but no \(\varepsilon_t\).  In the
quadratic part,
\[
\frac12\int_{\RR}\{(A_t+B_t)^2-A_t^2\}\,\omega(\dd y)
=\int_{\RR}A_tB_t\,\dd\omega+\frac12\int_{\RR}B_t^2\,\dd\omega .
\]
The component of \(A_tB_t\) coming from \(A_t-\mu_2u_2t^{-2}\) is
\[
O(t^{-4})\,O(t^{-L}\log t)=O(t^{-(L+4)}\log t)=O(t^{-(L+2)}),
\]
and
\[
\int_{\RR}B_t^2\,\dd\omega=O(t^{-2L}(\log t)^2)=O(t^{-(L+2)}).
\]
Thus the only logarithmic term at order \(t^{-(L+2)}\log t\) is
\[
\int_{\RR}\mu_2u_2(y)t^{-2}\,B_t(y)\,\omega(\dd y)
=2c\,\mu_2\,J(2,L)t^{-2N}\log t .
\]
Finally, for \(r\ge3\), every term of \((A_t+B_t)^r-A_t^r\) has at least one
\(B_t\) factor and at least two further density factors.  Since
\(\|A_t+B_t\|_\infty=O(t^{-2})\),
\[
C_r\|A_t+B_t\|_\infty^{r-1}\|B_t\|_{L^1(\omega)}
=O(t^{-2(r-1)}t^{-L}\log t)
=O(t^{-(L+4)}\log t)
=O(t^{-(L+2)}).
\]

Equivalently, with \(A_2(y):=\mu_2u_2(y)t^{-2}\), the preceding estimates are
the \(L^1(\omega)\) expansion
\begin{equation}\label{eq:phi-borderline-log-linearization}
\Phi(A_t+B_t+\varepsilon_t)
=\Phi(A_t)+A_2B_t+O(t^{-2N}).
\end{equation}
All terms in the \(O(t^{-2N})\) remainder are uniform after integration; the
only term carrying \(t^{-2N}\log t\) is \(A_2B_t\).

Combining the Taylor remainder, the pure-polynomial remainder, the
\(\varepsilon_t\)-terms, the nonlinear \(B_t\)-terms, and the quadratic
logarithmic cross term gives
\[
H(t)=S_{N-1}(t)+2c\,\mu_2 J(2,L)t^{-2N}\log t+O(t^{-2N}).
\]
The value \(J(2,L)=\kappa_N\) was computed in
\eqref{eq:top-moment-coefficient-sharpness}.  This proves
\eqref{eq:log-tail-counterexample-asymptotic} and hence
\eqref{eq:log-tail-counterexample-limit}.
\end{proof}

\begin{proposition}[The \(N=2\) borderline infinite-variance layer]
Let \(c>0\) satisfy \(c\mathrm e^{-2}<1\), and let \(\nu\) be the symmetric
probability law with density
\[
f(x)=\frac{1-c\mathrm e^{-2}}{2}\mathbf1_{\{|x|\le1\}}
      +c|x|^{-3}\mathbf1_{\{|x|\ge \mathrm e\}} .
\]
Then \(\nu\) has finite mean and infinite variance, and
\[
D_{\mathrm{KL}}(P_t*\nu\|P_t)
=\frac{c^2}{2}\,t^{-4}(\log t)^2+O(t^{-4}\log t).
\]
Thus the borderline \(N=2\) law is not a \(t^{-4}\log t\) replacement; its
leading layer is \(t^{-4}(\log t)^2\), in agreement with
Theorem~\ref{thm:sharp-leading-kl-variance-threshold}.
\end{proposition}

\begin{proof}
The moment statement follows from
\(\int_{\mathrm e}^{\infty}x^qc x^{-3}\dd x\).  Repeating
Lemma~\ref{lem:borderline-power-tail-quotient-expansion} with \(L=2\) gives
\[
\Delta_t(y):=\frac{(P_t*\nu)(ty)}{P_t(ty)}-1
=2c\,u_2(y)t^{-2}\log t+\varepsilon_t(y),
\qquad
\|\varepsilon_t\|_\infty+\|\varepsilon_t\|_{L^1(\omega)}=O(t^{-2}).
\]
The logarithm is still small enough that
\[
D_{\mathrm{KL}}(P_t*\nu\|P_t)
=\frac12\int_{\RR}\Delta_t(y)^2\,\omega(\dd y)+O(t^{-6}(\log t)^3).
\]
Since \(J(2,2)=\int u_2^2\,\dd\omega=1/4\), the leading term is
\[
\frac12(2c)^2J(2,2)t^{-4}(\log t)^2
=\frac{c^2}{2}t^{-4}(\log t)^2,
\]
and all cross terms with \(\varepsilon_t\) are \(O(t^{-4}\log t)\).
\end{proof}

\begin{lemma}[Second-mode transform of a Cauchy translate]%
\label{lem:second-mode-cauchy-translate-transform}
For every \(\varepsilon\in\RR\),
\begin{equation}\label{eq:second-mode-cauchy-translate-transform}
\int_{\RR}u_2(y)R_\varepsilon(y)\,\omega(\dd y)
=\frac{4\varepsilon^2}{(4+\varepsilon^2)^2}.
\end{equation}
Consequently, for
\[
h_t=P_t*\nu,\qquad g_t(x)=P_t(x-\bar{\gamma}),\qquad
\delta_t(x)=h_t(x)/g_t(x)-1,
\]
and \(x=\bar{\gamma}+ty\),
\begin{equation}\label{eq:second-mode-projection-exact}
\int_{\RR}u_2(y)\delta_t(\bar{\gamma}+ty)\,\omega(\dd y)
=
\EE\!\left[\frac{4(X_c/t)^2}{(4+(X_c/t)^2)^2}\right].
\end{equation}
\end{lemma}

\begin{proof}
The second identity follows from the first by Fubini, since \(u_2\) is
bounded and
\[
\delta_t(\bar{\gamma}+ty)=\EE[R_{X_c/t}(y)-1],
\qquad
\int_{\RR}u_2\,\dd\omega=0 .
\]
It remains to prove \eqref{eq:second-mode-cauchy-translate-transform}.  Using
\[
u_2(y)=\frac{3y^2-1}{(1+y^2)^2},\qquad
R_\varepsilon(y)=\frac{1+y^2}{1+(y-\varepsilon)^2},
\]
the left side is
\[
\frac1\pi\int_{\RR}
\frac{3y^2-1}{(1+y^2)^2(1+(y-\varepsilon)^2)}\,\dd y .
\]
For \(\varepsilon=0\) the integrand is
\((3y^2-1)/(1+y^2)^3\), whose integral is zero, for instance by the
substitution \(y=\tan\theta\).  Assume next that \(\varepsilon\ne0\), and set
\[
F_\varepsilon(z):=
\frac{3z^2-1}
{(z-\ii)^2(z+\ii)^2(z-\varepsilon-\ii)(z-\varepsilon+\ii)} .
\]
This rational function is \(O(|z|^{-4})\), so closing the contour in the
upper half-plane gives
\[
\frac1\pi\int_{\RR}
\frac{3y^2-1}{(1+y^2)^2(1+(y-\varepsilon)^2)}\,\dd y
=
2\ii\left\{
\operatorname{Res}_{z=\ii}F_\varepsilon
+\operatorname{Res}_{z=\varepsilon+\ii}F_\varepsilon
\right\}.
\]
The simple pole at \(z=\varepsilon+\ii\) gives
\[
\operatorname{Res}_{z=\varepsilon+\ii}F_\varepsilon
=-\frac{\ii(3\varepsilon^2+6\ii\varepsilon-4)}
{2\varepsilon^2(\varepsilon+2\ii)^2}.
\]
At the double pole \(z=\ii\),
\[
\operatorname{Res}_{z=\ii}F_\varepsilon
=
\left.
\frac{\dd}{\dd z}
\frac{3z^2-1}
{(z+\ii)^2(z-\varepsilon-\ii)(z-\varepsilon+\ii)}
\right|_{z=\ii}
=-\frac{\ii(\varepsilon^2+2\ii\varepsilon+4)}
{2\varepsilon^2(\varepsilon-2\ii)^2}.
\]
Adding the two displayed residues yields the cancellation
\[
\operatorname{Res}_{z=\ii}F_\varepsilon
+\operatorname{Res}_{z=\varepsilon+\ii}F_\varepsilon
=-\frac{2\ii\varepsilon^2}{(4+\varepsilon^2)^2}.
\]
Therefore
\[
\frac1\pi\int_{\RR}
\frac{3y^2-1}{(1+y^2)^2(1+(y-\varepsilon)^2)}\,\dd y
=\frac{4\varepsilon^2}{(4+\varepsilon^2)^2}.
\]
This proves the lemma.
\end{proof}

\begin{theorem}[Sharp analytic variance threshold for the leading KL coefficient]%
\label{thm:sharp-leading-kl-variance-threshold}
Let \(\nu\) be a probability measure on \(\RR\) with finite mean
\(\bar{\gamma}\).  Put
\[
h_t=P_t*\nu,\qquad g_t(x)=P_t(x-\bar{\gamma}),\qquad
H(t)=D_{\mathrm{KL}}(h_t\|g_t),
\]
and \(X_c=\gamma-\bar{\gamma}\).  Then the following are equivalent:
\[
\EE X_c^2<\infty
\qquad\Longleftrightarrow\qquad
\limsup_{t\to\infty}t^4H(t)<\infty .
\]
When these equivalent conditions hold, the limit exists and is
\begin{equation}\label{eq:sharp-leading-kl-variance-threshold-finite}
\lim_{t\to\infty}t^4H(t)=\frac{(\EE X_c^2)^2}{8}.
\end{equation}
When \(\EE X_c^2=\infty\), one has the stronger divergence
\begin{equation}\label{eq:sharp-leading-kl-variance-threshold-infinite}
\lim_{t\to\infty}t^4H(t)=+\infty .
\end{equation}
\end{theorem}

\begin{proof}
If \(\EE X_c^2<\infty\), Theorem~\ref{thm:poisson-kl-even-orders} with
\(N=2\) gives
\[
H(t)=A_4(\nu)t^{-4}+o(t^{-4}),
\qquad
A_4(\nu)=\frac18(\EE X_c^2)^2,
\]
which proves \eqref{eq:sharp-leading-kl-variance-threshold-finite}.

Assume now that \(\EE X_c^2=\infty\).  Write \(x=\bar{\gamma}+ty\) and
\[
\delta_t(\bar{\gamma}+ty)=h_t(\bar{\gamma}+ty)/g_t(\bar{\gamma}+ty)-1 .
\]
The Csisz{\'a}r--Pinsker inequality
\cite{Pinsker1964,Csiszar1967,BolleyVillani2005} in the form
\[
2H(t)\ge \left(\int_{\RR}|h_t-g_t|\,\dd x\right)^2
\]
and the bound
\(\|u_2\|_\infty\le1\) from \eqref{eq:cayley-first-modes} give
\[
2H(t)\ge
\left|
\int_{\RR}u_2(y)\delta_t(\bar{\gamma}+ty)\,\omega(\dd y)
\right|^2 .
\]
By Lemma~\ref{lem:second-mode-cauchy-translate-transform}, the last
projection equals
\[
B(t):=\EE\!\left[\frac{4(X_c/t)^2}{(4+(X_c/t)^2)^2}\right]\ge0.
\]
Therefore
\begin{equation}\label{eq:sharp-leading-threshold-pinsker-bound}
t^4H(t)\ge\frac12\,\{t^2B(t)\}^2 .
\end{equation}
For every \(M>0\), the function
\[
z\longmapsto \frac{4z^2}{(4+z^2)^2}
\]
satisfies, on \(|z|\le M/t\),
\[
\frac{4z^2}{(4+z^2)^2}
\ge
\frac{4z^2}{(4+(M/t)^2)^2}.
\]
Hence
\[
t^2B(t)
\ge
\frac{4}{(4+(M/t)^2)^2}\,
\EE[X_c^2\mathbf1_{\{|X_c|\le M\}}].
\]
Letting \(t\to\infty\) with \(M\) fixed gives
\[
\liminf_{t\to\infty}t^2B(t)
\ge
\frac14\,\EE[X_c^2\mathbf1_{\{|X_c|\le M\}}].
\]
Since the truncated second moment tends to \(+\infty\) as \(M\to\infty\),
\(\lim_{t\to\infty}t^2B(t)=+\infty\).  The lower bound
\eqref{eq:sharp-leading-threshold-pinsker-bound} then gives
\eqref{eq:sharp-leading-kl-variance-threshold-infinite}.  In particular
\(\limsup t^4H(t)<\infty\) is impossible without finite variance.
\end{proof}

\begin{remark}[Quantitative margins]
No positive-variance hypothesis is imposed in
Theorem~\ref{thm:poisson-kl-even-orders}; when the variance is zero the
displayed coefficients vanish and \(h_t=g_t\).
Every entropy monomial contributing to \(A_{2N}^{\rm form}\) has at least two
density factors, hence
uses centred moments only through order \(2N-2\).  With \(L=2N-2\), the finite
\(L\)th absolute moment gives an \(L^\infty\cap L^1(\omega)\) density
remainder \(o(t^{-L})\), and every entropy term containing that remainder also
contains the quadratic entropy scale.  The first discarded
density-remainder contribution is therefore \(o(t^{-(L+2)})=o(t^{-2N})\).
The lower-moment obstruction is precisely the coefficient-evaluation
obstruction of Proposition~\ref{prop:poisson-kl-moment-threshold-sharpness};
the boxed paragraph after Theorem~\ref{thm:poisson-kl-even-orders} records
the corresponding scope.  The \(+\eta\) moment
hypotheses in Appendix~\ref{subsubsec:finite-moment-constants-variants} are
used only for quantitative remainders and thresholds.
\end{remark}

\begin{theorem}[Analytic compact-support interchange]%
\label{thm:analytic-compact-support-interchange}
Let $\nu$ be a probability measure with mean $\bar{\gamma}$ and compactly
supported centred variable $X_c=\gamma-\bar{\gamma}$.  Then there is
$T_\nu\ge1$ such that, for $t\ge T_\nu$, the series
\begin{equation}\label{eq:analytic-density-series}
\delta_t(\bar{\gamma}+ty)
=\sum_{n=2}^{\infty}\mu_n\,u_n(y)t^{-n}
\end{equation}
converges absolutely and uniformly in $y\in\RR$.  The entropy Taylor series
\begin{equation}\label{eq:analytic-entropy-series}
D_{\mathrm{KL}}(P_t*\nu\|P_t(\cdot-\bar{\gamma}))
=\sum_{k=4}^{\infty}A_k(\nu)t^{-k}
\end{equation}
also converges absolutely for $t\ge T_\nu$, where $A_k(\nu)$ is the
finite-moment evaluation of $A_k^{\rm form}$ from
\eqref{eq:universal-entropy-coefficient-evaluation}.  In particular the
all-order coefficients are the genuine analytic coefficients under this
strong compact-support hypothesis, while
Theorem~\ref{thm:poisson-kl-even-orders} and the retained-polynomial form in
\S\ref{subsubsec:finite-moment-constants-variants} give the separate
finite-moment asymptotic justification.
\end{theorem}

\begin{proof}
Let $|X_c|\le M$ almost surely.  The generating function for the
Chebyshev polynomials gives
\[
R_\varepsilon(\tan\theta)
=\sum_{n=0}^{\infty}\cos^n\theta\,U_n(\sin\theta)\varepsilon^n.
\]
Since $|U_n(s)|\le n+1$ for $|s|\le1$ and $|\cos\theta|\le1$, the series is
dominated uniformly by $\sum_{n\ge0}(n+1)|\varepsilon|^n$ whenever
$|\varepsilon|<1$.  Choose $T_\nu\ge2M$.  Then
$|X_c|/t\le1/2$, so dominated convergence permits termwise integration
against~$\nu$ and gives the absolutely and uniformly convergent density
series.  The $n=0$ term is one and the $n=1$ term is zero because
$\EE X_c=0$, which proves \eqref{eq:analytic-density-series}.

The uniform convergence gives $\delta_t=O(t^{-2})$ uniformly.  Enlarging
$T_\nu$ if needed, $|\delta_t|\le1/2$, and the Taylor series for
$\Phi(s)=(1+s)\log(1+s)-s$ converges absolutely and uniformly after the
substitution $s=\delta_t(\bar{\gamma}+ty)$.  To make the interchange explicit,
write
\[
b_n(t):=|\mu_n|\,\|u_n\|_\infty t^{-n}
\qquad(n\ge2),
\qquad
B(t):=\sum_{n\ge2}b_n(t).
\]
The compact support bound and $\|u_n\|_\infty\le n+1$ give
$B(t)\le\sum_{n\ge2}(n+1)(M/t)^n<\infty$, and after increasing $T_\nu$ we
may assume $B(t)\le1/2$.  Hence every product occurring in the entropy Taylor
series has the summable absolute majorant
\[
\sum_{r=2}^{\infty}\frac1{r(r-1)}
\sum_{n_1,\dots,n_r\ge2}
b_{n_1}(t)\cdots b_{n_r}(t)
=\sum_{r=2}^{\infty}\frac{B(t)^r}{r(r-1)}<\infty.
\]
Since $\omega$ has total mass one and
\[
\int\bigl|u_{n_1}(y)\cdots u_{n_r}(y)\bigr|\,\omega(\dd y)
\le \prod_{j=1}^r\|u_{n_j}\|_\infty,
\]
this single majorant justifies both Fubini/Tonelli for
$\Phi(\delta_t)=\sum_{r\ge2}(-1)^r\delta_t^r/(r(r-1))$ and termwise
multiplication of the absolutely convergent density series.  For a fixed
power $t^{-k}$, only finitely many tuples $r\ge2$, $n_j\ge2$, and
$n_1+\cdots+n_r=k$ occur.  The degree-$k$ coefficient obtained by multiplying
the density series is therefore the finite-moment evaluation of the formal
expression in
\eqref{eq:universal-entropy-coefficient}; hence
\eqref{eq:analytic-entropy-series} follows.
\end{proof}
