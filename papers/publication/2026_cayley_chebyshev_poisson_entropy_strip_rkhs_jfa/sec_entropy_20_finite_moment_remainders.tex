\subsection{Finite-moment entropy asymptotics and remainders}%
\label{subsec:finite-moment-entropy}

\subsubsection{Density-to-entropy passage}

This subsection passes from formal coefficient algebra to genuine entropy
asymptotics.  The density-to-entropy passage is a finite operation at fixed
order $2N$.
First the density ratio is written as a finite mode polynomial plus an
$L^\infty\cap L^1(\omega)$ error.  Then
$\Phi(z)=(1+z)\log(1+z)-z$ is expanded only through degree~$N$, because the
density perturbation begins at order $t^{-2}$.  Thus a product containing one
density error also contains the entropy quadratic order and is two powers
smaller than the density error itself; products with more density-error
factors are smaller still.  The displayed two-scale constants below are
convenience envelopes for this refined product hierarchy and for the
high-degree finite-mode tail.  After integration, the remaining finite
polynomial is exactly the universal coefficient polynomial.

\begin{lemma}[Product hierarchy for the density remainder]%
\label{lem:density-remainder-product-hierarchy}
Let $p>2$, $2\le r$, and let $A,\rho$ be measurable functions satisfying
\[
\|A\|_{L^\infty(\RR)}\le M t^{-2},
\qquad
\|\rho\|_{L^\infty(\RR)}+\|\rho\|_{L^1(\omega)}
\le C t^{-p}
\]
for some $t\ge1$.  Then
\begin{equation}\label{eq:density-remainder-product-hierarchy}
\|(A+\rho)^r-A^r\|_{L^1(\omega)}
\le
\sum_{\ell=1}^{r}\binom r\ell
M^{r-\ell}C^\ell
t^{-\{\ell p+2(r-\ell)\}} .
\end{equation}
In particular, since $p>2$,
\begin{equation}\label{eq:density-remainder-product-envelope}
\|(A+\rho)^r-A^r\|_{L^1(\omega)}
\le
\sum_{\ell=1}^{r}\binom r\ell
M^{r-\ell}C^\ell
t^{-(p+2)} .
\end{equation}
\end{lemma}

\begin{proof}
Expand
\[
(A+\rho)^r-A^r
=\sum_{\ell=1}^r\binom r\ell A^{r-\ell}\rho^\ell .
\]
For a fixed $\ell$, put one factor of $\rho$ in $L^1(\omega)$ and all other
factors in $L^\infty$.  This gives
\[
\|A^{r-\ell}\rho^\ell\|_{L^1(\omega)}
\le
\|A\|_\infty^{r-\ell}\|\rho\|_\infty^{\ell-1}
\|\rho\|_{L^1(\omega)}
\le
M^{r-\ell}C^\ell
t^{-2(r-\ell)}t^{-\ell p},
\]
which is \eqref{eq:density-remainder-product-hierarchy}.  The exponent
$\ell p+2(r-\ell)$ is minimized at $\ell=1$ and $r=2$ when $p>2$ and
$r\ge2$, giving the envelope \eqref{eq:density-remainder-product-envelope}.
\end{proof}

\begin{lemma}[Exact-moment density truncation with little-oh remainder]%
\label{lem:exact-moment-density-truncation}
Let \(L\ge2\), and let \(\nu\) be a probability measure on \(\RR\) with
mean \(\bar\gamma\) and finite centred \(L\)th absolute moment.  Put
\(X_c=\gamma-\bar\gamma\) and
\[
\delta_t(\bar\gamma+ty)
:=
\frac{(P_t*\nu)(\bar\gamma+ty)}{P_t(ty)}-1 .
\]
Equivalently, for \(x=\bar\gamma+ty\),
\[
\frac{P_t(x-\gamma)}{P_t(x-\bar\gamma)}
=\frac{t^2+(x-\bar\gamma)^2}{t^2+(x-\gamma)^2}
=\frac{1+y^2}{1+(y-X_c/t)^2}
=R_{X_c/t}(y),
\]
and therefore
\[
\delta_t(\bar\gamma+ty)
=\int_{\RR}\bigl(R_{X_c/t}(y)-1\bigr)\dd\nu(\gamma).
\]
Define
\[
\rho_{L,t}(y):=
\int_{\RR}\left(
R_{X_c/t}(y)-\sum_{n=0}^{L}u_n(y)(X_c/t)^n
\right)\dd\nu(\gamma).
\]
Then, with \(\mu_n=\int X_c^n\dd\nu(\gamma)\) for \(0\le n\le L\),
\begin{equation}\label{eq:exact-moment-density-truncation}
\delta_t(\bar\gamma+ty)
=
\sum_{n=2}^{L}\mu_nu_n(y)t^{-n}+\rho_{L,t}(y),
\qquad
\|\rho_{L,t}\|_{\infty}+\|\rho_{L,t}\|_{L^1(\omega)}=o(t^{-L}).
\end{equation}
The \(n=0\) term is absent from the displayed sum because the quotient is
compared with \(1\), and the \(n=1\) term is absent because
\(\int X_c\,\dd\nu=0\).
Moreover \(\sup_y|\delta_t(\bar\gamma+ty)|=O(t^{-2})\).
In particular, when \(L=2N-2\), this is exactly the little-oh density input
used by Lemma~\ref{lem:exact-moment-entropy-transfer} in
Theorem~\ref{thm:poisson-kl-even-orders} for the universal polynomial
coefficient \(A_{2N}^{\rm alg}\); the same \(O(t^{-2})\) bound gives the
smallness threshold for the entropy Taylor expansion.  The statement uses
only the finite \(L\)th absolute moment: the proof does not assume
integrability of an \((L+1)\)-order Taylor remainder, and the tail split below
is the step that keeps the threshold at \(L\).  Thus the later exact-threshold
entropy hierarchy uses this lemma only through the displayed quotient
normalization, the cancellation of the \(0\) and \(1\) modes, and the
\(L^\infty\cap L^1(\omega)\) little-oh remainder.
\end{lemma}

\begin{proof}
The quotient normalization is the one used throughout this section:
\[
\frac{(P_t*\nu)(\bar\gamma+ty)}{P_t(ty)}
=\int_{\RR}R_{X_c/t}(y)\dd\nu(\gamma).
\]
We prove the \(L^\infty\) and \(L^1(\omega)\) remainders by the same split.
On \(|X_c|\le t\), Proposition~\ref{prop:cayley-mode-remainder} gives the
exact finite Taylor remainder
\[
R_{X_c/t}(y)-\sum_{n=0}^{L}u_n(y)(X_c/t)^n
=\left(\frac{X_c}{t}\right)^{L+1}v_L(y,X_c/t).
\]
With \(V_L=\|v_L\|_\infty<\infty\), uniformly in \(y\),
\[
\left|
\int_{|X_c|\le t}\left(
R_{X_c/t}(y)-\sum_{n=0}^{L}u_n(y)(X_c/t)^n
\right)\dd\nu
\right|
\le
V_Lt^{-L}\EE\!\left[
|X_c|^L\frac{|X_c|}{t}\mathbf1_{\{|X_c|\le t\}}
\right].
\]
The expectation tends to zero by dominated convergence, since the multiplier
\((|X_c|/t)\mathbf1_{\{|X_c|\le t\}}\) is bounded by \(1\) and converges
pointwise to \(0\).

On \(|X_c|>t\), put \(\varepsilon=X_c/t\).  The large-\(\varepsilon\)
estimate used in Theorem~\ref{thm:finite-moment-density-remainder} gives,
for \(L\ge2\),
\[
R_{\varepsilon}(y)\le 3(1+\varepsilon^2)\le6|\varepsilon|^L,
\qquad
\left|\sum_{n=0}^{L}u_n(y)\varepsilon^n\right|
\le (L+1)B_L|\varepsilon|^L
\]
whenever \(|\varepsilon|>1\), where
\(B_L=\max_{0\le n\le L}\|u_n\|_\infty\).  Hence, uniformly in \(y\),
\[
\left|
\int_{|X_c|>t}\left(
R_{X_c/t}(y)-\sum_{n=0}^{L}u_n(y)(X_c/t)^n
\right)\dd\nu
\right|
\le
\{6+(L+1)B_L\}t^{-L}\EE[
|X_c|^L\mathbf1_{\{|X_c|>t\}}],
\]
which is \(o(t^{-L})\).  Combining the two pieces gives
\(\|\rho_{L,t}\|_\infty=o(t^{-L})\).  Since \(\omega\) is a probability
measure and the same pointwise majorants dominate the \(y\)-integral, the
identical estimates give \(\|\rho_{L,t}\|_{L^1(\omega)}=o(t^{-L})\).

The constant term cancels and the \(n=1\) term integrates to zero because
\(\mu_1=0\), giving \eqref{eq:exact-moment-density-truncation}.

Finally, Lemma~\ref{lem:tail-uniform-cayley-mode-remainder} with \(N=1\)
applies because \(L\ge2\) implies \(\EE|X_c|^2<\infty\).  Since
\(\mu_1=0\), it gives
\(\sup_y|\delta_t(\bar\gamma+ty)|=O(t^{-2})\).
This is the density-level input used by the exact-moment entropy-transfer
lemma below when Theorem~\ref{thm:poisson-kl-even-orders} identifies the
universal moment-polynomial coefficient \(A_{2N}\) at the moment threshold
where the displayed coefficient polynomial is evaluable.
\end{proof}

\begin{proposition}[Finite-moment density-to-entropy transfer]%
\label{prop:density-to-entropy-descent}%
\label{lem:density-to-entropy-domination}%
\label{lem:explicit-entropy-constant-bound}
Let $N\ge2$, $0<\eta\le1$, and let
\[
\omega(\dd y)=\frac{\dd y}{\pi(1+y^2)}.
\]
Let $u_2,\ldots,u_{2N}$ be bounded measurable functions on~$\RR$, and let
$\mu_2,\ldots,\mu_{2N}$ be real numbers.  For
$2\le r\le N$ and $2\le n_j\le2N$, put
\[
I(n_1,\ldots,n_r)
:=\int_{\RR}u_{n_1}(y)\cdots u_{n_r}(y)\,\omega(\dd y),
\]
which is absolutely convergent because $\omega(\RR)=1$ and the modes are
bounded.  For $4\le m\le2N$ define the finite polynomial coefficient
\begin{equation}\label{eq:density-to-entropy-descent-coefficient}
\mathcal A_m(\mu,u)
:=
\sum_{r=2}^{\min\{N,\lfloor m/2\rfloor\}}
\frac{(-1)^r}{r(r-1)}
\sum_{\substack{2\le n_1,\ldots,n_r\le2N\\
n_1+\cdots+n_r=m}}
\mu_{n_1}\cdots\mu_{n_r}\,I(n_1,\ldots,n_r).
\end{equation}
Assume that for every $t\ge T$ there are measurable functions
$\delta_t,A_t,\rho_t$ such that
\begin{equation}\label{eq:density-to-entropy-descent-density}
\delta_t=A_t+\rho_t,\qquad
A_t(y)=\sum_{n=2}^{2N}\mu_nu_n(y)t^{-n},
\end{equation}
and that the following are the only analytic hypotheses:
\begin{equation}\label{eq:density-to-entropy-descent-hypotheses}
\|\rho_t\|_{L^\infty(\RR)}+\|\rho_t\|_{L^1(\omega)}
\le C_\rho t^{-(2N+\eta)},
\end{equation}
\[
\|\delta_t\|_{L^\infty(\RR)}\le\min\{1/2,S t^{-2}\},
\]
for fixed finite constants $C_\rho,S$.  Put
\[
M_A:=\sum_{n=2}^{2N}|\mu_n|\,\|u_n\|_{L^\infty(\RR)} .
\]
Then $\|A_t\|_{L^\infty(\RR)}\le M_A t^{-2}$ for $t\ge1$,
$\Phi(\delta_t)\in L^1(\omega)$ for
\[
\Phi(z)=(1+z)\log(1+z)-z,
\]
and, for every $t\ge\max\{1,T\}$,
\begin{equation}\label{eq:density-to-entropy-descent-expansion}
\int_{\RR}\Phi(\delta_t(y))\,\omega(\dd y)
=\sum_{m=4}^{2N}\mathcal A_m(\mu,u)t^{-m}
+\mathcal E_{N,t}.
\end{equation}
The remainder is controlled by the finite computable bound
\begin{equation}\label{eq:density-to-entropy-descent-remainder}
|\mathcal E_{N,t}|
\le
\mathfrak P_N(C_\rho,M_A)t^{-(2N+\eta+2)}
+\mathfrak Q_N(\mu,u)t^{-(2N+1)}
+2S^{N+1}t^{-(2N+2)},
\end{equation}
where
\begin{equation}\label{eq:entropy-constant-product-error}
\mathfrak P_N(C_\rho,M_A)
:=
\sum_{r=2}^{N}\frac{1}{r(r-1)}
\sum_{\ell=1}^{r}\binom r\ell M_A^{r-\ell}C_\rho^\ell
\end{equation}
and
\begin{equation}\label{eq:entropy-constant-truncation-error}
\mathfrak Q_N(\mu,u)
:=
\sum_{r=2}^{N}\frac{1}{r(r-1)}
\sum_{m=2N+1}^{2Nr}
\sum_{\substack{2\le n_1,\ldots,n_r\le2N\\
n_1+\cdots+n_r=m}}
|\mu_{n_1}\cdots\mu_{n_r}|\,
\|u_{n_1}\cdots u_{n_r}\|_{L^1(\omega)} .
\end{equation}
Thus the descent from an $L^\infty\cap L^1(\omega)$ density-ratio remainder to
the entropy integral is a finite calculation.  The finite
$(2N+\eta)$-moment input is used only before this proposition, to produce
\eqref{eq:density-to-entropy-descent-density} and the
$L^\infty\cap L^1(\omega)$ bound
\eqref{eq:density-to-entropy-descent-hypotheses}.  Inside the transfer,
Lemma~\ref{lem:density-remainder-product-hierarchy} shows that every product
containing at least one $\rho_t$ is in fact $O(t^{-(2N+\eta+2)})$ after the
entropy quadratic order is used.  The term
$\mathfrak Q_N(\mu,u)t^{-(2N+1)}$ is not a density-remainder loss: it is the
finite high-degree tail of the retained mode polynomial.  Since
$0<\eta\le1$, the coarser envelope
\begin{equation}\label{eq:density-to-entropy-convenience-envelope}
|\mathcal E_{N,t}|
\le
\bigl(\mathfrak P_N(C_\rho,M_A)+\mathfrak Q_N(\mu,u)\bigr)t^{-(2N+\eta)}
+2S^{N+1}t^{-(2N+2)}
\end{equation}
also follows for $t\ge1$.  Later displayed constants using
$C_{N,\eta}^{\rm dens}$ are this convenience envelope, not sharp product
constants.

In the Poisson finite-moment application of
Theorem~\ref{thm:finite-moment-density-remainder} with $L=2N$, where
$C_\rho=C_{\nu,2N,\eta}$ and $S=S_{\nu,2N,\eta}$, the fixed-law constants may
therefore be chosen as
\[
T_{N,\eta}^{\rm ent}(\nu):=\max\{1,T_{\nu,2N,\eta}\},
\]
and
\begin{equation}\label{eq:explicit-entropy-constant-bound}
C_{N,\eta}^{\rm dens}(\nu)
=\mathfrak P_N(C_{\nu,2N,\eta},M_A)+\mathfrak Q_N(\mu,u),
\qquad
C_{N,\eta}^{\rm Tay}(\nu)=2S_{\nu,2N,\eta}^{N+1}.
\end{equation}
The refined fixed-law bound supplied by
\eqref{eq:density-to-entropy-descent-remainder} is
\begin{equation}\label{eq:explicit-entropy-constant-refined-bound}
\mathfrak P_N(C_{\nu,2N,\eta},M_A)t^{-(2N+\eta+2)}
+\mathfrak Q_N(\mu,u)t^{-(2N+1)}
+C_{N,\eta}^{\rm Tay}(\nu)t^{-(2N+2)}.
\end{equation}
These are finite expressions in the density-remainder constant from
Theorem~\ref{thm:finite-moment-density-remainder}, the moments through order
$2N$, and the finite mode-product integrals.  If one wants a purely numerical
envelope, or the coarser two-term form
\eqref{eq:density-to-entropy-convenience-envelope},
Lemma~\ref{lem:closed-cayley-mode-constant-envelopes} gives
$\|u_n\|_\infty\le n+1$ and hence
\[
\|u_{n_1}\cdots u_{n_r}\|_{L^1(\omega)}
\le\prod_{j=1}^{r}(n_j+1).
\]
The proposition records usable bounds for each specified law.  The
exact-threshold identification of the universal moment-polynomial coefficient
\(A_{2N}\) in Theorem~\ref{thm:poisson-kl-even-orders} uses the standalone
exact-moment transfer lemma stated immediately before that theorem.
\end{proposition}

\begin{proof}
Since $\omega$ has total mass one and the functions $u_n$ are bounded, every
finite product in \eqref{eq:density-to-entropy-descent-coefficient} belongs to
$L^1(\omega)$.  Thus the displayed coefficient polynomial consists of
absolutely convergent integrals and finite sums.
For $t\ge1$ the finite-mode part satisfies
\[
\|A_t\|_\infty
\le\sum_{n=2}^{2N}|\mu_n|\,\|u_n\|_\infty t^{-n}
\le M_A t^{-2}.
\]

For $|z|\le1/2$,
\[
\Phi(z)=\sum_{r=2}^{N}\frac{(-1)^r}{r(r-1)}z^r+R_{N+1}(z),
\qquad |R_{N+1}(z)|\le2|z|^{N+1}.
\]
The smallness assumption on $\delta_t$ therefore implies
$\Phi(\delta_t)\in L^1(\omega)$ and
\[
\|R_{N+1}(\delta_t)\|_{L^1(\omega)}
\le2S^{N+1}t^{-(2N+2)}.
\]
For each $2\le r\le N$, apply
Lemma~\ref{lem:density-remainder-product-hierarchy} with
$p=2N+\eta$, $M=M_A$, and $C=C_\rho$.  Since $2N+\eta>2$, this gives
\[
\|\delta_t^r-A_t^r\|_{L^1(\omega)}
\le
\sum_{\ell=1}^{r}\binom r\ell
M_A^{r-\ell}C_\rho^\ell t^{-(2N+\eta+2)}.
\]
After multiplication by $1/(r(r-1))$ and summation in $r$, this gives the
$\mathfrak P_N(C_\rho,M_A)t^{-(2N+\eta+2)}$ part of the refined remainder.

It remains only to expand the finite polynomial $A_t^r$.  The exact identity
\[
\int_{\RR}A_t(y)^r\,\omega(\dd y)
=
\sum_{m=2r}^{2Nr}t^{-m}
\sum_{\substack{2\le n_1,\ldots,n_r\le2N\\
n_1+\cdots+n_r=m}}
\mu_{n_1}\cdots\mu_{n_r}\,I(n_1,\ldots,n_r)
\]
splits into the retained part $m\le2N$ and the discarded part $m>2N$.  The
retained part, summed over $2\le r\le N$ with Taylor coefficients, is exactly
$\sum_{m=4}^{2N}\mathcal A_m(\mu,u)t^{-m}$.  The discarded part is bounded in
absolute value by $\mathfrak Q_N(\mu,u)t^{-(2N+1)}$.  Combining this
high-degree polynomial tail with the product error and the Taylor tail proves
\eqref{eq:density-to-entropy-descent-expansion} and
\eqref{eq:density-to-entropy-descent-remainder}.
\end{proof}

\begin{lemma}[Closed moment-ball domination of density and entropy constants]%
\label{lem:moment-ball-uniform-constant-domination}
Fix $N\ge2$, $0<\eta\le1$, and $M<\infty$, and put $p=2N+\eta$.  There are
finite constants
\[
\overline C_{N,\eta,M}^{\rho},\qquad
\overline S_{N,\eta,M},
\qquad
\overline C_{N,\eta,M}^{\rm dens},
\qquad
\overline C_{N,\eta,M}^{\rm Tay},
\qquad
\overline T_{N,\eta,M}^{\rm ent}
\]
given by closed finite formulas in $N,\eta,M$ below, such that every
probability measure $\nu$ with finite mean $\bar\gamma$ and
\[
\int_{\RR}|\gamma-\bar\gamma|^p\dd\nu(\gamma)\le M
\]
satisfies
\begin{align*}
C_{\nu,2N,\eta}&\le \overline C_{N,\eta,M}^{\rho},
&
S_{\nu,2N,\eta}&\le \overline S_{N,\eta,M},
\\
C_{N,\eta}^{\rm dens}(\nu)&\le
\overline C_{N,\eta,M}^{\rm dens},
&
C_{N,\eta}^{\rm Tay}(\nu)&\le
\overline C_{N,\eta,M}^{\rm Tay},
&
T_{N,\eta}^{\rm ent}(\nu)&\le
\overline T_{N,\eta,M}^{\rm ent}.
\end{align*}
Consequently the density remainder
\eqref{eq:finite-moment-density-remainder}, the density smallness threshold
\eqref{eq:finite-moment-density-smallness}, and the entropy constants in
\eqref{eq:explicit-entropy-constant-bound} are uniformly dominated on the
centred $p$-moment ball by constants involving no unspecified mode norms.
\end{lemma}

\begin{proof}
Let
\[
\widehat B:=2N+1,\qquad
\widehat C:=\max\!\left\{\frac{10N+7}{2},\,6+(2N+1)^2\right\}.
\]
These are the closed envelopes for $B_{2N}$ and $C_{2N}$ from
Lemma~\ref{lem:closed-cayley-mode-constant-envelopes}.  Define
\[
\overline C_{N,\eta,M}^{\rho}:=2\widehat C M,
\qquad
\overline S_{N,\eta,M}:=
\widehat B\sum_{n=2}^{2N}M^{n/p}+2\widehat C M,
\]
\[
\overline M_{A,N,\eta,M}:=\sum_{n=2}^{2N}(n+1)M^{n/p}.
\]
For the finite high-degree mode tail set
\[
\overline{\mathfrak Q}_{N,\eta,M}:=
\sum_{r=2}^{N}\frac{1}{r(r-1)}
\sum_{m=2N+1}^{2Nr}
\sum_{\substack{2\le n_1,\dots,n_r\le2N\\
n_1+\cdots+n_r=m}}
\left(\prod_{j=1}^{r}M^{n_j/p}(n_j+1)\right),
\]
and set
\[
\overline{\mathfrak P}_{N,\eta,M}:=
\sum_{r=2}^{N}\frac{1}{r(r-1)}
\sum_{\ell=1}^{r}\binom r\ell
(\overline M_{A,N,\eta,M})^{r-\ell}
(\overline C_{N,\eta,M}^{\rho})^\ell .
\]
Finally define
\[
\overline C_{N,\eta,M}^{\rm dens}:=
\overline{\mathfrak P}_{N,\eta,M}
+\overline{\mathfrak Q}_{N,\eta,M},
\qquad
\overline C_{N,\eta,M}^{\rm Tay}:=
2(\overline S_{N,\eta,M})^{N+1},
\]
\[
\overline T_{N,\eta,M}^{\rm ent}:=
\max\{1,(2\overline S_{N,\eta,M})^{1/2}\}.
\]

For $2\le n\le2N$, Lyapunov's inequality gives
$|\mu_n|\le M^{n/p}$.  The density theorem with $L=2N$ and
Lemma~\ref{lem:closed-cayley-mode-constant-envelopes} give the closed
constant bound
\[
C_{\nu,2N,\eta}=2C_{2N}\EE|X_c|^p\le2\widehat C M
=\overline C_{N,\eta,M}^{\rho},
\]
and since
\[
S_{\nu,2N,\eta}=B_{2N}\sum_{n=2}^{2N}|\mu_n|+C_{\nu,2N,\eta},
\]
the same moment bounds give
\[
\overline S_{N,\eta,M}:=
\widehat B\sum_{n=2}^{2N}M^{n/p}+2\widehat C M .
\]
The threshold definition then yields
\[
T_{N,\eta}^{\rm ent}(\nu)=\max\{1,T_{\nu,2N,\eta}\}
\le \max\{1,(2\overline S_{N,\eta,M})^{1/2}\}
=:\overline T_{N,\eta,M}^{\rm ent}.
\]

It remains only to dominate the finite entropy sums in
Proposition~\ref{prop:density-to-entropy-descent}.  The closed mode bound
$\|u_n\|_\infty\le n+1$ gives
$M_A\le\overline M_{A,N,\eta,M}$.  Substituting this and
$C_\rho\le\overline C_{N,\eta,M}^{\rho}$ into the finite sum
\eqref{eq:entropy-constant-product-error} gives
$\mathfrak P_N(C_{\nu,2N,\eta},M_A)
\le\overline{\mathfrak P}_{N,\eta,M}$.
Substituting the bounds
$|\mu_{n_j}|\le M^{n_j/p}$ and
\[
\|u_{n_1}\cdots u_{n_r}\|_{L^1(\omega)}
\le \prod_{j=1}^{r}\|u_{n_j}\|_\infty
\le \prod_{j=1}^{r}(n_j+1),
\]
because $\omega$ has total mass one, gives
$\mathfrak Q_N(\mu,u)\le\overline{\mathfrak Q}_{N,\eta,M}$.  Hence
$C_{N,\eta}^{\rm dens}(\nu)\le
\overline C_{N,\eta,M}^{\rm dens}$.  Finally
$C_{N,\eta}^{\rm Tay}(\nu)=2S_{\nu,2N,\eta}^{N+1}$ is bounded by
$\overline C_{N,\eta,M}^{\rm Tay}$.  All constants are therefore controlled
by the displayed finite formulas in $N,\eta,M$ alone.
\end{proof}

\begin{lemma}[Exact-moment entropy transfer]\label{lem:exact-moment-entropy-transfer}
Let \(L\ge2\), let \(\omega\) be a probability measure, and let
\[
\delta_t=d_t+\rho_t,
\qquad
d_t(y):=\sum_{n=2}^{L}\mu_nu_n(y)t^{-n},
\]
where \(u_2,\ldots,u_L\) are bounded measurable functions.  Assume that
as \(t\to\infty\),
\[
\|d_t\|_\infty=O(t^{-2}),\qquad
\|\delta_t\|_\infty=O(t^{-2}),\qquad
\|\rho_t\|_\infty+\|\rho_t\|_{L^1(\omega)}=o(t^{-L}).
\]
Equivalently, there is a finite threshold \(T_\delta\) such that
\(\|\delta_t\|_\infty\le1/2\) for all \(t\ge T_\delta\); the expansion below
is asserted in this smallness regime and then as \(t\to\infty\).
For
\[
\Phi(s)=(1+s)\log(1+s)-s,
\]
one has
\begin{equation}\label{eq:exact-moment-entropy-transfer}
\int \Phi(\delta_t)\,\dd\omega
=
\sum_{m=4}^{L+2}
\sum_{r=2}^{\lfloor m/2\rfloor}
\frac{(-1)^r}{r(r-1)}
\sum_{\substack{n_1,\ldots,n_r\ge2\\
n_1+\cdots+n_r=m}}
\mu_{n_1}\cdots\mu_{n_r}\,I(n_1,\ldots,n_r)
t^{-m}
+o(t^{-(L+2)}),
\end{equation}
where
\[
I(n_1,\ldots,n_r):=\int u_{n_1}(y)\cdots u_{n_r}(y)\,\dd\omega(y).
\]
In the Poisson application, if
\[
\delta_t(y)=\frac{h_t(\bar\gamma+ty)}{g_t(\bar\gamma+ty)}-1,
\qquad g_t(x)=P_t(x-\bar\gamma),
\]
then \(g_t(\bar\gamma+ty)t\,\dd y=\dd\omega(y)\), and the left-hand side is
exactly
\[
D_{\mathrm{KL}}(h_t\|g_t)
=\int_{\RR}\Phi(\delta_t(y))\,\dd\omega(y).
\]
The inner sum is understood with the finite convention
\(\mu_nu_n=0\) for \(n>L\), equivalently only indices \(2\le n_j\le L\)
contribute.  In particular, when the \(u_n\) are the Cayley--Chebyshev modes
and the \(\mu_n\) are centred moments, the displayed coefficients are the
finite-moment evaluations \(A_m(\nu)\) for \(4\le m\le L+2\).
\end{lemma}

\begin{proof}
Choose \(T<\infty\) so that \(\|\delta_t\|_\infty\le1/2\) for \(t\ge T\);
this follows from \(\|\delta_t\|_\infty=O(t^{-2})\).  For \(t\ge T\),
\[
\Phi(s)=
\sum_{r=2}^{\lfloor (L+2)/2\rfloor}
\frac{(-1)^r}{r(r-1)}s^r
+O(|s|^{\lfloor (L+2)/2\rfloor+1})
\qquad(|s|\le1/2).
\]
Since \(\|\delta_t\|_\infty=O(t^{-2})\) and \(\omega\) is a probability
measure, the Taylor tail contributes
\[
O\!\left(t^{-2(\lfloor (L+2)/2\rfloor+1)}\right)=o(t^{-(L+2)}),
\]
because this exponent is \(L+3\) when \(L\) is odd and \(L+4\) when \(L\) is
even.

Fix \(2\le r\le\lfloor (L+2)/2\rfloor\).  In the binomial expansion of
\((d_t+\rho_t)^r-d_t^r\), every mixed term with \(1\le\ell\le r\) factors of
\(\rho_t\) satisfies the key \(L^\infty\)-\(L^1(\omega)\) placement
\begin{equation}\label{eq:exact-transfer-product-placement}
\|d_t^{\,r-\ell}\rho_t^\ell\|_{L^1(\omega)}
\le
\|d_t\|_\infty^{r-\ell}
\|\rho_t\|_\infty^{\ell-1}
\|\rho_t\|_{L^1(\omega)} .
\end{equation}
Thus
\[
\|d_t^{\,r-\ell}\rho_t^\ell\|_{L^1(\omega)}
=O(t^{-2(r-\ell)})\,o(t^{-L\ell}).
\]
The worst mixed term is at \(r=2,\ell=1\):
\[
\|d_t\rho_t\|_{L^1(\omega)}=O(t^{-2})\,o(t^{-L})
=o(t^{-(L+2)}).
\]
More generally, for \(\ell=1\) this gives
\[
O(t^{-2(r-1)})\,o(t^{-L})=o(t^{-(L+2r-2)})
=o(t^{-(L+2)}).
\]
For \(\ell\ge2\) the same bound is smaller: if \(r=\ell\), it is
\(o(t^{-L\ell})=o(t^{-(L+2)})\), while if \(r>\ell\) it contains the
additional factor \(t^{-2(r-\ell)}\); here \(L\ge2\).  Hence
\[
\|\delta_t^r-d_t^r\|_{L^1(\omega)}=o(t^{-(L+2)})
\]
for each retained Taylor degree \(r\).

The pure \(d_t\)-terms give the finite polynomial.  For every retained \(r\),
\[
\int d_t(y)^r\,\dd\omega(y)
=
\sum_{m=2r}^{Lr}t^{-m}
\sum_{\substack{2\le n_1,\ldots,n_r\le L\\
n_1+\cdots+n_r=m}}
\mu_{n_1}\cdots\mu_{n_r}\,I(n_1,\ldots,n_r).
\]
Terms with \(m>L+2\) are \(O(t^{-(L+3)})\), hence are absorbed in the
little-oh remainder.  Summing the retained pure terms with the Taylor
coefficients gives \eqref{eq:exact-moment-entropy-transfer}.  Finally, no
mode \(u_{L+1}\), or any higher mode, can enter a coefficient of total degree
\(m\le L+2\), because every entropy monomial has at least two factors and
every index is at least \(2\); an index \(L+1\) would force total degree at
least \(L+3\).  For the Cayley--Chebyshev modes this is precisely the
finite-moment coefficient evaluation
\eqref{eq:universal-entropy-coefficient-evaluation}.
\end{proof}

Before stating the coefficient theorem, recall the scope fixed in the
introduction.  The law \(\nu\) remains fixed, and the Poisson time \(t\) tends
to infinity in the scalar functional
\[
D_{\mathrm{KL}}(P_t*\nu\|P_t(\cdot-\bar\gamma)).
\]
Large-time moment expansions for heat, fractional heat, and geometric
heat-type flows give profile corrections for the evolved density or operator
kernel
\cite{DuoandikoetxeaZuazua1992,EscobedoZuazua1991,KarchSchonbek2001,IshigeKawakami2012,IshigeKawakamiMichihisa2017,AnkerPapageorgiouZhang2023HeatSymmetricSpaces,Papageorgiou2024FractionalLargeTime}.
Extension-method papers for the fractional Laplacian
\cite{CaffarelliSilvestre2007,StingaTorrea2010} fix a standard route from
nonlocal operators to local degenerate problems and provide normalization
background.
Stable-law entropy expansions and recent JFA entropic CLT work concern
convolution powers, normalized sums, or Esscher-transformed sums
\cite{BobkovChistyakovGotze2013,BobkovChistyakovGotzeEdgeworth2013,BobkovChistyakovGotzeBerryEsseen2014,BobkovGotzeEsscher2025}.
The theorems below treat fixed-law Poisson comparisons
\[
D_{\mathrm{KL}}(P_t*\nu\|P_t(\cdot-\bar\gamma)),
\]
and its radial \(d\)-dimensional analogue, after the mean-matched Poisson
normalization.  The first theorem records the finite-moment entropy jet in
all dimensions for the radial Poisson kernel.  The following scalar theorem
then evaluates the one-dimensional coefficients explicitly.  Within that
scalar functional, the Cayley--Chebyshev quotient
\((1+y^2)/(1+(y-X_c/t)^2)\) makes every retained entropy coefficient a
finite polynomial in centred moments, and the exact \((2N-2)\)-moment
threshold is the threshold for identifying the universal coefficient
\(A_{2N}\).  The increment is a finite-moment Poisson entropy-jet theorem,
with a one-dimensional Laurent extraction layer that makes the \(N=4\)
coefficients explicit.

\begin{unnumberedlemma}[Uniform radial Poisson Taylor remainder]
Let \(d\ge1\), put \(a=(d+1)/2\), and, for \(y,z\in\RR^d\), set
\[
F_y(z):=
\left(\frac{1+|y|^2}{1+|y-z|^2}\right)^a .
\]
For a multiindex \(\alpha\in\NN_0^d\), define
\[
U_\alpha^{(d)}(y):=\frac{1}{\alpha!}\,\partial_z^\alpha F_y(0).
\]
If \(L\ge d+1\), then there is a constant \(C_{d,L}<\infty\) such that
\begin{equation}\label{eq:multidim-uniform-taylor-remainder-bound}
\left|
F_y(z)-\sum_{|\alpha|\le L}U_\alpha^{(d)}(y)z^\alpha
\right|
\le C_{d,L}\Bigl(
|z|^{L+1}\mathbf1_{\{|z|\le1\}}
+|z|^L\mathbf1_{\{|z|>1\}}
\Bigr)
\end{equation}
for all \(y,z\in\RR^d\).  Moreover
\[
\sup_{y\in\RR^d}|U_\alpha^{(d)}(y)|<\infty
\qquad (|\alpha|\le L).
\]
Consequently, if \(X\) is an \(\RR^d\)-valued random vector with
\(\EE|X|^L<\infty\), \(\Omega_d\) is any probability measure on \(\RR^d\),
and
\[
r_t(y):=\EE\left[
F_y(X/t)-\sum_{|\alpha|\le L}U_\alpha^{(d)}(y)(X/t)^\alpha
\right],
\]
then
\begin{equation}\label{eq:multidim-uniform-density-remainder-consequence}
\|r_t\|_{L^\infty(\RR^d)}
+\|r_t\|_{L^1(\Omega_d)}
=o(t^{-L}).
\end{equation}
\end{unnumberedlemma}

\begin{proof}
For each multiindex \(\beta\), repeated differentiation gives the sharper
structure
\begin{equation}\label{eq:multidim-poisson-derivative-structure}
D_z^\beta F_y(w)
=(1+|y|^2)^a
\frac{P_\beta(y-w)}{(1+|y-w|^2)^{a+|\beta|}},
\end{equation}
where \(P_\beta\) is a polynomial, depending only on \(d\) and \(\beta\),
of degree at most \(|\beta|\).  Indeed, \(D_{z_j}=-D_{\xi_j}\) for
\(\xi=y-w\), and differentiating \((1+|\xi|^2)^{-b}\) once raises the
denominator exponent by one and multiplies by a polynomial of degree at most
one more.  If \(|w|\le1\) and \(|y|>2\), then \(|y-w|\ge |y|/2\), and
\[
|D_z^\beta F_y(w)|
\le
C_{d,\beta}
(1+|y|^2)^a
\frac{(1+|y|)^{|\beta|}}{(1+|y|^2)^{a+|\beta|}}
\le C_{d,\beta}.
\]
On \(|y|\le2\), the same derivative is bounded by compactness because
\(1+|y-w|^2\ge1\).  Hence, for every fixed \(L\),
\[
\sup_{\substack{y\in\RR^d\\ |w|\le1\\ |\beta|=L+1}}
|\partial_z^\beta F_y(w)|<\infty .
\]
The multivariable Taylor formula with integral remainder at \(z=0\) then
gives the first term on the right-hand side of
\eqref{eq:multidim-uniform-taylor-remainder-bound} when \(|z|\le1\).
Taking \(w=0\) in the preceding derivative bound, with
\(|\beta|\le L\), also gives the asserted boundedness of
\(U_\alpha^{(d)}\).

For the large-\(z\) bound, split according to whether
\(|y-z|\le |z|/2\).  If \(|y-z|\le |z|/2\), then \(|y|\le3|z|/2\), and
\[
F_y(z)\le (1+|y|^2)^a\le C_d|z|^{d+1}
\qquad(|z|>1).
\]
On the complementary region \(|y-z|>|z|/2\), the elementary inequality
\[
1+|y|^2
\le
2(1+|z|^2)(1+|y-z|^2),
\]
which follows from \(|y|\le |y-z|+|z|\).  Therefore
\[
F_y(z)\le C_d(1+|z|^2)^a\le C_d'|z|^{d+1}
\qquad (|z|>1),
\]
also on the complement.  Since \(L\ge d+1\), this is bounded by
\(C_{d,L}|z|^L\).  The boundedness of the coefficients gives, for
\(|z|>1\),
\[
\left|\sum_{|\alpha|\le L}U_\alpha^{(d)}(y)z^\alpha\right|
\le C_{d,L}\sum_{k=0}^L |z|^k
\le C_{d,L}'|z|^L.
\]
Combining the two estimates proves
\eqref{eq:multidim-uniform-taylor-remainder-bound}.

With \(z=X/t\), the bound just proved gives, uniformly in \(y\),
\[
|r_t(y)|
\le
C_{d,L}t^{-L}\EE\left[
|X|^L\frac{|X|}{t}\mathbf1_{\{|X|\le t\}}
+|X|^L\mathbf1_{\{|X|>t\}}
\right].
\]
The first expectation is \(o(1)\) by dominated convergence, and the second is
\(o(1)\) by integrability of \(|X|^L\).  This proves the
\(L^\infty\)-part of
\eqref{eq:multidim-uniform-density-remainder-consequence}.  The
\(L^1(\Omega_d)\)-part follows from the same uniform bound because
\(\Omega_d\) is a probability measure.
\end{proof}

\begin{theorem}[Multidimensional finite-moment Poisson entropy jets]%
\label{thm:multidimensional-poisson-entropy-jets}
Let \(d\ge1\), let
\[
P_t^{(d)}(x):=c_d\,\frac{t}{(t^2+|x|^2)^{(d+1)/2}},
\qquad x\in\RR^d,\ t>0,
\]
be the radial Poisson kernel on \(\RR^d\), normalized to integrate to one.
Let \(\nu\) be a probability measure on \(\RR^d\) with mean
\(\bar\gamma\), put \(X_c:=\gamma-\bar\gamma\), and set
\[
h_t:=P_t^{(d)}*\nu,\qquad
g_t(x):=P_t^{(d)}(x-\bar\gamma).
\]
Fix an integer \(L\ge\max\{2,d+1\}\), and assume
\(\EE|X_c|^L<\infty\).  For multiindices \(\alpha\in\NN_0^d\), write
\[
\mu_\alpha:=\EE[X_c^\alpha]\quad(|\alpha|\le L).
\]
Let
\[
\Omega_d(\dd y):=P_1^{(d)}(y)\dd y
\]
and define the universal functions \(U_\alpha^{(d)}\) by the Taylor expansion
at \(z=0\)
\begin{equation}\label{eq:multidim-poisson-ratio-taylor}
\left(\frac{1+|y|^2}{1+|y-z|^2}\right)^{(d+1)/2}
=\sum_{\alpha\in\NN_0^d}U_\alpha^{(d)}(y)z^\alpha .
\end{equation}
For \(2\le j\le L\), group the centred tensor moments of homogeneous order
\(j\) into the density layer
\begin{equation}\label{eq:multidim-density-layers}
D_j^{(d)}(y;\nu):=
\sum_{|\alpha|=j}\mu_\alpha U_\alpha^{(d)}(y).
\end{equation}
The preceding uniform radial Poisson Taylor lemma supplies, uniformly in
\(y\), both the finite Taylor remainder bound
\eqref{eq:multidim-uniform-taylor-remainder-bound} and its
\(L^\infty\cap L^1(\Omega_d)\) integrated consequence
\eqref{eq:multidim-uniform-density-remainder-consequence}; these are the only
large-\(z\) and density-remainder estimates used below.
For \(4\le m\le L+2\), define
\begin{equation}\label{eq:multidim-entropy-coefficient}
\mathcal A_m^{(d)}(\nu):=
\sum_{r=2}^{\lfloor m/2\rfloor}
\frac{(-1)^r}{r(r-1)}
\sum_{\substack{|\alpha_1|,\ldots,|\alpha_r|\ge2\\
|\alpha_1|+\cdots+|\alpha_r|=m\\
|\alpha_j|\le L}}
\mu_{\alpha_1}\cdots\mu_{\alpha_r}
\int_{\RR^d}\prod_{j=1}^rU_{\alpha_j}^{(d)}(y)\,\Omega_d(\dd y).
\end{equation}
Equivalently,
\begin{equation}\label{eq:multidim-entropy-coefficient-layers}
\mathcal A_m^{(d)}(\nu)=
\sum_{r=2}^{\lfloor m/2\rfloor}
\frac{(-1)^r}{r(r-1)}
\sum_{\substack{2\le j_1,\ldots,j_r\le L\\
j_1+\cdots+j_r=m}}
\int_{\RR^d}D_{j_1}^{(d)}(y;\nu)\cdots
D_{j_r}^{(d)}(y;\nu)\,\Omega_d(\dd y).
\end{equation}
Then
\begin{equation}\label{eq:multidim-poisson-entropy-jet}
D_{\rm KL}(h_t\|g_t)
=\sum_{m=4}^{L+2}\mathcal A_m^{(d)}(\nu)t^{-m}
+o(t^{-(L+2)})
\qquad(t\to\infty).
\end{equation}
The odd coefficients vanish:
\begin{equation}\label{eq:multidim-odd-parity-cancellation}
\mathcal A_{2q+1}^{(d)}(\nu)=0
\qquad(2q+1\le L+2).
\end{equation}
This cancellation is immediate from
\(F_{-y}(z)=F_y(-z)\), hence
\(U_\alpha^{(d)}(-y)=(-1)^{|\alpha|}U_\alpha^{(d)}(y)\), together with the
evenness of \(\Omega_d\).
The first potentially nonzero coefficients are therefore
\begin{align}
\mathcal A_4^{(d)}
&=\frac12\int_{\RR^d}D_2^2\,\dd\Omega_d,
\label{eq:multidim-first-coeff-a4}\\
\mathcal A_6^{(d)}
&=\int_{\RR^d}D_2D_4\,\dd\Omega_d
+\frac12\int_{\RR^d}D_3^2\,\dd\Omega_d
-\frac16\int_{\RR^d}D_2^3\,\dd\Omega_d,
\label{eq:multidim-first-coeff-a6}\\
\mathcal A_8^{(d)}
&=\int_{\RR^d}D_2D_6\,\dd\Omega_d
+\int_{\RR^d}D_3D_5\,\dd\Omega_d
+\frac12\int_{\RR^d}D_4^2\,\dd\Omega_d
\notag\\
&\quad
-\frac12\int_{\RR^d}D_2^2D_4\,\dd\Omega_d
-\frac12\int_{\RR^d}D_2D_3^2\,\dd\Omega_d
+\frac1{12}\int_{\RR^d}D_2^4\,\dd\Omega_d,
\label{eq:multidim-first-coeff-a8}
\end{align}
with the convention that a displayed term containing \(D_j\) is omitted when
\(j>L\).  Here and below in this theorem \(D_j=D_j^{(d)}(\,\cdot\,;\nu)\).
The coefficients in \eqref{eq:multidim-entropy-coefficient} are finite and
universal: they depend on \(d\), \(m\), and the centred tensor moments
\(\{\mu_\alpha:2\le|\alpha|\le m-2\}\), but not on any higher moment of
\(\nu\).  In particular a contribution of total entropy degree \(m\) cannot
contain a tensor moment of order above \(m-2\).  The leading term is
coercive in the covariance: Theorem~\ref{thm:multidim-leading-positive-defect}
gives an explicit positive decomposition of \(\mathcal A_4^{(d)}\), so
\(\mathcal A_4^{(d)}\ge0\) and equality holds exactly for zero covariance.
No fixed sign is asserted for the higher even coefficients; the formulas
\eqref{eq:multidim-first-coeff-a6}--\eqref{eq:multidim-first-coeff-a8}
display the alternating Bregman-Taylor contributions that destroy a general
positivity statement beyond the leading quadratic layer.  The tensor-moment
order \(m-2\) is the top order that can enter the universal coefficient
\(\mathcal A_m^{(d)}\), and for even \(m\) it enters only through the
quadratic-layer pairing \(\int D_2D_{m-2}\,\dd\Omega_d\); this is the
algebraic sharpness statement for the displayed universal coefficient, not a
claim that every component of the top tensor is detected for every covariance.
\end{theorem}

\begin{proof}
Write \(a=(d+1)/2\).  For \(x=\bar\gamma+ty\),
\[
\frac{P_t^{(d)}(x-\gamma)}{g_t(x)}
=
\left(\frac{t^2+|x-\bar\gamma|^2}{t^2+|x-\gamma|^2}\right)^a
=
\left(\frac{1+|y|^2}{1+|y-X_c/t|^2}\right)^a.
\]
Thus the density quotient in the scale \(\Omega_d\) is
\[
\widetilde\delta_t(y)
:=\frac{h_t(\bar\gamma+ty)}{g_t(\bar\gamma+ty)}-1
=\EE\!\left[
\left(\frac{1+|y|^2}{1+|y-X_c/t|^2}\right)^a-1
\right].
\]
The change of variables gives
\[
g_t(\bar\gamma+ty)t^d\dd y=P_1^{(d)}(y)\dd y=\Omega_d(\dd y),
\]
and therefore
\[
D_{\rm KL}(h_t\|g_t)
=\int_{\RR^d}\Phi(\widetilde\delta_t(y))\,\Omega_d(\dd y),
\qquad
\Phi(s)=(1+s)\log(1+s)-s.
\]

We first obtain the finite density expansion.  Apply
the uniform radial Poisson Taylor lemma with \(X=X_c\).
Its uniform Taylor-remainder consequence gives the error term below in both
\(L^\infty(\RR^d)\) and \(L^1(\Omega_d)\).  Since
\(U_0^{(d)}=1\) and the first-order terms integrate to zero against the
centred variable \(X_c\), we get
\begin{equation}\label{eq:multidim-density-expansion}
\widetilde\delta_t(y)
=\sum_{j=2}^{L}D_j^{(d)}(y;\nu)t^{-j}
+\rho_{L,t}(y),
\qquad
\|\rho_{L,t}\|_\infty+\|\rho_{L,t}\|_{L^1(\Omega_d)}=o(t^{-L}).
\end{equation}
The terms of orders \(0\) and \(1\) cancel because \(\nu\) has total mass one
and mean zero in the centred variable.
The proof of \eqref{eq:multidim-density-expansion} is exactly the tail split
in the Taylor lemma, with \(z=X_c/t\): the region \(|X_c|\le t\) gives the
integrable factor
\(|X_c|^L(|X_c|/t)\), while \(|X_c|>t\) gives the finite \(L\)th-moment tail.

The finite sum in \eqref{eq:multidim-density-expansion} starts at order
\(t^{-2}\).  The boundedness of the retained \(U_\alpha^{(d)}\)'s from
the uniform radial Poisson Taylor lemma, the finiteness of
the moments \(\mu_\alpha\) with \(|\alpha|\le L\), and
\(\|\rho_{L,t}\|_\infty=o(t^{-L})\) give
\(\|\widetilde\delta_t\|_\infty=O(t^{-2})\).  For large \(t\),
\(|\widetilde\delta_t|\le1/2\), so the Taylor expansion of \(\Phi\) may be
used through degree \(\lfloor (L+2)/2\rfloor\).  Products containing at least
one \(\rho_{L,t}\) are
\[
o(t^{-L})\,O(t^{-2(r-1)})=o(t^{-(L+2)})
\]
in \(L^1(\Omega_d)\), uniformly for each retained Taylor degree \(r\ge2\).
The Taylor tail is \(O(t^{-2q})\) with
\(q=\lfloor (L+2)/2\rfloor+1\), hence \(o(t^{-(L+2)})\).  Multiplying the
finite density polynomial and retaining total tensor degree \(m\le L+2\)
gives exactly \eqref{eq:multidim-entropy-coefficient}, proving
\eqref{eq:multidim-poisson-entropy-jet}.  Since every entropy monomial has
at least two density factors, each individual multiindex in a total degree
\(m\) contribution has length at most \(m-2\).

It remains only to record the parity and the displayed low-order formulas.
The identity \(F_{-y}(z)=F_y(-z)\) implies
\[
U_\alpha^{(d)}(-y)=(-1)^{|\alpha|}U_\alpha^{(d)}(y),
\qquad
D_j^{(d)}(-y;\nu)=(-1)^jD_j^{(d)}(y;\nu).
\]
Since \(\Omega_d\) is even, every product in
\eqref{eq:multidim-entropy-coefficient-layers} with odd total degree has zero
integral.  This proves \eqref{eq:multidim-odd-parity-cancellation}.  The
formulas \eqref{eq:multidim-first-coeff-a4}--\eqref{eq:multidim-first-coeff-a8}
are obtained by listing the ordered compositions of \(m\) by parts at least
two and applying the Taylor coefficients \(1/2,-1/6,1/12\) of \(\Phi\).
For even \(m\), the only compositions containing a part \(m-2\) are
\((2,m-2)\) and \((m-2,2)\); their combined contribution is
\(\int D_2D_{m-2}\,\dd\Omega_d\).  All remaining compositions use only
density layers of order at most \(m-3\).
\end{proof}

\begin{proposition}[First multidimensional entropy layers and odd cancellation]%
\label{prop:multidim-first-entropy-layers-parity}
In the notation of
Theorem~\ref{thm:multidimensional-poisson-entropy-jets}, put
\[
D_j(y)=\sum_{|\alpha|=j}\mu_\alpha U_\alpha^{(d)}(y).
\]
If \(L\ge4\), then
\begin{equation}\label{eq:multidim-kl-through-six-proposition}
D_{\rm KL}(h_t\|g_t)
=
\frac12t^{-4}\int_{\RR^d}D_2^2\,\dd\Omega_d
+t^{-6}\left(
\int_{\RR^d}D_2D_4\,\dd\Omega_d
+\frac12\int_{\RR^d}D_3^2\,\dd\Omega_d
-\frac16\int_{\RR^d}D_2^3\,\dd\Omega_d
\right)
+o(t^{-6}).
\end{equation}
Moreover, all odd entropy coefficients in the multidimensional jet vanish:
\[
\mathcal A_{2q+1}^{(d)}(\nu)=0.
\]
\end{proposition}

\begin{proof}
The expansion \eqref{eq:multidim-kl-through-six-proposition} is the
\(m=4\) and \(m=6\) part of
\eqref{eq:multidim-poisson-entropy-jet}, equivalently
\eqref{eq:multidim-first-coeff-a4} and
\eqref{eq:multidim-first-coeff-a6}.  For parity,
\[
F_{-y}(z)=F_y(-z)\quad\Rightarrow\quad
U_\alpha^{(d)}(-y)=(-1)^{|\alpha|}U_\alpha^{(d)}(y),
\]
while \(\Omega_d\) is even.  Hence every coefficient of odd total degree is
the integral of an odd function and is zero.
\end{proof}

\begin{theorem}[All-dimensional leading coefficient and covariance defect]%
\label{thm:multidim-leading-positive-defect}
Let the hypotheses of
Theorem~\ref{thm:multidimensional-poisson-entropy-jets} hold with
\(L\ge\max\{2,d+1\}\), and let
\[
\Sigma:=\EE[X_cX_c^{\mathsf T}]
\]
be the centred covariance matrix.  Put
\[
\Sigma_0:=\Sigma-\frac{\operatorname{tr}\Sigma}{d}I_d .
\]
Then the first nonzero entropy coefficient in
\[
D_{\rm KL}(P_t^{(d)}*\nu\|P_t^{(d)}(\cdot-\bar\gamma))
=\mathcal A_4^{(d)}(\nu)t^{-4}+o(t^{-4})
\]
has the explicit positive decomposition
\begin{equation}\label{eq:multidim-leading-positive-defect}
\mathcal A_4^{(d)}(\nu)
=
c_d^{\rm iso}\,(\operatorname{tr}\Sigma)^2
+c_d^{\rm tr}\,\|\Sigma_0\|_{\rm HS}^2,
\end{equation}
where
\begin{equation}\label{eq:multidim-leading-positive-constants}
c_d^{\rm iso}:=
\frac{3(d+1)(7d+9)}
{4d(d+3)(d+5)(d+7)},
\qquad
c_d^{\rm tr}:=
\frac{3(d+1)(d+3)}
{4(d+5)(d+7)} .
\end{equation}
Both constants are strictly positive.  Consequently
\(\mathcal A_4^{(d)}(\nu)\ge0\), and
\(\mathcal A_4^{(d)}(\nu)=0\) if and only if
\(\Sigma=0\).  In dimension \(d=1\) this gives
\(\mathcal A_4^{(1)}(\nu)=\mu_2^2/8\), the leading coefficient in
Theorem~\ref{thm:poisson-kl-even-orders}.
\end{theorem}

\begin{proof}
The multidimensional entropy-jet theorem identifies the \(t^{-4}\)
coefficient as
\[
\mathcal A_4^{(d)}(\nu)
=\frac12\int_{\RR^d} b_\Sigma(y)^2\,\Omega_d(\dd y),
\]
where \(b_\Sigma\) is the degree-two density coefficient.  It remains to
compute this coefficient and diagonalize the resulting quadratic form on
symmetric matrices.

Let \(a=(d+1)/2\) and \(A=1+|y|^2\).  Expanding
\[
\left(1+\frac{-2y\cdot z+|z|^2}{A}\right)^{-a}
\]
through second order gives, for a deterministic vector \(z\),
\[
F_y(z)
=1+\frac{2a\,y\cdot z}{A}
-\frac{a|z|^2}{A}
+\frac{2a(a+1)(y\cdot z)^2}{A^2}
+O(|z|^3).
\]
After centering, the linear term vanishes.  Averaging the quadratic term with
\(z=X_c/t\) gives
\begin{equation}\label{eq:multidim-leading-density-coefficient}
b_\Sigma(y)
=
a\left(
-\frac{\operatorname{tr}\Sigma}{A}
+\frac{(d+3)y^{\mathsf T}\Sigma y}{A^2}
\right).
\end{equation}
Write \(\Sigma=\tau I_d+\Sigma_0\), where
\(\tau=(\operatorname{tr}\Sigma)/d\) and
\(\operatorname{tr}\Sigma_0=0\).  Then
\[
b_\Sigma(y)
=a\tau\,\frac{3|y|^2-d}{(1+|y|^2)^2}
+a(d+3)\frac{y^{\mathsf T}\Sigma_0y}{(1+|y|^2)^2}.
\]
The mixed integral is zero by rotational symmetry, since
\[
\int_{\RR^d}
\frac{(3|y|^2-d)y^{\mathsf T}\Sigma_0y}
{(1+|y|^2)^4}\,\Omega_d(\dd y)
=K_d\,\operatorname{tr}\Sigma_0=0
\]
for a scalar \(K_d\).

It remains to compute the two radial constants.  If \(R=|Y|^2\) and
\(Y\sim\Omega_d\), then \(R\) has the beta-prime law with parameters
\((d/2,1/2)\).  Hence
\[
\EE\frac1{(1+R)^4}
=\frac{105}{(d+1)(d+3)(d+5)(d+7)},
\]
\[
\EE\frac{R}{(1+R)^4}
=\frac{15d}{(d+1)(d+3)(d+5)(d+7)},
\]
and
\[
\EE\frac{R^2}{(1+R)^4}
=\frac{3d(d+2)}{(d+1)(d+3)(d+5)(d+7)}.
\]
Therefore
\[
\EE\frac{(3R-d)^2}{(1+R)^4}
=
\frac{6d(7d+9)}
{(d+1)(d+3)(d+5)(d+7)}.
\]
The isotropic contribution to \(\frac12\int b_\Sigma^2\dd\Omega_d\) is
\[
\frac12 a^2\tau^2
\EE\frac{(3R-d)^2}{(1+R)^4}
=c_d^{\rm iso}(\operatorname{tr}\Sigma)^2 .
\]

For the traceless part, rotational symmetry on the sphere gives
\[
\EE_\Theta[(\Theta^{\mathsf T}\Sigma_0\Theta)^2]
=\frac{2}{d(d+2)}\|\Sigma_0\|_{\rm HS}^2,
\qquad |\Theta|=1,
\]
because \(\operatorname{tr}\Sigma_0=0\).  Thus
\[
\int_{\RR^d}
\frac{(y^{\mathsf T}\Sigma_0y)^2}{(1+|y|^2)^4}\,\Omega_d(\dd y)
=
\frac{6}{(d+1)(d+3)(d+5)(d+7)}
\|\Sigma_0\|_{\rm HS}^2.
\]
Multiplying by \(\frac12a^2(d+3)^2\) gives the coefficient
\(c_d^{\rm tr}\) in \eqref{eq:multidim-leading-positive-constants}.  This
proves \eqref{eq:multidim-leading-positive-defect}.

The constants are positive for every \(d\ge1\).  If the coefficient is zero,
then \(\operatorname{tr}\Sigma=0\) and \(\Sigma_0=0\), hence
\(\Sigma=0\).  Since \(\Sigma\) is positive semidefinite, this is equivalent
to \(X_c=0\) almost surely.  The one-dimensional specialization has
\(\Sigma_0=0\), \(\operatorname{tr}\Sigma=\mu_2\), and
\[
c_1^{\rm iso}
=\frac{3\cdot2\cdot16}{4\cdot1\cdot4\cdot6\cdot8}
=\frac18 .
\]
\end{proof}

\begin{theorem}[Finite-moment entropy hierarchy for Poisson entropy]%
\label{thm:poisson-kl-even-orders}
Let $N\ge2$ and let $\nu$ be a probability measure on~$\RR$ with mean
$\bar\gamma$.  Set
\[
X_c:=\gamma-\bar\gamma,
\qquad
\mu_n:=\int_{\RR}X_c^n\dd\nu(\gamma)\quad\text{whenever it exists},
\]
and define
\[
h_t:=P_t*\nu,
\qquad
g_t(x):=P_t(x-\bar\gamma).
\]
Write \(A_k^{\rm alg}(\nu)\) for the finite-moment evaluations defined in
\eqref{eq:universal-entropy-coefficient-evaluation}.

\textup{(E) Exact threshold.}
If \(\nu\) has finite centred \((2N-2)\)th absolute moment, then the
algebraic evaluations are the genuine large-time entropy coefficients through
order \(t^{-2N}\):
\begin{equation}\label{eq:poisson-kl-even-orders}
D_{\mathrm{KL}}(h_t\|g_t)
=\sum_{k=4}^{2N}A_k^{\rm alg}(\nu)t^{-k}+o(t^{-2N})
=\sum_{m=2}^{N}A_{2m}^{\rm alg}(\nu)t^{-2m}+o(t^{-2N})
\qquad(t\to\infty).
\end{equation}
This exact-threshold statement identifies the universal moment-polynomial
coefficient \(A_{2N}\) at the first moment level at which that displayed
polynomial is evaluable.  More precisely, the lower-moment obstruction in
Proposition~\ref{prop:poisson-kl-moment-threshold-sharpness} is algebraic: the
same universal polynomial coefficient is not determined from strictly lower
moment data alone.  This is not an analytic lower-bound claim about all
possible entropy asymptotics under other tail hypotheses.
After this identification we also write \(A_k^{\rm ent}(\nu)=A_k^{\rm alg}(\nu)\)
for these retained coefficients.
Equivalently, \(A_{2m-1}^{\rm ent}(\nu)=0\) for every retained odd order
\(2m-1\le 2N\).
At this exact-threshold level the theorem asserts coefficient identification
only: it combines Lemma~\ref{lem:exact-moment-density-truncation}, the finite
Taylor expansion of \(\Phi\), and the fact that \(\Phi(s)\) begins with
\(s^2/2\).  Quantitative tails and stability consequences require the
additional hypotheses in \textup{(PM)}, \textup{(MB)}, or the later certified
shell results.

\textup{(PM) Positive-margin retained order.}
Assume, in addition, that \(0<\eta\le1\) and that \(\nu\) has finite centred
\((2N-2+\eta)\)th absolute moment.  Put
\[
L:=2N-2,\qquad p:=L+\eta,
\]
and define the retained density polynomial
\[
d_t(y):=\sum_{n=2}^{L}\mu_nu_n(y)t^{-n}.
\]
Let \(C_L\) and \(B_L\) be the finite constants from
Theorem~\ref{thm:finite-moment-density-remainder}, and set
\[
C_\rho^{\rm opt}:=2C_L\EE|X_c|^p,\qquad
M_A^{\rm opt}:=\sum_{n=2}^{L}|\mu_n|\,\|u_n\|_\infty,
\]
\[
S^{\rm opt}:=B_L\sum_{n=2}^{L}|\mu_n|+C_\rho^{\rm opt},
\qquad
T^{\rm opt}:=\max\{1,(2S^{\rm opt})^{1/2}\}.
\]
Define
\[
P_N^{\rm opt}(C_\rho^{\rm opt},M_A^{\rm opt})
:=
\sum_{r=2}^{N}\frac{1}{r(r-1)}
\sum_{\ell=1}^{r}\binom r\ell
(M_A^{\rm opt})^{r-\ell}(C_\rho^{\rm opt})^\ell
\]
and
\[
Q_N^{\rm opt}(\mu,u)
:=
\sum_{r=2}^{N}\frac{1}{r(r-1)}
\sum_{m=2N+1}^{Lr}
\sum_{\substack{2\le n_1,\ldots,n_r\le L\\
n_1+\cdots+n_r=m}}
|\mu_{n_1}\cdots\mu_{n_r}|\,
\|u_{n_1}\cdots u_{n_r}\|_{L^1(\omega)} ,
\]
with the convention that an empty sum is zero.  Then, for every
\(t\ge T^{\rm opt}\),
\begin{align}
\label{eq:positive-margin-retained-order-theorem}
\left|
D_{\mathrm{KL}}(h_t\|g_t)
-\sum_{m=2}^{N}A_{2m}^{\rm ent}(\nu)t^{-2m}
\right|
&\le
P_N^{\rm opt}(C_\rho^{\rm opt},M_A^{\rm opt})\,t^{-(2N+\eta)}
\notag\\
&\quad
{}+Q_N^{\rm opt}(\mu,u)\,t^{-(2N+1)}
+2(S^{\rm opt})^{N+1}t^{-(2N+2)} .
\end{align}
Consequently the left-hand side is
\[
O_\nu(t^{-(2N+\eta)}).
\]

\textup{(MB) Moment-ball uniformity.}
Fix \(0<\eta\le1\) and \(M<\infty\).  There are finite constants
\[
C_{N,\eta,M}^{\rm dens},\qquad C_{N,\eta,M}^{\rm Tay},
\qquad T_{N,\eta,M}^{\rm ent},
\]
given by the closed finite formulas of
Lemma~\ref{lem:moment-ball-uniform-constant-domination}, depending only on
\(N,\eta,M\), such that every probability measure satisfying
\[
\int_{\RR}|\gamma-\bar\gamma|^{2N+\eta}\dd\nu(\gamma)\le M
\]
obeys, for every \(t\ge T_{N,\eta,M}^{\rm ent}\),
\begin{equation}\label{eq:uniform-moment-ball-entropy-remainder}
\left|
D_{\mathrm{KL}}(h_t\|g_t)-\sum_{m=2}^{N}A_{2m}^{\rm ent}(\nu)t^{-2m}
\right|
\le
C_{N,\eta,M}^{\rm dens}t^{-(2N+\eta)}
+C_{N,\eta,M}^{\rm Tay}t^{-(2N+2)} .
\end{equation}
Thus the exact-threshold expansion in \textup{(E)} is uniform on every
centred \((2N+\eta)\)-moment ball, with law-dependent coefficients but
constants and threshold depending only on the shell data.
\end{theorem}

\begin{remark}[Moment threshold and quantitative margins]
No positive-variance hypothesis is imposed in
Theorem~\ref{thm:poisson-kl-even-orders}; when the variance is zero the
displayed coefficients vanish and \(h_t=g_t\).
The coefficient theorem is at the exact fixed-law threshold for the universal
moment-polynomial coefficient \(A_{2N}\).  The displayed universal polynomial
\(A_{2N}\) is the coefficient identified at this threshold.  Every entropy monomial
contributing to \(A_{2N}^{\rm form}\) has at least two density factors, hence
uses centred moments only through order \(2N-2\).  With \(L=2N-2\), the finite
\(L\)th absolute moment gives an \(L^\infty\cap L^1(\omega)\) density
remainder \(o(t^{-L})\), and every entropy term containing that remainder also
contains the quadratic entropy scale.  The first discarded
density-remainder contribution is therefore \(o(t^{-(L+2)})=o(t^{-2N})\).
The \(+\eta\) moment hypotheses below are used only for quantitative
remainders and thresholds.
Thus the exact-threshold, positive-margin, and moment-ball variants are the
three labelled parts of the single consolidated hierarchy above.
\end{remark}


\begin{proof}
Put
\[
L:=2N-2 .
\]
We first fix the normalization in the rescaled variable.  For
$x=\bar\gamma+ty$,
\[
\frac{P_t(x-\gamma)}{g_t(x)}
=
\frac{t^2+(x-\bar\gamma)^2}{t^2+(x-\gamma)^2}
=
\frac{1+y^2}{1+(y-X_c/t)^2}
=R_{X_c/t}(y),
\]
where $X_c=\gamma-\bar\gamma$.  Hence the density quotient in the
Cauchy scale is the function
\[
\widetilde\delta_t(y)
:=\frac{h_t(\bar\gamma+ty)}{g_t(\bar\gamma+ty)}-1
=\int_{\RR}\bigl(R_{X_c/t}(y)-1\bigr)\dd\nu(\gamma).
\]
This is the same normalization as
\eqref{eq:cayley-comparison-kernel} and as the Laurent polynomials
\eqref{eq:mode-laurent-even}--\eqref{eq:mode-laurent-odd}; no reciprocal
kernel or extra factor of \(1+y^2\) is being introduced.

Apply Lemma~\ref{lem:exact-moment-density-truncation} with this value of
\(L\).  The constant \(n=0\) term cancels in \(R_{X_c/t}-1\), and the
\(n=1\) term integrates to zero because \(\mu_1=\int X_c\,\dd\nu=0\).
Thus the finite-mode part begins at order \(t^{-2}\).  Put
\[
d_t(y):=\sum_{n=2}^{L}\mu_n\,u_n(y)t^{-n}.
\]
The exact-moment density remainder gives
\[
\widetilde\delta_t(y)=d_t(y)+\rho_t(y),\qquad
\|\rho_t\|_{\infty}+\|\rho_t\|_{L^1(\omega)}
=o(t^{-L}),
\]
and \(\|\widetilde\delta_t\|_\infty=O(t^{-2})\).  The finite sum \(d_t\) is
also \(O(t^{-2})\) uniformly.

The change of variables is normalized by the same Cauchy law:
\[
g_t(\bar\gamma+ty)\,\dd(\bar\gamma+ty)
=P_t(ty)\,t\,\dd y
=\frac{\dd y}{\pi(1+y^2)}
=\omega(\dd y).
\]
Since \(\int g_t(\widetilde\delta_t\circ((\,\cdot-\bar\gamma)/t))\,\dd x=0\),
the KL entropy is
\[
D_{\mathrm{KL}}(h_t\|g_t)
=\int_{\RR}\Phi(\widetilde\delta_t(y))\,\omega(\dd y),
\qquad
\Phi(s)=(1+s)\log(1+s)-s.
\]

Since \(\|\widetilde\delta_t\|_\infty=O(t^{-2})\), choose
\(T_\delta<\infty\) so that
\(\|\widetilde\delta_t\|_\infty\le1/2\) for \(t\ge T_\delta\).  The
hypotheses of Lemma~\ref{lem:exact-moment-entropy-transfer} are therefore
satisfied with \(\delta_t=\widetilde\delta_t\) and \(L=2N-2\).  It gives
\[
D_{\mathrm{KL}}(h_t\|g_t)
=\sum_{m=4}^{2N}A_m(\nu)t^{-m}+o(t^{-2N}).
\]
This is precisely the finite extraction described in the statement: after
Lemma~\ref{lem:exact-moment-density-truncation} supplies
\(\widetilde\delta_t=d_t+\rho_t\), the missing constant and linear density
terms and the quadratic start of \(\Phi\) force every retained entropy monomial
to use only modes through \(u_{2N-2}\).
Here the transfer lemma also explains why no omitted density mode can
contribute to a retained entropy coefficient: an entropy monomial of total
degree \(m\le2N\) has at least two mode indices, all at least \(2\), hence each
individual index is at most \(m-2\le2N-2\).
The parity part of Theorem~\ref{thm:universal-entropy-coefficients} gives
$A_m(\nu)=0$ for odd~$m$, so only even powers remain.  Reindexing $m=2r$
gives
\[
D_{\mathrm{KL}}(h_t\|g_t)
=\sum_{r=2}^{N} A_{2r}(\nu)\,t^{-2r}+o(t^{-2N}),
\]
which proves \eqref{eq:poisson-kl-even-orders}.

We next prove \textup{(PM)}.  Under the additional finite
\((2N-2+\eta)\)th centred absolute moment, apply
Theorem~\ref{thm:finite-moment-density-remainder} with truncation
\(L=2N-2\).  In the same Cauchy scale this gives
\[
\widetilde\delta_t(y)=d_t(y)+\rho_t(y),\qquad
\|\rho_t\|_\infty+\|\rho_t\|_{L^1(\omega)}
\le C_\rho^{\rm opt}t^{-(L+\eta)},
\]
and \(\|\widetilde\delta_t\|_\infty\le S^{\rm opt}t^{-2}\).  For
\(t\ge T^{\rm opt}\), \(\|\widetilde\delta_t\|_\infty\le1/2\), and the Taylor
expansion of \(\Phi\) through degree \(N\) has tail bounded by
\(2(S^{\rm opt})^{N+1}t^{-(2N+2)}\).  For each \(2\le r\le N\),
Lemma~\ref{lem:density-remainder-product-hierarchy} puts every monomial with
at least one factor \(\rho_t\) at order
\[
(L+\eta)+2=2N+\eta,
\]
and summing the finite Taylor coefficients gives the
\(P_N^{\rm opt}(C_\rho^{\rm opt},M_A^{\rm opt})t^{-(2N+\eta)}\) term.  The
pure \(d_t\)-polynomial has retained total degrees \(m\le2N\) and a finite
tail \(m\ge2N+1\), bounded by
\(Q_N^{\rm opt}(\mu,u)t^{-(2N+1)}\).  No retained coefficient is lost by
truncating the density modes at \(L=2N-2\), because an entropy monomial of
total degree \(m\le2N\) has at least two factors and all indices are at
least \(2\).  Hence each retained index is at most \(2N-2\).  Parity removes
the odd retained coefficients and proves
\eqref{eq:positive-margin-retained-order-theorem}.

Finally assume the moment-ball hypothesis in \textup{(MB)}.  Apply
Theorem~\ref{thm:finite-moment-density-remainder} with truncation \(L=2N\),
then Proposition~\ref{prop:density-to-entropy-descent}.  This gives the
fixed-law finite estimate with the product, finite-tail, and Taylor constants
appearing in \eqref{eq:density-to-entropy-descent-remainder}.  The closed
domination lemma,
Lemma~\ref{lem:moment-ball-uniform-constant-domination}, bounds all those
law-dependent constants and the smallness threshold by finite formulas in
\(N,\eta,M\).  Substituting these dominating constants gives
\eqref{eq:uniform-moment-ball-entropy-remainder}.
\end{proof}

\subsubsection{Constants and variants}%
\label{subsubsec:finite-moment-constants-variants}

Theorem~\ref{thm:poisson-kl-even-orders} is the finite-moment hierarchy.  The
material in this subsection only fixes shorthand constants and common
remainder forms used later.  The coefficient-threshold proposition below is
kept numbered because it records the algebraic lower-moment obstruction for
the displayed universal polynomial coefficient.

\paragraph{Retained-polynomial form.}%
\phantomsection\label{thm:finite-moment-entropy-remainder}
Let \(N\ge2\), and let \(\nu\) be a probability measure on \(\RR\) with mean
\(\bar\gamma\) and finite centred \((2N-2)\)th absolute moment.  Put
\[
\mathcal R_N(t;\nu)
:=D_{\mathrm{KL}}(P_t*\nu\|P_t(\cdot-\bar\gamma))
-\sum_{m=2}^{N}A_{2m}(\nu)t^{-2m}.
\]
Part \textup{(E)} of Theorem~\ref{thm:poisson-kl-even-orders} gives
\[
\mathcal R_N(t;\nu)=o(t^{-2N})
\qquad(t\to\infty).
\]
After the universal even coefficient polynomial is subtracted, the exact
centred \((2N-2)\)th absolute moment hypothesis leaves no further contribution
at any order \(t^{-k}\) with \(k\le2N\).  The positive-margin quantitative
estimate is part \textup{(PM)} of Theorem~\ref{thm:poisson-kl-even-orders};
the moment-ball uniform estimate is part \textup{(MB)} of the same theorem.

\paragraph{Positive-margin constants.}%
\phantomsection\label{thm:positive-margin-retained-order-entropy}
Let \(N\ge2\), let \(0<\eta\le1\), and let \(\nu\) be a probability
measure on \(\RR\) with mean \(\bar\gamma\) and finite centred
\((2N-2+\eta)\)th absolute moment.  Put
\[
L:=2N-2,\qquad p:=L+\eta,\qquad
X_c:=\gamma-\bar\gamma.
\]
Let \(C_L\) and \(B_L\) be the finite constants from
Theorem~\ref{thm:finite-moment-density-remainder}, and set
\[
C_\rho:=2C_L\EE|X_c|^p,\qquad
M_A:=\sum_{n=2}^{L}|\mu_n|\,\|u_n\|_\infty,\qquad
S:=B_L\sum_{n=2}^{L}|\mu_n|+C_\rho .
\]
Define the density-remainder product constant
\[
C_{\nu,N,\eta}:=
\sum_{r=2}^{N}\frac{1}{r(r-1)}
\sum_{\ell=1}^{r}\binom r\ell
M_A^{r-\ell}C_\rho^\ell ,
\]
the finite retained-mode tail
\[
Q_{\nu,N}:=
\sum_{r=2}^{N}\frac{1}{r(r-1)}
\sum_{m=2N+1}^{Lr}
\sum_{\substack{2\le n_1,\ldots,n_r\le L\\
n_1+\cdots+n_r=m}}
|\mu_{n_1}\cdots\mu_{n_r}|\,
\|u_{n_1}\cdots u_{n_r}\|_{L^1(\omega)} ,
\]
with the convention that an empty sum is zero, and
\[
C'_{\nu,N}:=2S^{N+1},\qquad T_{\nu,N,\eta}:=\max\{1,(2S)^{1/2}\}.
\]
With these names, the positive-margin retained-order estimate is exactly
part \textup{(PM)} of Theorem~\ref{thm:poisson-kl-even-orders}.  After
removing the superscript ``opt'' from the corresponding quantities, the
identifications are
\[
C_{\nu,N,\eta}=P_N^{\rm opt}(C_\rho^{\rm opt},M_A^{\rm opt}),\qquad
Q_{\nu,N}=Q_N^{\rm opt}(\mu,u),
\]
\[
C'_{\nu,N}=2(S^{\rm opt})^{N+1},\qquad
T_{\nu,N,\eta}=T^{\rm opt}.
\]
The retained density modes are \(u_2,\ldots,u_{2N-2}\), the same modes used
in the exact-threshold coefficient identity.

\begin{proposition}[Algebraic threshold for the universal moment-polynomial coefficient]%
\label{prop:poisson-kl-moment-threshold-sharpness}
Fix \(N\ge2\) and put \(L=2N-2\).  The obstruction below is only a
coefficient-identification obstruction for the universal moment polynomial
\(A_{2N}^{\rm alg}\) in Theorem~\ref{thm:poisson-kl-even-orders}.  It says
that this specific polynomial coefficient cannot be determined from moments
below order \(L\).  It is not a sharp analytic lower bound for every possible
entropy asymptotic: a different renormalized, tail-dependent, or
nonuniversal coefficient under weaker hypotheses is not excluded here.

For \(N=2\), the leading coefficient is
\[
A_4(\nu)=\frac18\mu_2^2,
\]
so a universal statement through order \(t^{-4}\) must at least make the
variance finite.  For \(N\ge3\), the degree-\(2N\) formal coefficient has the
nonzero top-moment term
\begin{equation}\label{eq:top-moment-coefficient-sharpness}
A_{2N}^{\rm form}
=\kappa_N\,\mathfrak m_2\mathfrak m_{2N-2}
 +P_N(\mathfrak m_2,\ldots,\mathfrak m_{2N-3}),
\qquad
\kappa_N=(-1)^N(N-1)2^{-2N+2}\ne0,
\end{equation}
where \(P_N\) involves no \(\mathfrak m_{2N-2}\).  Hence this universal
moment-polynomial coefficient through order \(t^{-2N}\) is not determined by
moments below order \(2N-2\).  The assertion is only at the
coefficient-polynomial level.

Moreover, for every \(q<L\) there is a centred symmetric probability law
with finite \(q\)th absolute moment and infinite \(L\)th absolute moment.
For such a law the top universal moment-polynomial coefficient in
\eqref{eq:top-moment-coefficient-sharpness} is not a finite number.  Thus no
fixed-law theorem that keeps the same universal polynomial coefficient
\(A_{2N}^{\rm alg}\) can identify that coefficient from strictly lower
absolute moments alone.
\end{proposition}

\begin{proof}
The case \(N=2\) is the first line of the coefficient computation:
\[
A_4(\nu)=\frac12\mu_2^2 I(2,2)
=\frac12\mu_2^2\cdot\frac14
=\frac18\mu_2^2 .
\]
Assume \(N\ge3\).  In the formal coefficient
\eqref{eq:universal-entropy-coefficient}, a monomial containing
\(\mathfrak m_L\) and contributing to total degree \(L+2=2N\) must have
exactly two density factors, with indices \(2\) and \(L\).  All other
entropy monomials either have total degree different from \(2N\) or involve
only moments of order at most \(L-1\).  Since the quadratic Taylor
coefficient is \(1/2\) and the ordered pairs \((2,L)\) and \((L,2)\) both
occur, the coefficient of \(\mathfrak m_2\mathfrak m_L\) is \(I(2,L)\).

Write \(L=2m\), so \(m=N-1\).  From
\eqref{eq:mode-laurent-even},
\[
Q_{2m}(z)=(-1)^m2^{-2m}
\sum_{j=0}^{2m-1}\binom{2m-1}{j}
\bigl(z^{j+1}+z^{-(j+1)}\bigr),
\]
while \(Q_2(z)=-(z+z^{-1}+z^2+z^{-2})/4\).  Therefore only the
exponents \(\pm1\) and \(\pm2\) of \(Q_{2m}\) enter \(I(2,2m)\), and
\[
\begin{aligned}
I(2,2m)
&=[z^0]Q_2Q_{2m}  \\
&=-\frac12\,(-1)^m2^{-2m}
\left\{\binom{2m-1}{0}+\binom{2m-1}{1}\right\}  \\
&=(-1)^{m+1}m\,2^{-2m}
=(-1)^N(N-1)2^{-2N+2}.
\end{aligned}
\]
This proves \eqref{eq:top-moment-coefficient-sharpness}.

For the counterexample assertion, choose a symmetric probability density on
\(|x|\ge e\) proportional to
\[
\frac{1}{|x|^{L+1}\log |x|}
\]
and put any remaining mass symmetrically on a bounded interval so that the
total mass is one.  The law is centred by symmetry.  For every \(q<L\),
\(\int |x|^q\,\dd\nu(x)<\infty\), because the tail integral is dominated by
\(\int_e^\infty x^{q-L-1}\dd x\).  At order \(L\), however,
\[
\int |x|^L\,\dd\nu(x)
\asymp \int_e^\infty \frac{\dd x}{x\log x}
=\infty .
\]
Thus the moment required to evaluate the nonzero top term of the displayed
universal polynomial \(A_{2N}^{\rm alg}\) is genuinely absent although every
lower absolute moment is finite.
\end{proof}

\paragraph{Stronger \(2N+\eta\) truncation.}%
\phantomsection\label{cor:finite-moment-entropy-constant-remainder}
Let \(N\ge2\), let \(0<\eta\le1\), and let \(\nu\) be a probability measure
on \(\RR\) with mean \(\bar\gamma\) and finite centred
\((2N+\eta)\)th absolute moment.  This stronger hypothesis is not needed for
the coefficient identity in Theorem~\ref{thm:poisson-kl-even-orders}; it is a
convenient way to spell out a fixed-law constant obtained by retaining density
modes through \(2N\) instead of through \(2N-2\).  Put
\[
p:=2N+\eta,\qquad X_c:=\gamma-\bar\gamma .
\]
Let
\[
C_\rho:=2C_{2N}\EE|X_c|^p,
\qquad
M_A:=\sum_{n=2}^{2N}|\mu_n|\,\|u_n\|_\infty,
\]
where \(C_{2N}\) is the finite density constant from
Theorem~\ref{thm:finite-moment-density-remainder}.  Define
\[
P_N(C_\rho,M_A)
:=
\sum_{r=2}^{N}\frac1{r(r-1)}
\sum_{\ell=1}^{r}\binom r\ell
M_A^{r-\ell}C_\rho^\ell ,
\]
and
\[
Q_N(\mu,u)
:=
\sum_{r=2}^{N}\frac1{r(r-1)}
\sum_{m=2N+1}^{2Nr}
\sum_{\substack{2\le n_1,\ldots,n_r\le2N\\
n_1+\cdots+n_r=m}}
|\mu_{n_1}\cdots\mu_{n_r}|\,
\|u_{n_1}\cdots u_{n_r}\|_{L^1(\omega)} .
\]
Finally set
\[
S:=B_{2N}\sum_{n=2}^{2N}|\mu_n|+C_\rho,
\qquad
T:=\max\{1,(2S)^{1/2}\},
\]
where \(B_{2N}\) is the finite mode-size constant from
Theorem~\ref{thm:finite-moment-density-remainder}.  The same
density-to-entropy calculation used in part \textup{(MB)} of
Theorem~\ref{thm:poisson-kl-even-orders} gives, for every \(t\ge T\),
\begin{align}
\label{eq:finite-moment-entropy-remainder-estimate}
\left|
D_{\mathrm{KL}}(P_t*\nu\|P_t(\cdot-\bar\gamma))
-\sum_{m=2}^{N}A_{2m}(\nu)t^{-2m}
\right|
&\le
\{P_N(C_\rho,M_A)+Q_N(\mu,u)\}t^{-(2N+\eta)}
\notag\\
&\qquad
+2S^{N+1}t^{-(2N+2)} .
\end{align}
The refined three-term estimate behind this convenience envelope is
\[
P_N(C_\rho,M_A)t^{-(2N+\eta+2)}
+Q_N(\mu,u)t^{-(2N+1)}
+2S^{N+1}t^{-(2N+2)} .
\]
For a centred \(p\)-moment ball \(\EE|X_c|^p\le M\), the same estimate is
made uniform by replacing the law-dependent quantities with the closed finite
dominating constants of
Lemma~\ref{lem:moment-ball-uniform-constant-domination}.  In particular
\(\|u_n\|_\infty\le n+1\) and
\(\|u_{n_1}\cdots u_{n_r}\|_{L^1(\omega)}
\le\prod_j(n_j+1)\) give a version involving no unspecified mode norms.

\paragraph{Optimized retained-order notation and the \(N=4\) envelope.}%
\phantomsection\label{cor:optimized-retained-order-entropy-bridge}
Let \(N\ge2\), let \(0<\eta\le1\), and let \(\nu\) be a probability
measure on \(\RR\) with mean \(\bar\gamma\) and finite centred
\((2N-2+\eta)\)th absolute moment.  Put
\[
p:=2N-2+\eta,\qquad L:=2N-2,\qquad
X_c:=\gamma-\bar\gamma .
\]
Let
\[
C_\rho^{\rm opt}:=2C_L\EE|X_c|^p,
\qquad
M_A^{\rm opt}:=\sum_{n=2}^{L}|\mu_n|\,\|u_n\|_\infty,
\]
where \(C_L\) is the finite density constant from
Theorem~\ref{thm:finite-moment-density-remainder}.  Define
\[
P_N^{\rm opt}(C_\rho^{\rm opt},M_A^{\rm opt})
:=
\sum_{r=2}^{N}\frac{1}{r(r-1)}
\sum_{\ell=1}^{r}\binom r\ell
(M_A^{\rm opt})^{r-\ell}(C_\rho^{\rm opt})^\ell
\]
and
\[
Q_N^{\rm opt}(\mu,u)
:=
\sum_{r=2}^{N}\frac{1}{r(r-1)}
\sum_{m=2N+1}^{Lr}
\sum_{\substack{2\le n_1,\ldots,n_r\le L\\
n_1+\cdots+n_r=m}}
|\mu_{n_1}\cdots\mu_{n_r}|\,
\|u_{n_1}\cdots u_{n_r}\|_{L^1(\omega)} ,
\]
with the convention that an empty sum is zero.  Finally set
\[
S^{\rm opt}:=B_L\sum_{n=2}^{L}|\mu_n|+C_\rho^{\rm opt},
\qquad
T^{\rm opt}:=\max\{1,(2S^{\rm opt})^{1/2}\},
\]
where \(B_L\) is the finite mode-size constant in
Theorem~\ref{thm:finite-moment-density-remainder}.  These are exactly the
quantities used in part \textup{(PM)} of
Theorem~\ref{thm:poisson-kl-even-orders}; equivalently,
\[
C_{\nu,N,\eta}=P_N^{\rm opt},\qquad
Q_{\nu,N}=Q_N^{\rm opt},\qquad
C'_{\nu,N}=2(S^{\rm opt})^{N+1}.
\]
For reference we record the coarse envelope obtained from
\eqref{eq:positive-margin-retained-order-theorem}: for \(t\ge T^{\rm opt}\),
\begin{equation}\label{eq:optimized-retained-order-entropy-envelope}
\left|
D_{\mathrm{KL}}(P_t*\nu\|P_t(\cdot-\bar\gamma))
-\sum_{m=2}^{N}A_{2m}(\nu)t^{-2m}
\right|
\le
\bigl\{
P_N^{\rm opt}+Q_N^{\rm opt}+2(S^{\rm opt})^{N+1}
\bigr\}t^{-(2N+\eta)}
\end{equation}
  for \(t\ge T^{\rm opt}\).

  In the specialization \(N=4\), \(\eta=1\), assume
  \(\EE|X_c|^7<\infty\), set \(L=6\), and write
  \[
  M_7(\nu):=\EE|X_c|^7,\qquad
  \mathcal M_{\nu,8}:=\sum_{n=2}^{6}(n+1)|\mu_n|,
  \]
  \[
  \mathcal C_{\nu,8}:=110M_7(\nu),\qquad
  \mathcal S_{\nu,8}:=7\sum_{n=2}^{6}|\mu_n|+110M_7(\nu).
  \]
  The closed N=4 density-remainder product and retained-mode tail constants
  are
  \[
  \mathcal P_{\nu,8}:=
  \sum_{r=2}^{4}\frac{1}{r(r-1)}
  \sum_{\ell=1}^{r}\binom r\ell
  \mathcal M_{\nu,8}^{\,r-\ell}\mathcal C_{\nu,8}^{\,\ell}
  \]
  and
  \[
  \mathcal Q_{\nu,8}:=
  \sum_{r=2}^{4}\frac{1}{r(r-1)}
  \sum_{m=9}^{6r}
  \sum_{\substack{2\le n_1,\ldots,n_r\le6\\
  n_1+\cdots+n_r=m}}
  \prod_{j=1}^{r}\bigl((n_j+1)|\mu_{n_j}|\bigr),
  \]
  with empty sums interpreted as zero.  Set
  \[
  B_{\nu,8}:=\mathcal P_{\nu,8}+\mathcal Q_{\nu,8}
  +2\mathcal S_{\nu,8}^{5},
  \qquad
  T_{\nu,8}:=\max\{1,(2\mathcal S_{\nu,8})^{1/2}\}.
  \]
  These are closed bounds in terms of the centred moments through order
  \(6\) and the seventh absolute centred moment.  They are obtained from the
  preceding optimized quantities by applying
  Lemma~\ref{lem:closed-cayley-mode-constant-envelopes}: for \(L=6\),
  \(B_6\le7\), \(C_6\le55\), \(C_{\nu,6,1}\le110M_7(\nu)\), and
  \(\|u_n\|_\infty\le n+1\).
  Then, for every \(t\ge T_{\nu,8}\),
  \begin{equation}\label{eq:optimized-n4-seventh-envelope}
  \left|
  D_{\mathrm{KL}}(P_t*\nu\|P_t(\cdot-\bar\gamma))
  -A_4(\nu)t^{-4}
  -A_6(\nu)t^{-6}
  -A_8(\nu)t^{-8}
  \right|
  \le B_{\nu,8}t^{-9}.
  \end{equation}
  Thus the order-nine constant used below is dominated by the displayed
  closed envelope and can be checked directly from the law's moment data
  through order seven.
  The fixed-law coefficient identity is the \(N=4\) specialization of
  Theorem~\ref{thm:poisson-kl-even-orders}; after the explicit coefficient computation in
  Theorem~\ref{thm:poisson-kl-eighth}, the displayed
  \(A_8(\nu)\) is the actual \(t^{-8}\) entropy coefficient.  The displayed rate is the
  \(N=4,\eta=1\) specialization of the positive-margin envelope
  above.

\paragraph{Moment-ball constants.}%
\phantomsection\label{cor:uniform-moment-ball-entropy-remainder}
Fix $N\ge2$, $0<\eta\le1$, and $M<\infty$.  There are finite constants

C_{N,\eta,M}^{\rm dens},\qquad C_{N,\eta,M}^{\rm Tay},
\qquad T_{N,\eta,M}^{\rm ent}
\]
given by the closed finite formulas of

\begin{theorem}[Analytic compact-support interchange]%
\label{thm:analytic-compact-support-interchange}
Let $\nu$ be a probability measure with mean $\bar\gamma$ and compactly
supported centred variable $X_c=\gamma-\bar\gamma$.  Then there is
$T_\nu\ge1$ such that, for $t\ge T_\nu$, the series
\begin{equation}\label{eq:analytic-density-series}
\delta_t(\bar\gamma+ty)
=\sum_{n=2}^{\infty}\mu_n\,u_n(y)t^{-n}
\end{equation}
converges absolutely and uniformly in $y\in\RR$.  The entropy Taylor series
\begin{equation}\label{eq:analytic-entropy-series}
D_{\mathrm{KL}}(P_t*\nu\|P_t(\cdot-\bar\gamma))
=\sum_{k=4}^{\infty}A_k(\nu)t^{-k}
\end{equation}
also converges absolutely for $t\ge T_\nu$, where $A_k(\nu)$ is the
finite-moment evaluation of $A_k^{\rm form}$ from
\eqref{eq:universal-entropy-coefficient-evaluation}.  In particular the
all-order coefficients are the genuine analytic coefficients under this
strong compact-support hypothesis, while part \textup{(E)} of
Theorem~\ref{thm:poisson-kl-even-orders} and the retained-polynomial form in
\S\ref{subsubsec:finite-moment-constants-variants} give the separate
finite-moment asymptotic justification.
\end{theorem}

\begin{proof}
Let $|X_c|\le M$ almost surely.  The generating function for the
Chebyshev polynomials gives
\[
R_\varepsilon(\tan\theta)
=\sum_{n=0}^{\infty}\cos^n\theta\,U_n(\sin\theta)\varepsilon^n.
\]
Since $|U_n(s)|\le n+1$ for $|s|\le1$ and $|\cos\theta|\le1$, the series is
dominated uniformly by $\sum_{n\ge0}(n+1)|\varepsilon|^n$ whenever
$|\varepsilon|<1$.  Choose $T_\nu\ge2M$.  Then
$|X_c|/t\le1/2$, so dominated convergence permits termwise integration
against~$\nu$ and gives the absolutely and uniformly convergent density
series.  The $n=0$ term is one and the $n=1$ term is zero because
$\EE X_c=0$, which proves \eqref{eq:analytic-density-series}.

The uniform convergence gives $\delta_t=O(t^{-2})$ uniformly.  Enlarging
$T_\nu$ if needed, $|\delta_t|\le1/2$, and the Taylor series for
$\Phi(s)=(1+s)\log(1+s)-s$ converges absolutely and uniformly after the
substitution $s=\delta_t(\bar\gamma+ty)$.  To make the interchange explicit,
write
\[
b_n(t):=|\mu_n|\,\|u_n\|_\infty t^{-n}
\qquad(n\ge2),
\qquad
B(t):=\sum_{n\ge2}b_n(t).
\]
The compact support bound and $\|u_n\|_\infty\le n+1$ give
$B(t)\le\sum_{n\ge2}(n+1)(M/t)^n<\infty$, and after increasing $T_\nu$ we
may assume $B(t)\le1/2$.  Hence every product occurring in the entropy Taylor
series has the summable absolute majorant
\[
\sum_{r=2}^{\infty}\frac1{r(r-1)}
\sum_{n_1,\dots,n_r\ge2}
b_{n_1}(t)\cdots b_{n_r}(t)
=\sum_{r=2}^{\infty}\frac{B(t)^r}{r(r-1)}<\infty.
\]
Since $\omega$ has total mass one and
\[
\int\bigl|u_{n_1}(y)\cdots u_{n_r}(y)\bigr|\,\omega(\dd y)
\le \prod_{j=1}^r\|u_{n_j}\|_\infty,
\]
this single majorant justifies both Fubini/Tonelli for
$\Phi(\delta_t)=\sum_{r\ge2}(-1)^r\delta_t^r/(r(r-1))$ and termwise
multiplication of the absolutely convergent density series.  For a fixed
power $t^{-k}$, only finitely many tuples $r\ge2$, $n_j\ge2$, and
$n_1+\cdots+n_r=k$ occur.  The degree-$k$ coefficient obtained by multiplying
the density series is therefore the finite-moment evaluation of the formal
expression in
\eqref{eq:universal-entropy-coefficient}; hence
\eqref{eq:analytic-entropy-series} follows.
\end{proof}
