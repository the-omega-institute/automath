\noindent\emph{Sharpness of the uniform quotient clause.}
The threshold \(L\ge d+1\) is sharp for convergence in the
\(L^\infty\) quotient topology.  Indeed, fix an integer
\(1\le L<d+1\), let \(e_1\in\RR^d\), and choose \(R_n\uparrow\infty\) so
fast that
\[
  W:=\sum_{n\ge1}R_n^{-L}n^{-2}<1,
  \qquad
  t_n:=8^{-1/(d+1)}
  R_n^{(d+1-L)/(d+1)}n^{-2/(d+1)}\to\infty,
\]
and set \(w_n=R_n^{-L}n^{-2}\).  The symmetric probability law
\[
  \nu=(1-W)\delta_0
  +\sum_{n\ge1}\frac{w_n}{2}
  \bigl(\delta_{R_ne_1}+\delta_{-R_ne_1}\bigr)
\]
has mean zero and finite \(L\)th absolute moment because
\(\sum_n w_nR_n^L=\sum_n n^{-2}<\infty\).  At the space-time point
\(x_n=R_ne_1\), with \(g_t=P_t^{(d)}\) and \(h_t=P_t^{(d)}*\nu\),
\[
\frac{h_{t_n}(x_n)}{g_{t_n}(x_n)}
\ge
\frac{w_n}{2}\,
\frac{P_{t_n}^{(d)}(0)}{P_{t_n}^{(d)}(R_ne_1)}
\ge
\frac{w_n}{2}\left(\frac{R_n}{t_n}\right)^{d+1}
=4 .
\]
Hence \(\|\delta_{t_n}^{(d)}\|_\infty\ge3\) along \(t_n\to\infty\), so
the quotient does not even converge uniformly to one.  Thus \(d+1\) is the
sharp moment threshold for the uniform quotient route used above.  This
counterexample does not show that \(d+1\) moments are necessary for the
leading KL coefficient under a weaker \(L^2(\Omega_d)\) or entropy topology.
Those topologies are treated separately in
Section~\ref{sec:verified-A2-results}: Theorem
\ref{thm:multidim-l2-moment-threshold} gives the sharp \(L^2\) exponent,
while Theorem~\ref{thm:high-dimensional-kl-moment-threshold} gives the
optimal uniform sufficient KL exponent in the finite absolute-moment scale,
not a law-by-law moment criterion.

\begin{corollary}[Actual leading multidimensional KL coefficient]%
\label{cor:finite-dplusone-moment-leading-multidim-coefficient}
Let \(d\ge1\), let \(\nu\) be a probability measure on \(\RR^d\) with mean
\(\bar{\gamma}\), and assume
\[
  \EE |X_c|^{d+1}<\infty,\qquad X_c=\gamma-\bar{\gamma} .
\]
Put
\[
  h_t=P_t^{(d)}*\nu,\qquad
  g_t=P_t^{(d)}(\cdot-\bar{\gamma}),\qquad
  \Sigma=\EE[X_cX_c^{\mathsf T}],\qquad
  \Sigma_0=\Sigma-\frac{\operatorname{tr}\Sigma}{d}I_d .
\]
Then Theorem~\ref{thm:multidimensional-density-quotient} supplies the
second-order quotient criterion
\eqref{eq:second-order-quotient-criterion}.  Hence
\[
D_{\rm KL}\!\left(P_t^{(d)}*\nu\,\middle\|\,
P_t^{(d)}(\cdot-\bar{\gamma})\right)
=\mathcal Q_d(\Sigma)t^{-4}+o(t^{-4}),
\]
where
\[
\mathcal Q_d(\Sigma)
=c_d^{\rm iso}(\operatorname{tr}\Sigma)^2
+c_d^{\rm tr}\|\Sigma_0\|_{\rm HS}^2
\]
and
\[
c_d^{\rm iso}=
\frac{3(d+1)(7d+9)}
{4d(d+3)(d+5)(d+7)},
\qquad
c_d^{\rm tr}=
\frac{3(d+1)(d+3)}
{4(d+5)(d+7)} .
\]
\end{corollary}

\begin{proof}
Apply Theorem~\ref{thm:multidimensional-density-quotient} with
\(L=d+1\).  Thus \(\EE|X_c|^{d+1}<\infty\) implies the second-order quotient criterion
\eqref{eq:second-order-quotient-criterion}, including
\(\|\delta_t^{(d)}\|_\infty=O(t^{-2})\).  Theorem~\ref{thm:second-order-quotient-kl-coefficient}
then gives the KL leading term, and
Theorem~\ref{thm:multidim-covariance-algebra} identifies its
coefficient as \(\mathcal Q_d(\Sigma)\).
\end{proof}

\begin{corollary}[Internal bridge: leading coefficient under \((\mathrm{MPJ}_{d,L})+(\mathrm{EX}_{d,L})\)]%
\label{cor:multidim-leading-kl-coefficient}
Let \(d\ge1\), let \(L\in\mathbb N\) satisfy \(L\ge2\), and assume that
\((\nu,d,L)\) satisfies both \((\mathrm{MPJ}_{d,L})\) and
\((\mathrm{EX}_{d,L})\).  Let
\[
h_t=P_t^{(d)}*\nu,\qquad
g_t=P_t^{(d)}(\cdot-\bar{\gamma}).
\]
Since \(L\ge2\), the covariance
\[
\Sigma=\EE[X_cX_c^{\mathsf T}],\qquad X_c=\gamma-\bar{\gamma},
\]
exists.  Then
\[
D_{\rm KL}(h_t\|g_t)
=\mathcal Q_d(\Sigma)t^{-4}+o(t^{-4}).
\]
\end{corollary}

\begin{proof}
The hypotheses \((\mathrm{MPJ}_{d,L})+(\mathrm{EX}_{d,L})\) are exactly the
analytic input of Proposition~\ref{thm:multidimensional-poisson-entropy-jets}.
Applying that theorem gives the KL jet, whose \(m=4\) coefficient has only the quadratic
term
\[
\mathcal A_4^{(d)}(\nu)
=\frac12\int_{\RR^d}D_2^{(d)}(y;\nu)^2\,\Omega_d(\dd y).
\]
For centred data the degree-two density layer is
\[
D_2^{(d)}(y;\nu)=b_\Sigma(y),
\]
where \(b_\Sigma\) is the bounded covariance mode in
\eqref{eq:multidim-leading-theorem-density-mode}.  Theorem~\ref{thm:multidim-covariance-algebra}
computes
\[
\frac12\int_{\RR^d}b_\Sigma(y)^2\,\Omega_d(\dd y)=\mathcal Q_d(\Sigma).
\]
Thus \(D_{\rm KL}(h_t\|g_t)=\mathcal Q_d(\Sigma)t^{-4}+o(t^{-4})\).
\end{proof}

\begin{corollary}[Finite-moment covariance coefficient for larger \(L\)]%
\label{cor:multidim-finite-moment-covariance-coefficient}
Let \(d\ge1\), let \(L\in\mathbb N\) satisfy \(L\ge d+1\), and let \(\nu\) be
a probability measure on \(\RR^d\) with mean \(\bar{\gamma}\) and finite
centred \(L\)th absolute moment.  Put
\[
h_t=P_t^{(d)}*\nu,\qquad
g_t=P_t^{(d)}(\cdot-\bar{\gamma}),
\]
and let
\[
X_c=\gamma-\bar{\gamma},\qquad
\Sigma=\EE[X_cX_c^{\mathsf T}],\qquad
\Sigma_0=\Sigma-\frac{\operatorname{tr}\Sigma}{d}I_d .
\]
Then
\begin{equation}\label{eq:finite-moment-covariance-coefficient-through-mpj}
D_{\rm KL}(h_t\|g_t)=\mathcal Q_d(\Sigma)t^{-4}+o(t^{-4}),
\end{equation}
where \(\mathcal Q_d(\Sigma)\) is the quadratic covariance contraction computed
in Theorem~\ref{thm:multidim-covariance-algebra}:
\[
\mathcal Q_d(\Sigma)
=c_d^{\rm iso}(\operatorname{tr}\Sigma)^2
+c_d^{\rm tr}\|\Sigma_0\|_{\rm HS}^2 .
\]
\end{corollary}

\begin{proof}
Theorem~\ref{thm:multidimensional-density-quotient} turns the finite centred
\(L\)-absolute moment assumption into \((\mathrm{MPJ}_{d,L})\), and
Proposition~\ref{thm:multidimensional-poisson-entropy-jets} then gives the KL
coefficient jet.  The \(m=4\) coefficient in that jet is
\[
\mathcal A_4^{(d)}(\nu)
=\frac12\int_{\RR^d}D_2^{(d)}(y;\nu)^2\,\Omega_d(\dd y).
\]
The degree-two density layer depends only on the covariance:
\[
D_2^{(d)}(y;\nu)=
\frac{d+1}{2}\left(
-\frac{\operatorname{tr}\Sigma}{1+|y|^2}
+\frac{(d+3)y^{\mathsf T}\Sigma y}{(1+|y|^2)^2}
\right).
\]
Theorem~\ref{thm:multidim-covariance-algebra} evaluates the last
quadratic integral as \(\mathcal Q_d(\Sigma)\), proving the asserted leading
term.
\end{proof}
