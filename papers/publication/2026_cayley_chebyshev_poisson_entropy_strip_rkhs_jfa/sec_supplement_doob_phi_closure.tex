\section{General Doob--Bregman domain closure}\label{supp:doob-bregman-domain-closure}

This supplementary section records the range-local convex-\(\Phi\) closure for
the Poisson Doob quotient.  The main paper uses only the KL specialization in
Appendix~\ref{app:doob-bregman-domain}.

\begin{lemma}[Two-point Bregman algebra for the Doob derivative]%
\label{lem:doob-bregman-algebra}
Let $g_x,g_y>0$ and $u_x,u_y>0$.  Let
$\Phi\in C^1((0,\infty))$, set
\[
G(u):=\Phi(u)-u\Phi'(u),
\qquad
D_\Phi(a,b):=\Phi(a)-\Phi(b)-\Phi'(b)(a-b).
\]
Then
\begin{equation}\label{eq:doob-bregman-algebra}
\begin{split}
&(g_xu_x-g_yu_y)\bigl(\Phi'(u_x)-\Phi'(u_y)\bigr)
 +(g_x-g_y)\bigl(G(u_x)-G(u_y)\bigr)
\\
&\qquad
=g_xD_\Phi(u_x,u_y)+g_yD_\Phi(u_y,u_x).
\end{split}
\end{equation}
\end{lemma}

\begin{proof}
Collect the $g_x$ and $g_y$ coefficients separately.  The coefficient of
$g_x$ on the left of \eqref{eq:doob-bregman-algebra} is
\[
u_x\bigl(\Phi'(u_x)-\Phi'(u_y)\bigr)+G(u_x)-G(u_y).
\]
Substituting $G(u)=\Phi(u)-u\Phi'(u)$ gives
\[
\Phi(u_x)-\Phi(u_y)-\Phi'(u_y)(u_x-u_y)
=D_\Phi(u_x,u_y).
\]
The coefficient of $g_y$ is
\[
-u_y\bigl(\Phi'(u_x)-\Phi'(u_y)\bigr)-G(u_x)+G(u_y)
=\Phi(u_y)-\Phi(u_x)-\Phi'(u_x)(u_y-u_x)
=D_\Phi(u_y,u_x).
\]
This proves the identity.  No analytic domain assertion enters this step.
\end{proof}

\begin{lemma}[Compact quotient range and Bregman integrability]%
\label{lem:doob-range-bregman-integrability}
Let $[a,b]\subset(0,\infty)$ and let $\nu$ be a probability measure
on~$\RR$ with mean~$\bar{\gamma}$ and finite centred second moment
\[
\mu_2:=\int_{\RR}(\gamma-\bar{\gamma})^2\dd\nu(\gamma).
\]
Put $h_t=P_t*\nu$, $g_t=P_t(\cdot-\bar{\gamma})$, and
$u_t=h_t/g_t$.  Then, for every $t\in[a,b]$ and every $x\in\RR$,
\begin{equation}\label{eq:doob-compact-range-bound}
\frac{1}{2(1+\mu_2/a^2)}
\le u_t(x)\le
2+\frac{2\mu_2}{a^2}.
\end{equation}
Moreover $u_t$ is uniformly Lipschitz in $x$ on
$\RR\times[a,b]$.  Consequently, if
$\Phi:(0,\infty)\to\RR$ is convex, of class $C^1$, and $\Phi'$ is
Lipschitz on compact subintervals of $(0,\infty)$, then there is a
constant $C_{\Phi,a,b}<\infty$ such that, for all $t\in[a,b]$ and
$x,y\in\RR$,
\begin{equation}\label{eq:doob-bregman-pointwise-bound}
0\le D_\Phi(u_t(x),u_t(y))
\le C_{\Phi,a,b}\min\{|x-y|^2,1\}.
\end{equation}
In particular
\begin{equation}\label{eq:doob-bregman-integrability-bound}
\sup_{t\in[a,b]}
\iint_{\RR^2}
\frac{
g_t(x)D_\Phi(u_t(x),u_t(y))
+g_t(y)D_\Phi(u_t(y),u_t(x))}
{(x-y)^2}\dd x\dd y<\infty,
\end{equation}
and $\int_{\RR}g_t(x)|\Phi(u_t(x))|\dd x<\infty$ for every
$t\in[a,b]$.
\end{lemma}

\begin{proof}
The compact-range estimate \eqref{eq:doob-compact-range-bound} is the
Poisson quotient bound \eqref{eq:prelim-poisson-quotient-two-sided-bound} in
Lemma~\ref{lem:kl-bounded-test-and-poisson-quotient}.  The rest of this
lemma records the additional Lipschitz and Bregman consequences needed for
the Doob-domain closure.

Lemma~\ref{lem:cauchy-doob-cutoff-calculus} gives a constant
$L_{a,b}$ such that $|\partial_xu_t(x)|\le L_{a,b}$ for
$t\in[a,b]$.  Let
\[
m_{a,b}:=\frac{1}{2(1+\mu_2/a^2)},
\qquad
M_{a,b}:=2+\frac{2\mu_2}{a^2}.
\]
If $\Phi'$ has Lipschitz constant $L_\Phi$ on
$[m_{a,b},M_{a,b}]$, then the integral form of the Bregman remainder gives
\[
0\le D_\Phi(\alpha,\beta)\le \frac{L_\Phi}{2}|\alpha-\beta|^2
\qquad(m_{a,b}\le\alpha,\beta\le M_{a,b}).
\]
Combining this with the Lipschitz bound for $u_t$ and with the trivial
bound $|u_t(x)-u_t(y)|\le M_{a,b}-m_{a,b}$ proves
\eqref{eq:doob-bregman-pointwise-bound}.

Dividing \eqref{eq:doob-bregman-pointwise-bound} by $(x-y)^2$ gives the
majorant $C_{\Phi,a,b}\min\{1,|x-y|^{-2}\}$.  Since
\[
\int_{\RR}\min\{1,|x-y|^{-2}\}\dd y<\infty
\]
with a value independent of~$x$, and since $\int g_t=1$, the first term
in \eqref{eq:doob-bregman-integrability-bound} is bounded uniformly in
$t\in[a,b]$; the second term is identical after exchanging $x$ and~$y$.
Finally, \eqref{eq:doob-compact-range-bound} puts $u_t$ in a compact
subinterval of $(0,\infty)$, so $\Phi$ is bounded on the range and
$\int g_t|\Phi(u_t)|\le\sup_{[m_{a,b},M_{a,b}]}|\Phi|$.
\end{proof}

\begin{lemma}[Compact-range approximation for range-local $C^{1,1}$ entropies]%
\label{lem:compact-range-phi-approximation}
Let $[a,b]\subset(0,\infty)$, and let $h_t$, $g_t$, and $u_t=h_t/g_t$ be as
in Lemma~\ref{lem:doob-range-bregman-integrability}.  Let
$J=[m_{a,b},M_{a,b}]$ be the compact interval from its proof.  If
$\Phi:(0,\infty)\to\RR$ is convex, $C^1$, and $\Phi'$ is Lipschitz on compact
subintervals of $(0,\infty)$, then there are convex functions
$\Phi_\varepsilon\in C^\infty((0,\infty))$, defined for all sufficiently small
$\varepsilon>0$, such that
\[
\sup_{s\in J}|\Phi_\varepsilon(s)-\Phi(s)|
+\sup_{s\in J}|\Phi_\varepsilon'(s)-\Phi'(s)|\longrightarrow0,
\]
the functions $\Phi_\varepsilon'$ have a common Lipschitz constant on~$J$, and
\[
\sup_{\alpha,\beta\in J}
|D_{\Phi_\varepsilon}(\alpha,\beta)-D_\Phi(\alpha,\beta)|\longrightarrow0.
\]
Consequently the entropy functionals and the Bregman dissipation integrands in
Proposition~\ref{prop:doob-bregman-domain-closure} may be obtained as limits
from the same formulas for the smooth convex functions $\Phi_\varepsilon$,
with the same compact-range majorants on every time interval $[a,b]$.
\end{lemma}

\begin{proof}
Choose an open interval $J_*$ with compact closure in $(0,\infty)$ and
$J\subset J_*$.  Since $\Phi$ is convex and $C^1$, $\Phi'$ is nondecreasing;
by hypothesis it has some Lipschitz constant $L_*$ on~$J_*$.  Extend
$\Phi'|_{J_*}$ to a globally defined nondecreasing Lipschitz function
$\Psi$ on~$\RR$ with Lipschitz constant $L_*$, for instance by continuing it
linearly with slope~$0$ outside $J_*$.  Let $\rho_\varepsilon$ be a standard
nonnegative even mollifier and set
$\Psi_\varepsilon=\rho_\varepsilon*\Psi$.  For $\varepsilon$ smaller than the
distance from $J$ to $\RR\setminus J_*$, the functions
$\Psi_\varepsilon$ are smooth, nondecreasing, have Lipschitz constant at most
$L_*$ on~$J$, and converge uniformly to $\Phi'$ on~$J$.
Define
\[
\Phi_\varepsilon(s):=\Phi(1)+\int_1^s\Psi_\varepsilon(r)\dd r,
\]
after any smooth convex extension away from $J_*$ if necessary.  Then
$\Phi_\varepsilon$ is smooth and convex on $(0,\infty)$, and the displayed
uniform convergence of $\Phi_\varepsilon$ and $\Phi_\varepsilon'$ on~$J$
follows by integrating the uniform convergence of the derivatives on the
compact interval.  The Bregman convergence is immediate from
\[
|D_{\Phi_\varepsilon}(\alpha,\beta)-D_\Phi(\alpha,\beta)|
\le 2\sup_J|\Phi_\varepsilon-\Phi|
 +(M_{a,b}-m_{a,b})\sup_J|\Phi_\varepsilon'-\Phi'|.
\]
Lemma~\ref{lem:doob-range-bregman-integrability} gives the common bound
$0\le D_{\Phi_\varepsilon}(u_t(x),u_t(y))
\le C\min\{|x-y|^2,1\}$ with $C$ independent of~$\varepsilon$, and it also
puts $u_t$ in~$J$.  Since $\int g_t=1$, dominated convergence applies to the
one-variable entropy integrals and to the two-variable Bregman dissipation
integrals.  This proves the limiting assertion.
\end{proof}

\begin{lemma}[\(L^1\) chain rule for bounded Bregman quotients]%
\label{lem:l1-bounded-quotient-entropy-chain-rule}
Let \((X,\mathcal A,\mu)\) be a sigma-finite measure space, let
\(I=[a,b]\), and let \(h,g:I\times X\to(0,\infty)\) be measurable
functions such that, for a.e. \(x\), the maps \(t\mapsto h_t(x)\) and
\(t\mapsto g_t(x)\) are absolutely continuous on \(I\).  Assume
\[
h,g,\partial_th,\partial_tg\in L^1(I\times X),
\qquad
0<m\le \frac{h_t(x)}{g_t(x)}\le M<\infty
\]
for a.e. \((t,x)\).  Let \(\Phi\in C^1((0,\infty))\) and assume that
\(\Phi'\) is Lipschitz on \([m,M]\).  Define
\[
F(r,s):=s\Phi(r/s),\qquad
A(r,s):=\Phi'(r/s),\qquad
B(r,s):=\Phi(r/s)-(r/s)\Phi'(r/s).
\]
Equivalently, on the cone \(s>0\), \(m\le r/s\le M\),
\[
\partial_h\!\left[g\Phi(h/g)\right]=\Phi'(h/g),
\qquad
\partial_g\!\left[g\Phi(h/g)\right]
=\Phi(h/g)-(h/g)\Phi'(h/g).
\]
Then
\[
H(t):=\int_X F(h_t(x),g_t(x))\,\dd\mu(x)
\]
is absolutely continuous on \(I\), and, with \(u_t:=h_t/g_t\), for a.e.
\(t\in I\),
\begin{equation}\label{eq:l1-bounded-quotient-chain-rule-appendix}
\begin{split}
H'(t)
&=
\int_X
\Phi'(u_t(x))\,\partial_th_t(x)
\dd\mu(x)\\
&\quad+
\int_X
\bigl(\Phi(u_t(x))-u_t(x)\Phi'(u_t(x))\bigr)\,
\partial_tg_t(x)\dd\mu(x).
\end{split}
\end{equation}
Moreover, if \(0\le\eta_R\le1\) are measurable cutoffs, then
\[
H_R(t):=\int_X \eta_R(x)F(h_t(x),g_t(x))\,\dd\mu(x)
\]
is absolutely continuous and satisfies the same derivative formula with
\(\eta_R(x)\) inserted in both integrals.  If, in addition,
\(\sup_{t\in I}\int_X(1-\eta_R)g_t\,\dd\mu\to0\), then
\[
\sup_{t\in I}|H_R(t)-H(t)|\longrightarrow0 .
\]
\end{lemma}

\begin{proof}
On the compact interval \([m,M]\), the functions
\[
s\mapsto \Phi(s),\qquad s\mapsto \Phi'(s),\qquad
s\mapsto \Phi(s)-s\Phi'(s)
\]
are bounded.  Thus the two displayed partial derivatives of
\((h,g)\mapsto g\Phi(h/g)\) are bounded on the closed quotient cone
\(m\le h/g\le M\).  Hence there is \(C_{\Phi,m,M}<\infty\) such that, whenever
\(m\le r/s\le M\),
\[
|F(r,s)|\le C_{\Phi,m,M}s,\qquad
|A(r,s)|+|B(r,s)|\le C_{\Phi,m,M}.
\]
For a.e. \(x\), the scalar curve \(t\mapsto(h_t(x),g_t(x))\) is absolutely
continuous and remains in the closed cone.  The ordinary one-dimensional chain
rule gives, for a.e. \(t\),
\[
\frac{\dd}{\dd t}F(h_t(x),g_t(x))
=A(h_t(x),g_t(x))\,\partial_th_t(x)
+B(h_t(x),g_t(x))\,\partial_tg_t(x).
\]
The right side is dominated by
\[
C_{\Phi,m,M}\bigl(|\partial_th_t(x)|+|\partial_tg_t(x)|\bigr)
\]
in \(L^1(I\times X)\).  Fubini's theorem gives the integrated identity for
\(H(t)-H(s)\), hence absolute continuity and
\eqref{eq:l1-bounded-quotient-chain-rule-appendix}.  Multiplication by a fixed
cutoff preserves the same majorant.  Finally,
\[
|H_R(t)-H(t)|\le
C_{\Phi,m,M}\int_X(1-\eta_R)g_t\,\dd\mu,
\]
which proves the uniform cutoff convergence.
\end{proof}

\begin{proposition}[Smooth compactly supported Doob--Bregman identity]%
\label{prop:smooth-compact-doob-bregman}
Let $t\mapsto g_t$ be a positive $C^1$ curve of smooth densities on an open
time interval, and let $t\mapsto u_t$ be a positive $C^1$ curve of smooth
functions such that $u_t-1\in C_c^\infty(\RR)$ for each $t$, with one compact
support and one compact quotient range on compact time intervals.  Put
$h_t=g_tu_t$ and assume
\[
\partial_t h_t=Lh_t,\qquad \partial_tg_t=Lg_t,
\]
with all integrations below absolutely convergent; for instance this holds
after replacing the noncompact tests in Lemma~\ref{lem:cauchy-doob-cutoff-calculus}
by fixed spatial cutoffs.  Let $\Phi\in C^2((0,\infty))$ with $\Phi(1)=0$.
Then
\[
\mathcal H_\Phi(t):=\int_{\RR}g_t(x)\Phi(u_t(x))\dd x
\]
is $C^1$, and
\begin{equation}\label{eq:smooth-compact-phi-entropy-dissipation}
-\frac{\dd}{\dd t}\mathcal H_\Phi(t)
=
\frac{1}{2\pi}\iint_{\RR^2}
\frac{
g_t(x)D_\Phi(u_t(x),u_t(y))
+g_t(y)D_\Phi(u_t(y),u_t(x))}
{(x-y)^2}\dd x\dd y .
\end{equation}
\end{proposition}

\begin{proof}
Because $u_t-1$ has compact support and $\Phi(1)=0$, the entropy integrand is
supported where $u_t\ne1$.  Differentiating under the integral and using
$h_t=g_tu_t$ gives
\[
\frac{\dd}{\dd t}\mathcal H_\Phi(t)
=
\int_{\RR}\Phi'(u_t)\,Lh_t\,\dd x
+
\int_{\RR}\bigl(\Phi(u_t)-u_t\Phi'(u_t)\bigr)Lg_t\,\dd x .
\]
The functions $\Phi'(u_t)-\Phi'(1)$ and
$Q_\Phi(u_t)-Q_\Phi(1)$ are smooth and compactly supported, and constants do
not change the bilinear form.  The constant pairings with $Lh_t$ and $Lg_t$
vanish because the generator preserves total mass.  Applying the compactly
supported Cauchy bilinear identity
\[
\int_{\RR}qLp\,\dd x
=-\frac{1}{2\pi}\iint_{\RR^2}
\frac{(p(x)-p(y))(q(x)-q(y))}{(x-y)^2}\dd x\dd y
\]
first with $p=h_t=g_tu_t$, $q=\Phi'(u_t)$, and then with
$p=g_t$, $q=Q_\Phi(u_t)$, where
$Q_\Phi(s)=\Phi(s)-s\Phi'(s)$.  Thus
\[
\frac{\dd}{\dd t}\mathcal H_\Phi(t)
=-\frac{1}{2\pi}\iint_{\RR^2}\frac{E_t(x,y)}{(x-y)^2}\dd x\dd y,
\]
where, with $u_x=u_t(x)$, $u_y=u_t(y)$, $g_x=g_t(x)$, and $g_y=g_t(y)$,
\[
E_t(x,y)
=(g_xu_x-g_yu_y)\bigl(\Phi'(u_x)-\Phi'(u_y)\bigr)
 +(g_x-g_y)\bigl(Q_\Phi(u_x)-Q_\Phi(u_y)\bigr).
\]
Lemma~\ref{lem:doob-bregman-algebra} rewrites this exactly as
\[
E_t(x,y)=g_t(x)D_\Phi(u_t(x),u_t(y))
+g_t(y)D_\Phi(u_t(y),u_t(x)).
\]
Substitution gives \eqref{eq:smooth-compact-phi-entropy-dissipation}.
\end{proof}

\begin{lemma}[Doob--Bregman domain bounds]%
\label{lem:doob-bregman-domain-bounds}
Let $[a,b]\subset(0,\infty)$, let $\nu$ have finite centred second moment, and
use the notation
\[
h_t=P_t*\nu,
\qquad
g_t=P_t(\cdot-\bar{\gamma}),
\qquad
u_t=h_t/g_t .
\]
Let $\Phi$ be convex on $(0,\infty)$ and assume that, on every compact
subinterval of $(0,\infty)$, $\Phi$ is $C^1$ and $\Phi'$ is Lipschitz.  Then,
uniformly for $t\in[a,b]$, the following analytic bounds hold.
\begin{enumerate}[label=\textup{(\roman*)},leftmargin=*]
\item The principal-value identities for $Lh_t$, $Lg_t$, and
$L(g_tu_t)-u_tLg_t$ are the limits of the cutoff identities in
Lemma~\ref{lem:cauchy-doob-cutoff-calculus}; the associated bilinear Cauchy
forms are absolutely convergent.
\item The quotient range $u_t(\RR)$ is contained in one compact interval
$J_{a,b}\Subset(0,\infty)$, and the Bregman integrand
\[
\frac{g_t(x)D_\Phi(u_t(x),u_t(y))
      +g_t(y)D_\Phi(u_t(y),u_t(x))}{(x-y)^2}
\]
is absolutely integrable on $\RR^2$, with a bound independent of
$t\in[a,b]$.  More explicitly, there is
$C_{\Phi,a,b}<\infty$ such that, for
\[
K_t^\Phi(x,y):=
\frac{g_t(x)D_\Phi(u_t(x),u_t(y))}{(x-y)^2},
\qquad x\ne y,
\]
one has
\begin{align*}
0\le K_t^\Phi(x,y)
&\le C_{\Phi,a,b}g_t(x)\mathbf 1_{\{|x-y|<1\}},\\
0\le K_t^\Phi(x,y)
&\le C_{\Phi,a,b}g_t(x)|x-y|^{-2}
\mathbf 1_{\{|x-y|\ge1\}},
\end{align*}
and the same two bounds with \(x\) and \(y\) exchanged.  These are
\(L^1([a,b]\times\RR^2)\) majorants and therefore remove the symmetric
cutoffs \(\chi_R(x)\chi_R(y)\) by dominated convergence.
\item The map
\[
t\longmapsto \mathcal H_\Phi(t):=
\int_{\RR}g_t(x)\Phi(u_t(x))\dd x
\]
is absolutely continuous on $[a,b]$, and its derivative is obtained by
first proving the identity for the cutoff bilinear forms and then letting the
cutoff radius tend to infinity.
\item The passage from smooth convex entropies to the above range-local
$C^{1,1}$ class is justified by the approximants
$\Phi_\varepsilon$ of Lemma~\ref{lem:compact-range-phi-approximation}; on
$J_{a,b}$ the entropies, first derivatives, and Bregman divergences converge
under the same integrable majorants as in~\textup{(ii)}.
\end{enumerate}
Consequently the dissipation formula
\eqref{eq:phi-entropy-dissipation} is a domain statement in the stated
principal-value normalization, with the nonlocal product rule justified by the
cutoff and approximation limits above.
\end{lemma}

\begin{proof}
Part (i) is exactly Lemma~\ref{lem:cauchy-doob-cutoff-calculus}.  That lemma
first proves the compactly supported Cauchy-generator identities, then replaces
the test function by $\chi_Rq$ and passes to the limit using the near-diagonal
estimate controlled by $\partial_xp\in L^1$ and the far-field estimate
controlled by $p\in L^1$.  Thus the limiting forms are absolutely convergent
and fix the principal-value convention used here.

Lemma~\ref{lem:doob-range-bregman-integrability} proves the compact quotient
range.  Since \(\Phi'\) is Lipschitz on that range, the integral form of the
Bregman remainder gives
\[
0\le D_\Phi(u_t(x),u_t(y))
\le C_{\Phi,a,b}|u_t(x)-u_t(y)|^2.
\]
Lemma~\ref{lem:cauchy-doob-cutoff-calculus} and
Lemma~\ref{lem:appendix-poisson-quotient-kernel-calculus} give
\(|u_t(x)-u_t(y)|\le L_{a,b}|x-y|\), while compactness of the quotient range
also gives \(|u_t(x)-u_t(y)|\le M_{a,b}-m_{a,b}\).  Hence
\[
0\le D_\Phi(u_t(x),u_t(y))
\le C_{\Phi,a,b}\min\{|x-y|^2,1\}.
\]
Dividing by \((x-y)^2\) gives exactly the displayed near-diagonal and
far-field bounds for \(K_t^\Phi\), and the exchanged-variable estimates are
identical.  They are integrable because
\[
\int_a^b\int_{\RR}g_t(x)\dd x\dd t=b-a,\qquad
\int_{|z|<1}\dd z=2,\qquad
\int_{|z|\ge1}|z|^{-2}\dd z=2 .
\]
This proves (ii), including the cutoff convergence by dominated convergence.

For (iii), Lemma~\ref{lem:cauchy-doob-cutoff-calculus} gives uniform
$L^1$ bounds for $\partial_th_t$ and $\partial_tg_t$ on $[a,b]$.  Thus
$h,g,\partial_th,\partial_tg\in L^1([a,b]\times\RR)$, and
Lemma~\ref{lem:doob-range-bregman-integrability} gives
\[
m_{a,b}\le h_t(x)/g_t(x)\le M_{a,b}
\qquad(t\in[a,b],\ x\in\RR).
\]
The Poisson kernels are \(C^1\) in \(t\) pointwise, so \(h_t(x)\) and
\(g_t(x)\) are absolutely continuous in \(t\) for every fixed \(x\).
Lemma~\ref{lem:l1-bounded-quotient-entropy-chain-rule}, applied with
\(X=\RR\), \(F(h,g)=g\Phi(h/g)\), and
\([m,M]=[m_{a,b},M_{a,b}]\), therefore gives
\(\mathcal H_\Phi\in AC([a,b])\) and, for a.e. \(t\),
\[
\frac{\dd}{\dd t}\mathcal H_\Phi(t)
=
\int_{\RR}\Phi'(u_t)\,\partial_th_t\dd x
+
\int_{\RR}\bigl(\Phi(u_t)-u_t\Phi'(u_t)\bigr)\partial_tg_t\dd x .
\]
Its localized version gives the same formula with any fixed spatial cutoff.
Applying the cutoff identity from part (i) to the bounded Lipschitz tests
\(\Phi'(u_t)\) and \(\Phi(u_t)-u_t\Phi'(u_t)\) gives the displayed derivative
at the cutoff level; the absolute majorants from parts (i) and (ii) justify
passage to the limit in the cutoff radius.

Finally, Lemma~\ref{lem:compact-range-phi-approximation} constructs smooth
convex $\Phi_\varepsilon$ that agree with $\Phi$ to first order on the compact
range in the limit and have uniformly bounded second derivatives there.  The
one-variable entropy terms are dominated by $g_t$ times a constant, and the
two-variable Bregman terms are dominated by the majorant from (ii).  Dominated
convergence therefore passes the smooth identity to the range-local
$C^{1,1}$ entropy.  This proves all four assertions.
\end{proof}

\begin{proposition}[Domain closure of the Doob--Bregman identity]%
\label{prop:doob-bregman-domain-closure}
Let $\nu$ be a probability measure on~$\RR$ with mean~$\bar{\gamma}$ and
finite centred second moment, and use the notation of
Theorem~\ref{thm:poisson-doob-phi-entropy}.  Let
$\Phi:(0,\infty)\to\RR$ be convex, of class $C^1$, with range-local
$C^{1,1}$ control in the sense that $\Phi'$ is Lipschitz on every compact
subinterval of $(0,\infty)$, and assume $\Phi(1)=0$.  Define
\[
\mathcal H_\Phi(t):=\int_{\RR}g_t(x)\Phi(u_t(x))\dd x .
\]
Let \(0<a<b<\infty\).  Then there are constants
\[
0<m_{a,b}\le M_{a,b}<\infty,\qquad L_{a,b}<\infty,\qquad
C_{\Phi,a,b}<\infty,
\]
depending only on \(a,b,\Phi\) and the centred second moment of \(\nu\), such
that, for all \(t\in[a,b]\),
\begin{equation}\label{eq:phi-domain-core-range}
m_{a,b}\le u_t(x)\le M_{a,b},\qquad
|\partial_xu_t(x)|\le L_{a,b}\qquad(x\in\RR),
\end{equation}
and, for \(x\ne y\),
\begin{equation}\label{eq:phi-domain-core-majorant}
\frac{g_t(x)D_\Phi(u_t(x),u_t(y))}{(x-y)^2}
\le C_{\Phi,a,b}g_t(x)\min\{1,|x-y|^{-2}\}.
\end{equation}
Consequently $\mathcal H_\Phi(t)<\infty$ for every $t>0$, and
$t\mapsto\mathcal H_\Phi(t)$ is absolutely continuous on compact
subintervals of $(0,\infty)$.  If
\(\chi\in C_c^\infty(\RR)\), \(0\le\chi\le1\), \(\chi\equiv1\) on
\([-1,1]\), and \(\chi_R(x)=\chi(x/R)\), then
\[
H_R(t):=\int_{\RR}\chi_R(x)g_t(x)\Phi(u_t(x))\dd x
\]
satisfies \(H_R\in AC([a,b])\), \(H_R\to\mathcal H_\Phi\) uniformly on
\([a,b]\), and there are \(e_R\in L^1([a,b])\) with
\(\|e_R\|_{L^1([a,b])}\to0\) such that, for a.e. \(t\in[a,b]\),
\begin{equation}\label{eq:phi-domain-localized-cutoff-balance}
\begin{split}
-\frac{\dd}{\dd t}H_R(t)
&=
\frac{1}{2\pi}\iint_{\RR^2}
\chi_R(x)\chi_R(y)
\frac{
g_t(x)D_\Phi(u_t(x),u_t(y))
+g_t(y)D_\Phi(u_t(y),u_t(x))}
{(x-y)^2}\dd x\dd y
+e_R(t).
\end{split}
\end{equation}
Passing \(R\to\infty\) gives, for a.e. \(t\in[a,b]\),
\begin{equation}\label{eq:phi-entropy-dissipation}
-\frac{\dd}{\dd t}\mathcal H_\Phi(t)
=
\frac{1}{2\pi}\iint_{\RR^2}
\frac{
g_t(x)D_\Phi(u_t(x),u_t(y))
+g_t(y)D_\Phi(u_t(y),u_t(x))}
{(x-y)^2}\dd x\dd y,
\end{equation}
where
\[
D_\Phi(a,b):=\Phi(a)-\Phi(b)-\Phi'(b)(a-b)
\qquad(a,b>0).
\]
The right side is the finite cutoff-limit integral fixed by
Lemmas~\ref{lem:cauchy-doob-cutoff-calculus},
\ref{lem:doob-range-bregman-integrability}, and
\ref{lem:doob-bregman-domain-bounds}.  The convergence of the two-variable
cutoff integrals to the right side of \eqref{eq:phi-entropy-dissipation}
holds in \(L^1([a,b])\).  In particular \(t\mapsto\mathcal H_\Phi(t)\) is
nonincreasing.
\end{proposition}

\begin{proof}
Fix $[a,b]\subset(0,\infty)$.  Lemma~\ref{lem:doob-range-bregman-integrability}
places $u_t(\RR)$, $t\in[a,b]$, in the compact interval
\[
J_{a,b}:=[m_{a,b},M_{a,b}]
\Subset(0,\infty),
\qquad
m_{a,b}:=\frac{1}{2(1+\mu_2/a^2)},\quad
M_{a,b}:=2+\frac{2\mu_2}{a^2},
\]
and Lemma~\ref{lem:cauchy-doob-cutoff-calculus} supplies
\[
|\partial_xu_t(x)|\le L_{a,b}\qquad(t\in[a,b]).
\]
Since \(\Phi'\) is Lipschitz on \(J_{a,b}\), the integral form of the
Bregman remainder and the Lipschitz bound give
\[
0\le D_\Phi(u_t(x),u_t(y))
\le C_{\Phi,a,b}\min\{|x-y|^2,1\}.
\]
Dividing by \((x-y)^2\) proves the one-sided majorant
\eqref{eq:phi-domain-core-majorant}; the same estimate with \(x,y\)
exchanged controls the second Bregman term.  These bounds prove
\(\mathcal H_\Phi(t)<\infty\) and give the uniform Bregman double-integral
bound \eqref{eq:doob-bregman-integrability-bound}.
For the cutoff limit we shall use the following explicit decomposition of the
same majorant.  With
\[
K_t^\Phi(x,y):=
\frac{g_t(x)D_\Phi(u_t(x),u_t(y))}{(x-y)^2},
\qquad x\ne y,
\]
one has, uniformly in \(t\in[a,b]\),
\begin{align}
0\le K_t^\Phi(x,y)
&\le C_{\Phi,a,b}g_t(x)\mathbf 1_{\{|x-y|<1\}},
\label{eq:phi-cutoff-near-majorant}\\
0\le K_t^\Phi(x,y)
&\le C_{\Phi,a,b}g_t(x)|x-y|^{-2}
\mathbf 1_{\{|x-y|\ge1\}} .
\label{eq:phi-cutoff-far-majorant}
\end{align}
The corresponding estimates for
\(g_t(y)D_\Phi(u_t(y),u_t(x))/(x-y)^2\) are obtained by exchanging the
variables.  These are actual \(L^1([a,b]\times\RR^2)\) majorants, since
\[
\int_a^b\int_{\RR}g_t(x)\dd x\dd t=b-a,\qquad
\int_{|z|<1}\dd z=2,\qquad
\int_{|z|\ge1}|z|^{-2}\dd z=2 .
\]
The closure argument below deliberately separates the proof into three
steps.  First,
Proposition~\ref{prop:smooth-compact-doob-bregman} proves the formula for
smooth compactly supported quotients, with
Lemma~\ref{lem:doob-bregman-algebra} as the only nontrivial algebraic
identity.  Second, the cutoff estimates of
Lemma~\ref{lem:cauchy-doob-cutoff-calculus} pass the bilinear forms from
compactly supported tests to the Poisson quotient.  Third, the compact-range
approximation of Lemma~\ref{lem:compact-range-phi-approximation} passes from
smooth convex entropies to the stated range-local $C^{1,1}$ class.

Lemma~\ref{lem:cauchy-doob-cutoff-calculus} gives uniform $L^1$ bounds for
$\partial_th_t$ and $\partial_tg_t$, so $t\mapsto h_t$ and $t\mapsto g_t$ are
absolutely continuous as $L^1$-valued maps on $[a,b]$.  Together with the
compact quotient range \(m_{a,b}\le h_t/g_t\le M_{a,b}\), these are precisely
the integrability and range hypotheses of
Lemma~\ref{lem:l1-bounded-quotient-entropy-chain-rule}.  That lemma gives
\(\mathcal H_\Phi\in AC([a,b])\) and, for a.e. \(t\),
\[
\frac{\dd}{\dd t}\mathcal H_\Phi(t)
=
\int_{\RR}\Phi'(u_t)\,\partial_th_t\dd x
+
\int_{\RR}\bigl(\Phi(u_t)-u_t\Phi'(u_t)\bigr)\partial_tg_t\dd x.
\]
It also gives the same derivative formula after inserting a fixed cutoff
\(\chi_R\), and
\[
\sup_{t\in[a,b]}
\left|
\int_{\RR}(1-\chi_R(x))g_t(x)\Phi(u_t(x))\dd x
\right|
\le
\sup_{J_{a,b}}|\Phi|
\sup_{t\in[a,b]}\int_{\RR}(1-\chi_R(x))g_t(x)\dd x
\to0,
\]
where the final convergence follows from the explicit translated Poisson
kernel \(g_t(x)=P_t(x-\bar{\gamma})\), uniformly for \(t\in[a,b]\).
Substituting $\partial_th_t=L(u_tg_t)$ and $\partial_tg_t=Lg_t$, and applying
the cutoff bilinear identity \eqref{eq:cutoff-cauchy-bilinear} with the
bounded Lipschitz tests $\Phi'(u_t)$ and
$\Phi(u_t)-u_t\Phi'(u_t)$, gives the cutoff-limit version of the smooth
identity in Proposition~\ref{prop:smooth-compact-doob-bregman}.  The algebraic
step in that passage is precisely Lemma~\ref{lem:doob-bregman-algebra}, with
\[
u_x=u_t(x),\qquad u_y=u_t(y),\qquad g_x=g_t(x),\qquad g_y=g_t(y).
\]
Thus the cutoff limit is the Bregman integral in
\eqref{eq:phi-entropy-dissipation}.

The same argument with the cutoff inserted gives the localized statement.
For every \(R\ge1\),
\[
H_R(t):=\int_{\RR}\chi_R(x)g_t(x)\Phi(u_t(x))\dd x
\]
is absolutely continuous on \([a,b]\), and for a.e. \(t\)
\[
\frac{\dd}{\dd t}H_R(t)
=
\int_{\RR}\chi_R(x)\Phi'(u_t(x))\,\partial_th_t(x)\dd x
+\int_{\RR}\chi_R(x)
\bigl(\Phi(u_t(x))-u_t(x)\Phi'(u_t(x))\bigr)\partial_tg_t(x)\dd x .
\]
Applying the cutoff bilinear identity to the bounded Lipschitz tests
\(\chi_R\Phi'(u_t)\) and
\(\chi_R(\Phi(u_t)-u_t\Phi'(u_t))\), and then using
Lemma~\ref{lem:doob-bregman-algebra}, gives the symmetric Bregman expression
with the product cutoff \(\chi_R(x)\chi_R(y)\), plus the commutator
\(e_R(t)\) whose exact formula is
\eqref{eq:phi-cutoff-commutator-error-formula}.
Lemma~\ref{lem:doob-bregman-cutoff-commutator} splits this error into the
near-diagonal part \eqref{eq:phi-cutoff-commutator-near} and the far-field
part \eqref{eq:phi-cutoff-commutator-far}; the near estimate uses the
\(L^1\)-bounds for \(\partial_xh_t,\partial_xg_t\), and the far estimate uses
the probability-density bounds for \(h_t,g_t\) together with
\(\int_{|z|\ge1}|z|^{-2}\dd z<\infty\).  Its conclusion
\eqref{eq:phi-cutoff-commutator-l1-vanishing} gives
\(\|e_R\|_{L^1([a,b])}\to0\), and
\eqref{eq:phi-domain-localized-cutoff-balance} follows.  The same compact
range also gives
\[
\sup_{t\in[a,b]}|H_R(t)-\mathcal H_\Phi(t)|
\le \sup_{J_{a,b}}|\Phi|
\sup_{t\in[a,b]}\int_{\RR}|1-\chi_R(x)|g_t(x)\dd x\to0,
\]
so \(H_R\to\mathcal H_\Phi\) uniformly on \([a,b]\).

The remaining passages are exactly the closure statements in
Lemma~\ref{lem:doob-bregman-domain-bounds}.  The cutoff convergence follows
from dominated convergence with the explicit decomposition
\[
K_t^\Phi=K_t^\Phi\mathbf 1_{\{|x-y|<1\}}
        +K_t^\Phi\mathbf 1_{\{|x-y|\ge1\}},
\]
the near-diagonal majorant \eqref{eq:phi-cutoff-near-majorant}, the far-field
majorant \eqref{eq:phi-cutoff-far-majorant}, and their exchanged-variable
counterparts.  These are the exact majorants used above to prove
\[
\int_a^b\iint_{\RR^2}(1-\chi_R(x)\chi_R(y))K_t^\Phi(x,y)
\,\dd x\,\dd y\,\dd t\longrightarrow0
\]
for the first Bregman term, with the same argument after exchanging \(x\) and
\(y\).  Absolute convergence is the bound
\eqref{eq:doob-bregman-integrability-bound}; and the passage from smooth
convex entropies to the present range-local $C^{1,1}$ class uses the
approximants $\Phi_\varepsilon$ of
Lemma~\ref{lem:compact-range-phi-approximation}.  Their first derivatives have
a common Lipschitz constant on the compact quotient range, so the same
one-variable and Bregman majorants dominate both sides and dominated
convergence lets $\varepsilon\downarrow0$.  Convexity gives
$D_\Phi\ge0$, hence monotonicity.  Since $[a,b]$ was arbitrary, the proposition
follows.
\end{proof}

\begin{lemma}[Localized cutoff commutator for the Doob--Bregman identity]%
\label{lem:doob-bregman-cutoff-commutator}
Let the hypotheses and notation of
Lemma~\ref{lem:doob-bregman-domain-bounds} hold on
\([a,b]\Subset(0,\infty)\).  Put
\[
A_t(x):=\Phi'(u_t(x)),\qquad
G_t(x):=\Phi(u_t(x))-u_t(x)\Phi'(u_t(x)),
\]
and, for \(x\ne y\),
\begin{align*}
\mathcal B_t^\Phi(x,y)
&:=
g_t(x)D_\Phi(u_t(x),u_t(y))
+g_t(y)D_\Phi(u_t(y),u_t(x)),\\
\mathcal M_t^\Phi(x,y)
&:=
\bigl(h_t(x)-h_t(y)\bigr)A_t(y)
+\bigl(g_t(x)-g_t(y)\bigr)G_t(y).
\end{align*}
Let \(\chi_R(x)=\chi(x/R)\), where
\(\chi\in C_c^\infty(\RR)\), \(0\le\chi\le1\), and
\(\chi\equiv1\) on \([-1,1]\).  Define
\[
\mathcal C_R^\Phi(t,x,y)
:=
\chi_R(x)(1-\chi_R(y))\mathcal B_t^\Phi(x,y)
+(\chi_R(x)-\chi_R(y))\mathcal M_t^\Phi(x,y).
\]
Then the cutoff error in the localized entropy balance is exactly
\begin{equation}\label{eq:phi-cutoff-commutator-error-formula}
e_R(t)=
\frac{1}{2\pi}\iint_{\RR^2}
\frac{\mathcal C_R^\Phi(t,x,y)}{(x-y)^2}\dd x\dd y ,
\end{equation}
with the absolutely convergent near/far splitting
\begin{align}
e_R^{<}(t)
&:=
\frac{1}{2\pi}\iint_{|x-y|<1}
\frac{\mathcal C_R^\Phi(t,x,y)}{(x-y)^2}\dd x\dd y,
\label{eq:phi-cutoff-commutator-near}\\
e_R^{>}(t)
&:=
\frac{1}{2\pi}\iint_{|x-y|\ge1}
\frac{\mathcal C_R^\Phi(t,x,y)}{(x-y)^2}\dd x\dd y.
\label{eq:phi-cutoff-commutator-far}
\end{align}
Moreover
\begin{equation}\label{eq:phi-cutoff-commutator-l1-vanishing}
\int_a^b\bigl(|e_R^{<}(t)|+|e_R^{>}(t)|\bigr)\dd t
\le
\frac{1}{2\pi}\int_a^b\iint_{\RR^2}
\frac{|\mathcal C_R^\Phi(t,x,y)|}{|x-y|^2}\dd x\dd y\dd t
\longrightarrow0 .
\end{equation}
\end{lemma}

\begin{proof}
Write \(u_x=u_t(x)\), \(u_y=u_t(y)\), \(g_x=g_t(x)\),
\(g_y=g_t(y)\), \(h_x=g_xu_x\), \(h_y=g_yu_y\),
\(A_x=A_t(x)\), \(A_y=A_t(y)\), \(G_x=G_t(x)\), and \(G_y=G_t(y)\).
The cutoff bilinear computation for
\(\frac{\dd}{\dd t}H_R(t)\) produces the numerator
\[
\mathcal S_R(t,x,y)
:=
(h_x-h_y)\bigl(\chi_R(x)A_x-\chi_R(y)A_y\bigr)
+(g_x-g_y)\bigl(\chi_R(x)G_x-\chi_R(y)G_y\bigr).
\]
The desired product-cutoff numerator is
\[
\chi_R(x)\chi_R(y)
\Bigl[(h_x-h_y)(A_x-A_y)+(g_x-g_y)(G_x-G_y)\Bigr].
\]
By Lemma~\ref{lem:doob-bregman-algebra}, the bracket is
\(\mathcal B_t^\Phi(x,y)\).  Subtracting the product-cutoff numerator from
\(\mathcal S_R\) gives
\[
\begin{split}
&\mathcal S_R(t,x,y)-\chi_R(x)\chi_R(y)\mathcal B_t^\Phi(x,y)\\
&\qquad=
\chi_R(x)(1-\chi_R(y))\mathcal B_t^\Phi(x,y)
+(\chi_R(x)-\chi_R(y))
\bigl[(h_x-h_y)A_y+(g_x-g_y)G_y\bigr],
\end{split}
\]
which is exactly \(\mathcal C_R^\Phi(t,x,y)\).  This proves the formula for
the error once the bilinear identity is applied to the localized tests
\(\chi_R A_t\) and \(\chi_R G_t\).

It remains to justify the absolute convergence and the
\(L^1([a,b])\)-vanishing.  On the compact quotient range \(J_{a,b}\), the
functions \(\Phi\), \(\Phi'\), \(u_t\), \(A_t\), and \(G_t\) are uniformly
bounded, and \(A_t\), \(G_t\) are uniformly Lipschitz in \(x\).  The Bregman
part satisfies, by Lemma~\ref{lem:doob-bregman-domain-bounds},
\begin{align}
0\le \frac{\mathcal B_t^\Phi(x,y)}{|x-y|^2}
&\le C_{\Phi,a,b}(g_t(x)+g_t(y))\mathbf 1_{\{|x-y|<1\}},
\label{eq:phi-commutator-near-bregman-bound}\\
0\le \frac{\mathcal B_t^\Phi(x,y)}{|x-y|^2}
&\le C_{\Phi,a,b}(g_t(x)+g_t(y))|x-y|^{-2}
\mathbf 1_{\{|x-y|\ge1\}} .
\label{eq:phi-commutator-far-bregman-bound}
\end{align}
Both right-hand sides are integrable on
\([a,b]\times\RR^2\).

For the \(\mathcal M_t^\Phi\) part near the diagonal, the uniform boundedness
of \(A_t\) and \(G_t\) gives
\[
|\mathcal M_t^\Phi(x,y)|
\le C_{\Phi,a,b}\bigl(|h_t(x)-h_t(y)|+|g_t(x)-g_t(y)|\bigr).
\]
Using
\[
p_t(x)-p_t(y)
=(x-y)\int_0^1\partial_xp_t(y+s(x-y))\dd s,
\qquad p_t\in\{h_t,g_t\},
\]
and
\(|\chi_R(x)-\chi_R(y)|
\le \|\chi'\|_\infty |x-y|/R\le \|\chi'\|_\infty |x-y|\), we obtain
\begin{equation}\label{eq:phi-commutator-near-M-bound}
\frac{|(\chi_R(x)-\chi_R(y))\mathcal M_t^\Phi(x,y)|}{|x-y|^2}
\mathbf 1_{\{|x-y|<1\}}
\le
C_{\Phi,a,b,\chi}\int_0^1
\sum_{p\in\{h,g\}}|\partial_xp_t(y+s(x-y))|\dd s\,
\mathbf 1_{\{|x-y|<1\}} .
\end{equation}
The right-hand side is integrable on \([a,b]\times\RR^2\), because
Lemma~\ref{lem:cauchy-doob-cutoff-calculus} gives uniform \(L^1\)-bounds for
\(\partial_xh_t\) and \(\partial_xg_t\), and integration in the difference
variable contributes the finite factor \(\int_{|z|<1}\dd z=2\).
For each fixed \(t,x,y\), the factor
\(\chi_R(x)(1-\chi_R(y))\) and the factor
\(\chi_R(x)-\chi_R(y)\) tend to zero as \(R\to\infty\).  Dominated
convergence with
\eqref{eq:phi-commutator-near-bregman-bound} and
\eqref{eq:phi-commutator-near-M-bound} therefore gives
\[
\int_a^b\iint_{|x-y|<1}
\frac{|\mathcal C_R^\Phi(t,x,y)|}{|x-y|^2}
\dd x\dd y\dd t\longrightarrow0 .
\]

For the far field, boundedness of \(A_t\) and \(G_t\) gives
\[
|\mathcal M_t^\Phi(x,y)|
\le C_{\Phi,a,b}\bigl(h_t(x)+h_t(y)+g_t(x)+g_t(y)\bigr).
\]
Together with \(|\chi_R(x)-\chi_R(y)|\le1\) and
\eqref{eq:phi-commutator-far-bregman-bound}, this yields the integrable
majorant
\begin{equation}\label{eq:phi-commutator-far-M-bound}
\frac{|\mathcal C_R^\Phi(t,x,y)|}{|x-y|^2}
\mathbf 1_{\{|x-y|\ge1\}}
\le
C_{\Phi,a,b,\chi}
\frac{h_t(x)+h_t(y)+g_t(x)+g_t(y)}{|x-y|^2}
\mathbf 1_{\{|x-y|\ge1\}} .
\end{equation}
Indeed \(h_t\) and \(g_t\) are probability densities, so the right-hand side
has \([a,b]\times\RR^2\)-integral bounded by a constant times
\((b-a)\int_{|z|\ge1}|z|^{-2}\dd z\).  Pointwise again
\(\chi_R(x)(1-\chi_R(y))\to0\) and
\(\chi_R(x)-\chi_R(y)\to0\), so dominated convergence gives
\[
\int_a^b\iint_{|x-y|\ge1}
\frac{|\mathcal C_R^\Phi(t,x,y)|}{|x-y|^2}
\dd x\dd y\dd t\longrightarrow0 .
\]
Combining the near and far estimates proves
\eqref{eq:phi-cutoff-commutator-l1-vanishing}.
\end{proof}
