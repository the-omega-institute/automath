\subsection{Power divergences and a second moment phase}
\label{subsec:stable-power-divergence-phase}

For Gaussian smoothing, Chen and Niles-Weed proved matching asymptotics for
the chi-square divergence and relative entropy under exponential-square
tails \cite[Theorem~2.5]{ChenNilesWeed2022SmoothedWasserstein}.  Stable
tails distinguish these divergences already at the level of the optimal
uniform moment hypothesis.  We determine the resulting phase transition
for the power-divergence family.

For \(1<\theta\le2\), put
\begin{equation}\label{eq:stable-power-generator}
 \phi_\theta(t)
 :=\frac{t^\theta-\theta t+\theta-1}{\theta(\theta-1)},
 \qquad t\ge0,
\end{equation}
and, for positive probability densities \(f,g\), define
\begin{equation}\label{eq:stable-power-divergence}
 D_\theta(f\|g):=\int g\,\phi_\theta(f/g)\dd x.
\end{equation}
Thus \(D_2\) is one half of the Pearson chi-square divergence, while
\(\phi_\theta''(1)=1\) for every \(\theta\).

We first record the off-critical translate estimate needed below.  It is
the same stable-kernel calculation as
Lemma~\ref{lem:all-order-stable-critical-remainder}, with the integrability
exponent no longer fixed by the KL balance.

\begin{lemma}[Off-critical stable translation remainder]
\label{lem:off-critical-stable-translation-remainder}
Let \(r\ge1\) be an integer, \(1<q\le2\), and
\begin{equation}\label{eq:off-critical-stable-exponent-conditions}
 p\ge r,
 \qquad p\ge(d+\alpha)(1-1/q).
\end{equation}
Set \(m=\lfloor p\rfloor\).  Then
\begin{equation}\label{eq:off-critical-stable-remainder}
 \|F_\cdot^{(\alpha,d)}(z)-T_m^{(\alpha,d)}(\cdot,z)\|_{L^q(\Omega_{\alpha,d})}
 \le C
 \begin{cases}
  |z|^{m+1},&|z|\le1,\\
  |z|^p,&|z|>1,
 \end{cases}
 \le C|z|^p.
\end{equation}
Consequently, if a probability law \(\lambda\) has finite \(p\)-th
absolute moment, then
\begin{equation}\label{eq:off-critical-stable-integrated-remainder}
 q_s^\lambda
 =1+\sum_{j=1}^m s^{-j}B_j^\lambda+r_s^\lambda,
 \qquad
 \|r_s^\lambda\|_{L^q(\Omega_{\alpha,d})}=o(s^{-p}).
\end{equation}
\end{lemma}

\begin{proof}
The translate estimate used in the proof of
Lemma~\ref{lem:all-order-stable-critical-remainder} gives
\[
 \|F_\cdot^{(\alpha,d)}(z)\|_q
 \le C(1+|z|)^{(d+\alpha)(1-1/q)}.
\]
On the unit ball, Taylor's formula and the stable normalized-derivative
bounds give the first line of
\eqref{eq:off-critical-stable-remainder}.  Outside the unit ball, the
translate estimate, boundedness of the Taylor modes, \(m\le p\), and
\eqref{eq:off-critical-stable-exponent-conditions} give the second line.
Since \(m+1>p\), the final bound also holds on the unit ball.  Minkowski's
inequality and dominated convergence, exactly as in the last paragraph of
the proof of Lemma~\ref{lem:all-order-stable-critical-remainder}, prove
\eqref{eq:off-critical-stable-integrated-remainder}.
\end{proof}

The power generator has the following perturbation bound.  Its restriction
to \(q\ge\theta\) is the source of the additional moment phase.

\begin{lemma}[Power-divergence perturbation bound]
\label{lem:stable-power-perturbation}
Let \(1<\theta\le q\le2\), and write
\(\Psi_g(z)=g\phi_\theta(1+z/g)\).  There is a constant \(C\) such that
\begin{equation}\label{eq:stable-power-perturbation-bound}
 0\le \Psi_g(v+e)-\Psi_g(v)
 -\phi_\theta'(1+v/g)e\le C|e|^q
\end{equation}
whenever \(g\ge1/2\), \(|v|\le g/4\), and \(g+v+e\ge0\).  On the same
domain,
\begin{equation}\label{eq:stable-power-first-variation-bound}
 |\phi_\theta'(1+v/g)|\le C|v|.
\end{equation}
\end{lemma}

\begin{proof}
After division by \(g\), the left side of
\eqref{eq:stable-power-perturbation-bound} is \(g\) times the Bregman
remainder of \(\phi_\theta\), based at a point of \([3/4,5/4]\).  If
\(|e|/g\le1/2\), Taylor's theorem bounds this remainder by
\(C|e/g|^2\le C|e/g|^q\).  For negative increments of larger magnitude,
the admissible arguments range over a fixed compact subset of
\([0,5/4]\), including zero, and \(|e|/g\) is bounded below.  For positive
increments of larger magnitude, the explicit formula
\eqref{eq:stable-power-generator} bounds the remainder by
\(C\{(|e|/g)^\theta+|e|/g\}\le C|e/g|^q\).
Multiplication by \(g\) gives \(Cg^{1-q}|e|^q\le C|e|^q\).
Nonnegativity is convexity.  Finally,
\(\phi_\theta'(t)=(t^{\theta-1}-1)/(\theta-1)\), so the mean-value
theorem on \([3/4,5/4]\) proves
\eqref{eq:stable-power-first-variation-bound}.
\end{proof}

\begin{theorem}[Power-divergence moment phase transition under stable smoothing]
\label{thm:stable-power-divergence-moment-phase}
Let \(d\ge1\), \(0<\alpha<2\), \(r\ge1\) be an integer, and
\(1<\theta\le2\).  Put \(\beta=d+\alpha\) and define
\begin{equation}\label{eq:stable-power-optimal-moment-exponent}
 p_{r,\alpha,d}^{(\theta)}
 :=\max\left\{
 p_{r,\alpha,d},\frac{(d+\alpha)(\theta-1)}{\theta}
 \right\},
\end{equation}
where \(p_{r,\alpha,d}\) is given by
\eqref{eq:all-order-stable-critical-exponents}.  In particular,
\begin{equation}\label{eq:stable-chi-square-optimal-moment-exponent}
 p_{r,\alpha,d}^{(2)}
 =\max\left\{r,\frac{d+\alpha}{2}\right\}.
\end{equation}
Let \(\nu,\eta\) be
probability laws on \(\RR^d\) whose tensor moments agree below order \(r\),
such that
\begin{equation}\label{eq:stable-power-moment-hypotheses}
 \int_{\RR^d}|x|^{p_{r,\alpha,d}^{(\theta)}}
       (\nu+\eta)(\dd x)<\infty,
 \qquad
 \Delta_r=\int x^{\otimes r}(\nu-\eta)(\dd x)\ne0.
\end{equation}
Then \(D_\theta(p_s^{(\alpha,d)}*\nu\|
p_s^{(\alpha,d)}*\eta)\) is finite for every \(s>0\), and
\begin{equation}\label{eq:stable-power-first-unmatched-asymptotic}
 D_\theta(p_s^{(\alpha,d)}*\nu\|
          p_s^{(\alpha,d)}*\eta)
 =\mathcal Q_{\alpha,d,r}(\Delta_r)s^{-2r}+o(s^{-2r}).
\end{equation}

The exponent in \eqref{eq:stable-power-optimal-moment-exponent} is optimal
as a uniform complete-moment hypothesis, with two distinct sharpness
mechanisms.
\begin{enumerate}[label=\textup{(\roman*)}]
\item If
\[
 \max\left\{r,\frac{\beta(\theta-1)}\theta\right\}
 \le\rho<p_{r,\alpha,d}^{(\theta)},
\]
there are laws with finite \(\rho\)-th absolute moments, matching all
tensor moments below order \(r\), and having finite nonzero \(\Delta_r\),
for which the power divergence is finite at every scale but
\begin{equation}\label{eq:stable-power-analytic-sharpness}
 \limsup_{s\to\infty}s^{2r}
 D_\theta(p_s^{(\alpha,d)}*\nu\|
          p_s^{(\alpha,d)}*\eta)=+\infty.
\end{equation}

\item If
\[
 r\le\rho<\frac{\beta(\theta-1)}\theta,
\]
there are laws with the same moment-matching and tensor properties for
which
\begin{equation}\label{eq:stable-power-domain-sharpness}
 D_\theta(p_s^{(\alpha,d)}*\nu\|
          p_s^{(\alpha,d)}*\eta)=+\infty
 \qquad\hbox{for every }s>0.
\end{equation}
\end{enumerate}
For \(0<\rho<r\), the order-\(r\) tensor is not defined throughout the
moment class.  Thus the first term in
\eqref{eq:stable-power-optimal-moment-exponent} is the KL-type asymptotic
barrier, while the second is the stable-tail integrability barrier of the
power divergence.
\end{theorem}

\begin{proof}
Write \(q_0=q_{r,\alpha,d}\), and set
\[
 q=\max\{\theta,q_0\},
 \qquad p=p_{r,\alpha,d}^{(\theta)}.
\]
If \(q=q_0\), then \(p=2r/q_0\) and
\(p\ge\beta(1-1/q_0)\).  If \(q=\theta>q_0\), then
\(p=\beta(1-1/\theta)\) and \(p\theta=\beta(\theta-1)>2r\).
In both cases
\begin{equation}\label{eq:stable-power-balance-relations}
 p\ge r,\qquad p\ge\beta(1-1/q),\qquad pq\ge2r.
\end{equation}

Apply Lemma~\ref{lem:off-critical-stable-translation-remainder} to both
laws, with \(m=\lfloor p\rfloor\).  Moment matching gives
\[
 q_s^\nu-q_s^\eta=v_s+e_s,
 \qquad
 v_s=\sum_{j=r}^m s^{-j}(B_j^\nu-B_j^\eta),
 \qquad
 \|e_s\|_q=o(s^{-p}),
\]
and hence
\begin{equation}\label{eq:stable-power-leading-mode-convergence}
 v_s=s^{-r}B_{\Delta_r}+o(s^{-r})
 \quad\hbox{in }L^\infty(\Omega_{\alpha,d}).
\end{equation}
Put \(g_s=q_s^\eta\).  The bounded stable score and Jensen's inequality
give
\[
 g_s(y)\ge
 \int e^{-S_{\alpha,d}|x|/s}\eta(\dd x)
 \ge e^{-S_{\alpha,d}\int|x|\eta(\dd x)/s}=:c_s\longrightarrow1.
\]
Translation continuity and dominated convergence also give
\begin{equation}\label{eq:stable-power-background-convergence}
 \|g_s-1\|_{L^1(\Omega_{\alpha,d})}\longrightarrow0.
\end{equation}

For large \(s\), apply
Lemma~\ref{lem:stable-power-perturbation} pointwise to
\(g=g_s\), \(v=v_s\), and \(e=e_s\).  H\"older's inequality,
\eqref{eq:stable-power-balance-relations}, and
\eqref{eq:stable-power-leading-mode-convergence} yield
\begin{align*}
 &\left|D_\theta(q_s^\nu\Omega_{\alpha,d}\|
                   q_s^\eta\Omega_{\alpha,d})
 -\int\Psi_{g_s}(v_s)\dd\Omega_{\alpha,d}\right|\\
 &\qquad\le C\|v_s\|_{q/(q-1)}\|e_s\|_q
       +C\|e_s\|_q^q
 =o(s^{-r-p})+o(s^{-pq})=o(s^{-2r}).
\end{align*}
This bound also proves finiteness for all sufficiently large \(s\).  For a
fixed \(s>0\), the global translate estimate and Minkowski's inequality give
\[
 \|q_s^\lambda\|_{L^q(\Omega_{\alpha,d})}
 \le C_s\int(1+|x|)^p\lambda(\dd x)<\infty
 \qquad(\lambda\in\{\nu,\eta\}).
\]
Since \(q\ge\theta\), \(g_s\) has a positive uniform lower bound, and
\(\phi_\theta(t)\le C_\theta(1+t^\theta)\), this proves finiteness at every
scale.

Uniform Taylor expansion at zero, followed by
\eqref{eq:stable-power-background-convergence}, gives
\begin{align*}
 \int\Psi_{g_s}(v_s)\dd\Omega_{\alpha,d}
 &=\frac12\int\frac{v_s^2}{g_s}\dd\Omega_{\alpha,d}
   +O(\|v_s\|_\infty^3)\\
 &=\frac{s^{-2r}}2\int B_{\Delta_r}^2
       \dd\Omega_{\alpha,d}+o(s^{-2r}).
\end{align*}
Here \(g_s\ge c_s\to1\) implies
\(\|g_s^{-1}-1\|_1=o(1)\).  This proves
\eqref{eq:stable-power-first-unmatched-asymptotic}.

We prove the two sharpness assertions using the finite-difference blocks
of Lemma~\ref{lem:stable-finite-difference-blocks}.  The power divergence
obeys data processing by conditional Jensen's inequality.  Moreover, if
Bernoulli parameters satisfy \(u\ge Aw\) and \(v\le Bw\), where
\(0<B<A\), then
\begin{equation}\label{eq:stable-power-bernoulli-linear-bound}
 D_\theta(\operatorname{Ber}(u)\|
          \operatorname{Ber}(v))\ge c_{A,B,\theta}w.
\end{equation}
Indeed, both Bernoulli summands are nonnegative and
\(v\phi_\theta(u/v)=u\phi_\theta(u/v)/(u/v)\); the last quotient has a
positive infimum on \([A/B,\infty)\).  If instead \(u/v\) is sufficiently
large, the explicit power growth gives
\begin{equation}\label{eq:stable-power-bernoulli-power-bound}
 D_\theta(\operatorname{Ber}(u)\|
          \operatorname{Ber}(v))\ge c_\theta u^\theta v^{1-\theta}.
\end{equation}

For part~\textup{(i)}, the displayed interval can be nonempty only when
\(p_{r,\alpha,d}^{(\theta)}=p_{r,\alpha,d}>\max\{r,
\beta(\theta-1)/\theta\}\).  Use the laws and scales in the sharpness
proof of Theorem~\ref{thm:all-order-stable-first-unmatched-moment}:
\(w_j=a_jR_j^{-\rho}\) and
\(s_j=\kappa R_jw_j^{1/\beta}\).  The lower moment bound in part~\textup{(i)}
and the off-critical translate estimate with \(q=\theta\) make the power
divergence finite at every scale.  Lemma~\ref{lem:stable-moving-ball-probability-separation},
data processing, and \eqref{eq:stable-power-bernoulli-linear-bound} give
\[
 s_j^{2r}D_\theta(p_{s_j}^{(\alpha,d)}*\nu\|
                   p_{s_j}^{(\alpha,d)}*\eta)
 \ge cR_j^{2r}w_j^{1+2r/\beta}
 =ca_j^{q_0}R_j^{2r-\rho q_0}.
\]
The exponent is positive because \(\rho<p_{r,\alpha,d}=2r/q_0\).
Recursive enlargement of the radii proves
\eqref{eq:stable-power-analytic-sharpness}.

For part~\textup{(ii)}, take the same block laws with
\(w_j=a_jR_j^{-\rho}\), where \(a_j>0\) is summable, and choose the
radii recursively so fast that
\begin{equation}\label{eq:stable-power-domain-radii}
 a_j^\theta R_j^{\beta(\theta-1)-\theta\rho}\longrightarrow\infty.
\end{equation}
They have finite \(\rho\)-th moments, match below order \(r\), and have a
finite nonzero order-\(r\) tensor difference.  Fix \(s>0\), and let \(E_j\)
be the ball of radius \(s\) about the positive endpoint
\((r+1)R_je\).  The proof of
Lemma~\ref{lem:stable-moving-ball-probability-separation}, with this fixed
radius, gives for all large \(j\)
\[
 (p_s^{(\alpha,d)}*\nu)(E_j)\ge c_rw_j,
 \qquad
 (p_s^{(\alpha,d)}*\eta)(E_j)\le C_{s,\alpha,d,r}R_j^{-\beta}.
\]
Data processing and \eqref{eq:stable-power-bernoulli-power-bound} now give
\[
 D_\theta(p_s^{(\alpha,d)}*\nu\|
          p_s^{(\alpha,d)}*\eta)
 \ge C_s a_j^\theta
 R_j^{\beta(\theta-1)-\theta\rho}\longrightarrow\infty.
\]
This proves \eqref{eq:stable-power-domain-sharpness}.  The final assertion
below order \(r\) is the same tensor-domain obstruction as in
Theorem~\ref{thm:all-order-stable-first-unmatched-moment}.
\end{proof}
