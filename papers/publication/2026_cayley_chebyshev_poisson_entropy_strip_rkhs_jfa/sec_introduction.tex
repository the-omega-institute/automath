\section{Introduction}\label{sec:introduction}

The Cayley transform
\[
\iota(x)=\frac{1+\ii x}{1-\ii x}
\]
identifies the one-point compactification of \(\RR\) with \(\TT\), and pulls
normalized Haar measure back to the Cauchy law
\[
\omega(\dd x)=\frac{1}{\pi(1+x^{2})}\dd x=P_1(x)\dd x .
\]
This is the endpoint-normalized form of the classical Cauchy/wrapped-Cauchy
M{\"o}bius correspondence in circular statistics
\cite{KentTyler1988WrappedCauchy,McCullagh1992CauchyModels,
McCullagh1996CauchyMobius,
KatoJones2010MobiusCircle,KatoMcCullagh2020SphereCauchyMobius}.
This normalization makes the fixed-law Poisson comparison
\begin{equation}\label{eq:intro-main-functional}
D_{\rm KL}\bigl(P_t*\nu\,\|\,P_t(\cdot-\bar\gamma)\bigr)
\end{equation}
a finite coefficient problem in Cayley--Chebyshev modes; \(D_{\rm KL}\)
denotes Kullback--Leibler divergence
\cite{KullbackLeibler1951,CoverThomas2006}.  The input law
\(\nu\) is fixed, \(\bar\gamma\) is its mean, and \(t\to\infty\) is the
Poisson scale.

\subsection*{Main theorem statements}

The following theorem-level statements give the map of the paper.
They are stated with their operative assumptions and coefficients.
The multidimensional coefficients are denoted
\(\mathcal A_m^{(d)}(\nu)\); the one-dimensional coefficients use the
unsuperscripted symbols \(A_k(\nu)\).

\begin{theorem}[One-dimensional coefficient identification at finite \((2N-2)\)-moment level]%
\label{thm:intro-one-dimensional-universal-expansion}
Let \(N\ge2\), let \(\nu\) be a probability measure on \(\RR\) with mean
\(\bar\gamma\), and assume that \(X_c=\gamma-\bar\gamma\) has finite
\((2N-2)\)th absolute moment.  Put
\[
h_t=P_t*\nu,\qquad g_t(x)=P_t(x-\bar\gamma),\qquad
\mu_n=\int X_c^n\,\dd\nu(\gamma).
\]
Then
\begin{equation}\label{eq:intro-one-dimensional-transfer}
D_{\mathrm{KL}}\bigl(P_t*\nu\,\|\,P_t(\cdot-\bar\gamma)\bigr)
=
\sum_{m=2}^{N}A_{2m}(\nu)t^{-2m}+o(t^{-2N}).
\end{equation}
Each \(A_{2m}\) is the finite-moment evaluation of a universal polynomial in
the centred moments, computed by the Cayley mode integrals in
Theorem~\ref{thm:universal-entropy-coefficients}; all retained odd
coefficients vanish.  More explicitly, for \(4\le k\le2N\),
\[
A_k(\nu)=
\sum_{r=2}^{\lfloor k/2\rfloor}
\frac{(-1)^r}{r(r-1)}
\sum_{\substack{n_1+\cdots+n_r=k\\2\le n_j\le 2N-2}}
\mu_{n_1}\cdots\mu_{n_r}
\int_{\RR}u_{n_1}\cdots u_{n_r}\,\dd\omega .
\]
This is Theorem~\ref{thm:poisson-kl-even-orders}.
\end{theorem}

\begin{theorem}[Sharp leading tail threshold in dimension one]%
\label{thm:intro-sharp-leading-threshold}
Let \(\nu\) be a probability measure on \(\RR\) with finite mean
\(\bar\gamma\), and put
\[
h_t=P_t*\nu,\qquad g_t(x)=P_t(x-\bar\gamma),\qquad
H(t)=D_{\mathrm{KL}}(h_t\|g_t).
\]
If the centred variance \(\mu_2=\int(\gamma-\bar\gamma)^2\,\dd\nu(\gamma)\)
is finite, then
\[
\lim_{t\to\infty}t^4H(t)=\frac{\mu_2^2}{8}.
\]
If the centred variance is infinite, then
\[
\lim_{t\to\infty}t^4H(t)=+\infty.
\]
Equivalently, finite centred second moment is necessary and sufficient for a
finite leading \(t^{-4}\) matched-Poisson KL coefficient, and the finite value
is the universal coefficient \(A_4(\nu)=\mu_2^2/8\).  This is
Theorem~\ref{thm:sharp-leading-kl-variance-threshold}.
\end{theorem}

\begin{theorem}[Multidimensional finite-\(L\) Poisson entropy jet, \(L\ge d+1\)]%
\label{thm:intro-multidimensional-jet}
Let \(d\ge1\), let \(L\ge\max\{2,d+1\}\) be an integer, and let
\(\nu\) be a probability measure on \(\RR^d\) with mean \(\bar\gamma\) and
finite centred \(L\)th absolute moment.  Then
\[
D_{\rm KL}\bigl(P_t^{(d)}*\nu\,\|\,P_t^{(d)}(\cdot-\bar\gamma)\bigr)
=\sum_{m=4}^{L+2}\mathcal A_m^{(d)}(\nu)t^{-m}
+o(t^{-(L+2)}),
\]
where \(\mathcal A_m^{(d)}\) is given by the tensor formula
\[
\mathcal A_m^{(d)}(\nu)=
\sum_{r=2}^{\lfloor m/2\rfloor}
\frac{(-1)^r}{r(r-1)}
\sum_{\substack{|\alpha_1|,\ldots,|\alpha_r|\ge2\\
|\alpha_1|+\cdots+|\alpha_r|=m\\
|\alpha_j|\le L}}
\mu_{\alpha_1}\cdots\mu_{\alpha_r}
\int_{\RR^d}\prod_{j=1}^rU_{\alpha_j}^{(d)}(y)\,\Omega_d(\dd y).
\]
Here \(U_\alpha^{(d)}\) are the Taylor modes of
\[
\left(\frac{1+|y|^2}{1+|y-z|^2}\right)^{(d+1)/2},
\qquad
\mu_\alpha=\EE[X_c^\alpha],
\]
and \(\Omega_d=P_1^{(d)}(y)\dd y\).  This is
Theorem~\ref{thm:multidimensional-poisson-entropy-jets}.
\end{theorem}

\begin{proposition}[Formal multidimensional covariance layer under finite covariance]%
\label{prop:intro-multidimensional-formal-covariance-layer}
\label{thm:intro-multidimensional-leading-coefficient}
Let \(d\ge1\), and let \(\nu\) be a probability measure on \(\RR^d\) with
finite centred second moment.  Put
\[
X_c=\gamma-\bar\gamma,
\]
where \(\bar\gamma=\int_{\RR^d}\gamma\,\dd\nu(\gamma)\), and let
\[
\Sigma=\EE[X_cX_c^{\mathsf T}],
\qquad
\Sigma_0=\Sigma-d^{-1}(\operatorname{tr}\Sigma)I_d .
\]
Then the formal quadratic covariance layer obtained by inserting the
degree-two density mode into the quadratic term of the KL integrand is
\begin{equation}\label{eq:intro-multidim-leading}
A_{4,\mathrm{quad}}^{(d)}(\nu)
=
c_d^{\rm iso}(\operatorname{tr}\Sigma)^2
+c_d^{\rm tr}\|\Sigma_0\|_{\rm HS}^2,
\end{equation}
where
\[
c_d^{\rm iso}=
\frac{3(d+1)(7d+9)}
{4d(d+3)(d+5)(d+7)},
\qquad
c_d^{\rm tr}=
\frac{3(d+1)(d+3)}
{4(d+5)(d+7)}.
\]
The formula is positive and vanishes exactly when \(\Sigma=0\).  This is
Proposition~\ref{prop:multidim-formal-covariance-layer}.
\end{proposition}

\begin{corollary}[Leading multidimensional KL coefficient under a finite jet hypothesis]%
\label{cor:intro-multidimensional-leading-kl-coefficient}
Let \(d\ge1\), let \(L\ge\max\{2,d+1\}\), and assume that
\(\nu\) has mean \(\bar\gamma\), centred tensor moments
\(\mu_\alpha\) for \(2\le|\alpha|\le L\), and satisfies
\[
\begin{aligned}
(\mathrm{MPJ}_{d,L}):\qquad
\delta_t^{(d)}(y)
&=\sum_{2\le|\alpha|\le L}
\mu_\alpha U_\alpha^{(d)}(y)t^{-|\alpha|}
+\rho_{L,t}^{(d)}(y),\\
\|\rho_{L,t}^{(d)}\|_\infty+
\|\rho_{L,t}^{(d)}\|_{L^1(\Omega_d)}
&=o(t^{-L}),\qquad
\|\delta_t^{(d)}\|_\infty=O(t^{-2}).
\end{aligned}
\]
The finite centred \(L\)th absolute moment hypothesis in
Theorem~\ref{thm:intro-multidimensional-jet} is a sufficient condition for
this jet.
With \(A_{4,\mathrm{quad}}^{(d)}(\nu)\) as in
\eqref{eq:intro-multidim-leading},
\[
D_{\rm KL}\bigl(P_t^{(d)}*\nu\,\|\,P_t^{(d)}(\cdot-\bar\gamma)\bigr)
=A_{4,\mathrm{quad}}^{(d)}(\nu)t^{-4}+o(t^{-4}).
\]
\end{corollary}

\begin{theorem}[Poisson quotient KL tail identity]%
\label{thm:intro-kl-tail-identity}
Let \(\nu\) be a probability measure on \(\RR\) with mean \(\bar\gamma\) and
finite centred second moment.  For
\[
h_t=P_t*\nu,\qquad g_t(x)=P_t(x-\bar\gamma),\qquad u_t=h_t/g_t,
\]
define
\[
I_\nu(t)=
\frac1\pi\iint_{\RR^2}
\frac{g_t(x)\left(
u_t(x)\log\!\bigl(u_t(x)/u_t(y)\bigr)-u_t(x)+u_t(y)
\right)}
{(x-y)^2}\dd x\dd y .
\]
Then \(I_\nu\in L^1_{\rm loc}(0,\infty)\) and
\[
D_{\mathrm{KL}}\bigl(P_t*\nu\,\|\,P_t(\cdot-\bar\gamma)\bigr)
=\int_t^\infty I_\nu(s)\dd s,
\]
for every \(t>0\).  This is Theorem~\ref{thm:poisson-kl-doob-tail}.
\end{theorem}

\begin{theorem}[Explicit \(N=4\) sixth/eighth coefficient layer]%
\label{thm:intro-explicit-n4-layer}
Let \(\nu\) be a probability measure on \(\RR\) with mean \(\bar\gamma\),
finite variance \(\sigma^2\ge0\), and finite centred sixth absolute moment.
Then
Theorem~\ref{thm:poisson-kl-eighth} identifies
\[
D_{\mathrm{KL}}\bigl(P_t*\nu\,\|\,P_t(\cdot-\bar\gamma)\bigr)
=
\frac{\sigma^4}{8t^4}
+\frac{A_6(\nu)}{t^6}
+\frac{A_8(\nu)}{t^8}
+o(t^{-8}),
\]
where
\[
A_6(\nu)
=\frac{\sigma^6+6\mu_3^2-8\sigma^2\mu_4}{64},
\]
and
\[
A_8(\nu)
=
\frac{3\sigma^8-12\sigma^4\mu_4+9\sigma^2\mu_3^2
+12\sigma^2\mu_6-30\mu_3\mu_5+20\mu_4^2}{256}.
\]
If the centred seventh absolute moment is finite, then
Theorem~\ref{thm:poisson-kl-eighth-seventh-remainder} improves the last
remainder to \(O(t^{-9})\).  When \(\sigma^2=0\),
\(\nu=\delta_{\bar\gamma}\) and the expansion is the trivial identity
\(0=0\).  These are Theorems~\ref{thm:poisson-kl-eighth} and
\ref{thm:poisson-kl-eighth-seventh-remainder}.
\end{theorem}

\begin{corollary}[Coefficient defect stability at the fourth-moment layer]%
\label{cor:intro-defect-certificate}
Let \(\nu\) have mean \(\bar\gamma\), variance \(\sigma^2>0\), and finite
centred fourth absolute moment.  The \(N=3\) case of
Theorem~\ref{thm:intro-one-dimensional-universal-expansion} identifies
\[
A_6(\nu)=\frac{\sigma^6+6\mu_3^2-8\sigma^2\mu_4}{64}
\]
as the genuine \(t^{-6}\) coefficient.  Define the sixth-order defect
\[
\Delta_6(\nu)=-\frac{7}{64}\sigma^6-A_6(\nu).
\]
Corollary~\ref{cor:coefficient-defect-coupling} converts this fixed-variance
defect into a \(W_2\) bound against
\(\rho_\sigma=\frac12(\delta_{-\sigma}+\delta_\sigma)\).
\end{corollary}

\subsection*{Cayley coefficient mechanism}

After the rescaling \(x=\bar\gamma+ty\), a translate by \(X_c=\gamma-\bar\gamma\)
is encoded by
\[
R_\varepsilon(y)=\frac{1+y^2}{1+(y-\varepsilon)^2},
\qquad \varepsilon=X_c/t .
\]
The Cayley angle \(y=\tan\theta\) converts the Taylor modes of
\(R_\varepsilon\) into finite trigonometric, equivalently Laurent, modes.
Theorem~\ref{thm:cayley-chebyshev-mode} gives the mode formula,
Theorem~\ref{thm:universal-entropy-coefficients} defines the formal entropy
coefficients by finite Cauchy-weighted mode integrals, and
Theorem~\ref{thm:finite-fourier-mode-integral-algorithm} reduces those
integrals to Laurent constant-term extraction.

The analytic input is separate from the algebra.  The finite-moment quotient
remainder is proved in Theorem~\ref{thm:finite-moment-density-remainder} and
Lemma~\ref{lem:exact-moment-density-truncation}; the retained-order nonlinear
transfer through \((1+u)\log(1+u)-u\) is
Proposition~\ref{prop:density-to-entropy-descent} and
Theorem~\ref{prop:coefficient-identification-poisson-entropy-transfer}.  The
finite Laurent arithmetic needed for the explicit \(N=4\) layer is displayed
in Lemmas~\ref{lem:n4-normalized-laurent-representatives}--\ref{lem:n4-entropy-coefficient-substitution}.

\subsection*{Related entropy-asymptotic literature}

The closest entropy-asymptotic literature studies information functionals
along limiting or approximating families.  Barron's entropic central limit
theorem and its quantitative refinements
\cite{Barron1986,ArtsteinBallBartheNaor2004,JohnsonBarron2004,
MadimanBarron2007,BobkovChistyakovGotzeEdgeworth2013,
BobkovChistyakovGotzeBerryEsseen2014} concern normalized sums and
Gaussian-relative entropy; the stable-law and de Bruijn variants
\cite{BobkovChistyakovGotze2013,BobkovChistyakovGotzeStableFisher2014,
JohnsonStableDeBruijn2013,Toscani2015,Toscani2016} change the limiting
semigroup or attractor.  Discrete and Poisson-approximation analogues
\cite{KontoyiannisHarremoesJohnson2005,GavalakisKontoyiannis2024DiscreteCLT}
and direct Shannon/R\'enyi entropy expansions for classical discrete
families \cite{Knessl1998EntropyAsymptotics} are adjacent in method and
motivation.  The object here is different: the input law \(\nu\) is fixed,
the large parameter is the Cauchy/Poisson scale \(t\), and the expansion is
of the matched quotient
\[
D_{\rm KL}(P_t*\nu\|P_t(\cdot-\bar\gamma)),
\]
including finite-moment polynomial coefficients and explicit heavy-tail
replacement layers.
Entropy-method asymptotics for porous-medium and fast-diffusion flows form a
second adjacent literature.  In that setting relative entropy and entropy
dissipation measure convergence of a time-evolving solution toward a
self-similar Barenblatt profile or an equilibrium profile
\cite{CarrilloToscani2000PorousMedium,CarrilloJungelMarkowichToscaniUnterreiter2001EntropyDissipation,
DolbeaultToscani2011FastDiffusion,Vazquez2014NonlinearFractionalHeat}.  Here
there is no evolving input law and no entropy convergence theorem toward a
nonlinear diffusion attractor: the fixed law is smoothed once by the
Cauchy/Poisson kernel, and the question is the coefficient expansion of a
matched KL quotient in powers of \(t^{-1}\), including the moment-polynomial
coefficients and the logarithmic replacements when a borderline moment is
missing.

\subsection*{Organization}

Section~\ref{sec:preliminaries} fixes notation.  Sections~\ref{sec:cayley-gate}
and~\ref{sec:haar-pullback} record the Cayley and Haar-pullback
normalizations.  Section~\ref{sec:entropy-asymptotics} proves the
multidimensional entropy jets, the one-dimensional coefficient hierarchy, the
sharp leading variance threshold, the explicit sixth/eighth-order layer, and
the coefficient-defect consequence.
Section~\ref{sec:doob-phi-entropy}
proves the Poisson quotient KL tail identity by the cutoff Green/Bregman
calculation and then proves the dissipation-tail coefficient expansion.
Appendix~\ref{app:related-work-scope} gives related-work context;
Appendix~\ref{app:entropy-integrals} records the weighted trigonometric
normalization behind the Laurent arithmetic;
Appendix~\ref{app:haar-pullback-uniqueness}
records the endpoint-normalized Cayley uniqueness statement; and
Appendix~\ref{app:remarks-nonclaims} collects scope remarks and nonclaims,
including the status of conditional finite-time post-processing.
