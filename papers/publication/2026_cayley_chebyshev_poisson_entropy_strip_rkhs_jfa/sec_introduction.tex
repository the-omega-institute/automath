\section{Introduction}\label{sec:introduction}

The Cayley transform
\[
\iota(x)=\frac{1+\ii x}{1-\ii x}
\]
identifies the one-point compactification of \(\RR\) with \(\TT\), and pulls
normalized Haar measure back to the Cauchy law
\[
\omega(\dd x)=\frac{1}{\pi(1+x^{2})}\dd x=P_1(x)\dd x .
\]
This is the endpoint-normalized form of the classical Cauchy/wrapped-Cauchy
M{\"o}bius correspondence in circular statistics
\cite{KentTyler1988WrappedCauchy,McCullagh1992CauchyModels,
McCullagh1996CauchyMobius,
KatoJones2010MobiusCircle,KatoMcCullagh2020SphereCauchyMobius}.
This normalization makes the fixed-law Poisson comparison
\begin{equation}\label{eq:intro-main-functional}
D_{\rm KL}\bigl(P_t*\nu\,\|\,P_t(\cdot-\bar\gamma)\bigr)
\end{equation}
a finite coefficient problem in Cayley--Chebyshev modes; \(D_{\rm KL}\)
denotes Kullback--Leibler divergence
\cite{KullbackLeibler1951,CoverThomas2006}.  The input law
\(\nu\) is fixed, \(\bar\gamma\) is its mean, and \(t\to\infty\) is the
Poisson scale.

\subsection*{Main theorem statements}

The following theorem-level statements give the map of the paper.
They are stated with their operative assumptions and coefficients; proof
mechanisms and comparisons follow after the theorem block, while conditional
finite-time post-processing is kept outside the main manuscript in the
supplementary source files.  The multidimensional coefficients are denoted
\(\mathcal A_m^{(d)}(\nu)\) throughout; the one-dimensional coefficients use
the unsuperscripted symbols \(A_k(\nu)\).

The one-dimensional coefficient hierarchy is stated first because it is the
core mechanism of the paper.  The quotient identity and the finite-moment
density truncation are Lemma~\ref{lem:exact-moment-density-truncation}; the
nonlinear transfer through the KL integrand is
Theorem~\ref{thm:poisson-kl-even-orders}.  The leading \(A_4\) scale is
separated from the higher hierarchy because it has the sharp analytic variance
threshold in Theorem~\ref{thm:sharp-leading-kl-variance-threshold}; the higher
\(A_{2N}\) statements are coefficient-evaluability statements for the
displayed universal polynomials.

\begin{theorem}[One-dimensional coefficient identification at finite \((2N-2)\)-moment level]%
\label{thm:intro-one-dimensional-universal-expansion}
Let \(N\ge2\), let \(\nu\) be a probability measure on \(\RR\) with mean
\(\bar\gamma\), and assume that \(X_c=\gamma-\bar\gamma\) has finite
\((2N-2)\)th absolute moment.  Put
\[
h_t=P_t*\nu,\qquad g_t(x)=P_t(x-\bar\gamma),\qquad
\mu_n=\int X_c^n\,\dd\nu(\gamma).
\]
Then
\begin{equation}\label{eq:intro-one-dimensional-transfer}
D_{\mathrm{KL}}\bigl(P_t*\nu\,\|\,P_t(\cdot-\bar\gamma)\bigr)
=
\sum_{m=2}^{N}A_{2m}(\nu)t^{-2m}+o(t^{-2N}).
\end{equation}
Each \(A_{2m}\) is the finite-moment evaluation of a universal polynomial in
the centred moments, computed by the Cayley mode integrals in
Theorem~\ref{thm:universal-entropy-coefficients}; all retained odd
coefficients vanish.  More explicitly, for \(4\le k\le2N\),
\[
A_k(\nu)=
\sum_{r=2}^{\lfloor k/2\rfloor}
\frac{(-1)^r}{r(r-1)}
\sum_{\substack{n_1+\cdots+n_r=k\\2\le n_j\le 2N-2}}
\mu_{n_1}\cdots\mu_{n_r}
\int_{\RR}u_{n_1}\cdots u_{n_r}\,\dd\omega .
\]
This is Theorem~\ref{thm:poisson-kl-even-orders}; the boxed paragraph
following it fixes the coefficient-evaluability scope.  The algebraic
obstruction to evaluating the displayed top polynomial from lower moments is
isolated in
Proposition~\ref{prop:poisson-kl-moment-threshold-sharpness}, while
Proposition~\ref{prop:log-tail-counterexample-top-moment} gives explicit
power-tail examples in which the missing top moment is replaced by a
tail-dependent \(t^{-2N}\log t\) entropy layer.
\end{theorem}

\begin{theorem}[Sharp leading tail threshold in dimension one]%
\label{thm:intro-sharp-leading-threshold}
Let \(\nu\) be a probability measure on \(\RR\) with finite mean
\(\bar\gamma\), and put
\[
h_t=P_t*\nu,\qquad g_t(x)=P_t(x-\bar\gamma),\qquad
H(t)=D_{\mathrm{KL}}(h_t\|g_t).
\]
If the centred variance \(\mu_2=\int(\gamma-\bar\gamma)^2\,\dd\nu(\gamma)\)
is finite, then
\[
\lim_{t\to\infty}t^4H(t)=\frac{\mu_2^2}{8}.
\]
If the centred variance is infinite, then
\[
\lim_{t\to\infty}t^4H(t)=+\infty.
\]
Equivalently, finite centred second moment is necessary and sufficient for a
finite leading \(t^{-4}\) matched-Poisson KL coefficient, and the finite value
is the universal coefficient \(A_4(\nu)=\mu_2^2/8\).  This is
Theorem~\ref{thm:sharp-leading-kl-variance-threshold}.  It is the analytic
threshold result in the paper; the higher \(A_{2N}\) hierarchy is a family of
coefficient-evaluability statements for the displayed universal polynomials.
\end{theorem}

\begin{theorem}[Multidimensional finite-\(L\) Poisson entropy jet, \(L\ge d+1\)]%
\label{thm:intro-multidimensional-jet}
Let \(d\ge1\), let \(L\ge\max\{2,d+1\}\) be an integer, and let
\(\nu\) be a probability measure on \(\RR^d\) with mean \(\bar\gamma\) and
finite centred \(L\)th absolute moment.  Then
\[
D_{\rm KL}\bigl(P_t^{(d)}*\nu\,\|\,P_t^{(d)}(\cdot-\bar\gamma)\bigr)
=\sum_{m=4}^{L+2}\mathcal A_m^{(d)}(\nu)t^{-m}
+o(t^{-(L+2)}),
\]
where \(\mathcal A_m^{(d)}\) is given by the tensor formula
\[
\mathcal A_m^{(d)}(\nu)=
\sum_{r=2}^{\lfloor m/2\rfloor}
\frac{(-1)^r}{r(r-1)}
\sum_{\substack{|\alpha_1|,\ldots,|\alpha_r|\ge2\\
|\alpha_1|+\cdots+|\alpha_r|=m\\
|\alpha_j|\le L}}
\mu_{\alpha_1}\cdots\mu_{\alpha_r}
\int_{\RR^d}\prod_{j=1}^rU_{\alpha_j}^{(d)}(y)\,\Omega_d(\dd y).
\]
Here \(U_\alpha^{(d)}\) are the Taylor modes of
\[
\left(\frac{1+|y|^2}{1+|y-z|^2}\right)^{(d+1)/2},
\qquad
\mu_\alpha=\EE[X_c^\alpha],
\]
and \(\Omega_d=P_1^{(d)}(y)\dd y\).  Equivalently, the radial Poisson quotient
\((P_t^{(d)}*\nu)/P_t^{(d)}(\cdot-\bar\gamma)\) has the finite-moment
Taylor-mode expansion of
Theorem~\ref{thm:multidimensional-density-quotient}, and
Theorem~\ref{thm:multidimensional-poisson-entropy-jets} transfers it through
the KL integrand.
\end{theorem}

\begin{proposition}[Formal multidimensional covariance layer under finite covariance]%
\label{prop:intro-multidimensional-formal-covariance-layer}
\label{thm:intro-multidimensional-leading-coefficient}
Let \(d\ge1\), and let \(\nu\) be a probability measure on \(\RR^d\) with
finite centred second moment.  Put
\[
X_c=\gamma-\bar\gamma,
\]
where \(\bar\gamma=\int_{\RR^d}\gamma\,\dd\nu(\gamma)\), and let
\[
\Sigma=\EE[X_cX_c^{\mathsf T}],
\qquad
\Sigma_0=\Sigma-d^{-1}(\operatorname{tr}\Sigma)I_d .
\]
Then the formal quadratic covariance layer obtained by inserting the
degree-two density mode into the quadratic term of the KL integrand is
\begin{equation}\label{eq:intro-multidim-leading}
A_{4,\mathrm{quad}}^{(d)}(\nu)
=
c_d^{\rm iso}(\operatorname{tr}\Sigma)^2
+c_d^{\rm tr}\|\Sigma_0\|_{\rm HS}^2,
\end{equation}
where
\[
c_d^{\rm iso}=
\frac{3(d+1)(7d+9)}
{4d(d+3)(d+5)(d+7)},
\qquad
c_d^{\rm tr}=
\frac{3(d+1)(d+3)}
{4(d+5)(d+7)}.
\]
This is Proposition~\ref{prop:multidim-formal-covariance-layer}.  The coefficient
formula assumes only \(\EE|X_c|^2<\infty\); it is a covariance-layer identity
and depends on no higher tensor moments.  The formula is positive and vanishes
exactly when \(\Sigma=0\).  No KL asymptotic is asserted from finite covariance
alone.
\end{proposition}

\begin{corollary}[Leading multidimensional KL coefficient under \((\mathrm{MPJ}_{d,L})\)]%
\label{cor:intro-multidimensional-leading-kl-coefficient}
Let \(d\ge1\), let \(L\ge\max\{2,d+1\}\), and assume that
\((\nu,d,L)\) satisfies \((\mathrm{MPJ}_{d,L})\).  With
\(A_{4,\mathrm{quad}}^{(d)}(\nu)\) as in
\eqref{eq:intro-multidim-leading},
\[
D_{\rm KL}\bigl(P_t^{(d)}*\nu\,\|\,P_t^{(d)}(\cdot-\bar\gamma)\bigr)
=A_{4,\mathrm{quad}}^{(d)}(\nu)t^{-4}+o(t^{-4}).
\]
\end{corollary}

\begin{remark}[Multidimensional hypothesis barrier]
Theorem~\ref{thm:intro-multidimensional-jet},
Proposition~\ref{prop:intro-multidimensional-formal-covariance-layer}, and
Corollary~\ref{cor:intro-multidimensional-leading-kl-coefficient} have separate
hypothesis levels.  The finite-\(L\) theorem supplies the full entropy jet and
uses the dimension-dependent quotient-remainder hypothesis.  The proposition is
only the formal quadratic-layer calculation under finite covariance.  The
corollary is where the proposition's coefficient is inserted into the entropy
jet.  Thus neither a full finite jet nor the asymptotic statement
\(D_{\rm KL}=A_{4,\mathrm{quad}}^{(d)}t^{-4}+o(t^{-4})\) is inferred from
\(\EE|X_c|^2<\infty\); only the displayed covariance functional
\(A_{4,\mathrm{quad}}^{(d)}\) is a covariance-only identity.
\end{remark}

\begin{theorem}[Poisson quotient KL tail identity]%
\label{thm:intro-kl-tail-identity}
Let \(\nu\) be a probability measure on \(\RR\) with mean \(\bar\gamma\) and
finite centred second moment.  For
\[
h_t=P_t*\nu,\qquad g_t(x)=P_t(x-\bar\gamma),\qquad u_t=h_t/g_t,
\]
define
\[
I_\nu(t)=
\frac1\pi\iint_{\RR^2}
\frac{g_t(x)\left(
u_t(x)\log\!\bigl(u_t(x)/u_t(y)\bigr)-u_t(x)+u_t(y)
\right)}
{(x-y)^2}\dd x\dd y .
\]
Then \(I_\nu\in L^1_{\rm loc}(0,\infty)\) and
\[
D_{\mathrm{KL}}\bigl(P_t*\nu\,\|\,P_t(\cdot-\bar\gamma)\bigr)
=\int_t^\infty I_\nu(s)\dd s,
\]
for every \(t>0\).  This is Theorem~\ref{thm:poisson-kl-doob-tail}.  The
underlying Bregman differential identity is proved here from the moving
quotient domain and the Cauchy cutoff Green formula.  Lemma~
\ref{lem:poisson-kl-l1-ac-chain-rule} proves the
\(L^1\)-absolute-continuity chain rule for the KL path; Lemma~
\ref{lem:kl-tail-chain-rule-cutoff-passage} proves the bounded quotient
range, cutoff passage, and \(L^1\) Bregman dominator for
\(u_t=(P_t*\nu)/P_t(\cdot-\bar\gamma)\); Proposition~
\ref{prop:self-contained-cauchy-kl-hardystein} applies the two Green pairings
\((g_tu_t,\log u_t)\) and \((g_t,u_t)\) and the pointwise logarithmic
Bregman identity.  Consequently
Proposition~\ref{prop:dissipation-tail-coefficient-expansion} transfers the
already proved entropy coefficient expansion to the dissipation tail.
\end{theorem}

\begin{theorem}[Explicit \(N=4\) sixth/eighth coefficient layer]%
\label{thm:intro-explicit-n4-layer}
Let \(\nu\) be a probability measure on \(\RR\) with mean \(\bar\gamma\),
finite variance \(\sigma^2\ge0\), and finite centred sixth absolute moment.
Then
Theorem~\ref{thm:poisson-kl-eighth} identifies
\[
D_{\mathrm{KL}}\bigl(P_t*\nu\,\|\,P_t(\cdot-\bar\gamma)\bigr)
=
\frac{\sigma^4}{8t^4}
+\frac{A_6(\nu)}{t^6}
+\frac{A_8(\nu)}{t^8}
+o(t^{-8}),
\]
where
\[
A_6(\nu)
=\frac{\sigma^6+6\mu_3^2-8\sigma^2\mu_4}{64},
\]
and
\[
A_8(\nu)
=
\frac{3\sigma^8-12\sigma^4\mu_4+9\sigma^2\mu_3^2
+12\sigma^2\mu_6-30\mu_3\mu_5+20\mu_4^2}{256}.
\]
If the centred seventh absolute moment is finite, then
Theorem~\ref{thm:poisson-kl-eighth-seventh-remainder} improves the last
remainder to \(O(t^{-9})\).  The coefficient computation is the finite
zero-mode Laurent arithmetic displayed in the paper from the representatives
\[
Q_n(w)=\sum_{1\le |k|\le n}q_{n,k}w^k,\qquad
q_{n,k}=\lambda_n\,\operatorname{sgn}(k)^{\,n\bmod 2}
\binom{n-1}{|k|-1}
\]
from Lemma~\ref{lem:n4-normalized-laurent-representatives}, together with the
zero-sum convolution rule for Laurent constant terms.
When \(\sigma^2=0\), \(\nu=\delta_{\bar\gamma}\) and this is the trivial
identity \(0=0\); the positive-variance assumption enters only in the
normalized defect corollary below and the sixth/eighth-order defect
inequalities.
\end{theorem}

\begin{corollary}[Coefficient defect stability at the fourth-moment layer]%
\label{cor:intro-defect-certificate}
Let \(\nu\) have mean \(\bar\gamma\), variance \(\sigma^2>0\), and finite
centred fourth absolute moment.  The \(N=3\) case of
Theorem~\ref{thm:intro-one-dimensional-universal-expansion} identifies
\[
A_6(\nu)=\frac{\sigma^6+6\mu_3^2-8\sigma^2\mu_4}{64}
\]
as the genuine \(t^{-6}\) coefficient.  Define the sixth-order defect
\[
\Delta_6(\nu)=-\frac{7}{64}\sigma^6-A_6(\nu).
\]
Corollary~\ref{cor:coefficient-defect-coupling} converts this fixed-variance
defect into a \(W_2\) bound against
\(\rho_\sigma=\frac12(\delta_{-\sigma}+\delta_\sigma)\).  This is a
coefficient-level statement under the finite fourth absolute moment: no
finite-time KL value, eighth-order remainder bound, sixth absolute moment, or
approximation radius is part of the corollary.  The stronger \(N=4\) theorem
above is needed only for the \(A_8\) layer and for the seventh-moment
quantitative remainder.
\end{corollary}

\subsection*{Scope and nonclaims}

The results above concern the fixed-law matched Poisson KL functional
\eqref{eq:intro-main-functional}.  Theorem~\ref{thm:intro-sharp-leading-threshold}
is a genuine analytic iff threshold for the leading \(t^{-4}\) scale.  The
higher one-dimensional hierarchy is a coefficient-evaluability hierarchy in
the sense of the boxed paragraph after
Theorem~\ref{thm:poisson-kl-even-orders}.  Conditional finite-time
bookkeeping is not part of the main proof architecture and is kept in the
supplementary source files.

\subsection*{Cayley coefficient mechanism}

After the rescaling \(x=\bar\gamma+ty\), a translate by \(X_c=\gamma-\bar\gamma\)
is encoded by
\[
R_\varepsilon(y)=\frac{1+y^2}{1+(y-\varepsilon)^2},
\qquad \varepsilon=X_c/t .
\]
The Cayley angle \(y=\tan\theta\) converts the Taylor modes of
\(R_\varepsilon\) into finite trigonometric, equivalently Laurent, modes.
Theorem~\ref{thm:cayley-chebyshev-mode} gives the mode formula,
Theorem~\ref{thm:universal-entropy-coefficients} defines the formal entropy
coefficients by finite Cauchy-weighted mode integrals, and
Theorem~\ref{thm:finite-fourier-mode-integral-algorithm} reduces those
integrals to Laurent constant-term extraction.

The analytic input is separate from the algebra.  The finite-moment quotient
remainder is proved in Theorem~\ref{thm:finite-moment-density-remainder} and
Lemma~\ref{lem:exact-moment-density-truncation}; the retained-order nonlinear
transfer through \((1+u)\log(1+u)-u\) is
Proposition~\ref{prop:density-to-entropy-descent} and
Theorem~\ref{prop:coefficient-identification-poisson-entropy-transfer}.  The
finite Laurent arithmetic needed for the explicit \(N=4\) layer is displayed
in Lemmas~\ref{lem:n4-normalized-laurent-representatives}--\ref{lem:n4-entropy-coefficient-substitution}.

\subsection*{Position relative to existing work}

The comparison is organized by mathematical object, not by journal venue.
Each cited class below is labelled by what it actually studies: a fixed input
law, a changing-law CLT or stable-limit sequence, a linear heat or
fractional-heat profile, an abstract Bregman identity, or a
functional-inequality deficit.  This is important because the main
coefficients here belong to a fixed-law Cauchy/Poisson relative entropy
\[
D_{\rm KL}(P_t*\nu\|P_t(\cdot-\bar\gamma)),
\]
where the law \(\nu\) is not resampled, normalized, or sent through a CLT.
The Hardy--Stein literature supplies the dissipation identity; the
heat-profile and entropy-CLT literatures supply comparison techniques and
analogous asymptotic questions, but not the fixed-law Cauchy coefficient
calculation itself.  Appendix~\ref{app:related-work-scope} keeps the same
classification in compact form.

\medskip
\noindent\textbf{Fixed-law Cauchy/Poisson entropy asymptotics and Cauchy
information formulae.}

\noindent
The fixed-law object of this paper is
\(D_{\rm KL}(P_t*\nu\|P_t(\cdot-\bar\gamma))\) for a single input law
\(\nu\).  The finite formulas for KL or related divergences between
parametric Cauchy laws in
\cite{ChyzakNielsen2019CauchyKL,NielsenOkamura2023Cauchy,
BouhlelRousseau2022MultivariateCauchyKL,Verdu2023Cauchy} address fixed
Cauchy families, not a general convolution \(P_t*\nu\).  They therefore do
not contain the moment-dependent Cayley--Chebyshev expansion, the sharp
leading variance threshold, or the borderline \(t^{-2N}\log t\) layer for
non-Cauchy fixed input laws.

\medskip
\noindent\textbf{Stable-law entropy/de Bruijn identities and entropic CLT
expansions.}

\noindent
The Gaussian and stable entropy-production line
\cite{Barron1986,CarlenSoffer1991,JohnsonBarron2004,MadimanBarron2007,
JohnsonStableDeBruijn2013,Toscani2015,Toscani2016} addresses entropy
differentiation along Gaussian or stable smoothing semigroups.  The works of
Bobkov--Chistyakov--G{\"o}tze
\cite{BobkovChistyakovGotze2013,BobkovChistyakovGotzeStableFisher2014,
BobkovChistyakovGotzeEdgeworth2013,BobkovChistyakovGotzeBerryEsseen2014,
BobkovChistyakovGotzeFisher2014} and Bobkov--G{\"o}tze
\cite{BobkovGotzeRenyi2025,BobkovGotzeRenyiSurvey2025,
BobkovGotzeEsscher2025} address changing-law CLT, Edgeworth, stable-limit,
Fisher-information, and R\'enyi/Esscher asymptotics.  The common surface is
information asymptotics with explicit coefficients.  The separation is that
their laws vary with the normalization index, whereas here \(\nu\) is fixed
and the large parameter is the Poisson/Cauchy scale \(t\).

\medskip
\noindent\textbf{Heat/fractional-heat large-time moment expansions.}

\noindent
Duoandikoetxea--Zuazua \cite{DuoandikoetxeaZuazua1992},
Escobedo--Zuazua \cite{EscobedoZuazua1991},
Karch--Schonbek \cite{KarchSchonbek2001},
V\'azquez \cite{Vazquez2014NonlinearFractionalHeat},
Ishige--Kawakami--Michihisa
\cite{IshigeKawakamiMichihisa2017,IshigeKawakamiMichihisa2022RefinedFractional},
and Anker--Papageorgiou--Zhang
\cite{AnkerPapageorgiouZhang2023HeatSymmetricSpaces} address linear heat,
fractional-heat, nonlinear fractional diffusion, or symmetric-space heat
profile expansions.  Their object is an evolved profile or kernel in PDE
norms.  The present multidimensional theorem instead expands a fixed-law
Poisson quotient and then applies the nonlinear KL transfer; the covariance
coefficient is a beta-prime/spherical integral of a degree-two quotient mode.

\medskip
\noindent\textbf{Hardy--Stein/Bregman identities.}

\noindent
Ba{\~n}uelos--Bogdan--Luks
\cite{BanuelosBogdanLuks2016HardyStein},
Bogdan--Gutowski--Pietruska-Pa{\l}uba
\cite{BogdanGutowskiPietruskaPaluba2025PolarizedHardyStein},
Gutowski \cite{Gutowski2023HardySteinPureJump},
Gutowski--Kwa\'snicki
\cite{GutowskiKwasnicki2025BeurlingDenySobolevBregman}, and Chafa{\"i}
\cite{Chafai2004PhiEntropies} address Hardy--Stein, Sobolev--Bregman, and
\(\Phi\)-entropy identities for semigroups and Dirichlet forms.  This paper
uses that class as the identity input, but the proof in
Section~\ref{sec:doob-phi-entropy} also verifies the moving Cauchy quotient
hypotheses: generator normalization, \(L^1\)-absolute-continuity chain rule,
bounded positive quotient, cutoff convergence, Bregman domination, and the
tail limit \(H(t)\to0\).  The coefficient asymptotics are then supplied by
the Cayley expansion, not by the abstract Hardy--Stein theorem.

\medskip
\noindent\textbf{Stability/defect estimates.}

\noindent
The stability works
\cite{ChenLuTang2023LogSobolevStability,BrigatiDolbeaultSimonov2024LogSobolev,
Carlen2025LogHLS} and Poisson transport estimates such as
\cite{LopezRiveraShenfeld2025PoissonTransport} address
functional-inequality deficits, extremizer stability, or transport
constructions.  Corollary~\ref{cor:intro-defect-certificate} is different:
its defect is the explicit sixth Cauchy--Poisson coefficient
\(\Delta_6=-7\sigma^6/64-A_6\), its target is the symmetric two-point law,
and any finite-time use is conditional bookkeeping after independent
coefficient, remainder, and KL error inputs are supplied.

\subsection*{Organization}

Section~\ref{sec:preliminaries} fixes notation.  Sections~\ref{sec:cayley-gate}
and~\ref{sec:haar-pullback} record the Cayley and Haar-pullback
normalizations.  Section~\ref{sec:entropy-asymptotics} proves the
multidimensional entropy jets, the one-dimensional coefficient hierarchy, the
sharp leading variance threshold, the explicit sixth/eighth-order layer, and
the coefficient-defect consequence.
