\begin{corollary}[A sufficient moment condition for defect vanishing]
\label{cor:arbitrary-tail-defect-positive-moment-condition}
Under the hypotheses of
Corollary~\ref{thm:stable-arbitrary-order-law-by-law-decomposition}, suppose in
addition that \(\Delta_r\ne0\) and
\begin{equation}\label{eq:arbitrary-tail-defect-positive-moment-condition}
 \int_{\RR^d}|x|^{p_{r,\alpha,d}}(\nu+\eta)(\dd x)<\infty.
\end{equation}
Then
\begin{equation}\label{eq:arbitrary-tail-defect-positive-limit}
 s^{2r}\mathcal E_{r,s}(\nu,\eta)\longrightarrow0.
\end{equation}
This condition includes noncompact pairs of laws.
\end{corollary}

\begin{proof}
Theorem~\ref{thm:all-order-stable-first-unmatched-moment} and
\eqref{eq:arbitrary-tail-defect-decomposition} give, respectively,
\begin{align*}
 D_{\rm KL}(p_s^{(\alpha,d)}*\nu\|p_s^{(\alpha,d)}*\eta)
 &=\mathcal Q_{\alpha,d,r}(\Delta_r)s^{-2r}+o(s^{-2r}),\\
 D_{\rm KL}(p_s^{(\alpha,d)}*\nu\|p_s^{(\alpha,d)}*\eta)
 &=\mathcal Q_{\alpha,d,r}(\Delta_r)s^{-2r}
   +\mathcal E_{r,s}(\nu,\eta)+o(s^{-2r}).
\end{align*}
Subtracting proves \eqref{eq:arbitrary-tail-defect-positive-limit}.

For a concrete noncompact class, use the finite-support probability blocks
from Lemma~\ref{lem:stable-finite-difference-blocks}.  Convolve \(A_R^+\) and
\(A_R^-\) with the same nondegenerate Gaussian law.  The multinomial formula
shows that the convolved laws still match below order \(r\), while their
order-\(r\) difference is the nonzero difference of the two blocks: every
other term contains a block-moment difference of order below \(r\).  Both
convolved laws are noncompact and have moments of every order, so
\eqref{eq:arbitrary-tail-defect-positive-moment-condition} applies.
\end{proof}

\begin{corollary}[Non-vacuousness of the tail-defect criterion]
\label{cor:arbitrary-tail-defect-nonvacuousness}
If \(d+\alpha>2r\), then for every
\(r\le\rho<p_{r,\alpha,d}\) there are probability laws \(\nu,\eta\) with
finite \(\rho\)-th absolute moments, matching every tensor moment below
order \(r\), and having a finite nonzero order-\(r\) tensor difference, for
which
\begin{equation}\label{eq:arbitrary-tail-defect-unbounded-limsup}
 \limsup_{s\to\infty}s^{2r}\mathcal E_{r,s}(\nu,\eta)=+\infty.
\end{equation}
In particular, vanishing of the explicit nonnegative tail-defect energy is
not automatic under the minimal finite \(r\)-th moment hypothesis.
\end{corollary}

\begin{proof}
Use the laws in the sharpness assertion
\eqref{eq:all-order-stable-sharpness}.  They satisfy the hypotheses of the
law-by-law decomposition because \(\rho\ge r\).  Multiplying
\eqref{eq:arbitrary-tail-defect-decomposition} by \(s^{2r}\) gives
\[
 s^{2r}D_{\rm KL}(p_s^{(\alpha,d)}*\nu\|p_s^{(\alpha,d)}*\eta)
 =\mathcal Q_{\alpha,d,r}(\Delta_r)
  +s^{2r}\mathcal E_{r,s}(\nu,\eta)+o(1).
\]
The quadratic term is finite and independent of \(s\), whereas the left
side has infinite limsup by \eqref{eq:all-order-stable-sharpness}.  This
forces \eqref{eq:arbitrary-tail-defect-unbounded-limsup}.
\end{proof}
