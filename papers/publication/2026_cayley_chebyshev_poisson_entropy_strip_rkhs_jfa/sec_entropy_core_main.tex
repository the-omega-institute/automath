\section{Poisson Smoothing and Entropy Asymptotics}%
\label{sec:entropy-asymptotics}

This section contains the entropy coefficient mechanism, the multidimensional
covariance calculation, the leading one-dimensional threshold, and the
coefficient-defect consequence.  The proofs below are internal to the stated
lemmas and theorems; they do not require auxiliary finite-grid data or
external certificate files.  Pointwise dissipation asymptotics are not part of
this coefficient hierarchy; the only retained statement in that direction is
the short differentiated-remainder remark immediately before
Proposition~\ref{prop:dissipation-tail-coefficient-expansion}.

\subsection*{Entropy proof architecture and hypotheses}
\label{sec:entropy-theorem-map}

The hypotheses used below are separated by theorem rather than inferred from
neighbouring statements.  Probability-law and matched-barycentre assumptions
are repeated in the statements that use them, and the extended KL convention
is fixed in \eqref{eq:extended-kl-definition}.  The finite centred
\((2N-2)\)th absolute moment in
Theorem~\ref{thm:poisson-kl-even-orders} is the level needed to evaluate the
retained coefficient polynomial.  The finite side of
Theorem~\ref{thm:sharp-leading-kl-variance-threshold} and the dissipation
identity in Theorem~\ref{thm:poisson-kl-doob-tail} assume finite centred
variance; the infinite-variance side is read in the extended KL sense.  The
tail replacements are split into the principal conditional quotient-transfer
proposition and three verified subclass results: exact power tail,
perturbative power tail, and monotone slowly varying borderline tail.  Each
subclass statement repeats the quotient expansion or quotient controls used in
that result.  In several
dimensions, finite covariance is used only to evaluate the algebraic
quadratic layer; KL asymptotics use \((\mathrm{MPJ}_{d,L})\), with finite
centred \(L\)th absolute moment as a sufficient condition.

\subsection{Cayley modes and coefficient algebra}

Throughout the one-dimensional part,
\[
  \omega(\dd y)=\frac{\dd y}{\pi(1+y^2)}.
\]
For a law with mean \(\bar{\gamma}\), write
\[
  X_c=\gamma-\bar{\gamma},\qquad
  \mu_n=\int_{\RR} X_c^n\,\dd\nu(\gamma)
\]
whenever the moment exists.  For \(y\in\RR\) and \(\varepsilon\in\RR\), set
\[
R_\varepsilon(y):=\frac{1+y^2}{1+(y-\varepsilon)^2}.
\]
Let \(u_n(y)\) be the Taylor coefficient of \(R_\varepsilon(y)\) at
\(\varepsilon=0\):
\[
  R_\varepsilon(y)\sim\sum_{n\ge0}u_n(y)\varepsilon^n .
\]
\begin{lemma}[Uniform one-dimensional Cayley Taylor remainder]%
\label{lem:uniform-one-dimensional-cayley-taylor-remainder}
Let \(L\ge1\).  The Taylor coefficients of \(R_\varepsilon(y)\) at
\(\varepsilon=0\) satisfy
\[
  R_\varepsilon(y)=\sum_{n=0}^{L}u_n(y)\varepsilon^n
  +\varepsilon^{L+1}v_L(y,\varepsilon)
\]
for all \(y,\varepsilon\in\RR\), where \(v_L\) is a rational function on
\(\RR^2\) and
\begin{equation}\label{eq:uniform-vL-one-dimensional-bound}
  \sup_{y\in\RR,\ |\varepsilon|\le1}|v_L(y,\varepsilon)|
  \le \sup_{y,\varepsilon\in\RR}|v_L(y,\varepsilon)|
  \le \frac{5L+7}{2}.
\end{equation}
Moreover
\begin{equation}\label{eq:uniform-un-one-dimensional-bound}
  \|u_n\|_{L^\infty(\RR)}\le n+1,\qquad 0\le n\le L+1 .
\end{equation}
\end{lemma}

\begin{proof}
Write \(y=\tan\theta\), \(c=\cos\theta\), and \(s=\sin\theta\).  Then
\[
R_\varepsilon(\tan\theta)
=\{1-2(\varepsilon c)s+(\varepsilon c)^2\}^{-1}.
\]
The finite Chebyshev identity for the Chebyshev polynomials \(U_n\) of the
second kind,
\[
\sum_{n=0}^{L}U_n(x)r^n
=\frac{1-r^{L+1}U_{L+1}(x)+r^{L+2}U_L(x)}
       {1-2xr+r^2}
\]
with \(x=s\) and \(r=\varepsilon c\) gives
\[
  u_n(\tan\theta)=c^nU_n(s)
\]
and
\[
v_L(\tan\theta,\varepsilon)
=\frac{c^{L+1}U_{L+1}(s)-\varepsilon c^{L+2}U_L(s)}
       {1-2(\varepsilon c)s+(\varepsilon c)^2}.
\]
Since
\[
1-2(\varepsilon c)s+(\varepsilon c)^2
=\frac{1+(\tan\theta-\varepsilon)^2}{1+\tan^2\theta},
\]
this is the rational function
\[
v_L(y,\varepsilon)
=\frac{(1+y^2)u_{L+1}(y)-\varepsilon u_L(y)}
       {1+(y-\varepsilon)^2}.
\]
It remains only to prove that this function is uniformly bounded on the whole
\((y,\varepsilon)\)-plane.  Put \(d=\tan\theta-\varepsilon\).  Since
\(\varepsilon c=s-dc\), the preceding trigonometric expression becomes
\[
v_L(\tan\theta,\varepsilon)
=\frac{c^{L-1}\{U_{L+1}(s)-sU_L(s)+dc\,U_L(s)\}}
       {1+d^2}.
\]
For \(|s|\le1\), \(\sup |U_L(s)|=L+1\) and
\(\sup |U_{L+1}(s)|=L+2\).  Hence, using
\(|c|\le1\) and \(|d|/(1+d^2)\le1/2\),
\[
|v_L(\tan\theta,\varepsilon)|
\le (L+2)+(L+1)+\frac{L+1}{2}
=\frac{5L+7}{2}.
\]
This proves \eqref{eq:uniform-vL-one-dimensional-bound}.  Finally,
\[
  |u_n(\tan\theta)|\le |c|^n\sup_{|x|\le1}|U_n(x)|\le n+1,
\]
which proves \eqref{eq:uniform-un-one-dimensional-bound}.
\end{proof}

The following derivative form is the uniform Taylor input used in
\eqref{eq:one-dimensional-two-region-taylor-remainder}.  It is recorded
separately because it is independent of the Chebyshev representation.

\begin{lemma}[Uniform derivative bound for the Cayley quotient]%
\label{lem:uniform-cayley-derivative-bound}
For every \(L\ge2\),
\[
\sup_{y\in\RR,\ |\varepsilon|\le1}
\left|
\partial_\varepsilon^{L+1}
\frac{1+y^2}{1+(y-\varepsilon)^2}
\right|<\infty .
\]
\end{lemma}

\begin{proof}
For each \(k\ge0\), repeated differentiation of
\((1+(y-\varepsilon)^2)^{-1}\) gives
\[
\partial_\varepsilon^k
\frac{1}{1+(y-\varepsilon)^2}
=
\frac{P_k(y-\varepsilon)}
{(1+(y-\varepsilon)^2)^{k+1}},
\]
where \(P_k\) is a polynomial of degree at most \(k\).  When
\(|\varepsilon|\le1\),
\[
1+y^2\le C\bigl(1+(y-\varepsilon)^2\bigr)
\]
with an absolute constant \(C\).  Hence, with \(z=y-\varepsilon\),
\[
(1+y^2)
\frac{|P_k(y-\varepsilon)|}{(1+(y-\varepsilon)^2)^{k+1}}
\le
C_k\frac{1+|z|^{k+2}}{(1+z^2)^{k+1}}
\le C_k .
\]
Taking \(k=L+1\) proves the claim.
\end{proof}

Thus, for each \(L\ge1\),
\[
R_\varepsilon(y)=\sum_{n=0}^{L}u_n(y)\varepsilon^n
 +\varepsilon^{L+1}v_L(y,\varepsilon)
\]
with \(v_L\) as in
Lemma~\ref{lem:uniform-one-dimensional-cayley-taylor-remainder}.  The
finite-order estimate used below is the
following two-region form: for every \(L\ge2\) there is a constant \(C_L\)
such that, uniformly in \(y\in\RR\),
\begin{equation}\label{eq:one-dimensional-two-region-taylor-remainder}
\left|R_\varepsilon(y)-\sum_{n=0}^{L}u_n(y)\varepsilon^n\right|
\le C_L
\begin{cases}
|\varepsilon|^{L+1},& |\varepsilon|\le1,\\
|\varepsilon|^L,& |\varepsilon|>1.
\end{cases}
\end{equation}
Indeed, Lemma~\ref{lem:uniform-cayley-derivative-bound} and Taylor's formula
give the first line; equivalently the explicit remainder
Lemma~\ref{lem:uniform-one-dimensional-cayley-taylor-remainder}
gives it with constant \((5L+7)/2\).  For
\(|\varepsilon|>1\), the elementary bound
\[
R_\varepsilon(y)\le3(1+\varepsilon^2)
\]
and \(\|u_n\|_\infty\le n+1\) give
\[
\left|R_\varepsilon(y)-\sum_{n=0}^{L}u_n(y)\varepsilon^n\right|
\le
\left(6+\sum_{n=0}^{L}(n+1)\right)|\varepsilon|^L .
\]
Thus \eqref{eq:one-dimensional-two-region-taylor-remainder} holds with a
constant depending only on \(L\), uniformly in \(y\).

We use \(\theta\) for the half-angle Cayley coordinate \(y=\tan\theta\) and
\(z=e^{2\ii\theta}\) for the Laurent phase.  Thus
\(\dd\omega(y)=\dd y/(\pi(1+y^2))=\dd\theta/\pi\), and
\(\cos(2k\theta)\), \(\sin(2k\theta)\) become \(z^{\pm k}\) modes.  The
Laurent representative of \(u_n\) is denoted \(Q_n\), so
\(Q_n(e^{2\ii\theta})=u_n(\tan\theta)\).  Mode integrals are written
\(J(n_1,\ldots,n_r)\); the dissipation is always
\(\mathcal I_\nu(t)\).  The one-dimensional universal coefficient polynomial
is \(A_k^{\rm form}\), its evaluation on \(\nu\) is \(A_k(\nu)\), and
multidimensional retained coefficients are denoted
\(\mathcal A_m^{(d)}(\nu)\).  The separate symbol
\(\mathcal Q_d(\Sigma)\) refers only to the algebraic covariance quadratic layer,
which equals \(\mathcal A_4^{(d)}(\nu)\) after a quotient jet has identified
the KL expansion.

\begin{theorem}[Cayley--Chebyshev mode formula]%
\label{thm:cayley-chebyshev-mode}
Let \(U_n\) be the Chebyshev polynomial of the second kind.  If
\(y=\tan\theta\), then
\begin{equation}\label{eq:cayley-chebyshev-mode}
u_n(\tan\theta)=\cos^n\theta\,U_n(\sin\theta).
\end{equation}
Equivalently, for \(n\ge1\),
\[
u_n(\tan\theta)=
\cos^{n-1}\theta\,
\sin\!\left((n+1)\left(\frac{\pi}{2}-\theta\right)\right),
\]
and therefore
\begin{equation}\label{eq:cayley-mode-parity}
u_n(-y)=(-1)^nu_n(y).
\end{equation}
\end{theorem}

\begin{proof}
With \(y=\tan\theta\),
\[
R_\varepsilon(\tan\theta)
=\{1-2(\varepsilon\cos\theta)\sin\theta
+(\varepsilon\cos\theta)^2\}^{-1}.
\]
The generating function
\(\sum_{n\ge0}U_n(x)r^n=(1-2xr+r^2)^{-1}\)
\cite[Chapters~1 and~5]{MasonHandscomb2002} gives
\eqref{eq:cayley-chebyshev-mode}.  The trigonometric form follows from the
standard identity for \(U_n(\sin\theta)\), and parity is immediate.
\end{proof}

For \(r\ge2\) and \(n_j\ge2\), define
\begin{equation}\label{eq:mode-integral-definition}
J(n_1,\ldots,n_r):=
\int_{\RR}u_{n_1}(y)\cdots u_{n_r}(y)\,\omega(\dd y).
\end{equation}
In the formal polynomial algebra
\(\QQ[\mathfrak m_2,\ldots,\mathfrak m_k]\), put
\begin{equation}\label{eq:universal-entropy-coefficient}
A_k^{\rm form}:=
\sum_{r=2}^{\lfloor k/2\rfloor}
\frac{(-1)^r}{r(r-1)}
\sum_{\substack{n_1+\cdots+n_r=k\\ n_j\ge2}}
\mathfrak m_{n_1}\cdots\mathfrak m_{n_r}\,J(n_1,\ldots,n_r).
\end{equation}
When the required centred moments exist,
\begin{equation}\label{eq:universal-entropy-coefficient-evaluation}
A_k(\nu):=\operatorname{ev}_{\nu,k}(A_k^{\rm form}),
\qquad
\operatorname{ev}_{\nu,k}(\mathfrak m_n)=\mu_n .
\end{equation}

\begin{theorem}[Universal entropy coefficients]%
\label{thm:universal-entropy-coefficients}
The coefficient of \(t^{-k}\) in the formal expression
\[
\int_{\RR}\left\{
\left(1+\sum_{n=2}^{k}\mathfrak m_nu_n(y)t^{-n}\right)
\log\left(1+\sum_{n=2}^{k}\mathfrak m_nu_n(y)t^{-n}\right)
-\sum_{n=2}^{k}\mathfrak m_nu_n(y)t^{-n}
\right\}\omega(\dd y)
\]
is \(A_k^{\rm form}\).  If the moments needed in
\eqref{eq:universal-entropy-coefficient} exist for \(\nu\), then
\[
A_k(\nu)=
\sum_{r=2}^{\lfloor k/2\rfloor}
\frac{(-1)^r}{r(r-1)}
\sum_{\substack{n_1+\cdots+n_r=k\\ n_j\ge2}}
\mu_{n_1}\cdots\mu_{n_r}\,J(n_1,\ldots,n_r).
\]
All odd formal coefficients vanish.
\end{theorem}

\begin{proof}
Use
\[
(1+s)\log(1+s)-s=\sum_{r=2}^{\infty}\frac{(-1)^r}{r(r-1)}s^r.
\]
The coefficient of total degree \(k\) is finite and is exactly
\eqref{eq:universal-entropy-coefficient}.  If \(k\) is odd, then each mode
product in the coefficient has odd total parity by
\eqref{eq:cayley-mode-parity}, hence integrates to zero against the even
measure \(\omega\).
\end{proof}

\begin{theorem}[Finite Fourier algorithm for mode integrals]%
\label{thm:finite-fourier-mode-integral-algorithm}
Define \(Q_n(e^{2\ii\theta})=u_n(\tan\theta)\).  Equivalently, for every
\(n\ge1\),
\begin{equation}\label{eq:mode-laurent-polynomial-definition}
Q_n(z)=2^{-n}(-\ii)^n
\sum_{k=1}^{n}\binom{n-1}{k-1}
\bigl(z^k+(-1)^nz^{-k}\bigr).
\end{equation}
Thus, if \(Q_n(z)=\sum_kq_{n,k}z^k\), then \(q_{n,0}=0\) and
\begin{equation}\label{eq:mode-laurent-coefficients}
q_{n,k}=
2^{-n}(-\ii)^n\operatorname{sgn}(k)^{\,n\bmod2}
\binom{n-1}{|k|-1},
\qquad 0<|k|\le n,
\end{equation}
with \(q_{n,k}=0\) for \(|k|>n\).
Consequently
\begin{equation}\label{eq:mode-integral-constant-term}
J(n_1,\ldots,n_r)=[z^0]\prod_{j=1}^r Q_{n_j}(z).
\end{equation}
\end{theorem}

\begin{proof}
Put \(w=e^{\ii\theta}\) and \(z=w^2\).  The trigonometric form in
Theorem~\ref{thm:cayley-chebyshev-mode} gives
\[
u_n(\tan\theta)
=\frac{\cos^{n-1}\theta}{2\ii}
\left\{\ii^{\,n+1}w^{-(n+1)}
-(-\ii)^{n+1}w^{n+1}\right\}.
\]
Since
\(\cos^{n-1}\theta=2^{-(n-1)}w^{-(n-1)}(1+w^2)^{n-1}\), this becomes
\[
2^{-n}(1+z)^{n-1}
\left\{\ii^nz^{-n}+(-\ii)^nz\right\}.
\]
Expanding \((1+z)^{n-1}\) and reindexing the negative powers gives
\eqref{eq:mode-laurent-polynomial-definition}; the coefficient formula
\eqref{eq:mode-laurent-coefficients} is the same identity read termwise.
Since \(\omega(\dd y)=\dd\theta/\pi\), the map \(z=e^{2\ii\theta}\) sends
the integral to Haar integration on \(\TT\), hence to constant-term
extraction.
\end{proof}

\subsection{One-dimensional coefficient theorem}

\begin{lemma}[Exact-moment density truncation]%
\label{lem:exact-moment-density-truncation}
Let \(L\ge2\), and let \(\nu\) have mean \(\bar{\gamma}\) and finite centred
\(L\)th absolute moment.  For
\[
\delta_t(y):=
\frac{(P_t*\nu)(\bar{\gamma}+ty)}{P_t(ty)}-1
\]
one has
\begin{equation}\label{eq:exact-moment-density-truncation}
\delta_t(y)=\sum_{n=2}^{L}\mu_nu_n(y)t^{-n}+\rho_{L,t}(y),
\qquad
\|\rho_{L,t}\|_{\infty}+\|\rho_{L,t}\|_{L^1(\omega)}=o(t^{-L}).
\end{equation}
Also \(\|\delta_t\|_\infty=O(t^{-2})\).
\end{lemma}

\begin{proof}
For \(x=\bar{\gamma}+ty\),
\[
\frac{P_t(x-\gamma)}{P_t(x-\bar{\gamma})}
=R_{X_c/t}(y).
\]
Subtract the Taylor polynomial in \(X_c/t\) through order \(L\).  On
\(|X_c|\le t\), the small-translation part of
\eqref{eq:one-dimensional-two-region-taylor-remainder} gives
\[
C_Lt^{-L}\EE\!\left[|X_c|^L(|X_c|/t)\mathbf1_{\{|X_c|\le t\}}\right]
=o(t^{-L}).
\]
On \(|X_c|>t\), the large-translation part gives
\[
\left|R_{X_c/t}(y)-\sum_{n=0}^{L}u_n(y)(X_c/t)^n\right|
\le C_L|X_c/t|^L,
\]
and hence \(C_Lt^{-L}\EE[|X_c|^L\mathbf1_{\{|X_c|>t\}}]=o(t^{-L})\).  The same
uniform estimates prove both the \(L^\infty\) and \(L^1(\omega)\) bounds.  The
\(n=0\) term is removed by subtracting \(1\), and the \(n=1\) term vanishes
because \(\EE X_c=0\).  The \(O(t^{-2})\) bound follows by applying the two
displayed estimates with \(L=2\): the small-translation part is bounded by
\(Ct^{-2}\EE[|X_c|^2\mathbf1_{\{|X_c|\le t\}}]\), and the
large-translation part by
\(Ct^{-2}\EE[|X_c|^2\mathbf1_{\{|X_c|>t\}}]\).  These two bounds are
uniform in \(y\) and hence also hold in \(L^1(\omega)\).
\end{proof}

\begin{proposition}[Finite quotient jet implies KL coefficient jet]%
\label{prop:finite-quotient-jet-implies-kl-coefficient-jet}
Let \((E,\mathcal E,\lambda)\) be a probability space, let \(J\ge2\), and
let \(b_2,\ldots,b_J\in L^\infty(\lambda)\).  Put
\[
  q_t=\sum_{j=2}^{J}b_jt^{-j},\qquad
  \delta_t=q_t+\rho_t,
\]
and assume
\[
  \|\delta_t\|_\infty=O(t^{-2}),\qquad
  \|\rho_t\|_\infty+\|\rho_t\|_{L^1(\lambda)}=o(t^{-J}).
\]
Then, with \(\Phi(s)=(1+s)\log(1+s)-s\),
\begin{equation}\label{eq:finite-quotient-jet-transfer}
  \int_E\Phi(\delta_t)\,\dd\lambda
  =
  \sum_{m=4}^{J+2}C_m t^{-m}+o(t^{-(J+2)}),
\end{equation}
where
\begin{equation}\label{eq:finite-quotient-jet-coefficients}
  C_m=
  \sum_{r=2}^{\lfloor m/2\rfloor}
  \frac{(-1)^r}{r(r-1)}
  \sum_{\substack{j_1+\cdots+j_r=m\\2\le j_1,\ldots,j_r\le J}}
  \int_E b_{j_1}\cdots b_{j_r}\,\dd\lambda .
\end{equation}
If a symmetry of \(\lambda\) and the functions \(b_j\) makes the integrals in
\eqref{eq:finite-quotient-jet-coefficients} vanish for odd \(m\), then the
corresponding odd coefficients in \eqref{eq:finite-quotient-jet-transfer}
vanish.

More generally, let \(s_t\downarrow0\) be a comparison scale.  Suppose
\[
  \delta_t=a_t+B_t+e_t,\qquad
  a_t=\sum_{j=2}^{J}b_jt^{-j},
\]
\[
  \|a_t\|_\infty=O(t^{-2}),\qquad
  \|B_t\|_\infty+\|e_t\|_\infty=o(t^{-2}),\qquad
  \|e_t\|_{L^1(\lambda)}=o(s_t/t^{-2}).
\]
If, for some fixed Taylor order \(R_0\), the finite polynomial
\[
  \sum_{r=2}^{R_0}\frac{(-1)^r}{r(r-1)}(a_t+B_t)^r
\]
has only finitely many monomial contributions at scale \(s_t\), then
\(\int_E\Phi(\delta_t)\,\dd\lambda\) has exactly those finite monomial
contributions at scale \(s_t\); every monomial containing \(e_t\) is
\(o(s_t)\), and the Taylor remainder is \(o(s_t)\) provided
\(\|\delta_t\|_\infty^{R_0}\|\delta_t\|_{L^1(\lambda)}=o(s_t)\).
\end{proposition}

\begin{proof}
The first part is the finite algebraic transfer used in the finite-jet
theorems.  Since \(\|\delta_t\|_\infty=O(t^{-2})\), for all large \(t\)
\[
\Phi(\delta_t)=
\sum_{r=2}^{R}\frac{(-1)^r}{r(r-1)}\delta_t^r
+O(|\delta_t|^{R+1}),
\qquad
R=\left\lfloor\frac{J+2}{2}\right\rfloor+1 .
\]
The integrated Taylor tail is
\(O(t^{-2(R+1)})=o(t^{-(J+2)})\).  For \(2\le r\le R\),
\[
\|\delta_t^r-q_t^r\|_{L^1(\lambda)}
\le
C_r\sum_{\ell=1}^{r}
\|q_t\|_\infty^{r-\ell}
\|\rho_t\|_\infty^{\ell-1}
\|\rho_t\|_{L^1(\lambda)}
=o(t^{-(J+2)}),
\]
with the largest possible layer \(O(t^{-2})o(t^{-J})\).  Hence only pure
powers of the finite quotient polynomial \(q_t\) remain through order
\(J+2\).  Collecting the coefficient of \(t^{-m}\) gives
\eqref{eq:finite-quotient-jet-coefficients}.  The parity assertion is then
immediate from the displayed coefficient formula.

For the scale \(s_t\), the same Taylor expansion through order \(R_0\) gives
the asserted rule.
Any product containing \(e_t\) and another factor is bounded in \(L^1\) by
\(O(t^{-2})\|e_t\|_{L^1}=o(s_t)\).  A product made only of powers of \(e_t\)
is \(o(s_t)\) because \(\|e_t\|_\infty=o(t^{-2})\) and
\(\|e_t\|_{L^1}=o(s_t/t^{-2})\).  The assumed Taylor-tail bound removes the
remainder, leaving precisely the finite monomials in \(a_t+B_t\) whose size
is comparable to \(s_t\).
\end{proof}

\begin{proposition}[Finite density jet entropy transfer]%
\label{prop:finite-density-jet-entropy-transfer}
Let \(N\ge2\), put \(L=2N-2\), and let \((E,\mathcal E,\lambda)\) be a
probability space.  Suppose
\[
\delta_t=d_t+\rho_t,\qquad
d_t(y)=\sum_{n=2}^{L}a_n(y)t^{-n},
\]
where \(a_2,\ldots,a_L\in L^\infty(\lambda)\),
\(\|d_t\|_\infty=O(t^{-2})\), and
\[
\|\rho_t\|_\infty+\|\rho_t\|_{L^1(\lambda)}=o(t^{-L}).
\]
Then, with \(\Phi(s)=(1+s)\log(1+s)-s\),
\[
\int_E\Phi(\delta_t)\,\dd\lambda
=
\sum_{r=2}^{N}\frac{(-1)^r}{r(r-1)}
\int_E d_t(y)^r\,\dd\lambda
+o(t^{-(L+2)}).
\]
\end{proposition}

\begin{proof}
Since \(\|d_t\|_\infty=O(t^{-2})\) and
\(\|\rho_t\|_\infty=o(t^{-L})\), also
\(\|\delta_t\|_\infty=O(t^{-2})\).  Hence, uniformly in \(y\),
\[
\Phi(\delta_t)=
\sum_{r=2}^{N}\frac{(-1)^r}{r(r-1)}\delta_t^r
+O(|\delta_t|^{N+1}).
\]
The Taylor tail is \(O(t^{-2N-2})=o(t^{-(L+2)})\).  For every
\(2\le r\le N\), the binomial expansion and one \(L^1(\lambda)\) placement
of the remainder give
\[
\|\delta_t^r-d_t^r\|_{L^1(\lambda)}
\le
C_r\sum_{\ell=1}^{r}
\|d_t\|_\infty^{r-\ell}
\|\rho_t\|_\infty^{\ell-1}
\|\rho_t\|_{L^1(\lambda)}
=o(t^{-(L+2)}).
\]
Thus the Taylor polynomial may be evaluated on the finite density polynomial
\(d_t\) through order \(L+2\), which proves the claim.
\end{proof}

\begin{theorem}[Finite-moment Poisson entropy transfer and coefficient formula]%
\label{thm:poisson-kl-even-orders}
Let \(N\ge2\), put \(L=2N-2\), and let \(\nu\) be a probability law on
\(\RR\) with finite mean \(\bar{\gamma}\) and finite centred \(L\)th absolute
moment.  Let
\[
\begin{aligned}
X_c&=\gamma-\bar{\gamma},\\
\mu_n&=\int_{\RR} X_c^n\,\dd\nu(\gamma)\qquad(2\le n\le L),\\
h_t&=P_t*\nu,\qquad
g_t(x)&=P_t(x-\bar{\gamma}),
\end{aligned}
\]
then
\begin{equation}\label{eq:poisson-kl-even-orders}
D_{\mathrm{KL}}(h_t\|g_t)
=\sum_{m=2}^{N}A_{2m}(\nu)t^{-2m}+o(t^{-2N}).
\end{equation}
For \(4\le k\le2N\),
\begin{equation}\label{eq:coefficient-identification-transfer-expansion}
A_k(\nu)=
\sum_{r=2}^{\lfloor k/2\rfloor}
\frac{(-1)^r}{r(r-1)}
\sum_{\substack{n_1+\cdots+n_r=k\\2\le n_j\le2N-2}}
\mu_{n_1}\cdots\mu_{n_r}
\int_{\RR}u_{n_1}\cdots u_{n_r}\,\dd\omega .
\end{equation}
All retained odd \(A_k\) vanish.
\end{theorem}

For \(N\ge3\), the \((2N-2)\)-moment assumption is the first finite moment
level at which the universal polynomial \(A_{2N}\) can be evaluated for all
laws; this is not asserted to be an analytic iff threshold for the existence
of a possibly non-polynomial \(t^{-2N}\)-scale expansion.  The algebraic
reason is isolated in
Proposition~\ref{prop:poisson-kl-moment-threshold-sharpness}.

\begin{proof}
Write
\[
d_t(y)=\sum_{n=2}^{L}\mu_nu_n(y)t^{-n}.
\]
Lemma~\ref{lem:exact-moment-density-truncation} gives
\(\delta_t=d_t+\rho_t\),
\(\|\rho_t\|_\infty+\|\rho_t\|_{L^1(\omega)}=o(t^{-L})\).  The Cayley modes
\(u_2,\ldots,u_L\) are bounded, hence \(\|d_t\|_\infty=O(t^{-2})\).  Since
\[
g_t(\bar{\gamma}+ty)t\,\dd y=\omega(\dd y),
\]
and \(\int\delta_t\,\dd\omega=0\),
\[
D_{\mathrm{KL}}(h_t\|g_t)
=\int_{\RR}\Phi(\delta_t(y))\,\omega(\dd y),
\qquad
\Phi(s)=(1+s)\log(1+s)-s .
\]
Applying
Proposition~\ref{prop:finite-density-jet-entropy-transfer} with
\((E,\lambda)=(\RR,\omega)\) gives
\[
D_{\mathrm{KL}}(h_t\|g_t)
=
\sum_{r=2}^{N}\frac{(-1)^r}{r(r-1)}
\int_{\RR}\left(\sum_{n=2}^{L}\mu_nu_n(y)t^{-n}\right)^r\omega(\dd y)
+o(t^{-2N}).
\]
Extracting the Laurent coefficient of \(t^{-k}\) gives
\eqref{eq:coefficient-identification-transfer-expansion}, and the odd
orders vanish by Theorem~\ref{thm:universal-entropy-coefficients}.
\end{proof}

\begin{lemma}[Positive-margin quotient remainder and entropy transfer]
\label{lem:positive-margin-quotient-remainder}
\label{rem:positive-margin-retained-order}
Let \(L\ge2\), \(0<\eta\le1\), and let \(\nu\) have mean
\(\bar{\gamma}\) and finite centred \((L+\eta)\)th absolute moment.  With
\[
\delta_t(y):=
\frac{(P_t*\nu)(\bar{\gamma}+ty)}{P_t(ty)}-1
\]
there is a remainder \(\rho_{L,t}^{(\eta)}\) such that
\begin{equation}\label{eq:positive-margin-density-remainder}
\delta_t(y)=
\sum_{n=2}^{L}\mu_nu_n(y)t^{-n}+\rho_{L,t}^{(\eta)}(y),
\qquad
\|\rho_{L,t}^{(\eta)}\|_{\infty}
+\|\rho_{L,t}^{(\eta)}\|_{L^1(\omega)}
=O_\nu(t^{-(L+\eta)}).
\end{equation}
Moreover \(\|\delta_t\|_\infty=O_\nu(t^{-2})\).
More precisely, the remainder satisfies
\[
\|\rho_{L,t}^{(\eta)}\|_{\infty}
+\|\rho_{L,t}^{(\eta)}\|_{L^1(\omega)}
\le
C_{L,\eta}\,\EE|X_c|^{L+\eta}\,t^{-(L+\eta)},
\]
where \(C_{L,\eta}\) depends only on \(L,\eta\) and the fixed Cayley-mode
envelopes.  For \(0<\eta<1\), this is a genuine fractional-moment
improvement; for \(\eta=1\), it is the usual \((L+1)\)-moment estimate.

Consequently, if \(N\ge2\), \(L=2N-2\), and
\(\EE|X_c|^{2N-2+\eta}<\infty\), then
\begin{equation}\label{eq:positive-margin-retained-order-theorem}
D_{\mathrm{KL}}(P_t*\nu\|P_t(\cdot-\bar{\gamma}))
=\sum_{m=2}^{N}A_{2m}(\nu)t^{-2m}
+O_\nu(t^{-(2N+\eta)}).
\end{equation}
\end{lemma}

\begin{proof}
The quotient identity is
\[
\frac{P_t(\bar{\gamma}+ty-\gamma)}{P_t(ty)}
=R_{X_c/t}(y).
\]
Subtract the Taylor polynomial
\(\sum_{n=0}^{L}u_n(y)(X_c/t)^n\) and split the expectation into
\(|X_c|\le t\) and \(|X_c|>t\).  Both sides of the split are estimated by the
same \(L+\eta\) moment.

On \(|X_c|\le t\), the first line of
\eqref{eq:one-dimensional-two-region-taylor-remainder} gives, uniformly in
\(y\),
\[
\left|
\EE\!\left[
\left(R_{X_c/t}(y)-\sum_{n=0}^{L}u_n(y)(X_c/t)^n\right)
\mathbf1_{\{|X_c|\le t\}}
\right]\right|
\le
C_Lt^{-(L+1)}
\EE\!\left[|X_c|^{L+1}\mathbf1_{\{|X_c|\le t\}}\right].
\]
Because \(0<\eta\le1\) and \(|X_c|\le t\), the last term is bounded by
\[
C_Lt^{-(L+\eta)}
\EE\!\left[
|X_c|^{L+\eta}
\left(\frac{|X_c|}{t}\right)^{1-\eta}
\mathbf1_{\{|X_c|\le t\}}
\right]
\le
C_L\EE|X_c|^{L+\eta}\,t^{-(L+\eta)} .
\]
At the endpoint \(\eta=1\) this is precisely the assumed
\((L+1)\)-moment bound; for \(0<\eta<1\) it uses only the lower
\((L+\eta)\)-moment.

On \(|X_c|>t\), the second line of
\eqref{eq:one-dimensional-two-region-taylor-remainder} gives
\[
\left|
\EE\!\left[
\left(R_{X_c/t}(y)-\sum_{n=0}^{L}u_n(y)(X_c/t)^n\right)
\mathbf1_{\{|X_c|>t\}}
\right]\right|
\le
C_Lt^{-L}\EE\!\left[|X_c|^L\mathbf1_{\{|X_c|>t\}}\right].
\]
Since \(|X_c|>t\) implies
\(|X_c|^L\le t^{-\eta}|X_c|^{L+\eta}\), this is also
\[
O_\nu(t^{-(L+\eta)}).
\]
The same uniform bounds give the \(L^1(\omega)\) estimate because
\(\omega\) is a probability measure.  The \(n=0\) term cancels in
\(\delta_t\), and the \(n=1\) term vanishes after integration against
\(\nu\) because \(\EE X_c=0\).  This proves
\eqref{eq:positive-margin-density-remainder}.  The bound
\(\|\delta_t\|_\infty=O_\nu(t^{-2})\) follows from
Lemma~\ref{lem:exact-moment-density-truncation}, since
\(\EE|X_c|^{L+\eta}<\infty\) and \(L\ge2\).

Now put \(L=2N-2\) and
\[
d_t(y)=\sum_{n=2}^{L}\mu_nu_n(y)t^{-n}.
\]
As in the proof of Theorem~\ref{thm:poisson-kl-even-orders},
\[
D_{\mathrm{KL}}(P_t*\nu\|P_t(\cdot-\bar{\gamma}))
=\int_{\RR}\Phi(\delta_t(y))\,\omega(\dd y),
\qquad
\Phi(s)=(1+s)\log(1+s)-s .
\]
The proof of
Proposition~\ref{prop:finite-quotient-jet-implies-kl-coefficient-jet}, with
the stronger bound
\(\|\rho_{L,t}^{(\eta)}\|_\infty+
\|\rho_{L,t}^{(\eta)}\|_{L^1(\omega)}
=O_\nu(t^{-(L+\eta)})\), gives the same finite monomial coefficients as
Theorem~\ref{thm:poisson-kl-even-orders}; every product containing the
quotient remainder is
\(O_\nu(t^{-(L+\eta+2)})=O_\nu(t^{-(2N+\eta)})\), and the Taylor tail is
\(O_\nu(t^{-2N-2})\).  This proves
\eqref{eq:positive-margin-retained-order-theorem}.
\end{proof}

\begin{lemma}[Lower moments do not identify the next even moment]%
\label{lem:lower-moment-realizability-obstruction}
Let \(L\ge4\) be even.  There exist two centred symmetric compactly
supported probability laws \(\nu_+\) and \(\nu_-\) on \(\RR\) such that
\[
  \int x^j\,\nu_+(\dd x)=\int x^j\,\nu_-(\dd x)
  \qquad(0\le j\le L-1),
\]
but
\[
  \int x^L\,\nu_+(\dd x)\ne \int x^L\,\nu_-(\dd x).
\]
In particular, even within compactly supported centred symmetric laws, the
moment vector through order \(L-1\) does not determine the \(L\)th centred
moment.
\end{lemma}

\begin{proof}
Set \(x_i=i-L/2\), \(0\le i\le L\), and define the signed measure
\[
  \sigma=\sum_{i=0}^{L}(-1)^{L-i}\binom{L}{i}\delta_{x_i}.
\]
This is the \(L\)th finite-difference functional translated by \(L/2\).
Hence, for every polynomial \(p\) of degree \(<L\),
\[
  \int p(x)\,\sigma(\dd x)
  =\sum_{i=0}^{L}(-1)^{L-i}\binom{L}{i}p(i-L/2)=0,
\]
whereas
\[
  \int x^L\,\sigma(\dd x)=L!\ne0 .
\]
Because \(L\) is even, the coefficient at \(x_i\) equals the coefficient at
\(-x_i=x_{L-i}\); thus \(\sigma\) is symmetric and has total mass zero.

Let
\[
  \lambda=\frac1{L+1}\sum_{i=0}^{L}\delta_{x_i}.
\]
Choose \(\varepsilon>0\) so small that all weights of
\(\lambda\pm\varepsilon\sigma\) are nonnegative, and put
\(\nu_\pm=\lambda\pm\varepsilon\sigma\).  Since \(\sigma(\RR)=0\), these are
probability laws.  They are symmetric, hence centred.  The moment identities
up to order \(L-1\) follow from the annihilation of lower-degree
polynomials by \(\sigma\), while their \(L\)th moments differ by
\[
  \int x^L\,\nu_+(\dd x)-\int x^L\,\nu_-(\dd x)
  =2\varepsilon L!\ne0 .
\]
\end{proof}

\begin{proposition}[Top-moment dependence of the universal coefficient]%
\label{prop:poisson-kl-moment-threshold-sharpness}
\label{prop:analytic-first-evaluability-obstruction}
For \(N\ge3\), the universal top coefficient \(A_{2N}^{\rm form}\) contains
the nonzero monomial
\[
\kappa_N\,\mathfrak m_2\mathfrak m_{2N-2},
\qquad
\kappa_N=(-1)^N(N-1)2^{-2N+2}\ne0 .
\]
More precisely, with \(L=2N-2\),
\[
  A_{2N}^{\rm form}
  =\kappa_N\,\mathfrak m_2\mathfrak m_L
  +P_N(\mathfrak m_2,\ldots,\mathfrak m_{L-1}),
\]
where \(P_N\) involves no \(\mathfrak m_L\).  Consequently the displayed
universal polynomial \(A_{2N}\) cannot be evaluated from centred moments of
order at most \(2N-3\) on the full class of laws; two compactly supported
centred symmetric laws can have the same such lower moments but different
values of \(A_{2N}\).

This is only a coefficient-evaluability obstruction for the universal
finite-moment polynomial in
Theorem~\ref{thm:poisson-kl-even-orders}.  The proposition proves exactly the
following negative statement: if the object to be recovered is the same
polynomial \(A_{2N}\) on the full finite-moment class, then centred moments
through order \(2N-3\) do not contain enough information.  No analytic
if-and-only-if assertion is part of this proposition.  In particular, it does
not rule out law-dependent tail expansions, conditional truncated-moment
layers, cancellations on special subclasses, or renormalized non-polynomial
asymptotics under weaker hypotheses.  The only analytic if-and-only-if tail
threshold proved in this paper is the leading variance threshold,
Theorem~\ref{thm:sharp-leading-kl-variance-threshold}.
\end{proposition}

\begin{proof}
Put \(L=2N-2\).  In the formal coefficient
\eqref{eq:universal-entropy-coefficient}, a term involving
\(\mathfrak m_L\) and contributing to total order \(2N=L+2\) must come from
the quadratic part of
\((1+s)\log(1+s)-s\), with the ordered index pairs \((2,L)\) and \((L,2)\).
All terms with three or more factors have total index at least \(2+2+2\) and,
if one factor is \(L\), total index at least \(L+4> L+2\).  Therefore
\[
  A_{2N}^{\rm form}
  =J(2,L)\,\mathfrak m_2\mathfrak m_L
  +P_N(\mathfrak m_2,\ldots,\mathfrak m_{L-1}).
\]
Writing \(L=2m\), the uniform coefficient formula
\eqref{eq:mode-laurent-coefficients} gives
\(q_{2,\pm1}=q_{2,\pm2}=-1/4\) and
\(q_{2m,\pm k}=(-1)^m2^{-2m}\binom{2m-1}{k-1}\).  Hence only the exponents
\(\pm1\) and \(\pm2\) of \(Q_{2m}\) contribute to \(J(2,2m)\), and
\[
\begin{aligned}
J(2,2m)
&=[z^0]Q_2Q_{2m}  \\
&=-\frac12\,(-1)^m2^{-2m}
\left\{\binom{2m-1}{0}+\binom{2m-1}{1}\right\}  \\
&=(-1)^{m+1}m\,2^{-2m}
=(-1)^N(N-1)2^{-2N+2}.
\end{aligned}
\]
This proves the displayed top-moment coefficient.

It remains to justify the stated lower-moment non-identifiability.  Apply
Lemma~\ref{lem:lower-moment-realizability-obstruction} with this
\(L=2N-2\).  The resulting compactly supported symmetric laws \(\nu_+\) and
\(\nu_-\) have the same centred moments through order \(L-1\), in particular
the same positive second moment, but they have different \(L\)th centred
moments.  Since
\[
  A_{2N}(\nu_+)-A_{2N}(\nu_-)
  =\kappa_N\,\mu_2\{\mu_L(\nu_+)-\mu_L(\nu_-)\}
  \ne0,
\]
no rule depending only on centred moments of orders at most \(L-1=2N-3\)
can evaluate \(A_{2N}\) on the full class of laws.
\end{proof}

\subsubsection*{Conditional quotient transfer}

\medskip
\noindent\fbox{%
\begin{minipage}{0.94\linewidth}
\begin{definition}[Admissible borderline tail classes]%
\label{def:admissible-borderline-tail-classes}
Fix an even integer \(p\ge4\), \(R>0\), and \(c>0\).  Every class in this
definition uses the same disjoint core-plus-tail convention: \(\nu\) has a
nonnegative normalized density
\[
  f(x)=g(x)+\{c|x|^{-p-1}L_f(|x|)+r_f(x)\}
       \mathbf1_{\{|x|\ge R\}},
\]
where \(g\ge0\) is symmetric, compactly supported, and
\(\operatorname{supp}g\subset(-R,R)\), \(L_f\ge0\) is measurable on
\([R,\infty)\), and \(r_f\) is an even measurable perturbation on
\(\{|x|\ge R\}\).  Put
\[
  M_p(t):=\int_{|x|\le t}x^pf(x)\,\dd x,\qquad
  \mathcal R_t^{(p)}(y):=
  \int_{|x|\le t}
  \left\{R_{x/t}(y)-\sum_{n=0}^{p}u_n(y)(x/t)^n\right\}f(x)\,\dd x .
\]

\(\nu\) belongs to the admissible class \(\mathcal A_{p,R,c}\) if it is
symmetric, has finite absolute moments of every order \(q<p\), and satisfies
the four quotient controls
\begin{align}
t^{-n}\int_{|x|>t}|x|^nf(x)\,\dd x
  &=o(t^{-p}M_p(t))\qquad(0\le n<p), \label{eq:admissible-lower-tail-control}\\
t^{-p}\int_{|x|\le t}x^pf(x)\,\dd x
  &=t^{-p}M_p(t), \label{eq:admissible-top-mode-control}\\
\|\mathcal R_t^{(p)}\|_\infty+\|\mathcal R_t^{(p)}\|_{L^1(\omega)}
  &=o(t^{-p}M_p(t)), \label{eq:admissible-local-remainder-control}\\
\int_{|x|>t}\left(1+\frac{x^2}{t^2}\right)f(x)\,\dd x
  &=o(t^{-p}M_p(t)). \label{eq:admissible-far-tail-control}
\end{align}
together with the growth bounds
\begin{equation}\label{eq:admissible-top-moment-growth}
  M_p(t)\to\infty,\qquad M_p(t)=o(t^{p-2}).
\end{equation}
The statement \(M_p(t)\to\infty\) alone is not an admissible quotient
hypothesis and gives no entropy conclusion in this subsection; the displayed
quotient controls are the hypotheses that make the truncated top moment
visible to KL.
The three named subclasses used below are the following.

\emph{Exact power tail.}
\(\mathcal A_{p,R,c}^{\rm ex}\) is the class with \(L_f\equiv1\) and
\(r_f=0\).  Equivalently, after the compact core,
\[
  f(x)=g(x)+c|x|^{-p-1}\mathbf1_{\{|x|\ge R\}} .
\]

\emph{Perturbative power tail.}
\(\mathcal A_{p,R,c}^{\rm pert}\) is the class with \(L_f\equiv1\) and
\(r_f=r\), where \(r\) is even and satisfies
\[
\int_{|x|\ge R}|x|^q|r(x)|\,\dd x<\infty\quad(0\le q<p),
\qquad
\int_{R\le |x|\le t}|x|^pr(x)\,\dd x=O(1),
\]
\[
t^{-(p+1)}\int_{R\le |x|\le t}|x|^{p+1}|r(x)|\,\dd x=O(t^{-p}),
\qquad
\int_{|x|>t}\left(1+\frac{x^2}{t^2}\right)|r(x)|\,\dd x=O(t^{-p}).
\]

\emph{Monotone slowly varying tail.}
\(\mathcal A_{p,R,c}^{\rm sv}\) is the class with \(r_f=0\) and
\(L_f=L\), where \(L\) is nonnegative on \([R,\infty)\), eventually positive,
eventually monotone, slowly varying, and \(M_p(t)\to\infty\).

The exact, perturbative, and slowly varying theorems below verify membership
in these three subclasses.  Outside the three displayed subclasses the paper
uses only the admissible quotient controls defining \(\mathcal A_{p,R,c}\).
\end{definition}
\end{minipage}}
\medskip

Thus membership in \(\mathcal A_{p,R,c}\) records, in one place, the
symmetry, lower moments, moving truncated top moment, and exact estimates
needed for the quotient Taylor calculation.  In particular, for
\[
  \Delta_t(y):=\frac{(P_t*\nu)(ty)}{P_t(ty)}-1
\]
and \(\mu_n=\int x^nf(x)\,\dd x\), \(2\le n<p\), Taylor expansion on
\(|x|\le t\), the lower-tail replacement
\eqref{eq:admissible-lower-tail-control}, the local remainder
\eqref{eq:admissible-local-remainder-control}, and the far-tail control
\eqref{eq:admissible-far-tail-control} give the admissible quotient expansion
\begin{equation}\label{eq:admissible-borderline-quotient-expansion}
  \Delta_t(y)=
  \sum_{n=2}^{p-1}\mu_nu_n(y)t^{-n}
  +u_p(y)t^{-p}M_p(t)+e_t(y),
  \qquad
  \|e_t\|_\infty+\|e_t\|_{L^1(\omega)}
  =o(t^{-p}M_p(t)).
\end{equation}

The following proposition is the principal theorem of the borderline-tail
subsection.  It is only a transfer theorem: its hypothesis is the exact
quotient expansion \eqref{eq:conditional-truncated-top-mode-quotient} with the
error scale \eqref{eq:conditional-truncated-top-mode-error}; it is not a
regular-variation theorem and supplies no quotient estimate by itself.
When \(\nu\in\mathcal A_{p,R,c}\) in the sense of
Definition~\ref{def:admissible-borderline-tail-classes}, the quotient
hypothesis is supplied by \eqref{eq:admissible-borderline-quotient-expansion}.

\begin{proposition}[Principal theorem: conditional truncated top-mode quotient transfer]%
\label{thm:tail-replacements-proved-here}
\label{prop:conditional-truncated-top-moment-replacement}
Let \(N\ge3\), put \(p=2N-2\), and let \(\nu\) be a symmetric probability
law on \(\RR\) with finite absolute moments of every order \(q<p\).  Let
\[
  M_p(t):=\EE\!\left[X^p\mathbf1_{\{|X|\le t\}}\right],
  \qquad
  \Delta_t(y):=\frac{(P_t*\nu)(ty)}{P_t(ty)}-1 .
\]
Assume first that the quotient admits the \(L^\infty(\RR)\cap L^1(\omega)\)
expansion
\begin{equation}\label{eq:conditional-truncated-top-mode-quotient}
  \Delta_t(y)=
  A_t(y)
  +u_p(y)t^{-p}M_p(t)+e_t(y),
  \qquad
  A_t(y):=\sum_{n=2}^{p-1}\mu_nu_n(y)t^{-n},
\end{equation}
where
\begin{equation}\label{eq:conditional-truncated-top-mode-error}
  \|e_t\|_\infty+\|e_t\|_{L^1(\omega)}
  =o(t^{-p}M_p(t)).
\end{equation}
Assume also the growth bounds
\begin{equation}\label{eq:conditional-truncated-moment-growth}
  M_p(t)\to\infty,\qquad M_p(t)=o(t^{p-2}).
\end{equation}
The growth \(M_p(t)\to\infty\) has no standalone consequence here; the entropy
conclusion below uses the quotient expansion and error estimate above.
Then, with \(H(t)=D_{\mathrm{KL}}(P_t*\nu\|P_t)\),
\begin{equation}\label{eq:conditional-truncated-top-moment-entropy}
H(t)=
\sum_{m=2}^{N-1}A_{2m}(\nu)t^{-2m}
+\kappa_N\mu_2\,t^{-(p+2)}M_p(t)
+o(t^{-(p+2)}M_p(t)),
\end{equation}
where
\[
  \kappa_N=J(2,p)=(-1)^N(N-1)2^{-2N+2}.
\]
\end{proposition}

\subsubsection*{Exact power-tail verification}

For the exact power-law corollary below, the proof verifies
\(\nu\in\mathcal A_{p,R,c}^{\rm ex}\) from
Definition~\ref{def:admissible-borderline-tail-classes} and then applies the
conditional transfer.
Lemma~\ref{lem:borderline-exact-power-tail-quotient-expansion} gives the
sharper \(O(t^{-p})\) quotient remainder used in that verification.  The
quotient expansion is repeated in the theorem statement so that the entropy
conclusion cannot be read as a consequence of regular variation alone.

\begin{corollary}[Verified subclass: exact symmetric power tail]%
\label{prop:log-tail-counterexample-top-moment}
Let \(N\ge3\), put \(p=2N-2\), and choose \(R>0\) and \(c>0\) such that
\[
  \frac{2c}{pR^p}<1 .
\]
Let \(g\in L^1(\RR)\) be any symmetric nonnegative compactly supported
function with \(\operatorname{supp}g\subset(-R,R)\) and total mass
\(1-2c/(pR^p)\), and define the symmetric probability density
\begin{equation}\label{eq:normalized-borderline-tail-density}
  f(x)=g(x)+c|x|^{-(p+1)}\mathbf1_{\{|x|\ge R\}} .
\end{equation}
Let \(\nu\) be the law with density \(f\).  Then
\(\nu\in\mathcal A_{p,R,c}^{\rm ex}\subset\mathcal A_{p,R,c}\), where the
classes are those of
Definition~\ref{def:admissible-borderline-tail-classes}.  In particular
\(\nu\) is centred,
\[
\EE|X|^q<\infty\quad(q<p),\qquad \EE|X|^p=\infty ,
\]
and
\[
\EE\!\left[X^p\mathbf1_{\{|X|\le t\}}\right]=2c\log t+O(1).
\]
For
\[
  \Delta_t(y):=\frac{(P_t*\nu)(ty)}{P_t(ty)}-1
\]
the exact quotient expansion
\begin{equation}\label{eq:exact-power-tail-theorem-quotient}
  \Delta_t(y)=
  \sum_{n=2}^{p-1}\mu_nu_n(y)t^{-n}
  +2c\,u_p(y)t^{-p}\log t+E_t(y)
\end{equation}
holds in \(L^\infty(\RR)\cap L^1(\omega)\), with
\begin{equation}\label{eq:exact-power-tail-theorem-quotient-error}
  \|E_t\|_\infty+\|E_t\|_{L^1(\omega)}=O(t^{-p}).
\end{equation}
Since \(M_p(t)=2c\log t+O(1)\), this displayed quotient expansion verifies
the conditional quotient hypothesis of
Proposition~\ref{thm:tail-replacements-proved-here}.  The entropy assertion
below is a consequence of that verification, not of the divergence of
\(M_p(t)\) alone.
With \(H(t)=D_{\mathrm{KL}}(P_t*\nu\|P_t)\),
\begin{equation}\label{eq:core-log-tail-counterexample-asymptotic}
H(t)=\sum_{m=2}^{N-1}A_{2m}(\nu)t^{-2m}
    +2c\,\kappa_N\,\mu_2\,t^{-2N}\log t
    +o(t^{-2N}\log t),
\end{equation}
where
\[
\kappa_N=J(2,p)=(-1)^N(N-1)2^{-2N+2}\ne0
\]
is the quadratic cross coefficient, equivalently the coefficient of the
monomial
\(\mathfrak m_2\mathfrak m_{2N-2}\) in
Proposition~\ref{prop:poisson-kl-moment-threshold-sharpness}.  Thus, for this
explicit power-tail law, the missing \(A_{2N}t^{-2N}\) polynomial layer is
replaced by the displayed tail-dependent logarithmic layer.  The case \(N=2\)
is excluded from this statement.
\end{corollary}

\begin{theorem}[Borderline infinite-variance power tail]%
\label{prop:n2-borderline-infinite-variance-layer}
Let \(c>0\) satisfy \(c\mathrm e^{-2}<1\), and let \(\nu\) be the symmetric
probability law with density
\[
f(x)=\frac{1-c\mathrm e^{-2}}{2}\mathbf1_{\{|x|\le1\}}
      +c|x|^{-3}\mathbf1_{\{|x|\ge \mathrm e\}} .
\]
Then \(\nu\) has finite mean and infinite variance, and
\[
\EE\!\left[X^2\mathbf1_{\{|X|\le t\}}\right]\sim 2c\log t,
\]
while
\[
D_{\mathrm{KL}}(P_t*\nu\|P_t)
=\frac{c^2}{2}\,t^{-4}(\log t)^2+O(t^{-4}\log t).
\]
Thus the borderline \(N=2\) law is not a \(t^{-4}\log t\) replacement; its
leading layer is \(t^{-4}(\log t)^2\), in agreement with the variance
threshold theorem below.
\end{theorem}

\subsubsection*{Perturbative power-tail verification}

The perturbative corollary
\ref{thm:exact-perturbative-power-tail-log-replacement} is separated
from the exact power law.  For later reference, when \(p\ge4\), \(R>0\), and
\(r\) is an even integrable function on \(\{|x|\ge R\}\), the four
perturbative quotient controls are
\begin{equation}\label{eq:perturbative-power-tail-lower-moments}
  \int_{|x|\ge R}|x|^q|r(x)|\,\dd x<\infty
  \qquad(0\le q<p),
\end{equation}
\begin{equation}\label{eq:perturbative-power-tail-top-moment}
  \int_{R\le |x|\le t}|x|^p r(x)\,\dd x=O(1),
\end{equation}
\begin{equation}\label{eq:perturbative-power-tail-local-remainder}
  t^{-(p+1)}\int_{R\le |x|\le t}|x|^{p+1}|r(x)|\,\dd x=O(t^{-p}),
\end{equation}
and
\begin{equation}\label{eq:perturbative-power-tail-far-remainder}
  \int_{|x|>t}\left(1+\frac{x^2}{t^2}\right)|r(x)|\,\dd x=O(t^{-p}).
\end{equation}
These are precisely the perturbative subclass conditions in
Definition~\ref{def:admissible-borderline-tail-classes}; they verify
membership in \(\mathcal A_{p,R,c}^{\rm pert}\) after the perturbation has
been absorbed into the compact-plus-tail decomposition.
Lemma~\ref{lem:perturbative-power-tail-quotient-expansion}
records the resulting quotient expansion.  The corollary below repeats the
controls and the quotient expansion before transferring it to entropy.  No
conclusion is claimed for a perturbation that fails these controls.

\begin{lemma}[Pointwise sufficient condition for the perturbative tail]%
\label{lem:pointwise-perturbative-power-tail-sufficient-condition}
Let \(p\ge4\), \(R>0\), and let \(r\) be a measurable function on
\(\{|x|\ge R\}\).  Suppose that, for some \(C<\infty\) and \(\eta>0\),
\begin{equation}\label{eq:pointwise-perturbative-tail-condition}
  |r(x)|\le C|x|^{-p-1-\eta}\qquad(|x|\ge R).
\end{equation}
Then \(r\) satisfies
\eqref{eq:perturbative-power-tail-lower-moments}--\eqref{eq:perturbative-power-tail-far-remainder}.
\end{lemma}

\begin{proof}
For \(0\le q<p\),
\[
  \int_{|x|\ge R}|x|^q|r(x)|\,\dd x
  \le 2C\int_R^\infty x^{q-p-1-\eta}\,\dd x<\infty,
\]
which proves \eqref{eq:perturbative-power-tail-lower-moments}.  Also
\[
  \left|\int_{R\le |x|\le t}|x|^pr(x)\,\dd x\right|
  \le 2C\int_R^\infty x^{-1-\eta}\,\dd x=O(1),
\]
so \eqref{eq:perturbative-power-tail-top-moment} holds.  For the local
Taylor-remainder condition,
\[
  t^{-(p+1)}\int_{R\le |x|\le t}|x|^{p+1}|r(x)|\,\dd x
  \le
  2Ct^{-(p+1)}\int_R^t x^{-\eta}\,\dd x
  =
  \begin{cases}
    O(t^{-p-\eta}),&0<\eta<1,\\
    O(t^{-p-1}\log t),&\eta=1,\\
    O(t^{-p-1}),&\eta>1,
  \end{cases}
\]
and each alternative is \(O(t^{-p})\).  Finally,
\[
\begin{aligned}
  \int_{|x|>t}\left(1+\frac{x^2}{t^2}\right)|r(x)|\,\dd x
  &\le
  2C\int_t^\infty x^{-p-1-\eta}\,\dd x
  +2Ct^{-2}\int_t^\infty x^{1-p-\eta}\,\dd x  \\
  &=O(t^{-p-\eta})=O(t^{-p}),
\end{aligned}
\]
because \(p\ge4\).  This proves
\eqref{eq:perturbative-power-tail-far-remainder}.
\end{proof}

\begin{proof}[Proof of Proposition~\ref{thm:tail-replacements-proved-here}]
Set
\[
  A_t(y)=\sum_{n=2}^{p-1}\mu_nu_n(y)t^{-n},
  \qquad
  B_t(y)=u_p(y)t^{-p}M_p(t).
\]
The Cayley modes are bounded, so \(\|A_t\|_\infty=O(t^{-2})\).  By
\eqref{eq:conditional-truncated-moment-growth},
\(\|B_t\|_\infty=o(t^{-2})\); by
\eqref{eq:conditional-truncated-top-mode-error},
\(\|e_t\|_\infty=o(t^{-2})\).  Hence
\(\|\Delta_t\|_\infty\to0\), and the KL identity gives
\[
  H(t)=\int_{\RR}\Phi(\Delta_t(y))\,\omega(\dd y),
  \qquad
  \Phi(s)=(1+s)\log(1+s)-s .
\]
Set
\[
  s_t:=t^{-2N}M_p(t)=t^{-(p+2)}M_p(t).
\]
The following bookkeeping is the finite quotient-transfer argument at this
moving scale, written out to isolate the conditional tail hypothesis.  Expand
\(\Phi\) through degree \(N\):
\[
\Phi(\Delta_t)=
\sum_{r=2}^{N}\frac{(-1)^r}{r(r-1)}\Delta_t^r
+O(|\Delta_t|^{N+1}).
\]
The Taylor tail satisfies
\[
  O(\|\Delta_t\|_\infty^{N}\|\Delta_t\|_{L^1(\omega)})
  =O(t^{-2N-2})+o(t^{-2N}M_p(t))
  =o(s_t),
\]
because \(M_p(t)\to\infty\).  Every monomial containing \(e_t\) is also
negligible at scale \(s_t\).  Indeed, the remaining factors are
\(O(t^{-2})\) in \(L^\infty\), while
\[
  \|e_t\|_{L^1(\omega)}
  =o(t^{-p}M_p(t))=o(s_t/t^{-2});
\]
pure powers of \(e_t\) are smaller because
\(\|e_t\|_\infty=o(t^{-2})\).  Thus only finite monomials in \(A_t+B_t\)
can contribute at scale \(s_t\).
Terms containing \(B_t\) at least twice are also negligible:
\[
  \|B_t\|_\infty^2
  =t^{-2p}M_p(t)^2
  =o(t^{-(p+2)}M_p(t))
  =o(t^{-2N}M_p(t)),
\]
where the middle equality uses \(M_p(t)=o(t^{p-2})\).  If a term contains one
factor \(B_t\) and at least two factors from \(A_t\), its size is
\[
  O(t^{-4})\,t^{-p}M_p(t)
  =o(t^{-(p+2)}M_p(t)).
\]
Thus the only surviving top-mode term is the quadratic cross term between
\(\mu_2u_2t^{-2}\) and \(u_pt^{-p}M_p(t)\):
\[
  \int_{\RR}\mu_2u_2(y)t^{-2}\,u_p(y)t^{-p}M_p(t)\,\omega(\dd y)
  =
  \mu_2J(2,p)t^{-2N}M_p(t).
\]
The pure \(A_t\) terms of total order \(<2N\) give
\(\sum_{m=2}^{N-1}A_{2m}(\nu)t^{-2m}\).  Pure \(A_t\) terms of total order
\(2N\), if present, are \(O(t^{-2N})=o(t^{-2N}M_p(t))\), and higher pure
terms are smaller.  Finally, the computation in
Proposition~\ref{prop:poisson-kl-moment-threshold-sharpness} gives
\(J(2,p)=\kappa_N\).  This proves
\eqref{eq:conditional-truncated-top-moment-entropy}.
\end{proof}

\begin{lemma}[Borderline exact power-tail quotient expansion]%
\label{lem:borderline-exact-power-tail-quotient-expansion}
Let \(p\ge4\) be even.  Let
\[
  f(x)=g(x)+c|x|^{-(p+1)}\mathbf1_{\{|x|\ge R\}},
\]
where \(c>0\), \(R>0\), and \(g\) is a symmetric nonnegative integrable
compactly supported function.  Assume that \(f\) is normalized to have mass
one, and let \(\nu\) be the corresponding symmetric law.  For
\[
  \Delta_t(y):=\frac{(P_t*\nu)(ty)}{P_t(ty)}-1
  =\int_{\RR}\{R_{x/t}(y)-1\}f(x)\,\dd x ,
\]
and for \(2\le n\le p-1\), let \(\mu_n=\int x^nf(x)\,\dd x\).  Then, in
\(L^\infty(\RR)\cap L^1(\omega)\),
\begin{equation}\label{eq:borderline-exact-power-tail-quotient-expansion}
  \Delta_t(y)=
  \sum_{n=2}^{p-1}\mu_nu_n(y)t^{-n}
  +2c\,u_p(y)t^{-p}\log t+E_t(y),
\end{equation}
where
\begin{equation}\label{eq:borderline-exact-power-tail-quotient-error}
  \|E_t\|_\infty+\|E_t\|_{L^1(\omega)}=O(t^{-p}).
\end{equation}
The implicit constant may depend on \(p,R,c\), and \(g\), but not on \(t\).
\end{lemma}

\begin{proof}
Choose \(B\ge R\) with \(\operatorname{supp}g\subset[-B,B]\).  It is enough
to consider \(t\ge \max\{B,1\}\).  Split the compact core as
\[
  g=g_{\rm in}+g_{\rm out},\qquad
  g_{\rm in}=g\,\mathbf1_{\{|x|<R\}},\qquad
  \operatorname{supp}g_{\rm out}\subset\{R\le |x|\le B\}.
\]
We decompose \(\Delta_t=C_t^{\rm in}+C_t^{\rm out}+T_t+F_t\), where
\(C_t^{\rm in}\) and \(C_t^{\rm out}\) are the contributions of
\(g_{\rm in}\) and \(g_{\rm out}\), \(T_t\) is the explicit power-tail
contribution on \(R\le |x|\le t\), and \(F_t\) is the explicit far tail
\(|x|>t\).  All estimates below are uniform in \(y\); since \(\omega\) is a
probability measure, the same bounds also give the \(L^1(\omega)\) estimates.

\emph{Compact core.}  On the full compact set \(|x|\le B\), the local Cayley
Taylor estimate gives
\[
  R_{x/t}(y)-1
  =\sum_{n=1}^{p}u_n(y)(x/t)^n
   +O_{p,B}(|x/t|^{p+1})
\]
in \(L^\infty_y\).  Hence
\[
  C_t^{\rm in}(y)+C_t^{\rm out}(y)
  =\sum_{n=1}^{p}u_n(y)t^{-n}\int_{\RR}x^ng(x)\,\dd x
          +O(t^{-(p+1)}).
\]
The odd core terms vanish by symmetry.  The \(n=p\) core term is \(O(t^{-p})\)
and will be absorbed into \(E_t\); the terms \(2\le n\le p-1\) contribute the
core parts of the moments \(\mu_n\).

This explicitly includes the possible compact outside piece.  Equivalently,
for \(t>B\),
\[
\int g_{\rm out}(x)\{R_{x/t}(y)-1\}\,\dd x
=\sum_{n=1}^{p}u_n(y)t^{-n}
  \int x^ng_{\rm out}(x)\,\dd x+O(t^{-(p+1)}).
\]
Evenness removes odd \(n\).  The terms \(2\le n<p\) are part of the lower
moments \(\mu_n\).  If
\[
  C_p:=\int x^pg_{\rm out}(x)\,\dd x ,
\]
then the compact outside top contribution is \(C_pu_p(y)t^{-p}\).  In the
later entropy transfer with \(p=2N-2\), its only possible logarithmic-scale
interaction with the leading \(\mu_2u_2t^{-2}\)-mode has size
\[
  \mu_2C_pJ(2,p)t^{-(p+2)}=O(t^{-2N})=o(t^{-2N}\log t),
\]
so this compact \(p\)-mode cannot change the logarithmic coefficient
\(2c\kappa_N\mu_2\) in
Corollary~\ref{prop:log-tail-counterexample-top-moment}.

\emph{Truncated tail moments.}  On \(R\le |x|\le t\), the same expansion gives
\[
  T_t(y)=
  \sum_{n=1}^{p}u_n(y)t^{-n}
  \int_{R\le |x|\le t}x^nc|x|^{-p-1}\,\dd x
  +\mathcal R_t(y).
\]
Again the odd terms vanish.  For \(2\le n<p\), replacing the truncated tail
moment by the full tail moment costs
\[
  t^{-n}\int_{|x|>t}|x|^nc|x|^{-p-1}\,\dd x
  \le Ct^{-n}\int_t^\infty x^{n-p-1}\,\dd x
  =O(t^{-p}).
\]
Thus the core and truncated-tail terms together give
\(\sum_{n=2}^{p-1}\mu_nu_n(y)t^{-n}\), up to an \(O(t^{-p})\) error.  The
borderline top tail mode is
\[
  t^{-p}u_p(y)\int_{R\le |x|\le t}x^pc|x|^{-p-1}\,\dd x
  =
  2c\,u_p(y)t^{-p}\int_R^t\frac{\dd x}{x}
  =2c\,u_p(y)t^{-p}\log t+O(t^{-p}).
\]

\emph{Taylor remainder on the truncated tail.}  The remainder satisfies
\[
  \|\mathcal R_t\|_\infty
  \le
  Ct^{-(p+1)}
  \int_{R\le |x|\le t}|x|^{p+1}c|x|^{-p-1}\,\dd x
  \le Ct^{-p}.
\]
This is the only place where the nonintegrable \((p+1)\)-moment would have
entered; truncation at \(t\) leaves exactly the \(O(t^{-p})\) remainder needed
for \eqref{eq:borderline-exact-power-tail-quotient-error}.

\emph{Far tail.}  For all \(\varepsilon\in\RR\),
\[
  |R_\varepsilon(y)-1|\le C(1+\varepsilon^2)
\]
uniformly in \(y\).  Since \(p\ge4\),
\[
  \|F_t\|_\infty
  \le
  C\int_{|x|>t}\left(1+\frac{x^2}{t^2}\right)c|x|^{-p-1}\,\dd x
  \le
  C\left\{\int_t^\infty x^{-p-1}\,\dd x
  +t^{-2}\int_t^\infty x^{1-p}\,\dd x\right\}
  =O(t^{-p}).
\]
The already-assigned polynomial moments omitted from \(|x|>t\) are also
\(O(t^{-p})\) by the estimate in the truncated-tail paragraph.  Combining the
compact-core, truncated-tail, Taylor-remainder, and far-tail estimates proves
\eqref{eq:borderline-exact-power-tail-quotient-expansion} and
\eqref{eq:borderline-exact-power-tail-quotient-error}.
\end{proof}

\begin{proof}[Proof of Corollary~\ref{prop:log-tail-counterexample-top-moment}]
The mass condition makes \eqref{eq:normalized-borderline-tail-density} a
probability density:
\[
  \int_{\RR}f(x)\,\dd x
  =\left(1-\frac{2c}{pR^p}\right)
    +2c\int_R^\infty x^{-p-1}\,\dd x
  =1 .
\]
The compact core is symmetric and nonnegative by construction, the tail is
symmetric, and therefore \(\nu\) is centred.  The moment assertions are
immediate from compact support of \(g\) and
\[
2c\int_R^\infty x^q x^{-p-1}\,\dd x<\infty
\quad\Longleftrightarrow\quad q<p.
\]
The truncated top moment is
\[
\EE\!\left[X^p\mathbf1_{\{|X|\le t\}}\right]
=O(1)+2c\int_R^t x^{-1}\,\dd x
=2c\log t+O(1).
\]

For \(x=ty\), write
\[
\Delta_t(y)=\frac{(P_t*\nu)(ty)}{P_t(ty)}-1
=\int_{\RR}\{R_{x/t}(y)-1\}f(x)\dd x .
\]
Lemma~\ref{lem:borderline-exact-power-tail-quotient-expansion} gives the
quotient expansion \eqref{eq:exact-power-tail-theorem-quotient} with
\eqref{eq:exact-power-tail-theorem-quotient-error}, namely
\[
\Delta_t(y)=\sum_{n=2}^{p-1}\mu_nu_n(y)t^{-n}
            +2c\,u_p(y)t^{-p}\log t+\varepsilon_t(y),
\]
with
\(\|\varepsilon_t\|_\infty+\|\varepsilon_t\|_{L^1(\omega)}\le Ct^{-p}\).
The compact-core, truncated-tail, local-remainder, and far-tail estimates in
Lemma~\ref{lem:borderline-exact-power-tail-quotient-expansion} are precisely
the four controls
\eqref{eq:admissible-lower-tail-control}--\eqref{eq:admissible-far-tail-control};
together with \(M_p(t)=2c\log t+O(1)\), they prove
\(\nu\in\mathcal A_{p,R,c}\).  Because \(M_p(t)=2c\log t+O(1)\), this verifies
\eqref{eq:conditional-truncated-moment-growth}--\eqref{eq:conditional-truncated-top-mode-error}
with \(M_p(t)\) in place of \(2c\log t\): the difference between
\(M_p(t)u_pt^{-p}\) and \(2c(\log t)u_pt^{-p}\) is \(O(t^{-p})\), which is
absorbed into the conditional error.  The conditional quotient-transfer
Proposition~\ref{thm:tail-replacements-proved-here} therefore gives
\[
H(t)=
\sum_{m=2}^{N-1}A_{2m}(\nu)t^{-2m}
+\kappa_N\mu_2\,t^{-2N}M_p(t)
+o(t^{-2N}M_p(t)).
\]
Substituting \(M_p(t)=2c\log t+O(1)\) proves
\eqref{eq:core-log-tail-counterexample-asymptotic}.
\end{proof}

\begin{lemma}[Perturbative power-tail quotient expansion]%
\label{lem:perturbative-power-tail-quotient-expansion}
Let \(p\ge4\) be even.  Let \(R>0\), \(c>0\), let
\(g\in L^1(\RR)\) be symmetric and nonnegative with
\(\operatorname{supp}g\subset(-R,R)\), and let \(r\) be an even integrable
function on \(\{|x|\ge R\}\).  Assume that
\[
  f(x)=g(x)+\bigl(c|x|^{-p-1}+r(x)\bigr)\mathbf1_{\{|x|\ge R\}}
\]
is nonnegative and normalized to have mass one, and assume
\eqref{eq:perturbative-power-tail-lower-moments}--\eqref{eq:perturbative-power-tail-far-remainder}.
Let \(\nu\) be the law with density \(f\), put
\[
  M_p(t):=\int_{|x|\le t}x^pf(x)\,\dd x,
  \qquad
  \Delta_t(y):=\frac{(P_t*\nu)(ty)}{P_t(ty)}-1 ,
\]
and, for \(2\le n\le p-1\), let
\(\mu_n=\int_{\RR}x^nf(x)\,\dd x\).  Then \(\nu\) is centred, has finite
absolute moments of every order \(q<p\), and
\begin{equation}\label{eq:perturbative-power-tail-top-truncated-moment}
  M_p(t)=2c\log t+O(1).
\end{equation}
Moreover, in \(L^\infty(\RR)\cap L^1(\omega)\),
\begin{equation}\label{eq:perturbative-power-tail-quotient-expansion}
  \Delta_t(y)=
  \sum_{n=2}^{p-1}\mu_nu_n(y)t^{-n}
  +u_p(y)t^{-p}M_p(t)+E_t(y),
\end{equation}
where
\begin{equation}\label{eq:perturbative-power-tail-quotient-error}
  \|E_t\|_\infty+\|E_t\|_{L^1(\omega)}=O(t^{-p}).
\end{equation}
\end{lemma}

\begin{proof}
It is enough to consider \(t\ge\max\{R,1\}\).  The symmetry of \(g\), of
\(|x|^{-p-1}\), and of \(r\) makes \(f\) symmetric; hence \(\nu\) is centred
and every odd truncated or full moment below vanishes.  The compact core has
all moments, the explicit tail has finite \(q\)th absolute moment exactly for
\(q<p\), and
\eqref{eq:perturbative-power-tail-lower-moments} gives the corresponding
moments of \(r\).  Thus the lower moments \(\mu_n\), \(2\le n<p\), are
well-defined.

We first record the four perturbative estimates used in the quotient
calculation.  From \eqref{eq:perturbative-power-tail-local-remainder} there is
a constant \(C\) such that
\[
  A(T):=\int_{R\le |x|\le T}|x|^{p+1}|r(x)|\,\dd x\le CT
  \qquad(T\ge R).
\]
For \(0\le n<p\), a dyadic decomposition gives the lower-moment tail
replacement
\begin{equation}\label{eq:perturbative-lower-moment-tail-replacement}
\begin{aligned}
  \int_{|x|>t}|x|^n|r(x)|\,\dd x
  &\le
  \sum_{k=0}^{\infty}(2^kt)^{\,n-p-1}
  \int_{2^kt<|x|\le2^{k+1}t}|x|^{p+1}|r(x)|\,\dd x  \\
  &\le C_n t^{\,n-p}.
\end{aligned}
\end{equation}
Consequently
\[
  t^{-n}\int_{R\le |x|\le t}x^nr(x)\,\dd x
  =
  t^{-n}\int_{|x|\ge R}x^nr(x)\,\dd x+O(t^{-p})
  \qquad(0\le n<p).
\]
The top signed moment of the perturbation is controlled separately by
\eqref{eq:perturbative-power-tail-top-moment}:
\begin{equation}\label{eq:perturbative-top-signed-moment-bound}
  \int_{R\le |x|\le t}x^pr(x)\,\dd x=O(1).
\end{equation}
Hence
\[
  M_p(t)
  =
  \int_{|x|<R}x^pg(x)\,\dd x
  +2c\int_R^t x^{-1}\,\dd x
  +\int_{R\le |x|\le t}x^pr(x)\,\dd x
  =2c\log t+O(1),
\]
which proves \eqref{eq:perturbative-power-tail-top-truncated-moment}.

The local Taylor remainder for the perturbation is also explicit.  On
\(|x|\le t\), \eqref{eq:one-dimensional-two-region-taylor-remainder} gives
\[
  \left|R_{x/t}(y)-\sum_{n=0}^{p}u_n(y)(x/t)^n\right|
  \le C_p |x/t|^{p+1}
  \qquad(y\in\RR),
\]
and therefore
\begin{equation}\label{eq:perturbative-local-taylor-remainder-bound}
\left\|
\int_{R\le |x|\le t}
\left\{R_{x/t}-\sum_{n=0}^{p}u_n(x/t)^n\right\}r(x)\,\dd x
\right\|_{\infty}
\le
C_pt^{-(p+1)}
\int_{R\le |x|\le t}|x|^{p+1}|r(x)|\,\dd x
=O(t^{-p}).
\end{equation}
Since \(\omega\) is a probability measure, the same estimate holds in
\(L^1(\omega)\).  Finally, the elementary bound
\(|R_\varepsilon(y)-1|\le C(1+\varepsilon^2)\), uniform in \(y\), and
\eqref{eq:perturbative-power-tail-far-remainder} give the far-tail quotient
bound
\begin{equation}\label{eq:perturbative-far-tail-quotient-bound}
\left\|
\int_{|x|>t}\{R_{x/t}-1\}r(x)\,\dd x
\right\|_\infty
+
\left\|
\int_{|x|>t}\{R_{x/t}-1\}r(x)\,\dd x
\right\|_{L^1(\omega)}
=O(t^{-p}).
\end{equation}

We now assemble the quotient.  Split the integral defining \(\Delta_t\) into
\(|x|\le t\) and \(|x|>t\).  On \(|x|\le t\), Taylor expansion through order
\(p\) gives
\[
  \int_{|x|\le t}\{R_{x/t}(y)-1\}f(x)\,\dd x
  =
  \sum_{n=1}^{p}u_n(y)t^{-n}\int_{|x|\le t}x^nf(x)\,\dd x
  +\mathcal R_t(y).
\]
The compact core and the explicit \(c|x|^{-p-1}\) tail have local remainders
\(O(t^{-p})\), by the same estimates displayed in
Lemma~\ref{lem:borderline-exact-power-tail-quotient-expansion}; the
perturbative part is controlled by
\eqref{eq:perturbative-local-taylor-remainder-bound}.  Thus
\(\|\mathcal R_t\|_\infty+\|\mathcal R_t\|_{L^1(\omega)}=O(t^{-p})\).
For \(2\le n<p\), replacing the truncated \(n\)th moment by the full moment
costs \(O(t^{-p})\): for the explicit power tail this is
\[
  t^{-n}\int_{|x|>t}|x|^n c|x|^{-p-1}\,\dd x=O(t^{-p}),
\]
and for \(r\) it is
\eqref{eq:perturbative-lower-moment-tail-replacement}.  The odd terms vanish
by symmetry, and the \(n=p\) term is exactly \(u_p(y)t^{-p}M_p(t)\).

It remains only to estimate the far tail.  For the explicit power tail,
\[
  \int_{|x|>t}(1+x^2/t^2)c|x|^{-p-1}\,\dd x=O(t^{-p}),
\]
and the perturbative far tail is
\eqref{eq:perturbative-far-tail-quotient-bound}.  Combining the local
expansion, the lower-moment replacements, the top truncated moment, and the
far-tail bound gives \eqref{eq:perturbative-power-tail-quotient-expansion} and
\eqref{eq:perturbative-power-tail-quotient-error}.
\end{proof}

\begin{corollary}[Verified subclass: perturbative symmetric power tail]%
\label{thm:exact-perturbative-power-tail-log-replacement}
\label{cor:perturbative-power-tail-log-replacement}
Let \(N\ge3\), put \(p=2N-2\), and let \(R>0\), \(c>0\).  Let \(f\) be an
even probability density of the form
\[
f(x)=g(x)+\{c|x|^{-p-1}+r(x)\}\mathbf1_{\{|x|\ge R\}},
\]
where \(g\in L^1(\RR)\) is symmetric and nonnegative with
\(\operatorname{supp}g\subset(-R,R)\), while \(r\) is even and satisfies
\[
\int_{|x|\ge R}|x|^q|r(x)|\,\dd x<\infty\quad(0\le q<p),
\qquad
\int_{R\le |x|\le t}|x|^pr(x)\,\dd x=O(1),
\]
\[
t^{-(p+1)}\int_{R\le |x|\le t}|x|^{p+1}|r(x)|\,\dd x=O(t^{-p}),
\qquad
\int_{|x|>t}\left(1+\frac{x^2}{t^2}\right)|r(x)|\,\dd x=O(t^{-p}).
\]
Let \(\nu\) be the law with density \(f\), set
\[
H(t):=D_{\mathrm{KL}}(P_t*\nu\|P_t),
\qquad
\mu_2:=\int_{\RR}x^2f(x)\,\dd x,
\]
and let
\[
\kappa_N:=J(2,p)=(-1)^N(N-1)2^{-2N+2}.
\]
Then \(\nu\) is centred, has finite absolute moments of every order \(q<p\),
belongs to
\(\mathcal A_{p,R,c}^{\rm pert}\subset\mathcal A_{p,R,c}\) in the sense of
Definition~\ref{def:admissible-borderline-tail-classes}, and
\[
  M_p(t):=\EE[X^p\mathbf1_{\{|X|\le t\}}]=2c\log t+O(1).
\]
Moreover, for
\[
  \Delta_t(y):=\frac{(P_t*\nu)(ty)}{P_t(ty)}-1,
\]
the quotient expansion
\begin{equation}\label{eq:perturbative-power-tail-theorem-quotient}
\Delta_t(y)=
\sum_{n=2}^{p-1}\mu_nu_n(y)t^{-n}
+u_p(y)t^{-p}M_p(t)+E_t(y)
\end{equation}
holds in \(L^\infty(\RR)\cap L^1(\omega)\), with
\begin{equation}\label{eq:perturbative-power-tail-theorem-quotient-error}
\|E_t\|_\infty+\|E_t\|_{L^1(\omega)}=O(t^{-p}).
\end{equation}
Since \(M_p(t)=2c\log t+O(1)\), this quotient expansion verifies the
conditional quotient hypothesis of
Proposition~\ref{thm:tail-replacements-proved-here}.  The following entropy
asymptotic is not asserted from \(M_p(t)\to\infty\) alone.  Consequently
\[
D_{\mathrm{KL}}(P_t*\nu\|P_t)
=\sum_{m=2}^{N-1}A_{2m}(\nu)t^{-2m}
+2c\,\kappa_N\mu_2\,t^{-2N}\log t
+o(t^{-2N}\log t).
\]
No regularly varying theorem is claimed without the quotient controls above.
\end{corollary}

\begin{proof}
The support condition on \(g\) makes the displayed formula a disjoint
core-plus-tail decomposition: \(g\) contributes only on \((-R,R)\), while the
explicit power tail and the perturbation \(r\) contribute only on
\(\{|x|\ge R\}\).  Therefore the hypotheses are exactly those of
Lemma~\ref{lem:perturbative-power-tail-quotient-expansion} with this same
cutoff \(R\).  That lemma proves the centering, the lower moment assertions,
and
\[
  M_p(t)=\EE[X^p\mathbf1_{\{|X|\le t\}}]=2c\log t+O(1).
\]
It also gives the quotient expansion
\eqref{eq:perturbative-power-tail-theorem-quotient} with
\eqref{eq:perturbative-power-tail-theorem-quotient-error}, namely
\[
\Delta_t(y)=
\sum_{n=2}^{p-1}\mu_nu_n(y)t^{-n}
+u_p(y)t^{-p}M_p(t)+E_t(y),
\qquad
\|E_t\|_\infty+\|E_t\|_{L^1(\omega)}=O(t^{-p}).
\]
The same lemma proves the lower-moment replacement, the top truncated-moment
identity, the local Taylor-remainder bound, and the far-tail bound that form
\eqref{eq:admissible-lower-tail-control}--\eqref{eq:admissible-far-tail-control}.
Since \(M_p(t)\sim2c\log t\), these estimates verify
\(\nu\in\mathcal A_{p,R,c}^{\rm pert}\subset\mathcal A_{p,R,c}\) from
Definition~\ref{def:admissible-borderline-tail-classes}.
Since \(M_p(t)\sim2c\log t\), the error is
\(o(t^{-p}M_p(t))\), while
\[
  M_p(t)\to\infty,\qquad M_p(t)=o(t^{p-2})
\]
because \(p\ge4\).  Thus
\eqref{eq:conditional-truncated-moment-growth}--\eqref{eq:conditional-truncated-top-mode-error}
hold, and the conditional quotient-transfer
Proposition~\ref{thm:tail-replacements-proved-here} gives
\[
H(t)=
\sum_{m=2}^{N-1}A_{2m}(\nu)t^{-2m}
+\kappa_N\mu_2\,t^{-2N}M_p(t)
+o(t^{-2N}M_p(t)).
\]
Substituting \(M_p(t)=2c\log t+O(1)\) proves the displayed asymptotic in
the theorem.
\end{proof}

\subsubsection*{Monotone slowly varying verification}

The next lemma is the only place where slow variation is used.  The corollary
that follows is not a general regular-variation statement: it proves
membership in the monotone slowly varying subclass
\(\mathcal A_{p,R,c}^{\rm sv}\) of
Definition~\ref{def:admissible-borderline-tail-classes}, and hence the same
conditional quotient-transfer hypotheses
\eqref{eq:conditional-truncated-moment-growth}--\eqref{eq:conditional-truncated-top-mode-error},
from eventual monotonicity and slow variation for the displayed borderline
density.

\begin{lemma}[Slow variation supplies the quotient controls]%
\label{lem:slowly-varying-quotient-controls}
Let \(p\ge4\), \(R>0\), and \(c>0\).  Let \(L\) be eventually positive,
eventually monotone, and slowly varying on \([R,\infty)\).  Put
\[
  \mathcal M_L(t):=\int_R^t\frac{L(s)}s\,\dd s,
\]
and assume \(\mathcal M_L(t)\to\infty\).  Then, as \(t\to\infty\),
\begin{equation}\label{eq:slowly-varying-top-moment-growth}
  L(t)=o(\mathcal M_L(t)),\qquad
  \mathcal M_L(t)=o(t^\alpha)\quad\text{for every }\alpha>0 .
\end{equation}
Moreover, for every \(0\le n<p\), Karamata's theorem gives the sharper
tail equivalent
\begin{equation}\label{eq:slowly-varying-karamata-tail-equivalent}
t^{-n}\int_t^\infty x^{n-p-1}L(x)\,\dd x
\sim \frac{1}{p-n}t^{-p}L(t)
=o(t^{-p}\mathcal M_L(t)),
\end{equation}
and hence
\begin{equation}\label{eq:slowly-varying-lower-potter-control}
t^{-n}\int_t^\infty x^n c x^{-p-1}L(x)\,\dd x
=o(t^{-p}\mathcal M_L(t)),
\end{equation}
and, with \(r_t(x):=c x^{-p-1}\{L(x)-L(t)\}\) on \(R\le x\le t\),
\begin{equation}\label{eq:slowly-varying-local-potter-control}
t^{-(p+1)}\int_R^t x^{p+1}|r_t(x)|\,\dd x
=o(t^{-p}\mathcal M_L(t)),
\end{equation}
while
\begin{equation}\label{eq:slowly-varying-far-potter-control}
\int_t^\infty\left(1+\frac{x^2}{t^2}\right)
c x^{-p-1}L(x)\,\dd x
=O(t^{-p}L(t))=o(t^{-p}\mathcal M_L(t)).
\end{equation}
Finally, eventual monotonicity and Karamata's theorem imply
\begin{equation}\label{eq:slowly-varying-local-karamata-control}
t^{-(p+1)}\int_R^t |L(x)-L(t)|\,\dd x
\le
t^{-(p+1)}\left(\int_R^t L(x)\,\dd x+tL(t)\right)
=O(t^{-p}L(t))
=o(t^{-p}\mathcal M_L(t)).
\end{equation}
For \(0\le n<p\), the lower tail moment
\(\int_R^\infty x^{n-p-1}L(x)\,\dd x\) is finite.
\end{lemma}

\begin{proof}
We use only standard consequences of slow variation, recorded here at the
scales needed in the quotient argument.  Finite initial intervals do not affect
any of the asymptotic estimates, so we may increase \(R\) once so that \(L\) is
positive and monotone on \([R,\infty)\).  Potter's estimate says that, for
each \(\delta>0\), after this enlargement there is a constant \(C_\delta\) such
that, for \(x,t\ge R\),
\begin{equation}\label{eq:potter-bound-used-locally}
  \frac{L(x)}{L(t)}
  \le C_\delta\max\{(x/t)^\delta,(t/x)^\delta\}.
\end{equation}
This estimate is a consequence of the uniform convergence theorem for slowly
varying functions and is not an additional hypothesis.

First we compare \(L(t)\) with \(\mathcal M_L(t)\).  Fix \(A>1\).  By the
uniform convergence theorem, \(L(tu)/L(t)\to1\) uniformly for
\(u\in[1/A,1]\).  Hence, for \(t\) large,
\[
  \mathcal M_L(t)\ge\int_{t/A}^{t}\frac{L(s)}s\,\dd s
  \ge \frac12L(t)\log A .
\]
Since \(A\) is arbitrary and \(\mathcal M_L(t)\to\infty\), this gives
\(L(t)=o(\mathcal M_L(t))\).  Taking
\(\delta=\alpha/2\) in \eqref{eq:potter-bound-used-locally} with \(t=R\)
gives \(L(s)\le C_\alpha s^{\alpha/2}\) for large \(s\), and therefore
\[
  \mathcal M_L(t)\le C_\alpha\int_R^t s^{\alpha/2-1}\,\dd s
  =O(t^{\alpha/2})=o(t^\alpha).
\]
This proves \eqref{eq:slowly-varying-top-moment-growth}.

Let \(0\le n<p\) and set \(\beta=p-n>0\).  Choose
\(0<\delta<\beta\).  For \(x\ge t\), Potter's estimate gives
\[
  L(x)\le C_\delta L(t)(x/t)^\delta .
\]
Thus
\[
\begin{aligned}
t^{-n}\int_t^\infty x^n c x^{-p-1}L(x)\,\dd x
&\le C t^{-n}L(t)t^{-\delta}
   \int_t^\infty x^{-\beta-1+\delta}\,\dd x  \\
&\le C_{\beta,\delta}t^{-p}L(t)
 =o(t^{-p}\mathcal M_L(t)).
\end{aligned}
\]
The exact equivalent in
\eqref{eq:slowly-varying-karamata-tail-equivalent} is the standard Karamata
tail theorem applied to the regularly varying function
\(x\mapsto x^{n-p-1}L(x)\), whose index is less than \(-1\).
The same estimate with \(t=R\), or equivalently Potter with
\(\delta<\beta\), gives
\(\int_R^\infty x^{n-p-1}L(x)\,\dd x<\infty\).

For the local Taylor remainder, eventual monotonicity and Karamata's theorem
give \(\int_R^tL(x)\,\dd x=O(tL(t))\).  Consequently
\[
\begin{aligned}
t^{-(p+1)}\int_R^t x^{p+1}|r_t(x)|\,\dd x
&=ct^{-(p+1)}\int_R^t |L(x)-L(t)|\,\dd x  \\
&\le Ct^{-(p+1)}\left(\int_R^t L(x)\,\dd x+tL(t)\right)  \\
&=O(t^{-p}L(t))
 =o(t^{-p}\mathcal M_L(t)).
\end{aligned}
\]
This is also the displayed control
\eqref{eq:slowly-varying-local-karamata-control}.

Finally, applying
\eqref{eq:slowly-varying-karamata-tail-equivalent} with \(n=0\) and \(n=2\)
gives
\[
  \int_t^\infty x^{-p-1}L(x)\,\dd x
  \sim \frac1p t^{-p}L(t),
  \qquad
  t^{-2}\int_t^\infty x^{1-p}L(x)\,\dd x
  \sim \frac1{p-2}t^{-p}L(t).
\]
Together with \(L(t)=o(\mathcal M_L(t))\), this proves
\eqref{eq:slowly-varying-far-potter-control}.
\end{proof}

\begin{corollary}[Verified subclass: monotone slowly varying borderline tail]%
\label{cor:conditional-slowly-varying-borderline-tail}
Let \(N\ge3\), put \(p=2N-2\), and let \(R>0\), \(c>0\).  Let \(L\) be
slowly varying, eventually positive, eventually monotone, and nonnegative on
\([R,\infty)\), and let \(g\in L^1(\RR)\) be symmetric and nonnegative with
\(\operatorname{supp}g\subset(-R,R)\).  Suppose
\[
  f(x)=g(x)+c|x|^{-p-1}L(|x|)\mathbf1_{\{|x|\ge R\}}
\]
is normalized to have mass one, and let \(\nu\) be the corresponding
symmetric probability law.  Put
\[
M_p(t):=\int_{|x|\le t}x^pf(x)\,\dd x
       =\int_{|x|<R}x^pg(x)\,\dd x
         +2c\int_R^t\frac{L(s)}s\,\dd s .
\]
Within this monotone slowly varying subclass, assume that \(M_p(t)\to\infty\).
This divergence is used only to prove the quotient estimates displayed below;
it is not, by itself, an entropy hypothesis.
Then \(\nu\in\mathcal A_{p,R,c}^{\rm sv}\subset\mathcal A_{p,R,c}\), where
the classes are those of
Definition~\ref{def:admissible-borderline-tail-classes}.  With
\[
  \Delta_t(y):=\frac{(P_t*\nu)(ty)}{P_t(ty)}-1,
  \qquad H(t):=D_{\mathrm{KL}}(P_t*\nu\|P_t),
\]
one has, in \(L^\infty(\RR)\cap L^1(\omega)\),
\begin{equation}\label{eq:slowly-varying-quotient-expansion}
  \Delta_t(y)=
  \sum_{n=2}^{p-1}\mu_nu_n(y)t^{-n}
  +u_p(y)t^{-p}M_p(t)
  +o(t^{-p}M_p(t)),
\end{equation}
where \(\mu_n=\int x^nf(x)\,\dd x\) for \(2\le n<p\).  Consequently
\[
  M_p(t)\to\infty,\qquad M_p(t)=o(t^{p-2}),
\]
and the quotient expansion is exactly of the conditional-transfer form
\[
  \Delta_t(y)=
  \sum_{n=2}^{p-1}\mu_nu_n(y)t^{-n}
  +u_p(y)t^{-p}M_p(t)+e_t(y),
  \qquad
  \|e_t\|_\infty+\|e_t\|_{L^1(\omega)}=o(t^{-p}M_p(t)).
\]
Only after this quotient verification does
Proposition~\ref{thm:tail-replacements-proved-here} apply.  Therefore
\begin{equation}\label{eq:slowly-varying-entropy-expansion}
H(t)=
\sum_{m=2}^{N-1}A_{2m}(\nu)t^{-2m}
+\kappa_N\mu_2\,t^{-2N}M_p(t)
+o(t^{-2N}M_p(t)),
\end{equation}
where
\[
  \kappa_N=J(2,p)=(-1)^N(N-1)2^{-2N+2}.
\]
\end{corollary}

\begin{proof}
The law is symmetric, hence centred, and all odd moments below vanish.  The
compact core has all moments.  Since
\[
  M_p(t)=O(1)+2c\,\mathcal M_L(t),
  \qquad
  \mathcal M_L(t):=\int_R^t\frac{L(s)}s\,\dd s,
\]
the assumption \(M_p(t)\to\infty\) is equivalent to
\(\mathcal M_L(t)\to\infty\).  Lemma~\ref{lem:slowly-varying-quotient-controls}
therefore gives
\[
  L(t)=o(M_p(t)),\qquad M_p(t)=o(t^{p-2}),
\]
as well as the three quotient-scale estimates below with \(M_p(t)\) in place
of \(\mathcal M_L(t)\).  It also gives the finiteness of the lower tail
moments, so \(\mu_n=\int x^nf(x)\,\dd x\) exists for \(0\le n<p\).
The lower-moment tail replacement is
\begin{equation}\label{eq:slowly-varying-lower-moment-tail-replacement}
  t^{-n}\int_t^\infty x^n c x^{-p-1}L(x)\,\dd x=o(t^{-p}M_p(t)),
  \qquad 0\le n<p.
\end{equation}

For \(R\le x\le t\), write
\[
  r_t(x):=c x^{-p-1}\{L(x)-L(t)\}.
\]
The local Taylor-remainder control supplied by
Lemma~\ref{lem:slowly-varying-quotient-controls} is
\begin{equation}\label{eq:slowly-varying-local-taylor-bound}
t^{-(p+1)}\int_R^t x^{p+1}|r_t(x)|\,\dd x
=o(t^{-p}M_p(t)).
\end{equation}
The top truncated moment is the displayed \(M_p(t)\); equivalently
\begin{equation}\label{eq:slowly-varying-top-mode-identity}
  \int_{R\le |x|\le t}x^p c|x|^{-p-1}L(|x|)\,\dd x
  =M_p(t)+O(1),
\end{equation}
where the \(O(1)\) term is the compact core contribution.  The far-tail
quotient bound is exactly
\begin{equation}\label{eq:slowly-varying-far-tail-bound}
\left\|
\int_{|x|>t}\{R_{x/t}-1\}c|x|^{-p-1}L(|x|)\,\dd x
\right\|_{L^\infty\cap L^1(\omega)}
=o(t^{-p}M_p(t)),
\end{equation}
because \(|R_\varepsilon(y)-1|\le C(1+\varepsilon^2)\) uniformly in \(y\)
and \(\omega\) is a probability measure.

It remains to assemble the quotient.  On \(|x|<R\), the compact-core Taylor
expansion contributes the moments \(2\le n<p\), while its \(p\)th mode and
remainder are \(O(t^{-p})=o(t^{-p}M_p(t))\).  On \(R\le |x|\le t\), Taylor
expansion through degree \(p\), paired by symmetry, gives
\[
\sum_{n=2}^{p}u_n(y)t^{-n}
\int_{R\le |x|\le t}x^n c|x|^{-p-1}L(|x|)\,\dd x
+\mathcal R_t(y).
\]
The remainder \(\mathcal R_t\) is \(o(t^{-p}M_p(t))\) in
\(L^\infty\cap L^1(\omega)\): replacing \(L(x)\) by \(L(t)\) gives
\[
  t^{-(p+1)}L(t)\int_R^t \dd x
  =O(t^{-p}L(t))=o(t^{-p}M_p(t)),
\]
and the difference is controlled by
\eqref{eq:slowly-varying-local-taylor-bound}.  For \(2\le n<p\), replacing
the truncated lower moments by the full lower moments costs
\eqref{eq:slowly-varying-lower-moment-tail-replacement}.  The \(n=p\) term
is \(u_p(y)t^{-p}M_p(t)+O(t^{-p})\), hence has the asserted leading form.
Finally the far tail is \eqref{eq:slowly-varying-far-tail-bound}.  These
estimates prove \eqref{eq:slowly-varying-quotient-expansion}.  The growth
conclusions supplied by Lemma~\ref{lem:slowly-varying-quotient-controls} and
the quotient expansion verify
\eqref{eq:conditional-truncated-moment-growth}--\eqref{eq:conditional-truncated-top-mode-error},
so the conditional quotient-transfer
Proposition~\ref{thm:tail-replacements-proved-here} gives
\eqref{eq:slowly-varying-entropy-expansion}.
\end{proof}

\begin{lemma}[Symmetric \(p=2\) quotient tail split]
\label{lem:symmetric-p2-far-tail-quotient}
Let
\[
  S_a(y):=R_a(y)+R_{-a}(y)-2 .
\]
There is a universal constant \(C<\infty\) such that, for all \(t\ge \mathrm e\),
\begin{equation}\label{eq:symmetric-p2-far-tail-quotient}
  \left\|
  \int_t^\infty S_{x/t}(y)\,x^{-3}\,\dd x
  \right\|_{L^\infty(\RR)}
  +
  \left\|
  \int_t^\infty S_{x/t}(y)\,x^{-3}\,\dd x
  \right\|_{L^1(\omega)}
  \le Ct^{-2}.
\end{equation}
Equivalently, for each fixed \(c_0\),
\[
\sup_y\left|
c_0t^{-2}\int_1^\infty
\{R_\varepsilon(y)+R_{-\varepsilon}(y)-2\}\varepsilon^{-3}\,\dd\varepsilon
\right|
+\int_{\RR}\left|
c_0t^{-2}\int_1^\infty
\{R_\varepsilon(y)+R_{-\varepsilon}(y)-2\}\varepsilon^{-3}\,\dd\varepsilon
\right|\omega(\dd y)
\le C(c_0)t^{-2}.
\]
Moreover, if \(R\ge1\), then as \(t\to\infty\)
\begin{equation}\label{eq:symmetric-p2-full-tail-split}
\int_R^\infty S_{x/t}(y)x^{-3}\,\dd x
=2u_2(y)t^{-2}\log t+O(t^{-2})
\end{equation}
in \(L^\infty(\RR)\cap L^1(\omega)\), with the constant depending on \(R\)
but not on \(y\) or \(t\).
\end{lemma}

\begin{proof}
First prove the far-tail estimate.  After the change of variables \(a=x/t\),
the integral over \(x>t\) is
\[
  t^{-2}\int_1^\infty S_a(y)a^{-3}\,\dd a .
\]
It is therefore enough to prove a uniform bound in \(y\) for the last
integral.  Pairing the two translates before estimating is essential: the
unpaired estimate with \((1+x^2/t^2)x^{-3}\) would produce the divergent
\(\int_t^\infty x^{-1}\dd x\).  With
\(A=1+y^2+a^2\),
\[
\begin{aligned}
R_a(y)+R_{-a}(y)-2
&=(1+y^2)\left(\frac1{A-2ay}+\frac1{A+2ay}\right)-2  \\
&=
\frac{2a^2(3y^2-1-a^2)}
     {\{1+(y-a)^2\}\{1+(y+a)^2\}} .
\end{aligned}
\]
Thus, writing \(Y=|y|\),
\[
 |S_a(y)|a^{-3}
 \le
 \frac{2(a^2+3Y^2+1)}
      {a\{1+(a-Y)^2\}\{1+(a+Y)^2\}}
 \le
 \frac{C}{a\{1+(a-Y)^2\}},
\]
because \(a^2+3Y^2+1\le C\{1+(a+Y)^2\}\).
If \(Y\le2\), then
\[
  \int_1^\infty \frac{\dd a}{a\{1+(a-Y)^2\}}
  \le
  \int_1^\infty \frac{\dd a}{1+(a-2)^2}<\infty .
\]
If \(Y>2\), split the integral over
\[
[1,Y/2],\qquad
[Y/2,2Y]\cap\{|a-Y|>1\},\qquad
\{|a-Y|\le1\},\qquad
[2Y,\infty).
\]
On the first interval \(|a-Y|\ge Y/2\), so the contribution is
\[
  O\!\left(Y^{-2}\int_1^{Y/2}a^{-1}\dd a\right)
  =O(Y^{-2}\log Y).
\]
On the middle interval \(a\asymp Y\) and \(|a-Y|>1\), whence
\[
  \int_{[Y/2,2Y]\cap\{|a-Y|>1\}}
  \frac{\dd a}{a\{1+(a-Y)^2\}}
  \le \frac{C}{Y}\int_{|u|>1}\frac{\dd u}{1+u^2}
  =O(Y^{-1}).
\]
On the local interval \(|a-Y|\le1\),
\[
  \int_{|a-Y|\le1}
  \frac{\dd a}{a\{1+(a-Y)^2\}}
  \le \frac{C}{Y}\int_{-1}^{1}\frac{\dd u}{1+u^2}
  =O(Y^{-1}).
\]
This local \(O(Y^{-1})\) term is also \(L^1(\omega)\), since
\(\omega(\dd y)\asymp Y^{-2}\dd Y\) for \(Y>2\).  Finally, on
\([2Y,\infty)\), \(|a-Y|\ge a/2\), so the contribution is
\[
  O\!\left(\int_{2Y}^{\infty}a^{-3}\dd a\right)=O(Y^{-2}).
\]
These four bounds are uniform in \(Y\).  Hence
\[
  \sup_{y\in\RR}\int_1^\infty |S_a(y)|a^{-3}\,\dd a<\infty .
\]
This gives the \(L^\infty\) estimate in
\eqref{eq:symmetric-p2-far-tail-quotient}; the \(L^1(\omega)\) estimate
follows because \(\omega\) is a probability measure.

It remains to record the logarithmic near-tail part.  For \(R\le x\le t\),
Taylor expansion through order \(3\), paired at \(x\) and \(-x\), cancels the
odd modes and yields
\[
S_{x/t}(y)=2u_2(y)(x/t)^2+O((x/t)^4)
\qquad(0\le x/t\le1),
\]
uniformly in \(y\).  Therefore
\[
\int_R^t S_{x/t}(y)x^{-3}\,\dd x
=2u_2(y)t^{-2}\int_R^t x^{-1}\,\dd x
+O\!\left(t^{-4}\int_R^t x\,\dd x\right)
=2u_2(y)t^{-2}\log t+O(t^{-2}).
\]
Adding \eqref{eq:symmetric-p2-far-tail-quotient} gives
\eqref{eq:symmetric-p2-full-tail-split}.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{prop:n2-borderline-infinite-variance-layer}]
The law is symmetric.  Its absolute first moment is finite, while its second
moment is infinite, since
\[
  2c\int_{\mathrm e}^{\infty}x^{-2}\,\dd x<\infty,
  \qquad
  2c\int_{\mathrm e}^{\infty}x^{-1}\,\dd x=\infty .
\]
The truncated second moment is
\[
  \EE[X^2\mathbf1_{\{|X|\le t\}}]
  =O(1)+2c\int_{\mathrm e}^{t}x^{-1}\,\dd x
  \sim 2c\log t .
\]

For \(t\ge \mathrm e\), put
\[
  \Delta_t(y)=\frac{(P_t*\nu)(ty)}{P_t(ty)}-1 .
\]
We prove the quotient expansion
\begin{equation}\label{eq:n2-borderline-quotient-expansion}
\Delta_t(y)=2c\,u_2(y)t^{-2}\log t+E_t(y),
\qquad
\|E_t\|_\infty+\|E_t\|_{L^1(\omega)}=O(t^{-2})
\end{equation}
in \(L^\infty\cap L^1(\omega)\).

The compact core \(|x|\le1\) has bounded support and is symmetric.  Expanding
the paired expression \(R_{x/t}(y)+R_{-x/t}(y)-2\) through order \(3\) gives a
contribution
\[
  t^{-2}u_2(y)
  \int_{|x|\le1}x^2\,\frac{1-c\mathrm e^{-2}}{2}\,\dd x
  +O(t^{-4}),
\]
uniformly in \(y\), which is part of \(E_t\).  For the tail, symmetry gives
\[
  c\int_{\mathrm e}^{\infty}
  \{R_{x/t}(y)+R_{-x/t}(y)-2\}x^{-3}\,\dd x .
\]
Lemma~\ref{lem:symmetric-p2-far-tail-quotient}, with \(R=\mathrm e\), gives
the full tail contribution
\[
  2c\,u_2(y)t^{-2}\log t+O(t^{-2})
\]
in both \(L^\infty\) and \(L^1(\omega)\).  The pairing in that lemma is the
separate \(p=2\) tail argument: the unpaired far-tail quotient bound would be
too crude for the infinite-variance layer.
This proves \eqref{eq:n2-borderline-quotient-expansion}.

Since
\(\|\Delta_t\|_\infty=O(t^{-2}\log t)\),
\[
D_{\mathrm{KL}}(P_t*\nu\|P_t)
=\frac12\int_{\RR}\Delta_t(y)^2\,\omega(\dd y)+O(t^{-6}(\log t)^3).
\]
Using \(J(2,2)=\int u_2^2\,\dd\omega=1/4\), and using
\(\|E_t\|_\infty+\|E_t\|_{L^1(\omega)}=O(t^{-2})\) for the cross and error
terms, the quadratic term is
\[
\frac12(2c)^2J(2,2)t^{-4}(\log t)^2+O(t^{-4}\log t)
=\frac{c^2}{2}t^{-4}(\log t)^2+O(t^{-4}\log t).
\]
\end{proof}

\subsection{Sharp leading threshold}

\begin{lemma}[Second-mode transform of a Cauchy translate]%
\label{lem:second-mode-cauchy-translate-transform}
For \(\varepsilon\in\RR\),
\[
\int_{\RR}u_2(y)\{R_\varepsilon(y)-1\}\,\omega(\dd y)
=\frac{4\varepsilon^2}{(4+\varepsilon^2)^2}.
\]
\end{lemma}

\begin{proof}
Since \(\int_{\RR}u_2\,\dd\omega=0\), it is enough to compute
\(\int u_2R_\varepsilon\,\dd\omega\).  We use the same Cayley/Fourier
normalization as in Theorem~\ref{thm:finite-fourier-mode-integral-algorithm}.
Put \(y=\tan\theta\) and \(z=e^{2\ii\theta}\).  Then
\(\omega(\dd y)=\dd\theta/\pi\), Haar integration on the \(z\)-circle is
constant-term extraction, and
\[
  Q_2(z)=u_2(\tan\theta)
  =-\frac14(z+z^{-1}+z^2+z^{-2}).
\]
For \(\varepsilon=0\), \(R_0\equiv1\) and the integral is zero.  Assume
\(\varepsilon\ne0\).  From
\[
R_\varepsilon(\tan\theta)
=\{1-\varepsilon\sin2\theta+\varepsilon^2\cos^2\theta\}^{-1}
\]
one obtains, after multiplying numerator and denominator by \(z\),
\[
R_\varepsilon(\tan\theta)
=
\frac{z}
{\frac{\varepsilon(\varepsilon+2\ii)}4z^2
+(1+\varepsilon^2/2)z
+\frac{\varepsilon(\varepsilon-2\ii)}4}
=
\frac{\alpha}{z-\alpha}-\frac{\beta}{z-\beta},
\]
where
\[
\alpha=-\frac{\varepsilon}{\varepsilon+2\ii},
\qquad
\beta=\frac{2\ii-\varepsilon}{\varepsilon}.
\]
Here \(|\alpha|<1<|\beta|\), so on the unit circle the Laurent expansion is
\[
R_\varepsilon(\tan\theta)
=\sum_{n=0}^{\infty}\beta^{-n}z^n
+\sum_{n=1}^{\infty}\alpha^nz^{-n}.
\]
Hence
\[
\begin{aligned}
\int_{\RR}u_2(y)R_\varepsilon(y)\,\omega(\dd y)
&=[z^0]\,Q_2(z)R_\varepsilon(\tan\theta)\\
&=-\frac14\left(\alpha+\beta^{-1}+\alpha^2+\beta^{-2}\right)\\
&=-\frac14\left\{
-\frac{\varepsilon}{\varepsilon+2\ii}
-\frac{\varepsilon}{\varepsilon-2\ii}
+\frac{\varepsilon^2}{(\varepsilon+2\ii)^2}
+\frac{\varepsilon^2}{(\varepsilon-2\ii)^2}
\right\}  \\
&=-\frac14\left\{
-\frac{2\varepsilon^2}{4+\varepsilon^2}
+\frac{2\varepsilon^2(\varepsilon^2-4)}
{(4+\varepsilon^2)^2}
\right\}  \\
&=-\frac14\left\{
-\frac{16\varepsilon^2}{(4+\varepsilon^2)^2}
\right\}  \\
&=\frac{4\varepsilon^2}{(4+\varepsilon^2)^2}.
\end{aligned}
\]
The last line is an elementary rational simplification; equivalently, the
bracket equals \(-16\varepsilon^2/(4+\varepsilon^2)^2\).  This proves the
claimed transform.
\end{proof}

\begin{lemma}[Bounded-test Pinsker for the second-mode projection]%
\label{lem:sharp-threshold-bounded-test-pinsker}
Let \(P\ll Q\), let \(D(P\|Q)\in[0,\infty]\) be the extended
Kullback--Leibler divergence, and let \(\|\varphi\|_\infty\le1\).  Then
\[
D(P\|Q)
\ge
\frac12\left|\int \varphi\,\dd(P-Q)\right|^2 .
\]
\end{lemma}

\begin{proof}
If \(D(P\|Q)=\infty\), the assertion is trivial.  Otherwise the usual
Csisz{\'a}r--Pinsker inequality gives
\(\|P-Q\|_1^2\le2D(P\|Q)\).  Since
\(\left|\int\varphi\,\dd(P-Q)\right|\le\|P-Q\|_1\), the claim follows.
\end{proof}

\begin{theorem}[Sharp analytic variance threshold for the leading KL coefficient]%
\label{thm:sharp-leading-kl-variance-threshold}
Let \(\nu\) have finite mean \(\bar{\gamma}\), and put
\[
H(t)=D_{\rm KL}(P_t*\nu\|P_t(\cdot-\bar{\gamma})).
\]
With the extended nonnegative convention
\eqref{eq:extended-kl-definition} for \(D_{\rm KL}\),
\[
H(t)\in[0,\infty]\qquad(t>0).
\]
Then
\[
\EE X_c^2<\infty
\quad\Longleftrightarrow\quad
\limsup_{t\to\infty}t^4H(t)<\infty .
\]
When the variance is finite,
\[
H(t)<\infty\qquad(t>0)
\]
and
\[
\lim_{t\to\infty}t^4H(t)=\frac{(\EE X_c^2)^2}{8},
\]
and when the variance is infinite,
\[
\lim_{t\to\infty}t^4H(t)=+\infty .
\]
The last limit is meant in the extended sense; no finite-time finiteness of
\(H(t)\) is asserted in the infinite-variance case, and the conclusion
includes the case \(H(t)=+\infty\) along an unbounded set of times.
\end{theorem}

\begin{proof}
Finite variance is Theorem~\ref{thm:poisson-kl-even-orders} with \(N=2\).
In that case Lemma~\ref{lem:finite-time-poisson-kl-finite} also gives
\(H(t)<\infty\) for every \(t>0\).  Hence the finite-variance limit is an
ordinary real limit.

For the converse, write
\[
h_t=P_t*\nu,\qquad g_t=P_t(\cdot-\bar{\gamma}),\qquad
\delta_t(y)=\frac{h_t(\bar{\gamma}+ty)}{g_t(\bar{\gamma}+ty)}-1 .
\]
Let \(\varphi_t(x)=u_2((x-\bar{\gamma})/t)\).  Since
\(\|\varphi_t\|_\infty=\|u_2\|_\infty\le1\),
Lemma~\ref{lem:sharp-threshold-bounded-test-pinsker} gives, in the
extended-valued sense,
\begin{equation}\label{eq:bounded-test-pinsker-second-mode}
2H(t)\ge
\left|
\int_{\RR}\varphi_t(x)\{h_t(x)-g_t(x)\}\dd x
\right|^2
=
\left|
\int_{\RR}u_2(y)\delta_t(y)\,\omega(\dd y)
\right|^2 .
\end{equation}
This use of Pinsker involves only the bounded test \(\varphi_t\); it does not
assume that \(H(t)\) is finite.

It remains to identify the projection without using an \(X_c^2\)-integrability
hypothesis.  For fixed \(t>0\),
\[
\delta_t(y)=\int_{\RR}\{R_{X_c/t}(y)-1\}\,\dd\nu(\gamma),
\]
where the integrand is bounded as a function of \(\gamma\) for each fixed
\(y\).  Moreover
\[
R_\varepsilon(y)\omega(\dd y)
=\frac{\dd y}{\pi(1+(y-\varepsilon)^2)},
\qquad
\int_{\RR}|R_\varepsilon-1|\,\dd\omega\le2 .
\]
Since \(\|u_2\|_\infty\le1\), Tonelli gives the uniform bound
\[
\int_{\RR}\!\!\int_{\RR}
|u_2(y)|\,|R_{X_c/t}(y)-1|\,\omega(\dd y)\,\nu(\dd\gamma)
\le2 .
\]
Thus Fubini is justified against the possibly infinite-variance law \(\nu\),
and Lemma~\ref{lem:second-mode-cauchy-translate-transform} gives
\[
B(t):=
\int_{\RR}u_2(y)\delta_t(y)\,\omega(\dd y)
=
\EE\!\left[\frac{4(X_c/t)^2}{(4+(X_c/t)^2)^2}\right]\ge0 .
\]
Combining this identity with \eqref{eq:bounded-test-pinsker-second-mode}
yields
\[
t^4H(t)\ge\frac12\{t^2B(t)\}^2 .
\]
For fixed \(M<\infty\),
\[
t^2B(t)\ge
\frac{4}{(4+(M/t)^2)^2}
\EE[X_c^2\mathbf1_{\{|X_c|\le M\}}],
\]
and the right side tends to
\(\frac14\EE[X_c^2\mathbf1_{\{|X_c|\le M\}}]\).  Letting
\(M\to\infty\) proves divergence when the variance is infinite.
\end{proof}

\subsection{Multidimensional D1--D4 quotient and covariance package}

For \(d\ge1\), let
\[
P_t^{(d)}(x)=c_d\,\frac{t}{(t^2+|x|^2)^{(d+1)/2}},
\qquad
c_d=\frac{\Gamma((d+1)/2)}{\pi^{(d+1)/2}},
\qquad
\Omega_d(\dd y)=P_1^{(d)}(y)\dd y.
\]
Thus \(P_t^{(d)}\) is normalized to have total mass one and
\[
P_t^{(d)}(ty)t^d\,\dd y=\Omega_d(\dd y)
=c_d(1+|y|^2)^{-(d+1)/2}\dd y .
\]
For \(z\in\RR^d\), write
\[
F_y(z)=\left(\frac{1+|y|^2}{1+|y-z|^2}\right)^{(d+1)/2}.
\]
Let \(U_\alpha^{(d)}\) be the Taylor coefficients of \(F_y(z)\) at \(z=0\).

\begin{lemma}[Local multidimensional quotient--Taylor remainder]%
\label{lem:local-multidim-quotient-taylor-remainder-core}
Let \(d\ge1\), and let
\[
F_y(z)=\left(\frac{1+|y|^2}{1+|y-z|^2}\right)^{(d+1)/2}.
\]
For a multiindex \(\alpha\), put
\[
U_\alpha^{(d)}(y)=\frac1{\alpha!}\partial_z^\alpha F_y(0).
\]
For each integer \(L\ge2\) there is a constant \(C_{d,L}<\infty\),
depending only on \(d\) and \(L\), such that
\begin{equation}\label{eq:local-multidim-coefficient-bound}
\sup_{y\in\RR^d}|U_\alpha^{(d)}(y)|\le C_{d,L}
\qquad (|\alpha|\le L),
\end{equation}
and
\begin{equation}\label{eq:local-multidim-taylor-remainder}
\sup_{y\in\RR^d}\left|
F_y(z)-\sum_{|\alpha|\le L}U_\alpha^{(d)}(y)z^\alpha
\right|
\le C_{d,L}|z|^{L+1}
\qquad (|z|\le1).
\end{equation}
\end{lemma}

\begin{proof}
Put \(a=(d+1)/2\).  For each multiindex \(\beta\), repeated
differentiation in \(z\) gives
\begin{equation}\label{eq:local-multidim-derivative-structure}
\partial_z^\beta F_y(w)
=(1+|y|^2)^a
\frac{P_\beta(y-w)}{(1+|y-w|^2)^{a+|\beta|}},
\end{equation}
where \(P_\beta\) is a polynomial of degree at most \(|\beta|\), depending
only on \(d\) and \(\beta\).  Indeed, \(z\)-differentiation is
\(-\xi\)-differentiation for \(\xi=y-z\), and each differentiation of
\((1+|\xi|^2)^{-b}\) raises the denominator exponent by one and multiplies
by a polynomial whose degree increases by at most one.

If \(|w|\le1\) and \(|y|>2\), then \(|y-w|\ge |y|/2\), and
\[
|\partial_z^\beta F_y(w)|
\le C_{d,\beta}
(1+|y|^2)^a
\frac{(1+|y|)^{|\beta|}}{(1+|y|^2)^{a+|\beta|}}
\le C_{d,\beta}.
\]
If \(|y|\le2\), the same derivative is bounded uniformly for \(|w|\le1\)
because the denominator in \eqref{eq:local-multidim-derivative-structure}
is at least \(1\).  Thus
\[
\sup_{\substack{y\in\RR^d,\ |w|\le1\\ |\beta|\le L+1}}
|\partial_z^\beta F_y(w)|<\infty .
\]
Taking \(w=0\) and \(|\beta|\le L\) proves
\eqref{eq:local-multidim-coefficient-bound}.  The multivariable Taylor
formula at \(z=0\), with integral remainder and the displayed derivative
bound for \(|\beta|=L+1\), gives
\eqref{eq:local-multidim-taylor-remainder}.
\end{proof}

\medskip
\noindent\textbf{Multidimensional radial Poisson jet and existence
hypotheses.}
For \(d\ge1\) and \(L\in\mathbb N\) with \(L\ge2\), first define the
large-\(t\) entropy-existence clause \((\mathrm{EX}_{d,L})\) as follows.
With
\[
h_t=P_t^{(d)}*\nu,\qquad g_t=P_t^{(d)}(\cdot-\bar{\gamma}),
\qquad
\delta_t^{(d)}(y)=
\frac{h_t(\bar{\gamma}+ty)}{g_t(\bar{\gamma}+ty)}-1,
\]
one has, for all sufficiently large \(t\),
\[
1+\delta_t^{(d)}(y)>0,\qquad
\int_{\RR^d}\delta_t^{(d)}(y)\,\Omega_d(\dd y)=0,\qquad
\int_{\RR^d}\Phi(\delta_t^{(d)}(y))\,\Omega_d(\dd y)<\infty,
\]
where \(\Phi(s)=(1+s)\log(1+s)-s\).  Equivalently, the matched KL integral is
finite for all large \(t\) and is represented by
\[
D_{\rm KL}(h_t\|g_t)
=\int_{\RR^d}\Phi(\delta_t^{(d)}(y))\,\Omega_d(\dd y).
\]
This clause is recorded separately because the covariance algebra below does
not imply it.  It is supplied either by the quotient hypothesis itself or, in
Theorem~\ref{thm:multidimensional-density-quotient}, by the finite
\(L\)-moment estimate.

We say that \((\nu,d,L)\) satisfies the matched Poisson quotient jet
\((\mathrm{MPJ}_{d,L})\) when \((\mathrm{EX}_{d,L})\) holds, \(\nu\) is a
probability law on \(\RR^d\), the mean \(\bar{\gamma}\) exists, the centred
tensor moments
\[
\mu_\alpha=\int_{\RR^d}(\gamma-\bar{\gamma})^\alpha\,\dd\nu(\gamma)
\qquad(2\le|\alpha|\le L)
\]
exist, and the scaled quotient has the jet
\begin{equation}\label{eq:mpj-dl-definition}
\begin{aligned}
(\mathrm{MPJ}_{d,L}):\qquad
\delta_t^{(d)}(y)
&=
\sum_{2\le\lvert\alpha\rvert\le L}
\mu_\alpha U_\alpha^{(d)}(y)t^{-\lvert\alpha\rvert}
+\rho_{L,t}^{(d)}(y),\\
\lVert\rho_{L,t}^{(d)}\rVert_\infty+
\lVert\rho_{L,t}^{(d)}\rVert_{L^1(\Omega_d)}
&=o(t^{-L}),\qquad
\lVert\delta_t^{(d)}\rVert_\infty=O(t^{-2}).
\end{aligned}
\end{equation}
This displayed quotient expansion is the hypothesis used by the entropy
transfer below; its definition has no dimension-dependent lower bound beyond
\(L\ge2\).  Lemma~\ref{lem:global-multidim-quotient-remainder-core} is the
deterministic and integrated quotient-remainder input used to verify this
hypothesis from finite moments, with constants depending only on \(d,L\);
then Theorem~\ref{thm:multidimensional-density-quotient} proves that finite
centred \(L\)th absolute moment implies \((\mathrm{MPJ}_{d,L})\) in the
range \(L\ge d+1\).

The four front-end statements below keep the logical layers separate:
D1 transfers the conditional quotient jet to entropy, D2 supplies that jet
from finite moment, D3 evaluates the covariance algebra under finite
covariance alone, and D4 is the finite \((d+1)\)-moment leading KL
coefficient consequence.

\begin{lemma}[Global multidimensional quotient remainder]%
\label{lem:global-multidim-quotient-remainder-core}
Let \(d\ge1\), \(a=(d+1)/2\), and
\[
F_y(z)=\left(\frac{1+|y|^2}{1+|y-z|^2}\right)^a .
\]
For a multiindex \(\alpha\), put
\[
U_\alpha^{(d)}(y)=\frac1{\alpha!}\partial_z^\alpha F_y(0).
\]
If \(L\in\mathbb N\) and \(L\ge d+1\), then there is a constant
\(C_{d,L}<\infty\), depending only on \(d\) and \(L\), such that
\begin{equation}\label{eq:multidim-coefficient-bound}
\sup_{y\in\RR^d}|U_\alpha^{(d)}(y)|\le C_{d,L}
\qquad (|\alpha|\le L),
\end{equation}
\begin{equation}\label{eq:multidim-small-translation-remainder}
\sup_{y\in\RR^d}\left|
F_y(z)-\sum_{|\alpha|\le L}U_\alpha^{(d)}(y)z^\alpha
\right|
\le C_{d,L}|z|^{L+1}
\qquad (|z|\le1),
\end{equation}
and
\begin{equation}\label{eq:multidim-large-translation-remainder}
\sup_{y\in\RR^d}\left|
F_y(z)-\sum_{|\alpha|\le L}U_\alpha^{(d)}(y)z^\alpha
\right|
\le C_{d,L}|z|^{L}
\qquad (|z|>1).
\end{equation}
Equivalently, for all \(y,z\),
\begin{equation}\label{eq:multidim-global-translation-remainder}
\left|
F_y(z)-\sum_{|\alpha|\le L}U_\alpha^{(d)}(y)z^\alpha
\right|
\le C_{d,L}\left(
|z|^{L+1}{\bf1}_{\{|z|\le1\}}
{}
+{}|z|^L{\bf1}_{\{|z|>1\}}
\right).
\end{equation}
Consequently, if \(X\) is an \(\RR^d\)-valued random vector with
\(\EE|X|^L<\infty\), \(\Omega_d\) is the Cauchy probability measure above,
and
\[
r_t(y):=\EE\left[
F_y(X/t)-\sum_{|\alpha|\le L}U_\alpha^{(d)}(y)(X/t)^\alpha
\right],
\]
then
\begin{equation}\label{eq:multidim-integrated-remainder-bound}
\|r_t\|_{L^\infty(\RR^d)}+\|r_t\|_{L^1(\Omega_d)}
\le C_{d,L}t^{-L}\EE\left[
|X|^L\frac{|X|}{t}{\bf1}_{\{|X|\le t\}}
{}
+{}|X|^L{\bf1}_{\{|X|>t\}}
\right]=o(t^{-L}).
\end{equation}
\end{lemma}

\begin{proof}
The coefficient bound \eqref{eq:multidim-coefficient-bound} and the
small-translation remainder
\eqref{eq:multidim-small-translation-remainder} are exactly
Lemma~\ref{lem:local-multidim-quotient-taylor-remainder-core}.  Thus the
constant in these two estimates depends only on \(d,L\).

It remains to handle large translations.  Fix \(|z|>1\).  First suppose
\(|y-z|\le |z|/2\).  Then \(|y|\le3|z|/2\), and hence
\[
F_y(z)\le (1+|y|^2)^a\le C_d|z|^{d+1}.
\]
On the complementary region \(|y-z|>|z|/2\), the elementary estimate
\[
1+|y|^2
\le 2(1+|y-z|^2+|z|^2)
\le 2(1+|z|^2)(1+|y-z|^2)
\]
gives
\[
F_y(z)
\le 2^a(1+|z|^2)^a
\le C_d|z|^{d+1}
\qquad(|z|>1).
\]
Thus, uniformly in \(y\),
\[
\sup_y F_y(z)\le C_d|z|^{d+1}\le C_{d,L}|z|^L
\qquad(|z|>1),
\]
where the last inequality is precisely where \(L\ge d+1\) is used.  The
coefficient bound from the local lemma gives, again uniformly in \(y\),
\[
\sum_{|\alpha|\le L}|U_\alpha^{(d)}(y)|\,|z|^{|\alpha|}
\le C_{d,L}\sum_{k=0}^{L}|z|^k
\le C_{d,L}'|z|^L
\qquad(|z|>1).
\]
Consequently
\[
\sup_y\left(
|F_y(z)|
+\sum_{|\alpha|\le L}|U_\alpha^{(d)}(y)|\,|z|^{|\alpha|}
\right)
\le C_{d,L}|z|^L
\qquad(|z|>1),
\]
which proves \eqref{eq:multidim-large-translation-remainder}.  Combining
this estimate with the small-translation remainder proves the global form
\eqref{eq:multidim-global-translation-remainder}.

Set \(z=X/t\) in \eqref{eq:multidim-global-translation-remainder}.  Taking
expectations gives the pointwise bound, uniform in \(y\),
\[
|r_t(y)|
\le
C_{d,L}\left[
t^{-(L+1)}
\EE\!\left(|X|^{L+1}{\bf1}_{\{|X|\le t\}}\right)
+t^{-L}
\EE\!\left(|X|^L{\bf1}_{\{|X|>t\}}\right)
\right].
\]
Equivalently,
\[
t^{-(L+1)}
\EE\!\left(|X|^{L+1}{\bf1}_{\{|X|\le t\}}\right)
=
t^{-L}
\EE\!\left[
|X|^L\frac{|X|}{t}{\bf1}_{\{|X|\le t\}}
\right]
=o(t^{-L}),
\]
because \((|X|/t){\bf1}_{\{|X|\le t\}}\le1\) and converges pointwise to
zero, while \(|X|^L\) is integrable.  Also
\[
t^{-L}\EE\!\left(|X|^L{\bf1}_{\{|X|>t\}}\right)=o(t^{-L})
\]
by the absolute continuity of the integral for the integrable function
\(|X|^L\).  These two estimates are independent of \(y\), and therefore give
the \(L^\infty\) part of \eqref{eq:multidim-integrated-remainder-bound}.
The \(L^1(\Omega_d)\) part follows from the same uniform bound because
\(\Omega_d\) is a probability measure.
\end{proof}

\begin{theorem}[D1 conditional jet transfer]%
\label{thm:multidimensional-poisson-entropy-jets}
Let \(d\ge1\), let \(L\in\mathbb N\) satisfy \(L\ge2\), let \(\nu\) be a
probability law on \(\RR^d\), and assume that \((\nu,d,L)\) satisfies the full
\((\mathrm{MPJ}_{d,L})\) package.  Thus the large-\(t\) existence clause
\((\mathrm{EX}_{d,L})\) holds, and
\[
\delta_t^{(d)}(y)
=\sum_{2\le\lvert\alpha\rvert\le L}
\mu_\alpha U_\alpha^{(d)}(y)t^{-\lvert\alpha\rvert}
+\rho_{L,t}^{(d)}(y),
\]
with
\[
\lVert\rho_{L,t}^{(d)}\rVert_\infty+
\lVert\rho_{L,t}^{(d)}\rVert_{L^1(\Omega_d)}=o(t^{-L}),
\qquad
\lVert\delta_t^{(d)}\rVert_\infty=O(t^{-2}).
\]
Then
\[
D_{\mathrm{KL}}\bigl(P_t^{(d)}*\nu\,\|\,P_t^{(d)}(\cdot-\bar{\gamma})\bigr)
=\sum_{m=4}^{L+2}\mathcal A_m^{(d)}(\nu)t^{-m}
+o(t^{-(L+2)}),
\]
where
\[
\mathcal A_m^{(d)}(\nu)=
\sum_{r=2}^{\lfloor m/2\rfloor}
\frac{(-1)^r}{r(r-1)}
\sum_{\substack{\alpha_1,\ldots,\alpha_r\in\NN_0^d\\
|\alpha_1|+\cdots+|\alpha_r|=m\\
2\le|\alpha_j|\le L\ \mathrm{for}\ 1\le j\le r}}
\mu_{\alpha_1}\cdots\mu_{\alpha_r}
\int_{\RR^d}\prod_{j=1}^r
U_{\alpha_j}^{(d)}(y)\,\Omega_d(\dd y).
\]
Moreover, all odd retained coefficients vanish:
\[
\mathcal A_m^{(d)}(\nu)=0
\qquad\text{for every odd }m,\quad 4\le m\le L+2.
\]
\end{theorem}

\paragraph{Remark.}
The coefficient formula includes the multidimensional parity cancellation:
\(U_\alpha^{(d)}(-y)=(-1)^{|\alpha|}U_\alpha^{(d)}(y)\), while
\(\Omega_d\) is even.  Hence every summand of \(\mathcal A_m^{(d)}\) with
odd total order \(m\) integrates to zero, so all retained odd layers vanish.
For \(d=1\), \(\Sigma_0=0\), \(\operatorname{tr}\Sigma=\mu_2\), and the
quadratic form evaluated below gives \(c_1^{\rm iso}=1/8\).  Therefore, once
the quotient-jet hypothesis of this theorem is in force,
\(\mathcal A_4^{(1)}=A_4=\mu_2^2/8\), matching the one-dimensional
Cayley--Chebyshev coefficient.

\begin{proof}
Write
\[
D_j^{(d)}(y)=\sum_{|\alpha|=j}\mu_\alpha U_\alpha^{(d)}(y),\qquad
d_t(y)=\sum_{j=2}^{L}D_j^{(d)}(y)t^{-j}.
\]
\((\mathrm{MPJ}_{d,L})\) gives
\[
\delta_t^{(d)}=d_t+\rho_{L,t}^{(d)},\qquad
\|d_t\|_\infty=O(t^{-2}),\qquad
\|\rho_{L,t}^{(d)}\|_\infty+
\|\rho_{L,t}^{(d)}\|_{L^1(\Omega_d)}=o(t^{-L}),
\]
and also \(\|\delta_t^{(d)}\|_\infty=O(t^{-2})\).  Since
\[
P_t^{(d)}(ty)t^d\,\dd y=P_1^{(d)}(y)\dd y=\Omega_d(\dd y)
\]
and \((\mathrm{EX}_{d,L})\) gives the large-\(t\) entropy representation
\[
D_{\rm KL}\bigl(P_t^{(d)}*\nu\,\|\,P_t^{(d)}(\cdot-\bar{\gamma})\bigr)
=\int_{\RR^d}\Phi(\delta_t^{(d)}(y))\,\Omega_d(\dd y),
\]
where \(\Phi(s)=(1+s)\log(1+s)-s\).
By Proposition~\ref{prop:finite-quotient-jet-implies-kl-coefficient-jet},
applied with \((E,\lambda)=(\RR^d,\Omega_d)\), \(J=L\), and
\(b_j=D_j^{(d)}\), only pure powers of the finite quotient polynomial
\(d_t\) contribute through order \(L+2\).  Collecting the coefficient of
\(t^{-m}\) in
\[
\sum_{r=2}^{\lfloor m/2\rfloor}
\frac{(-1)^r}{r(r-1)}d_t(y)^r
\]
gives precisely the displayed formula for \(\mathcal A_m^{(d)}(\nu)\), and
all omitted layers are \(o(t^{-(L+2)})\).
\end{proof}

The restriction \(L\ge d+1\) is not used in this conditional entropy-transfer
argument.  It appears in the following finite-moment verification, where the
large-translation remainder must be controlled by the available
\(L\)-absolute moment.

\begin{theorem}[D2 moment supply]%
\label{thm:multidimensional-density-quotient}
Let \(d\ge1\), let \(L\in\mathbb N\) satisfy \(L\ge d+1\), and let \(\nu\) be
a probability measure on \(\RR^d\) with mean \(\bar{\gamma}\) and finite centred
\(L\)th absolute moment.  Put
\[
\begin{gathered}
X_c=\gamma-\bar{\gamma},\\
\mu_\alpha=\int_{\RR^d}X_c^\alpha\,\dd\nu(\gamma)
\qquad(|\alpha|\le L).
\end{gathered}
\]
Then, for
\[
\delta_t^{(d)}(y):=
\frac{(P_t^{(d)}*\nu)(\bar{\gamma}+ty)}
{P_t^{(d)}(ty)}-1,
\]
\begin{equation}\label{eq:multidim-density-quotient-expansion-main}
\delta_t^{(d)}(y)=
\sum_{2\le|\alpha|\le L}\mu_\alpha U_\alpha^{(d)}(y)t^{-|\alpha|}
+\rho_{L,t}^{(d)}(y),
\end{equation}
where the remainder satisfies the explicit \(L^\infty\cap L^1(\Omega_d)\)
bound
\begin{equation}\label{eq:multidim-density-quotient-explicit-remainder}
\begin{aligned}
\|\rho_{L,t}^{(d)}\|_\infty+
\|\rho_{L,t}^{(d)}\|_{L^1(\Omega_d)}
&\le C_{d,L}t^{-L}\EE\left[
|X_c|^L\frac{|X_c|}{t}{\bf1}_{\{|X_c|\le t\}}
{}
+{}|X_c|^L{\bf1}_{\{|X_c|>t\}}
\right]  \\
&=o(t^{-L}).
\end{aligned}
\end{equation}
Moreover \(\|\delta_t^{(d)}\|_\infty=O(t^{-2})\), so
\((\mathrm{EX}_{d,L})\) holds for all sufficiently large \(t\).  The first
term in
\eqref{eq:multidim-density-quotient-explicit-remainder} is the local Taylor
remainder on \(|X_c|\le t\); the second term is the large-translation
remainder on \(|X_c|>t\), and the estimate for that term uses precisely
\(L\ge d+1\).  In particular, finite centred \(L\)-absolute moment is a
sufficient condition for the full \((\mathrm{MPJ}_{d,L})\) package, including
the large-\(t\) entropy-existence clause \((\mathrm{EX}_{d,L})\).
\end{theorem}

\begin{proof}
Put \(h_t=P_t^{(d)}*\nu\) and \(g_t=P_t^{(d)}(\cdot-\bar{\gamma})\).
For \(x=\bar{\gamma}+ty\),
\[
\frac{P_t^{(d)}(x-\gamma)}{P_t^{(d)}(x-\bar{\gamma})}
=
\left(\frac{t^2+|x-\bar{\gamma}|^2}{t^2+|x-\gamma|^2}\right)^{(d+1)/2}
=F_y(X_c/t).
\]
Thus
\[
\delta_t^{(d)}(y)=\int_{\RR^d}\{F_y(X_c/t)-1\}\,\dd\nu(\gamma).
\]
Subtract the Taylor polynomial through order \(L\).  By
Lemma~\ref{lem:global-multidim-quotient-remainder-core}, the remainder
\[
\rho_{L,t}^{(d)}(y):=
\int_{\RR^d}\left\{
F_y(X_c/t)-\sum_{|\alpha|\le L}U_\alpha^{(d)}(y)(X_c/t)^\alpha
\right\}\dd\nu(\gamma)
\]
satisfies \eqref{eq:multidim-density-quotient-explicit-remainder}.
This is the displayed \(L^\infty\cap L^1(\Omega_d)\) tail split.  The
\(|X_c|\le t\) region is controlled by the uniform Taylor remainder
\(O(|X_c/t|^{L+1})\); the tail region \(|X_c|>t\) is controlled by the
large-translation bound \(O(|X_c/t|^L)\), where \(L\ge d+1\) enters.

The constant term equals \(1\) and is removed by the definition of
\(\delta_t^{(d)}\).  The first-order Taylor layer is
\[
\sum_{|\alpha|=1}\mu_\alpha U_\alpha^{(d)}(y)t^{-1}=0
\]
because \(\nu\) is centred.  The remaining Taylor terms give
\eqref{eq:multidim-density-quotient-expansion-main}.  Finally,
\eqref{eq:multidim-coefficient-bound} and the displayed expansion give
\(\|\delta_t^{(d)}\|_\infty=O(t^{-2})+o(t^{-L})=O(t^{-2})\).
For all sufficiently large \(t\), therefore, \(\|\delta_t^{(d)}\|_\infty\le
1/2\), so \(1+\delta_t^{(d)}\) is bounded above and below by positive
constants.  Also
\[
\int_{\RR^d}\delta_t^{(d)}(y)\,\Omega_d(\dd y)
=\int_{\RR^d}\{h_t(\bar{\gamma}+ty)-g_t(\bar{\gamma}+ty)\}t^d\,\dd y=0,
\]
because both densities have total mass one.  Since \(\Phi\) is bounded by a
constant multiple of \(s^2\) on \(|s|\le1/2\), the integral
\(\int\Phi(\delta_t^{(d)})\,\dd\Omega_d\) is finite for all large \(t\), and
the change of variables \(x=\bar{\gamma}+ty\) gives the displayed KL
representation in \((\mathrm{EX}_{d,L})\).
\end{proof}

\begin{corollary}[Finite-moment multidimensional KL jet]%
\label{thm:multidim-finite-moment-poisson-kl-asymptotic}%
\label{cor:multidim-finite-moment-entropy-jet}
Let \(d\ge1\), let \(L\in\mathbb N\) satisfy \(L\ge d+1\), and let \(\nu\) be
a probability measure on \(\RR^d\) with mean \(\bar{\gamma}\) and finite centred
\(L\)th absolute moment.  Set
\[
\begin{aligned}
h_t&=P_t^{(d)}*\nu,\\
g_t&=P_t^{(d)}(\cdot-\bar{\gamma}),
\end{aligned}
\]
and define the centred variable and moments by
\[
\begin{aligned}
X_c&=\gamma-\bar{\gamma},\\
\mu_\alpha&=\EE[X_c^\alpha]\qquad(|\alpha|\le L).
\end{aligned}
\]
Then the finite moment hypothesis first gives the quotient jet
\((\mathrm{MPJ}_{d,L})\), and this quotient jet gives the matched Poisson KL
coefficient expansion
\[
D_{\mathrm{KL}}(h_t\|g_t)
=\sum_{m=4}^{L+2}\mathcal A_m^{(d)}(\nu)t^{-m}
+o(t^{-(L+2)}),
\]
where
\[
\mathcal A_m^{(d)}(\nu)=
\sum_{r=2}^{\lfloor m/2\rfloor}
\frac{(-1)^r}{r(r-1)}
\sum_{\substack{\alpha_1,\ldots,\alpha_r\in\NN_0^d\\
|\alpha_1|+\cdots+|\alpha_r|=m\\
2\le|\alpha_j|\le L\ \mathrm{for}\ 1\le j\le r}}
\mu_{\alpha_1}\cdots\mu_{\alpha_r}
\int_{\RR^d}\prod_{j=1}^r
U_{\alpha_j}^{(d)}(y)\,\Omega_d(\dd y).
\]
All odd retained coefficients vanish:
\[
\mathcal A_m^{(d)}(\nu)=0
\qquad\text{for every odd }m,\quad 4\le m\le L+2.
\]
\end{corollary}

\begin{proof}
Theorem~\ref{thm:multidimensional-density-quotient} proves that the finite
centred \(L\)th absolute moment hypothesis implies \((\mathrm{MPJ}_{d,L})\),
with density layers
\[
D_j^{(d)}(y;\nu)=\sum_{|\alpha|=j}\mu_\alpha U_\alpha^{(d)}(y).
\]
Theorem~\ref{thm:multidimensional-poisson-entropy-jets} applies to exactly
that named quotient jet and gives the displayed entropy expansion and
coefficient formula.  The odd-coefficient cancellation is the parity statement
proved in that theorem.  No covariance contraction is used in this proof.
\end{proof}

\noindent
The next lemma and proposition form only the separate covariance-algebra
layer.  They evaluate the quadratic density mode under finite covariance and
do not by themselves prove any KL asymptotic.

\begin{lemma}[Radial--angular contractions for the Cauchy weight]%
\label{lem:radial-angular-contractions}
Let \(Y\) have law
\(\Omega_d(\dd y)=P_1^{(d)}(y)\dd y\), write the polar decomposition
\[
Y=\sqrt R\,\Theta,\qquad R=|Y|^2,\qquad \Theta\in S^{d-1}.
\]
Then \(R\) and \(\Theta\) are independent, \(\Theta\) is uniform on
\(S^{d-1}\), and \(R\) has beta-prime law with parameters \(d/2\) and
\(1/2\).  The angular fourth moments are
\begin{equation}\label{eq:spherical-fourth-contraction}
\EE_\Theta[\Theta_i\Theta_j\Theta_k\Theta_l]
=
\frac{\delta_{ij}\delta_{kl}
+\delta_{ik}\delta_{jl}
+\delta_{il}\delta_{jk}}{d(d+2)}
\qquad(1\le i,j,k,l\le d).
\end{equation}
Consequently, for real symmetric matrices \(A,B\),
\begin{equation}\label{eq:spherical-quadratic-contraction}
\EE_\Theta[
(\Theta^{\mathsf T}A\Theta)(\Theta^{\mathsf T}B\Theta)]
=
\frac{(\operatorname{tr}A)(\operatorname{tr}B)
+2\operatorname{tr}(AB)}{d(d+2)}.
\end{equation}
In particular, if \(A\) is traceless, then
\begin{equation}\label{eq:spherical-traceless-square}
\EE_\Theta[(\Theta^{\mathsf T}A\Theta)^2]
=\frac{2}{d(d+2)}\|A\|_{\rm HS}^2 .
\end{equation}
For every real \(k\) with \(-d/2<k<9/2\),
\begin{equation}\label{eq:radial-beta-prime-contraction-general}
\EE\frac{R^k}{(1+R)^4}
=
\frac{B(d/2+k,\,9/2-k)}{B(d/2,\,1/2)}
=
\frac{\Gamma(d/2+k)\Gamma(9/2-k)}
{\Gamma(d/2)\Gamma(1/2)}
\frac{\Gamma((d+1)/2)}{\Gamma((d+9)/2)} .
\end{equation}
Thus, with \(D_d=(d+1)(d+3)(d+5)(d+7)\),
\begin{equation}\label{eq:radial-beta-prime-contractions-low}
\EE\frac{1}{(1+R)^4}=\frac{105}{D_d},\qquad
\EE\frac{R}{(1+R)^4}=\frac{15d}{D_d},\qquad
\EE\frac{R^2}{(1+R)^4}=\frac{3d(d+2)}{D_d}.
\end{equation}
\end{lemma}

\begin{proof}
The density \(P_1^{(d)}(y)\) is radial and proportional to
\((1+|y|^2)^{-(d+1)/2}\).  Passing to polar coordinates shows that
\(\Theta\) is uniform, independent of \(R\), and that \(R\) has density
\[
\frac{1}{B(d/2,1/2)}\,r^{d/2-1}(1+r)^{-(d+1)/2},
\qquad r>0.
\]
This is the beta-prime distribution with parameters \(d/2\) and \(1/2\).
Multiplying this density by \(r^k(1+r)^{-4}\) and integrating gives
\eqref{eq:radial-beta-prime-contraction-general}; the three rational values
in \eqref{eq:radial-beta-prime-contractions-low} are the cases
\(k=0,1,2\) after expanding the gamma ratios.  For example the
\(d=1,\ k=0\) value is \(35/128\), the one-dimensional Cauchy value.

For the angular identity, rotational invariance implies that the tensor
\(\EE_\Theta[\Theta_i\Theta_j\Theta_k\Theta_l]\) is a multiple of
\(\delta_{ij}\delta_{kl}+\delta_{ik}\delta_{jl}+\delta_{il}\delta_{jk}\).
Contracting \(i=j\) and \(k=l\) gives
\[
1=\EE_\Theta|\Theta|^4=C(d^2+2d),
\]
so \(C=1/(d(d+2))\), proving
\eqref{eq:spherical-fourth-contraction}.  Contracting this identity with
\(A_{ij}B_{kl}\) gives \eqref{eq:spherical-quadratic-contraction}; taking
\(\operatorname{tr}A=0\) and \(B=A\) gives
\eqref{eq:spherical-traceless-square}.
\end{proof}

Finite covariance alone is not used here to prove

D_{\rm KL}\bigl(P_t^{(d)}*\nu\,\|\,P_t^{(d)}(\cdot-\bar{\gamma})\bigr)
=\mathcal Q_d(\Sigma)t^{-4}+o(t^{-4});
\]
it gives only the algebraic number \(\mathcal Q_d(\Sigma)\).  The KL

The next proposition evaluates the finite quadratic form determined by the
degree-two density mode.  It has no entropy-existence content: neither
\((\mathrm{EX}_{d,L})\) nor \((\mathrm{MPJ}_{d,L})\) follows from finite
covariance alone.  The bridge corollary after the proposition records how the
same algebra enters once \((\mathrm{MPJ}_{d,L})\) is separately present, and
D4 is the finite-moment statement that supplies that hypothesis.

\begin{proposition}[D3 covariance algebra under finite covariance]%
\label{prop:multidim-formal-covariance-layer}
Let \(d\ge1\), and let \(\nu\) be a probability measure on \(\RR^d\) with
finite centred second moment.  Put
\[
\bar{\gamma}=\int_{\RR^d}\gamma\,\dd\nu(\gamma),
\qquad
X_c=\gamma-\bar{\gamma},
\]
and
\[
\Sigma=\EE[X_cX_c^{\mathsf T}],\qquad
\Sigma_0=\Sigma-\frac{\operatorname{tr}\Sigma}{d}I_d.
\]
The degree-two centred density mode associated with \(\Sigma\) is
\begin{equation}\label{eq:multidim-leading-theorem-density-mode}
b_\Sigma(y)=
\frac{d+1}{2}\left(
-\frac{\operatorname{tr}\Sigma}{1+|y|^2}
+\frac{(d+3)y^{\mathsf T}\Sigma y}{(1+|y|^2)^2}
\right).
\end{equation}
Define its algebraic quadratic entropy-integrand layer by
\begin{equation}\label{eq:multidim-leading-positive-defect}
\mathcal Q_d(\Sigma)
:=\frac12\int_{\RR^d}b_\Sigma(y)^2\,\Omega_d(\dd y)
=c_d^{\rm iso}(\operatorname{tr}\Sigma)^2
+c_d^{\rm tr}\|\Sigma_0\|_{\rm HS}^2,
\end{equation}
with
\begin{equation}\label{eq:multidim-leading-positive-constants}
c_d^{\rm iso}=
\frac{3(d+1)(7d+9)}
{4d(d+3)(d+5)(d+7)},
\qquad
c_d^{\rm tr}=
\frac{3(d+1)(d+3)}
{4(d+5)(d+7)} .
\end{equation}
Two immediate checks are
\[
d=1:\qquad \Sigma_0=0,\qquad c_1^{\rm iso}=\frac18,\qquad
\mathcal Q_1(\Sigma)=\frac{(\operatorname{tr}\Sigma)^2}{8},
\]
and, for an isotropic covariance \(\Sigma=\tau I_d\),
\[
\Sigma_0=0,\qquad
\mathcal Q_d(\Sigma)=c_d^{\rm iso}d^2\tau^2 .
\]
The radial moments and angular contractions entering the two constants are
isolated in Lemma~\ref{lem:radial-angular-contractions}.
\end{proposition}

\begin{proof}
First compute the formal degree-two density layer.  Let
\[
  a=\frac{d+1}{2},\qquad A=1+|y|^2 .
\]
For a deterministic \(z\in\RR^d\), the radial Poisson quotient is
\[
F_y(z):=
\left(\frac{1+|y|^2}{1+|y-z|^2}\right)^a
=
\left(1+\frac{-2y\cdot z+|z|^2}{A}\right)^{-a}.
\]
Write
\[
  w=\frac{-2y\cdot z+|z|^2}{A}=w_1+w_2,\qquad
  w_1=-\frac{2y\cdot z}{A},\qquad
  w_2=\frac{|z|^2}{A}.
\]
The binomial expansion gives
\[
  (1+w)^{-a}=1-aw+\frac{a(a+1)}2w^2+O(|w|^3),
\]
and the homogeneous degree-two contribution comes only from \(-aw_2\) and
\(\frac{a(a+1)}2w_1^2\).  Hence that degree-two part is
\[
a\left(-\frac{|z|^2}{A}
+\frac{(d+3)(y\cdot z)^2}{A^2}\right).
\]
Indeed,
\[
  -aw_2+\frac{a(a+1)}2w_1^2
  =
  -a\frac{|z|^2}{A}
  +2a(a+1)\frac{(y\cdot z)^2}{A^2},
\]
and \(2(a+1)=d+3\).  The degree-zero term is the matched reference density and
the degree-one term vanishes after centering.  Averaging the displayed
degree-two term with \(z=X_c/t\) gives
\[
  D_2^{(d)}(y;\nu)=b_\Sigma(y)=
  \frac{d+1}{2}\left(
  -\frac{\operatorname{tr}\Sigma}{1+|y|^2}
  +\frac{(d+3)y^{\mathsf T}\Sigma y}{(1+|y|^2)^2}
  \right),
\]
since
\[
\EE |X_c|^2=\operatorname{tr}\Sigma,\qquad
\EE (y\cdot X_c)^2=y^{\mathsf T}\Sigma y .
\]

It remains to evaluate \(\frac12\int b_\Sigma^2\,\dd\Omega_d\).  Decompose
\(\Sigma=\tau I_d+\Sigma_0\), \(\tau=d^{-1}\operatorname{tr}\Sigma\).  Then
\[
b_\Sigma(y)=
a\tau\frac{3|y|^2-d}{(1+|y|^2)^2}
+a(d+3)\frac{y^{\mathsf T}\Sigma_0y}{(1+|y|^2)^2}.
\]
Let \(Y=\sqrt R\,\Theta\) be distributed according to \(\Omega_d\).  In these
variables,
\[
b_\Sigma(Y)=
a\tau\frac{3R-d}{(1+R)^2}
+a(d+3)\frac{R\,\Theta^{\mathsf T}\Sigma_0\Theta}{(1+R)^2}.
\]
The mixed term in \(\EE b_\Sigma(Y)^2\) is zero because \(R\) and \(\Theta\)
are independent and
\[
\EE_\Theta[\Theta^{\mathsf T}\Sigma_0\Theta]
=\frac{\operatorname{tr}\Sigma_0}{d}=0 .
\]

The isotropic part uses only the low radial moments from
Lemma~\ref{lem:radial-angular-contractions}.  With
\[
  D_d=(d+1)(d+3)(d+5)(d+7),
\]
those moments are
\[
  \EE\frac1{(1+R)^4}=\frac{105}{D_d},\qquad
  \EE\frac{R}{(1+R)^4}=\frac{15d}{D_d},\qquad
  \EE\frac{R^2}{(1+R)^4}=\frac{3d(d+2)}{D_d}.
\]
Therefore
\[
\begin{aligned}
\EE\frac{(3R-d)^2}{(1+R)^4}
&=
9\EE\frac{R^2}{(1+R)^4}
-6d\EE\frac{R}{(1+R)^4}
+d^2\EE\frac1{(1+R)^4} \\
&=
\frac{27d(d+2)-90d^2+105d^2}{D_d} \\
&=
\frac{6d(7d+9)}
{(d+1)(d+3)(d+5)(d+7)} .
\end{aligned}
\]
The isotropic contribution to \(\frac12\int b_\Sigma^2\,\dd\Omega_d\) is
\[
\begin{aligned}
\frac12 a^2\frac{(\operatorname{tr}\Sigma)^2}{d^2}
\EE\frac{(3R-d)^2}{(1+R)^4}
&=
\frac{3(d+1)(7d+9)}
{4d(d+3)(d+5)(d+7)}
(\operatorname{tr}\Sigma)^2 .
\end{aligned}
\]
For the traceless part, the radial and angular factors separate.  The
spherical fourth-moment contraction gives
\[
\EE_\Theta[(\Theta^{\mathsf T}\Sigma_0\Theta)^2]
=\frac{2}{d(d+2)}\|\Sigma_0\|_{\rm HS}^2,
\]
because \(\operatorname{tr}\Sigma_0=0\).  Hence
\[
\begin{aligned}
\int_{\RR^d}
\frac{(y^{\mathsf T}\Sigma_0y)^2}{(1+|y|^2)^4}\,\Omega_d(\dd y)
&=
\EE\frac{R^2}{(1+R)^4}\,
\EE_\Theta[(\Theta^{\mathsf T}\Sigma_0\Theta)^2] \\
&=
\frac{3d(d+2)}{D_d}\,
\frac{2}{d(d+2)}
\|\Sigma_0\|_{\rm HS}^2 \\
&=
\frac{6}{(d+1)(d+3)(d+5)(d+7)}
\|\Sigma_0\|_{\rm HS}^2 .
\end{aligned}
\]
Here the first factor is the \(k=2\) value in
\eqref{eq:radial-beta-prime-contractions-low}, and the second factor is the
traceless angular contraction
\eqref{eq:spherical-traceless-square}.
Multiplication by \(\frac12a^2(d+3)^2\) gives
\[
\frac{3(d+1)(d+3)}{4(d+5)(d+7)}
\|\Sigma_0\|_{\rm HS}^2 .
\]
This proves \eqref{eq:multidim-leading-positive-constants}.  The constants are
positive, so the layer vanishes exactly when \(\Sigma=0\).
\end{proof}

\noindent\fbox{\begin{minipage}{\dimexpr\linewidth-2\fboxsep-2\fboxrule\relax}
\textbf{Open finite-covariance quotient problem.}
Proposition~\ref{prop:multidim-formal-covariance-layer} proves the algebraic
implication
\[
\EE|X_c|^2<\infty
\quad\Longrightarrow\quad
\mathcal Q_d(\Sigma)
=\frac12\int_{\RR^d}b_\Sigma^2\,\dd\Omega_d<\infty .
\]
It does not prove the leading KL asymptotic
\[
D_{\rm KL}\bigl(P_t^{(d)}*\nu\,\|\,P_t^{(d)}(\cdot-\bar{\gamma})\bigr)
=\mathcal Q_d(\Sigma)t^{-4}+o(t^{-4}).
\]
The open quotient question is whether finite covariance alone implies
\[
  \delta_t^{(d)}(y)=t^{-2}b_\Sigma(y)+r_t(y),\qquad
  \|r_t\|_\infty+\|r_t\|_{L^1(\Omega_d)}=o(t^{-2}).
\]
Equivalently, if \(\EE|X_c|^2<\infty\), does this displayed quotient
expansion hold?
For the KL conclusion this paper requires the second-order quotient criterion
\eqref{eq:second-order-quotient-criterion}, which also includes
\(\|\delta_t^{(d)}\|_\infty=O(t^{-2})\), or a stronger hypothesis such as
\((\mathrm{MPJ}_{d,L})\).  Theorem~\ref{thm:multidimensional-density-quotient}
proves the quotient input under the sufficient condition
\(\EE|X_c|^{d+1}<\infty\), but not under finite covariance alone.
\end{minipage}}

\begin{proposition}[Second-order quotient criterion]%
\label{prop:second-order-quotient-criterion}
Let \(d\ge1\), and let \(\nu\) be a probability measure on \(\RR^d\) with
mean \(\bar{\gamma}\) and finite centred covariance \(\Sigma\).  Put
\[
  h_t=P_t^{(d)}*\nu,\qquad
  g_t=P_t^{(d)}(\cdot-\bar{\gamma}),\qquad
  \delta_t^{(d)}(y)=\frac{h_t(\bar{\gamma}+ty)}
  {g_t(\bar{\gamma}+ty)}-1 .
\]
Assume that the second-order quotient estimate
\begin{equation}\label{eq:second-order-quotient-criterion}
  \delta_t^{(d)}(y)=t^{-2}b_\Sigma(y)+r_t(y),\qquad
  \|r_t\|_\infty+\|r_t\|_{L^1(\Omega_d)}=o(t^{-2}),\qquad
  \|\delta_t^{(d)}\|_\infty=O(t^{-2})
\end{equation}
holds.  Then
\[
D_{\rm KL}(h_t\|g_t)=\mathcal Q_d(\Sigma)t^{-4}+o(t^{-4}).
\]
\end{proposition}

\begin{proof}
Since \(\|\delta_t^{(d)}\|_\infty=O(t^{-2})\), for all sufficiently large \(t\)
\[
\Phi(s)=(1+s)\log(1+s)-s=\frac{s^2}{2}+O(s^3)
\]
uniformly for \(s=\delta_t^{(d)}(y)\).  Also \(b_\Sigma\) is bounded and
\eqref{eq:second-order-quotient-criterion} gives
\(\|\delta_t^{(d)}\|_{L^1(\Omega_d)}=O(t^{-2})\).  The quadratic part obeys
\[
\left\|\frac12\{(t^{-2}b_\Sigma+r_t)^2-t^{-4}b_\Sigma^2\}\right\|_{L^1(\Omega_d)}
\le
t^{-2}\|b_\Sigma\|_\infty\|r_t\|_{L^1(\Omega_d)}
+\frac12\|r_t\|_\infty\|r_t\|_{L^1(\Omega_d)}
=o(t^{-4}).
\]
The Taylor remainder has \(L^1\)-norm at most
\[
C\|\delta_t^{(d)}\|_\infty^2\|\delta_t^{(d)}\|_{L^1(\Omega_d)}=O(t^{-6}).
\]
Thus
\[
\Phi(\delta_t^{(d)})=\frac12t^{-4}b_\Sigma^2+o(t^{-4})
\quad\text{in }L^1(\Omega_d).
\]
For all large \(t\), the quotient is positive and the displayed \(L^1\)
estimate also gives \(\Phi(\delta_t^{(d)})\in L^1(\Omega_d)\).  Changing
variables in the density quotient and integrating,
\[
\begin{aligned}
D_{\rm KL}(h_t\|g_t)
&=\int_{\RR^d}\Phi(\delta_t^{(d)}(y))\,\Omega_d(\dd y)\\
&=\frac12t^{-4}\int_{\RR^d}b_\Sigma(y)^2\,\Omega_d(\dd y)+o(t^{-4})\\
&=\mathcal Q_d(\Sigma)t^{-4}+o(t^{-4}),
\end{aligned}
\]
because Proposition~\ref{prop:multidim-formal-covariance-layer} identifies
\(\frac12\int b_\Sigma^2\,\dd\Omega_d\) with \(\mathcal Q_d(\Sigma)\).
\end{proof}

\noindent\fbox{\begin{minipage}{\dimexpr\linewidth-2\fboxsep-2\fboxrule\relax}
\textbf{Finite covariance is not enough here.}
Finite covariance alone is not proved to imply the quotient criterion
\eqref{eq:second-order-quotient-criterion}; this is open here.  Thus
Proposition~\ref{prop:multidim-formal-covariance-layer} supplies the number
\(\mathcal Q_d(\Sigma)\), while
\(\mathcal Q_d(\Sigma)t^{-4}+o(t^{-4})\) is a KL asymptotic only under an
additional quotient estimate such as
\eqref{eq:second-order-quotient-criterion}.
\end{minipage}}

\begin{corollary}[Internal bridge: leading coefficient under \((\mathrm{MPJ}_{d,L})\)]%
\label{cor:multidim-leading-kl-coefficient}
Let \(d\ge1\), let \(L\in\mathbb N\) satisfy \(L\ge2\), and assume that
\((\nu,d,L)\) satisfies the full \((\mathrm{MPJ}_{d,L})\), including
\((\mathrm{EX}_{d,L})\).  Let
\[
h_t=P_t^{(d)}*\nu,\qquad
g_t=P_t^{(d)}(\cdot-\bar{\gamma}).
\]
Since \(L\ge2\), the covariance
\[
\Sigma=\EE[X_cX_c^{\mathsf T}],\qquad X_c=\gamma-\bar{\gamma},
\]
exists.  Then
\[
D_{\rm KL}(h_t\|g_t)
=\mathcal Q_d(\Sigma)t^{-4}+o(t^{-4}).
\]
\end{corollary}

\begin{proof}
The hypothesis \((\mathrm{MPJ}_{d,L})\) is the analytic input of
Theorem~\ref{thm:multidimensional-poisson-entropy-jets}.  Applying that
theorem gives the KL jet, whose \(m=4\) coefficient has only the quadratic
term
\[
\mathcal A_4^{(d)}(\nu)
=\frac12\int_{\RR^d}D_2^{(d)}(y;\nu)^2\,\Omega_d(\dd y).
\]
For centred data the degree-two density layer is
\[
D_2^{(d)}(y;\nu)=b_\Sigma(y),
\]
where \(b_\Sigma\) is the bounded covariance mode in
\eqref{eq:multidim-leading-theorem-density-mode}.  Proposition~\ref{prop:multidim-formal-covariance-layer}
computes
\[
\frac12\int_{\RR^d}b_\Sigma(y)^2\,\Omega_d(\dd y)=\mathcal Q_d(\Sigma).
\]
Thus \(D_{\rm KL}(h_t\|g_t)=\mathcal Q_d(\Sigma)t^{-4}+o(t^{-4})\).
\end{proof}

\begin{corollary}[D4 actual leading KL coefficient]%
\label{cor:finite-dplusone-moment-leading-multidim-coefficient}
Let \(d\ge1\), let \(\nu\) be a probability measure on \(\RR^d\) with mean
\(\bar{\gamma}\), and assume
\[
  \EE |X_c|^{d+1}<\infty,\qquad X_c=\gamma-\bar{\gamma} .
\]
Put
\[
  h_t=P_t^{(d)}*\nu,\qquad
  g_t=P_t^{(d)}(\cdot-\bar{\gamma}),\qquad
  \Sigma=\EE[X_cX_c^{\mathsf T}],\qquad
  \Sigma_0=\Sigma-\frac{\operatorname{tr}\Sigma}{d}I_d .
\]
Then Theorem~\ref{thm:multidimensional-density-quotient} supplies
\((\mathrm{MPJ}_{d,d+1})\), including the large-\(t\) entropy-existence
clause \((\mathrm{EX}_{d,d+1})\).  Consequently
\[
D_{\rm KL}\!\left(P_t^{(d)}*\nu\,\middle\|\,
P_t^{(d)}(\cdot-\bar{\gamma})\right)
=\mathcal Q_d(\Sigma)t^{-4}+o(t^{-4}),
\]
where
\[
\mathcal Q_d(\Sigma)
=c_d^{\rm iso}(\operatorname{tr}\Sigma)^2
+c_d^{\rm tr}\|\Sigma_0\|_{\rm HS}^2
\]
and
\[
c_d^{\rm iso}=
\frac{3(d+1)(7d+9)}
{4d(d+3)(d+5)(d+7)},
\qquad
c_d^{\rm tr}=
\frac{3(d+1)(d+3)}
{4(d+5)(d+7)} .
\]
\end{corollary}
The hypothesis \(\EE |X_c|^{d+1}<\infty\) in D4 is a sufficient
finite-moment route to \((\mathrm{MPJ}_{d,d+1})\) and
\((\mathrm{EX}_{d,d+1})\), not a sharpness assertion.  In particular, this
corollary does not claim a converse or an optimal moment threshold in
dimension \(d\).

\begin{proof}
Apply Theorem~\ref{thm:multidimensional-density-quotient} with
\(L=d+1\).  The finite \((d+1)\)-absolute moment supplies
\((\mathrm{MPJ}_{d,d+1})\).  In the introduction-level notation, this is the
D2 input: by D2, \(\EE|X_c|^{d+1}<\infty\) implies the displayed quotient
criterion, and by D3 the coefficient is \(\mathcal Q_d(\Sigma)\).
Corollary~\ref{cor:multidim-leading-kl-coefficient} then applies to that
quotient jet and identifies the \(t^{-4}\) coefficient with the covariance
contraction computed in
Proposition~\ref{prop:multidim-formal-covariance-layer}.
\end{proof}

\begin{corollary}[Finite-moment covariance coefficient for larger \(L\)]%
\label{cor:multidim-finite-moment-covariance-coefficient}
Let \(d\ge1\), let \(L\in\mathbb N\) satisfy \(L\ge d+1\), and let \(\nu\) be
a probability measure on \(\RR^d\) with mean \(\bar{\gamma}\) and finite
centred \(L\)th absolute moment.  Put
\[
h_t=P_t^{(d)}*\nu,\qquad
g_t=P_t^{(d)}(\cdot-\bar{\gamma}),
\]
and let
\[
X_c=\gamma-\bar{\gamma},\qquad
\Sigma=\EE[X_cX_c^{\mathsf T}],\qquad
\Sigma_0=\Sigma-\frac{\operatorname{tr}\Sigma}{d}I_d .
\]
Then the finite \(L\)-moment hypothesis first supplies
\((\mathrm{MPJ}_{d,L})\), including the entropy-existence clause
\((\mathrm{EX}_{d,L})\), and only through that full quotient package one has
\begin{equation}\label{eq:finite-moment-covariance-coefficient-through-mpj}
D_{\rm KL}(h_t\|g_t)=\mathcal Q_d(\Sigma)t^{-4}+o(t^{-4}),
\end{equation}
where \(\mathcal Q_d(\Sigma)\) is the quadratic covariance contraction computed
in Proposition~\ref{prop:multidim-formal-covariance-layer}:
\[
\mathcal Q_d(\Sigma)
=c_d^{\rm iso}(\operatorname{tr}\Sigma)^2
+c_d^{\rm tr}\|\Sigma_0\|_{\rm HS}^2 .
\]
Equivalently, finite covariance gives the displayed algebraic quantity
\(\mathcal Q_d(\Sigma)\), but it becomes the leading KL coefficient in
\eqref{eq:finite-moment-covariance-coefficient-through-mpj} only because the
stronger finite \(L\)-moment assumption has supplied
\((\mathrm{MPJ}_{d,L})\).
\end{corollary}

\begin{proof}
Theorem~\ref{thm:multidimensional-density-quotient} turns the finite centred
\(L\)-absolute moment assumption into \((\mathrm{MPJ}_{d,L})\), and
Theorem~\ref{thm:multidimensional-poisson-entropy-jets} then gives the KL
coefficient jet.  The \(m=4\) coefficient in that jet is
\[
\mathcal A_4^{(d)}(\nu)
=\frac12\int_{\RR^d}D_2^{(d)}(y;\nu)^2\,\Omega_d(\dd y).
\]
The degree-two density layer depends only on the covariance:
\[
D_2^{(d)}(y;\nu)=
\frac{d+1}{2}\left(
-\frac{\operatorname{tr}\Sigma}{1+|y|^2}
+\frac{(d+3)y^{\mathsf T}\Sigma y}{(1+|y|^2)^2}
\right).
\]
Proposition~\ref{prop:multidim-formal-covariance-layer} evaluates the last
quadratic integral as \(\mathcal Q_d(\Sigma)\), proving the asserted leading
term.
\end{proof}
