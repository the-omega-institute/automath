\subsection{Coefficient-level moment-defect corollaries}
\label{subsec:fourth-moment-stability}

This subsection records the sixth-order coefficient defect.  Its fourth-moment
algebra begins with the evaluated polynomial \(A_6\) from
\eqref{eq:n4-notation-a6}; after the sixth-coefficient identification in
Theorem~\ref{thm:poisson-kl-eighth}, this same polynomial is the genuine
\(t^{-6}\) entropy coefficient, and \(\Delta_6(\nu)\) is its two-point defect.
The finite-time certificate at the end of the subsection is only conditional
bookkeeping: after \(A_8\) and the \(O(t^{-9})\) remainder have been bounded
independently of the observed KL value, the displayed sixth-defect identity
converts that observation, together with its certified error radius, into a
conditional bound for \(\Delta_6\), and then the coefficient-defect coupling
below gives the \(W_2\) estimate.

\paragraph{Fourth-moment algebra.}

\begin{lemma}[Fourth-moment algebraic defect]%
\label{lem:fourth-moment-algebraic-defect}
Let $\nu$ have mean~$\bar\gamma$, variance $\sigma^2>0$, and finite
centred fourth moment.  Define
\[
A_6(\nu):=
\frac{\sigma^6+6\mu_3^2-8\sigma^2\mu_4}{64},
\qquad
\Delta_6(\nu):=
-\frac{7}{64}\sigma^6-A_6(\nu).
\]
These are the quantities from
\eqref{eq:n4-notation-a6}--\eqref{eq:n4-notation-delta6}, used here only as
fourth-moment polynomials until the coefficient identification is invoked.
On $(\RR,\nu)$ set
\begin{align*}
X&:=\frac{\gamma-\bar\gamma}{\sigma},
&
\beta_3&:=\EE[X^3],
&
\beta_4&:=\EE[X^4],
\\
Q_2&:=X^2-1-\beta_3X,
&
\kappa&:=\|Q_2\|_{L^2(\nu)}^2.
\end{align*}
Then the pure fourth-moment identity is
\begin{equation}\label{eq:defect-sixth-fourth-moment}
\Delta_6(\nu)
=\frac{\sigma^6}{32}\beta_3^2+\frac{\sigma^6}{8}\kappa,
\qquad
\kappa=\beta_4-1-\beta_3^2\ge0.
\end{equation}
Moreover, \(\Delta_6(\nu)=0\) if and only if the centred law of
$\gamma-\bar\gamma$ is $\rho_\sigma=\frac12(\delta_{-\sigma}
+\delta_\sigma)$.
\end{lemma}

\begin{proof}
The identity for $\kappa$ follows by expanding $Q_2^2$ and using
$\EE[X]=0$, $\EE[X^2]=1$, and $\EE[X^3]=\beta_3$:
\[
\kappa
=\EE[(X^2-1-\beta_3X)^2]
=\beta_4-1-\beta_3^2.
\]
Since $\mu_3=\sigma^3\beta_3$ and $\mu_4=\sigma^4\beta_4$,
\[
-\frac{7}{64}\sigma^6-A_6(\nu)
=\frac{\sigma^6}{64}\bigl(-7-1-6\beta_3^2+8\beta_4\bigr)
=\frac{\sigma^6}{32}\beta_3^2+\frac{\sigma^6}{8}\kappa.
\]
This proves \eqref{eq:defect-sixth-fourth-moment} and non-negativity.  If
\(\Delta_6(\nu)=0\), then $\beta_3=0$ and $\kappa=0$, so
$X^2=1$ almost surely.  The conditions $\EE[X]=0$ and $\EE[X^2]=1$ then
force $X=\pm1$ with equal probabilities, i.e. the centred law is
$\rho_\sigma$.  The converse is immediate by substitution.
\end{proof}

\begin{lemma}[Fourth-moment algebraic coupling]%
\label{lem:fourth-moment-algebraic-coupling}
Let $\nu$ have mean~$\bar\gamma$, variance $\sigma^2>0$, and finite
centred fourth moment.  Let $\nu_c$ be the centred law of
$X_c=\gamma-\bar\gamma$ and set
\[
\rho_\sigma:=\frac12(\delta_{-\sigma}+\delta_\sigma).
\]
For the fourth-moment polynomial $\Delta_6(\nu)$ defined above,
\[
\Delta_6(\nu)\ge0
\]
and
\begin{equation}\label{eq:fourth-moment-algebraic-coupling}
W_2(\nu_c,\rho_\sigma)
\le
4\,\sigma^{-1/2}\Delta_6(\nu)^{1/4}
+8\,\sigma^{-2}\Delta_6(\nu)^{1/2}.
\end{equation}
Moreover, $\Delta_6(\nu)=0$ if and only if
$\nu_c=\rho_\sigma$.
\end{lemma}

\begin{proof}
All random variables below live on $(\RR,\nu)$.  Set
\[
X:=\frac{\gamma-\bar\gamma}{\sigma},
\qquad
\beta_3:=\EE[X^3],
\qquad
Q_2:=X^2-1-\beta_3X,
\qquad
\kappa:=\|Q_2\|_{L^2(\nu)}^2 .
\]
By \eqref{eq:defect-sixth-fourth-moment},
\begin{equation}\label{eq:algebraic-coupling-delta-six}
\Delta_6(\nu)
=\frac{\sigma^6}{32}\beta_3^2+\frac{\sigma^6}{8}\kappa .
\end{equation}
Define
\[
Y:=\mathbf{1}_{\{X\ge 0\}}-\mathbf{1}_{\{X<0\}}\in\{-1,1\}.
\]
For every real $x$, with $\operatorname{sgn}(0)=1$,
\[
|x-\operatorname{sgn}(x)|\le |x^2-1|.
\]
Therefore
\[
\EE[(X-Y)^2]\le \EE[(X^2-1)^2].
\]
The decomposition $X^2-1=\beta_3X+Q_2$ is orthogonal in $L^2(\nu)$ because
\[
\EE[XQ_2]=\EE[X^3]-\EE[X]-\beta_3\EE[X^2]=0.
\]
Thus
\[
\EE[(X^2-1)^2]
=\beta_3^2+\kappa
\le 2\beta_3^2+8\kappa
=\frac{64}{\sigma^6}\Delta_6(\nu),
\]
where the last equality is \eqref{eq:algebraic-coupling-delta-six}.  Since
$\EE[X]=0$,
\[
|\EE[Y]|=|\EE[Y-X]|
\le \EE[|Y-X|]
\le \EE[(Y-X)^2]^{1/2}
\le \frac{8}{\sigma^3}\Delta_6(\nu)^{1/2}.
\]
Let $Z$ be a symmetric Rademacher random variable.  If
$p=\mathbb P(Y=1)$, then $\EE[Y]=2p-1$ and the optimal transport between
$\mathcal L(Y)$ and the symmetric law on $\{-1,1\}$ moves
$|p-1/2|$ mass across distance~$2$.  Hence
\[
W_2(\mathcal L(Y),\mathcal L(Z))^2
=4|p-1/2|=2|\EE[Y]|
\le \frac{16}{\sigma^3}\Delta_6(\nu)^{1/2}.
\]
Also
\[
W_2(\mathcal L(X),\mathcal L(Y))
\le \EE[(X-Y)^2]^{1/2}
\le \frac{8}{\sigma^3}\Delta_6(\nu)^{1/2}.
\]
The laws satisfy $\nu_c=\mathcal L(\sigma X)$ and
$\rho_\sigma=\mathcal L(\sigma Z)$.  Taking square roots in the preceding
$W_2$ bound, then using scaling and the triangle inequality, gives
\eqref{eq:fourth-moment-algebraic-coupling}.
The zero statement follows from
Lemma~\ref{lem:fourth-moment-algebraic-defect}.
\end{proof}

\begin{remark}[Algebraic layer]
Lemma~\ref{lem:fourth-moment-algebraic-coupling} belongs to the
fourth-moment algebra layer.  The entropy coefficient interpretation enters
only through Lemma~\ref{lem:seventh-moment-bridge}, and the finite-time
estimates enter later through certified eighth-order data.
\end{remark}

\paragraph{Fourth-moment coefficient bridge.}

\begin{lemma}[Fourth-moment coefficient identification bridge]%
\label{lem:seventh-moment-bridge}
Assume that $\nu$ has mean~$\bar\gamma$, variance $\sigma^2>0$, and finite
centred fourth absolute moment.  Theorem~\ref{thm:poisson-kl-even-orders},
applied with \(N=3\), identifies the notation-block polynomial \(A_6(\nu)\)
as the universal \(t^{-6}\) entropy coefficient and gives
\[
D_{\mathrm{KL}}(P_t*\nu\|P_t(\cdot-\bar\gamma))
=\frac{\sigma^4}{8t^4}+\frac{A_6(\nu)}{t^6}
+o(t^{-6}).
\]
Equivalently,
\begin{equation}\label{eq:c6-entropy-bridge-explicit}
A_6(\nu)
=\frac{\sigma^6+6\mu_3^2-8\sigma^2\mu_4}{64}.
\end{equation}
Consequently \(\Delta_6(\nu)\) is the corresponding sixth-order
coefficient defect.  The seventh moment enters later through the
\(N=4,\eta=1\) quantitative \(O(t^{-9})\) remainder used in the finite-time
estimate.
\end{lemma}

\begin{proof}
The finite fourth absolute moment is precisely the \(N=3\) hypothesis in
Theorem~\ref{thm:poisson-kl-even-orders}; it is the coefficient-evaluation threshold for
evaluating the displayed universal moment-polynomial coefficient \(A_6\).
That theorem gives
\[
D_{\mathrm{KL}}(P_t*\nu\|P_t(\cdot-\bar\gamma))
=\frac{\sigma^4}{8t^4}
+\frac{\sigma^6+6\mu_3^2-8\sigma^2\mu_4}{64\,t^6}
+o(t^{-6}),
\]
using the evaluated universal coefficients \(A_4(\nu)\) and
\(A_6(\nu)\).  The explicit value of \(A_6(\nu)\) is the sixth-order arithmetic already recorded
in Lemma~\ref{lem:n4-entropy-coefficient-substitution}; it involves only
\(\mu_2,\mu_3,\mu_4\) and hence is available under the fourth-moment
hypothesis.
The coefficient in an expansion of the form
\[
F(t)=a_4t^{-4}+a_6t^{-6}+o(t^{-6})
\]
is unique: if the same \(F\) is also
\(\tilde a_4t^{-4}+\tilde a_6t^{-6}+o(t^{-6})\), then multiplying the
difference first by \(t^4\) and then, after \(a_4=\tilde a_4\), by \(t^6\)
gives \(a_6=\tilde a_6\).  Hence the genuine entropy coefficient is exactly
the displayed polynomial \(A_6(\nu)\), proving
\eqref{eq:c6-entropy-bridge-explicit}.  Substituting this identity into
\(-7\sigma^6/64-A_6(\nu)\) gives the displayed polynomial
\(\Delta_6(\nu)\), now read as the coefficient defect.
\end{proof}

\begin{corollary}[Coefficient-defect coupling corollary]%
\label{cor:coefficient-defect-coupling}
Let $\nu$ be a probability measure on~$\RR$ with mean~$\bar\gamma$,
variance $\sigma^2>0$, and finite centred fourth absolute moment.
By Lemma~\ref{lem:seventh-moment-bridge}, \(A_6(\nu)\) is the genuine
\(t^{-6}\) entropy coefficient under this fourth-moment hypothesis, and
\(\Delta_6(\nu)\) is the corresponding entropy-coefficient defect.
On $(\RR,\nu)$ set
\[
X=\frac{\gamma-\bar\gamma}{\sigma},
\qquad
\beta_3=\EE X^3,
\qquad
Q_2=X^2-1-\beta_3X,
\qquad
\kappa=\|Q_2\|_{L^2(\nu)}^2 .
\]
Under this fourth-moment hypothesis,
\begin{equation}\label{eq:w2-theorem-moment-defect-form}
\Delta_6(\nu)
=\frac{\sigma^6}{32}\beta_3^2+\frac{\sigma^6}{8}\kappa .
\end{equation}
Let $\nu_c$ be the centred law of $X_c=\gamma-\bar\gamma$ and set
\[
\rho_\sigma:=\frac12(\delta_{-\sigma}+\delta_\sigma).
\]
Then the coupling argument gives
\[
W_2(\nu_c,\rho_\sigma)
\le
4\,\sigma^{-1/2}\Delta_6(\nu)^{1/4}
+8\,\sigma^{-2}\Delta_6(\nu)^{1/2}.
\]
In particular \(\Delta_6(\nu)\ge0\).  This is a coefficient-defect estimate for
closeness to the symmetric two-point law.  The numerical constants $4$ and
$8$ are universal.  At this coefficient level the only non-universal
parameters displayed in the \(W_2\) modulus are the variance scale $\sigma$
and the coefficient defect \(\Delta_6(\nu)\) itself; no finite-time remainder
constant, observed KL value, or eighth-order input is involved.
In particular, for fixed $\sigma$,
\[
W_2(\nu_c,\rho_\sigma)=O\!\bigl(\Delta_6(\nu)^{1/4}\bigr)
\qquad\text{as }\Delta_6(\nu)\downarrow 0.
\]
Moreover \(\Delta_6(\nu)=0\) if and only if
\(\nu_c=\rho_\sigma\).
\end{corollary}

\begin{proof}
The displayed fourth absolute moment condition is exactly the hypothesis
of Lemma~\ref{lem:seventh-moment-bridge}.  Hence \(A_6(\nu)\) is the
sixth-order entropy coefficient and \(\Delta_6(\nu)\) is its
coefficient defect.
The identity \eqref{eq:w2-theorem-moment-defect-form} is
\eqref{eq:defect-sixth-fourth-moment} after applying this bridge.
Substituting
\eqref{eq:fourth-moment-algebraic-coupling} gives the stated $W_2$ estimate,
and the non-negativity and zero characterization are those of
Lemma~\ref{lem:fourth-moment-algebraic-defect}.
The \(O((\Delta_6)^{1/4})\) statement for fixed \(\sigma\)
follows immediately from the displayed estimate.
\end{proof}

\subsection{Appendix-level finite-time certificate bookkeeping}
\label{subsec:finite-time-bookkeeping}

The preceding coefficient result is the stability theorem: it converts the
sixth-order coefficient defect \(\Delta_6(\nu)\) into a \(W_2\) estimate
against the symmetric two-point law.  A finite-time KL number can be used only
after separate coefficient and remainder certificates have already been fixed.
Appendix~\ref{app:finite-time-defect-estimates} records this deterministic
error propagation as Lemma~\ref{lem:controlled-shell-conditional-certificate}.
The lemma is not an additional entropy asymptotic; it is bookkeeping for the
identity
\[
\mathfrak D_t^{(6)}(\nu)=\Delta_6(\nu)-A_8(\nu)t^{-2}-t^6E(t)
\]
with a separately certified observation error.
