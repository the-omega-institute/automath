\noindent\textbf{Notation bridge.}
We keep the notation convention from the table preceding
Subsection~\ref{subsec:eighth-order-defects}: \(C_8^{(7)}(\nu)\) denotes the
seventh-moment realization of \(A^{\mathrm{form}}_8\) and is the same number
as \(A_8(\nu)\) in the \(N=4,\eta=1\) expansion.  The symbol \(C_8\) is not
used as a standing notation.  The fourth-moment polynomial
\(\Delta_6^{(4)}(\nu)\) remains superscripted until
Lemma~\ref{lem:seventh-moment-bridge} identifies it with the genuine
sixth-order entropy coefficient defect \(\Delta_6(\nu)\).

\subsection{Coefficient-level moment-defect corollaries}
\label{subsec:fourth-moment-stability}

This subsection records only coefficient-level consequences of the
large-time entropy expansion.  The first two lemmas are purely algebraic:
with a finite centred fourth moment, $C_6^{(4)}(\nu)$ and
$\Delta_6^{(4)}(\nu)$ are fourth-moment polynomials and make no claim about
an entropy asymptotic.  The bridge paragraph then uses the finite centred
seventh absolute moment hypothesis of Theorem~\ref{thm:poisson-kl-eighth} to
identify the genuine entropy coefficient $C_6=A_6(\nu)$ with that
fourth-moment polynomial.  After this identification, and only after it, the
same fourth-moment algebra gives the \(W_2\) estimate below.  No global
stability of an entropy functional, no finite-time entropy-functional
stability statement, and no logarithmic-HLS or transportation-inequality
stability theorem is proved here.

\paragraph{Fourth-moment algebra.}

\begin{lemma}[Fourth-moment algebraic defect]%
\label{lem:fourth-moment-algebraic-defect}
Let $\nu$ have mean~$\bar\gamma$, variance $\sigma^2>0$, and finite
centred fourth moment.  Define
\[
C_6^{(4)}(\nu):=
\frac{\sigma^6+6\mu_3^2-8\sigma^2\mu_4}{64},
\qquad
\Delta_6^{(4)}(\nu):=
-\frac{7}{64}\sigma^6-C_6^{(4)}(\nu).
\]
The superscript $(4)$ is part of the notation: it records that the
quantity is a fourth-moment polynomial and carries no asymptotic assertion
by itself.  On $(\RR,\nu)$ set
\begin{align*}
X&:=\frac{\gamma-\bar\gamma}{\sigma},
&
\beta_3&:=\EE[X^3],
&
\beta_4&:=\EE[X^4],
\\
Q_2&:=X^2-1-\beta_3X,
&
\kappa&:=\|Q_2\|_{L^2(\nu)}^2.
\end{align*}
Then the pure fourth-moment identity is
\begin{equation}\label{eq:defect-sixth-fourth-moment}
\Delta_6^{(4)}(\nu)
=\frac{\sigma^6}{32}\beta_3^2+\frac{\sigma^6}{8}\kappa,
\qquad
\kappa=\beta_4-1-\beta_3^2\ge0.
\end{equation}
Moreover, $\Delta_6^{(4)}(\nu)=0$ if and only if the centred law of
$\gamma-\bar\gamma$ is $\rho_\sigma=\frac12(\delta_{-\sigma}
+\delta_\sigma)$.
\end{lemma}

\begin{proof}
The identity for $\kappa$ follows by expanding $Q_2^2$ and using
$\EE[X]=0$, $\EE[X^2]=1$, and $\EE[X^3]=\beta_3$:
\[
\kappa
=\EE[(X^2-1-\beta_3X)^2]
=\beta_4-1-\beta_3^2.
\]
Since $\mu_3=\sigma^3\beta_3$ and $\mu_4=\sigma^4\beta_4$,
\[
-\frac{7}{64}\sigma^6-C_6^{(4)}(\nu)
=\frac{\sigma^6}{64}\bigl(-7-1-6\beta_3^2+8\beta_4\bigr)
=\frac{\sigma^6}{32}\beta_3^2+\frac{\sigma^6}{8}\kappa.
\]
This proves \eqref{eq:defect-sixth-fourth-moment} and non-negativity.  If
$\Delta_6^{(4)}(\nu)=0$, then $\beta_3=0$ and $\kappa=0$, so
$X^2=1$ almost surely.  The conditions $\EE[X]=0$ and $\EE[X^2]=1$ then
force $X=\pm1$ with equal probabilities, i.e. the centred law is
$\rho_\sigma$.  The converse is immediate by substitution.
\end{proof}

\begin{lemma}[Fourth-moment algebraic coupling only]%
\label{lem:fourth-moment-algebraic-coupling}
Let $\nu$ have mean~$\bar\gamma$, variance $\sigma^2>0$, and finite
centred fourth moment.  Let $\nu_c$ be the centred law of
$\Delta=\gamma-\bar\gamma$ and set
\[
\rho_\sigma:=\frac12(\delta_{-\sigma}+\delta_\sigma).
\]
For the fourth-moment polynomial $\Delta_6^{(4)}(\nu)$ defined above,
\[
\Delta_6^{(4)}(\nu)\ge0
\]
and
\begin{equation}\label{eq:fourth-moment-algebraic-coupling}
W_2(\nu_c,\rho_\sigma)
\le
4\,\sigma^{-1/2}\Delta_6^{(4)}(\nu)^{1/4}
+8\,\sigma^{-2}\Delta_6^{(4)}(\nu)^{1/2}.
\end{equation}
Moreover, $\Delta_6^{(4)}(\nu)=0$ if and only if
$\nu_c=\rho_\sigma$.  This is only an algebraic fourth-moment statement; it
does not introduce an entropy coefficient and is not a finite-time stability
or transport theorem.
\end{lemma}

\begin{proof}
All random variables below live on $(\RR,\nu)$.  Set
\[
X:=\frac{\gamma-\bar\gamma}{\sigma},
\qquad
\beta_3:=\EE[X^3],
\qquad
Q_2:=X^2-1-\beta_3X,
\qquad
\kappa:=\|Q_2\|_{L^2(\nu)}^2 .
\]
By \eqref{eq:defect-sixth-fourth-moment},
\begin{equation}\label{eq:algebraic-coupling-delta-six}
\Delta_6^{(4)}(\nu)
=\frac{\sigma^6}{32}\beta_3^2+\frac{\sigma^6}{8}\kappa .
\end{equation}
Define
\[
Y:=\mathbf{1}_{\{X\ge 0\}}-\mathbf{1}_{\{X<0\}}\in\{-1,1\}.
\]
For every real $x$, with $\operatorname{sgn}(0)=1$,
\[
|x-\operatorname{sgn}(x)|\le |x^2-1|.
\]
Therefore
\[
\EE[(X-Y)^2]\le \EE[(X^2-1)^2].
\]
The decomposition $X^2-1=\beta_3X+Q_2$ is orthogonal in $L^2(\nu)$ because
\[
\EE[XQ_2]=\EE[X^3]-\EE[X]-\beta_3\EE[X^2]=0.
\]
Thus
\[
\EE[(X^2-1)^2]
=\beta_3^2+\kappa
\le 2\beta_3^2+8\kappa
=\frac{64}{\sigma^6}\Delta_6^{(4)}(\nu),
\]
where the last equality is \eqref{eq:algebraic-coupling-delta-six}.  Since
$\EE[X]=0$,
\[
|\EE[Y]|=|\EE[Y-X]|
\le \EE[|Y-X|]
\le \EE[(Y-X)^2]^{1/2}
\le \frac{8}{\sigma^3}\Delta_6^{(4)}(\nu)^{1/2}.
\]
Let $Z$ be a symmetric Rademacher random variable.  If
$p=\mathbb P(Y=1)$, then $\EE[Y]=2p-1$ and the optimal transport between
$\mathcal L(Y)$ and the symmetric law on $\{-1,1\}$ moves
$|p-1/2|$ mass across distance~$2$.  Hence
\[
W_2(\mathcal L(Y),\mathcal L(Z))^2
=4|p-1/2|=2|\EE[Y]|
\le \frac{16}{\sigma^3}\Delta_6^{(4)}(\nu)^{1/2}.
\]
Also
\[
W_2(\mathcal L(X),\mathcal L(Y))
\le \EE[(X-Y)^2]^{1/2}
\le \frac{8}{\sigma^3}\Delta_6^{(4)}(\nu)^{1/2}.
\]
The laws satisfy $\nu_c=\mathcal L(\sigma X)$ and
$\rho_\sigma=\mathcal L(\sigma Z)$.  Taking square roots in the preceding
$W_2$ bound, then using scaling and the triangle inequality, gives
\eqref{eq:fourth-moment-algebraic-coupling}.
The zero statement follows from
Lemma~\ref{lem:fourth-moment-algebraic-defect}.
\end{proof}

\paragraph{Seventh-moment bridge.}

\begin{lemma}[Seventh-moment coefficient identification bridge]%
\label{lem:seventh-moment-bridge}
Assume that $\nu$ has mean~$\bar\gamma$, variance $\sigma^2>0$, and
\[
\int_{\RR}\abs{\gamma-\bar\gamma}^{7}\,\dd\nu(\gamma)<\infty.
\]
This lemma is only the coefficient-identification step needed before the
algebraic fourth-moment estimate can be read as an entropy-coefficient
statement.  Let $C_6$ be the genuine $t^{-6}$ coefficient in the asymptotic expansion
of \(D_{\mathrm{KL}}(P_t*\nu\|P_t(\cdot-\bar\gamma))\) supplied by
Theorem~\ref{thm:poisson-kl-eighth}.  Equivalently,
\[
D_{\mathrm{KL}}(P_t*\nu\|P_t(\cdot-\bar\gamma))
=\frac{\sigma^4}{8t^4}+\frac{C_6}{t^6}+O_\nu(t^{-8}).
\]
Let $C_6^{(4)}(\nu)$ and $\Delta_6^{(4)}(\nu)$ be the fourth-moment
polynomials defined in Lemma~\ref{lem:fourth-moment-algebraic-defect}.
Then
\begin{equation}\label{eq:delta-six-bridge}
C_6=C_6^{(4)}(\nu),
\qquad
\Delta_6(\nu)
:=-\frac{7}{64}\sigma^6-C_6
=\Delta_6^{(4)}(\nu).
\end{equation}
More explicitly, the coefficient identified by the entropy asymptotic is
\begin{equation}\label{eq:c6-entropy-bridge-explicit}
C_6
=A_6(\nu)
=\frac{\sigma^6+6\mu_3^2-8\sigma^2\mu_4}{64}.
\end{equation}
Thus the unsuperscripted entropy defect $\Delta_6$ is introduced only under
the finite seventh absolute moment hypothesis used to make the coefficient
identification a statement about the entropy expansion.
\end{lemma}

\begin{proof}
The finite seventh absolute moment implies the fourth moment needed to
define $C_6^{(4)}(\nu)$ and $\Delta_6^{(4)}(\nu)$.  It also implies the
hypothesis of Theorem~\ref{thm:poisson-kl-eighth}, which is
Corollary~\ref{cor:optimized-retained-order-entropy-bridge} with
\(N=4\) and \(\eta=1\) followed by the explicit eighth-order coefficient
audit.  That theorem gives
\[
D_{\mathrm{KL}}(P_t*\nu\|P_t(\cdot-\bar\gamma))
=\frac{\sigma^4}{8t^4}
+\frac{\sigma^6+6\mu_3^2-8\sigma^2\mu_4}{64\,t^6}
+O_\nu(t^{-8}).
\]
The coefficient in an expansion of the form
\[
F(t)=a_4t^{-4}+a_6t^{-6}+O(t^{-8})
\]
is unique: if the same \(F\) is also
\(\tilde a_4t^{-4}+\tilde a_6t^{-6}+O(t^{-8})\), then multiplying the
difference first by \(t^4\) and then, after \(a_4=\tilde a_4\), by \(t^6\)
gives \(a_6=\tilde a_6\).  Hence the genuine entropy coefficient is exactly
\[
C_6
=\frac{\sigma^6+6\mu_3^2-8\sigma^2\mu_4}{64}
=C_6^{(4)}(\nu),
\]
which proves \eqref{eq:c6-entropy-bridge-explicit}.
Substituting this identity into
\(\Delta_6(\nu):=-7\sigma^6/64-C_6\) and using
\eqref{eq:defect-sixth-fourth-moment} gives
\(\Delta_6(\nu)=\Delta_6^{(4)}(\nu)\), completing
\eqref{eq:delta-six-bridge}.
\end{proof}

\qquad

Thus \(C_6\), \(A_6(\nu)\), and \(C_6^{(4)}(\nu)\) name the same

\begin{corollary}[Coefficient-defect coupling corollary]%
\label{cor:coefficient-defect-coupling}
Let $\nu$ be a probability measure on~$\RR$ with mean~$\bar\gamma$,
variance $\sigma^2>0$, and finite centred seventh absolute moment, i.e.
\[
\int_{\RR}\abs{\gamma-\bar\gamma}^{7}\,\dd\nu(\gamma)<\infty.
\]
Let \(C_6=A_6(\nu)\) be the genuine \(t^{-6}\) coefficient in the
large-time expansion of
\(D_{\mathrm{KL}}(P_t*\nu\|P_t(\cdot-\bar\gamma))\), as identified in
Theorem~\ref{thm:poisson-kl-eighth} and
Lemma~\ref{lem:seventh-moment-bridge}.  Define the sixth-order entropy
defect by
\[
\Delta_6(\nu):=-\frac{7}{64}\sigma^6-C_6.
\]
This definition is not the fourth-moment algebraic polynomial by fiat; under
the displayed seventh-moment hypothesis, the bridge lemma proves that the two
quantities coincide.
On $(\RR,\nu)$ set
\[
X=\frac{\gamma-\bar\gamma}{\sigma},
\qquad
\beta_3=\EE X^3,
\qquad
Q_2=X^2-1-\beta_3X,
\qquad
\kappa=\|Q_2\|_{L^2(\nu)}^2 .
\]
Under the same seventh-moment hypothesis needed to identify the entropy
coefficient,
\begin{equation}\label{eq:w2-theorem-moment-defect-form}
\Delta_6(\nu)
=\frac{\sigma^6}{32}\beta_3^2+\frac{\sigma^6}{8}\kappa .
\end{equation}
Let $\nu_c$ be the centred law of $\Delta=\gamma-\bar\gamma$ and set
\[
\rho_\sigma:=\frac12(\delta_{-\sigma}+\delta_\sigma).
\]
Then the coupling argument gives
\[
W_2(\nu_c,\rho_\sigma)
\le
4\,\sigma^{-1/2}\Delta_6(\nu)^{1/4}
+8\,\sigma^{-2}\Delta_6(\nu)^{1/2}.
\]
In particular $\Delta_6(\nu)\ge0$.  This is a coefficient-defect estimate for
closeness to the symmetric two-point law.  The numerical constants $4$ and
$8$ are universal.  At this coefficient level the only non-universal
parameters displayed in the \(W_2\) modulus are the variance scale $\sigma$
and the coefficient defect $\Delta_6(\nu)$ itself; no finite-time remainder
constant is involved.
In particular, for fixed $\sigma$,
\[
W_2(\nu_c,\rho_\sigma)=O\!\bigl(\Delta_6(\nu)^{1/4}\bigr)
\qquad\text{as }\Delta_6(\nu)\downarrow 0.
\]
Moreover $\Delta_6(\nu)=0$ if and only if $\nu_c=\rho_\sigma$.
\end{corollary}

\begin{proof}
The displayed seventh absolute moment condition is exactly the hypothesis
of Lemma~\ref{lem:seventh-moment-bridge}.  Hence the coefficient appearing in
the entropy expansion is
\[
C_6=C_6^{(4)}(\nu)
\]
and therefore
\[
\Delta_6(\nu)=\Delta_6^{(4)}(\nu).
\]
The identity \eqref{eq:w2-theorem-moment-defect-form} is therefore
\eqref{eq:defect-sixth-fourth-moment} with the superscripted fourth-moment
polynomial replaced by the entropy coefficient defect through the bridge.
The finite fourth moment needed in
Lemma~\ref{lem:fourth-moment-algebraic-coupling} is automatic.  Substituting
the bridge identity into
\eqref{eq:fourth-moment-algebraic-coupling} gives the stated $W_2$ estimate,
and the non-negativity and zero characterization are those of
Lemma~\ref{lem:fourth-moment-algebraic-defect} after the same substitution.
The asymptotic
$O(\Delta_6^{1/4})$ statement for fixed $\sigma$ follows immediately from
the displayed estimate.
\end{proof}

Finite-time translations of the same coefficient estimate are bookkeeping
consequences of the eighth-order remainder in
Theorem~\ref{thm:poisson-kl-eighth}.  They are recorded separately in
Appendix~\ref{app:finite-time-defect-readouts} so that the main subsection
contains only the algebraic corollaries of the entropy coefficients.
