\subsection{Coefficient-level moment-defect corollaries}
\label{subsec:fourth-moment-stability}

This subsection has two layers.  The first layer records the sixth-order
coefficient defect.  The notation
\(A_6(\nu)\) and \(\Delta_6(\nu)\) is the notation block
\eqref{eq:n4-notation-a6}--\eqref{eq:n4-notation-delta6}: before an entropy
coefficient theorem is invoked these are elementary fourth-moment
polynomials, and after Theorem~\ref{thm:poisson-kl-even-orders} with \(N=3\)
they are the genuine \(t^{-6}\) entropy coefficient and its two-point defect.
The second layer is the primary finite-time application,
Theorem~\ref{thm:finite-time-poisson-entropy-stability}: after fixing a
variance-and-seventh-moment shell, it combines a measured KL value with
the shell constants explicitly displayed in that theorem,
\(C_{\sigma,M}^{(8)},B_{\sigma,M}^{(8)},T_{\sigma,M}^{(8)}\).  The measured
KL value supplies only the statistic \(\mathfrak D_t^{(6)}(\nu)\).  The subsidiary
fixed-law oracle form, Theorem~\ref{thm:oracle-finite-time-poisson-entropy-stability},
instead takes the particular law's supplied data \(A_8,B,T\).  In both
finite-time forms, the coefficient and remainder constants are fixed before
the measured KL value is inserted.

\paragraph{Fourth-moment algebra.}

\begin{lemma}[Fourth-moment algebraic defect]%
\label{lem:fourth-moment-algebraic-defect}
Let $\nu$ have mean~$\bar\gamma$, variance $\sigma^2>0$, and finite
centred fourth moment.  Define
\[
A_6(\nu):=
\frac{\sigma^6+6\mu_3^2-8\sigma^2\mu_4}{64},
\qquad
\Delta_6(\nu):=
-\frac{7}{64}\sigma^6-A_6(\nu).
\]
These are the \(A_6\) and \(\Delta_6\) entries of the notation block
\eqref{eq:n4-notation-a6}--\eqref{eq:n4-notation-delta6}, used here only as
fourth-moment polynomials.  On $(\RR,\nu)$ set
\begin{align*}
X&:=\frac{\gamma-\bar\gamma}{\sigma},
&
\beta_3&:=\EE[X^3],
&
\beta_4&:=\EE[X^4],
\\
Q_2&:=X^2-1-\beta_3X,
&
\kappa&:=\|Q_2\|_{L^2(\nu)}^2.
\end{align*}
Then the pure fourth-moment identity is
\begin{equation}\label{eq:defect-sixth-fourth-moment}
\Delta_6(\nu)
=\frac{\sigma^6}{32}\beta_3^2+\frac{\sigma^6}{8}\kappa,
\qquad
\kappa=\beta_4-1-\beta_3^2\ge0.
\end{equation}
Moreover, \(\Delta_6(\nu)=0\) if and only if the centred law of
$\gamma-\bar\gamma$ is $\rho_\sigma=\frac12(\delta_{-\sigma}
+\delta_\sigma)$.
\end{lemma}

\begin{proof}
The identity for $\kappa$ follows by expanding $Q_2^2$ and using
$\EE[X]=0$, $\EE[X^2]=1$, and $\EE[X^3]=\beta_3$:
\[
\kappa
=\EE[(X^2-1-\beta_3X)^2]
=\beta_4-1-\beta_3^2.
\]
Since $\mu_3=\sigma^3\beta_3$ and $\mu_4=\sigma^4\beta_4$,
\[
-\frac{7}{64}\sigma^6-A_6(\nu)
=\frac{\sigma^6}{64}\bigl(-7-1-6\beta_3^2+8\beta_4\bigr)
=\frac{\sigma^6}{32}\beta_3^2+\frac{\sigma^6}{8}\kappa.
\]
This proves \eqref{eq:defect-sixth-fourth-moment} and non-negativity.  If
\(\Delta_6(\nu)=0\), then $\beta_3=0$ and $\kappa=0$, so
$X^2=1$ almost surely.  The conditions $\EE[X]=0$ and $\EE[X^2]=1$ then
force $X=\pm1$ with equal probabilities, i.e. the centred law is
$\rho_\sigma$.  The converse is immediate by substitution.
\end{proof}

\begin{lemma}[Fourth-moment algebraic coupling]%
\label{lem:fourth-moment-algebraic-coupling}
Let $\nu$ have mean~$\bar\gamma$, variance $\sigma^2>0$, and finite
centred fourth moment.  Let $\nu_c$ be the centred law of
$X_c=\gamma-\bar\gamma$ and set
\[
\rho_\sigma:=\frac12(\delta_{-\sigma}+\delta_\sigma).
\]
For the fourth-moment polynomial $\Delta_6(\nu)$ defined above,
\[
\Delta_6(\nu)\ge0
\]
and
\begin{equation}\label{eq:fourth-moment-algebraic-coupling}
W_2(\nu_c,\rho_\sigma)
\le
4\,\sigma^{-1/2}\Delta_6(\nu)^{1/4}
+8\,\sigma^{-2}\Delta_6(\nu)^{1/2}.
\end{equation}
Moreover, $\Delta_6(\nu)=0$ if and only if
$\nu_c=\rho_\sigma$.  This is only an algebraic fourth-moment statement; it
does not introduce an entropy coefficient and is not a finite-time stability
or transport theorem.
\end{lemma}

\begin{proof}
All random variables below live on $(\RR,\nu)$.  Set
\[
X:=\frac{\gamma-\bar\gamma}{\sigma},
\qquad
\beta_3:=\EE[X^3],
\qquad
Q_2:=X^2-1-\beta_3X,
\qquad
\kappa:=\|Q_2\|_{L^2(\nu)}^2 .
\]
By \eqref{eq:defect-sixth-fourth-moment},
\begin{equation}\label{eq:algebraic-coupling-delta-six}
\Delta_6(\nu)
=\frac{\sigma^6}{32}\beta_3^2+\frac{\sigma^6}{8}\kappa .
\end{equation}
Define
\[
Y:=\mathbf{1}_{\{X\ge 0\}}-\mathbf{1}_{\{X<0\}}\in\{-1,1\}.
\]
For every real $x$, with $\operatorname{sgn}(0)=1$,
\[
|x-\operatorname{sgn}(x)|\le |x^2-1|.
\]
Therefore
\[
\EE[(X-Y)^2]\le \EE[(X^2-1)^2].
\]
The decomposition $X^2-1=\beta_3X+Q_2$ is orthogonal in $L^2(\nu)$ because
\[
\EE[XQ_2]=\EE[X^3]-\EE[X]-\beta_3\EE[X^2]=0.
\]
Thus
\[
\EE[(X^2-1)^2]
=\beta_3^2+\kappa
\le 2\beta_3^2+8\kappa
=\frac{64}{\sigma^6}\Delta_6(\nu),
\]
where the last equality is \eqref{eq:algebraic-coupling-delta-six}.  Since
$\EE[X]=0$,
\[
|\EE[Y]|=|\EE[Y-X]|
\le \EE[|Y-X|]
\le \EE[(Y-X)^2]^{1/2}
\le \frac{8}{\sigma^3}\Delta_6(\nu)^{1/2}.
\]
Let $Z$ be a symmetric Rademacher random variable.  If
$p=\mathbb P(Y=1)$, then $\EE[Y]=2p-1$ and the optimal transport between
$\mathcal L(Y)$ and the symmetric law on $\{-1,1\}$ moves
$|p-1/2|$ mass across distance~$2$.  Hence
\[
W_2(\mathcal L(Y),\mathcal L(Z))^2
=4|p-1/2|=2|\EE[Y]|
\le \frac{16}{\sigma^3}\Delta_6(\nu)^{1/2}.
\]
Also
\[
W_2(\mathcal L(X),\mathcal L(Y))
\le \EE[(X-Y)^2]^{1/2}
\le \frac{8}{\sigma^3}\Delta_6(\nu)^{1/2}.
\]
The laws satisfy $\nu_c=\mathcal L(\sigma X)$ and
$\rho_\sigma=\mathcal L(\sigma Z)$.  Taking square roots in the preceding
$W_2$ bound, then using scaling and the triangle inequality, gives
\eqref{eq:fourth-moment-algebraic-coupling}.
The zero statement follows from
Lemma~\ref{lem:fourth-moment-algebraic-defect}.
\end{proof}

\paragraph{Fourth-moment coefficient bridge.}

\begin{lemma}[Fourth-moment coefficient identification bridge]%
\label{lem:seventh-moment-bridge}
Assume that $\nu$ has mean~$\bar\gamma$, variance $\sigma^2>0$, and finite
centred fourth absolute moment.  Theorem~\ref{thm:poisson-kl-even-orders},
applied with \(N=3\), identifies the notation-block polynomial \(A_6(\nu)\)
as the universal \(t^{-6}\) entropy coefficient and gives
\[
D_{\mathrm{KL}}(P_t*\nu\|P_t(\cdot-\bar\gamma))
=\frac{\sigma^4}{8t^4}+\frac{A_6(\nu)}{t^6}
+o(t^{-6}).
\]
Equivalently,
\begin{equation}\label{eq:c6-entropy-bridge-explicit}
A_6(\nu)
=\frac{\sigma^6+6\mu_3^2-8\sigma^2\mu_4}{64}.
\end{equation}
Consequently \(\Delta_6(\nu)\) is the corresponding sixth-order
coefficient defect.  The seventh moment enters later through the
\(N=4,\eta=1\) quantitative \(O(t^{-9})\) remainder used in the finite-time
estimate.
\end{lemma}

\begin{proof}
The finite fourth absolute moment is precisely the \(N=3\) hypothesis in
Theorem~\ref{thm:poisson-kl-even-orders}; it is the algebraic threshold for
evaluating the displayed universal moment-polynomial coefficient \(A_6\).
That theorem gives
\[
D_{\mathrm{KL}}(P_t*\nu\|P_t(\cdot-\bar\gamma))
=\frac{\sigma^4}{8t^4}
+\frac{\sigma^6+6\mu_3^2-8\sigma^2\mu_4}{64\,t^6}
+o(t^{-6}),
\]
using the evaluated universal coefficients \(A_4(\nu)\) and
\(A_6(\nu)\).  The explicit value of \(A_6(\nu)\) is the sixth-order arithmetic already recorded
in Lemma~\ref{lem:n4-entropy-coefficient-substitution}; it involves only
\(\mu_2,\mu_3,\mu_4\) and hence is available under the fourth-moment
hypothesis.
The coefficient in an expansion of the form
\[
F(t)=a_4t^{-4}+a_6t^{-6}+o(t^{-6})
\]
is unique: if the same \(F\) is also
\(\tilde a_4t^{-4}+\tilde a_6t^{-6}+o(t^{-6})\), then multiplying the
difference first by \(t^4\) and then, after \(a_4=\tilde a_4\), by \(t^6\)
gives \(a_6=\tilde a_6\).  Hence the genuine entropy coefficient is exactly
the displayed polynomial \(A_6(\nu)\), proving
\eqref{eq:c6-entropy-bridge-explicit}.  Substituting this identity into
\(-7\sigma^6/64-A_6(\nu)\) gives the displayed polynomial
\(\Delta_6(\nu)\), now read as the coefficient defect.
\end{proof}

\begin{corollary}[Coefficient-defect coupling corollary]%
\label{cor:coefficient-defect-coupling}
Let $\nu$ be a probability measure on~$\RR$ with mean~$\bar\gamma$,
variance $\sigma^2>0$, and finite centred fourth absolute moment.
By Lemma~\ref{lem:seventh-moment-bridge}, \(A_6(\nu)\) is the genuine
\(t^{-6}\) entropy coefficient under this fourth-moment hypothesis, and
\(\Delta_6(\nu)\) is the corresponding entropy-coefficient defect.
On $(\RR,\nu)$ set
\[
X=\frac{\gamma-\bar\gamma}{\sigma},
\qquad
\beta_3=\EE X^3,
\qquad
Q_2=X^2-1-\beta_3X,
\qquad
\kappa=\|Q_2\|_{L^2(\nu)}^2 .
\]
Under this fourth-moment hypothesis,
\begin{equation}\label{eq:w2-theorem-moment-defect-form}
\Delta_6(\nu)
=\frac{\sigma^6}{32}\beta_3^2+\frac{\sigma^6}{8}\kappa .
\end{equation}
Let $\nu_c$ be the centred law of $X_c=\gamma-\bar\gamma$ and set
\[
\rho_\sigma:=\frac12(\delta_{-\sigma}+\delta_\sigma).
\]
Then the coupling argument gives
\[
W_2(\nu_c,\rho_\sigma)
\le
4\,\sigma^{-1/2}\Delta_6(\nu)^{1/4}
+8\,\sigma^{-2}\Delta_6(\nu)^{1/2}.
\]
In particular \(\Delta_6(\nu)\ge0\).  This is a coefficient-defect estimate for
closeness to the symmetric two-point law.  The numerical constants $4$ and
$8$ are universal.  At this coefficient level the only non-universal
parameters displayed in the \(W_2\) modulus are the variance scale $\sigma$
and the coefficient defect \(\Delta_6(\nu)\) itself; no finite-time remainder
constant or eighth-order oracle input is involved.
In particular, for fixed $\sigma$,
\[
W_2(\nu_c,\rho_\sigma)=O\!\bigl(\Delta_6(\nu)^{1/4}\bigr)
\qquad\text{as }\Delta_6(\nu)\downarrow 0.
\]
Moreover \(\Delta_6(\nu)=0\) if and only if
\(\nu_c=\rho_\sigma\).
\end{corollary}

\begin{proof}
The displayed fourth absolute moment condition is exactly the hypothesis
of Lemma~\ref{lem:seventh-moment-bridge}.  Hence \(A_6(\nu)\) is the
sixth-order entropy coefficient and \(\Delta_6(\nu)\) is its
coefficient defect.
The identity \eqref{eq:w2-theorem-moment-defect-form} is
\eqref{eq:defect-sixth-fourth-moment} after applying this bridge.
Substituting
\eqref{eq:fourth-moment-algebraic-coupling} gives the stated $W_2$ estimate,
and the non-negativity and zero characterization are those of
Lemma~\ref{lem:fourth-moment-algebraic-defect}.
The \(O((\Delta_6)^{1/4})\) statement for fixed \(\sigma\)
follows immediately from the displayed estimate.
\end{proof}

\begin{theorem}[Conditional certified-shell finite-time sixth-defect estimate]%
\label{thm:finite-time-poisson-entropy-stability}
Fix \(\sigma>0\) and \(M>0\).  The theorem is conditional on the
variance-and-seventh-moment shell
\[
\mathcal S_{\sigma,M}:=
\left\{\nu:\ \mu_2(\nu)=\sigma^2,\quad
\EE_\nu|\gamma-\bar\gamma|^7\le M\right\}.
\]
The constants below are part of the certificate and are fixed before any KL
value is inserted.  Set
\[
\mathcal M_{\sigma,M}^{(8)}
:=3\sigma^2+\sum_{n=3}^{6}(n+1)M^{n/7},
\qquad
\mathcal R_{\sigma,M}^{(8)}:=110M,
\]
\[
\mathcal S_{\sigma,M}^{(8)}
:=7\left(\sigma^2+\sum_{n=3}^{6}M^{n/7}\right)+110M.
\]
Define
\[
\mathcal P_{\sigma,M}^{(8)}:=
\sum_{r=2}^{4}\frac{1}{r(r-1)}
\sum_{\ell=1}^{r}\binom r\ell
(\mathcal M_{\sigma,M}^{(8)})^{r-\ell}
(\mathcal R_{\sigma,M}^{(8)})^\ell
\]
and
\[
\mathcal Q_{\sigma,M}^{(8)}:=
\sum_{r=2}^{4}\frac{1}{r(r-1)}
\sum_{m=9}^{6r}
\sum_{\substack{2\le n_1,\ldots,n_r\le6\\
n_1+\cdots+n_r=m}}
\prod_{j=1}^{r}(n_j+1)\,\Lambda_{n_j},
\]
where
\[
\Lambda_2:=\sigma^2,\qquad
\Lambda_n:=M^{n/7}\quad(3\le n\le6),
\]
and where empty inner sums are interpreted as zero.  Finally put
\[
B_{\sigma,M}^{(8)}
:=\mathcal P_{\sigma,M}^{(8)}+\mathcal Q_{\sigma,M}^{(8)}
+2(\mathcal S_{\sigma,M}^{(8)})^5,
\qquad
T_{\sigma,M}^{(8)}
:=\max\{1,(2\mathcal S_{\sigma,M}^{(8)})^{1/2}\}.
\]
These are finite, directly checkable constants depending only on
\(\sigma\) and \(M\).  They are chosen before any entropy observation is
inserted.
On the controlled shell they dominate the fixed-law seventh-moment constants \(B_{\nu,8}\) and
\(T_{\nu,8}\) from
Theorem~\ref{thm:poisson-kl-eighth-seventh-remainder}.  Define also the
explicit eighth-coefficient shell bound
\begin{equation}\label{eq:main-controlled-shell-c8}
C_{\sigma,M}^{(8)}
:=
\frac{1}{256}\left(
3\sigma^8+12\sigma^4M^{4/7}
+21\sigma^2M^{6/7}+50M^{8/7}
\right).
\end{equation}
Thus the theorem's finite-time certificate is the shell datum
\((\sigma,M)\), together with the displayed constants
\(C_{\sigma,M}^{(8)}\), \(B_{\sigma,M}^{(8)}\), and
\(T_{\sigma,M}^{(8)}\).  This certificate is fixed before any KL value is
inserted.
Let \(\nu\in\mathcal S_{\sigma,M}\).  Put
\[
h_t:=P_t*\nu,\qquad g_t(x):=P_t(x-\bar\gamma),
\]
let \(\nu_c\) be the centred law of \(\gamma-\bar\gamma\), and let
\(\rho_\sigma=\frac12(\delta_{-\sigma}+\delta_\sigma)\).  Let
\(A_6(\nu)\) be the sixth-order entropy coefficient identified by
Theorem~\ref{thm:poisson-kl-even-orders} with \(N=3\), and set
\[
\Delta_6(\nu)
:=-\frac{7}{64}\sigma^6-A_6(\nu).
\]
Define the finite-time sixth-order entropy statistic
\begin{equation}\label{eq:main-finite-time-sixth-deficit}
\mathfrak D_t^{(6)}(\nu)
:=
t^6\left(
\frac{\sigma^4}{8t^4}
-\frac{7\sigma^6}{64t^6}
-D_{\mathrm{KL}}(h_t\|g_t)
\right).
\end{equation}
For every \(t\ge T_{\sigma,M}^{(8)}\), the certified shell supplies numbers
\(A_8(\nu)\) and \(E_\nu(t)\) with
\[
D_{\mathrm{KL}}(h_t\|g_t)
=\frac{\sigma^4}{8t^4}
+\frac{A_6(\nu)}{t^6}
+\frac{A_8(\nu)}{t^8}
+E_\nu(t),
\qquad
|A_8(\nu)|\le C_{\sigma,M}^{(8)},\quad
|E_\nu(t)|\le B_{\sigma,M}^{(8)}t^{-9}.
\]
The statistic is therefore governed first by the exact certified-shell identity
\begin{equation}\label{eq:main-certified-shell-sixth-identity}
\mathfrak D_t^{(6)}(\nu)
=\Delta_6(\nu)-A_8(\nu)t^{-2}-t^6E_\nu(t),
\end{equation}
and only then by the bound
\begin{equation}\label{eq:main-finite-time-defect-estimate}
0\le\Delta_6(\nu)
\le
\bigl|\mathfrak D_t^{(6)}(\nu)\bigr|
+C_{\sigma,M}^{(8)}t^{-2}
+B_{\sigma,M}^{(8)}t^{-3}.
\end{equation}
Consequently, for every \(t\ge T_{\sigma,M}^{(8)}\),
\begin{equation}\label{eq:main-finite-time-w2-stability}
W_2(\nu_c,\rho_\sigma)
\le
4\,\sigma^{-1/2}\mathcal A_t^{1/4}
+8\,\sigma^{-2}\mathcal A_t^{1/2},
\end{equation}
where
\[
\mathcal A_t
:=
\bigl|\mathfrak D_t^{(6)}(\nu)\bigr|
+C_{\sigma,M}^{(8)}t^{-2}
+B_{\sigma,M}^{(8)}t^{-3}.
\]

In particular, if
\[
\bigl|\mathfrak D_t^{(6)}(\nu)\bigr|
\le \varepsilon,
\qquad
C_{\sigma,M}^{(8)}t^{-2}\le\varepsilon,
\qquad
B_{\sigma,M}^{(8)}t^{-3}\le\varepsilon,
\]
then
\[
W_2(\nu_c,\rho_\sigma)
\le
4\,\sigma^{-1/2}(3\varepsilon)^{1/4}
+8\,\sigma^{-2}(3\varepsilon)^{1/2}.
\]

Without a certified bound for the eighth coefficient \(A_8(\nu)\) and for the
remainder \(E_\nu(t)\), the single value
\(\mathfrak D_t^{(6)}(\nu)\) gives no bound on \(\Delta_6(\nu)\).
\end{theorem}

\begin{remark}
The observable inputs for \(\mathfrak D_t^{(6)}(\nu)\) are
\(D_{\mathrm{KL}}(h_t\|g_t)\), \(t\), and \(\sigma^2\).  They define only the
finite-time statistic.  The shell constants
\(C_{\sigma,M}^{(8)}\), \(B_{\sigma,M}^{(8)}\), and
\(T_{\sigma,M}^{(8)}\) are the displayed finite functions of the
variance-and-seventh-moment shell
before the KL value at time \(t\) is inserted into
\(\mathfrak D_t^{(6)}(\nu)\).  The theorem is therefore not a
one-observation stability theorem: the measured KL value is used only after
the controlled shell \((\sigma,M)\) has certified the possible sizes of
\(A_8(\nu)\) and \(E_\nu(t)\).
\end{remark}

\begin{proof}
Theorem~\ref{thm:poisson-kl-eighth-seventh-remainder} gives, for each fixed
\(\nu\in\mathcal S_{\sigma,M}\),
\[
D_{\mathrm{KL}}(h_t\|g_t)
=\frac{\sigma^4}{8t^4}
+\frac{A_6(\nu)}{t^6}
+\frac{A_8(\nu)}{t^8}
+E_\nu(t),
\qquad
|E_\nu(t)|\le B_{\nu,8}t^{-9}
\]
for \(t\ge T_{\nu,8}\).  We first check the shell domination asserted in the
statement.  If \(\nu\in\mathcal S_{\sigma,M}\), then Lyapunov's inequality
gives \(|\mu_n|\le M^{n/7}\) for \(3\le n\le6\), while
\(\mu_2=\sigma^2\) and \(M_7(\nu)\le M\).  Hence the definitions of
\(\mathcal M_{\nu,8}\), \(\mathcal C_{\nu,8}\), and
\(\mathcal S_{\nu,8}\) in
Theorem~\ref{thm:poisson-kl-eighth-seventh-remainder} give
\[
\mathcal M_{\nu,8}\le\mathcal M_{\sigma,M}^{(8)},\qquad
\mathcal C_{\nu,8}\le\mathcal R_{\sigma,M}^{(8)},\qquad
\mathcal S_{\nu,8}\le\mathcal S_{\sigma,M}^{(8)}.
\]
The displayed finite sums defining \(\mathcal P_{\nu,8}\) and
\(\mathcal Q_{\nu,8}\) are monotone in these absolute moment bounds, so
\[
B_{\nu,8}\le B_{\sigma,M}^{(8)},\qquad
T_{\nu,8}\le T_{\sigma,M}^{(8)}.
\]
Therefore the quantitative expansion holds for
every \(t\ge T_{\sigma,M}^{(8)}\) with
\(|E_\nu(t)|\le B_{\sigma,M}^{(8)}t^{-9}\).  By
Theorem~\ref{thm:poisson-kl-eighth},
\[
A_8(\nu)
=\frac{1}{256}\left(
3\sigma^8-12\sigma^4\mu_4
+9\sigma^2\mu_3^2+12\sigma^2\mu_6
-30\mu_3\mu_5+20\mu_4^2
\right).
\]
Lyapunov's inequality gives \(|\mu_j|\le M^{j/7}\) for \(3\le j\le6\),
because \(\EE|X_c|^7\le M\).  Therefore
\[
|A_8(\nu)|
\le C_{\sigma,M}^{(8)}
\]
with \(C_{\sigma,M}^{(8)}\) as in
\eqref{eq:main-controlled-shell-c8}.  Substitution into
\eqref{eq:main-finite-time-sixth-deficit} gives the identity
\eqref{eq:main-certified-shell-sixth-identity}.  Combining this identity with
the certified bounds for \(A_8(\nu)\) and \(E_\nu(t)\) yields
\[
0\le \Delta_6(\nu)
\le
\bigl|\mathfrak D_t^{(6)}(\nu)\bigr|
+C_{\sigma,M}^{(8)}t^{-2}
+B_{\sigma,M}^{(8)}t^{-3},
\]
where the non-negativity is Corollary~\ref{cor:coefficient-defect-coupling}.
This proves \eqref{eq:main-finite-time-defect-estimate}.  The same corollary
also gives
\[
W_2(\nu_c,\rho_\sigma)
\le
4\,\sigma^{-1/2}\Delta_6(\nu)^{1/4}
+8\,\sigma^{-2}\Delta_6(\nu)^{1/2}.
\]
Inserting the preceding defect bound into the coefficient-defect coupling
proves
\eqref{eq:main-finite-time-w2-stability}, and the
\(\varepsilon\)-form is immediate.
\end{proof}

The algebra behind the oracle finite-time form is the identity
\begin{equation}\label{eq:oracle-sixth-deficit-identity-before-theorem}
\mathfrak D_t^{(6)}(\nu)
=\Delta_6(\nu)-A_8(\nu)t^{-2}-t^6E_\nu(t).
\end{equation}
The conclusion requires a certified bound on \(|A_8|\) and
\(|E_\nu(t)|\); a single KL value at one time does not supply those bounds.

\begin{theorem}[Oracle finite-time bound from oracle/shell eighth-order expansion data]%
\label{thm:oracle-finite-time-poisson-entropy-stability}
The quantities \(A_8\in\RR\), \(B\ge0\), and \(T>0\) are oracle or shell
inputs supplied before the measured KL value is used.  Without such certified
data, the value of \(\mathfrak D_t^{(6)}(\nu)\) at one time does not bound
\(\Delta_6(\nu)\).
Let $\nu$ be a probability measure on~$\RR$ with mean~$\bar\gamma$, variance
\(\sigma^2>0\), and finite centred fourth absolute moment.  Put
\[
h_t:=P_t*\nu,\qquad g_t(x):=P_t(x-\bar\gamma),
\]
let \(\nu_c\) be the centred law of \(\gamma-\bar\gamma\), and let
\(\rho_\sigma=\frac12(\delta_{-\sigma}+\delta_\sigma)\).  Let
\(A_6(\nu)\) be the sixth-order entropy coefficient identified by
Theorem~\ref{thm:poisson-kl-even-orders} with \(N=3\), and set
\[
\Delta_6(\nu)
:=-\frac{7}{64}\sigma^6-A_6(\nu).
\]
Assume that these supplied data satisfy
\begin{equation}\label{eq:main-oracle-eighth-expansion}
D_{\mathrm{KL}}(h_t\|g_t)
=\frac{\sigma^4}{8t^4}
+\frac{A_6(\nu)}{t^6}
+\frac{A_8}{t^8}
+E(t),
\qquad
|E(t)|\le Bt^{-9}
\end{equation}
for every \(t\ge T\).  Define \(\mathfrak D_t^{(6)}(\nu)\) by
\eqref{eq:main-finite-time-sixth-deficit}.  Then, for every \(t\ge T\),
\begin{equation}\label{eq:main-oracle-finite-time-defect-estimate}
\left|
\Delta_6(\nu)-\mathfrak D_t^{(6)}(\nu)
-\frac{A_8}{t^2}
\right|
\le Bt^{-3}.
\end{equation}
Consequently, for every \(t\ge T\),
\begin{equation}\label{eq:main-oracle-finite-time-w2-stability}
W_2(\nu_c,\rho_\sigma)
\le
4\,\sigma^{-1/2}\mathcal A_{t,A_8,B}^{1/4}
+8\,\sigma^{-2}\mathcal A_{t,A_8,B}^{1/2},
\end{equation}
where
\[
\mathcal A_{t,A_8,B}
:=
\bigl|\mathfrak D_t^{(6)}(\nu)+A_8t^{-2}\bigr|
+Bt^{-3}.
\]

In particular, if
\[
\bigl|\mathfrak D_t^{(6)}(\nu)+A_8t^{-2}\bigr|
\le \varepsilon,
\qquad
Bt^{-3}\le\varepsilon,
\]
then
\[
W_2(\nu_c,\rho_\sigma)
\le
4\,\sigma^{-1/2}(2\varepsilon)^{1/4}
+8\,\sigma^{-2}(2\varepsilon)^{1/2}.
\]

\end{theorem}

\begin{remark}
The fixed-law theorem is the oracle form of the preceding controlled-shell
bound.  In applications justified by
Theorem~\ref{thm:poisson-kl-eighth-seventh-remainder}, one may take
\(A_8=A_8(\nu)\), \(B=B_{\nu,8}\), and \(T=T_{\nu,8}\).  These are
law-specific expansion data; the measured KL value supplies only the
finite-time statistic \(\mathfrak D_t^{(6)}(\nu)\).
\end{remark}

\begin{proof}
Substitution of \eqref{eq:main-oracle-eighth-expansion} into
\eqref{eq:main-finite-time-sixth-deficit} gives, for \(t\ge T\),
\[
\begin{aligned}
\mathfrak D_t^{(6)}(\nu)
&=-\frac{7}{64}\sigma^6-A_6(\nu)
-\frac{A_8}{t^2}-t^6E(t)\\
&=\Delta_6(\nu)-\frac{A_8}{t^2}-t^6E(t).
\end{aligned}
\]
Hence
\[
0\le \Delta_6(\nu)
\le \bigl|\mathfrak D_t^{(6)}(\nu)+A_8t^{-2}\bigr|+Bt^{-3},
\]
where the non-negativity is Corollary~\ref{cor:coefficient-defect-coupling}.
This proves \eqref{eq:main-oracle-finite-time-defect-estimate}.  The same
corollary also gives
\[
W_2(\nu_c,\rho_\sigma)
\le
4\,\sigma^{-1/2}\Delta_6(\nu)^{1/4}
+8\,\sigma^{-2}\Delta_6(\nu)^{1/2}.
\]
Inserting the preceding bound for \(\Delta_6(\nu)\) proves
\eqref{eq:main-oracle-finite-time-w2-stability}, and the
\(\varepsilon\)-form is immediate.
\end{proof}

\begin{corollary}[Controlled-shell restatement of the primary finite-time bound]\label{cor:controlled-shell-finite-time-oracle}
Under the hypotheses and notation of
Theorem~\ref{thm:finite-time-poisson-entropy-stability}, for every
\(t\ge T_{\sigma,M}^{(8)}\),
\[
W_2(\nu_c,\rho_\sigma)
\le
4\,\sigma^{-1/2}\mathcal A_{t,\sigma,M}^{1/4}
+8\,\sigma^{-2}\mathcal A_{t,\sigma,M}^{1/2},
\]
where
\[
\mathcal A_{t,\sigma,M}:=
\bigl|\mathfrak D_t^{(6)}(\nu)\bigr|
+C_{\sigma,M}^{(8)}t^{-2}+B_{\sigma,M}^{(8)}t^{-3}.
\]
Here the shell constants \(C_{\sigma,M}^{(8)}\),
\(B_{\sigma,M}^{(8)}\), and \(T_{\sigma,M}^{(8)}\) are the explicit functions
of \((\sigma,M)\) displayed in
Theorem~\ref{thm:finite-time-poisson-entropy-stability}; they are fixed
before the KL value entering \(\mathfrak D_t^{(6)}(\nu)\) is used.  The
right-hand side therefore uses exactly the measured KL statistic together
with those controlled-shell constants, not fixed-law oracle data
\(A_8,B,T\).
\end{corollary}

\begin{proof}
This is exactly \eqref{eq:main-finite-time-w2-stability} with the displayed
definition of \(\mathcal A_t\).
\end{proof}

Appendix~\ref{app:finite-time-defect-estimates} records the constant-regime
bookkeeping and checks that the explicit shell formulas used here dominate
the fixed-law constants uniformly on the controlled moment shell.
