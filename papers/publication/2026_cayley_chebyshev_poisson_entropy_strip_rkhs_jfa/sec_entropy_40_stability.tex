\subsection{Coefficient-level moment-defect corollaries}
\label{subsec:fourth-moment-stability}

This subsection has two layers.  The first layer records coefficient-level
consequences of the large-time entropy expansion.  The first two lemmas are
purely algebraic: with a finite centred fourth moment, \(C_6(\nu)\) and
\(\Delta_6(\nu)\) are elementary fourth-moment polynomials and make no claim
about an entropy asymptotic by themselves.  The theorem statements below add
the finite-moment hypotheses under which these same polynomials are identified
with the sixth-order entropy coefficient and its defect.  The second layer is
a finite-time readout theorem: it uses the eighth-order remainder to bound the
coefficient defect from a single large-time entropy observation, with an
explicit \(A_8(\nu)t^{-2}\) correction and an \(O(t^{-3})\) residual on the
renormalized sixth-order scale.  This is a finite-time stability theorem for
this Poisson entropy jet.  It is not a logarithmic-HLS, Talagrand, or
point-process transport stability theorem for the different functionals in
the comparison literature.

\paragraph{Fourth-moment algebra.}

\begin{lemma}[Fourth-moment algebraic defect]%
\label{lem:fourth-moment-algebraic-defect}
Let $\nu$ have mean~$\bar\gamma$, variance $\sigma^2>0$, and finite
centred fourth moment.  Define
\[
C_6(\nu):=
\frac{\sigma^6+6\mu_3^2-8\sigma^2\mu_4}{64},
\qquad
\Delta_6(\nu):=
-\frac{7}{64}\sigma^6-C_6(\nu).
\]
In this lemma these are only fourth-moment polynomials; no entropy
coefficient is asserted until a later theorem imposes the finite-moment
hypothesis needed for the entropy expansion.  On $(\RR,\nu)$ set
\begin{align*}
X&:=\frac{\gamma-\bar\gamma}{\sigma},
&
\beta_3&:=\EE[X^3],
&
\beta_4&:=\EE[X^4],
\\
Q_2&:=X^2-1-\beta_3X,
&
\kappa&:=\|Q_2\|_{L^2(\nu)}^2.
\end{align*}
Then the pure fourth-moment identity is
\begin{equation}\label{eq:defect-sixth-fourth-moment}
\Delta_6(\nu)
=\frac{\sigma^6}{32}\beta_3^2+\frac{\sigma^6}{8}\kappa,
\qquad
\kappa=\beta_4-1-\beta_3^2\ge0.
\end{equation}
Moreover, $\Delta_6(\nu)=0$ if and only if the centred law of
$\gamma-\bar\gamma$ is $\rho_\sigma=\frac12(\delta_{-\sigma}
+\delta_\sigma)$.
\end{lemma}

\begin{proof}
The identity for $\kappa$ follows by expanding $Q_2^2$ and using
$\EE[X]=0$, $\EE[X^2]=1$, and $\EE[X^3]=\beta_3$:
\[
\kappa
=\EE[(X^2-1-\beta_3X)^2]
=\beta_4-1-\beta_3^2.
\]
Since $\mu_3=\sigma^3\beta_3$ and $\mu_4=\sigma^4\beta_4$,
\[
-\frac{7}{64}\sigma^6-C_6(\nu)
=\frac{\sigma^6}{64}\bigl(-7-1-6\beta_3^2+8\beta_4\bigr)
=\frac{\sigma^6}{32}\beta_3^2+\frac{\sigma^6}{8}\kappa.
\]
This proves \eqref{eq:defect-sixth-fourth-moment} and non-negativity.  If
$\Delta_6(\nu)=0$, then $\beta_3=0$ and $\kappa=0$, so
$X^2=1$ almost surely.  The conditions $\EE[X]=0$ and $\EE[X^2]=1$ then
force $X=\pm1$ with equal probabilities, i.e. the centred law is
$\rho_\sigma$.  The converse is immediate by substitution.
\end{proof}

\begin{lemma}[Fourth-moment algebraic coupling]%
\label{lem:fourth-moment-algebraic-coupling}
Let $\nu$ have mean~$\bar\gamma$, variance $\sigma^2>0$, and finite
centred fourth moment.  Let $\nu_c$ be the centred law of
$\Delta=\gamma-\bar\gamma$ and set
\[
\rho_\sigma:=\frac12(\delta_{-\sigma}+\delta_\sigma).
\]
For the fourth-moment polynomial $\Delta_6(\nu)$ defined above,
\[
\Delta_6(\nu)\ge0
\]
and
\begin{equation}\label{eq:fourth-moment-algebraic-coupling}
W_2(\nu_c,\rho_\sigma)
\le
4\,\sigma^{-1/2}\Delta_6(\nu)^{1/4}
+8\,\sigma^{-2}\Delta_6(\nu)^{1/2}.
\end{equation}
Moreover, $\Delta_6(\nu)=0$ if and only if
$\nu_c=\rho_\sigma$.  This is only an algebraic fourth-moment statement; it
does not introduce an entropy coefficient and is not a finite-time stability
or transport theorem.
\end{lemma}

\begin{proof}
All random variables below live on $(\RR,\nu)$.  Set
\[
X:=\frac{\gamma-\bar\gamma}{\sigma},
\qquad
\beta_3:=\EE[X^3],
\qquad
Q_2:=X^2-1-\beta_3X,
\qquad
\kappa:=\|Q_2\|_{L^2(\nu)}^2 .
\]
By \eqref{eq:defect-sixth-fourth-moment},
\begin{equation}\label{eq:algebraic-coupling-delta-six}
\Delta_6(\nu)
=\frac{\sigma^6}{32}\beta_3^2+\frac{\sigma^6}{8}\kappa .
\end{equation}
Define
\[
Y:=\mathbf{1}_{\{X\ge 0\}}-\mathbf{1}_{\{X<0\}}\in\{-1,1\}.
\]
For every real $x$, with $\operatorname{sgn}(0)=1$,
\[
|x-\operatorname{sgn}(x)|\le |x^2-1|.
\]
Therefore
\[
\EE[(X-Y)^2]\le \EE[(X^2-1)^2].
\]
The decomposition $X^2-1=\beta_3X+Q_2$ is orthogonal in $L^2(\nu)$ because
\[
\EE[XQ_2]=\EE[X^3]-\EE[X]-\beta_3\EE[X^2]=0.
\]
Thus
\[
\EE[(X^2-1)^2]
=\beta_3^2+\kappa
\le 2\beta_3^2+8\kappa
=\frac{64}{\sigma^6}\Delta_6(\nu),
\]
where the last equality is \eqref{eq:algebraic-coupling-delta-six}.  Since
$\EE[X]=0$,
\[
|\EE[Y]|=|\EE[Y-X]|
\le \EE[|Y-X|]
\le \EE[(Y-X)^2]^{1/2}
\le \frac{8}{\sigma^3}\Delta_6(\nu)^{1/2}.
\]
Let $Z$ be a symmetric Rademacher random variable.  If
$p=\mathbb P(Y=1)$, then $\EE[Y]=2p-1$ and the optimal transport between
$\mathcal L(Y)$ and the symmetric law on $\{-1,1\}$ moves
$|p-1/2|$ mass across distance~$2$.  Hence
\[
W_2(\mathcal L(Y),\mathcal L(Z))^2
=4|p-1/2|=2|\EE[Y]|
\le \frac{16}{\sigma^3}\Delta_6(\nu)^{1/2}.
\]
Also
\[
W_2(\mathcal L(X),\mathcal L(Y))
\le \EE[(X-Y)^2]^{1/2}
\le \frac{8}{\sigma^3}\Delta_6(\nu)^{1/2}.
\]
The laws satisfy $\nu_c=\mathcal L(\sigma X)$ and
$\rho_\sigma=\mathcal L(\sigma Z)$.  Taking square roots in the preceding
$W_2$ bound, then using scaling and the triangle inequality, gives
\eqref{eq:fourth-moment-algebraic-coupling}.
The zero statement follows from
Lemma~\ref{lem:fourth-moment-algebraic-defect}.
\end{proof}

\paragraph{Seventh-moment bridge.}

\begin{lemma}[Seventh-moment coefficient identification bridge]%
\label{lem:seventh-moment-bridge}
Assume that $\nu$ has mean~$\bar\gamma$, variance $\sigma^2>0$, and
\[
\int_{\RR}\abs{\gamma-\bar\gamma}^{7}\,\dd\nu(\gamma)<\infty.
\]
This lemma is only the coefficient-identification step needed before the
algebraic fourth-moment estimate can be read as an entropy-coefficient
statement.  The genuine \(t^{-6}\) coefficient in the asymptotic expansion
of \(D_{\mathrm{KL}}(P_t*\nu\|P_t(\cdot-\bar\gamma))\) supplied by
Theorem~\ref{thm:poisson-kl-eighth} is \(C_6(\nu)\).  Equivalently,
\[
D_{\mathrm{KL}}(P_t*\nu\|P_t(\cdot-\bar\gamma))
=\frac{\sigma^4}{8t^4}+\frac{C_6(\nu)}{t^6}+O_\nu(t^{-8}).
\]
Let \(C_6(\nu)\) and \(\Delta_6(\nu)\) be the fourth-moment polynomials
defined in Lemma~\ref{lem:fourth-moment-algebraic-defect}.  Then
\begin{equation}\label{eq:delta-six-bridge}
-\frac{7}{64}\sigma^6-C_6(\nu)=\Delta_6(\nu).
\end{equation}
More explicitly, the coefficient identified by the entropy asymptotic is
\begin{equation}\label{eq:c6-entropy-bridge-explicit}
C_6(\nu)
=A_6(\nu)
=\frac{\sigma^6+6\mu_3^2-8\sigma^2\mu_4}{64}.
\end{equation}
Thus, under the displayed finite seventh absolute moment hypothesis, the
elementary polynomial \(C_6(\nu)\) is the actual sixth-order entropy
coefficient and \(\Delta_6(\nu)\) is the corresponding entropy-coefficient
defect.
\end{lemma}

\begin{proof}
The finite seventh absolute moment implies the fourth moment needed to define
\(C_6(\nu)\) and \(\Delta_6(\nu)\).  It also implies the finite-sixth
hypothesis of Theorem~\ref{thm:poisson-kl-eighth}, whose explicit coefficient
audit gives
\[
D_{\mathrm{KL}}(P_t*\nu\|P_t(\cdot-\bar\gamma))
=\frac{\sigma^4}{8t^4}
+\frac{\sigma^6+6\mu_3^2-8\sigma^2\mu_4}{64\,t^6}
+O_\nu(t^{-8}).
\]
The coefficient in an expansion of the form
\[
F(t)=a_4t^{-4}+a_6t^{-6}+O(t^{-8})
\]
is unique: if the same \(F\) is also
\(\tilde a_4t^{-4}+\tilde a_6t^{-6}+O(t^{-8})\), then multiplying the
difference first by \(t^4\) and then, after \(a_4=\tilde a_4\), by \(t^6\)
gives \(a_6=\tilde a_6\).  Hence the genuine entropy coefficient is exactly
\[
C_6(\nu)
=\frac{\sigma^6+6\mu_3^2-8\sigma^2\mu_4}{64}
\]
which proves \eqref{eq:c6-entropy-bridge-explicit}.  Substituting this
identity into \(-7\sigma^6/64-C_6(\nu)\) gives \(\Delta_6(\nu)\),
completing \eqref{eq:delta-six-bridge}.
\end{proof}

\begin{corollary}[Coefficient-defect coupling corollary]%
\label{cor:coefficient-defect-coupling}
Let $\nu$ be a probability measure on~$\RR$ with mean~$\bar\gamma$,
variance $\sigma^2>0$, and finite centred seventh absolute moment, i.e.
\[
\int_{\RR}\abs{\gamma-\bar\gamma}^{7}\,\dd\nu(\gamma)<\infty.
\]
Let \(C_6(\nu)\) and \(\Delta_6(\nu)\) be the polynomial quantities from
Lemma~\ref{lem:fourth-moment-algebraic-defect}.  By
Lemma~\ref{lem:seventh-moment-bridge}, \(C_6(\nu)\) is the genuine
\(t^{-6}\) entropy coefficient under the displayed hypothesis, and
\(\Delta_6(\nu)\) is the corresponding entropy-coefficient defect.
On $(\RR,\nu)$ set
\[
X=\frac{\gamma-\bar\gamma}{\sigma},
\qquad
\beta_3=\EE X^3,
\qquad
Q_2=X^2-1-\beta_3X,
\qquad
\kappa=\|Q_2\|_{L^2(\nu)}^2 .
\]
Under the same seventh-moment hypothesis needed to identify the entropy
coefficient,
\begin{equation}\label{eq:w2-theorem-moment-defect-form}
\Delta_6(\nu)
=\frac{\sigma^6}{32}\beta_3^2+\frac{\sigma^6}{8}\kappa .
\end{equation}
Let $\nu_c$ be the centred law of $\Delta=\gamma-\bar\gamma$ and set
\[
\rho_\sigma:=\frac12(\delta_{-\sigma}+\delta_\sigma).
\]
Then the coupling argument gives
\[
W_2(\nu_c,\rho_\sigma)
\le
4\,\sigma^{-1/2}\Delta_6(\nu)^{1/4}
+8\,\sigma^{-2}\Delta_6(\nu)^{1/2}.
\]
In particular $\Delta_6(\nu)\ge0$.  This is a coefficient-defect estimate for
closeness to the symmetric two-point law.  The numerical constants $4$ and
$8$ are universal.  At this coefficient level the only non-universal
parameters displayed in the \(W_2\) modulus are the variance scale $\sigma$
and the coefficient defect $\Delta_6(\nu)$ itself; no finite-time remainder
constant is involved.
In particular, for fixed $\sigma$,
\[
W_2(\nu_c,\rho_\sigma)=O\!\bigl(\Delta_6(\nu)^{1/4}\bigr)
\qquad\text{as }\Delta_6(\nu)\downarrow 0.
\]
Moreover $\Delta_6(\nu)=0$ if and only if $\nu_c=\rho_\sigma$.
\end{corollary}

\begin{proof}
The displayed seventh absolute moment condition is exactly the hypothesis
of Lemma~\ref{lem:seventh-moment-bridge}.  Hence the polynomial \(C_6(\nu)\)
is the entropy coefficient and \(\Delta_6(\nu)\) is its coefficient defect.
The identity \eqref{eq:w2-theorem-moment-defect-form} is
\eqref{eq:defect-sixth-fourth-moment}.
The finite fourth moment needed in
Lemma~\ref{lem:fourth-moment-algebraic-coupling} is automatic.  Substituting
\eqref{eq:fourth-moment-algebraic-coupling} gives the stated $W_2$ estimate,
and the non-negativity and zero characterization are those of
Lemma~\ref{lem:fourth-moment-algebraic-defect}.
The asymptotic
$O(\Delta_6^{1/4})$ statement for fixed $\sigma$ follows immediately from
the displayed estimate.
\end{proof}

\begin{theorem}[Finite-time Poisson entropy stability readout]%
\label{thm:finite-time-poisson-entropy-stability}
Let $\nu$ be a probability measure on~$\RR$ with mean~$\bar\gamma$, variance
$\sigma^2>0$, and finite centred seventh absolute moment.  Put
\[
h_t:=P_t*\nu,\qquad g_t(x):=P_t(x-\bar\gamma),
\]
let \(\nu_c\) be the centred law of \(\gamma-\bar\gamma\), and let
\(\rho_\sigma=\frac12(\delta_{-\sigma}+\delta_\sigma)\).  Let \(C_6(\nu)\)
and \(A_8(\nu)\) be the audited sixth- and eighth-order coefficients from
Theorem~\ref{thm:poisson-kl-eighth}.  Let
\(B_{\nu,8},T_{\nu,8}\) be the constants from that quantitative theorem, so
that the eighth-order remainder estimate
\eqref{eq:poisson-kl-eighth-remainder} holds.  Define
\[
\Delta_6(\nu):=-\frac{7}{64}\sigma^6-C_6(\nu)
\]
and the finite-time sixth-order entropy readout
\begin{equation}\label{eq:main-finite-time-sixth-deficit}
\mathfrak D_t^{(6)}(\nu)
:=
t^6\left(
\frac{\sigma^4}{8t^4}
-\frac{7\sigma^6}{64t^6}
-D_{\mathrm{KL}}(h_t\|g_t)
\right).
\end{equation}
Then for every \(t\ge T_{\nu,8}\),
\begin{equation}\label{eq:main-finite-time-defect-readout}
\left|
\Delta_6(\nu)-\mathfrak D_t^{(6)}(\nu)
-\frac{A_8(\nu)}{t^2}
\right|
\le B_{\nu,8}t^{-3}.
\end{equation}
Consequently, for every \(t\ge T_{\nu,8}\),
\begin{equation}\label{eq:main-finite-time-w2-stability}
W_2(\nu_c,\rho_\sigma)
\le
4\,\sigma^{-1/2}\mathcal A_t^{1/4}
+8\,\sigma^{-2}\mathcal A_t^{1/2},
\end{equation}
where
\[
\mathcal A_t
:=
\bigl|\mathfrak D_t^{(6)}(\nu)\bigr|
+|A_8(\nu)|t^{-2}
+B_{\nu,8}t^{-3}.
\]
In particular, if the finite-time entropy lies within
\[
\varepsilon t^{-6}
\]
of the two-point sixth-order model after the displayed \(A_8(\nu)t^{-8}\)
correction, and if \(B_{\nu,8}t^{-3}\le\varepsilon\), then
\[
W_2(\nu_c,\rho_\sigma)
\le
4\,\sigma^{-1/2}(2\varepsilon)^{1/4}
+8\,\sigma^{-2}(2\varepsilon)^{1/2}.
\]
\end{theorem}

\begin{proof}
Theorem~\ref{thm:poisson-kl-eighth-seventh-remainder} gives, for
\(t\ge T_{\nu,8}\),
\[
D_{\mathrm{KL}}(h_t\|g_t)
=\frac{\sigma^4}{8t^4}
+\frac{C_6(\nu)}{t^6}
+\frac{A_8(\nu)}{t^8}
+E_{\nu,8}(t),
\qquad
|E_{\nu,8}(t)|\le B_{\nu,8}t^{-9}.
\]
Substitution into \eqref{eq:main-finite-time-sixth-deficit} gives
\[
\mathfrak D_t^{(6)}(\nu)
=-\frac{7}{64}\sigma^6-C_6(\nu)
-\frac{A_8(\nu)}{t^2}
-t^6E_{\nu,8}(t)
=\Delta_6(\nu)
-\frac{A_8(\nu)}{t^2}
-t^6E_{\nu,8}(t),
\]
which proves \eqref{eq:main-finite-time-defect-readout}.  By
Corollary~\ref{cor:coefficient-defect-coupling}, \(\Delta_6(\nu)\ge0\) and
\[
W_2(\nu_c,\rho_\sigma)
\le
4\,\sigma^{-1/2}\Delta_6(\nu)^{1/4}
+8\,\sigma^{-2}\Delta_6(\nu)^{1/2}.
\]
The readout estimate implies \(0\le\Delta_6(\nu)\le\mathcal A_t\).  Inserting
this bound into the coefficient-defect coupling proves
\eqref{eq:main-finite-time-w2-stability}.  The final displayed
\(\varepsilon\)-form is the special case
\[
\bigl|\mathfrak D_t^{(6)}(\nu)+A_8(\nu)t^{-2}\bigr|
\le\varepsilon,
\qquad
B_{\nu,8}t^{-3}\le\varepsilon .
\]
\end{proof}
