\subsection{Coefficient-level moment-defect corollaries}
\label{subsec:fourth-moment-stability}

This subsection has two layers.  The first layer records the sixth-order
coefficient defect.  The first two lemmas are purely algebraic: with a finite
centred fourth moment, \(C_6^{\rm alg}(\nu)\) and
\(\Delta_6^{\rm alg}(\nu)\) are elementary fourth-moment polynomials.
The bridge lemma then invokes Theorem~\ref{thm:poisson-kl-even-orders} with
\(N=3\), sharp for identification of the universal moment-polynomial
coefficient \(A_6\), so the same fourth-moment hypothesis identifies these
polynomials as the genuine \(t^{-6}\) entropy coefficient and its two-point
defect.  No
seventh moment is used in this coefficient-defect
\(W_2\) estimate.  The second layer is an a posteriori finite-time
sixth-defect bound: it uses the eighth-order remainder to bound the coefficient defect from
the observed renormalized entropy statistic together with supplied
eighth-order expansion data \(A_8,B,T\).  Under the seventh-moment remainder
theorem one may take \(A_8=A_8^{\rm ent}(\nu)\),
\(B=B_{\nu,8}\), and \(T=T_{\nu,8}\).  Here
\(A_8^{\rm ent}(\nu)=A_8^{\rm alg}(\nu)\) is the sixth-moment polynomial already identified as the
eighth entropy coefficient, while \(B_{\nu,8}\) and
\(T_{\nu,8}\) are the seventh-moment remainder constant and threshold
supplied by the optimized positive-margin theorem.  Thus the fixed-law
statement is an oracle coefficient-defect estimate, not a claim that these
quantities are read from the observed entropy value.  The shell-uniform
version uses one entropy observation together with preassigned shell
constants; it is stated in
Corollary~\ref{cor:controlled-shell-finite-time-oracle} after the fixed-law
estimate and is recorded again in
Appendix~\ref{app:finite-time-defect-estimates}, where
\(C_{\sigma,M}^{(8)}\), \(B_{\sigma,M}^{(8)}\), and
\(T_{\sigma,M}^{(8)}\) are supplied in advance by a controlled
seventh-moment shell.  This is a finite-time coefficient-defect estimate for
this Poisson entropy jet.  It is not a logarithmic-HLS, Talagrand, or
point-process transport stability theorem for the different functionals in
the comparison literature.

\paragraph{Fourth-moment algebra.}

\begin{lemma}[Fourth-moment algebraic defect]%
\label{lem:fourth-moment-algebraic-defect}
Let $\nu$ have mean~$\bar\gamma$, variance $\sigma^2>0$, and finite
centred fourth moment.  Define
\[
C_6^{\rm alg}(\nu):=
\frac{\sigma^6+6\mu_3^2-8\sigma^2\mu_4}{64},
\qquad
\Delta_6^{\rm alg}(\nu):=
-\frac{7}{64}\sigma^6-C_6^{\rm alg}(\nu).
\]
In this lemma these are only fourth-moment polynomials; no entropy
coefficient is asserted until a later theorem imposes the finite-moment
hypothesis needed for the entropy expansion.  On $(\RR,\nu)$ set
\begin{align*}
X&:=\frac{\gamma-\bar\gamma}{\sigma},
&
\beta_3&:=\EE[X^3],
&
\beta_4&:=\EE[X^4],
\\
Q_2&:=X^2-1-\beta_3X,
&
\kappa&:=\|Q_2\|_{L^2(\nu)}^2.
\end{align*}
Then the pure fourth-moment identity is
\begin{equation}\label{eq:defect-sixth-fourth-moment}
\Delta_6^{\rm alg}(\nu)
=\frac{\sigma^6}{32}\beta_3^2+\frac{\sigma^6}{8}\kappa,
\qquad
\kappa=\beta_4-1-\beta_3^2\ge0.
\end{equation}
Moreover, \(\Delta_6^{\rm alg}(\nu)=0\) if and only if the centred law of
$\gamma-\bar\gamma$ is $\rho_\sigma=\frac12(\delta_{-\sigma}
+\delta_\sigma)$.
\end{lemma}

\begin{proof}
The identity for $\kappa$ follows by expanding $Q_2^2$ and using
$\EE[X]=0$, $\EE[X^2]=1$, and $\EE[X^3]=\beta_3$:
\[
\kappa
=\EE[(X^2-1-\beta_3X)^2]
=\beta_4-1-\beta_3^2.
\]
Since $\mu_3=\sigma^3\beta_3$ and $\mu_4=\sigma^4\beta_4$,
\[
-\frac{7}{64}\sigma^6-C_6^{\rm alg}(\nu)
=\frac{\sigma^6}{64}\bigl(-7-1-6\beta_3^2+8\beta_4\bigr)
=\frac{\sigma^6}{32}\beta_3^2+\frac{\sigma^6}{8}\kappa.
\]
This proves \eqref{eq:defect-sixth-fourth-moment} and non-negativity.  If
\(\Delta_6^{\rm alg}(\nu)=0\), then $\beta_3=0$ and $\kappa=0$, so
$X^2=1$ almost surely.  The conditions $\EE[X]=0$ and $\EE[X^2]=1$ then
force $X=\pm1$ with equal probabilities, i.e. the centred law is
$\rho_\sigma$.  The converse is immediate by substitution.
\end{proof}

\begin{lemma}[Fourth-moment algebraic coupling]%
\label{lem:fourth-moment-algebraic-coupling}
Let $\nu$ have mean~$\bar\gamma$, variance $\sigma^2>0$, and finite
centred fourth moment.  Let $\nu_c$ be the centred law of
$X_c=\gamma-\bar\gamma$ and set
\[
\rho_\sigma:=\frac12(\delta_{-\sigma}+\delta_\sigma).
\]
For the fourth-moment polynomial $\Delta_6^{\rm alg}(\nu)$ defined above,
\[
\Delta_6^{\rm alg}(\nu)\ge0
\]
and
\begin{equation}\label{eq:fourth-moment-algebraic-coupling}
W_2(\nu_c,\rho_\sigma)
\le
4\,\sigma^{-1/2}\Delta_6^{\rm alg}(\nu)^{1/4}
+8\,\sigma^{-2}\Delta_6^{\rm alg}(\nu)^{1/2}.
\end{equation}
Moreover, $\Delta_6^{\rm alg}(\nu)=0$ if and only if
$\nu_c=\rho_\sigma$.  This is only an algebraic fourth-moment statement; it
does not introduce an entropy coefficient and is not a finite-time stability
or transport theorem.
\end{lemma}

\begin{proof}
All random variables below live on $(\RR,\nu)$.  Set
\[
X:=\frac{\gamma-\bar\gamma}{\sigma},
\qquad
\beta_3:=\EE[X^3],
\qquad
Q_2:=X^2-1-\beta_3X,
\qquad
\kappa:=\|Q_2\|_{L^2(\nu)}^2 .
\]
By \eqref{eq:defect-sixth-fourth-moment},
\begin{equation}\label{eq:algebraic-coupling-delta-six}
\Delta_6^{\rm alg}(\nu)
=\frac{\sigma^6}{32}\beta_3^2+\frac{\sigma^6}{8}\kappa .
\end{equation}
Define
\[
Y:=\mathbf{1}_{\{X\ge 0\}}-\mathbf{1}_{\{X<0\}}\in\{-1,1\}.
\]
For every real $x$, with $\operatorname{sgn}(0)=1$,
\[
|x-\operatorname{sgn}(x)|\le |x^2-1|.
\]
Therefore
\[
\EE[(X-Y)^2]\le \EE[(X^2-1)^2].
\]
The decomposition $X^2-1=\beta_3X+Q_2$ is orthogonal in $L^2(\nu)$ because
\[
\EE[XQ_2]=\EE[X^3]-\EE[X]-\beta_3\EE[X^2]=0.
\]
Thus
\[
\EE[(X^2-1)^2]
=\beta_3^2+\kappa
\le 2\beta_3^2+8\kappa
=\frac{64}{\sigma^6}\Delta_6^{\rm alg}(\nu),
\]
where the last equality is \eqref{eq:algebraic-coupling-delta-six}.  Since
$\EE[X]=0$,
\[
|\EE[Y]|=|\EE[Y-X]|
\le \EE[|Y-X|]
\le \EE[(Y-X)^2]^{1/2}
\le \frac{8}{\sigma^3}\Delta_6^{\rm alg}(\nu)^{1/2}.
\]
Let $Z$ be a symmetric Rademacher random variable.  If
$p=\mathbb P(Y=1)$, then $\EE[Y]=2p-1$ and the optimal transport between
$\mathcal L(Y)$ and the symmetric law on $\{-1,1\}$ moves
$|p-1/2|$ mass across distance~$2$.  Hence
\[
W_2(\mathcal L(Y),\mathcal L(Z))^2
=4|p-1/2|=2|\EE[Y]|
\le \frac{16}{\sigma^3}\Delta_6^{\rm alg}(\nu)^{1/2}.
\]
Also
\[
W_2(\mathcal L(X),\mathcal L(Y))
\le \EE[(X-Y)^2]^{1/2}
\le \frac{8}{\sigma^3}\Delta_6^{\rm alg}(\nu)^{1/2}.
\]
The laws satisfy $\nu_c=\mathcal L(\sigma X)$ and
$\rho_\sigma=\mathcal L(\sigma Z)$.  Taking square roots in the preceding
$W_2$ bound, then using scaling and the triangle inequality, gives
\eqref{eq:fourth-moment-algebraic-coupling}.
The zero statement follows from
Lemma~\ref{lem:fourth-moment-algebraic-defect}.
\end{proof}

\paragraph{Fourth-moment coefficient bridge.}

\begin{lemma}[Fourth-moment coefficient identification bridge]%
\label{lem:seventh-moment-bridge}
Assume that $\nu$ has mean~$\bar\gamma$, variance $\sigma^2>0$, and finite
centred fourth absolute moment.  This lemma is the coefficient-identification
step needed before the algebraic fourth-moment estimate can be read as an
entropy-coefficient statement.  Theorem~\ref{thm:poisson-kl-even-orders},
applied with \(N=3\), identifies the universal moment-polynomial coefficient
\(A_6\) and gives
\[
D_{\mathrm{KL}}(P_t*\nu\|P_t(\cdot-\bar\gamma))
=\frac{\sigma^4}{8t^4}+\frac{A_6(\nu)}{t^6}
+o(t^{-6}).
\]
Let \(C_6^{\rm alg}(\nu)\) and \(\Delta_6^{\rm alg}(\nu)\) denote the
fourth-moment polynomials defined in
Lemma~\ref{lem:fourth-moment-algebraic-defect}.  Then the coefficient
identified by the entropy asymptotic is
\begin{equation}\label{eq:c6-entropy-bridge-explicit}
C_6^{\rm ent}(\nu)
=A_6(\nu)
=C_6^{\rm alg}(\nu)
=\frac{\sigma^6+6\mu_3^2-8\sigma^2\mu_4}{64}.
\end{equation}
Thus the elementary fourth-moment polynomial \(C_6^{\rm alg}(\nu)\) becomes
the actual sixth-order entropy coefficient after this bridge.  We then set
\[
\Delta_6^{\rm ent}(\nu):=-\frac{7}{64}\sigma^6-C_6^{\rm ent}(\nu)
=\Delta_6^{\rm alg}(\nu).
\]
A seventh moment is not part of this identification; it is only used later
when the finite-time estimate calls on the \(N=4,\eta=1\) quantitative
\(O(t^{-9})\) remainder.
\end{lemma}

\begin{proof}
The finite fourth absolute moment is precisely the hypothesis, sharp for
identification of the universal moment-polynomial coefficient \(A_6\), in
Theorem~\ref{thm:poisson-kl-even-orders} with \(N=3\).  That theorem gives
\[
D_{\mathrm{KL}}(P_t*\nu\|P_t(\cdot-\bar\gamma))
=\frac{\sigma^4}{8t^4}
+\frac{\sigma^6+6\mu_3^2-8\sigma^2\mu_4}{64\,t^6}
+o(t^{-6}),
\]
using the evaluated universal coefficients \(A_4(\nu)\) and
\(A_6(\nu)\).  The explicit value of \(A_6(\nu)\) is the sixth-order arithmetic already recorded
in Lemma~\ref{lem:eighth-order-coefficient-arithmetic}; it involves only
\(\mu_2,\mu_3,\mu_4\) and hence is available under the fourth-moment
hypothesis.
The coefficient in an expansion of the form
\[
F(t)=a_4t^{-4}+a_6t^{-6}+o(t^{-6})
\]
is unique: if the same \(F\) is also
\(\tilde a_4t^{-4}+\tilde a_6t^{-6}+o(t^{-6})\), then multiplying the
difference first by \(t^4\) and then, after \(a_4=\tilde a_4\), by \(t^6\)
gives \(a_6=\tilde a_6\).  Hence the genuine entropy coefficient is exactly
\[
C_6^{\rm ent}(\nu)=C_6^{\rm alg}(\nu)
=\frac{\sigma^6+6\mu_3^2-8\sigma^2\mu_4}{64}
\]
which proves \eqref{eq:c6-entropy-bridge-explicit}.  Substituting this
identity into \(-7\sigma^6/64-C_6^{\rm ent}(\nu)\), and comparing
with the algebraic definition of \(\Delta_6^{\rm alg}(\nu)\), gives
the same displayed polynomial for the coefficient defect.
\end{proof}

\begin{corollary}[Coefficient-defect coupling corollary]%
\label{cor:coefficient-defect-coupling}
Let $\nu$ be a probability measure on~$\RR$ with mean~$\bar\gamma$,
variance $\sigma^2>0$, and finite centred fourth absolute moment.
Let \(C_6^{\rm alg}(\nu)\) and
\(\Delta_6^{\rm alg}(\nu)\) be the polynomial quantities defined in
Lemma~\ref{lem:fourth-moment-algebraic-defect}.  By
Lemma~\ref{lem:seventh-moment-bridge},
\(C_6^{\rm ent}(\nu)=C_6^{\rm alg}(\nu)\) is the genuine \(t^{-6}\)
entropy coefficient under this fourth-moment hypothesis, and
\(\Delta_6^{\rm ent}(\nu)=\Delta_6^{\rm alg}(\nu)\) is the corresponding
entropy-coefficient defect.
On $(\RR,\nu)$ set
\[
X=\frac{\gamma-\bar\gamma}{\sigma},
\qquad
\beta_3=\EE X^3,
\qquad
Q_2=X^2-1-\beta_3X,
\qquad
\kappa=\|Q_2\|_{L^2(\nu)}^2 .
\]
Under this fourth-moment hypothesis,
\begin{equation}\label{eq:w2-theorem-moment-defect-form}
\Delta_6^{\rm ent}(\nu)
=\frac{\sigma^6}{32}\beta_3^2+\frac{\sigma^6}{8}\kappa .
\end{equation}
Let $\nu_c$ be the centred law of $X_c=\gamma-\bar\gamma$ and set
\[
\rho_\sigma:=\frac12(\delta_{-\sigma}+\delta_\sigma).
\]
Then the coupling argument gives
\[
W_2(\nu_c,\rho_\sigma)
\le
4\,\sigma^{-1/2}\Delta_6^{\rm ent}(\nu)^{1/4}
+8\,\sigma^{-2}\Delta_6^{\rm ent}(\nu)^{1/2}.
\]
In particular \(\Delta_6^{\rm ent}(\nu)\ge0\).  This is a coefficient-defect estimate for
closeness to the symmetric two-point law.  The numerical constants $4$ and
$8$ are universal.  At this coefficient level the only non-universal
parameters displayed in the \(W_2\) modulus are the variance scale $\sigma$
and the coefficient defect \(\Delta_6^{\rm ent}(\nu)\) itself; no finite-time remainder
constant or eighth-order oracle input is involved.
In particular, for fixed $\sigma$,
\[
W_2(\nu_c,\rho_\sigma)=O\!\bigl(\Delta_6^{\rm ent}(\nu)^{1/4}\bigr)
\qquad\text{as }\Delta_6^{\rm ent}(\nu)\downarrow 0.
\]
Moreover \(\Delta_6^{\rm ent}(\nu)=0\) if and only if
\(\nu_c=\rho_\sigma\).
\end{corollary}

\begin{proof}
The displayed fourth absolute moment condition is exactly the hypothesis
of Lemma~\ref{lem:seventh-moment-bridge}.  Hence \(C_6^{\rm ent}(\nu)\) is the
sixth-order entropy coefficient and \(\Delta_6^{\rm ent}(\nu)\) is its
coefficient defect.
The identity \eqref{eq:w2-theorem-moment-defect-form} is
\eqref{eq:defect-sixth-fourth-moment} after applying this bridge.
Substituting
\eqref{eq:fourth-moment-algebraic-coupling} gives the stated $W_2$ estimate,
and the non-negativity and zero characterization are those of
Lemma~\ref{lem:fourth-moment-algebraic-defect}.
The \(O((\Delta_6)^{1/4})\) statement for fixed \(\sigma\)
follows immediately from the displayed estimate.
\end{proof}

\begin{theorem}[Finite-time sixth-defect bound conditional on supplied eighth-order expansion data]%
\label{thm:finite-time-poisson-entropy-stability}
Assume that numbers \(A_8\in\RR\), \(B\ge0\), and \(T>0\) are supplied
as oracle or shell inputs before the entropy observation is used; they are
not estimated from, fitted to, or determined by the single entropy value in
\(\mathfrak D_t^{(6)}(\nu)\) below.
Let $\nu$ be a probability measure on~$\RR$ with mean~$\bar\gamma$, variance
\(\sigma^2>0\), and finite centred fourth absolute moment.  Put
\[
h_t:=P_t*\nu,\qquad g_t(x):=P_t(x-\bar\gamma),
\]
let \(\nu_c\) be the centred law of \(\gamma-\bar\gamma\), and let
\(\rho_\sigma=\frac12(\delta_{-\sigma}+\delta_\sigma)\).  Let
\(C_6^{\rm ent}(\nu)\) be the sixth-order entropy coefficient identified by
Theorem~\ref{thm:poisson-kl-even-orders} with \(N=3\), and set
\[
\Delta_6^{\rm ent}(\nu)
:=-\frac{7}{64}\sigma^6-C_6^{\rm ent}(\nu).
\]
Assume that these supplied data satisfy
\begin{equation}\label{eq:main-oracle-eighth-expansion}
D_{\mathrm{KL}}(h_t\|g_t)
=\frac{\sigma^4}{8t^4}
+\frac{C_6^{\rm ent}(\nu)}{t^6}
+\frac{A_8}{t^8}
+E(t),
\qquad
|E(t)|\le Bt^{-9}
\end{equation}
for every \(t\ge T\).  Define the finite-time sixth-order entropy statistic
\begin{equation}\label{eq:main-finite-time-sixth-deficit}
\mathfrak D_t^{(6)}(\nu)
:=
t^6\left(
\frac{\sigma^4}{8t^4}
-\frac{7\sigma^6}{64t^6}
-D_{\mathrm{KL}}(h_t\|g_t)
\right).
\end{equation}
Then, for every \(t\ge T\),
\begin{equation}\label{eq:main-finite-time-defect-estimate}
\left|
\Delta_6^{\rm ent}(\nu)-\mathfrak D_t^{(6)}(\nu)
-\frac{A_8}{t^2}
\right|
\le Bt^{-3}.
\end{equation}
Consequently, for every \(t\ge T\),
\begin{equation}\label{eq:main-finite-time-w2-stability}
W_2(\nu_c,\rho_\sigma)
\le
4\,\sigma^{-1/2}\mathcal A_t^{1/4}
+8\,\sigma^{-2}\mathcal A_t^{1/2},
\end{equation}
where
\[
\mathcal A_t
:=
\bigl|\mathfrak D_t^{(6)}(\nu)+A_8t^{-2}\bigr|
+Bt^{-3}.
\]

In particular, if
\[
\bigl|\mathfrak D_t^{(6)}(\nu)+A_8t^{-2}\bigr|
\le \varepsilon,
\qquad
Bt^{-3}\le\varepsilon,
\]
then
\[
W_2(\nu_c,\rho_\sigma)
\le
4\,\sigma^{-1/2}(2\varepsilon)^{1/4}
+8\,\sigma^{-2}(2\varepsilon)^{1/2}.
\]

\end{theorem}

\begin{remark}
The observable inputs for \(\mathfrak D_t^{(6)}(\nu)\) are
\(D_{\mathrm{KL}}(h_t\|g_t)\), \(t\), and \(\sigma^2\).  The constants
\(A_8,B,T\) in \eqref{eq:main-oracle-eighth-expansion} are separate supplied
inputs, not consequences of the observed entropy value.  In the fixed-law
application they are the law-specific eighth-order coefficient and
seventh-moment remainder data.
In the shell version, \(A_8^{\rm ent}=A_8^{\rm alg}\) is supplied by the
coefficient-identification theorem, and the bounds for \(B_{\nu,8}\) and
\(T_{\nu,8}\) come from the positive-margin seventh-moment remainder theorem.
\end{remark}

\begin{proof}
Substitution of \eqref{eq:main-oracle-eighth-expansion} into
\eqref{eq:main-finite-time-sixth-deficit} gives, for \(t\ge T\),
\[
\begin{aligned}
\mathfrak D_t^{(6)}(\nu)
&=-\frac{7}{64}\sigma^6-C_6^{\rm ent}(\nu)
-\frac{A_8}{t^2}-t^6E(t)\\
&=\Delta_6^{\rm ent}(\nu)-\frac{A_8}{t^2}-t^6E(t).
\end{aligned}
\]
Hence
\[
0\le \Delta_6^{\rm ent}(\nu)
\le \bigl|\mathfrak D_t^{(6)}(\nu)+A_8t^{-2}\bigr|+Bt^{-3},
\]
where the non-negativity is Corollary~\ref{cor:coefficient-defect-coupling}.
This proves \eqref{eq:main-finite-time-defect-estimate}.  The same corollary
also gives
\[
W_2(\nu_c,\rho_\sigma)
\le
4\,\sigma^{-1/2}\Delta_6^{\rm ent}(\nu)^{1/4}
+8\,\sigma^{-2}\Delta_6^{\rm ent}(\nu)^{1/2}.
\]
Inserting this bound into the coefficient-defect coupling proves
\eqref{eq:main-finite-time-w2-stability}, and the
\(\varepsilon\)-form is immediate.
\end{proof}

\begin{corollary}[Controlled-shell finite-time bound from supplied shell data]\label{cor:controlled-shell-finite-time-oracle}
Fix \(\sigma>0\) and \(M>0\), and assume the shell constants below are
preassigned shell inputs, not estimated from the entropy observation.
Restrict to probability laws with \(\mu_2=\sigma^2\) and
\(\EE|\gamma-\bar\gamma|^7\le M\).  Let
\[
C_{\sigma,M}^{(8)}
:=
\frac{1}{256}\left(
3\sigma^8+12\sigma^4M^{4/7}
+21\sigma^2M^{6/7}+50M^{8/7}
\right),
\]
and let \(B_{\sigma,M}^{(8)}\) and \(T_{\sigma,M}^{(8)}\) be preassigned
finite shell constants dominating \(B_{\nu,8}\) and \(T_{\nu,8}\), respectively,
on this class.  Then, for every \(t\ge T_{\sigma,M}^{(8)}\),
\[
W_2(\nu_c,\rho_\sigma)
\le
4\,\sigma^{-1/2}\mathcal A_{t,\sigma,M}^{1/4}
+8\,\sigma^{-2}\mathcal A_{t,\sigma,M}^{1/2},
\]
where
\[
\mathcal A_{t,\sigma,M}:=
\bigl|\mathfrak D_t^{(6)}(\nu)\bigr|
+C_{\sigma,M}^{(8)}t^{-2}+B_{\sigma,M}^{(8)}t^{-3}.
\]
Here the single entropy value in \(\mathfrak D_t^{(6)}(\nu)\) does not
determine the shell constants \(C_{\sigma,M}^{(8)}\),
\(B_{\sigma,M}^{(8)}\), or \(T_{\sigma,M}^{(8)}\).
\end{corollary}

\begin{proof}
Theorem~\ref{thm:poisson-kl-eighth-seventh-remainder}
supplies \eqref{eq:main-oracle-eighth-expansion} with
\(A_8=A_8^{\rm ent}(\nu)\), \(B=B_{\nu,8}\), and \(T=T_{\nu,8}\).  On the
shell, Lyapunov's inequality gives \(|\mu_j|\le M^{j/7}\) for
\(3\le j\le6\), and insertion in the polynomial for \(A_8^{\rm alg}\) gives
\(|A_8^{\rm ent}(\nu)|\le C_{\sigma,M}^{(8)}\).  Applying
Theorem~\ref{thm:finite-time-poisson-entropy-stability} and replacing
\(|A_8^{\rm ent}(\nu)|\), \(B_{\nu,8}\), and \(T_{\nu,8}\) by the displayed
shell constants gives the result.
\end{proof}
