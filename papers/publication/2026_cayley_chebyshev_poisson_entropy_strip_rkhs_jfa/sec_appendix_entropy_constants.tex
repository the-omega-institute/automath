\section{Finite-moment entropy constant regimes}
\label{app:finite-moment-entropy-constants}

\subsection{Constants and variants for the finite-moment entropy hierarchy}%
\label{subsubsec:finite-moment-constants-variants}

Theorem~\ref{thm:poisson-kl-even-orders} is the exact finite-moment
coefficient-identification theorem.  This subsection fixes the quantitative
constant regimes and common remainder forms used later.
The algebraic lower-moment obstruction for the displayed universal polynomial
coefficient is stated next to the main theorem as
Proposition~\ref{prop:poisson-kl-moment-threshold-sharpness}; the appendix
only records the constant regimes and variants.

\paragraph{Retained-polynomial form.}%
\phantomsection\label{thm:finite-moment-entropy-remainder}
Let \(N\ge2\), and let \(\nu\) be a probability measure on \(\RR\) with mean
\(\bar\gamma\) and finite centred \((2N-2)\)th absolute moment.  Put
\[
\mathcal R_N(t;\nu)
:=D_{\mathrm{KL}}(P_t*\nu\|P_t(\cdot-\bar\gamma))
-\sum_{m=2}^{N}A_{2m}(\nu)t^{-2m}.
\]
Theorem~\ref{thm:poisson-kl-even-orders} gives
\[
\mathcal R_N(t;\nu)=o(t^{-2N})
\qquad(t\to\infty).
\]
After the universal even coefficient polynomial is subtracted, the exact
centred \((2N-2)\)th absolute moment hypothesis leaves no further contribution
at any order \(t^{-k}\) with \(k\le2N\).  The positive-margin and moment-ball
estimates below add quantitative remainders; they are not separate
coefficient-identification theorems.

\paragraph{Positive-margin constants.}%
\phantomsection\label{thm:positive-margin-retained-order-entropy}
Let \(N\ge2\), let \(0<\eta\le1\), and let \(\nu\) be a probability
measure on \(\RR\) with mean \(\bar\gamma\) and finite centred
\((2N-2+\eta)\)th absolute moment.  Put
\[
L:=2N-2,\qquad p:=L+\eta,\qquad
X_c:=\gamma-\bar\gamma.
\]
Let \(C_L\) and \(B_L\) be the finite constants from
Theorem~\ref{thm:finite-moment-density-remainder}, and set
\[
C_\rho:=2C_L\EE|X_c|^p,\qquad
M_A:=\sum_{n=2}^{L}|\mu_n|\,\|u_n\|_\infty,\qquad
S:=B_L\sum_{n=2}^{L}|\mu_n|+C_\rho .
\]
Define the density-remainder product constant
\[
C_{\nu,N,\eta}:=
\sum_{r=2}^{N}\frac{1}{r(r-1)}
\sum_{\ell=1}^{r}\binom r\ell
M_A^{r-\ell}C_\rho^\ell ,
\]
the finite retained-mode tail
\[
Q_{\nu,N}:=
\sum_{r=2}^{N}\frac{1}{r(r-1)}
\sum_{m=2N+1}^{Lr}
\sum_{\substack{2\le n_1,\ldots,n_r\le L\\
n_1+\cdots+n_r=m}}
|\mu_{n_1}\cdots\mu_{n_r}|\,
\|u_{n_1}\cdots u_{n_r}\|_{L^1(\omega)} ,
\]
with the convention that an empty sum is zero, and
\[
C'_{\nu,N}:=2S^{N+1},\qquad T_{\nu,N,\eta}:=\max\{1,(2S)^{1/2}\}.
\]
With these names, the positive-margin retained-order estimate is
\begin{align}
\label{eq:positive-margin-retained-order-theorem}
\left|
D_{\mathrm{KL}}(P_t*\nu\|P_t(\cdot-\bar\gamma))
-\sum_{m=2}^{N}A_{2m}(\nu)t^{-2m}
\right|
&\le
C_{\nu,N,\eta}\,t^{-(2N+\eta)}
\notag\\
&\quad
{}+Q_{\nu,N}\,t^{-(2N+1)}
+C'_{\nu,N}t^{-(2N+2)}
\end{align}
for every \(t\ge T_{\nu,N,\eta}\).  Equivalently, in the optimized notation
used in the derivation,
\[
C_{\nu,N,\eta}=P_N^{\rm opt}(C_\rho^{\rm opt},M_A^{\rm opt}),\qquad
Q_{\nu,N}=Q_N^{\rm opt}(\mu,u),
\]
\[
C'_{\nu,N}=2(S^{\rm opt})^{N+1},\qquad
T_{\nu,N,\eta}=T^{\rm opt}.
\]
The coefficient-identification theorem uses the same retained density modes
\(u_2,\ldots,u_{2N-2}\).  The proof is the finite density-to-entropy
calculation: Theorem~\ref{thm:finite-moment-density-remainder} with
truncation \(L=2N-2\) gives a density remainder of size
\(t^{-(2N-2+\eta)}\); every entropy monomial containing that remainder also
contains the quadratic entropy scale, producing the first term above.  The
pure retained-mode polynomial has a finite tail of total degree \(>2N\),
which gives \(Q_{\nu,N}t^{-(2N+1)}\), and the Taylor tail gives
\(C'_{\nu,N}t^{-(2N+2)}\).

\paragraph{Stronger \(2N+\eta\) truncation.}%
\phantomsection\label{cor:finite-moment-entropy-constant-remainder}
Let \(N\ge2\), let \(0<\eta\le1\), and let \(\nu\) be a probability measure
on \(\RR\) with mean \(\bar\gamma\) and finite centred
\((2N+\eta)\)th absolute moment.  This stronger hypothesis gives a convenient
fixed-law constant obtained by retaining density modes through \(2N\) instead
of through \(2N-2\).  Put
\[
p:=2N+\eta,\qquad X_c:=\gamma-\bar\gamma .
\]
Let
\[
C_\rho:=2C_{2N}\EE|X_c|^p,
\qquad
M_A:=\sum_{n=2}^{2N}|\mu_n|\,\|u_n\|_\infty,
\]
where \(C_{2N}\) is the finite density constant from
Theorem~\ref{thm:finite-moment-density-remainder}.  Define
\[
P_N(C_\rho,M_A)
:=
\sum_{r=2}^{N}\frac1{r(r-1)}
\sum_{\ell=1}^{r}\binom r\ell
M_A^{r-\ell}C_\rho^\ell ,
\]
and
\[
Q_N(\mu,u)
:=
\sum_{r=2}^{N}\frac1{r(r-1)}
\sum_{m=2N+1}^{2Nr}
\sum_{\substack{2\le n_1,\ldots,n_r\le2N\\
n_1+\cdots+n_r=m}}
|\mu_{n_1}\cdots\mu_{n_r}|\,
\|u_{n_1}\cdots u_{n_r}\|_{L^1(\omega)} .
\]
Finally set
\[
S:=B_{2N}\sum_{n=2}^{2N}|\mu_n|+C_\rho,
\qquad
T:=\max\{1,(2S)^{1/2}\},
\]
where \(B_{2N}\) is the finite mode-size constant from
Theorem~\ref{thm:finite-moment-density-remainder}.  The same
density-to-entropy calculation gives, for every \(t\ge T\),
\begin{align}
\label{eq:finite-moment-entropy-remainder-estimate}
\left|
D_{\mathrm{KL}}(P_t*\nu\|P_t(\cdot-\bar\gamma))
-\sum_{m=2}^{N}A_{2m}(\nu)t^{-2m}
\right|
&\le
\{P_N(C_\rho,M_A)+Q_N(\mu,u)\}t^{-(2N+\eta)}
\notag\\
&\qquad
+2S^{N+1}t^{-(2N+2)} .
\end{align}
The refined three-term estimate behind this convenience envelope is
\[
P_N(C_\rho,M_A)t^{-(2N+\eta+2)}
+Q_N(\mu,u)t^{-(2N+1)}
+2S^{N+1}t^{-(2N+2)} .
\]
For a centred \(p\)-moment ball \(\EE|X_c|^p\le M\), the same estimate is
made uniform by replacing the law-dependent quantities with the closed finite
dominating constants of
Lemma~\ref{lem:moment-ball-uniform-constant-domination}.  In particular
\(\|u_n\|_\infty\le n+1\) and
\(\|u_{n_1}\cdots u_{n_r}\|_{L^1(\omega)}
\le\prod_j(n_j+1)\) give a version involving no unspecified mode norms.

\paragraph{Optimized retained-order notation and the \(N=4\) envelope.}%
\phantomsection\label{cor:optimized-retained-order-entropy-bridge}
Let \(N\ge2\), let \(0<\eta\le1\), and let \(\nu\) be a probability
measure on \(\RR\) with mean \(\bar\gamma\) and finite centred
\((2N-2+\eta)\)th absolute moment.  Put
\[
p:=2N-2+\eta,\qquad L:=2N-2,\qquad
X_c:=\gamma-\bar\gamma .
\]
Let
\[
C_\rho^{\rm opt}:=2C_L\EE|X_c|^p,
\qquad
M_A^{\rm opt}:=\sum_{n=2}^{L}|\mu_n|\,\|u_n\|_\infty,
\]
where \(C_L\) is the finite density constant from
Theorem~\ref{thm:finite-moment-density-remainder}.  Define
\[
P_N^{\rm opt}(C_\rho^{\rm opt},M_A^{\rm opt})
:=
\sum_{r=2}^{N}\frac{1}{r(r-1)}
\sum_{\ell=1}^{r}\binom r\ell
(M_A^{\rm opt})^{r-\ell}(C_\rho^{\rm opt})^\ell
\]
and
\[
Q_N^{\rm opt}(\mu,u)
:=
\sum_{r=2}^{N}\frac{1}{r(r-1)}
\sum_{m=2N+1}^{Lr}
\sum_{\substack{2\le n_1,\ldots,n_r\le L\\
n_1+\cdots+n_r=m}}
|\mu_{n_1}\cdots\mu_{n_r}|\,
\|u_{n_1}\cdots u_{n_r}\|_{L^1(\omega)} ,
\]
with the convention that an empty sum is zero.  Finally set
\[
S^{\rm opt}:=B_L\sum_{n=2}^{L}|\mu_n|+C_\rho^{\rm opt},
\qquad
T^{\rm opt}:=\max\{1,(2S^{\rm opt})^{1/2}\},
\]
where \(B_L\) is the finite mode-size constant in
Theorem~\ref{thm:finite-moment-density-remainder}.  These are exactly the
quantities in the positive-margin estimate above; equivalently,
\[
C_{\nu,N,\eta}=P_N^{\rm opt},\qquad
Q_{\nu,N}=Q_N^{\rm opt},\qquad
C'_{\nu,N}=2(S^{\rm opt})^{N+1}.
\]
For reference we record the coarse envelope obtained from
\eqref{eq:positive-margin-retained-order-theorem}: for \(t\ge T^{\rm opt}\),
\begin{equation}\label{eq:optimized-retained-order-entropy-envelope}
\left|
D_{\mathrm{KL}}(P_t*\nu\|P_t(\cdot-\bar\gamma))
-\sum_{m=2}^{N}A_{2m}(\nu)t^{-2m}
\right|
\le
\bigl\{
P_N^{\rm opt}+Q_N^{\rm opt}+2(S^{\rm opt})^{N+1}
\bigr\}t^{-(2N+\eta)}
\end{equation}
  for \(t\ge T^{\rm opt}\).

  In the specialization \(N=4\), \(\eta=1\), assume
  \(\EE|X_c|^7<\infty\), set \(L=6\), and use the constants in
  Definition~\ref{def:n4-finite-time-constant-dictionary}.  Then, for every
  \(t\ge T_{\nu,8}\),
  \begin{equation}\label{eq:optimized-n4-seventh-envelope}
  \left|
  D_{\mathrm{KL}}(P_t*\nu\|P_t(\cdot-\bar\gamma))
  -A_4(\nu)t^{-4}
  -A_6(\nu)t^{-6}
  -A_8(\nu)t^{-8}
  \right|
  \le B_{\nu,8}t^{-9}.
  \end{equation}
  Thus the order-nine constant used below is dominated by the closed envelope
  in the dictionary and can be checked directly from the law's moment data
  through order seven.
  The fixed-law coefficient identity is the \(N=4\) specialization of
  Theorem~\ref{thm:poisson-kl-even-orders}; after the explicit coefficient computation in
  Theorem~\ref{thm:poisson-kl-eighth}, the displayed
  \(A_8(\nu)\) is the actual \(t^{-8}\) entropy coefficient.  The displayed rate is the
  \(N=4,\eta=1\) specialization of the positive-margin envelope
  above.

\begin{definition}[N=4 finite-time constant dictionary]%
\label{def:n4-finite-time-constant-dictionary}
For a law \(\nu\) on \(\RR\) with mean \(\bar\gamma\), centred moments
\(\mu_n=\EE(\gamma-\bar\gamma)^n\), and finite
\[
M_7(\nu):=\EE|\gamma-\bar\gamma|^7,
\]
put
\[
\mathcal M_{\nu,8}:=\sum_{n=2}^{6}(n+1)|\mu_n|,
\qquad
\mathcal C_{\nu,8}:=110M_7(\nu),
\]
\[
\mathcal S_{\nu,8}:=7\sum_{n=2}^{6}|\mu_n|+110M_7(\nu).
\]
Define
\[
\mathcal P_{\nu,8}:=
\sum_{r=2}^{4}\frac{1}{r(r-1)}
\sum_{\ell=1}^{r}\binom r\ell
\mathcal M_{\nu,8}^{\,r-\ell}\mathcal C_{\nu,8}^{\,\ell}
\]
and
\[
\mathcal Q_{\nu,8}:=
\sum_{r=2}^{4}\frac{1}{r(r-1)}
\sum_{m=9}^{6r}
\sum_{\substack{2\le n_1,\ldots,n_r\le6\\
n_1+\cdots+n_r=m}}
\prod_{j=1}^{r}\bigl((n_j+1)|\mu_{n_j}|\bigr),
\]
where an empty inner sum is zero.  The fixed-law order-nine remainder and
threshold constants are
\begin{equation}\label{eq:dictionary-fixed-law-n4-constants}
B_{\nu,8}:=\mathcal P_{\nu,8}+\mathcal Q_{\nu,8}
+2\mathcal S_{\nu,8}^{5},
\qquad
T_{\nu,8}:=\max\{1,(2\mathcal S_{\nu,8})^{1/2}\}.
\end{equation}

For the controlled shell \(\mu_2=\sigma^2\),
\(\EE|\gamma-\bar\gamma|^7\le M\), \(M\ge\sigma^7\), write
\(\widehat M=M/\sigma^7\).  For \(u\ge1\), set
\[
\mathcal M^{(8)}(u)
:=3+\sum_{n=3}^{6}(n+1)u^{n/7},
\qquad
\mathcal R^{(8)}(u):=110u,
\]
\[
\mathcal S^{(8)}(u)
:=7\left(1+\sum_{n=3}^{6}u^{n/7}\right)+110u .
\]
Define
\[
\mathcal P^{(8)}(u):=
\sum_{r=2}^{4}\frac{1}{r(r-1)}
\sum_{\ell=1}^{r}\binom r\ell
\bigl(\mathcal M^{(8)}(u)\bigr)^{r-\ell}
\bigl(\mathcal R^{(8)}(u)\bigr)^\ell
\]
and
\[
\mathcal Q^{(8)}(u):=
\sum_{r=2}^{4}\frac{1}{r(r-1)}
\sum_{m=9}^{6r}
\sum_{\substack{2\le n_1,\ldots,n_r\le6\\
n_1+\cdots+n_r=m}}
\prod_{j=1}^{r}(n_j+1)\,\Lambda_{n_j}(u),
\]
with
\[
\Lambda_2(u):=1,\qquad
\Lambda_n(u):=u^{n/7}\quad(3\le n\le6),
\]
and empty inner sums again interpreted as zero.  The normalized controlled
shell functions are
\begin{equation}\label{eq:dictionary-normalized-shell-functions}
\begin{aligned}
C(u)
&:=\frac{1}{256}\left(
3+12u^{4/7}+21u^{6/7}+50u^{8/7}
\right),\\
B(u)
&:=\mathcal P^{(8)}(u)+\mathcal Q^{(8)}(u)
+2\bigl(\mathcal S^{(8)}(u)\bigr)^5,\\
T(u)
&:=\max\{1,\bigl(2\mathcal S^{(8)}(u)\bigr)^{1/2}\}.
\end{aligned}
\end{equation}
Their dimensional scale restores are
\begin{equation}\label{eq:dictionary-controlled-shell-c8}
C_{\sigma,M}^{(8)}:=\sigma^8C(\widehat M),\qquad
B_{\sigma,M}^{(8)}:=\sigma^9B(\widehat M),\qquad
T_{\sigma,M}^{(8)}:=\sigma T(\widehat M).
\end{equation}
All constants in this dictionary are obtained from the optimized
positive-margin quantities by applying
Lemma~\ref{lem:closed-cayley-mode-constant-envelopes}: for \(L=6\),
\(B_6\le7\), \(C_6\le55\), \(C_{\nu,6,1}\le110M_7(\nu)\), and
\(\|u_n\|_\infty\le n+1\).
\end{definition}

\paragraph{Moment-ball constants.}%
\phantomsection\label{cor:uniform-moment-ball-entropy-remainder}
Fix $N\ge2$, $0<\eta\le1$, and $M<\infty$.  There are finite constants
\[
C_{N,\eta,M}^{\rm dens},\qquad C_{N,\eta,M}^{\rm Tay},
\qquad T_{N,\eta,M}^{\rm ent}
\]
given by the closed finite formulas of
Lemma~\ref{lem:moment-ball-uniform-constant-domination}, depending only on
$N$, $\eta$, and $M$, such that every probability measure satisfying
\[
\int_{\RR}|\gamma-\bar\gamma|^{2N+\eta}\dd\nu(\gamma)\le M
\]
obeys, for every \(t\ge T_{N,\eta,M}^{\rm ent}\),
\begin{equation}\label{eq:uniform-moment-ball-entropy-remainder}
\left|
D_{\mathrm{KL}}(P_t*\nu\|P_t(\cdot-\bar\gamma))
-\sum_{m=2}^{N}A_{2m}(\nu)t^{-2m}
\right|
\le
C_{N,\eta,M}^{\rm dens}t^{-(2N+\eta)}
+C_{N,\eta,M}^{\rm Tay}t^{-(2N+2)} .
\end{equation}
These formulas are only the constant dictionary for this uniform estimate;
the coefficients remain the moment polynomials of the individual law.
