\section{Exact Lattice Reconstruction for Poisson Observation Data}%
\label{sec:separate-lattice-reconstruction-note}

The entropy theorems in Section~\ref{sec:entropy-asymptotics} are proved from
centred moments.  This note records exact external data from which the finite
centred moment jet \((\mu_2,\ldots,\mu_6)\) may be supplied before the entropy
theorems are invoked.

Fix \(t_0>0\), put
\[
X_c:=\gamma-\bar\gamma,\qquad
\kappa_{t_0}:=\operatorname{Law}(X_c/t_0),
\]
and define the signed finite measure
\[
\lambda_{t_0}(\dd s):=s\,\kappa_{t_0}(\dd s).
\]
The continuous observation profile used below is
\begin{equation}\label{eq:note-observation-profile-definition}
q_{t_0}(\xi):=
\pi\int_{\RR}s e^{-\ii s\xi}\,\kappa_{t_0}(\dd s)
=\pi\widehat{\lambda_{t_0}}(\xi).
\end{equation}
Thus full knowledge of \(q_{t_0}\) is exactly full knowledge of the Fourier
transform of \(\pi\lambda_{t_0}\).  When the corresponding moments are finite,
the full profile gives them directly by differentiation at the origin:
\begin{equation}\label{eq:note-full-profile-moment-derivatives}
  \int_{\RR}s^j\,\kappa_{t_0}(\dd s)
  =
  \frac{\ii^{\,j-1}}{\pi}\,q_{t_0}^{(j-1)}(0),
  \qquad j\ge1.
\end{equation}

\begin{theorem}[Exact lattice recovery from integer observations]%
\label{thm:note-poisson-observation-lattice-sampling}
\label{prop:note-poisson-law-lattice-periodicity}
Let \(\nu\) have finite centred first absolute moment and fix \(t_0>0\).
Assume the centred scaled law is supported on the integer lattice,
\begin{equation}\label{eq:note-poisson-law-side-lattice-support}
\nu\{\gamma:\gamma-\bar\gamma\in t_0\ZZ\}=1 .
\end{equation}
Let \(p_m:=\kappa_{t_0}(\{m\})\) and define the integer observation sequence
\begin{equation}\label{eq:note-integer-observation-samples}
a_n:=
\int_{\ZZ} s\,K(n,s)\,\kappa_{t_0}(\dd s),
\qquad
K(a,b):=\frac{2}{4+(a-b)^2},
\qquad n\in\ZZ .
\end{equation}
Then the sequence \((a_n)_{n\in\ZZ}\) determines the signed lattice masses
\[
b_m:=m p_m\qquad(m\in\ZZ),
\]
hence determines \(\kappa_{t_0}\) by
\[
p_m=\frac{b_m}{m}\quad(m\ne0),
\qquad
p_0=1-\sum_{m\ne0}p_m .
\]
More explicitly, with
\[
A(\theta):=\sum_{n\in\ZZ}a_ne^{-\ii n\theta},
\qquad
B(\theta):=\sum_{m\in\ZZ}b_me^{-\ii m\theta},
\]
one has
\begin{equation}\label{eq:note-lattice-symbol}
A(\theta)=\sigma(\theta)B(\theta),\qquad
\sigma(\theta):=\sum_{k\in\ZZ}\frac{2}{4+k^2}e^{-\ii k\theta}
=\pi\frac{\cosh(2(\pi-|\theta|))}{\sinh(2\pi)}
\end{equation}
for \(-\pi\le\theta\le\pi\).  Since \(\sigma(\theta)>0\), the recovery is
\[
B(\theta)=\frac{A(\theta)}{\sigma(\theta)},\qquad
b_m=\frac{1}{2\pi}\int_{-\pi}^{\pi}
\frac{A(\theta)}{\sigma(\theta)}e^{\ii m\theta}\dd\theta .
\]
Conversely, for this convolutional reconstruction formula to represent the
full continuous profile \eqref{eq:note-observation-profile-definition}, the
scaled law \(\kappa_{t_0}\) must be supported on \(\ZZ\), equivalently
\eqref{eq:note-poisson-law-side-lattice-support}.
\end{theorem}

\begin{proof}
Under \eqref{eq:note-poisson-law-side-lattice-support}, finite first absolute
moment gives \((b_m)\in\ell^1(\ZZ)\).  Since
\(\sum_{k\in\ZZ}2/(4+k^2)<\infty\), the samples are the absolutely convergent
discrete convolution
\[
a_n=\sum_{m\in\ZZ}b_m\,\frac{2}{4+(n-m)^2}.
\]
Taking Fourier series on \(\TT\) gives \(A=\sigma B\).  The computation of
\(\sigma\) is Poisson summation applied to \(x\mapsto2/(4+x^2)=\pi P_2(x)\):
\[
\sigma(\theta)
=\pi\sum_{\ell\in\ZZ}e^{-2|\theta+2\pi\ell|}
=\pi\frac{\cosh(2(\pi-|\theta|))}{\sinh(2\pi)},
\qquad -\pi\le\theta\le\pi.
\]
The displayed formula is strictly positive, so \(B=A/\sigma\) and its Fourier
coefficients recover \(b_m\).  Division by \(m\) recovers all off-zero masses,
and total mass recovers \(p_0\).

For the converse just stated, fix the reconstruction formula above: it first
recovers a lattice signed sequence \(b=(b_m)\) and then represents the profile
by the Fourier series of the lattice signed measure \(\sum_m b_m\delta_m\).
Any profile represented by this integer-orbit convolutional formula is
\(2\pi\)-periodic.  Thus, if that formula represents the full continuous
profile \(q_{t_0}\) in \eqref{eq:note-observation-profile-definition}, then
\(q_{t_0}\) is \(2\pi\)-periodic.  By
\eqref{eq:note-observation-profile-definition}, this periodicity says that the
finite signed measure
\[
(e^{-\ii2\pi s}-1)s\,\kappa_{t_0}(\dd s)
\]
has zero Fourier transform.  Fourier uniqueness for finite signed measures
therefore gives
\((e^{-\ii2\pi s}-1)s\,\kappa_{t_0}(\dd s)=0\), so the off-zero part of
\(\kappa_{t_0}\) is supported on \(\ZZ\); the remaining point \(0\) is already in
\(\ZZ\).
\end{proof}

\begin{theorem}[Exact moment recovery from profile or lattice data]%
\label{thm:note-observation-entropy-interface}
Let \(\nu\) be a probability measure on \(\RR\) with mean \(\bar\gamma\),
positive variance \(\sigma^2\), and finite centred sixth absolute moment.
Fix \(t_0>0\) and define \(\kappa_{t_0}\) and \(q_{t_0}\) by
\eqref{eq:note-observation-profile-definition}.  The following exact data
determine the centred moment jet
\((\mu_2,\ldots,\mu_6)\), where
\(\mu_j=\int_{\RR}(\gamma-\bar\gamma)^j\dd\nu(\gamma)\):
\begin{enumerate}[label=\textup{(\roman*)},leftmargin=*]
\item the full continuous profile \(q_{t_0}(\xi)\) on \(\RR\);
\item the integer samples \((a_n)_{n\in\ZZ}\) from
      \eqref{eq:note-integer-observation-samples}, but only together with the
      exact lattice condition
      \[
      \nu\{\gamma:\gamma-\bar\gamma\in t_0\ZZ\}=1 .
      \]
\end{enumerate}
For full profile data, the direct formula is
\begin{equation}\label{eq:note-observation-interface-full-profile-moments}
\mu_j
=
t_0^j\,\frac{\ii^{\,j-1}}{\pi}\,
q_{t_0}^{(j-1)}(0),
\qquad 2\le j\le6.
\end{equation}
For exact lattice data, first recover \(b_m=mp_m\) by
Theorem~\ref{thm:note-poisson-observation-lattice-sampling}; then
\begin{equation}\label{eq:note-observation-interface-lattice-moments}
\mu_j
=
t_0^j\sum_{m\in\ZZ}m^j p_m
=
t_0^j\sum_{m\ne0}m^{j-1}b_m,
\qquad 2\le j\le6.
\end{equation}
Equivalently, in either case the moments are recovered from the scaled law by
\begin{equation}\label{eq:note-observation-interface-moments}
\mu_j=t_0^j\int_{\RR}s^j\,\kappa_{t_0}(\dd s),
\qquad 2\le j\le6.
\end{equation}
\end{theorem}

\begin{proof}
Full profile data determine \(\lambda_{t_0}\) by Fourier uniqueness for
finite signed measures, since \(q_{t_0}=\pi\widehat{\lambda_{t_0}}\).  On any
Borel set bounded away from zero,
\[
\kappa_{t_0}(B)=\int_B s^{-1}\lambda_{t_0}(\dd s).
\]
A monotone exhaustion of \(\RR\setminus\{0\}\) recovers the off-zero
restriction of \(\kappa_{t_0}\), and
\[
\kappa_{t_0}(\{0\})=1-\kappa_{t_0}(\RR\setminus\{0\})
\]
recovers the atom at the origin.  Thus the full profile determines
\(\kappa_{t_0}\).  Since \(\nu\) has a finite centred sixth absolute moment,
the signed measure \(s\,\kappa_{t_0}(\dd s)\) has finite moments through order
five.  Differentiating \eqref{eq:note-observation-profile-definition} at the
origin gives
\[
q_{t_0}^{(j-1)}(0)
=\pi(-\ii)^{j-1}\int_{\RR}s^j\,\kappa_{t_0}(\dd s),
\qquad 2\le j\le6,
\]
which is \eqref{eq:note-observation-interface-full-profile-moments} after
rescaling by \(t_0^j\).

If the integer samples are accompanied by the lattice-support condition,
Theorem~\ref{thm:note-poisson-observation-lattice-sampling} recovers the same
scaled law directly.  With \(p_m=\kappa_{t_0}(\{m\})\) and \(b_m=mp_m\),
absolute convergence of the sums in
\eqref{eq:note-observation-interface-lattice-moments} follows from the sixth
absolute moment, and
\[
\int_{\RR}s^j\,\kappa_{t_0}(\dd s)
=\sum_{m\in\ZZ}m^j p_m
=\sum_{m\ne0}m^{j-1}b_m,
\qquad 2\le j\le6 .
\]

The finite sixth absolute moment makes the moments in
\eqref{eq:note-observation-interface-moments} finite, and the scaling identity
\(X_c=t_0s\) gives precisely the centred moments of \(\nu\) through order
six.
\end{proof}
