\subsection{Primitive orbit counts}

Let \(M\) be a primitive non-negative matrix with Perron root
\(\lambda=\rho(M)\). The associated dynamical \(\zeta\)-function is
\[
  \zeta_M(z)
  =\exp\!\left(\sum_{n\ge 1}\frac{\Tr(M^n)}{n}z^n\right)
  =\det(I-zM)^{-1}.
\]
The number of primitive orbits of least period \(n\) is
\begin{equation}\label{eq:primitive-mobius}
  p_n(M)=\frac{1}{n}\sum_{d\mid n}\mu(d)\Tr(M^{n/d}).
\end{equation}

\begin{lemma}\label{lem:primitive-orbit-asymptotic}
If
\[
  \Lambda(M):=\max\{|\mu|:\mu\in\spec(M),\ \mu\neq\lambda\},
\]
then
\[
  p_n(M)=\frac{\lambda^n}{n}
  +O\!\left(\frac{\max\{\Lambda(M)^n,\lambda^{n/2}\}}{n}\right).
\]
\end{lemma}

\begin{proof}
By the standard Perron--Frobenius trace estimate for shifts of finite
type~\cite{LindMarcus2021,Kitchens1998},
\(
  \Tr(M^n)=\lambda^n+O(\Lambda(M)^n)
\).
Substituting this into \eqref{eq:primitive-mobius}, the divisor \(d=1\)
gives the main term and every divisor \(d\ge 2\) contributes at most
\[
  O\!\left(\frac{\lambda^{n/2}}{n}\right).
\]
\end{proof}

\subsection{Self-dual kernels and completion}
\fi

Let \(B(u)=B_0+uB_1\) with integer matrices \(B_0,B_1\), and suppose there is a permutation matrix \(\Pi\) such that
\[
  \Pi^{-1}B_0\Pi=B_1,
  \qquad
  \Pi^{-1}B_1\Pi=B_0.
\]
Write
\[\Delta(z,u):=\det(I-zB(u)).\]
The following observation packages the formal consequences of this self-duality.

\begin{lemma}[Self-dual compression]\label{lem:self-dual-compression}
With \(B(u)\) as above,
\[
  \Pi^{-1}B(u)\Pi=u\,B(1/u),
  \qquad
  \Delta(z,u)=\Delta(uz,u^{-1}).
\]
After the change of variables \(u=r^2\), \(w=zr\), and
\(s=r+r^{-1}\), the expression
\(\widehat\Delta(w,s):=\Delta(w/r,r^2)\) belongs to \(\ZZ[s][w]\).
Writing
\[
  \widehat\Delta(w,s)=\sum_{j=0}^d c_j(s)w^j,
\]
each \(c_j(s)\) satisfies \(\deg_s c_j\le j\) and
\(c_j(-s)=(-1)^j c_j(s)\). In particular,
\(\widehat\Delta(w,-s)=\widehat\Delta(-w,s)\).
\end{lemma}

\begin{proof}
The matrix identity follows from the hypotheses:
\[
  \Pi^{-1}B(u)\Pi
  =\Pi^{-1}(B_0+uB_1)\Pi
  =B_1+uB_0
  =u\,B(1/u).
\]
Taking determinants of both sides gives the functional equation for
\(\Delta(z,u)\).

Write
\[
  \Delta(z,u)=\sum_{j=0}^d a_j(u)z^j
  \qquad\text{with }a_j(u)\in\ZZ[u].
\]
Comparing coefficients in
\(\Delta(z,u)=\Delta(uz,u^{-1})\) gives
\[
  a_j(u)=u^j a_j(u^{-1})
  \qquad (0\le j\le d).
\]
Therefore \(r^{-j}a_j(r^2)\) is a Laurent polynomial in \(r\) that is
invariant under \(r\mapsto r^{-1}\), whose exponents lie in
\(\{-j,-j+2,\dots,j-2,j\}\). Hence there is a unique polynomial
\(c_j(s)\in\ZZ[s]\) with \(\deg_s c_j\le j\) such that
\[
  r^{-j}a_j(r^2)=c_j(r+r^{-1})=c_j(s).
\]
Substituting \(u=r^2\) and \(z=w/r\) yields
\[
  \widehat\Delta(w,s)=\Delta(w/r,r^2)=\sum_{j=0}^d c_j(s)w^j\in\ZZ[s][w].
\]
Replacing \(r\) by \(-r\) sends \((w,s)\) to \((-w,-s)\), so
\[
  \widehat\Delta(w,-s)=\widehat\Delta(-w,s).
\]
Comparing coefficients of \(w^j\) gives
\(c_j(-s)=(-1)^j c_j(s)\).
\end{proof}
For reference we keep the terminology of Section~\ref{sec:introduction}:
the polynomial \(\widehat\Delta(w,s)\) is the completed determinant. For
\(m\ge2\), \(\Phi_m^+\) denotes the real cyclotomic polynomial of
\(2\cos(\pi/m)\), i.e.
\[
  \Phi_m^+(s)
  =
  \prod_{\alpha\in A_m}(s-\alpha),
  \qquad
  A_m=\Bigl\{2\cos\!\tfrac{\pi a}{m}:
      a\in(\ZZ/2m\ZZ)^\times/\{\pm1\}\Bigr\}.
\]
Whenever a cyclotomic factor is written as a resultant in this paper,
the convention is the ordered one
\[
  \Res_s\bigl(\Phi_m^+(s),H(w,s)\bigr)
\]
with \(\Phi_m^+\) in the first slot. The reversed order is always
understood to carry the Sylvester sign described in
\Cref{lem:ordered-cyclotomic-resultant}.

\begin{lemma}[Row-sum Perron bound]
  \label{lem:row-sum-perron-bound}
  Let \(M\) be a finite non-negative matrix. If every row of \(M\) has
  sum at most \(q\), and \(M\mathbf1=q\mathbf1\), then \(\rho(M)=q\).
  If, in addition, \(M\) is primitive, then \(q\) is a simple eigenvalue
  and no other eigenvalue has modulus \(q\).
\end{lemma}

\begin{proof}
The equality \(M\mathbf1=q\mathbf1\) shows that \(q\) is an eigenvalue.
For any eigenvalue \(\lambda\) of \(M\),
\[
  \abs{\lambda}\le \|M\|_\infty
  =\max_i\sum_j \abs{M_{ij}}
  =\max_i\sum_j M_{ij}\le q,
\]
so \(\rho(M)=q\). The final assertion is the primitive
Perron--Frobenius theorem.
\end{proof}

\ifshortcicm\else
\begin{lemma}[Fixed-field transfer sequences are exponentially bounded]
  \label{lem:fixed-field-transfer-exponential-bound}
  Let \(K\) be a fixed coefficient field, let
  \(\iota:K\hookrightarrow\CC\) be a fixed complex embedding, let
  \(A\in M_d(K)\), and let \(u,v\in K^d\).  For
  \[
    a_m:=\iota\!\left(u^{\mathsf T}A^m v\right)\qquad(m\ge0),
  \]
  there are constants \(C,R>0\), depending on \(A,u,v\) and on the fixed
  embedding \(\iota\) but not on \(m\), such that
  \[
    \abs{a_m}\le C R^m\qquad(m\ge0).
  \]
  In particular, over a fixed coefficient field with a fixed complex
  embedding, every fixed transfer sequence \(u^{\mathsf T}A^m v\) is
  exponentially bounded and therefore cannot realise super-exponential
  growth, such as factorial growth, along an infinite unbounded range.
\end{lemma}

\begin{proof}
Apply \(\iota\) entrywise and set
\(\widetilde A=\iota(A)\), \(\widetilde u=\iota(u)\), and
\(\widetilde v=\iota(v)\).  For any submultiplicative matrix norm and
its compatible vector norm,
\[
  \abs{a_m}
  =\abs{\widetilde u^{\mathsf T}\widetilde A^m\widetilde v}
  \le \|\widetilde u\|\,\|\widetilde A\|^m\,\|\widetilde v\|.
\]
Taking \(R=\max\{1,\|\widetilde A\|\}\) and
\(C=\|\widetilde u\|\,\|\widetilde v\|\) gives the claimed bound, after
enlarging \(C\) if necessary in the trivial case \(C=0\).  Since
\(m!/R^m\to\infty\) for every fixed \(R\), no sequence satisfying such a
bound can agree with factorial growth along an infinite unbounded range.
\end{proof}

\begin{lemma}[Ordered root-product resultant convention]
  \label{lem:ordered-cyclotomic-resultant}
  Let \(K\) be a field of characteristic \(0\), let \(A\) be a finite
  Galois-stable set of algebraic numbers, and put
  \[
    \Phi_A(s):=\prod_{\alpha\in A}(s-\alpha)\in K[s].
  \]
  For every polynomial \(H(w,s)\in K[w,s]\), define the root product
  \[
    \mathcal R_A[H](w):=\prod_{\alpha\in A}H(w,\alpha).
  \]
  With the standard Sylvester convention \cite{CoxLittleOShea2005}
  \[
    \Res_s(f,g)=a_f^{\deg g}\prod_{f(\alpha)=0}g(\alpha),
    \qquad f(s)=a_f\prod_{f(\alpha)=0}(s-\alpha),
  \]
  one has the exact product identity
  \[
    \mathcal R_A[H](w)
    =\Res_s\bigl(\Phi_A(s),H(w,s)\bigr).
  \]
  Thus the root product is defined independently of resultants and is
  represented by a resultant only in this ordered form, with the monic
  polynomial \(\Phi_A\) as the first argument. This is the convention
  used throughout the paper whenever a cyclotomic root product is
  denoted by a resultant.
  If \(e=\deg_s H\), where the degree is taken in \(K(w)[s]\), then
  reversing the two resultant arguments gives
  \[
    \Res_s\bigl(H(w,s),\Phi_A(s)\bigr)
      =(-1)^{e|A|}\mathcal R_A[H](w).
  \]
\end{lemma}

\begin{proof}
Since \(\Phi_A\) is monic, the displayed Sylvester convention gives
\[
  \Res_s\bigl(\Phi_A(s),H(w,s)\bigr)
    =\prod_{\alpha\in A}H(w,\alpha).
\]
The reversed-order formula is the standard symmetry
\[
  \Res_s(g,f)=(-1)^{\deg f\deg g}\Res_s(f,g),
\]
applied in \(K(w)[s]\) to \(f=\Phi_A\) and \(g=H\). In particular,
no statement in the sequel identifies
\(\Res_s(H(w,s),\Phi_A(s))\) with the root product unless the displayed
sign is also present.
\end{proof}

\begin{lemma}[Parity normal form for the sign quotient]
  \label{lem:sign-quotient-normal-form}
\ifshortcicm
  Let \(k\) have characteristic different from \(2\), and let
  \(H(w,s)=\sum c_j(s)w^j\in k[w,s]\) satisfy
  \(\deg_s c_j\le j\) and \(c_j(-s)=(-1)^jc_j(s)\). Then there is a
  unique polynomial \(G(x,y)\in k[x,y]\) with
  \(H(w,s)=G(w^2,sw)\). If \(H\) is irreducible and \(c_0\ne0\), this
  gives the function field of the quotient by
  \((w,s)\mapsto(-w,-s)\).
\end{lemma}

\begin{proof}
Parity rewrites each monomial \(s^bw^j\) as a monomial in
\(x=w^2\) and \(y=sw\).  These monomials are independent, giving
existence and uniqueness.  The invariant rational field is
\(k(w^2,sw)\); irreducibility and \(c_0\ne0\) ensure that the induced
order-two action on the curve is nontrivial, so this invariant field is
the quotient function field.
\else
  Let \(k\) be a field of characteristic different from \(2\), and let
  \[
    H(w,s)=\sum_{j=0}^d c_j(s)w^j\in k[w,s]
  \]
  satisfy \(\deg_s c_j\le j\) and \(c_j(-s)=(-1)^j c_j(s)\) for every
  \(j\). Then there is a unique polynomial \(G(x,y)\in k[x,y]\) such
  that
  \[
    H(w,s)=G(w^2,sw).
  \]
  If \(H\) is irreducible and \(c_0\ne0\), then the quotient of the
  affine curve \(H(w,s)=0\) by the involution
  \((w,s)\mapsto(-w,-s)\) is birational to the affine plane curve
  \(G(x,y)=0\).
\end{lemma}

\begin{proof}
For each \(j\), the parity condition implies that \(c_j(s)\) is a
linear combination of monomials \(s^b\) with \(0\le b\le j\) and
\(b\equiv j\pmod 2\). Hence every monomial \(s^bw^j\) occurring in
\(H\) can be rewritten as
\[
  s^bw^j=(w^2)^{(j-b)/2}(sw)^b=x^{(j-b)/2}y^b,
\]
where \(x=w^2\) and \(y=sw\). This gives a polynomial
\(G(x,y)\in k[x,y]\) with \(H(w,s)=G(w^2,sw)\). Uniqueness follows
because the monomials \(x^ay^b\) are linearly independent in
\(k[w,s]\).

If \(H\) is irreducible and \(c_0\ne0\), then \(G\) is irreducible as
well: otherwise a proper factorisation of \(G\) would pull back to a
proper factorisation of \(H\). Thus the image of \(k[x,y]\) in
\(k[w,s]/(H)\) has kernel \((G)\). Let
\[
  K:=\operatorname{Frac}\bigl(k[w,s]/(H)\bigr)
\]
be its function field. The involution \((w,s)\mapsto(-w,-s)\) acts on
\(K\). In the ambient rational field,
\[
  k(w,s)^{(w,s)\mapsto(-w,-s)}=k(w^2,sw):
\]
the even rational invariants are generated by \(w^2\), \(ws\), and
\(s^2=(ws)^2/w^2\), so no further generator is needed. Passing to the
quotient is legitimate as follows. Let \(L\) be the subfield of \(K\)
generated by the images of \(x=w^2\) and \(y=sw\). The field \(K\) is
generated over \(L\) by the image of \(w\), with \(w^2=x\) and
\(s=y/w\). The induced involution is nontrivial, since otherwise the image
of \(w\) would be equal to its negative; as \(c_0\ne0\), the polynomial
\(H\) is not divisible by \(w\), so the image of \(w\) is nonzero in the
domain \(k[w,s]/(H)\).
Because \(\operatorname{char} k\ne2\), this order-two action gives
\([K:K^\iota]=2\). Since \(K=L(w)\) and \(w^2=x\), also \([K:L]\le2\);
the nontriviality just noted forces \(L=K^\iota\). Therefore
\[
  K^\iota=\operatorname{Frac}\bigl(k[x,y]/(G)\bigr),
\]
and the quotient is birational to the plane curve \(G(x,y)=0\).
\fi
\end{proof}

\begin{theorem}[Sign-quotient plane model and genus]
  \label{thm:sign-quotient-plane-genus}
  Let \(k\) be a field of characteristic different from \(2\), and let
  \[
    H(w,s)=\sum_{j=0}^d c_j(s)w^j\in k[w,s]
  \]
  be irreducible with \(c_0\ne0\). Assume
  \[
    \deg_s c_j\le j,\qquad c_j(-s)=(-1)^jc_j(s)
    \quad (0\le j\le d).
  \]
  Let \(G(x,y)\in k[x,y]\) be the unique polynomial satisfying
  \[
    H(w,s)=G(w^2,sw),
  \]
  and let \(Y_{\mathrm{aff}}\) be the affine plane curve \(G(x,y)=0\).
  Then the quotient of the affine curve \(H(w,s)=0\) by the involution
  \((w,s)\mapsto(-w,-s)\) has function field \(k(Y_{\mathrm{aff}})\).
  If the projective closure \(Y\subset\mathbb P^2_k\) of
  \(Y_{\mathrm{aff}}\) is nonsingular of degree \(n\), then \(Y\) is the
  smooth projective quotient model and, after extending scalars to an
  algebraic closure of \(k\),
  \[
    g(Y)=\frac{(n-1)(n-2)}2.
  \]
\end{theorem}

\begin{proof}
\ifshortcicm
\Cref{lem:sign-quotient-normal-form} identifies the invariant function
field.  A nonsingular projective plane curve is already the smooth
projective model, and its genus is
\((n-1)(n-2)/2\) \cite[Ch.~IV, Example~1.3.4]{Hartshorne1977}.
\else
\Cref{lem:sign-quotient-normal-form} gives the polynomial \(G\) and
identifies the invariant function field with \(k(Y_{\mathrm{aff}})\).
If the projective closure \(Y\) is nonsingular, it is already complete
and regular, hence it is the smooth projective model of that function
field. The genus formula is the standard formula for a nonsingular
plane curve of degree \(n\)
\cite[Ch.~IV, Example~1.3.4]{Hartshorne1977}.
\fi
\end{proof}

\begin{proposition}[Self-duality]\label{prop:sync-self-duality}
Let \(\iota\) be the involution of \(\mathcal S\) defined by
\[
  \st{000}\leftrightarrow\st{101},\qquad
  \st{001}\leftrightarrow\st{100},\qquad
  \st{002}\leftrightarrow\st{010},
\]
\[
  \st{0-12}\leftrightarrow\st{11-1},\qquad
  \st{1-12}\leftrightarrow\st{01-1},
\]
and let \(\Pi\) be the associated permutation matrix. Then
\[
  \Pi^{-1}B_0\Pi=B_1,
  \qquad
  \Pi^{-1}B_1\Pi=B_0.
\]
Consequently,
\[
  \Pi^{-1}B(u)\Pi=u\,B(1/u),
  \qquad
  \Delta(z,u):=\det(I-zB(u))=\Delta(uz,u^{-1}).
\]
\end{proposition}

\begin{proof}
Inspection of the transition list shows that, whenever
\(x\to y\) is an edge with output \(e\), the transition
\(\iota(x)\to\iota(y)\) is an edge with output \(1-e\). This gives the
matrix identities for \(B_0\) and \(B_1\). The identity for \(B(u)\)
follows immediately:
\[
  \Pi^{-1}B(u)\Pi
  =\Pi^{-1}(B_0+uB_1)\Pi
  =B_1+uB_0
  =u\,B(1/u).
\]
Taking determinants of \(I-zB(u)\) and \(I-zu\,B(1/u)\) yields the
functional equation for \(\Delta(z,u)\).
\end{proof}

\begin{definition}\label{def:perron-residue-constant}
We write \(B:=B(1)\) for the unweighted kernel and
\[
  C_B:=\lim_{z\to 1/\rho(B)}(1-\rho(B)z)\det(I-zB)^{-1}
\]
for its Perron residue constant.
\end{definition}
