\subsection{Optimal injective relabeling of the realized Sturmian law}

\begin{theorem}[Optimal injective relabeling of the realized Sturmian law]
\label{thm:optimal-injective-relabeling}
Assume \(\beta=\alpha\). For \(m\ge1\), let
\[
\mathcal I_m(\alpha):=
\{\iota:S_m(\alpha)\to X_m:\ \iota\text{ is injective}\},
\]
and for \(\iota\in\mathcal I_m(\alpha)\) write
\[
\nu_\iota:=\iota_*\mu_m.
\]
If \(m\ge6\), then for every \(q\in[0,\infty]\),
\[
\inf_{\iota\in\mathcal I_m(\alpha)}D_q(\nu_\iota\|\pi_m)
=
(m-1)\log\varphi-\log\pi(0)-H_q(\mu_m).
\]
Every injection with image contained in \(X_m^{00}\) attains this value. For
every \(q<\infty\), these are exactly the optimal injections. For
\(q=\infty\), write
\[
p_*:=\max_{w\in S_m(\alpha)}\mu_m(w),\qquad
\tau(u):=\frac{\pi_m(u)}{\pi(0)\varphi^{-(m-1)}}\in\{1,\varphi^{-1},\varphi^{-2}\}.
\]
An injection \(\iota\) is optimal if and only if
\[
\mu_m(w)\le p_*\tau(\iota(w))
\qquad\text{for every }w\in S_m(\alpha).
\]
Thus \(q=\infty\) may have additional optimizers, but precisely the additional
placements of non-maximal source atoms whose loss of Parry mass is no larger
than their drop below the maximal source mass. Since \(|X_m^{00}|=F_m\), an
all-\(X_m^{00}\) optimal placement exists if and only if \(m\ge6\).
\end{theorem}

\begin{proof}
Fix \(\iota\in\mathcal I_m(\alpha)\), and put \(\nu:=\nu_\iota\). Since
\(\iota\) is injective on the \(m+1\)-point Sturmian support \(S_m(\alpha)\),
it only relabels the atoms of \(\mu_m\). Therefore
\[
H_q(\nu)=H_q(\mu_m)
\qquad (q\in[0,\infty]).
\]

Set
\[
c_{00}:=\pi(0)\varphi^{-(m-1)}.
\]
By Proposition~\ref{prop:parry-endpoint-collapse},
\[
\pi_m(u)=c_{00}\tau(u),
\qquad
\tau(u)\in\{1,\varphi^{-1},\varphi^{-2}\},
\]
and \(\tau(u)=1\) exactly on \(X_m^{00}\).

First let \(q\in(0,\infty)\setminus\{1\}\). Then
\[
\begin{aligned}
D_q(\nu\|\pi_m)
&=
\frac{1}{q-1}
\log\!\left(\sum_{u\in X_m}\nu(u)^q(c_{00}\tau(u))^{1-q}\right)\\
&=
-\log c_{00}-H_q(\nu)
+
\frac{1}{q-1}
\log\!\left(
\frac{\sum_{u\in X_m}\nu(u)^q\tau(u)^{1-q}}
{\sum_{u\in X_m}\nu(u)^q}
\right).
\end{aligned}
\]
Because \(0<\tau(u)\le1\) for every \(u\), the last term is nonnegative. It
vanishes if and only if \(\tau(u)=1\) on \(\supp(\nu)\), equivalently
\(\supp(\nu)\subset X_m^{00}\). Hence
\[
D_q(\nu\|\pi_m)
\ge
(m-1)\log\varphi-\log\pi(0)-H_q(\mu_m),
\]
with equality precisely when \(\iota(S_m(\alpha))\subset X_m^{00}\).

For \(q=1\), Theorem~\ref{thm:parry-kl-identity} gives
\[
\begin{aligned}
\KL(\nu\|\pi_m)
&=
m\log\varphi-\log\pi(0)-H_1(\nu)
+
\bigl(\eta_{11}(\nu)-\eta_{00}(\nu)\bigr)\log\varphi\\
&=
(m-1)\log\varphi-\log\pi(0)-H_1(\mu_m)\\
&\quad
+
\bigl(\eta_{01}(\nu)+\eta_{10}(\nu)+2\eta_{11}(\nu)\bigr)\log\varphi.
\end{aligned}
\]
The correction term is nonnegative and vanishes exactly when
\(\eta_{01}(\nu)=\eta_{10}(\nu)=\eta_{11}(\nu)=0\), that is, exactly when
\(\supp(\nu)\subset X_m^{00}\).

For \(q=0\), \(\nu\) has support size \(m+1\), so
\[
D_0(\nu\|\pi_m)
=
-\log\pi_m(\supp\nu)
\ge
-\log\bigl((m+1)c_{00}\bigr)
=
(m-1)\log\varphi-\log\pi(0)-H_0(\mu_m).
\]
Equality holds if and only if every point of \(\supp(\nu)\) has Parry mass
\(c_{00}\), namely when \(\supp(\nu)\subset X_m^{00}\).

For \(q=\infty\),
\[
\begin{aligned}
D_\infty(\nu\|\pi_m)
&=
\log\max_{u\in\supp(\nu)}\frac{\nu(u)}{\pi_m(u)}\\
&\ge
\log\max_{u\in\supp(\nu)}\frac{\nu(u)}{c_{00}}
=
-\log c_{00}-H_\infty(\nu)\\
&=
(m-1)\log\varphi-\log\pi(0)-H_\infty(\mu_m).
\end{aligned}
\]
The displayed lower bound is sharp exactly when
\[
\max_{u\in\supp(\nu)}\frac{\nu(u)}{c_{00}\tau(u)}
=
\frac{\max_{u\in\supp(\nu)}\nu(u)}{c_{00}}.
\]
Since \(\nu=\iota_*\mu_m\) and \(\iota\) is injective, this condition is
equivalent to
\[
\frac{\mu_m(w)}{\tau(\iota(w))}\le p_*
\qquad (w\in S_m(\alpha)),
\]
or, equivalently, to
\(\mu_m(w)\le p_*\tau(\iota(w))\) for every realized word \(w\).  This proves
the asserted \(q=\infty\) optimizer characterization.  In particular, any
injection with image in \(X_m^{00}\), for which \(\tau\equiv1\) on the image,
attains equality here as well; other optimizers occur exactly when every atom
placed outside \(X_m^{00}\) is small enough to satisfy the displayed
inequality.

It remains to characterize the admissible range. The Sturmian complexity is
\(|S_m(\alpha)|=m+1\), while Lemma~\ref{lem:endpoint-class-counts} gives
\(|X_m^{00}|=F_m\). Since \(F_5=5<6\) and \(F_6=8\ge7\), and
\[
F_{m+1}-(m+2)=\bigl(F_m-(m+1)\bigr)+F_{m-1}-1,
\]
an induction on \(m\) shows that \(F_m\ge m+1\) exactly for \(m\ge6\). Thus
an injective placement inside \(X_m^{00}\) exists exactly in that range, and
the displayed infimum formula follows.
\end{proof}

\begin{proposition}[Finite small-\(m\) endpoint budgets]
\label{prop:small-m-endpoint-budget}
Assume \(\beta=\alpha\), fix \(1\le m\le5\), and write the masses of the
Sturmian \(m\)-block law in nonincreasing order as
\[
p_1\ge p_2\ge\cdots\ge p_{m+1}>0.
\]
Let
\[
b_0(m):=|X_m^{00}|,\qquad
b_1(m):=|X_m^{01}|+|X_m^{10}|,\qquad
b_2(m):=|X_m^{11}|,
\]
with the convention \(X_1^{00}=\{0\}\), \(X_1^{11}=\{1\}\), and
\(X_1^{01}=X_1^{10}=\varnothing\).  The endpoint capacities and the
support-optimal endpoint budgets for an injective placement of the \(m+1\)
Sturmian atoms are
\[
\begin{array}{c|c|c|c}
m & (b_0,b_1,b_2) & \text{best support budget }(n_0,n_1,n_2)
& \text{endpoint classes used}\\
\hline
1 & (1,0,1) & (1,0,1) & X_1^{00},X_1^{11}\\
2 & (1,2,0) & (1,2,0) & X_2^{00},X_2^{01},X_2^{10}\\
3 & (2,2,1) & (2,2,0) & X_3^{00},X_3^{01},X_3^{10}\\
4 & (3,4,1) & (3,2,0) & X_4^{00}\text{ and two middle-class words}\\
5 & (5,6,2) & (5,1,0) & X_5^{00}\text{ and one middle-class word}.
\end{array}
\]
Here \(n_0+n_1+n_2=m+1\), and the classes are filled in the order
\[
X_m^{00},\qquad X_m^{01}\cup X_m^{10},\qquad X_m^{11}.
\]
Write
\[
\tau(u):=\varphi^{-\mathbf 1_{\{u_1=1\}}-\mathbf 1_{\{u_m=1\}}},
\]
so \(\tau=1\) on \(X_m^{00}\), \(\tau=\varphi^{-1}\) on
\(X_m^{01}\cup X_m^{10}\), and \(\tau=\varphi^{-2}\) on \(X_m^{11}\).
For the \(q=\infty\) formula below set
\[
R_{m,\infty}:=
\max\!\left(
\{1\}
\cup
\left\{\varphi\,\frac{p_{n_0+1}}{p_1}: n_1>0\right\}
\cup
\left\{\varphi^2\,\frac{p_{n_0+n_1+1}}{p_1}: n_2>0\right\}
\right).
\]

For \(q\in(0,\infty)\setminus\{1\}\), the minimum over injective placements
\(T_m:S_m(\alpha)\to X_m\) is
\[
\inf_{T_m}D_q((T_m)_*\mu_m\|\pi_m)
=
(m-1)\log\varphi-\log\pi(0)-H_q(\mu_m)+\gamma_{m,q},
\]
where
\[
\gamma_{m,q}
=
\frac{1}{q-1}\log
\frac{
\sum_{r=1}^{n_0}p_r^q
\;+\;\varphi^{q-1}\sum_{r=n_0+1}^{n_0+n_1}p_r^q
\;+\;\varphi^{2(q-1)}\sum_{r=n_0+n_1+1}^{m+1}p_r^q}
{\sum_{r=1}^{m+1}p_r^q}.
\]
For \(q=1\),
\[
\inf_{T_m}\KL((T_m)_*\mu_m\|\pi_m)
=
(m-1)\log\varphi-\log\pi(0)-H_1(\mu_m)+\gamma_{m,1},
\]
where
\[
\gamma_{m,1}
=
\left(\sum_{r=n_0+1}^{n_0+n_1}p_r
+2\sum_{r=n_0+n_1+1}^{m+1}p_r\right)\log\varphi.
\]
For \(q=\infty\),
\[
\inf_{T_m}D_\infty((T_m)_*\mu_m\|\pi_m)
=
(m-1)\log\varphi-\log\pi(0)-H_\infty(\mu_m)+\gamma_{m,\infty},
\]
where
\[
\gamma_{m,\infty}
=
\log R_{m,\infty}.
\]
For \(q=0\), the minimum is
\[
\inf_{T_m}D_0((T_m)_*\mu_m\|\pi_m)
=
-\log\!\left(
\pi(0)\varphi^{-(m-1)}
\bigl(n_0+n_1\varphi^{-1}+n_2\varphi^{-2}\bigr)
\right).
\]
For \(q\in(0,\infty)\), the minimizers are exactly the placements with the
endpoint budgets displayed above, with the largest source atoms assigned to the
heaviest available endpoint classes.  For \(q=\infty\), those placements attain
the displayed value, and an injective placement is minimizing if and only if
\[
\max_{a\in S_m(\alpha)}
\frac{\mu_m(a)}{p_1\,\tau(T_m(a))}
\le R_{m,\infty}.
\]
For \(q=0\), only the support budget matters.
\end{proposition}

\begin{proof}
The capacity table follows from direct enumeration for \(m=1\) and from
Lemma~\ref{lem:endpoint-class-counts} for \(2\le m\le5\).  Since Parry masses
on \(X_m^{00}\), \(X_m^{01}\cup X_m^{10}\), and \(X_m^{11}\) are respectively
\[
\pi(0)\varphi^{-(m-1)},\qquad
\pi(0)\varphi^{-m},\qquad
\pi(0)\varphi^{-(m+1)},
\]
the support of largest Parry mass is obtained by filling the endpoint classes
in the displayed order.  Applying this rule with \(k=m+1\) gives the five
rows of the table.

Let \(q\in(0,\infty)\setminus\{1\}\).  The endpoint-penalty identity proved in
Theorem~\ref{thm:lossless-coarse-graining-envelope} shows that minimizing the
divergence is the same as optimizing
\[
\sum_{a\in S_m(\alpha)}\mu_m(a)^q\tau(T_m(a))^{1-q},
\]
with the sign of \(1/(q-1)\) taken into account.  If \(q>1\), the prefactor is
positive and the coefficients
\[
1,\quad \varphi^{q-1},\quad \varphi^{2(q-1)}
\]
increase from the \(00\)-class to the \(11\)-class.  The rearrangement
inequality therefore assigns the largest weights \(p_r^q\) to the smallest
available coefficients.  If \(0<q<1\), the prefactor is negative and the
coefficients
\[
1,\quad \varphi^{q-1},\quad \varphi^{2(q-1)}
\]
decrease from the \(00\)-class to the \(11\)-class; maximizing the weighted
sum, which minimizes the logarithm after multiplication by the negative
prefactor, gives the same assignment.  This proves the displayed formula for
\(\gamma_{m,q}\).

For \(q=1\), Theorem~\ref{thm:lossless-coarse-graining-envelope} reduces the
endpoint penalty to the expectation of the class costs \(0,\log\varphi\), and
\(2\log\varphi\).  The same rearrangement argument assigns the largest masses
to the lowest costs, giving \(\gamma_{m,1}\).  For \(q=\infty\), the problem is
to minimize
\[
\max_{a\in S_m(\alpha)}\frac{\mu_m(a)}{\tau(T_m(a))}.
\]
The displayed class filling gives bottleneck \(p_1R_{m,\infty}\): its largest
ratios in the three endpoint levels are \(p_1\), \(\varphi p_{n_0+1}\), and
\(\varphi^2p_{n_0+n_1+1}\), with empty levels omitted.  Conversely, if a
placement has bottleneck at most \(p_1R_{m,\infty}\), then it is optimal.  No
smaller bottleneck is possible.  Indeed, a bound \(B<p_1R_{m,\infty}\) must
fail in one of three ways: \(B<p_1\), so the largest atom already exceeds the
bound even in \(X_m^{00}\); or \(n_1>0\) and \(B<\varphi p_{n_0+1}\), so the
first \(n_0+1\) atoms would all have to lie in \(X_m^{00}\), whose capacity is
only \(n_0\); or \(n_2>0\) and
\(B<\varphi^2p_{n_0+n_1+1}\), so the first \(n_0+n_1+1\) atoms would all have
to lie outside \(X_m^{11}\), while \(X_m^{00}\cup X_m^{01}\cup X_m^{10}\) has
capacity only \(n_0+n_1\) in the support-optimal budget.  This proves the
\(q=\infty\) value and the stated minimizer criterion.  For \(q=0\), the source weights
drop out and
\[
D_0((T_m)_*\mu_m\|\pi_m)=-\log\pi_m(T_m(S_m(\alpha))),
\]
so the best support budget is exactly the support budget already listed in
the table.  The equality cases stated in the proposition are precisely the
equality cases in these rearrangement steps.
\end{proof}

\begin{remark}[Injective relabeling and support budgets]
\label{rem:injective-relabeling-stronger}
Theorem~\ref{thm:optimal-injective-relabeling} does more than constrain the
image support size. It fixes the realized Sturmian weight vector \(\mu_m\) and
allows only injective placements of those \(m+1\) probabilities inside the
ambient language \(X_m\). The resulting optimum is therefore a coding
invariance statement for the realized source, not just a best \((m+1)\)-state
approximation problem over arbitrary laws on \(X_m\).  The finite range
\(m<6\) is the endpoint-budget obstruction classified in
Proposition~\ref{prop:small-m-endpoint-budget}.
\end{remark}
