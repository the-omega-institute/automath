\subsection{Support budgets and bounded-type asymptotics}

\begin{proposition}[Bounded-type Sturmian R\'enyi entropy]\label{prop:bounded-type-sturmian-entropy}
Assume \(\beta=\alpha\), and suppose that the partial quotients of \(\alpha\)
are bounded by \(A\). Then, for every \(m\ge1\) and every \(q\in[0,\infty]\),
\[
\log m-\log\!\bigl(2(A+1)\bigr)
\le
H_q(\mu_m)
\le
\log(m+1)
\le
\log m+\log 2.
\]
In particular,
\[
H_q(\mu_m)=\log m+O_\alpha(1)
\qquad (q\in[0,\infty]),
\]
uniformly in the R\'enyi order.
\end{proposition}

\begin{proof}
Choose \(n\) so that \(q_n\le m<q_{n+1}\). Since the partial quotients are at
most \(A\),
\[
q_{n+1}=a_{n+1}q_n+q_{n-1}\le (A+1)q_n.
\]
Proposition~\ref{thm:spg-sturmian-cylinder-measure-continued-fraction-sandwich}
therefore yields, for every \(u\in S_m(\alpha)\),
\[
\mu_m(u)<\frac{2}{q_n}\le \frac{2(A+1)}{m}.
\]
Hence
\[
H_\infty(\mu_m)
=
-\log\max_{u\in S_m(\alpha)}\mu_m(u)
\ge
\log m-\log\!\bigl(2(A+1)\bigr).
\]

The same Sturmian cylinder partition has exactly \(m+1\) nonempty atoms, so
\[
H_0(\mu_m)=\log|S_m(\alpha)|=\log(m+1).
\]
R\'enyi entropy is monotone decreasing in its order on a finite alphabet, so
\[
H_\infty(\mu_m)\le H_q(\mu_m)\le H_0(\mu_m)
\qquad (q\in[0,\infty]).
\]
Combining the preceding bounds proves the claim.
\end{proof}

\begin{corollary}[Bounded-type injective encodings]
\label{cor:injective-relabeling-bounded-type}
Assume \(\beta=\alpha\), suppose that the partial quotients of \(\alpha\) are
bounded by \(A\), and let
\[
L_q(m):=\inf_{\iota\in\mathcal I_m(\alpha)}D_q(\iota_*\mu_m\|\pi_m).
\]
Then, for every \(m\ge6\) and every \(q\in[0,\infty]\),
\[
(m-1)\log\varphi-\log(m+1)-\log\pi(0)
\le
L_q(m)
\]
\[
\le
(m-1)\log\varphi-\log m-\log\pi(0)
+
\log\!\bigl(2(A+1)\bigr).
\]
Hence
\[
L_q(m)=m\log\varphi-\log m+O_\alpha(1)
\qquad (q\in[0,\infty]),
\]
uniformly in the R\'enyi order. Moreover,
\[
0
\le
D_q(\pi_m^{\mathrm{fold}}\|\pi_m)-L_q(m)
\le
2\log\varphi.
\]
\end{corollary}

\begin{proof}
The exact injective-placement formula from
Theorem~\ref{thm:optimal-injective-relabeling} gives
\[
L_q(m)
=
(m-1)\log\varphi-\log\pi(0)-H_q(\mu_m).
\]
Applying Proposition~\ref{prop:bounded-type-sturmian-entropy} to
\(H_q(\mu_m)\) yields the two-sided estimate for \(L_q(m)\).

Since \(\pi_m^{\mathrm{fold}}\) itself is an injective relabeling of
\(\mu_m\) by Corollary~\ref{cor:sturmian-exact-relabeling},
\[
L_q(m)\le D_q(\pi_m^{\mathrm{fold}}\|\pi_m).
\]
Subtracting the exact formula for \(L_q(m)\) from the upper bound in
Proposition~\ref{prop:parry-renyi-window}, again with
\(\nu=\pi_m^{\mathrm{fold}}\), gives
\[
D_q(\pi_m^{\mathrm{fold}}\|\pi_m)-L_q(m)\le 2\log\varphi.
\]
\end{proof}

\begin{lemma}[Golden-slope coding audit]
\label{lem:golden-slope-coding-audit}
Let \(\alpha=\varphi^{-1}\), \(I=[0,\alpha)\), and code the rotation by
\(s_j(x)=\mathbf 1_I(x+j\alpha)\).  This is the Fibonacci-word symbol
convention in which the substitution is
\[
\sigma(1)=10,\qquad \sigma(0)=1.
\]
For the finite endpoint calculations used below, the following convention
checks hold.
\begin{enumerate}
\item The length-three factors are exactly
\[
010,\qquad 011,\qquad 101,\qquad 110,
\]
and the allowed right extensions are
\[
010\mapsto 0101,\qquad
110\mapsto 1101,\qquad
011\mapsto 0110,\qquad
101\mapsto 1010\text{ or }1011 .
\]
\item The length-six factors, in the same coding convention, are exactly
\[
010110,\quad 011010,\quad 011011,\quad 101011,\quad 101101,\quad
110101,\quad 110110 .
\]
\item With the half-open window \(I=[0,\alpha)\),
\[
[10]=[0,1-\alpha),\qquad
[010]=[\alpha,2-2\alpha),\qquad
[110]=[1-\alpha,\alpha).
\]
Hence
\[
\mu([10])=\varphi^{-2},\qquad
\mu([010])=\varphi^{-4},\qquad
\mu([110])=\varphi^{-3}.
\]
\end{enumerate}
\end{lemma}

\begin{proof}
The standard identification of the half-open coding of the rotation of slope
\(\varphi^{-1}\) with the Fibonacci substitution language gives the displayed
substitution convention \cite{QueffelecSubstitution,Fogg2002}.  The finite
facts needed in the endpoint argument can also be checked directly from the
rotation partition, which fixes the half-open endpoint choices.

For words of length \(n\), the discontinuities of the map
\[
x\longmapsto
\bigl(\mathbf 1_I(x),\mathbf 1_I(x+\alpha),\ldots,
\mathbf 1_I(x+(n-1)\alpha)\bigr)
\]
are the points \(-j\alpha\pmod1\) with \(-1\le j\le n-1\).  On each
half-open atom between consecutive discontinuities the word is constant, and
the left endpoint belongs to the atom.

For \(n=4\), the cyclic order of the discontinuities is
\[
0<2-3\alpha<1-\alpha<\alpha<2-2\alpha<1.
\]
Sampling the corresponding atoms gives
\[
\begin{array}{c|c}
\text{atom} & s_0s_1s_2s_3\\
\hline
{[0,2-3\alpha)} & 1010\\
{[2-3\alpha,1-\alpha)} & 1011\\
{[1-\alpha,\alpha)} & 1101\\
{[\alpha,2-2\alpha)} & 0101\\
{[2-2\alpha,1)} & 0110 .
\end{array}
\]
Taking the first three symbols and the fourth symbol in this table proves the
terminal triples and their allowed right extensions.

For \(n=6\), the cyclic order is
\[
0<2-3\alpha<1-\alpha<3-4\alpha<\alpha<2-2\alpha<4-5\alpha<1.
\]
The half-open atoms give
\[
\begin{array}{c|c}
\text{atom} & s_0s_1s_2s_3s_4s_5\\
\hline
{[0,2-3\alpha)} & 101011\\
{[2-3\alpha,1-\alpha)} & 101101\\
{[1-\alpha,3-4\alpha)} & 110101\\
{[3-4\alpha,\alpha)} & 110110\\
{[\alpha,2-2\alpha)} & 010110\\
{[2-2\alpha,4-5\alpha)} & 011010\\
{[4-5\alpha,1)} & 011011 .
\end{array}
\]
This is the asserted length-six factor list.

It remains to spell out the half-open endpoint cylinders.  The word \(10\)
means \(x\in I\) and \(x+\alpha\notin I\), hence
\[
[10]=[0,\alpha)\cap R_{-\alpha}(\TT\setminus I)=[0,1-\alpha).
\]
Similarly,
\[
[010]
=
(\TT\setminus I)\cap R_{-\alpha}I\cap R_{-2\alpha}(\TT\setminus I)
=[\alpha,2-2\alpha),
\]
and
\[
[110]
=
I\cap R_{-\alpha}I\cap R_{-2\alpha}(\TT\setminus I)
=[1-\alpha,\alpha).
\]
The endpoints are exactly as written because \(I\) contains \(0\) and excludes
\(\alpha\).  Taking lengths and using
\(\alpha=\varphi^{-1}\), \(1-\alpha=\varphi^{-2}\),
\(2-3\alpha=\varphi^{-4}\), and \(2\alpha-1=\varphi^{-3}\) proves the three
mass identities.
\end{proof}

\begin{lemma}[Golden-slope terminal residue skeleton]
\label{lem:golden-terminal-residue-skeleton}
Assume \(\beta=\alpha=\varphi^{-1}\), and let \(m\ge6\).  For
\(a\in S_m(\varphi^{-1})\),
\[
N_m(a)\bmod F_{m+2}\in [F_{m+1},F_{m+2})
\]
if and only if the last three symbols of \(a\) are \(010\) or \(110\).
\end{lemma}

\begin{proof}
Lemma~\ref{lem:golden-slope-coding-audit} fixes the golden-slope symbol
convention and gives the possible terminal triples together with their
one-step right extensions.  In terms of the new terminal triple after appending
the next symbol, these transitions are
\[
010\mapsto 101,\qquad
110\mapsto 101,\qquad
011\mapsto 110,\qquad
101\mapsto 010\text{ or }011 .
\]

The audited length-six factors are
\[
010110,\quad 011010,\quad 011011,\quad 101011,\quad 101101,\quad 110101,
\quad 110110 .
\]
A direct valuation gives the base case
\[
\begin{array}{c|c|c}
b\in S_6(\varphi^{-1}) & b_4b_5b_6 & N_6(b)\\
\hline
010110&110&15\\
011010&010&13\\
011011&011&26\\
101011&011&25\\
101101&101&22\\
110101&101&21\\
110110&110&16 .
\end{array}
\]
Since \(F_7=13\), \(F_8=21\), and \(F_9=34\), this is exactly the pair of
interval statements
\[
\begin{array}{ll}
N_6(b)\in[F_7,F_8), & b_4b_5b_6\in\{010,110\},\\
N_6(b)\in[F_8,F_9), & b_4b_5b_6\in\{011,101\}.
\end{array}
\]

We now propagate these two interval assertions.  Suppose they hold at length
\(m\).  If a length-\((m+1)\) factor is obtained from \(a\in S_m(\varphi^{-1})\)
by appending \(c\in\{0,1\}\), then
\[
N_{m+1}(ac)=N_m(a)+cF_{m+2}.
\]
Using the transition graph above:
\[
\begin{array}{c|c|c}
\text{old terminal triple} & c & \text{new terminal triple}\\
\hline
010&1&101\\
110&1&101\\
011&0&110\\
101&0&010\\
101&1&011 .
\end{array}
\]
The corresponding interval inclusions are
\[
[F_{m+1},F_{m+2})+F_{m+2}\subset [F_{m+3},F_{m+4}),
\]
\[
[F_{m+2},F_{m+3})\subset [F_{m+2},F_{m+3}),
\]
and
\[
[F_{m+2},F_{m+3})+F_{m+2}\subset [F_{m+3},F_{m+4}),
\]
where the last inclusion uses \(2F_{m+2}\ge F_{m+3}\).  Hence, at length
\(m+1\),
\[
\begin{array}{ll}
N_{m+1}(b)\in[F_{m+2},F_{m+3}), & b_{m-1}b_mb_{m+1}\in\{010,110\},\\
N_{m+1}(b)\in[F_{m+3},F_{m+4}), & b_{m-1}b_mb_{m+1}\in\{011,101\}.
\end{array}
\]
This closes the induction.

For terminal triples \(010\) and \(110\), the actual value already lies in
\([F_{m+1},F_{m+2})\), so its residue modulo \(F_{m+2}\) lies in that same
interval.  For terminal triples \(011\) and \(101\), the actual value lies in
\([F_{m+2},F_{m+3})\); subtracting one \(F_{m+2}\) puts the residue in
\([0,F_{m+1})\).  This proves the criterion.
\end{proof}

\begin{lemma}[Golden-slope endpoint balance of the canonical fold]
\label{lem:golden-fold-endpoint-masses}
Assume \(\beta=\alpha=\varphi^{-1}\) and \(m\ge6\). Then
\[
\pi_m^{\mathrm{fold}}\{u:u_1=1\}
=
\pi_m^{\mathrm{fold}}\{u:u_m=1\}
=
\varphi^{-2}.
\]
Consequently,
\[
\eta_{11}(\pi_m^{\mathrm{fold}})
-
\eta_{00}(\pi_m^{\mathrm{fold}})
=
-\varphi^{-3}.
\]
\end{lemma}

\begin{proof}
For the golden slope, the Sturmian language is the set of length-\(m\)
factors of the Fibonacci word, in the symbol convention determined by the
window \([0,\alpha)\); this convention and the half-open endpoint choices are
audited in Lemma~\ref{lem:golden-slope-coding-audit}.  We shall use two
elementary endpoint consequences of this presentation.  First, every realized
word begins with one of
\[
01,\qquad 10,\qquad 110.
\]
Second, Lemma~\ref{lem:golden-terminal-residue-skeleton} gives the right-end
residue criterion.  A legal word \(x\in X_m\) has last digit \(1\) exactly when
its Fibonacci value lies in \([F_{m+1},F_{m+2})\).  By
Proposition~\ref{prop:fold-residue-characterization}, the folded value has the
same residue as \(N_m(a)\) modulo \(F_{m+2}\).  Therefore
\[
(\Fold_m(a))_m=1
\quad\Longleftrightarrow\quad
a_{m-2}a_{m-1}a_m\in\{010,110\}
\qquad (m\ge6).
\]

Let \(u=\Fold_m(a)\) with \(a\in S_m(\varphi^{-1})\).  The first endpoint is
controlled by the same finite-transition induction used for the terminal
residue skeleton.  In the audited length-six list of
Lemma~\ref{lem:golden-slope-coding-audit}, a direct Zeckendorf normalization
gives first folded digit \(1\) exactly for the two words beginning with
\(10\), and first folded digit \(0\) for the words beginning with \(01\) or
\(110\).  When a legal golden factor \(a\) is extended on the right by an
  allowed symbol \(c\), the valuation changes by
  \[
  N_{m+1}(ac)=N_m(a)+cF_{m+2}.
  \]
  The first folded digit is unchanged by this extension.  We verify this at the
  level of the raw Fibonacci digit vector before invoking uniqueness of the
  terminal normal form.  The audited transition graph has exactly the five
  right-extension cases
  \[
  010+1\to101,\qquad
  110+1\to101,\qquad
  011+0\to110,\qquad
  101+0\to010,\qquad
  101+1\to011 .
  \]
  In the two \(c=0\) cases the integer is unchanged, so the lower Zeckendorf
  digits, and in particular the \(F_2\)-digit, are unchanged.  In the cases
  \(010+1\) and \(110+1\), Lemma~\ref{lem:golden-terminal-residue-skeleton}
  puts \(N_m(a)\) in \([F_{m+1},F_{m+2})\).  Its normal form therefore has the
  top digit \(z_m=1\) and no digit in position \(m+1\).  Adding \(F_{m+2}\)
  creates the adjacent pair at positions \(m,m+1\), and the local carry
  \[
  F_{m+1}+F_{m+2}\longmapsto F_{m+3}
  \]
  is supported in positions \(m,m+1,m+2\).  It cannot alter any position
  \(\le m-1\).

  The remaining \(c=1\) transition is the exceptional branch
  \(101+1\to011\).  Here the appended raw vector has adjacent \(1\)'s at
  positions \(m,m+1\).  Applying the same adjacent carry first replaces
  \(d_m=d_{m+1}=1\) by \(d_m=d_{m+1}=0\), \(d_{m+2}=1\).  This removes the only
  new contact between the appended digit and the old word.  Because the old
  suffix is \(101\), the digit in position \(m-1\) is \(0\).  After the carry,
  all altered digits are in positions \(m,m+1,m+2\), and positions \(m-1\) and
  \(m\) are both \(0\).  Every remaining forbidden adjacent pair, if any, lies
  wholly inside the unchanged prefix \(\{1,\ldots,m-2\}\): there is no adjacent
  pair crossing from this prefix into the zero pair at \(m-1,m\), and the single
  high digit at \(m+2\) is separated from it by the zero at \(m+1\).  Hence the
  rest of the Zeckendorf normalization factors as the normalization of the
  unchanged lower prefix together with the isolated high digit \(F_{m+3}\).
  In particular the \(F_2\)-digit is the same as at length \(m\).
  The normal form is order-independent by
  Theorem~\ref{thm:local-zeckendorf-normalization}, so these local
  normalizations compute the canonical fold.  Induction from the length-six
  audit gives
  \[
  (\Fold_m(a))_1=1
  \quad\Longleftrightarrow\quad
  a_1a_2=10.
\]
The cylinder \([10]\) for the rotation coding is
\[
\{x\in\TT:\ x\in[0,\alpha),\ x+\alpha\notin[0,\alpha)\}
=
[0,1-\alpha),
\]
and has measure \(1-\alpha=\varphi^{-2}\). Therefore
\[
\pi_m^{\mathrm{fold}}\{u:u_1=1\}=\varphi^{-2}.
\]

By the right-end criterion, the last folded endpoint has mass
\(\mu_m([010])+\mu_m([110])\).  Lemma~\ref{lem:golden-slope-coding-audit}
computes these two cylinders in the same half-open convention:
\[
[010]=(\TT\setminus I)\cap R_{-\alpha}I\cap R_{-2\alpha}(\TT\setminus I)
=[\alpha,2-2\alpha),
\]
and
\[
[110]=I\cap R_{-\alpha}I\cap R_{-2\alpha}(\TT\setminus I)
=[1-\alpha,\alpha),
\]
so
\[
\mu_m([010])=\varphi^{-4},
\qquad
\mu_m([110])=\varphi^{-3},
\]
and
\[
\pi_m^{\mathrm{fold}}\{u:u_m=1\}
=
\varphi^{-4}+\varphi^{-3}
=
\varphi^{-2}.
\]

Finally,
\[
\eta_{11}(\pi_m^{\mathrm{fold}})
-
\eta_{00}(\pi_m^{\mathrm{fold}})
=
\pi_m^{\mathrm{fold}}\{u:u_1=1\}
+
\pi_m^{\mathrm{fold}}\{u:u_m=1\}
-
1
=
2\varphi^{-2}-1
=
-\varphi^{-3},
\]
using \(\varphi^2=\varphi+1\).  This proves the asserted endpoint difference.
\end{proof}

\begin{proposition}[Canonical golden-slope gap above the injective floor]
\label{prop:canonical-golden-slope-gap}
Assume \(\beta=\alpha=\varphi^{-1}\).  For every \(m\ge6\),
\[
\KL(\pi_m^{\mathrm{fold}}\|\pi_m)-L_1(m)
=
\bigl(1-\varphi^{-3}\bigr)\log\varphi,
\]
where \(L_1(m)\) is the optimal injective value from
Corollary~\ref{cor:injective-relabeling-bounded-type}.
\end{proposition}

\begin{proof}
By Corollary~\ref{cor:injective-relabeling-bounded-type} and
Theorem~\ref{thm:optimal-injective-relabeling},
\[
L_1(m)
=
(m-1)\log\varphi-\log\pi(0)-H_1(\mu_m).
\]
On the Sturmian slice the fold is an exact relabeling of \(\mu_m\), so
Corollary~\ref{cor:no-fold-entropy-loss} gives
\[
\KL(\pi_m^{\mathrm{fold}}\|\pi_m)
=
m\log\varphi-\log\pi(0)-H_1(\mu_m)
+
\bigl(\eta_{11}(\pi_m^{\mathrm{fold}})
-\eta_{00}(\pi_m^{\mathrm{fold}})\bigr)\log\varphi .
\]
Subtracting the two formulas and using
Lemma~\ref{lem:golden-fold-endpoint-masses} yields
\[
\KL(\pi_m^{\mathrm{fold}}\|\pi_m)-L_1(m)
=
\bigl(1-\varphi^{-3}\bigr)\log\varphi.
\]
\end{proof}

Theorem~\ref{thm:optimal-injective-relabeling} gives the dominant comparison
result of this section: once \(m\ge6\), the best injective placement is the
one supported on \(X_m^{00}\).  The next theorem is a secondary refinement. It
keeps the same injective setting but resolves the exact additive penalty
incurred by the endpoint classes, including the finite exceptional range
\(m<6\).

\begin{theorem}[Exact endpoint penalty for injective deterministic placements]
\label{thm:lossless-coarse-graining-envelope}
Assume \(\beta=\alpha\), let \(m\ge1\), and let
\[
T_m:S_m(\alpha)\longrightarrow X_m
\]
be injective. Put \(\nu_m:=(T_m)_*\mu_m\). Then, for every \(m\ge1\) and every
\(u\in X_m\), define
\[
\tau(u):=\varphi^{-\mathbf 1_{\{u_1=1\}}-\mathbf 1_{\{u_m=1\}}}
\in \{1,\varphi^{-1},\varphi^{-2}\}.
\]
Equivalently,
\[
\tau(u)=1 \text{ on } X_m^{00},\qquad
\tau(u)=\varphi^{-1} \text{ on } X_m^{01}\cup X_m^{10},\qquad
\tau(u)=\varphi^{-2} \text{ on } X_m^{11}.
\]

For every \(q\in(0,\infty)\setminus\{1\}\),
\[
D_q(\nu_m\|\pi_m)
=
(m-1)\log\varphi-\log\pi(0)-H_q(\mu_m)+\Gamma_q(T_m),
\]
where
\[
\Gamma_q(T_m):=
\frac{1}{q-1}
\log\!\left(
\frac{\sum_{a\in S_m(\alpha)}\mu_m(a)^q\tau(T_m(a))^{1-q}}
{\sum_{a\in S_m(\alpha)}\mu_m(a)^q}
\right).
\]
For \(q=1\),
\[
\KL(\nu_m\|\pi_m)
=
(m-1)\log\varphi-\log\pi(0)-H_1(\mu_m)+\Gamma_1(T_m),
\]
where
\[
\Gamma_1(T_m):=
\bigl(
\eta_{01}(\nu_m)+\eta_{10}(\nu_m)+2\eta_{11}(\nu_m)
\bigr)\log\varphi.
\]
For \(q=0\),
\[
D_0(\nu_m\|\pi_m)
=
(m-1)\log\varphi-\log\pi(0)-H_0(\mu_m)+\Gamma_0(T_m),
\]
where
\[
\Gamma_0(T_m):=
-\log\!\left(
\frac{1}{m+1}\sum_{a\in S_m(\alpha)}\tau(T_m(a))
\right).
\]
For \(q=\infty\),
\[
D_\infty(\nu_m\|\pi_m)
=
(m-1)\log\varphi-\log\pi(0)-H_\infty(\mu_m)+\Gamma_\infty(T_m),
\]
where
\[
\Gamma_\infty(T_m):=
\log\max_{a\in S_m(\alpha)}
\frac{\mu_m(a)}
{e^{-H_\infty(\mu_m)}\tau(T_m(a))}.
\]
For every \(q\in[0,\infty]\),
\[
0\le \Gamma_q(T_m)\le 2\log\varphi.
\]
For finite \(q\in[0,\infty)\),
\[
\Gamma_q(T_m)=0
\iff
T_m(S_m(\alpha))\subset X_m^{00},
\]
and
\[
\Gamma_q(T_m)=2\log\varphi
\iff
T_m(S_m(\alpha))\subset X_m^{11}.
\]
For \(q=\infty\), the containments \(T_m(S_m(\alpha))\subset X_m^{00}\) and
\(T_m(S_m(\alpha))\subset X_m^{11}\) are sufficient for the lower and upper
equalities, respectively.

If, in addition, the partial quotients of \(\alpha\) are bounded by \(A\),
then, for every \(q\in[0,\infty]\),
\[
m\log\varphi-\log(m+1)-\log\pi(0)-\log\varphi
\le
D_q(\nu_m\|\pi_m)
\]
\[
\le
m\log\varphi-\log m-\log\pi(0)
+
\log\varphi+\log\!\bigl(2(A+1)\bigr).
\]
Equivalently,
\[
D_q(\nu_m\|\pi_m)=m\log\varphi-\log m+O_\alpha(1)
\qquad (q\in[0,\infty]),
\]
uniformly in the R\'enyi order.
\end{theorem}

\begin{proof}
Because \(T_m\) is injective on \(S_m(\alpha)\), it is a relabeling of the
realized Sturmian law. Hence
\[
H_q(\nu_m)=H_q(\mu_m)
\qquad (q\in[0,\infty]).
\]

Put
\[
c_{00}:=\pi(0)\varphi^{-(m-1)}.
\]
By Proposition~\ref{prop:parry-endpoint-collapse},
\[
\pi_m(u)=c_{00}\tau(u)
\qquad (u\in X_m).
\]

First let \(q\in(0,\infty)\setminus\{1\}\). Since
\(\nu_m(T_m(a))=\mu_m(a)\) for every \(a\in S_m(\alpha)\),
\[
\begin{aligned}
D_q(\nu_m\|\pi_m)
&=
\frac{1}{q-1}
\log\!\left(
\sum_{a\in S_m(\alpha)}\mu_m(a)^q(c_{00}\tau(T_m(a)))^{1-q}
\right)\\
&=
-\log c_{00}-H_q(\mu_m)+\Gamma_q(T_m),
\end{aligned}
\]
which is the claimed identity.  Put
\[
w_a=\frac{\mu_m(a)^q}{\sum_{b\in S_m(\alpha)}\mu_m(b)^q}.
\]
The logarithm in \(\Gamma_q(T_m)\) is applied to the weighted mean
\[
\sum_{a\in S_m(\alpha)}w_a\,\tau(T_m(a))^{1-q}.
\]
If \(q>1\), then \(1-q<0\), and this mean lies in
\([1,\varphi^{2(q-1)}]\); multiplying its logarithm by the positive number
\(1/(q-1)\) gives a value in \([0,2\log\varphi]\).  If \(0<q<1\), then
\(1-q>0\), and the same mean lies in \([\varphi^{-2(1-q)},1]\);
multiplying by the negative number \(1/(q-1)\) reverses the interval and
again gives \(0\le \Gamma_q(T_m)\le2\log\varphi\).
The lower endpoint occurs if and only if every \(\tau(T_m(a))=1\), namely
\(T_m(S_m(\alpha))\subset X_m^{00}\). The upper endpoint occurs if and only if
every \(\tau(T_m(a))=\varphi^{-2}\), namely
\(T_m(S_m(\alpha))\subset X_m^{11}\).

For \(q=1\), Theorem~\ref{thm:parry-kl-identity} gives
\[
\begin{aligned}
\KL(\nu_m\|\pi_m)
&=
m\log\varphi-\log\pi(0)-H_1(\nu_m)
+
\bigl(\eta_{11}(\nu_m)-\eta_{00}(\nu_m)\bigr)\log\varphi\\
&=
(m-1)\log\varphi-\log\pi(0)-H_1(\mu_m)\\
&\quad
+
\bigl(\eta_{01}(\nu_m)+\eta_{10}(\nu_m)+2\eta_{11}(\nu_m)\bigr)\log\varphi,
\end{aligned}
\]
because
\[
\eta_{11}(\nu_m)-\eta_{00}(\nu_m)
=
-1+\eta_{01}(\nu_m)+\eta_{10}(\nu_m)+2\eta_{11}(\nu_m).
\]
This is the stated identity for \(\Gamma_1(T_m)\). Since
\[
\eta_{01}(\nu_m)+\eta_{10}(\nu_m)+2\eta_{11}(\nu_m)\in[0,2],
\]
one has \(0\le \Gamma_1(T_m)\le2\log\varphi\). The lower endpoint holds
exactly when \(\eta_{00}(\nu_m)=1\), equivalently
\(T_m(S_m(\alpha))\subset X_m^{00}\), and the upper endpoint holds exactly
when \(\eta_{11}(\nu_m)=1\), equivalently
\(T_m(S_m(\alpha))\subset X_m^{11}\).

For \(q=0\), injectivity of \(T_m\) and the Sturmian complexity
\(|S_m(\alpha)|=m+1\) give
\[
\supp(\nu_m)=T_m(S_m(\alpha)),
\qquad
H_0(\mu_m)=\log(m+1).
\]
Therefore
\[
\begin{aligned}
D_0(\nu_m\|\pi_m)
&=
-\log \pi_m(T_m(S_m(\alpha)))\\
&=
-\log\!\left(
c_{00}\sum_{a\in S_m(\alpha)}\tau(T_m(a))
\right)\\
&=
(m-1)\log\varphi-\log\pi(0)-H_0(\mu_m)+\Gamma_0(T_m).
\end{aligned}
\]
Since the arithmetic mean of numbers in \([\varphi^{-2},1]\) again lies in
\([\varphi^{-2},1]\), we obtain \(0\le\Gamma_0(T_m)\le2\log\varphi\), with the
same equality cases as above.

For \(q=\infty\),
\[
\begin{aligned}
D_\infty(\nu_m\|\pi_m)
&=
\log\max_{a\in S_m(\alpha)}
\frac{\mu_m(a)}{c_{00}\tau(T_m(a))}\\
&=
(m-1)\log\varphi-\log\pi(0)-H_\infty(\mu_m)\\
&\quad
+
\log\max_{a\in S_m(\alpha)}
\frac{\mu_m(a)}{e^{-H_\infty(\mu_m)}\tau(T_m(a))}.
\end{aligned}
\]
Because \(\mu_m(a)\le e^{-H_\infty(\mu_m)}\) and
\(\tau(T_m(a))\in[\varphi^{-2},1]\), the last logarithm lies in
\([0,2\log\varphi]\). If \(T_m(S_m(\alpha))\subset X_m^{00}\), then
\(\tau(T_m(a))=1\) for every \(a\), so the lower bound is attained. If
\(T_m(S_m(\alpha))\subset X_m^{11}\), then \(\tau(T_m(a))=\varphi^{-2}\) for
every \(a\), so the upper bound is attained.

Now assume the partial quotients of \(\alpha\) are bounded by \(A\). The lower
bound for \(q\in[0,\infty)\) follows from \(\Gamma_q(T_m)\ge0\) and
Proposition~\ref{prop:bounded-type-sturmian-entropy}, while the upper bound for
\(q\in[0,\infty)\) follows from \(\Gamma_q(T_m)\le2\log\varphi\) and the same
entropy estimate. The case \(q=\infty\) is covered by the preceding
two-sided bound together with Proposition~\ref{prop:bounded-type-sturmian-entropy}.
This proves the displayed bounded-type window. The asymptotic
\[
D_q(\nu_m\|\pi_m)=m\log\varphi-\log m+O_\alpha(1)
\]
is exactly the same estimate rewritten with constants uniform in
\(q\in[0,\infty]\).
\end{proof}
