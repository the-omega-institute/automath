\section{Window rigidity and conjugacy on the Sturmian slice}

\subsection{Sharp window classification}

The collision question has a complete answer for the interval windows of
Section~2.  At the two extremes the source or its coordinatewise complement
already lies in the golden-mean language.  Between them, the unique collision
of the two-letter fold is realized with positive mass.  Thus losslessness at
all resolutions is decided at resolution two.

For the general window, write
\[
S_m(\alpha,\beta):=\supp(\mu_m),
\qquad
\delta(\alpha):=\|\alpha\|=\min\{\alpha,1-\alpha\}.
\]

\begin{theorem}[Sharp classification of lossless rotation windows]
\label{thm:window-injectivity-classification}
Let \(\alpha\in(0,1)\) be irrational and \(\beta\in(0,1)\).  The following
conditions are equivalent.
\begin{enumerate}
\item For every \(m\ge1\), the restriction
\[
\Fold_m|_{S_m(\alpha,\beta)}:S_m(\alpha,\beta)\longrightarrow X_m
\]
is injective.
\item The restriction of \(\Fold_2\) to \(S_2(\alpha,\beta)\) is injective.
\item The window length satisfies
\[
\beta\in(0,\delta(\alpha)]\cup[1-\delta(\alpha),1).
\]
\end{enumerate}
More precisely, if \(\beta\le\delta(\alpha)\), then
\(S_m(\alpha,\beta)\subset X_m\) and \(\Fold_m\) is the identity on the
realized language.  If \(\beta\ge1-\delta(\alpha)\), then the coordinatewise
complement of every word in \(S_m(\alpha,\beta)\) belongs to \(X_m\), and
\(\Fold_m\) is injective on the realized language.  If
\(\delta(\alpha)<\beta<1-\delta(\alpha)\), then \(00,11\in
S_2(\alpha,\beta)\) and
\[
\Fold_2(00)=\Fold_2(11)=00.
\]
\end{theorem}

\begin{proof}
Put \(I=[0,\beta)\).  Two circle intervals of common length \(\ell\), whose
initial points differ by \(\alpha\), have an intersection of positive
Lebesgue measure exactly when \(\ell>\delta(\alpha)\).  Indeed, after choosing
the shorter orientation between their initial points, their displacement is
\(\delta(\alpha)\in(0,1/2)\); the two half-open arcs are disjoint up to
endpoints precisely when their common length is at most that displacement.
Applying this first to \(I\) and \(I-\alpha\), and then to their complementary
intervals, gives
\[
\mu_2(11)>0\quad\Longleftrightarrow\quad
\beta>\delta(\alpha),
\]
and
\[
\mu_2(00)>0\quad\Longleftrightarrow\quad
1-\beta>\delta(\alpha).
\]

Suppose first that \(\beta\le\delta(\alpha)\).  The word \(11\) has zero
mass.  Any positive-mass word containing adjacent \(1\)'s would have its
cylinder contained in a translate of the two-letter \(11\)-cylinder, which
is impossible.  Hence \(S_m(\alpha,\beta)\subset X_m\) for every \(m\), and
Proposition~\ref{prop:canonical-fold-basic} shows that \(\Fold_m\) fixes this
language pointwise.

Now suppose that \(\beta\ge1-\delta(\alpha)\).  The word \(00\) has zero
mass, so the coordinatewise complement
\[
\widetilde u:=(1-u_1)\cdots(1-u_m)
\]
belongs to \(X_m\) for every \(u\in S_m(\alpha,\beta)\).  If \(u\ne v\),
then \(\widetilde u\ne\widetilde v\), and the bijectivity of
\(N_m|_{X_m}\) from Proposition~\ref{prop:canonical-fold-basic} gives
\[
0<\left|N_m(\widetilde u)-N_m(\widetilde v)\right|<F_{m+2}.
\]
Since
\[
N_m(u)-N_m(v)
=-\bigl(N_m(\widetilde u)-N_m(\widetilde v)\bigr),
\]
the values \(N_m(u)\) and \(N_m(v)\) are distinct modulo \(F_{m+2}\).
Proposition~\ref{prop:fold-residue-characterization} therefore gives
\(\Fold_m(u)\ne\Fold_m(v)\).

It remains to consider
\(\delta(\alpha)<\beta<1-\delta(\alpha)\).  The two intersection criteria
above show that both \(00\) and \(11\) have positive mass.  Moreover,
\[
N_2(00)=0,
\qquad
N_2(11)=F_2+F_3=F_4,
\]
so Proposition~\ref{prop:fold-residue-characterization} gives
\(\Fold_2(00)=\Fold_2(11)=00\).  Thus condition~(2) fails.  We have proved
that (3) implies (1), that (1) implies (2), and that failure of (3) implies
failure of (2), which establishes the equivalence and all stated refinements.
\end{proof}

\begin{corollary}[No collisions on the Sturmian language]
\label{thm:sturmian-no-collision-rigidity}
Assume \(\beta=\alpha\). For every \(m\ge1\), the restriction
\[
J_m:=\Fold_m|_{S_m(\alpha)}:S_m(\alpha)\longrightarrow X_m
\]
is injective.
\end{corollary}

\begin{proof}
If \(\alpha<1/2\), then \(\delta(\alpha)=\alpha=\beta\).  If
\(\alpha>1/2\), then \(1-\delta(\alpha)=\alpha=\beta\).  The remaining case
\(\alpha=1/2\) is excluded by irrationality, so
Theorem~\ref{thm:window-injectivity-classification} applies.
\end{proof}

\begin{corollary}[Exact relabeling on the realized Sturmian language]\label{cor:sturmian-exact-relabeling}
Assume \(\beta=\alpha\), and write
\[
Y_m(\alpha):=\Fold_m(S_m(\alpha))\subset X_m,
\qquad
J_m:=\Fold_m|_{S_m(\alpha)}.
\]
Then \(J_m:S_m(\alpha)\to Y_m(\alpha)\) is a bijection. For every probability
law \(\nu\) supported on \(S_m(\alpha)\), the pushforward \(J_{m*}\nu\) is an
exact relabeling of \(\nu\). In particular,
\[
\pi_m^{\mathrm{fold}}(J_m(u))=\mu_m(u)
\qquad (u\in S_m(\alpha)),
\]
and for every \(q\in[0,\infty]\),
\[
H_q(\pi_m^{\mathrm{fold}})=H_q(\mu_m).
\]
\end{corollary}

\begin{proof}
Corollary~\ref{thm:sturmian-no-collision-rigidity} shows that \(J_m\) is
injective, and surjectivity onto \(Y_m(\alpha)\) is tautological. If
\(\nu\in\Delta(S_m(\alpha))\), then \(J_m\) merely renames the atoms of
\(\nu\), so the masses are preserved one-for-one:
\[
(J_{m*}\nu)(J_m(u))=\nu(u)
\qquad (u\in S_m(\alpha)).
\]
Applying this to \(\nu=\mu_m\) gives the formula for
\(\pi_m^{\mathrm{fold}}=(\Fold_m)_*\mu_m\), and every R\'enyi entropy is
invariant under bijective relabeling.
\end{proof}

\subsection{Higher-block conjugacy on the Sturmian source}

\begin{theorem}[Higher-block conjugacy on the Sturmian source]
\label{thm:sequence-level-transparency}
Assume \(\beta=\alpha\), and let \(X_\alpha\subset\{0,1\}^{\mathbb Z}\) be the
Sturmian subshift of slope \(\alpha\). For \(m\ge1\), let
\[
\Phi_m:X_\alpha\longrightarrow S_m(\alpha)^{\mathbb Z},
\qquad
(\Phi_m(s))_t:=s_t\cdots s_{t+m-1},
\]
be the usual \(m\)-block presentation, and let
\[
\Psi_m:X_\alpha\longrightarrow Y_m(\alpha)^{\mathbb Z},
\qquad
(\Psi_m(s))_t:=\Fold_m(s_t\cdots s_{t+m-1}),
\]
be the folded block process. Then
\[
\Psi_m=J_m^{\mathbb Z}\circ \Phi_m,
\]
where \(J_m^{\mathbb Z}\) is the coordinatewise relabeling induced by the
bijection \(J_m:S_m(\alpha)\to Y_m(\alpha)\). Consequently,
\(J_m^{\mathbb Z}\) is a topological conjugacy from the usual \(m\)-block
presentation \(\Phi_m(X_\alpha)\) onto the folded image \(\Psi_m(X_\alpha)\).
\end{theorem}

\begin{proof}
For every \(s\in X_\alpha\) and every \(t\in\mathbb Z\),
\[
(\Psi_m(s))_t
=
\Fold_m(s_t\cdots s_{t+m-1})
=
J_m\bigl((\Phi_m(s))_t\bigr),
\]
which is exactly the identity \(\Psi_m=J_m^{\mathbb Z}\circ\Phi_m\). By
Corollary~\ref{cor:sturmian-exact-relabeling}, \(J_m\) is a bijection on the
realized alphabet, so \(J_m^{\mathbb Z}\) and \((J_m^{-1})^{\mathbb Z}\) are
inverse one-block codes. They commute with the shift by construction, hence
define a topological conjugacy between the two subshifts.
\end{proof}
\subsection{Transfer to conjugate circle maps}

\begin{theorem}[Conjugacy invariance of the comparison]
\label{thm:circle-map-conjugacy-transfer}
Let \(f:\TT\to\TT\) be an orientation-preserving circle homeomorphism, and
assume that there exists a circle homeomorphism \(h\) such that
\[
h\circ f=R_\alpha\circ h,
\qquad
R_\alpha(y)=y+\alpha \pmod 1,
\]
with \(\alpha\) irrational. Fix \(\beta\in(0,1)\), set
\[
I:=h^{-1}([0,\beta)),
\]
and define the itinerary of \(x_0\in\TT\) by
\[
s_t^f:=\mathbf 1_I(f^t(x_0)),
\qquad t\ge0.
\]
If \(y_0:=h(x_0)\), then
\[
s_t^f=\mathbf 1_{[0,\beta)}(y_0+t\alpha)
\qquad (t\ge0).
\]
Consequently the empirical microstate laws, folded laws, and projective
residuals built from the \(f\)-orbit of \(x_0\) coincide exactly with the
corresponding quantities for the rotation orbit \(y_0+t\alpha\). If one uses
the \(f\)-invariant measure \(h^{-1}_*\leb\) as the reference law on the
state space of \(f\), then the limiting microstate and folded laws coincide
with \(\mu_m\) and \(\pi_m^{\mathrm{fold}}\). The transfer is
hypothesis-preserving in the following precise sense.
\begin{enumerate}
\item For arbitrary \(\beta\), every general-window statement proved above for
the half-open rotation window \([0,\beta)\), including the constants \(2m\),
\(4m+2\), the separation scale \(\lambda_m(\alpha,\beta)\), the sharp
lossless-window classification, and the boundary residual criterion, applies
to the pulled-back window \(I=h^{-1}([0,\beta))\) with the same constants.
\item The Sturmian-slice statements, including no-collision rigidity,
one-block higher-block conjugacy, entropy preservation under the fold, the
Sturmian \(m+1\) discrepancy constant, and the injective-placement comparisons
for the realized Sturmian law, transfer only in the case
\(\beta=\alpha\), equivalently \(I=h^{-1}([0,\alpha))\).
\item The finite-\(m\) Parry comparison identities transfer to the folded law
obtained from the corresponding pulled-back itinerary; when a comparison
theorem assumes the Sturmian slice or the golden slope, those assumptions are
carried along unchanged.
\end{enumerate}
If \(f\) is a \(C^2\) circle diffeomorphism with irrational rotation number
\(\alpha\), the required conjugacy exists by Denjoy's theorem
\cite{Denjoy1932CourbesTore,KatokHasselblatt1995}.
\end{theorem}

\begin{proof}
For every \(t\ge0\),
\[
\begin{aligned}
s_t^f
&=
\mathbf 1_I(f^t(x_0))
=
\mathbf 1_{[0,\beta)}(h(f^t(x_0)))\\
&=
\mathbf 1_{[0,\beta)}(R_\alpha^t(h(x_0)))
=
\mathbf 1_{[0,\beta)}(y_0+t\alpha).
\end{aligned}
\]
Thus the entire symbolic sequence generated by \(f\) agrees term-by-term with
the rotation coding of \(y_0\). Every length-\(m\) block, every folded block,
and every finite-sample histogram is therefore identical in the two models, so
the empirical microstate laws, empirical folded laws, and projective residuals
are the same.

For the limiting laws, the conjugacy pushes the \(f\)-invariant measure
\(h^{-1}_*\leb\) to Lebesgue measure on the rotation model. Hence the
length-\(m\) block law generated by \(f\) with respect to \(I\) is exactly the
rotation law \(\mu_m\), and its folded pushforward is
\(\pi_m^{\mathrm{fold}}\).

It remains only to check that no hypothesis is lost in the transfer.  The
general-window estimates in Sections~3--4 use the orbit, the half-open window
\([0,\beta)\), the induced finite partition, the atom separation constants,
and the deterministic fold.  All of these objects are carried to the
\(f\)-model by \(h^{-1}\), so their statements and constants are identical.
The rigidity and relabeling statements in Section~4 also assume the special
identity \(\beta=\alpha\).  Under conjugacy this is exactly the assertion that
the \(f\)-window is \(h^{-1}([0,\alpha))\).  In that case the symbolic language
is the Sturmian language of slope \(\alpha\), so the support
\(S_m(\alpha)\), the map \(J_m=\Fold_m|_{S_m(\alpha)}\), and the
coordinatewise conjugacy \(J_m^{\mathbb Z}\) are the same objects as in the
rotation proof.  For \(\beta\ne\alpha\) those Sturmian hypotheses are absent,
and the theorem asserts only the general-window conclusions.  Finally, the
Parry formulas in Section~5 are functions of the folded \(m\)-block law and
of the fixed golden-mean reference law; hence they transfer to the conjugate
model precisely under the hypotheses with which they were proved.
\end{proof}
