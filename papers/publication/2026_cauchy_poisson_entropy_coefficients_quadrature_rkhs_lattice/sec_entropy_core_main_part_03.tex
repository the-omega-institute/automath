It remains to identify the projection without using an \(X_c^2\)-integrability
hypothesis.  For fixed \(t>0\),
\[
\delta_t(y)=\int_{\RR}\{R_{X_c/t}(y)-1\}\,\dd\nu(\gamma),
\]
where the integrand is bounded as a function of \(\gamma\) for each fixed
\(y\).  Moreover
\[
R_\varepsilon(y)\omega(\dd y)
=\frac{\dd y}{\pi(1+(y-\varepsilon)^2)},
\qquad
\int_{\RR}|R_\varepsilon-1|\,\dd\omega\le2 .
\]
Since \(\|u_2\|_\infty\le1\), Tonelli gives the uniform bound
\[
\int_{\RR}\!\!\int_{\RR}
|u_2(y)|\,|R_{X_c/t}(y)-1|\,\omega(\dd y)\,\nu(\dd\gamma)
\le2 .
\]
Thus Fubini is justified against the possibly infinite-variance law \(\nu\),
and Lemma~\ref{lem:second-mode-cauchy-translate-transform} gives
\[
B(t):=
\int_{\RR}u_2(y)\delta_t(y)\,\omega(\dd y)
=
\EE\!\left[\frac{4(X_c/t)^2}{(4+(X_c/t)^2)^2}\right]\ge0 .
\]
Combining this identity with \eqref{eq:bounded-test-pinsker-second-mode}
yields
\[
t^4H(t)\ge\frac12\{t^2B(t)\}^2 .
\]
For fixed \(M<\infty\),
\[
t^2B(t)\ge
\frac{4}{(4+(M/t)^2)^2}
\EE[X_c^2\mathbf1_{\{|X_c|\le M\}}],
\]
and the right side tends to
\(\frac14\EE[X_c^2\mathbf1_{\{|X_c|\le M\}}]\).  Letting
\(M\to\infty\) proves divergence when the variance is infinite.
\end{proof}

\begin{remark}
This theorem supplies the threshold clause in
Theorem~\ref{thm:sharp-leading-kl-variance-threshold}.
Its proof uses the coefficient theorem only in the finite-variance direction;
the infinite-variance direction is a bounded-test lower bound and does not
assert a finite Laurent expansion.
\end{remark}

\subsection{Multidimensional quotient and covariance package}

For \(d\ge1\), let
\[
P_t^{(d)}(x)=c_d\,\frac{t}{(t^2+|x|^2)^{(d+1)/2}},
\qquad
c_d=\frac{\Gamma((d+1)/2)}{\pi^{(d+1)/2}},
\qquad
\Omega_d(\dd y)=P_1^{(d)}(y)\dd y.
\]
Thus \(P_t^{(d)}\) is normalized to have total mass one and
\[
P_t^{(d)}(ty)t^d\,\dd y=\Omega_d(\dd y)
=c_d(1+|y|^2)^{-(d+1)/2}\dd y .
\]
For \(z\in\RR^d\), write
\[
F_y(z)=\left(\frac{1+|y|^2}{1+|y-z|^2}\right)^{(d+1)/2}.
\]
Let \(U_\alpha^{(d)}\) be the Taylor coefficients of \(F_y(z)\) at \(z=0\).

\begin{lemma}[Local multidimensional quotient--Taylor remainder]%
\label{lem:local-multidim-quotient-taylor-remainder-core}
Let \(d\ge1\), and let
\[
F_y(z)=\left(\frac{1+|y|^2}{1+|y-z|^2}\right)^{(d+1)/2}.
\]
For a multiindex \(\alpha\), put
\[
U_\alpha^{(d)}(y)=\frac1{\alpha!}\partial_z^\alpha F_y(0).
\]
For each integer \(L\ge2\) there is a constant \(C_{d,L}<\infty\),
depending only on \(d\) and \(L\), such that
\begin{equation}\label{eq:local-multidim-coefficient-bound}
\sup_{y\in\RR^d}|U_\alpha^{(d)}(y)|\le C_{d,L}
\qquad (|\alpha|\le L),
\end{equation}
and
\begin{equation}\label{eq:local-multidim-taylor-remainder}
\sup_{y\in\RR^d}\left|
F_y(z)-\sum_{|\alpha|\le L}U_\alpha^{(d)}(y)z^\alpha
\right|
\le C_{d,L}|z|^{L+1}
\qquad (|z|\le1).
\end{equation}
\end{lemma}

\begin{proof}
Put \(a=(d+1)/2\).  For each multiindex \(\beta\), repeated
differentiation in \(z\) gives
\begin{equation}\label{eq:local-multidim-derivative-structure}
\partial_z^\beta F_y(w)
=(1+|y|^2)^a
\frac{P_\beta(y-w)}{(1+|y-w|^2)^{a+|\beta|}},
\end{equation}
where \(P_\beta\) is a polynomial of degree at most \(|\beta|\), depending
only on \(d\) and \(\beta\).  Indeed, \(z\)-differentiation is
\(-\xi\)-differentiation for \(\xi=y-z\), and each differentiation of
\((1+|\xi|^2)^{-b}\) raises the denominator exponent by one and multiplies
by a polynomial whose degree increases by at most one.

If \(|w|\le1\) and \(|y|>2\), then \(|y-w|\ge |y|/2\), and
\[
|\partial_z^\beta F_y(w)|
\le C_{d,\beta}
(1+|y|^2)^a
\frac{(1+|y|)^{|\beta|}}{(1+|y|^2)^{a+|\beta|}}
\le C_{d,\beta}.
\]
If \(|y|\le2\), the same derivative is bounded uniformly for \(|w|\le1\)
because the denominator in \eqref{eq:local-multidim-derivative-structure}
is at least \(1\).  Thus
\[
\sup_{\substack{y\in\RR^d,\ |w|\le1\\ |\beta|\le L+1}}
|\partial_z^\beta F_y(w)|<\infty .
\]
Taking \(w=0\) and \(|\beta|\le L\) proves
\eqref{eq:local-multidim-coefficient-bound}.  The multivariable Taylor
formula at \(z=0\), with integral remainder and the displayed derivative
bound for \(|\beta|=L+1\), gives
\eqref{eq:local-multidim-taylor-remainder}.
\end{proof}

\subsubsection*{Covariance algebra}\leavevmode\par\smallskip

The next lemma and theorem evaluate the quadratic layer associated with a
finite covariance matrix.

\begin{lemma}[Radial--angular contractions for the Cauchy weight]%
\label{lem:radial-angular-contractions}
Let \(Y\) have law
\(\Omega_d(\dd y)=P_1^{(d)}(y)\dd y\), write the polar decomposition
\[
Y=\sqrt R\,\Theta,\qquad R=|Y|^2,\qquad \Theta\in S^{d-1}.
\]
Then \(R\) and \(\Theta\) are independent, \(\Theta\) is uniform on
\(S^{d-1}\), and \(R\) has beta-prime law with parameters \(d/2\) and
\(1/2\).  The angular fourth moments are
\begin{equation}\label{eq:spherical-fourth-contraction}
\EE_\Theta[\Theta_i\Theta_j\Theta_k\Theta_l]
=
\frac{\delta_{ij}\delta_{kl}
+\delta_{ik}\delta_{jl}
+\delta_{il}\delta_{jk}}{d(d+2)}
\qquad(1\le i,j,k,l\le d).
\end{equation}
Consequently, for real symmetric matrices \(A,B\),
\begin{equation}\label{eq:spherical-quadratic-contraction}
\EE_\Theta[
(\Theta^{\mathsf T}A\Theta)(\Theta^{\mathsf T}B\Theta)]
=
\frac{(\operatorname{tr}A)(\operatorname{tr}B)
+2\operatorname{tr}(AB)}{d(d+2)}.
\end{equation}
In particular, if \(A\) is traceless, then
\begin{equation}\label{eq:spherical-traceless-square}
\EE_\Theta[(\Theta^{\mathsf T}A\Theta)^2]
=\frac{2}{d(d+2)}\|A\|_{\rm HS}^2 .
\end{equation}
For every real \(k\) with \(-d/2<k<9/2\),
\begin{equation}\label{eq:radial-beta-prime-contraction-general}
\EE\frac{R^k}{(1+R)^4}
=
\frac{B(d/2+k,\,9/2-k)}{B(d/2,\,1/2)}
=
\frac{\Gamma(d/2+k)\Gamma(9/2-k)}
{\Gamma(d/2)\Gamma(1/2)}
\frac{\Gamma((d+1)/2)}{\Gamma((d+9)/2)} .
\end{equation}
Thus, with \(D_d=(d+1)(d+3)(d+5)(d+7)\),
\begin{equation}\label{eq:radial-beta-prime-contractions-low}
\EE\frac{1}{(1+R)^4}=\frac{105}{D_d},\qquad
\EE\frac{R}{(1+R)^4}=\frac{15d}{D_d},\qquad
\EE\frac{R^2}{(1+R)^4}=\frac{3d(d+2)}{D_d}.
\end{equation}
\end{lemma}

\begin{proof}
The density \(P_1^{(d)}(y)\) is radial and proportional to
\((1+|y|^2)^{-(d+1)/2}\).  Passing to polar coordinates shows that
\(\Theta\) is uniform, independent of \(R\), and that \(R\) has density
\[
\frac{1}{B(d/2,1/2)}\,r^{d/2-1}(1+r)^{-(d+1)/2},
\qquad r>0.
\]
This is the beta-prime distribution with parameters \(d/2\) and \(1/2\).
Multiplying this density by \(r^k(1+r)^{-4}\) and integrating gives
\eqref{eq:radial-beta-prime-contraction-general}; the three rational values
in \eqref{eq:radial-beta-prime-contractions-low} are the cases
\(k=0,1,2\) after expanding the gamma ratios.  For example the
\(d=1,\ k=0\) value is \(35/128\), the one-dimensional Cauchy value.

For the angular identity, rotational invariance implies that the tensor
\(\EE_\Theta[\Theta_i\Theta_j\Theta_k\Theta_l]\) is a multiple of
\(\delta_{ij}\delta_{kl}+\delta_{ik}\delta_{jl}+\delta_{il}\delta_{jk}\).
Contracting \(i=j\) and \(k=l\) gives
\[
1=\EE_\Theta|\Theta|^4=C(d^2+2d),
\]
so \(C=1/(d(d+2))\), proving
\eqref{eq:spherical-fourth-contraction}.  Contracting this identity with
\(A_{ij}B_{kl}\) gives \eqref{eq:spherical-quadratic-contraction}; taking
\(\operatorname{tr}A=0\) and \(B=A\) gives
\eqref{eq:spherical-traceless-square}.
\end{proof}

\subsubsection*{Algebraic covariance layer}\leavevmode\par\smallskip

The following theorem evaluates the finite quadratic form determined by the
degree-two density mode.  The proof is written directly at the quotient level:
first the homogeneous degree-two part of
\((1+(-2y\cdot z+|z|^2)/(1+|y|^2))^{-(d+1)/2}\) is extracted, and only then
is the covariance average taken.

\begin{theorem}[Covariance algebra under finite covariance]%
\label{thm:multidim-covariance-algebra}
\label{prop:multidim-formal-covariance-layer}
Let \(d\ge1\), and let \(\nu\) be a probability measure on \(\RR^d\) with
finite centred second moment.  Put
\[
\bar{\gamma}=\int_{\RR^d}\gamma\,\dd\nu(\gamma),
\qquad
X_c=\gamma-\bar{\gamma},
\]
and
\[
\Sigma=\EE[X_cX_c^{\mathsf T}],\qquad
\Sigma_0=\Sigma-\frac{\operatorname{tr}\Sigma}{d}I_d.
\]
The degree-two centred density mode associated with \(\Sigma\) is
\begin{equation}\label{eq:multidim-leading-theorem-density-mode}
b_\Sigma(y)=
\frac{d+1}{2}\left(
-\frac{\operatorname{tr}\Sigma}{1+|y|^2}
+\frac{(d+3)y^{\mathsf T}\Sigma y}{(1+|y|^2)^2}
\right).
\end{equation}
Define its algebraic quadratic entropy-integrand layer by
\begin{equation}\label{eq:multidim-leading-positive-defect}
\mathcal Q_d(\Sigma)
:=\frac12\int_{\RR^d}b_\Sigma(y)^2\,\Omega_d(\dd y)
=c_d^{\rm iso}(\operatorname{tr}\Sigma)^2
+c_d^{\rm tr}\|\Sigma_0\|_{\rm HS}^2,
\end{equation}
with
\begin{equation}\label{eq:multidim-leading-positive-constants}
c_d^{\rm iso}=
\frac{3(d+1)(7d+9)}
{4d(d+3)(d+5)(d+7)},
\qquad
c_d^{\rm tr}=
\frac{3(d+1)(d+3)}
{4(d+5)(d+7)} .
\end{equation}
Two immediate checks are
\[
d=1:\qquad \Sigma_0=0,\qquad c_1^{\rm iso}=\frac18,\qquad
\mathcal Q_1(\Sigma)=\frac{(\operatorname{tr}\Sigma)^2}{8},
\]
and, for an isotropic covariance \(\Sigma=\tau I_d\),
\[
\Sigma_0=0,\qquad
\mathcal Q_d(\Sigma)=c_d^{\rm iso}d^2\tau^2 .
\]
This is the algebraic covariance calculation.  No KL asymptotic is asserted
in this theorem.  Lemma~\ref{lem:radial-angular-contractions} gives the radial
and angular contractions entering the constants.
\end{theorem}

\begin{proof}
First compute the formal degree-two density layer.  Let
\[
  a=\frac{d+1}{2},\qquad A=1+|y|^2 .
\]
For a deterministic \(z\in\RR^d\), the radial Poisson quotient is
\[
F_y(z):=
\left(\frac{1+|y|^2}{1+|y-z|^2}\right)^a
=
\left(1+\frac{-2y\cdot z+|z|^2}{A}\right)^{-a}.
\]
Write
\[
  w=\frac{-2y\cdot z+|z|^2}{A}=w_1+w_2,\qquad
  w_1=-\frac{2y\cdot z}{A},\qquad
  w_2=\frac{|z|^2}{A}.
\]
The binomial expansion gives
\[
  (1+w)^{-a}=1-aw+\frac{a(a+1)}2w^2+O(|w|^3),
\]
and the homogeneous degree-two contribution comes only from \(-aw_2\) and
\(\frac{a(a+1)}2w_1^2\).  Hence that degree-two part is
\[
a\left(-\frac{|z|^2}{A}
+\frac{(d+3)(y\cdot z)^2}{A^2}\right).
\]
Indeed,
\[
  -aw_2+\frac{a(a+1)}2w_1^2
  =
  -a\frac{|z|^2}{A}
  +2a(a+1)\frac{(y\cdot z)^2}{A^2},
\]
and \(2(a+1)=d+3\).  No other term contributes at degree two:
\(-aw_1\) is degree one, \(w_1w_2\) is degree three, and \(w_2^2\) is degree
four.  The degree-zero term is the matched reference density and the
degree-one term vanishes after centering.  Averaging the displayed
degree-two term with \(z=X_c/t\) gives
\[
  D_2^{(d)}(y;\nu)=b_\Sigma(y)=
  \frac{d+1}{2}\left(
  -\frac{\operatorname{tr}\Sigma}{1+|y|^2}
  +\frac{(d+3)y^{\mathsf T}\Sigma y}{(1+|y|^2)^2}
  \right),
\]
since
\[
\EE |X_c|^2=\operatorname{tr}\Sigma,\qquad
\EE (y\cdot X_c)^2=y^{\mathsf T}\Sigma y .
\]

It remains to evaluate \(\frac12\int b_\Sigma^2\,\dd\Omega_d\).  Decompose
\(\Sigma=\tau I_d+\Sigma_0\), \(\tau=d^{-1}\operatorname{tr}\Sigma\).  Then
\[
b_\Sigma(y)=
a\tau\frac{3|y|^2-d}{(1+|y|^2)^2}
+a(d+3)\frac{y^{\mathsf T}\Sigma_0y}{(1+|y|^2)^2}.
\]
Let \(Y=\sqrt R\,\Theta\) be distributed according to \(\Omega_d\).  In these
variables,
\[
b_\Sigma(Y)=
a\tau\frac{3R-d}{(1+R)^2}
+a(d+3)\frac{R\,\Theta^{\mathsf T}\Sigma_0\Theta}{(1+R)^2}.
\]
The mixed term in \(\EE b_\Sigma(Y)^2\) is zero because \(R\) and \(\Theta\)
are independent and
\[
\EE_\Theta[\Theta^{\mathsf T}\Sigma_0\Theta]
=\frac{\operatorname{tr}\Sigma_0}{d}=0 .
\]

The isotropic part uses only the low radial moments from
Lemma~\ref{lem:radial-angular-contractions}.  With
\[
  D_d=(d+1)(d+3)(d+5)(d+7),
\]
those moments are
\[
  \EE\frac1{(1+R)^4}=\frac{105}{D_d},\qquad
  \EE\frac{R}{(1+R)^4}=\frac{15d}{D_d},\qquad
  \EE\frac{R^2}{(1+R)^4}=\frac{3d(d+2)}{D_d}.
\]
Therefore
\[
\begin{aligned}
\EE\frac{(3R-d)^2}{(1+R)^4}
&=
9\EE\frac{R^2}{(1+R)^4}
-6d\EE\frac{R}{(1+R)^4}
+d^2\EE\frac1{(1+R)^4} \\
&=
\frac{27d(d+2)-90d^2+105d^2}{D_d} \\
&=
\frac{6d(7d+9)}
{(d+1)(d+3)(d+5)(d+7)} .
\end{aligned}
\]
The isotropic contribution to \(\frac12\int b_\Sigma^2\,\dd\Omega_d\) is
\[
\begin{aligned}
\frac12 a^2\frac{(\operatorname{tr}\Sigma)^2}{d^2}
\EE\frac{(3R-d)^2}{(1+R)^4}
&=
\frac{3(d+1)(7d+9)}
{4d(d+3)(d+5)(d+7)}
(\operatorname{tr}\Sigma)^2 .
\end{aligned}
\]
For the traceless part, the radial and angular factors separate.  The
spherical fourth-moment contraction gives
\[
\EE_\Theta[(\Theta^{\mathsf T}\Sigma_0\Theta)^2]
=\frac{2}{d(d+2)}\|\Sigma_0\|_{\rm HS}^2,
\]
because \(\operatorname{tr}\Sigma_0=0\).  Hence
\[
\begin{aligned}
\int_{\RR^d}
\frac{(y^{\mathsf T}\Sigma_0y)^2}{(1+|y|^2)^4}\,\Omega_d(\dd y)
&=
\EE\frac{R^2}{(1+R)^4}\,
\EE_\Theta[(\Theta^{\mathsf T}\Sigma_0\Theta)^2] \\
&=
\frac{3d(d+2)}{D_d}\,
\frac{2}{d(d+2)}
\|\Sigma_0\|_{\rm HS}^2 \\
&=
\frac{6}{(d+1)(d+3)(d+5)(d+7)}
\|\Sigma_0\|_{\rm HS}^2 .
\end{aligned}
\]
Here the first factor is the \(k=2\) value in
\eqref{eq:radial-beta-prime-contractions-low}, and the second factor is the
traceless angular contraction
\eqref{eq:spherical-traceless-square}.
Multiplication by \(\frac12a^2(d+3)^2\) gives
\[
\frac{3(d+1)(d+3)}{4(d+5)(d+7)}
\|\Sigma_0\|_{\rm HS}^2 .
\]
This proves \eqref{eq:multidim-leading-positive-constants}.  The constants are
positive, so the layer vanishes exactly when \(\Sigma=0\).
\end{proof}

\subsubsection*{Second-order quotient bridge}\leavevmode\par\smallskip

The next statements record a second-order quotient criterion for the leading
coefficient.  The finite \(d+1\)-moment theorem supplies the uniform version.

\noindent The following lemma shows that \(L^2\) convergence alone controls
the nonlinear part of a non-uniform second-order entropy transfer.

\begin{lemma}[\(L^2\) second-order entropy criterion]%
\label{lem:l2-second-order-entropy-criterion}
Let \((X,\mathcal A,\lambda)\) be a probability space and let
\(\delta_t:X\to[-1,\infty)\) be measurable with
\(\int_X\delta_t\,\dd\lambda=0\).  Let \(b\in L^2(\lambda)\), and put
\[
  \Phi(s)=(1+s)\log(1+s)-s .
\]
Assume only
\[
  t^2\delta_t\longrightarrow b
  \qquad\text{in }L^2(\lambda).
\]
Then
\[
  \int_X\Phi(\delta_t)\,\dd\lambda
  =\frac{t^{-4}}2\int_X b^2\,\dd\lambda+o(t^{-4}).
\]
\end{lemma}

\begin{proof}
Write \(z_t=t^2\delta_t\), and define
\[
r(s)=
\begin{cases}
\dfrac{\Phi(s)-s^2/2}{s^2},&s\ne0,\\[4pt]
0,&s=0.
\end{cases}
\]
The function \(r\) is bounded on \([-1,\infty)\) and continuous at zero.
Indeed, the Taylor expansion at zero gives \(r(s)\to0\), while the limits at
\(-1\) and at infinity are finite.  The \(L^2\) convergence gives
\[
  \int_X z_t^2\,\dd\lambda\to\int_X b^2\,\dd\lambda .
\]
It also implies that \(z_t^2\to b^2\) in \(L^1(\lambda)\), so
\(\{z_t^2\}\) is uniformly integrable, whereas
\(z_t/t^2=\delta_t\to0\) in probability.  Boundedness of \(r\) and Vitali's
theorem therefore yield
\[
t^4\int_X\left|\Phi(\delta_t)-\frac12\delta_t^2\right|\dd\lambda
=\int_X\left|r\!\left(\frac{z_t}{t^2}\right)\right|z_t^2\dd\lambda
\longrightarrow0.
\]
Thus
\[
  \frac12\int_X\delta_t^2\,\dd\lambda
  =\frac{t^{-4}}2\int_Xz_t^2\,\dd\lambda
  =\frac{t^{-4}}2\int_Xb^2\,\dd\lambda+o(t^{-4}).
\]
Combining the last two displays proves the claim.
\end{proof}

\medskip

\begin{theorem}[Second-order quotient criterion for the KL coefficient]%
\label{thm:second-order-quotient-kl-coefficient}
\label{prop:second-order-quotient-criterion}
Let \(d\ge1\), and let \(\nu\) be a probability measure on \(\RR^d\) with
mean \(\bar{\gamma}\) and finite centred covariance \(\Sigma\).  Put
\[
  h_t=P_t^{(d)}*\nu,\qquad
  g_t=P_t^{(d)}(\cdot-\bar{\gamma}),\qquad
  \delta_t^{(d)}(y)=\frac{h_t(\bar{\gamma}+ty)}
  {g_t(\bar{\gamma}+ty)}-1 .
\]
Assume the separate second-order quotient criterion
\begin{equation}\label{eq:second-order-quotient-criterion}
  \delta_t^{(d)}(y)=t^{-2}b_\Sigma(y)+r_t(y),\qquad
  \|r_t\|_\infty+\|r_t\|_{L^1(\Omega_d)}=o(t^{-2}),\qquad
  \|\delta_t^{(d)}\|_\infty=O(t^{-2})
\end{equation}
with \(b_\Sigma\) as in
\eqref{eq:multidim-leading-theorem-density-mode}, and assume that the matched
KL integral is represented by
\[
D_{\rm KL}(h_t\|g_t)
=\int_{\RR^d}\Phi(\delta_t^{(d)}(y))\,\Omega_d(\dd y)
\]
for all sufficiently large \(t\).  Then
\[
D_{\rm KL}(h_t\|g_t)=\mathcal Q_d(\Sigma)t^{-4}+o(t^{-4}).
\]
The same conclusion also follows from the non-uniform criterion
\[
  t^2\delta_t^{(d)}\to b_\Sigma\quad\text{in }L^2(\Omega_d)
\]
provided the KL integral is represented by
\(\int\Phi(\delta_t^{(d)})\,\dd\Omega_d\) for all sufficiently large \(t\).
\end{theorem}

\begin{proof}
Under \eqref{eq:second-order-quotient-criterion},
\(\|\delta_t^{(d)}\|_\infty=O(t^{-2})\).  Hence, for all sufficiently large
\(t\),
\[
\Phi(s)=(1+s)\log(1+s)-s=\frac{s^2}{2}+O(s^3)
\]
uniformly for \(s=\delta_t^{(d)}(y)\).  Also \(b_\Sigma\) is bounded and
\eqref{eq:second-order-quotient-criterion} gives
\(\|\delta_t^{(d)}\|_{L^1(\Omega_d)}=O(t^{-2})\).  The quadratic part obeys
\[
\left\|\frac12\{(t^{-2}b_\Sigma+r_t)^2-t^{-4}b_\Sigma^2\}\right\|_{L^1(\Omega_d)}
\le
t^{-2}\|b_\Sigma\|_\infty\|r_t\|_{L^1(\Omega_d)}
+\frac12\|r_t\|_\infty\|r_t\|_{L^1(\Omega_d)}
=o(t^{-4}).
\]
The Taylor remainder has \(L^1\)-norm at most
\[
C\|\delta_t^{(d)}\|_\infty^2\|\delta_t^{(d)}\|_{L^1(\Omega_d)}=O(t^{-6}).
\]
Thus
\[
\Phi(\delta_t^{(d)})=\frac12t^{-4}b_\Sigma^2+o(t^{-4})
\quad\text{in }L^1(\Omega_d).
\]
For all large \(t\), the quotient is positive and the displayed \(L^1\)
estimate also gives \(\Phi(\delta_t^{(d)})\in L^1(\Omega_d)\).  Changing
variables in the density quotient and integrating,
\[
\begin{aligned}
D_{\rm KL}(h_t\|g_t)
&=\int_{\RR^d}\Phi(\delta_t^{(d)}(y))\,\Omega_d(\dd y)\\
&=\frac12t^{-4}\int_{\RR^d}b_\Sigma(y)^2\,\Omega_d(\dd y)+o(t^{-4})\\
&=\mathcal Q_d(\Sigma)t^{-4}+o(t^{-4}),
\end{aligned}
\]
because Theorem~\ref{thm:multidim-covariance-algebra} identifies
\(\frac12\int b_\Sigma^2\,\dd\Omega_d\) with \(\mathcal Q_d(\Sigma)\).
Under the \(L^2\) alternative, the same conclusion follows
by applying Lemma~\ref{lem:l2-second-order-entropy-criterion} with
\((X,\lambda)=(\RR^d,\Omega_d)\) and \(b=b_\Sigma\), followed by the same
identification of \(\frac12\int b_\Sigma^2\,\dd\Omega_d\).
\end{proof}
