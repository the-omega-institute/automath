\section{The Covariance-Matched Poisson Defect}
\label{sec:covariance-proxy-defect}

Finite covariance determines the quadratic mode but, in dimensions at least
four, does not force the matched entropy to attain the corresponding
coefficient.  The following result identifies the complete nonnegative
remainder without imposing a tail class or an absolute moment above order
two.

\begin{theorem}[Covariance-proxy defect decomposition]
\label{thm:covariance-proxy-defect-decomposition}
\label{thm:finite-covariance-proxy-defect-decomposition}
Let \(d\ge1\), and let \(\nu\) be a probability law on \(\RR^d\) with mean
\(\bar\gamma\) and finite covariance matrix \(\Sigma\).  Put
\[
 h_t=P_t^{(d)}*\nu,
 \qquad g_t=P_t^{(d)}(\mathord\cdot-\bar\gamma),
 \qquad H_d(t)=D_{\rm KL}(h_t\|g_t),
\]
and let \(b_\Sigma\) be the bounded covariance mode in
\eqref{eq:multidim-leading-theorem-density-mode}.  For all sufficiently
large \(t\),
\begin{equation}\label{eq:covariance-poisson-proxy}
 k_{\Sigma,t}(x)
 :=g_t(x)\left[1+t^{-2}b_\Sigma
 \left(\frac{x-\bar\gamma}{t}\right)\right]
\end{equation}
is a probability density.  Define
\begin{equation}\label{eq:covariance-proxy-defect-functional}
 \mathcal C_{\nu,\Sigma}(t)
 :=t^4D_{\rm KL}(h_t\|k_{\Sigma,t}).
\end{equation}
Then
\begin{equation}\label{eq:covariance-proxy-defect-decomposition}
 t^4H_d(t)
 =\mathcal Q_d(\Sigma)+\mathcal C_{\nu,\Sigma}(t)+o(1)
 \qquad(t\to\infty).
\end{equation}
Consequently,
\begin{equation}\label{eq:covariance-proxy-defect-equivalence}
 t^4H_d(t)\longrightarrow\mathcal Q_d(\Sigma)
 \quad\Longleftrightarrow\quad
 \mathcal C_{\nu,\Sigma}(t)\longrightarrow0.
\end{equation}
No density of \(\nu\), regular-variation hypothesis, angular balance, or
higher absolute moment is required.
\end{theorem}

\begin{proof}
After translation, assume \(\bar\gamma=0\), and write
\[
 \Omega_d(\dd y)=P_1^{(d)}(y)\dd y,
 \qquad
 F_y(z)=\left(\frac{1+|y|^2}{1+|y-z|^2}\right)^{(d+1)/2}.
\]
In normalized coordinates,
\[
 q_t(y):=\frac{h_t(ty)}{g_t(ty)}
 =\EE F_y(X/t),
 \qquad \delta_t=q_t-1,
 \qquad u_t=t^{-2}b_\Sigma.
\]
The covariance calculation in
Theorem~\ref{thm:multidim-covariance-algebra} gives
\(\int b_\Sigma\dd\Omega_d=0\).  Since \(b_\Sigma\) is bounded,
\(1+u_t\ge1/2\) for all sufficiently large \(t\), proving that
\eqref{eq:covariance-poisson-proxy} is positive and has integral one.

We first establish the only asymptotic input needed below:
\begin{equation}\label{eq:covariance-proxy-weak-pairing}
 t^2\int_{\RR^d}b_\Sigma(y)\delta_t(y)\,\Omega_d(\dd y)
 \longrightarrow
 \int_{\RR^d}b_\Sigma(y)^2\,\Omega_d(\dd y).
\end{equation}
Set
\[
 J(z)=\int_{\RR^d}b_\Sigma(y)F_y(z)\,\Omega_d(\dd y)
 =\int_{\RR^d}b_\Sigma(y)P_1^{(d)}(y-z)\dd y.
\]
The function \(b_\Sigma\) is even, smooth, and has bounded second
derivatives.  Translation in the last integral shows that \(J\) is even,
\(J(0)=0\), and its Hessian is bounded on the unit ball.  Hence
\(|J(z)|\le C|z|^2\) for \(|z|\le1\).  For \(|z|\ge1\), positivity and
normalization of the translated Poisson density give
\(|J(z)|\le\|b_\Sigma\|_\infty\le
\|b_\Sigma\|_\infty|z|^2\).  Thus
\begin{equation}\label{eq:covariance-proxy-global-quadratic-pairing}
 |J(z)|\le C_{d,\Sigma}|z|^2,
 \qquad z\in\RR^d.
\end{equation}

Let \(B_2(y,z)\) denote the homogeneous degree-two Taylor term of
\(F_y(z)\) at zero.  Taylor expansion under the integral gives
\[
 J(z)=\int b_\Sigma(y)B_2(y,z)\,\Omega_d(\dd y)+o(|z|^2).
\]
The covariance-mode identity proved in
Theorem~\ref{thm:multidim-covariance-algebra} is
\(\EE B_2(y,X)=b_\Sigma(y)\).  Since \(\EE|X|^2<\infty\),
\eqref{eq:covariance-proxy-global-quadratic-pairing} and dominated
convergence imply
\[
 \EE[t^2J(X/t)]
 \longrightarrow
 \int b_\Sigma(y)\EE B_2(y,X)\,\Omega_d(\dd y)
 =\int b_\Sigma^2\dd\Omega_d.
\]
Because \(\int b_\Sigma\dd\Omega_d=0\), the left side is precisely the
left side of \eqref{eq:covariance-proxy-weak-pairing}.

We also have
\begin{equation}\label{eq:covariance-proxy-l1-smallness}
 \|\delta_t\|_{L^1(\Omega_d)}\longrightarrow0.
\end{equation}
Indeed, Minkowski's integral inequality bounds this norm by
\[
 \EE\left\|P_1^{(d)}(\mathord\cdot-X/t)-P_1^{(d)}
 \right\|_{L^1(\RR^d)},
\]
which tends to zero by translation continuity and dominated convergence.

Put \(r_t=\delta_t-u_t\).  Equations
\eqref{eq:covariance-proxy-weak-pairing} and
\eqref{eq:covariance-proxy-l1-smallness} give
\begin{equation}\label{eq:covariance-proxy-remainder-pairing}
 t^2\int b_\Sigma r_t\dd\Omega_d\longrightarrow0,
 \qquad \|r_t\|_{L^1(\Omega_d)}\longrightarrow0.
\end{equation}
Uniformly in \(y\),
\[
 \log(1+u_t)=u_t+O(t^{-4}).
\]
It follows from \eqref{eq:covariance-proxy-remainder-pairing} that
\begin{equation}\label{eq:covariance-proxy-cross-remainder}
 \int r_t\log(1+u_t)\dd\Omega_d=o(t^{-4}).
\end{equation}
On the other hand, with
\(\Phi(s)=(1+s)\log(1+s)-s\), boundedness of \(b_\Sigma\) gives
\begin{align}
 \int(1+u_t)\log(1+u_t)\dd\Omega_d
 &=\int\Phi(u_t)\dd\Omega_d \notag\\
 &=\frac{t^{-4}}2\int b_\Sigma^2\dd\Omega_d+O(t^{-6})
 =t^{-4}\mathcal Q_d(\Sigma)+O(t^{-6}).
 \label{eq:covariance-proxy-self-cross-entropy}
\end{align}
Combining \eqref{eq:covariance-proxy-cross-remainder} and
\eqref{eq:covariance-proxy-self-cross-entropy} yields
\begin{equation}\label{eq:covariance-proxy-cross-entropy-asymptotic}
 \int(1+\delta_t)\log(1+u_t)\dd\Omega_d
 =t^{-4}\mathcal Q_d(\Sigma)+o(t^{-4}).
\end{equation}

Finally, the exact logarithmic identity is
\[
 H_d(t)=D_{\rm KL}(h_t\|k_{\Sigma,t})
 +\int(1+\delta_t)\log(1+u_t)\dd\Omega_d.
\]
 All terms are finite for large \(t\).  Indeed, convexity of relative entropy
 and
 \[
  D_{\rm KL}\!\left(P_1^{(d)}(\mathord\cdot-z)\,\middle\|\,
  P_1^{(d)}\right)
  \le \frac{d+1}{2}\log\{2(1+|z|^2)\}
  \le C_d(1+|z|^2)
 \]
 bound \(H_d(t)\) by an integrable average under
 \(\EE|X|^2<\infty\).  The first inequality follows from
 \(1+|w+z|^2\le2(1+|w|^2)(1+|z|^2)\).  Moreover,
 \(\log(1+u_t)\) is bounded.  Substitution of
\eqref{eq:covariance-proxy-cross-entropy-asymptotic} proves
\eqref{eq:covariance-proxy-defect-decomposition}.  Nonnegativity of relative
entropy then gives \eqref{eq:covariance-proxy-defect-equivalence}.
\end{proof}

\begin{theorem}[Raw-tail Poisson-energy decomposition]
\label{thm:raw-tail-poisson-energy-decomposition}
Let \(d\ge1\), and let \(\nu\) be a probability law on \(\RR^d\) with
mean \(\bar\gamma\) and finite covariance matrix \(\Sigma\).  Write
\(\nu_c\) for its centred law, put \(a=(d+1)/2\), and define
\[
 \Omega_d(\dd y)=P_1^{(d)}(y)\dd y,
 \qquad
 F_y(z)=\left(\frac{1+|y|^2}{1+|y-z|^2}\right)^a,
 \qquad
 \Phi(v)=(1+v)\log(1+v)-v.
\]
For \(t>0\), let
\begin{equation}\label{eq:raw-tail-poisson-potential}
 V_t^\nu(y):=\int_{\{|x|>t\}}F_y(x/t)\,\nu_c(\dd x).
\end{equation}
Then
\begin{equation}\label{eq:raw-tail-entropy-decomposition}
 H_d(t)=\mathcal Q_d(\Sigma)t^{-4}
 +\int_{\RR^d}\Phi(V_t^\nu)\dd\Omega_d+o(t^{-4}).
\end{equation}
Consequently, the covariance-proxy defect in
\eqref{eq:covariance-proxy-defect-functional} satisfies
\begin{equation}\label{eq:raw-tail-proxy-decomposition}
 \mathcal C_{\nu,\Sigma}(t)
 =t^4\int_{\RR^d}\Phi(V_t^\nu)\dd\Omega_d+o(1),
\end{equation}
and hence
\begin{equation}\label{eq:raw-tail-proxy-equivalence}
 \mathcal C_{\nu,\Sigma}(t)\longrightarrow0
 \quad\Longleftrightarrow\quad
 t^4\int_{\RR^d}\Phi(V_t^\nu)\dd\Omega_d\longrightarrow0.
\end{equation}
No regular-variation or absolute-continuity hypothesis is required.
\end{theorem}

\begin{proof}
Translation permits us to assume that \(\nu\) is centred.  Set
\[
 \delta_t(y)=\int\{F_y(x/t)-1\}\nu(\dd x),
 \qquad u_t=t^{-2}b_\Sigma,
 \qquad \tau_t=\nu(|x|>t).
\]
Finite second moment gives
\begin{equation}\label{eq:raw-tail-mass-smallness}
 \tau_t\le t^{-2}\int_{|x|>t}|x|^2\nu(\dd x)=o(t^{-2}).
\end{equation}
Let \(\ell_y(z)\) and \(B_2(y,z)\) be the homogeneous degree-one and
degree-two Taylor terms of \(F_y(z)\) at zero.  Direct differentiation of
the rational function \(F_y\), together with translation continuity of
the first two derivatives of \(P_1^{(d)}\) in \(L^1(\RR^d)\), gives
\begin{equation}\label{eq:raw-tail-local-taylor-modulus}
 \|F_\cdot(z)-1-\ell_\cdot(z)-B_2(\cdot,z)\|_\infty
 +\|F_\cdot(z)-1-\ell_\cdot(z)-B_2(\cdot,z)\|_{L^1(\Omega_d)}
 \le \varepsilon_d(|z|)|z|^2
\end{equation}
for \(|z|\le1\), where \(\varepsilon_d\) is bounded on \([0,1]\) and
\(\varepsilon_d(r)\to0\) as \(r\downarrow0\).  The Taylor coefficients
themselves satisfy, for \(j=1,2\),
\begin{equation}\label{eq:raw-tail-taylor-coefficient-norms}
 \|B_j(\cdot,z)\|_\infty+
 \|B_j(\cdot,z)\|_{L^1(\Omega_d)}\le C_d|z|^j,
 \qquad B_1=\ell.
\end{equation}

Centering and the covariance-mode identity
\(\int B_2(y,x)\nu(\dd x)=b_\Sigma(y)\) now yield the exact split
\begin{equation}\label{eq:raw-tail-quotient-split}
 \delta_t=u_t+V_t^\nu-\tau_t+R_t,
\end{equation}
where
\[
 R_t=
 \int_{|x|\le t}\{F_\cdot(x/t)-1-\ell_\cdot(x/t)
                   -B_2(\cdot,x/t)\}\nu(\dd x)
 -\int_{|x|>t}\{\ell_\cdot(x/t)+B_2(\cdot,x/t)\}\nu(\dd x).
\]
Equations \eqref{eq:raw-tail-local-taylor-modulus} and
\eqref{eq:raw-tail-taylor-coefficient-norms} imply
\begin{equation}\label{eq:raw-tail-remainder-smallness}
 \|R_t\|_\infty+\|R_t\|_{L^1(\Omega_d)}=o(t^{-2}).
\end{equation}
Indeed, after multiplication by \(t^2\), the interior remainder is
dominated by \(C_d|x|^2\) and converges pointwise to zero.  The omitted
linear and quadratic tail terms are bounded respectively by
\[
 Ct^{-1}\int_{|x|>t}|x|\nu(\dd x)
 \le Ct^{-2}\int_{|x|>t}|x|^2\nu(\dd x)
 \quad\text{and}\quad
 Ct^{-2}\int_{|x|>t}|x|^2\nu(\dd x).
\]

The identity
\(F_y(z)\Omega_d(\dd y)=P_1^{(d)}(y-z)\dd y\) gives
\begin{equation}\label{eq:raw-tail-potential-normalization}
 V_t^\nu\ge0,
 \qquad \int V_t^\nu\dd\Omega_d=\tau_t.
\end{equation}
We next justify the nonlinear separation.  If \(|u|\le1/4\) and \(v\ge0\),
two integrations of \(\Phi''(s)=(1+s)^{-1}\) over the rectangle joining
\((0,0)\) to \((u,v)\) give
\begin{equation}\label{eq:raw-tail-phi-interaction}
 |\Phi(u+v)-\Phi(u)-\Phi(v)|\le C|u|v.
\end{equation}
By \eqref{eq:raw-tail-mass-smallness} and
\eqref{eq:raw-tail-remainder-smallness}, all negative arguments in
\eqref{eq:raw-tail-quotient-split} are uniformly larger than \(-1/2\) for
large \(t\).  The mean-value theorem, the bound
\(\log(1+v)\le v\), and \eqref{eq:raw-tail-potential-normalization} then give
\begin{align}
 \int|\Phi(\delta_t)-
 \Phi(u_t+V_t^\nu-\tau_t)|\dd\Omega_d&=o(t^{-4}),
 \label{eq:raw-tail-remove-remainder}\\
 \int|\Phi(u_t+V_t^\nu-\tau_t)-
 \Phi(u_t+V_t^\nu)|\dd\Omega_d&=o(t^{-4}),
 \label{eq:raw-tail-remove-mass}\\
 \int|\Phi(u_t+V_t^\nu)-\Phi(u_t)-
 \Phi(V_t^\nu)|\dd\Omega_d&=o(t^{-4}).
 \label{eq:raw-tail-remove-interaction}
\end{align}
For clarity, the first left side is bounded by
\(C\|R_t\|_\infty\tau_t+C t^{-2}\|R_t\|_1
 +C\|R_t\|_\infty\|R_t\|_1\); the second is bounded by
\(C\tau_t(\tau_t+t^{-2})\); and the third is at most
\(C\|u_t\|_\infty\tau_t\).  These bounds are all \(o(t^{-4})\).

Finally, boundedness and zero mean of \(b_\Sigma\) imply
\[
 \int\Phi(u_t)\dd\Omega_d
 =\frac{t^{-4}}2\int b_\Sigma^2\dd\Omega_d+O(t^{-6})
 =\mathcal Q_d(\Sigma)t^{-4}+O(t^{-6}).
\]
Since \(H_d(t)=\int\Phi(\delta_t)\dd\Omega_d\), equations
\eqref{eq:raw-tail-remove-remainder}--
\eqref{eq:raw-tail-remove-interaction} prove
\eqref{eq:raw-tail-entropy-decomposition}.  Comparing it with
\eqref{eq:covariance-proxy-defect-decomposition} proves
\eqref{eq:raw-tail-proxy-decomposition} and the equivalence.
\end{proof}

For a formulation involving only masses of moving sets, define
\[
 A_{t,0}(y)=\{|x|>t:\ |x-ty|<t\},
\]
and, for \(k\ge1\),
\[
 A_{t,k}(y)=\{|x|>t:\ 2^{k-1}t\le|x-ty|<2^kt\}.
\]
Put
\begin{equation}\label{eq:moving-annulus-tail-functional}
 M_t^\nu(y)=(1+|y|^2)^a\left\{
 \nu_c(A_{t,0}(y))+
 \sum_{k\ge1}2^{-k(d+1)}\nu_c(A_{t,k}(y))\right\}
\end{equation}
and
\begin{equation}\label{eq:moving-annulus-tail-energy}
 G_\nu(t)=t^4\int_{\RR^d}\Phi(M_t^\nu)\dd\Omega_d.
\end{equation}

\begin{corollary}[Moving-annulus raw-tail criterion]
\label{cor:moving-annulus-tail-criterion}
Under the hypotheses of
Theorem~\ref{thm:raw-tail-poisson-energy-decomposition},
\[
 t^4H_d(t)\longrightarrow\mathcal Q_d(\Sigma)
 \quad\Longleftrightarrow\quad
 \mathcal C_{\nu,\Sigma}(t)\longrightarrow0
 \quad\Longleftrightarrow\quad
 G_\nu(t)\longrightarrow0.
\]
This criterion applies without change to absolutely continuous, singular,
and purely atomic laws.
\end{corollary}

\begin{proof}
On \(A_{t,0}(y)\), the factor
\((1+|y-x/t|^2)^{-a}\) lies between \(2^{-a}\) and \(1\).  On
\(A_{t,k}(y)\), \(k\ge1\), it is comparable, with constants depending only
on \(d\), to \(2^{-k(d+1)}\).  Hence
\[
 c_dM_t^\nu(y)\le V_t^\nu(y)\le C_dM_t^\nu(y).
\]
For each fixed \(c>0\), the ratio \(\Phi(cv)/\Phi(v)\) has positive finite
limits \(c^2\) at zero and \(c\) at infinity and is continuous on
\((0,\infty)\).  Thus fixed multiplicative changes of the argument preserve
two-sided comparability of \(\Phi\).  The conclusion follows from
\eqref{eq:raw-tail-proxy-equivalence}.
\end{proof}

\begin{proposition}[Thin-shell aggregation precedes the entropy nonlinearity]
\label{prop:pre-phi-thin-shell-aggregation}
Fix \(d\ge4\), and set
\[
 p=\frac{4(d+1)}{d+5},\qquad q=\frac{d+5}{d+1}.
\]
Let \(B_n\to\infty\), choose \(R_n\to\infty\), and put
\(a_n=B_nR_n^{-p}\to0\) and
\(t_n=\kappa R_na_n^{1/(d+1)}\), where \(\kappa>0\) is fixed and
sufficiently small.  Let \(K_n\) be integers such that
\[
 B_n^qK_n^{1-q}\longrightarrow0.
\]
There are atoms \(\eta_{n,j}=(a_n/K_n)\delta_{r_{n,j}e_1}\),
\(1\le j\le K_n\), with
\(R_n\le r_{n,j}\le R_n+t_n/10\), for which
\begin{align}
 \sum_{j=1}^{K_n}\sup_{t>0}t^4
 \int\Phi(V_t^{\eta_{n,j}})\dd\Omega_d&\longrightarrow0,
 \label{eq:thin-shell-separated-energy}\\
 t_n^4\int\Phi\!\left(\sum_{j=1}^{K_n}
 V_{t_n}^{\eta_{n,j}}\right)\dd\Omega_d&\longrightarrow+\infty.
 \label{eq:thin-shell-aggregated-energy}
\end{align}
Thus summing shellwise nonlinear energies cannot replace aggregation of the
positive tail potential before applying \(\Phi\).
Here \(V_t^\eta\) denotes the expression on the right side of
\eqref{eq:raw-tail-poisson-potential} with the finite measure \(\eta\) in
place of \(\nu_c\).
\end{proposition}

\begin{proof}
The inequality \(\Phi(v)\le C_qv^q\), \(v\ge0\), and the exact translation
identity \eqref{eq:critical-lq-exact-translation-norm} show that an atom of
mass \(m\) at radius \(r\) satisfies
\[
 \sup_{t>0}t^4\int\Phi(V_t^{m\delta_{re_1}})\dd\Omega_d
 \le C_dr^4m^q.
\]
Indeed, the potential vanishes for \(t\ge r\); for \(t<r\), the right side
of \eqref{eq:critical-lq-exact-translation-norm} is bounded by
\(C_d\{1+(r/t)^4\}\).  Since \(r_{n,j}\asymp R_n\), summation gives
\[
 \sum_{j=1}^{K_n}\sup_{t>0}t^4
 \int\Phi(V_t^{\eta_{n,j}})\dd\Omega_d
 \le C_dR_n^4a_n^qK_n^{1-q}
 =C_dB_n^qK_n^{1-q},
\]
which proves \eqref{eq:thin-shell-separated-energy}.

At scale \(t_n\), all translated centres \(r_{n,j}e_1/t_n\) lie within
distance \(1/10\) of \(R_ne_1/t_n\).  On a fixed sufficiently small ball
about the latter point,
\[
 \sum_{j=1}^{K_n}V_{t_n}^{\eta_{n,j}}
 \ge c_da_n(R_n/t_n)^{d+1}=c_d\kappa^{-(d+1)}.
\]
The \(\Omega_d\)-measure of this ball is bounded below by
\(c_d(t_n/R_n)^{d+1}=c_d\kappa^{d+1}a_n\).  Therefore
\[
 t_n^4\int\Phi\!\left(\sum_jV_{t_n}^{\eta_{n,j}}\right)\dd\Omega_d
 \ge c_dt_n^4a_n
 =c_dR_n^4a_n^q=c_dB_n^q\longrightarrow\infty,
\]
after decreasing \(\kappa\), if necessary, so that the fixed lower value of
the potential is nonzero.  This proves
\eqref{eq:thin-shell-aggregated-energy}.
\end{proof}

\begin{corollary}[Proxy characterization at the critical vague boundary]
\label{cor:critical-rv-proxy-characterization}
Fix \(d\ge4\), set
\[
 p_d=\frac{4(d+1)}{d+5},
 \qquad
 \mu_\sigma(\dd\rho,\dd\theta)
 =p_d\rho^{-p_d-1}\dd\rho\,\sigma(\dd\theta),
\]
and let \(\nu\) have mean \(\bar\gamma\), finite covariance \(\Sigma\), and
centred law \(\nu_c\).  Suppose that, for a slowly varying \(L\),
\[
 \frac{r^{p_d}}{L(r)}\nu_c(r\,\mathord\cdot)
 \xrightarrow[r\to\infty]{v}\mu_\sigma
 \quad\text{on }\RR^d\setminus\{0\},
 \qquad
 \ell_L(t):=\int_{R_0}^t\frac{L(s)}s\dd s\longrightarrow\infty.
\]
Then
\[
 t^4H_d(t)\longrightarrow\mathcal Q_d(\Sigma)
 \quad\Longleftrightarrow\quad
 \mathcal C_{\nu,\Sigma}(t)\longrightarrow0.
\]
In particular, for the two laws \(\nu_{\rm ac}\) and \(\nu_{\rm cl}\) in
Theorem~\ref{thm:critical-vague-tail-nonidentifiability},
\[
 \mathcal C_{\nu_{\rm ac},\Sigma}(t)\longrightarrow0,
 \qquad
 \limsup_{t\to\infty}\mathcal C_{\nu_{\rm cl},\Sigma}(t)=+\infty.
\]
Thus the spectral measure and slowly varying factor do not determine the
covariance-proxy defect trajectory.
\end{corollary}

\begin{proof}
The equivalence is Theorem~\ref{thm:covariance-proxy-defect-decomposition};
the regular-variation assumptions specify the critical class but are not
needed for that decomposition.  Its combination with
\eqref{eq:critical-vague-tail-contrasting-entropy} gives the two final
assertions.
\end{proof}
