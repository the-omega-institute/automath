The finite-moment comparison theorem has a canonical extension at the
regularly varying boundary.  At that boundary the first unmatched moment is
not a Lebesgue moment; the signed difference of its truncated versions takes
its place, and relative entropy is quadratic in that difference.

\begin{lemma}[Critical \(L^2\) two-background transfer]
\label{lem:critical-l2-two-background-transfer}
Let \((E,\Omega)\) be a probability space.  Suppose that \(g_t\) is a
probability density with
\[
 g_t\ge c_t>0,\qquad c_t\longrightarrow1,
 \qquad \|g_t-1\|_{L^1(\Omega)}\longrightarrow0.
\]
Let \(B\in L^\infty(\Omega)\), let \(a_t\to0\), and suppose that
\[
 f_t=g_t+a_tB+e_t
\]
is a probability density with
\(\|e_t\|_{L^2(\Omega)}=o(|a_t|)\).  Then
\begin{equation}\label{eq:critical-l2-two-background-transfer}
 D_{\rm KL}(f_t\Omega\|g_t\Omega)
 =\frac{a_t^2}{2}\int_E B^2\dd\Omega+o(a_t^2).
\end{equation}
\end{lemma}

\begin{proof}
Put \(\Phi(s)=(1+s)\log(1+s)-s\) and
\(\Psi_g(z)=g\Phi(z/g)\).  The elementary convexity estimate
\begin{equation}\label{eq:critical-l2-bregman-bound}
 0\le \Psi_g(v+e)-\Psi_g(v)
       -\log(1+v/g)e\le Cg^{-1}e^2
\end{equation}
holds whenever \(g>0\), \(|v/g|\le1/4\), and \(g+v+e\ge0\).
Indeed, after scaling to \(g=1\), Taylor's theorem applies when
\(|e|\le1/2\); the admissible negative remainder is confined to a compact
set, while \((1+s)\log(1+s)=O(s^2)\) for positive \(s\).

For large \(t\), \(g_t\ge1/2\) and
\(\|a_tB/g_t\|_\infty\le1/4\).  Apply
\eqref{eq:critical-l2-bregman-bound} with \(v=a_tB\) and integrate.  Since
\[
 \|\log(1+a_tB/g_t)\|_2=O(|a_t|),
 \qquad \|e_t\|_2=o(|a_t|),
\]
H\"older's inequality gives
\[
 D_{\rm KL}(f_t\Omega\|g_t\Omega)
 =\int_E\Psi_{g_t}(a_tB)\dd\Omega+o(a_t^2).
\]
Uniform Taylor expansion at zero yields
\[
 \int_E\Psi_{g_t}(a_tB)\dd\Omega
 =\frac{a_t^2}{2}\int_E\frac{B^2}{g_t}\dd\Omega+O(|a_t|^3).
\]
Finally,
\[
 \left|\int_E B^2(g_t^{-1}-1)\dd\Omega\right|
 \le2\|B\|_\infty^2\|g_t-1\|_1\longrightarrow0,
\]
which proves \eqref{eq:critical-l2-two-background-transfer}.  The same
bounds also show that the relative entropy is finite for all sufficiently
large \(t\).
\end{proof}

\begin{theorem}[Two-law square law at the critical moment boundary]
\label{thm:two-law-critical-moment-square-law}
Let \(r\ge2\), and let \(\nu,\eta\) be probability laws on \(\RR\).  Suppose
that, for an eventually positive slowly varying function \(L\) and constants
\(c_{\lambda,+},c_{\lambda,-}\ge0\),
\begin{equation}\label{eq:two-law-critical-tail-hypotheses}
 \lambda((x,\infty))\sim c_{\lambda,+}x^{-r}L(x),
 \qquad
 \lambda(({-\infty},-x))\sim c_{\lambda,-}x^{-r}L(x),
 \qquad \lambda\in\{\nu,\eta\}.
\end{equation}
Constants equal to zero are allowed, with \(\sim0\) interpreted as
\(o(x^{-r}L(x))\).  Assume
\[
 m_j(\nu)=m_j(\eta)\quad(0\le j<r),
 \qquad
 \ell_L(t):=\int_{x_0}^t\frac{L(s)}s\dd s\longrightarrow\infty,
\]
and put
\begin{equation}\label{eq:two-law-critical-tail-imbalance}
 a_r:=c_{\nu,+}+(-1)^rc_{\nu,-}
      -c_{\eta,+}-(-1)^rc_{\eta,-}.
\end{equation}
If \(a_r\ne0\), then, with
\[
 M_{\lambda,r}(t):=
 \int_{|x|\le t}x^r\lambda(\dd x),
 \]
one has
\begin{align}
 M_{\nu,r}(t)-M_{\eta,r}(t)
 &\sim r a_r\ell_L(t),
 \label{eq:two-law-critical-truncated-moment}\\
 D_{\rm KL}(P_t*\nu\|P_t*\eta)
 &\sim C_r t^{-2r}
 \{M_{\nu,r}(t)-M_{\eta,r}(t)\}^2
 \label{eq:two-law-critical-square-law}\\
 &\sim C_r r^2a_r^2t^{-2r}\ell_L(t)^2,
 \notag
\end{align}
where
\[
 C_r=4^{-r}\binom{2r-2}{r-1}.
\]
Thus the critical coefficient depends on the signed tail imbalance: for
even \(r\) it compares total two-sided tail constants, whereas for odd
\(r\) it compares the right-minus-left tail constants.
\end{theorem}

\begin{proof}
The tail assumptions imply finite absolute moments of every order below
\(r\).  Karamata's theorem at the boundary gives
\[
 \int_{|x|\le t}x^r\lambda(\dd x)
 \sim r\{c_{\lambda,+}+(-1)^rc_{\lambda,-}\}\ell_L(t).
\]
Here the boundary terms are \(O(L(t))\), and
\(L(t)=o(\ell_L(t))\) because the latter integral diverges.  Subtraction
proves \eqref{eq:two-law-critical-truncated-moment}.

Write \(\Omega=\omega\),
\[
 R_z(y)=\frac{1+y^2}{1+(y-z)^2},
 \qquad
 q_t^\lambda(y)=\int_\RR R_{x/t}(y)\lambda(\dd x).
\]
We first establish the quotient expansion
\begin{equation}\label{eq:critical-two-law-quotient-expansion}
 q_t^\lambda
 =\sum_{j=0}^{r-1}m_j(\lambda)u_jt^{-j}
  +M_{\lambda,r}(t)u_rt^{-r}+e_t^\lambda,
 \qquad
 \|e_t^\lambda\|_{L^2(\Omega)}=O(t^{-r}L(t)).
\end{equation}
For \(|z|\le1\), the uniform Cayley remainder
\eqref{eq:one-dimensional-two-region-taylor-remainder} gives
\[
 \left\|R_z-\sum_{j=0}^ru_jz^j\right\|_2\le C_r|z|^{r+1}.
\]
For the exterior region, the pole expansion used in
Theorem~\ref{thm:first-cayley-mode-closure} gives
\[
 R_z(y)=1+\sum_{k\ge1}
 \bigl(\overline{b(z)}^{,k}\zeta^k+b(z)^k\zeta^{-k}\bigr),
 \qquad b(z)=\frac{\ii z}{2-\ii z},
\]
in the Cayley phase \(\zeta\).  Parseval therefore yields the exact identity
\[
 \|R_z-1\|_2^2
 =2\sum_{k\ge1}|b(z)|^{2k}=\frac{z^2}{2}.
\]
Since each \(u_j\) is bounded, for \(|z|>1\),
\begin{equation}\label{eq:critical-exterior-l2-jet-bound}
 \left\|R_z-\sum_{j=0}^{r-1}u_jz^j\right\|_2
 \le C_r|z|^{r-1}.
\end{equation}
Splitting the integral defining \(q_t^\lambda\) at \(|x|=t\), using the
order-\(r\) polynomial on the interior and the order-\((r-1)\) polynomial
on the exterior, gives
\[
 \|e_t^\lambda\|_2
 \le C_rt^{-(r+1)}
       \int_{|x|\le t}|x|^{r+1}\lambda(\dd x)
   +C_rt^{-(r-1)}
       \int_{|x|>t}|x|^{r-1}\lambda(\dd x).
\]
The two truncated-moment forms of Karamata's theorem bound the integrals by
\(O(tL(t))\) and \(O(t^{-1}L(t))\), respectively.  This proves
\eqref{eq:critical-two-law-quotient-expansion}.  In particular,
\[
 \|e_t^\nu-e_t^\eta\|_2
 =o(t^{-r}\ell_L(t)).
\]

The matched lower moments cancel in the difference of the two quotient
expansions.  Set
\[
 a_t=t^{-r}\{M_{\nu,r}(t)-M_{\eta,r}(t)\}.
\]
Then \(a_t\to0\), by slow variation, and
\[
 q_t^\nu-q_t^\eta=a_tu_r+o_{L^2(\Omega)}(|a_t|).
\]
It remains to verify the background hypotheses of
Lemma~\ref{lem:critical-l2-two-background-transfer}.  The score of the unit
Cauchy density has norm at most one, so Jensen's inequality gives
\[
 q_t^\eta(y)\ge\int e^{-|x|/t}\eta(\dd x)
 \ge e^{-\int|x|\eta(\dd x)/t}\longrightarrow1
\]
uniformly from below.  Translation continuity of \(P_1\) in \(L^1(\RR)\)
and dominated convergence give
\[
 \|q_t^\eta-1\|_{L^1(\Omega)}
 \le\int\|P_1(\mathord\cdot-x/t)-P_1\|_1\eta(\dd x)
 \longrightarrow0.
\]
Both quotients have \(\Omega\)-mass one.  The transfer lemma now gives
\[
 D_{\rm KL}(P_t*\nu\|P_t*\eta)
 =\frac{a_t^2}{2}\int_\RR u_r^2\dd\omega+o(a_t^2).
\]
The Parseval computation in
Theorem~\ref{thm:moment-matched-poisson-kl} identifies
\(\frac12\int u_r^2\dd\omega=C_r\).  Combining this identity with
\eqref{eq:two-law-critical-truncated-moment} proves both asymptotic
equivalences in \eqref{eq:two-law-critical-square-law}.
\end{proof}
