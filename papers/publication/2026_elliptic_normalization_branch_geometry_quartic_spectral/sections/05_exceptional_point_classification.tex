\section{Classification of real branch points and maximal-modulus roles}\label{sec:ep-classification}

Among the five finite branch values identified in Proposition~\ref{prop:elliptic-reduction}, exactly three are real.
By Lemma~\ref{lem:cyclic-companion}, every repeated root of \(\Pi(\cdot,y)\) corresponds to a defective eigenvalue of the companion family \(T(y)\).
This section is preparation for Theorem~\ref{thm:canonical-branch-dominance}: it determines the local collision structure at each real branch point and compares the modulus of the colliding eigenvalues against the remaining roots.
The assembled branch--dominance classification is cited canonically as Theorem~\ref{thm:canonical-branch-dominance}; the local statements below are its inputs, not competing theorem results.

The following elementary comparison is the only general local fact needed below:
a branch that leaves a repeated root at order \(t^{1/\nu}\) dominates any non-bifurcating simple branch whose modulus changes only at order \(t\).

\begin{lemma}[Local dominance from a fixed square-root splitting]\label{lem:local-dominance}
Let \(P(\lambda,t)=\prod_{j=1}^n(\lambda-\mu_j(t))\) be a monic degree-\(n\) polynomial whose coefficients are analytic in \(t\) near \(t=0\).
The conclusion is local on a prescribed real side of the base: throughout the statement take \(t>0\) sufficiently small and use the positive real square-root coordinate \(s=t^{1/2}>0\).
Suppose that at \(t=0\) exactly two roots collide:
\[
\mu_1(0)=\mu_2(0)=:\mu_*,
\qquad
\mu_3(0),\dots,\mu_n(0) \text{ simple and distinct from } \mu_*.
\]
Assume that for all \(j\ge 3\),
\begin{equation}\label{eq:local-dominance-hyp}
|\mu_j(0)|\le |\mu_*|.
\end{equation}
Assume that, after this square-root base change, one of the two bifurcating roots is represented by a holomorphic branch \(\mu_{1,\mathrm{sqrt}}(s)\), and write \(\mu_1(t)=\mu_{1,\mathrm{sqrt}}(t^{1/2})\) for the chosen positive-side branch.
Assume that this branch satisfies
\begin{equation}\label{eq:bifurc-form}
\abs{\mu_1(t)} \ge \abs{\mu_*} + a t^{1/2} - B t
\end{equation}
for some constants \(a,B>0\) and all real \(t>0\) sufficiently small on that side.
Then there exists \(\delta>0\) such that
\begin{equation}\label{eq:local-dominance-concl}
\abs{\mu_{1,\mathrm{sqrt}}(s)}>\abs{\mu_j(s^2)}
\qquad\text{for every } j\ge 3 \text{ and } 0<s<\delta^{1/2}.
\end{equation}
\end{lemma}

\begin{proof}
Since the non-colliding roots \(\mu_3(0),\dots,\mu_n(0)\) are simple, the implicit function theorem gives analytic continuations \(\mu_j(t)\) on a two-sided neighborhood of \(0\) and constants \(L_j>0\) such that
\[
\abs{\mu_j(t)-\mu_j(0)}\le L_j t
\qquad (j\ge 3,\ 0\le t\le t_0)
\]
for some \(t_0>0\).
\[
\abs{\mu_j(t)}
\le
\abs{\mu_j(0)}+L_j t
\le
\abs{\mu_*}+L_j t
=
\abs{\mu_*}+O(t)
\qquad (j\ge 3).
\]
On the other hand, the chosen square-root branch satisfies, with \(s=t^{1/2}>0\),
\[
\abs{\mu_{1,\mathrm{sqrt}}(s)}=\abs{\mu_1(t)}\ge \abs{\mu_*}+a t^{1/2}-Bt.
\]
Let \(L:=\max_{j\ge 3}L_j\).
Since \(t^{1/2}\gg t\) as \(t\to0^+\), choose \(0<\delta\le t_0\) so that
\[
Bt\le \frac{a}{2}t^{1/2},
\qquad
Lt\le \frac{a}{2}t^{1/2}
\qquad (0<t<\delta).
\]
Then for \(0<t<\delta\),
\[
\abs{\mu_{1,\mathrm{sqrt}}(s)}
\ge
\abs{\mu_*}+a t^{1/2}-Bt
\ge
\abs{\mu_*}+\frac{a}{2}t^{1/2}
>
\abs{\mu_*}+Lt
\ge
\abs{\mu_j(t)}
\]
for every \(j\ge 3\), proving \eqref{eq:local-dominance-concl} after writing \(t=s^2\).
\end{proof}

\begin{lemma}[Real-root count for the residual branch cubic]\label{lem:cubic-branch-real-root}
The cubic
\[
c(y)=256y^3+411y^2+165y+32
\]
has exactly one real root.
This root lies in \((-2,-1)\), and more sharply in
\begin{equation}\label{eq:ysub-certified-interval}
-\frac{2269}{2000}<y<-\frac{5672}{5000}.
\end{equation}
If this real root is denoted by \(\ySub\) and the two non-real roots by
\(\zeta_{+},\zeta_{-}\), then
\begin{equation}\label{eq:ysub-nonreal-modulus-bound}
\abs{\zeta_{+}}^2=
\zeta_{+}\zeta_{-}
=-
\frac{1}{8\ySub}
=\frac{1}{8\abs{\ySub}}
>
\frac1{16},
\qquad
\abs{\zeta_{+}}>\frac14 .
\end{equation}
\end{lemma}

\begin{proof}
For a cubic, the sign of the discriminant determines the number of real roots: a negative discriminant gives one real root and one non-real conjugate pair.
For
\[
c(y)=ay^3+by^2+cy+d,
\qquad
(a,b,c,d)=(256,411,165,32),
\]
the cubic discriminant
\[
\Delta=b^2c^2-4ac^3-4b^3d-27a^2d^2+18abcd
\]
is \(\Delta=-699868431<0\).
Thus \(c\) has exactly one real zero.
Finally,
\[
c(-2)=-702<0,
\qquad
c(-1)=22>0,
\]
so the intermediate value theorem places that unique real zero in \((-2,-1)\).
The refined interval follows from the rational evaluations
\[
c\!\left(-\frac{2269}{2000}\right)=-\frac{5301369}{500000000}<0,
\qquad
c\!\left(-\frac{5672}{5000}\right)=\frac{2804026}{244140625}>0,
\]
and uniqueness of the real zero.
Since the product of the three roots of \(c(y)\) is \(-32/256=-1/8\),
\(\ySub\zeta_{+}\zeta_{-}=-1/8\).  The interval \(-2<\ySub<-1\) gives
\(\abs{\ySub}<2\), and therefore \eqref{eq:ysub-nonreal-modulus-bound}.
\end{proof}

\begin{definition}[The subdominant real branch value]\label{def:ysub}
Throughout the paper, \(\ySub\) denotes the unique real root of
\begin{equation}\label{eq:ysub-definition}
256\ySub^3+411\ySub^2+165\ySub+32=0,
\qquad
-\frac{2269}{2000}<\ySub<-\frac{5672}{5000}.
\end{equation}
Equivalently, \(\ySub\in(-1.1345,-1.1344)\).  This is the negative real branch value that later occurs in the real oval images and in the real-root interval diagram.
\end{definition}

\begin{lemma}[Exact subdominance criterion at \(\ySub\)]\label{lem:ysub-exact-subdominance}
Let \(\alpha\) be the unique real root of
\[
h(\lambda)=16\lambda^3-9\lambda^2+1.
\]
Then \(-11/40<\alpha<-27/100\), and the real branch value \(\ySub\) is
\begin{equation}\label{eq:ysub-alpha-formula}
\ySub=\frac{4\alpha^3-3\alpha^2-2\alpha+1}{4\alpha}.
\end{equation}
At \(y=\ySub\) the spectral polynomial has the exact factorization in \(\QQ(\alpha)[\lambda]\)
\begin{equation}\label{eq:ysub-factorization}
\Pi(\lambda,\ySub)
=(\lambda-\alpha)^2
\left(
\lambda^2+\frac{(\alpha-1)(\alpha+1)}{8\alpha^2}\lambda
+\frac{9\alpha^2-144\alpha-1}{256\alpha^2}
\right).
\end{equation}
The quadratic factor is irreducible over \(\RR\), and if its two roots are \(z_{+},z_{-}\), then
\begin{equation}\label{eq:ysub-modulus-gap}
|z_{+}|^2
=\frac{9\alpha^2-144\alpha-1}{256\alpha^2}
>\alpha^2.
\end{equation}
Consequently the colliding double root at \(y=\ySub\) is strictly subdominant.  This supplies the \(\ySub\)-row of the branch-value real-root table used in Theorem~\ref{thm:theorem-b-real-projection-criterion} and in the positive-axis dominance theorem.
\end{lemma}

\begin{proof}
The derivative \(h'(\lambda)=3\lambda(16\lambda-6)\) has critical points \(0\) and \(3/8\).
Since \(h(0)=1\), \(h(3/8)=37/64>0\), and \(h(\lambda)\to-\infty\) as \(\lambda\to-\infty\), the cubic \(h\) has exactly one real root.
The rational evaluations
\[
h\!\left(-\frac{11}{40}\right)=-\frac{107}{8000}<0,
\qquad
h\!\left(-\frac{27}{100}\right)=\frac{7243}{250000}>0
\]
therefore isolate it in \((-11/40,-27/100)\).
The critical-value formula from the elliptic projection gives \eqref{eq:ysub-alpha-formula}, and substituting \eqref{eq:ysub-alpha-formula} into \(c(y)\) reduces its numerator to zero modulo \(h(\alpha)\); since \(\alpha\ne0\), this identifies the value with the unique real root \(\ySub\) of \(c\).

A direct division in the cubic field gives \eqref{eq:ysub-factorization}.
Equivalently, after multiplying by \(16\alpha^2\), the quotient of \(\Pi(\lambda,\ySub)\) by \((\lambda-\alpha)^2\) has coefficients
\[
16\alpha^2,
\qquad
2(\alpha-1)(\alpha+1),
\qquad
\frac{9\alpha^2-144\alpha-1}{16},
\]
and the remainder is zero in \(\QQ[\alpha]/(h)\).
The discriminant of the displayed quadratic is
\[
\frac{(\alpha-1)^2(\alpha+1)^2}{64\alpha^4}
-\frac{9\alpha^2-144\alpha-1}{64\alpha^2}
=\frac{2479\alpha^2+16\alpha-247}{8(81\alpha^2-16\alpha-9)}.
\]
On the isolating interval \((-11/40,-27/100)\), the numerator is negative and the denominator is positive; indeed
\[
2479\left(-\frac{11}{40}\right)^2+16\left(-\frac{11}{40}\right)-247=-\frac{102281}{1600}<0,
\]
while
\[
81\left(-\frac{27}{100}\right)^2-16\left(-\frac{27}{100}\right)-9=\frac{12249}{10000}>0.
\]
The derivative of the numerator, \(4958\lambda+16\), is negative throughout the interval, so the numerator stays negative after its left-endpoint check.  The denominator is the upward-opening quadratic \(81\lambda^2-16\lambda-9\), and its positive zero is larger than \(1/3\); hence the denominator is positive throughout \((-11/40,-27/100)\).
Thus the quadratic has negative discriminant and its two roots are a non-real conjugate pair.
Their squared modulus is the constant term of the quadratic, and
\[
|z_{+}|^2-\alpha^2
=\frac{9\alpha^2-144\alpha-1}{256\alpha^2}-\alpha^2
=-\frac{9\alpha^2+16\alpha-1}{32\alpha^2}.
\]
The numerator \(9\lambda^2+16\lambda-1\) has positive derivative on \((-11/40,-27/100)\).
Its value at \(-27/100\) is \(-46639/10000<0\), hence it is negative at \(\alpha\).
Since \(32\alpha^2>0\), the last display is positive.
Therefore the two non-real simple roots have modulus strictly larger than \(|\alpha|\), proving strict subdominance of the double root.
\end{proof}

\end{proof}

\begin{lemma}[Canonical real-oval covering count]\label{lem:canonical-real-oval-covering-count}
On the compact real oval \(E_{\mathrm c}(\RR)\), the only finite critical points of the projection \(y:E\to\mathbf P^1_y\) are the points with ramification abscissae
\[
\lambda=-1,
\qquad
\lambda=\alpha,
\]
where \(\alpha\in(-1/2,0)\) is the real root of \(16\lambda^3-9\lambda^2+1\).  Their critical values are respectively
\[
y(-1)=1,
\qquad
y(\alpha)=\ySub.
\]
Consequently
\[
y:E_{\mathrm c}(\RR)\longrightarrow[\ySub,1]
\]
is two-to-one over every value in \((\ySub,1)\).

On the noncompact real oval, compactified by the point at infinity, the only finite critical point is \(\lambda=1\), with value \(0\).  Along both real affine branches approaching infinity one has \(y\to+\infty\).  Consequently
\[
y:E_{\mathrm n}(\RR)\longrightarrow[0,\infty)
\]
is two-to-one over every value in \((0,\infty)\).
\end{lemma}
\begin{proof}
Proposition~\ref{cor:quartic-real-projection-framework} identifies the compact oval with the part of the Weierstrass model over \(J_{\mathrm c}=[\beta_1,\beta_2]\), the noncompact oval with the part over \(J_{\mathrm n}=[\beta_3,\infty)\) completed by infinity, and it locates the real finite ramification abscissae.  The order
\[
\beta_1<-1<\alpha<\beta_2<\beta_3<1
\]
places \(-1\) and \(\alpha\) on the compact oval and \(1\) on the noncompact oval.  The same proposition computes their values as \(1\), \(\ySub\), and \(0\), respectively.

Remove the two critical points from the compact oval.  The remaining real circle is the disjoint union of two open arcs.  On each arc \(dy\ne0\), so the real inverse-function theorem makes the projection a local diffeomorphism and the function is strictly monotone.  The two endpoints of each arc are the critical points with values \(\ySub\) and \(1\); hence each arc maps bijectively onto \((\ySub,1)\).  This gives exactly two compact-oval preimages for every interior regular value.

For the noncompact oval, remove the finite critical point \(\lambda=1\) and the point at infinity.  The compactified real oval is again a real circle, and the complement is two open affine arcs.  On both arcs \(dy\ne0\), so the projection is strictly monotone.  At the finite endpoint the value is \(0\).  At the point at infinity the local parameter computation in Proposition~\ref{cor:quartic-real-projection-framework} gives
\[
\lambda=t^{-2},
\qquad
\WeierY=2t^{-3}(1+O(t^2)),
\qquad
y=t^{-4}+O(t^{-3})\to+\infty
\]
on both real branches.  Therefore each noncompact arc maps bijectively onto \((0,\infty)\), giving exactly two noncompact-oval preimages for every interior regular value.
\end{proof}

\begin{theorem}[Canonical real projection]\label{thm:theorem-b-real-projection-criterion}
For the canonical quartic
\[
\Pi(\lambda,y)=\lambda^4-\lambda^3-(2y+1)\lambda^2+\lambda+y(y+1)
\]
and its elliptic normalization
\[
E:\quad \WeierY^2=4\lambda^3-4\lambda+1,
\qquad
y=\lambda^2-\frac12+\frac12\WeierY,
\]
let \(\beta_1<\beta_2<\beta_3\) be the three real roots of
\[
F(\lambda)=4\lambda^3-4\lambda+1,
\]
and put
\[
J_{\mathrm c}=[\beta_1,\beta_2],
\qquad
J_{\mathrm n}=[\beta_3,\infty).
\]
Let \(\ySub\) denote the unique real zero of
\[
256y^3+411y^2+165y+32.
\]
Then the real projection of the normalized canonical quartic has the following form.
\begin{enumerate}
\item The real locus has two components, denoted \(E_{\mathrm c}(\RR)\) and \(E_{\mathrm n}(\RR)\), lying above \(J_{\mathrm c}\) and \(J_{\mathrm n}\), respectively.
\item The complete finite transition framework is
\[
\mathcal V(J_{\mathrm c})=
\left\{\beta_1^2-\frac12,\ \beta_2^2-\frac12,\ 1,\ \ySub\right\},
\qquad
\mathcal V(J_{\mathrm n})=
\left\{\beta_3^2-\frac12,\ 0\right\},
\]
with extrema
\[
\min \mathcal V(J_{\mathrm c})=\ySub,
\qquad
\max \mathcal V(J_{\mathrm c})=1,
\qquad
\min \mathcal V(J_{\mathrm n})=0.
\]
The compact and noncompact component images are therefore
\[
y(E_{\mathrm c}(\RR))=[\ySub,1],
\qquad
y(E_{\mathrm n}(\RR))=[0,\infty).
\]
\item Consequently the canonical member has genuine overlap in the auxiliary fixed-member test: since $M_{\mathrm c}=1$ and $m_{\mathrm n}=0$, the two component images overlap nontrivially rather than merely touching or separating.  The preceding lemma proves the two-sheeted covering degrees on the interiors of the two component images, so the table below is a fibre count and not just an extrema list.  The total real image is
\[
y(E(\RR))=[\ySub,\infty),
\]
with overlap exactly on \([0,1]\).  For every regular real value of \(y\), the source of each real root is fixed by the table
\[
\begin{array}{c|c|c|c}
\text{range of }y & \#E_{\mathrm c} & \#E_{\mathrm n} & \#\Pi_y(\RR)\\
\hline
y<\ySub & 0 & 0 & 0\\
\ySub<y<0 & 2 & 0 & 2\\
0<y<1 & 2 & 2 & 4\\
y>1 & 0 & 2 & 2.
\end{array}
\]
At the three real branch values the singular-fibre factorization and count are included explicitly, so the count below is not merely the regular-fibre count continued across branch values:
\[
\begin{array}{c|c|c|c}
y & \text{factorization data} & \#\{\text{distinct real roots}\} & \#\{\text{real roots with multiplicity}\}\\
\hline
\ySub & (\lambda-\alpha)^2Q(\lambda),\ Q\text{ real irreducible} & 1 & 2\\
0 & \lambda(\lambda+1)(\lambda-1)^2 & 3 & 4\\
1 & (\lambda-2)(\lambda-1)(\lambda+1)^2 & 3 & 4.
\end{array}
\]
The component decomposition behind this count is recorded in Corollary~\ref{cor:real-root-interval-diagram}.
\item Equivalently, for regular real values, the canonical quartic has real \(\lambda\)-roots if and only if
\[
y>\ySub;
\]
the singular fibre at \(y=\ySub\) is the double-root fibre described in the preceding table.  For regular fibres, the number of real roots is the number in the last column of the displayed regular-value table.  Thus the only algebraic obstruction to real solvability below the regular branch levels is the single endpoint value \(\ySub\), while the only transition values in the real image are \(\ySub,0,1\).
\item The transition set \(\{\ySub,0,1\}\) is minimal for the canonical real-root interval diagram.  On each connected component of
\[
\RR\setminus\{\ySub,0,1\}
\]
both the number of real roots and the contributing real oval or ovals are constant; deleting any one of the three transition values merges two adjacent intervals with different regular real-root counts.  The displayed singular-fibre table records what happens at the transition values themselves, so the theorem gives the exact distinct and multiplicity-counted real-root count for every real \(y\), not only for regular fibres.
\end{enumerate}
\end{theorem}

\begin{remark}[Scope of the canonical real projection theorem]\label{rem:theorem-b-scope}
Theorem~\ref{thm:theorem-b-real-projection-criterion} is the canonical statement for the single member \(P_{-1,-1,1,0}\).  The auxiliary fixed-member lemma used in its proof is only a finite extrema test for that already selected member; it is not a chamber classification for varying balanced quartics, and no recurrence-theoretic or dominance input enters its proof.
\end{remark}

\begin{proof}
The polynomial discriminant of \(F(\lambda)=4\lambda^3-4\lambda+1\) is \({\rm Disc}(F)=592=16\cdot37>0\); equivalently the associated Weierstrass discriminant convention gives \(g_2^3-27g_3^2=37\).  Thus \(F\) has three distinct real roots.  Lemma~\ref{prop:balanced-real-projection-fixed-extrema}, applied only to this fixed member, gives two real components lying above the bounded and unbounded intervals \(J_{\mathrm c}\) and \(J_{\mathrm n}\), and it reduces the real-image and fibre-count problem to the finite sets \(\mathcal V(J_{\mathrm c})\) and \(\mathcal V(J_{\mathrm n})\).  Proposition~\ref{cor:quartic-real-projection-framework} computes the ramification values \(1\), \(0\), and \(\ySub\), while Lemma~\ref{lem:beta-endpoint-value-certificates} orders the endpoint values \(\beta_i^2-1/2\) relative to \(\ySub,0,1\).  These finite certificates give the two displayed sets and no exceptional \(\lambda=0\) contribution occurs for this specialization.  Hence the same proposition gives
\[
y(E_{\mathrm c}(\RR))=[\ySub,1],
\qquad
y(E_{\mathrm n}(\RR))=[0,\infty).
\]
Their union is \([\ySub,\infty)\), and their intersection is \([0,1]\).  In the notation of the fixed-member interval alternative, $M_{\mathrm c}=1$ and $m_{\mathrm n}=0$, so $M_{\mathrm c}>m_{\mathrm n}$.  Hence the canonical real image is the nondegenerate one-interval case, not the adjacent-touching case $M_{\mathrm c}=m_{\mathrm n}$ and not the two-disjoint-interval case $M_{\mathrm c}<m_{\mathrm n}$.  The only remaining endpoint degeneracies are the actual ramified endpoint levels $\ySub$, $0$, and $1$, proving the asserted finite real-solvability criterion and the transition-value list.
Let now \(y\) be a regular real value.  Lemma~\ref{lem:canonical-real-oval-covering-count} gives the covering degrees on the two real ovals by monotonicity on the complementary arcs: the compact oval contributes exactly two points for \(\ySub<y<1\) and none outside its image, while the noncompact oval contributes exactly two points for \(y>0\) and none outside its image.  Combining these two two-sheeted coverings over the intervals \([\ySub,1]\) and \([0,\infty)\) gives the four rows of the table.
The equality between real points on the normalization and real roots of \(\Pi(\cdot,y)\) follows from the normalization map away from branch values, where all roots are simple.
The four regular cells have real-root counts \(0,2,4,2\) in increasing order.  Hence crossing \(\ySub\) changes the count from \(0\) to \(2\), crossing \(0\) changes it from \(2\) to \(4\), and crossing \(1\) changes it from \(4\) to \(2\).  Since no transition can occur off the finite obstruction framework supplied by Lemma~\ref{prop:balanced-real-projection-fixed-extrema} for this member and evaluated in Proposition~\ref{cor:quartic-real-projection-framework}, the set \(\{\ySub,0,1\}\) is complete; since each of its three points separates adjacent cells with different counts, it is minimal for this real-root interval diagram.
At the transition values themselves the displayed branch-value table gives the count directly.  The factors at \(0\) and \(1\) show three distinct real roots and total real multiplicity four.  At \(\ySub\), Lemma~\ref{lem:ysub-exact-subdominance} gives one double real root \(\alpha\) and a residual non-real conjugate pair, so there is one distinct real root and real multiplicity two.
Corollary~\ref{cor:real-root-interval-diagram} retains the component source of the calculation.  Thus each clause follows from the auxiliary fixed-member test applied only to the canonical member and the finite canonical comparison supplied by Proposition~\ref{cor:quartic-real-projection-framework}.
\end{proof}

\begin{proposition}[Canonical real-solvability obstruction]\label{prop:canonical-real-solvability-obstruction}
For a real number \(y\) which is not one of \(\ySub,0,1\), the following are equivalent:
\begin{enumerate}
\item the polynomial \(\Pi(\lambda,y)\) has a real root;
\item the normalized real fibre \(E(\RR)\cap y^{-1}(y)\) is nonempty;
\item \(y>\ySub\).
\end{enumerate}
Moreover the number of real roots is \(2\) on \((\ySub,0)\), \(4\) on \((0,1)\), and \(2\) on \((1,\infty)\), while no real root occurs on \((-\infty,\ySub)\).
\end{proposition}

\begin{proof}
The normalization map in Proposition~\ref{prop:elliptic-reduction} identifies the affine normalized curve with \(\Pi(\lambda,y)=0\), and away from the branch values \(\ySub,0,1\) the fibre consists of simple points.  Hence real roots of \(\Pi(\cdot,y)\) are exactly real points of the normalized fibre.  Theorem~\ref{thm:theorem-b-real-projection-criterion} gives the total real image \([\ySub,\infty)\) and the regular-fibre table, which proves the equivalence and the three root counts.
\end{proof}

\begin{proposition}[Real roots at the real branch values]\label{prop:branch-value-real-root-counts}
The distinct and multiplicity-counted real roots of \(\Pi(\cdot,y)\) at the three real branch values are as follows; this is the local branch-value supplement to the regular-fibre count in Theorem~\ref{thm:theorem-b-real-projection-criterion}.
At \(y=0\),
\[
\Pi(\lambda,0)=\lambda(\lambda+1)(\lambda-1)^2,
\]
so the distinct real roots are \(-1,0,1\), with \(\lambda=1\) double.
At \(y=1\),
\[
\Pi(\lambda,1)=(\lambda-2)(\lambda-1)(\lambda+1)^2,
\]
so the distinct real roots are \(-1,1,2\), with \(\lambda=-1\) double.
At \(y=\ySub\),
\[
\Pi(\lambda,\ySub)=(\lambda-\alpha)^2 Q(\lambda),
\]
where
\[
Q(\lambda)=
\lambda^2+\frac{(\alpha-1)(\alpha+1)}{8\alpha^2}\lambda
+\frac{9\alpha^2-144\alpha-1}{256\alpha^2}
\]
is the real irreducible quadratic of Lemma~\ref{lem:ysub-exact-subdominance}; hence the only distinct real root is \(\alpha\), with multiplicity two, and the remaining two roots are a nonreal conjugate pair.
Equivalently,
\[
\begin{array}{c|c|c|c}
\text{branch value} & \text{distinct real roots} & \text{multiple real root} & \text{real roots with multiplicity}\\
\hline
\ySub & \alpha & \alpha\text{ double} & 2\\
0 & -1,0,1 & 1\text{ double} & 4\\
1 & -1,1,2 & -1\text{ double} & 4.
\end{array}
\]
\end{proposition}

\begin{proof}
The two factorizations at \(0\) and \(1\) are direct substitutions in \(\Pi\).
The factorization at \(\ySub\) and the irreducibility of \(Q\) are exactly Lemma~\ref{lem:ysub-exact-subdominance}.
Counting distinct real roots and then counting the real linear factors with multiplicity gives the table.
\end{proof}

\begin{proposition}[Real branch points and local spectral roles]\label{prop:ep-classification}
The three real branch values are
\[
\ySub < 0, \qquad y_0 = 0, \qquad y_1 = 1,
\]
where \(\ySub\) is the algebraic number fixed in Definition~\ref{def:ysub}.
Their local spectral roles, used later as inputs to the canonical Theorem~\ref{thm:canonical-branch-dominance}, are as follows.
\begin{enumerate}
\item At $y = 0$, the spectral polynomial factors as
\begin{equation}\label{eq:endpoint-factor-y0}
\Pi(\lambda, 0) = \lambda(\lambda + 1)(\lambda - 1)^2.
\end{equation}
The double root $\lambda=1$ ties in modulus with the simple root $\lambda=-1$.
Nevertheless, for $y>0$ sufficiently small, the pair of branches issuing from $\lambda=1$ contains the unique maximal-modulus root.
Thus $y=0$ is the endpoint of the maximal-modulus branch over $y>0$.

\item At $y = 1$, the spectral polynomial factors as
\begin{equation}\label{eq:endpoint-factor-y1}
\Pi(\lambda, 1) = (\lambda - 2)(\lambda - 1)(\lambda + 1)^2.
\end{equation}
The double root $\lambda = -1$ is not the maximal-modulus eigenvalue; the positive real root $\lambda = 2$ remains simple.
Hence $y = 1$ is not a branch point of the maximal-modulus branch over $y>0$.

\item At \(y=\ySub\), the colliding double root is \emph{strictly subdominant}: it is exceeded in modulus by a pair of complex-conjugate roots.
Therefore $\ySub$ is a subdominant branch point, not on the maximal-modulus envelope.
\end{enumerate}
\end{proposition}

\begin{proof}
Lemma~\ref{lem:cubic-branch-real-root} shows that the residual branch cubic \(c(y)\) contributes exactly one real branch value, and that it is negative.
Together with the two real branch values \(0\) and \(1\) from Proposition~\ref{prop:elliptic-reduction}, this gives precisely the three real branch values displayed in the proposition.

We begin with the local dominance claim in part~(1).

\medskip
\noindent\textbf{Continuity of roots at $y=0$.}
At $y=0$ the factorization is $\Pi(\lambda,0)=\lambda(\lambda+1)(\lambda-1)^2$.
The three simple roots $\lambda\in\{0,-1\}$ extend analytically in $y$ by the implicit function theorem, since $\partial_\lambda\Pi$ is nonzero at each of them.
The double root at $\lambda=1$ produces a square-root ramification as described next.

\medskip
\noindent\textbf{Self-contained local expansion.}
Write $y=s^2$ for real $s>0$ small.
Set $\lambda=1+h$ directly in $\Pi(\lambda,y)=\lambda^4-\lambda^3-(2y+1)\lambda^2+\lambda+y(y+1)$ and collect terms:
\begin{align*}
\Pi(1+h,y)
&= (1+4h+6h^2+4h^3+h^4) - (1+3h+3h^2+h^3)\\
&\quad - (1+2y)(1+2h+h^2) + (1+h) + y^2+y\\
&= 2h^2 - y - 4yh + O(y^2, h^3, yh^2).
\end{align*}
Setting $y=s^2$ and seeking $h=\sigma s+O(s^2)$, the leading terms give $2\sigma^2 s^2 - s^2 = 0$, hence $\sigma^2=\tfrac12$, i.e.\ $\sigma=\pm\tfrac{1}{\sqrt{2}}$.
The correct square-root opening is confirmed by evaluating the quadratic discriminant:
\[
\frac{\partial^2\Pi}{\partial\lambda^2}\bigg|_{(1,0)} = 2\cdot\frac{\partial}{\partial\lambda}(4\lambda^3-3\lambda^2-(2y+1)\cdot 2\lambda+1)\bigg|_{(1,0)}=2(12-6-4)=4,
\]
\[
\frac{\partial\Pi}{\partial y}\bigg|_{(1,0)} = (-2\lambda^2+2y+1)\bigg|_{(1,0)}=-2+1=-1.
\]
By the Puiseux implicit function theorem for a double root (see, e.g., \cite{BrieskornKnorrer1986}), and with the same positive real square-root coordinate \(s=y^{1/2}>0\) used in Lemma~\ref{lem:local-dominance}, the two branches issuing from $(1,0)$ satisfy
\begin{equation}\label{eq:local-bifurc}
\lambda_\pm(y)=1\pm\frac{1}{\sqrt{2}}\,y^{1/2}+O(y).
\end{equation}
The coefficient is
\(\bigl(-\partial_y\Pi/(\tfrac12\partial_\lambda^2\Pi)\bigr)^{1/2}_{(1,0)}=1/\sqrt2\).
In particular, setting $y=s^2$,
\begin{equation}\label{eq:lambda-pm-local}
\lambda_+(s) = 1 + \frac{1}{\sqrt{2}}\,s + O(s^2),
\qquad
\lambda_-(s) = 1 - \frac{1}{\sqrt{2}}\,s + O(s^2).
\end{equation}
These labels are the local positive-side labels only; in the endpoint zero theorem the same analytic branches are continued to the negative endpoint scale by taking \(s=iu/m\), as stated in Theorem~\ref{thm:endpoint-zero-quantization}.

\medskip
\noindent\textbf{Spectral-radius comparison.}
For the remaining two roots near $(-1,0)$ and $(0,0)$, the implicit function theorem gives simple analytic extensions:
\begin{equation}\label{eq:other-roots-local}
\lambda_{-1}(y) = -1 + O(y),
\qquad
\lambda_0(y) = O(y).
\end{equation}
More precisely, differentiating $\Pi(\lambda,y)=\lambda^4-\lambda^3-(2y+1)\lambda^2+\lambda+y(y+1)$ with respect to $\lambda$:
\[
\partial_\lambda\Pi(\lambda,y) = 4\lambda^3-3\lambda^2-2(2y+1)\lambda+1.
\]
At $(\lambda,y)=(-1,0)$: $\partial_\lambda\Pi(-1,0) = -4-3+2+1=-4\neq 0$, so the implicit function theorem applies.
For the $y$-derivative: $\partial_y\Pi(\lambda,y)=-2\lambda^2+2y+1$, giving at $(-1,0)$: $\partial_y\Pi(-1,0)=-2+1=-1$.
Hence
\[
\frac{d\lambda_{-1}}{dy}\bigg|_{y=0} = -\frac{\partial_y\Pi(-1,0)}{\partial_\lambda\Pi(-1,0)} = -\frac{-1}{-4} = -\frac{1}{4},
\]
and
\[
\lambda_{-1}(y) = -1 - \frac{1}{4}y + O(y^2).
\]
For the root near $\lambda=0$: $\partial_\lambda\Pi(0,0)=1$ and $\partial_y\Pi(0,0)=1$, so
\[
\lambda_0(y) = -y+O(y^2).
\]
Therefore $|\lambda_{-1}(s^2)|=1+\tfrac{1}{4}s^2+O(s^4)>1$ for $s>0$ small, while $|\lambda_+(s)|=1+\tfrac{1}{\sqrt{2}}s+O(s^2)>1$, and $|\lambda_0(s^2)|=s^2+O(s^4)\to 0$ for $s\to 0^+$.

The critical comparison is $|\lambda_+(s)|$ vs $|\lambda_{-1}(s^2)|$: the bifurcating root $\lambda_+$ departs from $1$ at rate $\tfrac{1}{\sqrt{2}}s$ (first order in $s$), while the non-bifurcating root $\lambda_{-1}$ departs from $-1$ (hence $|\lambda_{-1}|$ departs from $1$) at rate $\tfrac{1}{4}s^2$ (second order in $s$):
\[
|\lambda_+(s)| - |\lambda_{-1}(s^2)| = \frac{1}{\sqrt{2}}s - \frac{1}{4}s^2 + O(s^3) > 0
\]
for $0<s<4/\sqrt{2}=2\sqrt{2}$.
For $|\lambda_-|$ and $|\lambda_0|$: $|\lambda_-(s)|=1-\tfrac{1}{\sqrt{2}}s+O(s^2)<1$ and $|\lambda_0(s^2)|=s^2+O(s^4)\to 0$.
Hence, for $s>0$ sufficiently small,
\begin{equation}\label{eq:dominance-ineqs}
\abs{\lambda_+(s)}
>
\abs{\lambda_{-1}(s^2)},
\qquad
\abs{\lambda_+(s)}
>
\abs{\lambda_-(s)},
\qquad
\abs{\lambda_+(s)}
>
\abs{\lambda_0(s^2)}.
\end{equation}
Thus the branch $\lambda_+(y^{1/2})$ is the unique maximal-modulus root for $y>0$ small, proving part~(1).

Part~(2) is immediate from
\[
\Pi(\lambda,1)=(\lambda-2)(\lambda-1)(\lambda+1)^2,
\]
since the double root has modulus \(1\) while the simple root \(\lambda=2\) has modulus \(2\).
Part~(3) is exactly Lemma~\ref{lem:ysub-exact-subdominance}: the ramification abscissa \(\alpha\) maps to \(\ySub\), the residual quadratic at \(y=\ySub\) has a non-real conjugate pair, and \eqref{eq:ysub-modulus-gap} puts that pair at strictly larger modulus than the colliding root \(\alpha\).
\end{proof}

\begin{remark}\label{rem:ysub-local-gap}
The strict inequality $\sqrt{q}>|\alpha|$ gives a positive modulus gap at \(y=\ySub\).
By continuity of the roots of \(\Pi(\lambda,y)\) with respect to \(y\), there exists a neighborhood \(U\) of \(\ySub\) such that, for \(y\in U\setminus\{\ySub\}\), the two branches colliding at \(\ySub\) remain strictly below the maximal-modulus modulus.
Thus \(\ySub\) does not enter the maximal-modulus envelope even locally.
\end{remark}

\begin{remark}\label{rem:maximal-modulus-endpoint}
In the spectral decomposition of the polynomial sequence, the limiting zero-accumulation curve is governed by equimodular competition among maximal-modulus eigenvalues, not by the discriminant locus alone.
A branch point belongs to the maximal-modulus spectral boundary only when the colliding eigenvalue is maximal-modulus.
Proposition~\ref{prop:ep-classification} shows that $\ySub$ fails this dominance test: the colliding eigenvalue pair at $\lambda = \alpha$ is exceeded in modulus by the complex-conjugate pair of modulus $\sqrt{q}$.
Thus $\ySub$ is a branch point of the discriminant curve that does not enter the maximal-modulus envelope.
\end{remark}

\begin{remark}\label{rem:yang-lee-distinction}
The terminology of endpoint zeros is used here in the spectral sense of a maximal-modulus branch boundary.
The classical Yang--Lee and Fisher settings \cite{YangLee1952I,LeeYang1952II,Fisher1978,KortmanGriffiths1971} provide historical context for zero accumulation, but the present use of the term depends only on the quartic spectral curve.
Thus the point $y=0$ is a square-root endpoint of the maximal-modulus boundary for this quartic family.
\end{remark}

\subsection{Real ovals and the exact real-root interval diagram}

The elliptic normalization also determines the complete real-root count of the quartic transfer polynomial.
The next result is a preparatory real-oval corollary; the definitive branch-value, real-root, and dominance classification is Theorem~\ref{thm:canonical-branch-dominance}.

\begin{corollary}[Real-oval interval diagram]\label{cor:real-root-interval-diagram}
Let $\beta_1 < \beta_2 < \beta_3$ be the three real roots of
\[
4\lambda^3 - 4\lambda + 1 = 0.
\]
Then the real locus of the elliptic curve
\[
E:\quad \WeierY^2 = 4\lambda^3 - 4\lambda + 1
\]
decomposes as
\[
E(\RR)=E_{\mathrm{c}}(\RR)\sqcup E_{\mathrm{n}}(\RR),
\]
where
\[
E_{\mathrm{c}}(\RR)
=\{(\lambda,\WeierY)\in\RR^2:\beta_1\le \lambda\le \beta_2,\ \WeierY^2=4\lambda^3-4\lambda+1\}
\]
is a compact oval and
\[
E_{\mathrm{n}}(\RR)
=\{(\lambda,\WeierY)\in\RR^2:\lambda\ge \beta_3,\ \WeierY^2=4\lambda^3-4\lambda+1\}\cup\{O\}
\]
is the noncompact oval completed by the point at infinity $O$.
Under the degree-$4$ \(y\)-projection
\[
y=\lambda^2-\frac12+\frac12\WeierY,
\]
their images are exactly
\begin{equation}\label{eq:real-oval-images}
y\bigl(E_{\mathrm{c}}(\RR)\bigr)=[\ySub,1],
\qquad
y\bigl(E_{\mathrm{n}}(\RR)\bigr)=[0,\infty).
\end{equation}
Consequently the real-root count for regular real values is obtained by applying the two-to-one covering property on the interiors of these two intervals.
The definitive table is Theorem~\ref{thm:theorem-b-real-projection-criterion}, and the same table is incorporated into the final branch--dominance result in Theorem~\ref{thm:canonical-branch-dominance}.
\end{corollary}

\begin{proof}
Proposition~\ref{cor:quartic-real-projection-framework} supplies the complete oval computation used here: the isolations
\[
\beta_1<-1<\alpha<\beta_2<\beta_3<1,
\]
the assignment of the real ramification abscissae $-1$ and $\alpha$ to the compact oval and $1$ to the noncompact oval, the corresponding critical values $1$, $\ySub$, and $0$, and the pole behavior $y\to+\infty$ at the point at infinity on the noncompact component.
It follows immediately that the two real components are the compact and noncompact ovals displayed in the statement and that their images are exactly \eqref{eq:real-oval-images}.

For real $y$ away from the branch locus, each point of $y^{-1}(y)\cap E(\RR)$ is simple and contributes a distinct real root $\lambda$ of $\Pi(\lambda,y)=0$.
Moreover, Lemma~\ref{lem:real-weierstrass-oval-covering-count} gives the regular-value covering count on each oval; in the interiors of the two displayed oval images the degree is two.
Thus the displayed images are precisely the component source for the four rows of Theorem~\ref{thm:theorem-b-real-projection-criterion}; the count itself is left there as the citable real-projection theorem.
\end{proof}
