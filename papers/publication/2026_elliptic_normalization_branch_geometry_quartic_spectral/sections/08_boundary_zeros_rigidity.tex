\section{Supplementary boundary divisibility at \texorpdfstring{\(y=-1\)}{y=-1}}\label{sec:boundary-zeros}

This supplementary appendix records a recurrence-side divisibility fact for the same fixed sequence.  It explains the cancellation fibre $y=-1$ listed in Proposition~\ref{prop:minimal-order}; it is not used in the elliptic normalization, positive-axis dominance, or endpoint-zero proof.

\begin{proposition}[Uniform boundary zero of order at least two at $y = -1$]\label{prop:double-zero}
Let $u = y + 1$.
Then:
\begin{enumerate}
\item $Z_0(-1) = 1$ and $Z_1(y) = y + 1 = u$;
\item for all $m \ge 2$, the polynomial $Z_m(y)$ is divisible by $(y+1)^2$, i.e.,
\begin{equation}\label{eq:double-zero}
(y+1)^2 \mid Z_m(y) \qquad (m \ge 2).
\end{equation}
\end{enumerate}
In particular,
\[
Z_m(-1) = 0 \quad (m \ge 1), \qquad
\partial_y Z_m(-1) = 0 \quad (m \ge 2).
\]
\end{proposition}

\begin{proof}
We establish the factorization at the level of the generating function.
Writing $u = y + 1$ and $y = u - 1$, direct algebraic manipulation yields the identity
\begin{equation}\label{eq:N-D-identity}
N(t,y) = (1 + ut)\,D(t,y) + u^2\,t^2\,(1 - t^2)(1 + ty).
\end{equation}
Dividing both sides by $D(t,y)$:
\begin{equation}\label{eq:Z-factored}
Z(t,y) = \frac{N(t,y)}{D(t,y)}
= 1 + (y+1)\,t + (y+1)^2\,t^2 \cdot \frac{(1 - t^2)(1 + ty)}{D(t,y)}.
\end{equation}
The power-series expansion of the right-hand side in $t$ shows:
\begin{itemize}
\item the constant term is $1$, confirming $Z_0(y) = 1$;
\item the coefficient of $t$ is $y + 1$, confirming $Z_1(y) = y + 1$;
\item for $m \ge 2$, each coefficient $Z_m(y)$ receives a factor $(y+1)^2$ from the prefactor of $t^2$ in the last term, establishing \eqref{eq:double-zero}.
\end{itemize}
The evaluation $Z_m(-1) = 0$ for $m \ge 1$ follows from $(y+1) \mid Z_m(y)$ (which holds for all $m \ge 1$).
The vanishing of $\partial_y Z_m(-1)$ for $m \ge 2$ follows from the stronger divisibility $(y+1)^2 \mid Z_m(y)$.
\end{proof}

\begin{corollary}[Auxiliary boundary quotient degree law]\label{cor:boundary-quotient-degree-law}
For every $m\ge2$ define
\[
Q_m(y):=\frac{Z_m(y)}{(y+1)^2}.
\]
Then $Q_m(y)$ is a polynomial of degree
\begin{equation}\label{eq:boundary-quotient-degree}
\deg_y Q_m=\left\lfloor\frac{m-1}{2}\right\rfloor.
\end{equation}
Its leading coefficient is
\[
[y^{\deg Q_{2n+1}}]Q_{2n+1}=n\quad(n\ge1),
\qquad
[y^{\deg Q_{2n}}]Q_{2n}=\frac{n(n^2+5)}6\quad(n\ge1).
\]
Thus the uniform boundary zero at $y=-1$ removes exactly two powers from the polynomial degree and does not change the leading coefficient of the quotient polynomial.
\end{corollary}

\begin{proof}
Proposition~\ref{prop:double-zero} shows that $Q_m$ is a polynomial for every $m\ge2$.
Since $(y+1)^2$ is monic of degree two, division by $(y+1)^2$ lowers the degree by exactly two and preserves the leading coefficient.
Proposition~\ref{prop:degree-law} gives
\[
\deg_y Q_m=\deg_y Z_m-2
=\left\lfloor\frac{m+3}{2}\right\rfloor-2
=\left\lfloor\frac{m-1}{2}\right\rfloor,
\]
and gives the displayed leading coefficients for $Z_m$.
These are therefore also the leading coefficients of $Q_m$.
\end{proof}

\begin{remark}\label{rem:boundary-rigidity}
The divisibility at $y=-1$ is a check on the recurrence-side exceptional fibre, whereas the endpoint-zero theorem uses the different square-root collision at $y=0$.  The two boundary phenomena are therefore recorded separately and no asymptotic conclusion is drawn from the $y=-1$ divisibility.
\end{remark}
