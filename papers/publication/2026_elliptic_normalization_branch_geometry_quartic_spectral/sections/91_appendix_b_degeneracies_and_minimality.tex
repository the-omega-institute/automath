\section{Degenerate parameters and positive-axis Sturm data}\label{app:degeneracies}

This appendix records only the recurrence-side resultant, the exceptional fibres, and the primitive Sturm divisions used in Lemma~\ref{lem:positive-axis-dominance-criterion}.  The determinant and pseudo-remainder identities needed by the proof are displayed here, so the argument is independent of any ancillary verification file.

\subsection{Resultant and exceptional fibres}

For
\[
N(t,y)=1+yt+(y^2-y-1)t^2+(y^3-2y)t^3,
\qquad
D(t,y)=1-t-(2y+1)t^2+t^3+y(y+1)t^4,
\]
the exact resultant is
\[
\operatorname{Res}_t(N,D)=y^3(y-1)^3(y+1)^6.
\]
Thus a common factor of \(N(t,y)\) and \(D(t,y)\) over \(\CC[t]\) can occur only when \(y=-1,0,1\).  The reduced fibres are
\[
\begin{array}{c|c|c|c|c}
y & N(t,y) & D(t,y) & \text{reduced }Z(t,y) & \text{tail order}\\ \hline
1 & -(t-1)(t+1)^2 & (t-1)(t+1)^2(2t-1) & (1-2t)^{-1} & 1\\
0 & -(t-1)(t+1) & (t-1)^2(t+1) & (1-t)^{-1} & 1\\
-1 & t^3+t^2-t+1 & t^3+t^2-t+1 & 1 & 0
\end{array}
\]
For \(y\notin\{-1,0,1\}\), the displayed resultant is nonzero and \(N(t,y),D(t,y)\) are coprime.

\subsection{Positive-axis Sturm criterion}\label{app:positive-axis-sturm}

Let
\[
\Pi(\lambda,y)=\lambda^4-\lambda^3-(2y+1)\lambda^2+\lambda+y(y+1),
\qquad
c(y)=256y^3+411y^2+165y+32.
\]
The primitive chain used on the positive-axis cells is the following.
\begin{proposition}[Primitive Sturm chain for the positive-axis criterion]\label{prop:positive-axis-sturm-chain}
Set
\begin{align*}
S_0&=\Pi(\lambda,y),\\
S_1&=\partial_\lambda\Pi(\lambda,y),\\
16S_2&=(16y+11)\lambda^2+(4y-10)\lambda-16y^2-16y-1,\\
-\frac{(16y+11)^2}{16}S_3
&=4\lambda y^2-68\lambda y-8\lambda-64y^3-41y^2+25y+8,\\
S_4&=-\frac{y(y-1)(16y+11)^2c(y)}{256(y^2-17y-2)^2}.
\end{align*}
On each subinterval avoiding the possible rational denominator wall \(y^2-17y-2=0\), the ordinary Euclidean divisions in \(\QQ(y)[\lambda]\) satisfy
\[
\operatorname{rem}(S_0,S_1)=-S_2,
\qquad
\operatorname{rem}(S_1,S_2)=-S_3,
\qquad
\operatorname{rem}(S_2,S_3)=-S_4.
\]
Equivalently, after clearing denominators, the primitive pseudo-remainders are
\begin{align*}
\operatorname{prem}(S_0,S_1)
&= -16S_2,\\
\operatorname{prem}(S_1,16S_2)
&=16(4\lambda y^2-68\lambda y-8\lambda-64y^3-41y^2+25y+8),\\
\operatorname{prem}(16S_2,
&\ 4\lambda y^2-68\lambda y-8\lambda-64y^3-41y^2+25y+8)\\
&=y(y-1)(16y+11)^2c(y).
\end{align*}
The sign-controlling factors are
\[
\begin{array}{c|c|c}
\text{term} & \text{degree in }\lambda & \text{sign-controlling factor}\\ \hline
S_0&4&1\\
S_1&3&4\\
S_2&2&16y+11\\
S_3&1&-(y^2-17y-2)\\
S_4&0&-y(y-1)c(y).
\end{array}
\]
Here the omitted powers of \(16\), \((16y+11)^2\), and \((y^2-17y-2)^2\) are positive where the rational chain is evaluated.  With
\[
y_0=\frac{17+\sqrt{297}}2,
\]
the signs at \((-\infty,+\infty)\) are
\[
\begin{array}{c|c|c|c}
\text{cell} & -\infty & +\infty & \text{variation drop}\\ \hline
0<y<1 & (+,-,+,-,+) & (+,+,+,+,+) & 4\\
1<y<y_0 & (+,-,+,-,-) & (+,+,+,+,-) & 2\\
y>y_0 & (+,-,+,+,-) & (+,+,+,-,-) & 2.
\end{array}
\]
The apparent wall \(y_0\) is not a root-count boundary: at \(y^2-17y-2=0\) the discriminant \(-y(y-1)c(y)\) is nonzero, since \(y_0\ne0,1\) and the only real zero of \(c\) lies in \((-2,-1)\).  Hence the two rows for \(y>1\) extend to one constant real-root count on \((1,\infty)\).
\end{proposition}
