\section{Genus-one reduction of the quartic spectral curve}\label{sec:genus-one}

This section records the algebraic reduction used for the fixed quartic.  General balanced-family statements are kept only to the extent needed to identify the canonical member and to justify the finite real-projection test used later; no chamber decomposition in the four parameters is claimed.

\subsection{A balanced quartic family}

\begin{lemma}[Base-field compatibility of the balanced normalization]\label{lem:balanced-base-field}
Let $K$ be a field with $\operatorname{char}K\ne2,3$.  For $a,c,d,g\in K$, the change of variables
\[
\xi=y-\lambda^2,\qquad \WeierY=2\xi-c,
\qquad y=\lambda^2+\frac{\WeierY+c}{2}
\]
identifies $P_{a,c,d,g}(\lambda,y)=0$ with $\WeierY^2=c^2-4a\lambda^3-4d\lambda-4g$, and this identification, the cubic squarefreeness condition, and the finite ramification equation commute with scalar extension.
\end{lemma}
\begin{proof}
Substitution gives
$P_{a,c,d,g}=\xi^2-c\xi+a\lambda^3+d\lambda+g$, hence after multiplying by $4$ one obtains the displayed cubic.  The inverse formula recovers $y$.  Squarefreeness and ramification are determined by polynomial equations over $K$, so both commute with base change.
\end{proof}

\begin{proposition}[Balanced quartic family]\label{prop:balanced-family}
For
\begin{equation}\label{eq:balanced-family}
P_{a,c,d,g}(\lambda,y)=\lambda^4+a\lambda^3+(c-2y)\lambda^2+d\lambda+y(y-c)+g,
\end{equation}
put
\begin{equation}\label{eq:balanced-xi}
\xi=y-\lambda^2,
\qquad
\WeierY=2\xi-c.
\end{equation}
Then $P_{a,c,d,g}=0$ is affinely isomorphic to
\begin{equation}\label{eq:balanced-weierstrass}
\WeierY^2=c^2-4a\lambda^3-4d\lambda-4g,
\end{equation}
with inverse projection to the original parameter
\begin{equation}\label{eq:balanced-inverse}
y=\lambda^2+\frac{\WeierY+c}{2}.
\end{equation}
If the cubic on the right is squarefree, its smooth projective completion is a genus-one curve and the finite ramification of $y$ is cut out by
\begin{equation}\label{eq:balanced-ramification-eq}
2\lambda \WeierY=3a\lambda^2+d.
\end{equation}
Eliminating $(\lambda,\WeierY)$ gives the finite branch polynomial
\begin{equation}\label{eq:balanced-ramification}
\Res_\lambda\!\bigl(P_{a,c,d,g},\partial_\lambda P_{a,c,d,g}\bigr).
\end{equation}
\phantomsection\label{eq:balanced-quintic}
\end{proposition}
\begin{proof}
The affine isomorphism is Lemma~\ref{lem:balanced-base-field}.  Differentiating the Weierstrass equation and $y=\lambda^2+(\WeierY+c)/2$ shows that $dy=0$ exactly when \eqref{eq:balanced-ramification-eq} holds.  Eliminating the repeated root from $P$ and $\partial_\lambda P$ gives the same finite branch values.  The smoothness criterion is the nonvanishing cubic discriminant
\begin{equation}\label{eq:balanced-genus-condition}
\Disc_\lambda(c^2-4a\lambda^3-4d\lambda-4g)\ne0,
\end{equation}
which gives the genus-one completion.  This also records the content formerly separated as Corollary~\ref{cor:balanced-smooth}, Lemma~\ref{lem:balanced-projective-completion}, and Corollary~\ref{cor:balanced-ramification}.
\label{cor:balanced-smooth}\label{lem:balanced-projective-completion}\label{cor:balanced-ramification}
\end{proof}

\begin{theorem}[Balanced elliptic normalization and ramification]\label{thm:theorem-a-balanced-elliptic-normalization}
For every smooth balanced member \eqref{eq:balanced-family}, the affine curve is normalized by the smooth projective completion of \eqref{eq:balanced-weierstrass}; the map to the $y$-line has degree $4$, finite ramification satisfying \eqref{eq:balanced-ramification-eq}, and finite branch values given by \eqref{eq:balanced-ramification}.  In the canonical member $P_{-1,-1,1,0}=\Pi$, this normalization is
\[
E:\quad \WeierY^2=4\lambda^3-4\lambda+1,
\qquad
 y=\lambda^2-\frac12+\frac12\WeierY.
\]
\end{theorem}
\begin{proof}
Proposition~\ref{prop:balanced-family} gives the affine isomorphism and finite ramification equation.  Projectivizing the squarefree cubic gives a smooth genus-one curve with one point at infinity, and the rational function $y=\lambda^2+(\WeierY+c)/2$ has pole order $4$ at that point, hence degree $4$.  Substituting $(a,c,d,g)=(-1,-1,1,0)$ gives the displayed canonical model.
\end{proof}

\subsection{Finite real-projection test for one fixed member}

\begin{lemma}[Real Weierstrass ovals and regular values]\label{lem:real-weierstrass-oval-covering-count}
On a smooth real Weierstrass cubic, the image of a connected real component under a real regular function is an interval whose endpoints occur at infinity or at critical points of the function on that component.
\end{lemma}
\begin{proof}
Each real component is a circle or a line with the point at infinity added in the projective completion.  Away from critical points the map is a local diffeomorphism, and compactness/properness on the projective component gives interval images with endpoints among the listed values.
\end{proof}

\begin{lemma}[Proper real covering away from the finite transition set]\label{lem:real-projection-proper-covering}
For a fixed smooth real balanced member, the number of real preimages of a real value of $y$ is locally constant away from the finite set consisting of critical values and values attained at real points over infinity.
\end{lemma}
\begin{proof}
Remove the finite critical set.  The real map is then a proper submersion on each remaining component over compact subintervals, hence a covering; covering degree is locally constant.
\end{proof}

\begin{lemma}[Endpoint and critical values on a real component]\label{lem:real-component-extrema-enumeration}
For one fixed smooth real balanced member, the real image of each real component is determined by evaluating $y=\lambda^2+(\WeierY+c)/2$ at the finite critical points satisfying \eqref{eq:balanced-ramification-eq} and at the relevant points over infinity.
\end{lemma}
\begin{proof}
This is Lemma~\ref{lem:real-weierstrass-oval-covering-count} with the critical equation from Proposition~\ref{prop:balanced-family}.
\end{proof}

\begin{lemma}[Fixed-member endpoint-and-extrema test]\label{def:balanced-projection-datum}\label{prop:balanced-transition-support-reduction}\label{prop:balanced-projection-datum-complete}\label{prop:balanced-real-projection-fixed-extrema}\label{prop:balanced-real-projection}
Fix one smooth real balanced member.  Its real projection to the $y$-line is obtained by the following finite test: list the real Weierstrass components, solve \eqref{eq:balanced-ramification-eq} on each component, include the component endpoint values at infinity, and take the corresponding interval hulls.  This is only a finite extrema test for that member; it is not a chamber decomposition as the balanced parameters vary.
\begin{equation}\label{eq:balanced-real-extrema-set}
\mathcal T=\{y(P):P\in E(\RR),\ dy(P)=0\}\cup\{\text{real endpoint values over infinity}\}.
\end{equation}
\end{lemma}
\begin{proof}
Combine Lemmas~\ref{lem:real-projection-proper-covering} and~\ref{lem:real-component-extrema-enumeration}.  Between consecutive values of $\mathcal T$ the real preimage count is constant, and endpoints are read by evaluating $y$ at the finite list $\mathcal T$.  No conclusion is drawn for nearby members without repeating this finite computation.\label{rem:balanced-projection-reading}\label{rem:balanced-finite-comparison-algorithm}\label{rem:balanced-real-interval-alternatives}\label{rem:balanced-fixed-criterion-complete}\label{cor:fixed-member-projection-comparison}
\end{proof}

\begin{proposition}[Canonical quartic real projection input]\label{cor:quartic-real-projection-framework}\label{cor:two-interval-specialization}
For the canonical curve $E:\WeierY^2=4\lambda^3-4\lambda+1$, the real locus has two components.  The compact oval maps to
\[
[\ySub,1],
\]
and the noncompact oval maps to
\[
[0,\infty).
\]
The finite critical values are $\ySub$ and $1$ on the compact oval, and $0$ on the noncompact oval.  This proposition is the only place where the preceding fixed-member test is used in the main theorem chain.
\end{proposition}
\begin{proof}
Apply Lemma~\ref{prop:balanced-real-projection-fixed-extrema} to the canonical member.  The finite branch values are $0$, $1$, and the roots of $256y^3+411y^2+165y+32$.  Lemma~\ref{lem:cubic-branch-real-root} shows that exactly one cubic branch value is real, namely $\ySub<0$.  Direct evaluation at the branch points assigns $\ySub,1$ to the compact oval and $0$ to the noncompact oval; the noncompact component has $y\to+\infty$ at infinity.\label{rem:balanced-family-generality}
\end{proof}

\subsection{Double-root normalization inside the balanced family}

\begin{lemma}[Shift and scaling]\label{lem:balanced-affine}
The affine changes
\begin{equation}\label{eq:canonical-normalization-change}
\lambda=\alpha\Lambda+\beta,
\qquad
 y=\gamma \WeierY_0+\delta,
\qquad \alpha\gamma\ne0,
\end{equation}
preserve the balanced form after rescaling the equation, with transformed parameters
\begin{equation}\label{eq:canonical-transformed-parameters}
(a,c,d,g)\mapsto(a',c',d',g')
\end{equation}
obtained by coefficient comparison.
\end{lemma}
\begin{proof}
Substitute \eqref{eq:canonical-normalization-change} into \eqref{eq:balanced-family}, multiply by the inverse leading coefficient, and compare the four balanced coefficients.  No further geometry is used.
\end{proof}

\begin{definition}[Normalized collision data]\label{def:normalized-collision-data}
The normalized collision data are double roots at $(\lambda,y)=(1,0)$ and $(\lambda,y)=(-1,1)$ for a balanced quartic.
\end{definition}

\begin{lemma}[Fixed-coordinate double-root normalization]\label{thm:balanced-classification}\label{lem:balanced-classification}
Let \(P_{a,c,d,g}\) be balanced.  If \(P_{a,c,d,g}(\lambda,0)\) has a double root at \(\lambda=1\) and \(P_{a,c,d,g}(\lambda,1)\) has a double root at \(\lambda=-1\), then
\begin{equation}\label{eq:canonical-normalization-equation}
(a,c,d,g)=(-1,-1,1,0).
\end{equation}
Equivalently, the selected member is
\begin{equation}\label{eq:canonical-normalization-parameters}
P_{-1,-1,1,0}(\lambda,y)=\Pi(\lambda,y).
\end{equation}
\end{lemma}
\begin{proof}
The two double-root conditions are the four linear equations
\[
1+a+c+d+g=0,
\qquad
4+3a+2c+d=0,
\]
and
\[
-a-d+g=0,
\qquad
3a-2c+d=0.
\]
Solving this system gives \(a=-1\), \(c=-1\), \(d=1\), and \(g=0\).  Substitution in \eqref{eq:balanced-family} then gives \(P_{-1,-1,1,0}=\Pi\).
\end{proof}

\begin{corollary}[Canonical balanced member]\label{cor:canonical-balanced-member}
The normalized collision data of Definition~\ref{def:normalized-collision-data} select the unique balanced quartic \(P_{-1,-1,1,0}=\Pi\).
\end{corollary}
\begin{proof}
This is Lemma~\ref{lem:balanced-classification} restated for the normalized collision data.
\end{proof}

\begin{corollary}[Weighted pullback of the canonical double-root member]\label{cor:canonical-specialization-up-to-normalization}
Any real balanced member with two distinct double-root collisions that can be moved to the normalized positions by an affine change of the form \eqref{eq:canonical-normalization-change} is the corresponding weighted pullback of $\Pi$.
\end{corollary}
\begin{proof}
After the change of variables the member satisfies Lemma~\ref{lem:balanced-classification}; applying the inverse change gives the claim.
\end{proof}

\begin{lemma}[Real weighted pullback of positive-side dominance]\label{lem:real-weighted-pullback-dominance}
A real affine normalization with positive $y$-scale transports strict maximal-modulus dominance from the canonical positive branch to the corresponding pulled-back branch on any fixed compact positive interval avoiding branch values.
\end{lemma}
\begin{proof}
The root functions are carried by a real affine map in $\lambda$ and a monotone real reparametrization in $y$; on a fixed compact interval the canonical strict modulus gap has a positive minimum.
\end{proof}

\begin{corollary}[Real weighted pullback of canonical dominance]\label{cor:canonical-dominance-after-normalization}
The canonical dominance theorem, Theorem~\ref{thm:canonical-branch-dominance}, transfers to such normalized real pullbacks on fixed compact positive intervals.
\end{corollary}
\begin{proof}
This is Lemma~\ref{lem:real-weighted-pullback-dominance} applied to Theorem~\ref{thm:canonical-branch-dominance}.
\end{proof}

\subsection{Specialization to the quartic transfer family}

\begin{proposition}[Canonical affine model and branch geometry]\label{prop:elliptic-reduction}
For $\Pi$, set
\begin{equation}\label{eq:xi-def}
\xi=y-\lambda^2.
\end{equation}
Then
\begin{equation}\label{eq:elliptic-xi}
\Pi=\xi^2+\xi-\lambda^3+\lambda,
\end{equation}
and with $\WeierY=2\xi+1$ one obtains
\begin{equation}\label{eq:elliptic-weierstrass}
E:\quad \WeierY^2=4\lambda^3-4\lambda+1.
\end{equation}
The invariants are
\begin{equation}\label{eq:weierstrass-invariants}
g_2=4,\qquad g_3=-1,
\end{equation}
and the degree-four projection is
\begin{equation}\label{eq:y-from-E}
y=\lambda^2-\frac12+\frac12\WeierY.
\end{equation}
The finite branch values are
\begin{equation}\label{eq:branch-values}
0,
\quad 1,
\quad \text{and the roots of }c(y)=256y^3+411y^2+165y+32.
\end{equation}
\phantomsection\label{eq:branch-value-convention}
Equivalently,
\begin{equation}\label{eq:Pi-as-quadratic-in-y}
\Pi(\lambda,y)=y^2+(1-2\lambda^2)y+\lambda^4-\lambda^3-\lambda^2+\lambda,
\end{equation}
and finite ramification is certified by
\begin{equation}\label{eq:ramification-poly}
y(y-1)c(y)=0.
\end{equation}
\end{proposition}
\begin{proof}
The identities follow from the substitution in Theorem~\ref{thm:theorem-a-balanced-elliptic-normalization}.  Eliminating finite critical points using \eqref{eq:balanced-ramification-eq} gives \eqref{eq:ramification-poly}; Appendix~\ref{app:disc} records the exact Sylvester calculation.
\end{proof}

\begin{lemma}[Cubic branch elimination]\label{lem:cubic-branch-elimination}
The cubic $c(y)=256y^3+411y^2+165y+32$ has one real root $\ySub\in(-2,-1)$ and two nonreal roots.  This is a local elimination lemma for the canonical branch-value table, not a separate main theorem.  The elimination identity used to locate it is
\begin{equation}\label{eq:cubic-branch-elimination-identity}
c(y)=256(y-\ySub)(y^2+\alpha y+\beta),\qquad \alpha,\beta\in\RR,
\end{equation}
with the quadratic factor positive on $\RR$.
\end{lemma}
\begin{proof}
The signs $c(-2)<0<c(-1)$ give a real root in $(-2,-1)$.  The Sturm count for $c$ gives exactly one real root; the remaining factor is therefore an irreducible real quadratic, hence positive after normalization by its positive leading coefficient.
\end{proof}

\begin{lemma}[Simplicity of the finite ramification locus]\label{lem:finite-ramification-simple}
All finite ramification values in \eqref{eq:branch-values} are simple.  The common framework is
\begin{equation}\label{eq:finite-ramification-framework}
\Pi(\lambda,y)=\Pi_\lambda(\lambda,y)=0,
\qquad
\Pi_y(\lambda,y)=\WeierY.
\end{equation}
\end{lemma}
\begin{proof}
At a repeated root, $\Pi_y=\WeierY$.  If $\WeierY=0$ then the Weierstrass equation and ramification equation force $\lambda=\pm1/\sqrt3$, which is incompatible with $\WeierY^2=4\lambda^3-4\lambda+1=0$.  Thus no branch value is lost through the implicit denominator, and the displayed discriminant factors are squarefree.
\end{proof}

\begin{lemma}[Projective normalization of the quartic]\label{lem:projective-normalization}
The projective closure
\begin{equation}\label{eq:projective-quartic}
\Pi^{\mathrm{proj}}=0\subset\mathbf P^2
\end{equation}
is normalized by the morphism from $E$ induced by
\begin{equation}\label{eq:normalization-map}
(\lambda,\WeierY)\longmapsto (\lambda,y=\lambda^2-1/2+\WeierY/2).
\end{equation}
\end{lemma}
\begin{proof}
The affine map is birational by the inverse $\WeierY=2y+1-2\lambda^2$.  The source is smooth and projective, and the induced map to the projective closure is finite; hence it is the normalization.
\end{proof}

\subsection{Riemann--Hurwitz consistency}\label{subsec:riemann-hurwitz}
\begin{corollary}[Ramification budget for the degree-four projection]\label{cor:riemann-hurwitz}
The degree-four map $y:E\to\mathbf P^1_y$ has total ramification $8$, as required by Riemann--Hurwitz for a genus-one source and genus-zero target.
\end{corollary}
\begin{proof}
The finite ramification values in \eqref{eq:branch-values}, together with the point over infinity, account for the required total.  This is a consistency check on the explicit branch calculation.
\end{proof}

\subsection{Discriminant factorization as a corollary}\label{subsec:disc-corollary}
\begin{lemma}[Finite repeated roots and ramification]\label{lem:repeated-root-ramification}
For the canonical quartic, a finite value $y_0$ is a branch value of $y:E\to\mathbf P^1_y$ if and only if $\Pi(\cdot,y_0)$ has a repeated root.  The identity excluding denominator loss is
\begin{equation}\label{eq:Pi-y-equals-Weierstrass-coordinate}
\Pi_y(\lambda,y)=2y+1-2\lambda^2=\WeierY.
\end{equation}
\end{lemma}
\begin{proof}
A repeated root is a common zero of $\Pi$ and $\Pi_\lambda$.  Under the normalization, $dy/d\lambda$ has denominator $\WeierY=\Pi_y$; Lemma~\ref{lem:finite-ramification-simple} shows $\WeierY\ne0$ at finite ramification points, so the two criteria are equivalent.
\end{proof}

\begin{corollary}[Discriminant factorization]\label{cor:disc-factor}
\begin{equation}\label{eq:disc-factor-new}
\Disc_\lambda\Pi(\lambda,y)=-y(y-1)(256y^3+411y^2+165y+32).
\end{equation}
\end{corollary}
\begin{proof}
By Lemma~\ref{lem:repeated-root-ramification}, the zeros of the discriminant are precisely the finite branch values in Proposition~\ref{prop:elliptic-reduction}.  Appendix~\ref{app:disc} gives the leading coefficient $-256$, which fixes the factorization.
\end{proof}

\begin{corollary}[Canonical spectral-quartic specialization]\label{cor:canonical-theorem-a-specialization}
For $\Pi$, Theorem~\ref{thm:theorem-a-balanced-elliptic-normalization} specializes to $P_{-1,-1,1,0}$, the elliptic model \eqref{eq:elliptic-weierstrass}, the projection \eqref{eq:y-from-E}, the branch values \eqref{eq:branch-values}, and the discriminant \eqref{eq:disc-factor-new}.
\end{corollary}
\begin{proof}
This is the collection of Lemma~\ref{lem:balanced-classification}, Proposition~\ref{prop:elliptic-reduction}, Lemma~\ref{lem:projective-normalization}, and Corollary~\ref{cor:disc-factor}.
\end{proof}

\begin{remark}\label{rem:genus-one-upgrade}
For the present transfer polynomial, $\xi=y-\lambda^2$ is the balanced-family normalization specialized to the unique fixed member selected by the two double-root collisions.
\end{remark}
