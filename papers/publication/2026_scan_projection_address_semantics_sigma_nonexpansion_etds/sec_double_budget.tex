\section{Phase-resolved R\'enyi laws and collisions}
\label{sec:double-budget}

\subsection{Conditioned survivor power sums}

For \(a\in\mathcal L_m^H\), define the survivor-prefix law
\[
  \lambda_m^H(a)
  :=\frac{\nu_\varphi([a])}{\nu_\varphi(\tau_H\geq m)}
  =\frac{\nu_\varphi([a])}{Z_{H,m}(1)}.
\]
This is a probability distribution on \(\mathcal L_m^H\).  Its
R\'enyi power sums are
\[
  S_s^H(m):=\sum_{a\in\mathcal L_m^H}\lambda_m^H(a)^s
  =\frac{Z_{H,m}(s)}{Z_{H,m}(1)^s}.
\]
For \(s>0\) set
\begin{equation}
  h_{s,H}:=-\log\frac{\rho_s}{\rho_1^s}
  =sP_{Y_H}(\varphi)-P_{Y_H}(s\varphi),
  \label{eq:renyi-rate}
\end{equation}
where the pressure identity follows from \eqref{eq:rho-pressure}.  For
\(r\in\mathbb Z/d\mathbb Z\), define
\begin{equation}
  c_{s,r}
  :=\frac{A_{s,r}(\mathbf1)}{A_{1,r}(\mathbf1)^s}>0.
  \label{eq:phase-renyi-constant}
\end{equation}

\begin{theorem}[Periodic survivor R\'enyi asymptotics]
\label{thm:periodic-renyi}
For every \(s>0\), there is \(0<\kappa_s<1\) such that
\begin{equation}
  S_s^H(m)
  =c_{s,r}e^{-(m-1)h_{s,H}}
   (1+O(\kappa_s^{m-1})),
  \qquad m-1\equiv r\pmod d.
  \label{eq:conditioned-phase-power}
\end{equation}
In particular,
\[
  \lim_{m\to\infty}-\frac1m\log S_s^H(m)=h_{s,H},
\]
but the normalized quantity
\(e^{(m-1)h_{s,H}}S_s^H(m)\) need not converge unless the constants
\(c_{s,r}\) are independent of \(r\).
\end{theorem}

\begin{proof}
Apply \eqref{eq:unconditioned-phase-power} once with exponent \(s\) and
once with exponent \(1\), and divide.  The quotient of the exponential
terms is
\[
  \left(\frac{\rho_s}{\rho_1^s}\right)^{m-1}
  =e^{-(m-1)h_{s,H}},
\]
and the quotient of the leading coefficients is \(c_{s,r}\).  The
error remains exponentially small because the leading coefficients are
positive and there are only finitely many depth phases.
\end{proof}

\subsection{The strict R\'enyi gap}

The collision approximation requires higher power sums to be smaller
than the power predicted by a lower moment.  Periodicity does not
destroy this separation, but irreducibility alone is insufficient: it
fails on a deterministic cycle.

\begin{definition}[Branching survivor]
The safe graph is called branching if some \(i\in\mathcal S\) has at
least two safe successors \(j\) with \(K_{ij}>0\).  An irreducible safe
graph is nonbranching if and only if it is a single directed cycle.
\end{definition}

\begin{proposition}[Strict entrywise-power inequality]
\label{prop:strict-renyi-gap}
Assume that the safe graph is branching.  For all real
\(1\leq s<t\),
\begin{equation}
  h_{t,H}>\frac{t}{s}h_{s,H}.
  \label{eq:strict-renyi-gap}
\end{equation}
In particular \(h_{s,H}>0\) for every \(s>1\).
\end{proposition}

\begin{proof}
Let \(r_1>0\) be the Perron right vector of \(B_1\), and define the
irreducible stochastic matrix
\[
  P_{ij}:=\frac{B_1(i,j)r_1(j)}{\rho_1r_1(i)}.
\]
For \(s\geq1\), diagonal similarity gives
\begin{equation}
  \rho(P^{\circ s})=\frac{\rho_s}{\rho_1^s}=e^{-h_{s,H}},
  \label{eq:entrywise-spectral-radius}
\end{equation}
where \(P^{\circ s}\) denotes entrywise exponentiation.

Fix \(1\leq s<t\), put \(p=t/s>1\), and let \(x>0\) satisfy
\[
  P^{\circ s}x=\rho(P^{\circ s})x.
\]
For every state \(i\),
\begin{align*}
  \bigl(P^{\circ t}x^{\circ p}\bigr)_i
  &=\sum_j(P_{ij}^s x_j)^p\\
  &\leq\left(\sum_jP_{ij}^s x_j\right)^p
   =\rho(P^{\circ s})^p x_i^p.
\end{align*}
The inequality is strict in any branching row.  Since
\(P^{\circ t}\) is irreducible, the strict Collatz--Wielandt theorem
implies
\[
  \rho(P^{\circ t})<\rho(P^{\circ s})^{t/s}.
\]
Using \eqref{eq:entrywise-spectral-radius} and taking negative
logarithms proves \eqref{eq:strict-renyi-gap}.  Setting \(s=1\) and
noting that \(h_{1,H}=0\) gives the final assertion.
\end{proof}

\begin{remark}[Why the branching hypothesis is necessary]
\label{rem:deterministic-obstruction}
If the safe graph is a directed cycle, conditioning on survival leaves
only a bounded number of possible words at each depth.  Algebraically,
every row of \(P\) has a single nonzero entry, so equality holds in the
power-sum inequality and \(h_{s,H}=0\) for all \(s\).  The exponentially
growing birthday scale and the negligibility of overlap terms both
fail.  This is a structural obstruction, not a missing estimate.
\end{remark}

The obstruction can be described exactly.  Suppose the safe graph is
the cycle
\[
  i_0\longrightarrow i_1\longrightarrow\cdots
  \longrightarrow i_{d-1}\longrightarrow i_0,
\]
and write \(w_a:=K_{i_a i_{a+1}}\), with cyclic indices.  A safe word of
length \(m\) is determined by its initial state.  If \(n=m-1=qd+r\),
its unconditioned mass is
\[
  \pi_{i_a}
  \left(\prod_{b=0}^{d-1}w_b\right)^q
  \prod_{j=0}^{r-1}w_{a+j}.
\]
The common factor corresponding to \(q\) complete turns cancels after
conditioning on survival.  Hence along \(n\equiv r\pmod d\) the
survivor-prefix law is the fixed distribution
\begin{equation}
  \eta_r(a)
  :=\frac{\displaystyle
       \pi_{i_a}\prod_{j=0}^{r-1}w_{a+j}}
      {\displaystyle
       \sum_{b=0}^{d-1}\pi_{i_b}\prod_{j=0}^{r-1}w_{b+j}},
  \qquad a\in\mathbb Z/d\mathbb Z.
  \label{eq:cycle-conditioned-law}
\end{equation}
It follows that
\begin{equation}
  S_s^H(m)=\sum_{a=0}^{d-1}\eta_r(a)^s>0
  \qquad(n\equiv r\pmod d),
  \label{eq:cycle-power-sum}
\end{equation}
with no exponential decay.

\begin{proposition}[Failure on a deterministic survivor cycle]
\label{prop:cycle-failure}
For a deterministic survivor cycle, \(h_{s,H}=0\) for every \(s>0\).
For every fixed \(k\geq2\) and every sequence \(N_m\to\infty\),
\[
  \mathbb EW_{m,k}longrightarrow\infty.
\]
Consequently there is no birthday regime with a diverging sample size
and a finite positive Poisson mean.
\end{proposition}

\begin{proof}
Formula \eqref{eq:cycle-power-sum} is bounded below uniformly over the
finitely many phases.  Therefore
\[
  \mathbb EW_{m,k}
  =\binom{N_m}{k}S_k^H(m)\longrightarrow\infty.
\]
The identity \(h_{s,H}=0\) follows either from
\eqref{eq:cycle-power-sum} or because every entrywise power of the
normalized deterministic transition matrix has spectral radius one.
\end{proof}

\subsection{A total-variation collision theorem}

Fix an integer \(k\geq2\).  For each \(m\), let
\[
  X_m^{H,(1)},\ldots,X_m^{H,(N_m)}
\]
be independent random words with the common law \(\lambda_m^H\).  For a
\(k\)-subset \(I\subseteq\{1,\ldots,N_m\}\), let \(J_I\) be the
indicator that all words indexed by \(I\) are equal, and define
\[
  W_{m,k}:=\sum_{\lvert I\rvert=k}J_I.
\]
Then
\[
  \mathbb E J_I=S_k^H(m),
  \qquad
  \lambda_{m,k}:=\mathbb E W_{m,k}
  =\binom{N_m}{k}S_k^H(m).
\]
If \(I\) and \(J\) are disjoint, the corresponding indicators are
independent.  If \(\lvert I\cap J\rvert=\ell\geq1\), then
\begin{equation}
  \mathbb E[J_IJ_J]=S_{2k-\ell}^H(m),
  \label{eq:joint-collision-moment}
\end{equation}
because all \(2k-\ell\) involved samples must agree.

\begin{theorem}[Periodic survivor collision law]
\label{thm:periodic-collisions}
Assume that the safe graph is irreducible, periodic of period \(d\),
has at least two states, and is branching.  Fix an integer \(k\geq2\),
and use independent survivor prefixes with the common law
\(\lambda_m^H\) as above.  There is a
constant \(C_k<\infty\), depending only on \(k\), such that
\begin{align}
 d_{\rm TV}\bigl(\mathcal L(W_{m,k}),
     \operatorname{Poisson}(\lambda_{m,k})\bigr)
 &\leq C_k\min\{1,\lambda_{m,k}^{-1}\}
 \Bigl(
   N_m^{2k-1}S_k^H(m)^2 \notag\\
 &\hspace{42mm}+
   \sum_{\ell=1}^{k-1}
     N_m^{2k-\ell}S_{2k-\ell}^H(m)
 \Bigr).
 \label{eq:collision-tv-bound}
\end{align}
Consequently, if
\[
  \binom{N_m}{k}S_k^H(m)\longrightarrow\tau\in(0,\infty),
\]
then
\[
  W_{m,k}\Rightarrow\operatorname{Poisson}(\tau).
\]

For an explicit pressure normalization, fix
\(r\in\mathbb Z/d\mathbb Z\) and let \(m\to\infty\) through
\(m-1\equiv r\pmod d\).  For every \(\alpha\in(0,\infty)\),
\begin{equation}
  N_m e^{-(m-1)h_{k,H}/k}\longrightarrow\alpha
  \quad\Longrightarrow\quad
  W_{m,k}\Rightarrow
  \operatorname{Poisson}\!\left(
    \frac{c_{k,r}\alpha^k}{k!}\right).
  \label{eq:phase-poisson-limit}
\end{equation}
The collision threshold is sharp along every phase:
\begin{equation}
  \mathbb P(W_{m,k}\geq1)\longrightarrow
  \begin{cases}
    0,
      &N_m e^{-(m-1)h_{k,H}/k}\longrightarrow0,\\[2pt]
    1-\exp(-c_{k,r}\alpha^k/k!),
      &N_m e^{-(m-1)h_{k,H}/k}\longrightarrow\alpha\in(0,\infty),\\[2pt]
    1,
      &N_m e^{-(m-1)h_{k,H}/k}\longrightarrow\infty.
  \end{cases}
  \label{eq:phase-threshold}
\end{equation}
\end{theorem}

\begin{proof}
Join two \(k\)-subsets when they intersect.  This is a dependency graph
for \((J_I)\).  The Chen--Stein theorem of
\citet[Theorem~1]{ArratiaGoldsteinGordon1989}, together with
\eqref{eq:joint-collision-moment}, bounds the total variation distance
by a universal factor times \(\min\{1,\lambda_{m,k}^{-1}\}\) multiplied
by
\begin{align*}
 b_1
 &\leq \binom{N_m}{k}
   \sum_{\ell=1}^k
   \binom{k}{\ell}\binom{N_m-k}{k-\ell}S_k^H(m)^2,\\
 b_2
 &=\binom{N_m}{k}
   \sum_{\ell=1}^{k-1}
   \binom{k}{\ell}\binom{N_m-k}{k-\ell}
   S_{2k-\ell}^H(m).
\end{align*}
Bounding the binomial factors by powers of \(N_m\) gives
\eqref{eq:collision-tv-bound}.

Suppose first that \(\lambda_{m,k}\to\tau\in(0,\infty)\).  By
Theorem~\ref{thm:periodic-renyi}, the finitely many constants
\(c_{k,r}\) are bounded above and below by positive numbers.  Hence
\[
  N_m\asymp e^{(m-1)h_{k,H}/k}.
\]
The first term in \eqref{eq:collision-tv-bound} is
\(O(N_m^{-1}(N_m^kS_k^H(m))^2)=o(1)\).  For
\(t=2k-\ell>k\), Theorem~\ref{thm:periodic-renyi} and
Proposition~\ref{prop:strict-renyi-gap} give
\[
  N_m^tS_t^H(m)
  =O\!\left(
    \exp\left((m-1)\left(\frac{t}{k}h_{k,H}-h_{t,H}\right)\right)
  \right)
  =o(1).
\]
Thus the total-variation distance tends to zero.  Convergence of the
Poisson means proves the first limit.

If \(m-1\equiv r\pmod d\), then
\[
  \lambda_{m,k}
  =\binom{N_m}{k}c_{k,r}e^{-(m-1)h_{k,H}}(1+o(1)).
\]
The scaling in \eqref{eq:phase-poisson-limit} therefore gives
\(\lambda_{m,k}\to c_{k,r}\alpha^k/k!\), proving the phase-resolved
limit and the middle line of \eqref{eq:phase-threshold}.

In the subcritical regime, \(\mathbb EW_{m,k}=\lambda_{m,k}\to0\), so
Markov's inequality proves the first line.  In the supercritical regime,
write
\[
  N_m=e^{(m-1)h_{k,H}/k}L_m,
  \qquad L_m\longrightarrow\infty.
\]
The dependency graph and \eqref{eq:joint-collision-moment} give
\[
  \operatorname{Var}(W_{m,k})
  \leq\lambda_{m,k}
  +C_kN_m^{2k-1}S_k^H(m)^2
  +C_k\sum_{\ell=1}^{k-1}
      N_m^{2k-\ell}S_{2k-\ell}^H(m).
\]
Now \(\lambda_{m,k}\asymp L_m^k\to\infty\).  After division by
\(\lambda_{m,k}^2\), the first two terms tend to zero.  For
\(t=2k-\ell\), the remaining ratio is bounded by a constant times
\[
  L_m^{-\ell}
  \exp\left((m-1)\left(\frac{t}{k}h_{k,H}-h_{t,H}\right)\right),
\]
which tends to zero by \eqref{eq:strict-renyi-gap}.  Chebyshev's
inequality yields \(\mathbb P(W_{m,k}=0)\to0\), proving the last line.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:intro-collisions}]
This is the critical and threshold part of
Theorem~\ref{thm:periodic-collisions}.
\end{proof}

\begin{proposition}[Exponential approximation rate at criticality]
\label{prop:critical-tv-rate}
Fix \(k\geq2\) and define
\begin{equation}
  \Delta_k
  :=\min\left\{
    \frac{h_{k,H}}{k},
    \min_{k<t\leq2k-1}
      \left(h_{t,H}-\frac{t}{k}h_{k,H}\right)
  \right\}>0.
  \label{eq:critical-gap}
\end{equation}
Suppose that for some constants \(0<a\leq b<\infty\),
\begin{equation}
  a\leq N_m e^{-(m-1)h_{k,H}/k}\leq b
  \label{eq:critical-window}
\end{equation}
for all sufficiently large \(m\).  Then
\begin{equation}
  d_{\rm TV}\bigl(\mathcal L(W_{m,k}),
    \operatorname{Poisson}(\lambda_{m,k})\bigr)
  =O(e^{-(m-1)\Delta_k}).
  \label{eq:critical-tv-rate}
\end{equation}

Fix a phase \(r\) and suppose in addition that, along
\(m-1\equiv r\pmod d\),
\[
  N_m e^{-(m-1)h_{k,H}/k}
  =\alpha+O(e^{-\eta(m-1)})
\]
for some \(\alpha,\eta>0\).  Then there is \(\delta>0\) such that
\begin{equation}
  d_{\rm TV}\left(
    \mathcal L(W_{m,k}),
    \operatorname{Poisson}\!\left(\frac{c_{k,r}\alpha^k}{k!}\right)
  \right)
  =O(e^{-\delta(m-1)}).
  \label{eq:critical-target-rate}
\end{equation}
One may take \(\delta\) to be the minimum of \(\Delta_k\), \(\eta\),
and the finitely many Perron convergence rates entering
Theorem~\ref{thm:periodic-renyi}.
\end{proposition}

\begin{proof}
Write \(n=m-1\).  Under \eqref{eq:critical-window},
\(N_m\asymp e^{nh_{k,H}/k}\), while
Theorem~\ref{thm:periodic-renyi} gives
\(S_s^H(m)\asymp e^{-nh_{s,H}}\) uniformly over the finitely many
depth phases and moment orders in question.  The isolated-neighborhood
term in \eqref{eq:collision-tv-bound} satisfies
\[
  N_m^{2k-1}S_k^H(m)^2
  =O(e^{-nh_{k,H}/k}).
\]
For \(t=2k-\ell\), each overlap term satisfies
\[
  N_m^tS_t^H(m)
  =O\!\left(
    e^{-n(h_{t,H}-(t/k)h_{k,H})}
  \right).
\]
The Poisson means stay bounded above and below by positive constants, so
the prefactor \(\min\{1,\lambda_{m,k}^{-1}\}\) is harmless.
Definition \eqref{eq:critical-gap} and
\eqref{eq:collision-tv-bound} now prove
\eqref{eq:critical-tv-rate}.

Along a fixed phase, the precise Perron asymptotic and the assumed rate
of convergence of the sample normalization give
\[
  \lambda_{m,k}
  =\frac{c_{k,r}\alpha^k}{k!}+O(e^{-\delta'(m-1)})
\]
with \(\delta'\) equal to the minimum of \(\eta\) and the relevant
Perron rate.  The total variation distance between Poisson laws of means
\(x,y\geq0\) is at most \(|x-y|\).  Combining this estimate with
\eqref{eq:critical-tv-rate} proves \eqref{eq:critical-target-rate}.
\end{proof}

\begin{corollary}[Criterion for a full-sequence critical limit]
\label{cor:full-sequence-criterion}
Fix \(k\geq2\) and suppose
\[
  N_m e^{-(m-1)h_{k,H}/k}\longrightarrow\alpha\in(0,\infty)
\]
without restricting the depth phase.  Then \(W_{m,k}\) has a
full-sequence weak limit if and only if the phase constants
\(c_{k,r}\) are equal for every phase visited infinitely often.  When
they are equal to \(c_k\), the limit is
\(\operatorname{Poisson}(c_k\alpha^k/k!)\).
\end{corollary}

\begin{proof}
Theorem~\ref{thm:periodic-collisions} gives the stated Poisson limit
along each phase.  Poisson distributions are equal exactly when their
means are equal.  Since there are finitely many phases, equality of all
subsequential limits is necessary and sufficient for full-sequence
convergence.
\end{proof}

\begin{corollary}[Phase-adjusted sample normalization]
\label{cor:phase-adjusted-normalization}
Fix \(k\geq2\), let
\(r_m:=m-1\bmod d\), and suppose
\begin{equation}
  N_m c_{k,r_m}^{1/k}
  e^{-(m-1)h_{k,H}/k}
  \longrightarrow\alpha\in(0,\infty).
  \label{eq:phase-adjusted-scale}
\end{equation}
Then, without restricting \(m\) to a depth phase,
\[
  W_{m,k}\Rightarrow
  \operatorname{Poisson}\!\left(\frac{\alpha^k}{k!}\right).
\]
\end{corollary}

\begin{proof}
Theorem~\ref{thm:periodic-renyi} and
\eqref{eq:phase-adjusted-scale} give
\[
  \binom{N_m}{k}S_k^H(m)
  =\frac1{k!}\left(
    N_mc_{k,r_m}^{1/k}e^{-(m-1)h_{k,H}/k}
  \right)^k(1+o(1))
  \longrightarrow\frac{\alpha^k}{k!}.
\]
The phase constants for all higher power sums remain bounded above and
below because there are finitely many phases.  The same strict-gap
argument used in Theorem~\ref{thm:periodic-collisions} makes every
overlap term vanish, and the intrinsic-mean form of that theorem gives
the result.
\end{proof}

\begin{remark}[Observable versus corrected scale]
The constants \(c_{k,r}\) are part of the survivor law, not nuisance
parameters introduced by the approximation.  Corollary
\ref{cor:phase-adjusted-normalization} shows that they can be absorbed
into a phase-dependent sample budget if a single limiting Poisson law is
desired.  If the sample budget is fixed independently of the survivor
phase, Corollary~\ref{cor:full-sequence-criterion} shows exactly when the
oscillation remains visible.
\end{remark}
