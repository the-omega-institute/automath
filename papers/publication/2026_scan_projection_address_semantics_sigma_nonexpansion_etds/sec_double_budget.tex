\section{Consequences of the killed-matrix formula}\label{sec:double-budget}

The results in this section are secondary consequences of the
killed-matrix calculation from Section~\ref{sec:open-system}.  They use
the same survivor path sum with different terminal weights and a standard
Chen--Stein estimate for collisions of independent survivor prefixes.

\begin{corollary}[Survivor R\'enyi pressure spectrum]
\label{cor:survivor-renyi-pressure}
Assume the hypotheses of
Theorem~\ref{thm:first-entry-escape-rate}.  For \(s>0\) and \(m\geq 1\),
let \(\mathcal L_m^H\) denote the set of admissible words
\(a=a_0\cdots a_{m-1}\in A^m\) with \(a_t\in\mathcal S\) for every
\(0\leq t<m\), and define
\[
  \mathcal Z_{H,m}(s)
  := \sum_{a\in\mathcal L_m^H}\nu_\varphi([a])^s.
\]
Let
\[
  u_s(i):=\pi_i^s,
  \qquad
  B_{H,s}(i,j):=K_{ij}^s\mathbf 1_{\{j\in\mathcal S\}}
  \qquad (i,j\in\mathcal S).
\]
Then
\[
  \mathcal Z_{H,m}(s)=u_s^{\mathsf T}B_{H,s}^{m-1}\mathbf 1
  \qquad (m\geq 1),
\]
and
\[
  \lim_{m\to\infty}
  -\frac1m\log \mathcal Z_{H,m}(s)
  = sP_Y(\varphi)-P_{Y_H}(s\varphi).
\]

If
\[
  \lambda_m^H(a)
  := \frac{\nu_\varphi([a])}{\nu_\varphi(\tau_H\ge m)},
  \qquad a\in\mathcal L_m^H,
\]
denotes the survivor-prefix law conditioned on survival to depth \(m\),
then
\[
  \lim_{m\to\infty}
  -\frac1m\log
  \sum_{a\in\mathcal L_m^H}\lambda_m^H(a)^s
  = sP_{Y_H}(\varphi)-P_{Y_H}(s\varphi).
\]
In particular,
\[
  h_{2,H}:=2P_{Y_H}(\varphi)-P_{Y_H}(2\varphi)
\]
is the R\'enyi-\(2\) entropy rate of the conditioned survivor law.
\end{corollary}

\begin{proof}
For \(a=a_0\cdots a_{m-1}\in\mathcal L_m^H\), the Markov property gives
\[
  \nu_\varphi([a])
  = \pi_{a_0}K_{a_0a_1}\cdots K_{a_{m-2}a_{m-1}},
\]
so
\[
  \nu_\varphi([a])^s
  = u_s(a_0)\,
    B_{H,s}(a_0,a_1)\cdots B_{H,s}(a_{m-2},a_{m-1}).
\]
Summing over all safe words of length \(m\) yields
\[
  \mathcal Z_{H,m}(s)=u_s^{\mathsf T}B_{H,s}^{m-1}\mathbf 1.
\]

Order \(\mathcal S\) so that \(\mathcal S_\infty\) comes first and
\(\mathcal T\) comes last.  Because \(B_{H,s}\) has the same support as
\(B_H\), it has block form
\[
  B_{H,s}=
  \begin{pmatrix}
    B_{\infty,s} & C_s\\
    0 & Q_s
  \end{pmatrix},
\]
where \(Q_s\) is nilpotent and \(B_{\infty,s}:=B_{H,s}|_{\mathcal S_\infty}\)
is primitive.  Hence
\[
  \rho(B_{H,s})=\rho(B_{\infty,s}).
\]

Let \(h>0\) be a Perron right eigenvector of the ambient transfer matrix
\[
  L_\varphi(i,j):=M_{ij}e^{\varphi(i,j)}.
\]
For \(i,j\in\mathcal S_\infty\),
\[
  K_{ij}
  = M_{ij}e^{\varphi(i,j)-P_Y(\varphi)}\frac{h_j}{h_i}.
\]
Define
\[
  D_s(i,i):=h_i^s,
  \qquad
  L_{s\varphi,H}(i,j)
  := M_{ij}e^{s\varphi(i,j)}\mathbf 1_{\{j\in\mathcal S_\infty\}}.
\]
Then
\[
  B_{\infty,s}
  = e^{-sP_Y(\varphi)}D_s^{-1}L_{s\varphi,H}D_s.
\]
Consequently
\[
  \rho(B_{H,s})
  = e^{-sP_Y(\varphi)}\rho(L_{s\varphi,H})
  = e^{P_{Y_H}(s\varphi)-sP_Y(\varphi)}.
\]
Since \(u_s\) and \(\mathbf 1\) are non-negative and non-zero on
\(\mathcal S_\infty\), Perron--Frobenius yields
\[
  \mathcal Z_{H,m}(s)
  = c_s\,\rho(B_{H,s})^{m-1}+O(\vartheta_s^m)
\]
for some \(c_s>0\) and \(0\leq \vartheta_s<\rho(B_{H,s})\).  This proves
the first limit.

For the conditioned law,
\[
  \sum_{a\in\mathcal L_m^H}\lambda_m^H(a)^s
  = \frac{\mathcal Z_{H,m}(s)}
         {\nu_\varphi(\tau_H\ge m)^s}.
\]
By Theorem~\ref{thm:first-entry-escape-rate},
\[
  -\frac1m\log \nu_\varphi(\tau_H\ge m)\longrightarrow \gamma_H(\varphi)
  = P_Y(\varphi)-P_{Y_H}(\varphi).
\]
Subtracting \(s\gamma_H(\varphi)\) from the first limit gives
\[
  sP_{Y_H}(\varphi)-P_{Y_H}(s\varphi).
\]
\end{proof}

\begin{theorem}[Higher-order collision thresholds]
\label{thm:survivor-higher-collisions}
Assume the hypotheses of
Corollary~\ref{cor:survivor-renyi-pressure} and suppose that
\(\card(\mathcal S_\infty)\geq 2\).  For \(s\geq 1\), set
\[
  S_s^H(m):=\sum_{a\in\mathcal L_m^H}\lambda_m^H(a)^s,
  \qquad
  h_{s,H}:=sP_{Y_H}(\varphi)-P_{Y_H}(s\varphi).
\]
For every integer \(s\geq 2\), there is an explicitly computable
constant \(c_{s,H}>0\) such that
\[
  S_s^H(m)
  = c_{s,H}e^{-(m-1)h_{s,H}}(1+O(\kappa_s^m))
  \qquad (m\to\infty)
\]
for some \(0<\kappa_s<1\).  Moreover, for all real \(1\leq s<t\),
\[
  h_{t,H}>\frac{t}{s}h_{s,H}.
\]

Fix an integer \(k\geq 2\), let
\((X_m^{H,(i)})_{1\leq i\leq N_m}\) be independent survivor prefixes of
depth \(m\) with common law \(\lambda_m^H\), and define the number of
\(k\)-fold collisions by
\[
  W_{m,k}:=
  \sum_{\substack{I\subseteq\{1,\ldots,N_m\}\\ \card(I)=k}}
  \mathbf 1_{\{X_m^{H,(i)}=X_m^{H,(j)}\text{ for all }i,j\in I\}}.
\]
If
\[
  \binom{N_m}{k}S_k^H(m)\longrightarrow\tau\in(0,\infty),
\]
then
\[
  W_{m,k}\ \Rightarrow\ \mathrm{Poisson}(\tau).
\]
No additional moment or overlap hypothesis is required.  More
precisely, for every \(\alpha\in(0,\infty)\),
\[
  N_m e^{-(m-1)h_{k,H}/k}\longrightarrow\alpha
  \quad\Longrightarrow\quad
  W_{m,k}\ \Rightarrow\
  \mathrm{Poisson}\!\left(\frac{c_{k,H}\alpha^k}{k!}\right).
\]
The corresponding collision threshold is sharp:
\[
  \PP(W_{m,k}\geq 1)\longrightarrow
  \begin{cases}
    0,
      &N_m e^{-(m-1)h_{k,H}/k}\longrightarrow 0,\\[3pt]
    1-\exp(-c_{k,H}\alpha^k/k!),
      &N_m e^{-(m-1)h_{k,H}/k}\longrightarrow\alpha\in(0,\infty),\\[3pt]
    1,
      &N_m e^{-(m-1)h_{k,H}/k}\longrightarrow\infty.
  \end{cases}
\]
Thus every fixed collision multiplicity has its own pressure-determined
birthday scale \(N_m\asymp e^{(m-1)h_{k,H}/k}\).
\end{theorem}

\begin{proof}
Write \(B_{\infty,s}:=B_{H,s}|_{\mathcal S_\infty}\) and
\(\rho_s:=\rho(B_{\infty,s})\).  With the ordering used in the proof of
Corollary~\ref{cor:survivor-renyi-pressure}, write
\[
  B_{H,s}=\begin{pmatrix}B_{\infty,s}&E_s\\0&Q_s\end{pmatrix}.
\]
Let \(r_s\) and \(\ell_s\) be right and left Perron eigenvectors of
\(B_{\infty,s}\), normalized by \(\ell_s^{\mathsf T}r_s=1\), and set
\[
  L_s^{\mathsf T}:=
  \bigl(\ell_s^{\mathsf T},
  \ell_s^{\mathsf T}E_s(\rho_sI-Q_s)^{-1}\bigr),
  \qquad
  a_s:=(u_{s,\infty}^{\mathsf T}r_s)
       (L_s^{\mathsf T}\mathbf 1).
\]
Here \(u_{s,\infty}\) is the restriction of \(u_s\) to
\(\mathcal S_\infty\).  The same block calculation as in the proof of
Theorem~\ref{thm:quasistationary-ambiguity}, followed by the spectral
gap for the primitive matrix \(B_{\infty,s}\), gives
\[
  \mathcal Z_{H,m}(s)
  =a_s\rho_s^{m-1}(1+O(\kappa_s^m))
\]
after increasing \(\kappa_s<1\) if necessary.  Since
\(S_s^H(m)=\mathcal Z_{H,m}(s)/\mathcal Z_{H,m}(1)^s\), it follows that
\[
  S_s^H(m)
  =\frac{a_s}{a_1^s}
   \left(\frac{\rho_s}{\rho_1^s}\right)^{m-1}
   (1+O(\kappa_s^m)).
\]
Corollary~\ref{cor:survivor-renyi-pressure} identifies
\(\rho_s/\rho_1^s=e^{-h_{s,H}}\), so
\[
  c_{s,H}=\frac{a_s}{a_1^s}>0.
\]

It remains to prove the strict pressure inequality.  Let \(r_1>0\) be
the Perron right eigenvector of \(B_{\infty,1}\), and define
\[
  P_{ij}:=\frac{B_{\infty,1}(i,j)r_1(j)}{\rho_1r_1(i)}.
\]
Then \(P\) is a primitive stochastic matrix.  If \(P^{\circ s}\)
denotes the entrywise \(s\)-th power, diagonal similarity gives
\[
  \rho(P^{\circ s})=\frac{\rho_s}{\rho_1^s}=e^{-h_{s,H}}.
\]
Fix \(1\leq s<t\), put \(d=t/s>1\), and choose \(x>0\) with
\(P^{\circ s}x=\rho(P^{\circ s})x\).  For every state \(i\),
\begin{align*}
  \bigl(P^{\circ t}x^{\circ d}\bigr)_i
  &=\sum_j(P_{ij}^s x_j)^d\\
  &\leq\left(\sum_jP_{ij}^s x_j\right)^d
   =\rho(P^{\circ s})^d x_i^d.
\end{align*}
Because \(P\) is primitive on at least two states, some row has at
least two positive entries.  The displayed inequality is strict in that
row.  The matrix \(P^{\circ t}\) is irreducible, so the strict form of
the Collatz--Wielandt theorem yields
\[
  \rho(P^{\circ t})<\rho(P^{\circ s})^{t/s}.
\]
Taking negative logarithms proves
\(h_{t,H}>(t/s)h_{s,H}\); in particular, \(h_{k,H}>0\).

For the Poisson limit, index the summands of \(W_{m,k}\) by the
\(k\)-subsets \(I\) of \(\{1,\ldots,N_m\}\).  Two summands are
independent when their index sets are disjoint.  If
\(\card(I\cap J)=\ell\in\{1,\ldots,k-1\}\), their joint expectation is
\[
  S_{2k-\ell}^H(m).
\]
The dependency-graph estimate of
\citet[Theorem~1]{ArratiaGoldsteinGordon1989} therefore bounds the total
variation distance from a Poisson law of mean
\(\lambda_m=\binom{N_m}{k}S_k^H(m)\) by a constant depending only on
\(k\) times
\[
  N_m^{2k-1}S_k^H(m)^2
  +\sum_{\ell=1}^{k-1}
     N_m^{2k-\ell}S_{2k-\ell}^H(m).
\]
If \(\lambda_m\to\tau\in(0,\infty)\), then \(N_m\to\infty\), the first
term tends to zero, and for \(t=2k-\ell\),
\[
  N_m^tS_t^H(m)
  =\bigl(N_m^kS_k^H(m)\bigr)^{t/k}
    \frac{S_t^H(m)}{S_k^H(m)^{t/k}}
  \longrightarrow 0
\]
by the strict pressure inequality and the precise exponential
asymptotics above.  This proves the Poisson limit.  The explicit critical
scaling follows from the formula for \(S_k^H(m)\).

In the subcritical regime, \(\EE W_{m,k}\to0\), so the first threshold
claim follows from Markov's inequality.  In the supercritical regime,
\(\lambda_m\to\infty\).  Independence of disjoint index sets and the
same overlap count give
\[
  \Var(W_{m,k})
  \leq \lambda_m+C(k)\sum_{\ell=1}^{k-1}
      N_m^{2k-\ell}S_{2k-\ell}^H(m).
\]
After division by \(\lambda_m^2\), every term tends to zero by the same
strict pressure inequality.  Chebyshev's inequality then gives
\(\PP(W_{m,k}=0)\to0\), completing the proof.
\end{proof}

\begin{proof}[Proof of Corollary~\ref{cor:main-renyi}]
The pressure identity is Corollary~\ref{cor:survivor-renyi-pressure}.
The collision-scale statement is
Theorem~\ref{thm:survivor-higher-collisions}.
\end{proof}
