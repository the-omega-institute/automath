\subsection{Structural absence}

\begin{definition}[Undefined, total, and null]\label{def:undef-total-null}
Imported from \cite[Definition~4.10]{HomologicalVisibilityGluingObstructions2026}: $\Undef_s(r)$, $\Tot_s(r)$, and $\Null_s(r)$ respectively mean that $r$ is not admitted, is admitted with a global section, and is admitted without a global section.
\end{definition}

\begin{definition}[The three null modes]\label{def:null-failure-modes}
Imported from \cite[Definition~4.11]{HomologicalVisibilityGluingObstructions2026}: the modes $\Null^{\mathrm{loc}}_s$, $\Null^{\mathrm{cmp}}_s$, and $\Null^{\mathrm{glue}}_s$ distinguish no local witnesses, local-but-incompatible witnesses, and compatible local witnesses without a global section; the present paper supplies the preorder-site common-refinement meanings of $\LocSec_s$, $\CompSec_s$, and $\Sec_s$ used by those names.
\end{definition}

\begin{proposition}[The three modes lie inside null]\label{prop:null-modes-imply-null}
Imported from \cite[Proposition~4.12]{HomologicalVisibilityGluingObstructions2026}: each of $\Null^{\mathrm{loc}}_s(r)$, $\Null^{\mathrm{cmp}}_s(r)$, and $\Null^{\mathrm{glue}}_s(r)$ implies $\Null_s(r)$ under state forcing.
\end{proposition}

\begin{proposition}[Mutual exclusivity of the null modes]\label{prop:null-modes-exclusive}
No two distinct null modes can hold simultaneously for the same reference at the same state.
\end{proposition}

\begin{proof}
The mode $\Null^{\mathrm{loc}}_s(r)$ contains $\neg\LocSec_s(r)$, while the other two contain $\LocSec_s(r)$. The mode $\Null^{\mathrm{cmp}}_s(r)$ contains $\neg\CompSec_s(r)$, while $\Null^{\mathrm{glue}}_s(r)$ contains $\CompSec_s(r)$.
\end{proof}

\begin{definition}[Decomposition decidability]\label{def:decomposition-decidability}
Imported from \cite[Definition~4.14]{HomologicalVisibilityGluingObstructions2026}: decomposition decidability for $r$ at $p$ means that the two local-object predicates $\LocSec_s(r)$ and $\CompSec_s(r)$ are decided at $p$; no realization-prestack or gerbe data are part of this hypothesis.
\end{definition}

\begin{proposition}[Null trichotomy]\label{prop:null-trichotomy}
See \cite[Proposition~4.15]{HomologicalVisibilityGluingObstructions2026}.
\end{proposition}

\begin{theorem}[Classification after refinement]\label{thm:null-classification-under-refinement}
Let $q\le p$ and suppose
\[
M,p\Vdash \Null_s(r).
\]
Assume the local-object layer is refinement-stable along $q\le p$ in the sense
of \Cref{def:refinement-stable-local-object-layer}. If $q$ has decomposition
decidability for $r$, then the refined state $q$ realizes a unique one of the
three null modes.
\end{theorem}

\begin{remark}
The sibling classification is \cite[Theorem~4.16]{HomologicalVisibilityGluingObstructions2026}.
The concrete increment here is the explicit refinement-stability interface
needed before transporting the local-object predicates from $p$ to $q$.
\end{remark}

\begin{proof}
The refinement-stability interface of
\Cref{def:refinement-stable-local-object-layer} supplies the refined
local-object frame, covers, and section predicates for the same reference at
$q$. By persistence of forcing, $M,q\Vdash \Null_s(r)$. The hypothesis gives
decomposition decidability at $q$, so \Cref{prop:null-trichotomy} applies at the
refined state $q$.
\end{proof}

\begin{proposition}[A finite model for the three absence modes]\label{prop:finite-three-mode-model}
There is a finite information-state model with one state $p$, one
referenceable sort $s$, and three admitted references
\[
r_{\mathrm{loc}},\quad r_{\mathrm{cmp}},\quad r_{\mathrm{glue}}:\mathsf{Ref}_s
\]
such that, at $p$,
\[
M,p\Vdash \Null^{\mathrm{loc}}_s(r_{\mathrm{loc}}),
\qquad
M,p\Vdash \Null^{\mathrm{cmp}}_s(r_{\mathrm{cmp}}),
\qquad
M,p\Vdash \Null^{\mathrm{glue}}_s(r_{\mathrm{glue}}).
\]
The local reference frame may be chosen to be a finite preorder site in which
the compatibility failure is detected by two distinct common refinements, so
the example does not use or require binary meets.
\end{proposition}

\begin{proof}
Take a context $\Gamma_p$ with one valuation and let
$p=(\Gamma_p,R_p)$ with $R_p$ that singleton. Interpret the three references as
three distinct admitted classes
\[
a_{\mathrm{loc}}=[r_{\mathrm{loc}}]_p,\qquad
a_{\mathrm{cmp}}=[r_{\mathrm{cmp}}]_p,\qquad
a_{\mathrm{glue}}=[r_{\mathrm{glue}}]_p.
\]
Let $\mathcal B_{p,s}$ be the disjoint union of the following three finite
preorder components, and take the displayed nonempty families, together with
identity families and their transitive refinements, as the covers.

For $a_{\mathrm{loc}}$, use only the root object $a_{\mathrm{loc}}$ and its
identity cover. Put
\[
F(a_{\mathrm{loc}})=\varnothing .
\]
Then no covering family of $a_{\mathrm{loc}}$ has local sections on all of its
members, so $\LocSec_s(r_{\mathrm{loc}})$ fails. Since the reference is
admitted, \Cref{def:null-failure-modes} gives
\[
M,p\Vdash \Null^{\mathrm{loc}}_s(r_{\mathrm{loc}}).
\]

For $a_{\mathrm{cmp}}$, use objects
\[
u,\ v,\ c_0,\ c_1,\ a_{\mathrm{cmp}}
\]
with preorder arrows
\[
c_\epsilon\sqsubseteq u\sqsubseteq a_{\mathrm{cmp}},
\qquad
c_\epsilon\sqsubseteq v\sqsubseteq a_{\mathrm{cmp}}
\quad(\epsilon=0,1),
\]
and no imposed comparison between $c_0$ and $c_1$. Let
$\{u\to a_{\mathrm{cmp}},v\to a_{\mathrm{cmp}}\}$ be the only non-identity
root cover in this component; below $u$, $v$, $c_0$, and $c_1$, take only the
identity covers
\[
\{u\to u\},\qquad \{v\to v\},\qquad \{c_0\to c_0\},\qquad \{c_1\to c_1\}.
\]
This satisfies the pullback-stability clause because
the indexing sets over cover members not hit by the chosen lower object are
empty; the covering family itself remains nonempty. Thus the two objects
$c_0,c_1$ are common refinements used for the compatibility test, not a chosen
binary meet and not extra cover generators below $u$ or $v$.
Define the presheaf values by
\[
F(u)=\{\sigma_u\},\qquad
F(v)=\{\sigma_v\},\qquad
F(c_0)=F(c_1)=\{0,1\},\qquad
F(a_{\mathrm{cmp}})=\varnothing ,
\]
with restrictions
\[
\res_{c_0}^{u}(\sigma_u)=0,\quad
\res_{c_1}^{u}(\sigma_u)=0,\qquad
\res_{c_0}^{v}(\sigma_v)=0,\quad
\res_{c_1}^{v}(\sigma_v)=1 .
\]
The cover $\{u,v\}$ has local sections, so $\LocSec_s(r_{\mathrm{cmp}})$ holds.
However, the only possible local family on that cover is
$(\sigma_u,\sigma_v)$, and its restrictions disagree at the common refinement
$c_1$:
\[
\res_{c_1}^{u}(\sigma_u)=0\ne 1=\res_{c_1}^{v}(\sigma_v).
\]
The only root cover with local sections is therefore $\{u,v\}$, and its only
possible local family fails the common-refinement matching condition at $c_1$.
The identity cover cannot witness compatibility since
$F(a_{\mathrm{cmp}})=\varnothing$. Therefore $\CompSec_s(r_{\mathrm{cmp}})$
fails while $\LocSec_s(r_{\mathrm{cmp}})$ holds, and
\[
M,p\Vdash \Null^{\mathrm{cmp}}_s(r_{\mathrm{cmp}}).
\]
This is the finite-preorder increment over the sibling meet-semilattice
example: compatibility is tested against all common refinements
$c\sqsubseteq u,v$, not against a chosen meet $u\wedge v$.

For $a_{\mathrm{glue}}$, use objects
\[
x,\ y,\ d,\ a_{\mathrm{glue}}
\]
with $d\sqsubseteq x,y\sqsubseteq a_{\mathrm{glue}}$ and cover
$\{x\to a_{\mathrm{glue}},y\to a_{\mathrm{glue}}\}$. Define
\[
F(x)=F(y)=F(d)=\{*\},\qquad F(a_{\mathrm{glue}})=\varnothing ,
\]
with the unique possible restriction maps. The pair $(*,*)$ is compatible on
every common refinement of $x$ and $y$, so $\CompSec_s(r_{\mathrm{glue}})$
holds. Since $F(a_{\mathrm{glue}})=\varnothing$, $\Sec_s(r_{\mathrm{glue}})$
fails. Therefore
\[
M,p\Vdash \Null^{\mathrm{glue}}_s(r_{\mathrm{glue}}).
\]

The three components are finite and disjoint, so the prescriptions assemble
into a single finite preorder site and a single contravariant functor
$F_{p,s}:\mathcal B_{p,s}^{\mathrm{op}}\to\mathbf{Set}$. The cover axioms are
exactly identity and closure under the displayed pullback-stable and transitive
refinements, so \Cref{prop:local-reference-pretopology} applies.
\end{proof}

\subsection{Forcing necessity of the enrichment}

The three null modes show that information-state forcing distinguishes situations invisible to pointwise semantics. The following result makes this precise: the gluing predicates introduced at the local-object layer cannot be reduced to the lower layers.

The restriction on formulas is part of the claim. Work in the constant-free lower language except for the displayed free variable. Thus the symbols $r_1,\ldots,r_4$ may be used as elements substituted for that variable in the external comparison
\[
M,\rho\models\varphi(r_i),
\]
but they are not constants of the lower signature and cannot occur inside $\varphi$. The local-object functor, admitted-reference classes, sectionability predicates, and absence predicates are outside the lower reduct.

\begin{theorem}[Pointwise irreducibility of gluing predicates]\label{thm:pointwise-irreducibility}
\leanverified{paper_conservative_extension_pointwise_irreducibility_focused}
There exist a model $M$, a state $p=(\Gamma_p,R_p)$, and admitted reference
elements
\[
r_1,r_2,r_3,r_4:\mathsf{Ref}_s
\]
of the same sort such that:
\begin{enumerate}[label=(\roman*),nosep]
\item for every $\rho\in R_p$ and every $\mathrm{Form}_1$-formula $\varphi(x)$ with one free variable $x$, containing none of the local-object or absence predicates
\[
\Adm_s,\quad \LocSec_s,\quad \CompSec_s,\quad \Sec_s,\quad
\Null_s,\quad \Null^{\mathrm{loc}}_s,\quad
\Null^{\mathrm{cmp}}_s,\quad \Null^{\mathrm{glue}}_s,
\]
and containing no occurrence of the distinguished reference names $r_1,r_2,r_3,r_4$ other than as the externally substituted value of its free variable,
\[
M,\rho\models\varphi(r_1)
\quad\Longleftrightarrow\quad
M,\rho\models\varphi(r_2);
\]
\item under the same hypotheses on $\rho$ and $\varphi(x)$,
\[
M,\rho\models\varphi(r_3)
\quad\Longleftrightarrow\quad
M,\rho\models\varphi(r_4);
\]
\item $M,p\Vdash \CompSec_s(r_1)\land \Sec_s(r_1)$;
\item $M,p\Vdash \CompSec_s(r_2)\land \neg\Sec_s(r_2)$;
\item $M,p\Vdash \CompSec_s(r_3)$;
\item $M,p\Vdash \LocSec_s(r_4)\land \neg\CompSec_s(r_4)$.
\end{enumerate}
In particular, no formula of the information-state forcing language that is invariant under $\mathrm{Form}_1$-automorphisms, contains none of the local-object or absence predicates displayed above, and does not name the distinguished references except through its free variable can define the predicates
\[
\CompSec_s,
\qquad
\Sec_s,
\qquad
\Null^{\mathrm{glue}}_s.
\]
\end{theorem}

\begin{proof}
Take a single state $p$ and one referenceable sort $s$. In the ambient model
choose four reference elements
\[
r_1,r_2,r_3,r_4:\mathsf{Ref}_s
\]
with distinct reference classes
\[
[r_1]_p,\ [r_2]_p,\ [r_3]_p,\ [r_4]_p.
\]
We now define the lower-language reduct explicitly. Let $\mathbb L_{\mathrm{low}}$
contain the reference sort $\mathsf{Ref}_s$, equality, and any additional lower
sorts, functions, and relations already present in the information-state
fragment. It contains no constant symbols naming $r_1,\ldots,r_4$, no predicate
for admitted-reference classes, and none of the local-object or absence
predicates. Interpret
\[
D_{\mathsf{Ref}_s}^M=\{e_1,e_2,e_3,e_4\}
\]
and use the meta-notation $r_k=e_k$ only for the four elements at which formulas
are evaluated. The context $\Gamma_p$ consists only of variables of the
irrelevant lower sorts, and $R_p$ is any nonempty set of their valuations; in
particular no variable of $\Gamma_p$ has sort $\mathsf{Ref}_s$ and no assignment
in $R_p$ stores any of the elements $e_k$. Interpret every additional lower
relation and function so that the two permutations
\[
\alpha=(e_1\ e_2),
\qquad
\beta=(e_3\ e_4)
\]
on $D_{\mathsf{Ref}_s}$, extended by the identity on all other lower sorts, are
automorphisms of the $\mathbb L_{\mathrm{low}}$-reduct. For instance, all lower
relations involving $\mathsf{Ref}_s$ may be interpreted as unions of
$\langle\alpha,\beta\rangle$-orbits and all lower functions may be chosen
equivariantly; equality is automatically preserved. The admissibility and
local-object predicates are not in this reduct.

Equivalently, the lower reduct has a two-pair orbit decomposition
\[
\{e_1,e_2\}\sqcup\{e_3,e_4\},
\]
and the later local-object data are attached only after this reduct has been
fixed. Because admitted classes and local objects are not part of
$\mathbb L_{\mathrm{low}}$, changing the local-object component over $e_i$ does
not change the lower structure.

These automorphisms satisfy
\[
\alpha(r_1^M)=r_2^M,
\qquad
\alpha(r_2^M)=r_1^M,
\]
\[
\beta(r_3^M)=r_4^M,
\qquad
\beta(r_4^M)=r_3^M,
\]
where $r_k^M$ abbreviates the element $e_k$ for this displayed evaluation. They
preserve every realization $\rho\in R_p$ because they are the identity on
the sorts occurring in $\Gamma_p$. By invariance of first-order truth under
automorphisms, every $\mathrm{Form}_1$-formula $\varphi(x)$ satisfying the
restriction in~\textup{(i)} therefore obeys
\[
M,\rho\models\varphi(r_1)
\quad\Longleftrightarrow\quad
M,\rho\models\varphi(r_2)
\]
for every $\rho\in R_p$, which proves \textup{(i)}. The same argument with $\beta$ proves \textup{(ii)}.

Write $a_k=[r_k]_p$ for $1\le k\le 4$ in the separate admitted-reference layer,
and let the local reference frame be the disjoint union of four rooted
components, one above each $a_k$. This enrichment is not visible in
$\mathbb L_{\mathrm{low}}$. In each component choose a cover
\[
\{u_k\to a_k,v_k\to a_k\}
\]
with common refinement $c_k$. Define a single local object functor $F_{p,s}$ on the disjoint union by specifying it componentwise.

On the $a_1$-component, set
\[
F_{p,s}(u_1)=F_{p,s}(v_1)=F_{p,s}(c_1)=F_{p,s}(a_1)=\{*\},
\]
with all restrictions equal to the identity on $\{*\}$. The unique local family $(*,*)$ is compatible, and $*\in F_{p,s}(a_1)$ is a global section. Hence \textup{(iii)}.

On the $a_2$-component, set
\[
F_{p,s}(u_2)=F_{p,s}(v_2)=F_{p,s}(c_2)=\{*\},
\qquad
F_{p,s}(a_2)=\varnothing,
\]
with all restrictions to $c_2$ equal to the identity on $\{*\}$. The local family $(*,*)$ is compatible on the overlap, but $F_{p,s}(a_2)=\varnothing$. Hence \textup{(iv)}.

On the $a_3$-component, use the same definition as on the $a_1$-component. Then $r_3$ is compatibly locally sectionable, so \textup{(v)} holds.

On the $a_4$-component, set
\[
F_{p,s}(u_4)=\{\sigma_u\},\quad
F_{p,s}(v_4)=\{\sigma_v\},\quad
F_{p,s}(c_4)=\{0,1\},\quad
F_{p,s}(a_4)=\varnothing,
\]
with restriction maps
\[
\res_{c_4}^{u_4}(\sigma_u)=0,
\qquad
\res_{c_4}^{v_4}(\sigma_v)=1.
\]
Then local sections exist on the chosen cover, but the only local family is incompatible on the overlap. Hence \textup{(vi)}.

Finally, stipulate
\[
M,p\Vdash \Adm_s(r_1)\land \Adm_s(r_2)\land \Adm_s(r_3)\land \Adm_s(r_4)
\]
and identify the four reference classes with $a_1,a_2,a_3,a_4$ respectively.

Now let $\theta(x)$ be any lower-language information-state forcing formula satisfying the restrictions in the final sentence of the theorem. Because $\mathbb{L}_1$-forcing is flat over the realizations in $R_p$, the pointwise equivalences in \textup{(i)} and \textup{(ii)} imply
\[
M,p\Vdash \theta(r_1)\leftrightarrow\theta(r_2)
\qquad\text{and}\qquad
M,p\Vdash \theta(r_3)\leftrightarrow\theta(r_4).
\]
But \textup{(iii)} and \textup{(iv)} separate $\Sec_s$ and $\Null^{\mathrm{glue}}_s$, while \textup{(v)} and \textup{(vi)} separate $\CompSec_s$. Therefore no such $\theta$ can define any of the three predicates.
\end{proof}
