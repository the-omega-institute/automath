\section{Information States and Flat Team-Style Forcing}\label{sec:information-states}

This section defines the basic semantics used throughout. A state records a context together with the realizations still compatible with the information available.

\subsection{States, refinement, and forcing}

Fix a many-sorted first-order signature
\[
\Sigma=(S,F,R).
\]
For a model $M$, write $D_s^M$ for the interpretation of a sort $s\in S$. A \emph{context} is a finite typed list
\[
\Gamma=(x_1:s_1,\dots,x_n:s_n),
\]
and its realization set in $M$ is
\[
\Val_M(\Gamma):=\{\rho\mid \rho(x_i)\in D_{s_i}^M\text{ for }1\le i\le n\}.
\]
If $\Gamma\preceq\Delta$, every $\sigma\in\Val_M(\Delta)$ restricts to an element $\sigma|_{\Gamma}\in\Val_M(\Gamma)$.

\begin{definition}[Information state]\label{def:information-state}
See \cite[Definition~3.1]{HomologicalVisibilityGluingObstructions2026}; the same convention and notation $\St(M)$ are used here.
\end{definition}

\begin{definition}[Refinement]\label{def:refinement}
See \cite[Definition~3.2]{HomologicalVisibilityGluingObstructions2026}; throughout this paper $q\le p$ denotes that refinement relation.
\end{definition}

Refinement therefore combines two monotonicities: the context may become richer, and the set of still-admissible realizations may become smaller.

\begin{definition}[State forcing]\label{def:state-forcing}
See \cite[Definition~3.3]{HomologicalVisibilityGluingObstructions2026}; the forcing symbol $M,p\Vdash\varphi$ has exactly that imported meaning below.
\end{definition}

\begin{proposition}[Flat team-style reading of lower-layer forcing]\label{prop:flat-team-reading}
Let $p=(\Gamma_p,R_p)$ be an information state. If one reads $R_p$ as a nonempty team of $\Gamma_p$-valuations, then for every lower-layer formula $\varphi$ well formed in context $\Gamma_p$,
\[
M,p\Vdash\varphi
\quad\Longleftrightarrow\quad
\forall \rho\in R_p\;(M,\rho\models\varphi).
\]
\leanverified{paper_information_state_flat_team_reading_seeds}
\end{proposition}

\begin{proof}
This is exactly \Cref{def:state-forcing}.
\end{proof}

\begin{definition}[State theory]\label{def:state-theory}
For $p\in\St(M)$, define
\[
\Th_M(p):=\{\varphi\mid M,p\Vdash\varphi\}.
\]
\end{definition}

\begin{definition}[Decidedness]\label{def:decidedness}
See \cite[Definition~3.5]{HomologicalVisibilityGluingObstructions2026}; the notation $M,p\Downarrow\varphi$ is imported from that definition.
\end{definition}

\begin{proposition}[Persistence of forcing]\label{prop:forcing-persistence}
See \cite[Proposition~3.6]{HomologicalVisibilityGluingObstructions2026}.
\end{proposition}

\begin{corollary}[Monotone growth of state theories]\label{cor:state-theory-monotonicity}
If $q\le p$, then $\Th_M(p)\subseteq \Th_M(q)$.
\leanverified{paper_information_state_state_theory_monotonicity_focused_seeds}
\end{corollary}

\begin{proof}
Immediate from \Cref{prop:forcing-persistence}.
\end{proof}

\begin{proposition}[Singleton conservativity]\label{prop:singleton-conservativity}
See \cite[Proposition~3.8]{HomologicalVisibilityGluingObstructions2026}.
\end{proposition}

\subsection{Lifted consequence}

\begin{definition}[State semantic consequence]\label{def:state-semantic-consequence}
Let $\Phi$ be a set of formulas and let $\psi$ be a formula, all well formed in
a common context $\Gamma$. We write
\[
\Phi\models_{\St}\psi
\]
if for every model $M$ and every information state
$p=(\Gamma,R_p)\in\St(M)$,
\[
\bigl(\forall\varphi\in\Phi\; M,p\Vdash\varphi\bigr)
\quad\Longrightarrow\quad
M,p\Vdash\psi .
\]
For the empty set of premises we write $\models_{\St}\psi$. The corresponding
ordinary pointwise consequence relation is denoted $\Phi\models\psi$:
for every model $M$ and every valuation $\rho\in\Val_M(\Gamma)$,
\[
\bigl(\forall\varphi\in\Phi\; M,\rho\models\varphi\bigr)
\quad\Longrightarrow\quad
M,\rho\models\psi .
\]
\end{definition}

\begin{proposition}[Lifted soundness of pointwise reasoning]\label{prop:lifted-pointwise-soundness}
Let $\Phi$ be a set of lower-layer formulas and let $\psi$ be a lower-layer
formula in the same context $\Gamma$. If
\[
\Phi\models\psi,
\]
then
\[
\Phi\models_{\St}\psi .
\]
Equivalently, every pointwise sound inference rule for lower-layer formulas
remains sound after the formulas are evaluated by information-state forcing.
\end{proposition}

\begin{proof}
Let $M$ be a model and let $p=(\Gamma,R_p)\in\St(M)$ satisfy
$M,p\Vdash\varphi$ for every $\varphi\in\Phi$. To prove $M,p\Vdash\psi$, choose
an arbitrary $\rho\in R_p$. By \Cref{def:state-forcing}, the assumptions give
$M,\rho\models\varphi$ for every $\varphi\in\Phi$. Since
$\Phi\models\psi$ pointwise, $M,\rho\models\psi$. The choice of $\rho$ was
arbitrary, so another application of \Cref{def:state-forcing} yields
$M,p\Vdash\psi$.
\end{proof}

\Cref{prop:lifted-pointwise-soundness} is the bridge from ordinary satisfaction to state forcing. It shows that familiar sound rules continue to apply once formulas are evaluated against all realizations compatible with a state.

\begin{remark}[Where the new semantics begins]
\Cref{prop:flat-team-reading} explains the connection with flat team-style and inquisitive support semantics: lower-layer formulas are evaluated pointwise across a nonempty information state. The genuinely new phenomena studied in this paper begin only after the local-object predicates are added, because gluing failure and its visible/blind decomposition are not properties of the flat lower layer alone.
\end{remark}
