\subsection{Sheafified global sections and typed readouts}

\begin{definition}[Sheafified global sections]\label{def:visible-value-classes}
Let $r:\mathsf{Ref}_s$ be admissible at $p$, and write $a=[r]_p$. A \emph{plus-construction representative} of a section at $r$ is a local family of sections on some cover of $a$ in the pullback-based thin site of \Cref{def:local-reference-frame}; concretely, it is data
\[
\Bigl(\{a_i\to a\}_{i\in I},(\sigma_i)_{i\in I}\Bigr)
\]
with each $\sigma_i\in F_{p,s}(a_i)$ and with agreement after a covering
refinement of every pullback overlap $a_i\times_a a_j$. Two such
presentations are \emph{equivalent in the plus construction} when, after pulling both
families back to a common refinement of their covers and then to the
overlap refinements required by the plus construction, the resulting
local sections agree pointwise. A \emph{sheafified global section} is an
equivalence class under this refinement relation, and the set of all such
classes is denoted by $\Vis_{p,s}(r)$. If
$F_{p,s}|_{\mathcal C_a}$ is separated on the slice, the local-overlap
refinements in this definition are unnecessary, so plus-construction representatives
are the strict matching families of \Cref{def:existence-levels}(ii).
\end{definition}


\begin{proposition}[Sheafified global sections as sections of the sheaf of connected components]\label{thm:visible-value-components}
Assume that $r$ admits a banded groupoid lift in the sense of \Cref{def:banded-groupoid-lift} over the pullback-based slice site attached to \Cref{def:local-reference-frame}. Then the sheafified global sections of $r$ are exactly the sections at $a$ of the sheafification of the presheaf of local sections, and equivalently the sections at $a$ of the sheaf of connected components of $\mathfrak L_r$:
\[
\pi_0(\mathfrak L_r)(a)\cong \Vis_{p,s}(r)\cong \bigl(a_{p,s}F_{p,s}\bigr)(a).
\]
\end{proposition}

\begin{proof}
We use the plus--plus construction, not the one-step separated
presheaf of strict matching families. Let $F=F_{p,s}|_{\mathcal C_a}$.
The usual plus construction sends a presheaf $E$ to the quotient of
strict $E$-matching families by equality after common refinement, and
$aF=(F^+)^+$. Therefore a section of $(aF)(a)$ is represented first by a
strict $F^+$-matching family on a cover of $a$. Choosing plus
representatives for the entries of this family and then pulling them back
to the composite cover gives, by
\Cref{lem:strict-sheafification-representatives}, a family of
$F$-sections whose pairwise overlap restrictions agree only after further
covers of those overlaps. This flattening step is essential for an
arbitrary presheaf: no equality in $F$ on the original overlaps is being
assumed.

Conversely, suppose
\[
\Bigl(\{a_i\to a\}_{i\in I},(\sigma_i)_{i\in I}\Bigr)
\]
is a plus-construction representative in the sense of
\Cref{def:visible-value-classes}. For each pullback
$a_{ij}:=a_i\times_a a_j$ choose a cover on which the two restrictions of
$\sigma_i$ and $\sigma_j$ agree. These chosen local equalities are
precisely the data saying that the images of the $\sigma_i$ in
$F^+(a_i)$ form a strict $F^+$-matching family. Applying the second plus
construction gives a well-defined element of $(aF)(a)$. Refining the
cover of $a$, or refining the covers of the overlaps used to witness
compatibility, does not change this element because those refinements are
exactly the equivalence relation in the first or second plus construction.

It follows that evaluation of the sheafification on the object $a$ is the
quotient
\[
J_a/\!\sim,
\]
where $J_a$ consists of covers of $a$ equipped with $F$-sections that are
compatible after local refinement on every pullback overlap, and where
two such representatives are equivalent exactly when their pullbacks to a
common refinement agree after the further local refinements required in
the two plus steps. This is the equivalence relation imposed in
\Cref{def:visible-value-classes}; hence
\[
\Vis_{p,s}(r)\cong \bigl(a_{p,s}F_{p,s}\bigr)(a).
\]
The second identification is \Cref{prop:stackification-components} evaluated at $a$.
\end{proof}


\begin{definition}[Typed readout]\label{def:typed-readout}
Let $r:\mathsf{Ref}_s$ and a state $p$ be given. Define the \emph{typed readout} of $r$ at $p$ by
\[
\Read_{p,s}(r):=
\begin{cases}
\Inadm, & M,p\nVdash \Adm_s(r),\\[0.5ex]
v, & \Vis_{p,s}(r)=\{v\}\ \text{for some sheafified global section }v,\\[0.5ex]
\Indet, & M,p\Vdash \Adm_s(r)\ \text{and}\ \Vis_{p,s}(r)\neq\{v\}\ \text{for every }v.
\end{cases}
\]
Thus a determined readout occurs exactly when the admissible parameter carries a unique sheafified global section; $\Inadm$ denotes an inadmissible parameter and $\Indet$ denotes the non-unique-or-absent case.
\end{definition}

\begin{proposition}[Address before value]\label{prop:address-before-value}
If
\[
\Read_{p,s}(r)=v
\qquad\text{with}\qquad
v\notin\{\Inadm,\Indet\},
\]
then
\[
M,p\Vdash \Adm_s(r)
\qquad\text{and}\qquad
M,p\Vdash \CompSec_s(r).
\]
\end{proposition}

\begin{proof}
By \Cref{def:typed-readout}, a readout outside $\{\Inadm,\Indet\}$ occurs only in the singleton case
\[
\Vis_{p,s}(r)=\{v\}.
\]
Sheafified global sections are defined only for admissible parameters, so $M,p\Vdash \Adm_s(r)$. The section $v$ has a matching-family representative, hence $M,p\Vdash \CompSec_s(r)$ by \Cref{def:visible-value-classes}.
\end{proof}

\begin{corollary}[Criterion for a determined readout]\label{cor:nonnull-readout-criterion}
\[
\Read_{p,s}(r)=v
\quad\Longleftrightarrow\quad
\Vis_{p,s}(r)=\{v\}.
\]
Equivalently,
\[
\Read_{p,s}(r)=\Indet
\]
if and only if $r$ is admissible and either $\Vis_{p,s}(r)=\varnothing$ or $|\Vis_{p,s}(r)|>1$.
\end{corollary}

\begin{proof}
This is immediate from \Cref{def:typed-readout}.
\end{proof}

\begin{proposition}[Refinement stability of defined readouts]\label{prop:typed-readout-persistence}
Let $q\le p$, and suppose
\[
\Read_{p,s}(r)=v
\qquad\text{with}\qquad
v\notin\{\Inadm,\Indet\}.
\]
Then
\[
\Read_{q,s}(r)=v.
\]
\end{proposition}

\begin{proof}
Since $v\notin\{\Inadm,\Indet\}$, the readout at $p$ is represented by a unique sheafified global section. By the refinement-conservation hypothesis~\ref{hyp:refinement-conservation}, matching families restrict functorially from $q$ to $p$, and a uniquely determined section at $p$ cannot split under further refinement. Hence the same section $v$ remains unique at $q$, so \Cref{def:typed-readout} gives $\Read_{q,s}(r)=v$.
\end{proof}

\subsection{Failures of global existence}

\begin{definition}[Inadmissibility and existence of global sections]\label{def:undef-total-null}
For an element $r:\mathsf{Ref}_s$ at a state $p$, define
\begin{align*}
M,p\Vdash \Inadm_s(r)
&\quad:\Longleftrightarrow\quad
M,p\nVdash \Adm_s(r),\\
M,p\Vdash \HasGlob_s(r)
&\quad:\Longleftrightarrow\quad
M,p\Vdash \Adm_s(r)\land \Sec_s(r),\\
M,p\Vdash \NoGlob_s(r)
&\quad:\Longleftrightarrow\quad
M,p\Vdash \Adm_s(r)\land \neg\Sec_s(r).
\end{align*}
Thus admissible parameters are partitioned according to whether a global section exists, while $\Inadm_s$ records inadmissibility.
\end{definition}

\begin{definition}[Three failures of global existence]\label{def:null-failure-modes}
For an admissible parameter $r:\mathsf{Ref}_s$ at a state $p$, define
\begin{align*}
\NoLoc_s(r)
&\quad:\Longleftrightarrow\quad
\Adm_s(r)\land \neg\LocSec_s(r),\\
\NoMatch_s(r)
&\quad:\Longleftrightarrow\quad
\Adm_s(r)\land \LocSec_s(r)\land \neg\CompSec_s(r),\\
\NoAmalg_s(r)
&\quad:\Longleftrightarrow\quad
\Adm_s(r)\land \CompSec_s(r)\land \neg\Sec_s(r).
\end{align*}
They record, respectively, no local sections, local sections but no matching family, and a matching family without a global amalgamation.
\end{definition}

\begin{proposition}[The three cases imply failure of global existence]\label{prop:null-modes-imply-null}
For every admissible parameter $r:\mathsf{Ref}_s$ at a state $p$,
\[
M,p\Vdash \NoLoc_s(r)\ \lor\ \NoMatch_s(r)\ \lor\ \NoAmalg_s(r)
\quad\Longrightarrow\quad
M,p\Vdash \NoGlob_s(r).
\]
\end{proposition}

\begin{proof}
If $M,p\Vdash \NoLoc_s(r)$, then $r$ is admissible and $\LocSec_s(r)$ fails. By \Cref{prop:global-implies-compatible-local}, existence of a global section would imply local nonemptiness, so $\Sec_s(r)$ fails and hence $M,p\Vdash \NoGlob_s(r)$.

If $M,p\Vdash \NoMatch_s(r)$, then $r$ is admissible, local sections exist, but no matching family does. Again \Cref{prop:global-implies-compatible-local} shows that $\Sec_s(r)$ would imply $\CompSec_s(r)$, so $\Sec_s(r)$ fails and $M,p\Vdash \NoGlob_s(r)$.

If $M,p\Vdash \NoAmalg_s(r)$, then $r$ is admissible and $\Sec_s(r)$ fails by definition, so $M,p\Vdash \NoGlob_s(r)$.
\end{proof}

\begin{proposition}[Mutual exclusivity of the three cases]\label{prop:null-modes-exclusive}
No two distinct failure cases can hold simultaneously for the same parameter at the same state.
\end{proposition}

\begin{proof}
The case $\NoLoc_s(r)$ contains $\neg\LocSec_s(r)$, while the other two contain $\LocSec_s(r)$. The case $\NoMatch_s(r)$ contains $\neg\CompSec_s(r)$, while $\NoAmalg_s(r)$ contains $\CompSec_s(r)$.
\end{proof}

\begin{definition}[Decidability of the local-section predicates]\label{def:decomposition-decidability}
Let $p$ be a state and $s\in S_{\mathrm{ref}}$. The local-section predicates are \emph{decided at $p$ for the sort $s$} if, for every admissible parameter
\[
r:\mathsf{Ref}_s
\]
at $p$, both predicates
\[
\LocSec_s(r)
\qquad\text{and}\qquad
\CompSec_s(r)
\]
are decided at $p$ in the sense of \Cref{def:decidedness}.
\end{definition}

\begin{proposition}[Trichotomy for failure of global existence]\label{prop:null-trichotomy}
Assume that the local-section predicates are decided at $p$ for the sort $s$, and let
\[
r:\mathsf{Ref}_s
\]
be admissible at $p$. If
\[
M,p\Vdash \NoGlob_s(r),
\]
then exactly one of the three formulas
\[
\NoLoc_s(r),\qquad
\NoMatch_s(r),\qquad
\NoAmalg_s(r)
\]
holds at $p$.
\end{proposition}

\begin{proof}
Since
\[
M,p\Vdash \NoGlob_s(r),
\]
the parameter $r$ is admissible and
\[
M,p\nVdash \Sec_s(r).
\]
Because the local-section predicates are decided at $p$ for the sort $s$, the formula $\LocSec_s(r)$ is decided.

If
\[
M,p\nVdash \LocSec_s(r),
\]
then the definition of $\NoLoc_s(r)$ gives
\[
M,p\Vdash \NoLoc_s(r).
\]

Assume instead that
\[
M,p\Vdash \LocSec_s(r).
\]
The formula $\CompSec_s(r)$ is also decided. If
\[
M,p\nVdash \CompSec_s(r),
\]
then
\[
M,p\Vdash \NoMatch_s(r)
\]
by definition. If
\[
M,p\Vdash \CompSec_s(r),
\]
then, together with
\[
M,p\nVdash \Sec_s(r),
\]
the definition of $\NoAmalg_s(r)$ gives
\[
M,p\Vdash \NoAmalg_s(r).
\]
Hence at least one failure case holds.

By \Cref{prop:null-modes-exclusive}, no two distinct cases can hold simultaneously. Therefore exactly one of the three cases holds.
\end{proof}

\begin{theorem}[Classification after refinement]\label{thm:null-classification-under-refinement}
Let
\[
q\le p,
\]
assume that the local-section predicates are decided at $q$ for the sort $s$, and let
\[
r:\mathsf{Ref}_s
\]
be admissible at $q$. If
\[
M,q\Vdash \NoGlob_s(r),
\]
then exactly one of the formulas
\[
\NoLoc_s(r),\qquad
\NoMatch_s(r),\qquad
\NoAmalg_s(r)
\]
holds at $q$.
\end{theorem}

\begin{proof}
Apply \Cref{prop:null-trichotomy} at the refined state $q$.
\end{proof}

\subsection{Lower-fragment separation}

The three failure cases show that universal team satisfaction distinguishes situations not expressible in the pointwise fragment. By \Cref{prop:chain-instantiation}, the parameter sort $\mathsf{Ref}_s$ already belongs to the $\Sigma_1$-signature, whereas the predicates
\[
\Adm_s,\qquad \LocSec_s,\qquad \CompSec_s,\qquad \Sec_s
\]
belong to $\Sigma_2\setminus\Sigma_1$. The lower framework can therefore quantify over parameter elements but cannot mention the local-section predicates themselves. To keep track of the formulas allowed in that comparison, we fix the following terminology.

\begin{definition}[Automorphism-invariant one-variable fragment of the $\Sigma_1$-reduct]\label{def:lower-comparison-class}
Let $s\in S_{\mathrm{ref}}$. The automorphism-invariant one-variable fragment $\mathcal{C}_{s}^{\mathrm{inv}}$ consists of the constant-free $\Sigma_1$-formulas with exactly one free variable of sort $\mathsf{Ref}_s$, no parameters of that sort, and any number of bound variables from the ambient pointwise signature. A formula $\varphi(x)\in\mathcal{C}_{s}^{\mathrm{inv}}$ is evaluated inside the constant-free $\Sigma_1$-reduct of an $\mathbb L_1$-model, and it is required to be invariant under automorphisms of that reduct.
\end{definition}

\begin{definition}[Gluing-sensitive banded lift datum]\label{def:gluing-sensitive-banded-lift}
Fix an admissible parameter $r:\mathsf{Ref}_s$ at $p$ and write $a=[r]_p$. A \emph{gluing-sensitive banded lift datum} for $r$ is a tuple
\[
(\mathfrak P_r,\mathfrak L_r,\Band,\Cov_r^{\mathrm{acyc}})
\]
with the following properties:
\begin{enumerate}[label=(\roman*),nosep]
\item $\mathfrak P_r\to\mathcal C_{p,s}/a$ is a banded groupoid lift with band $\Band$ in the sense of \Cref{def:banded-groupoid-lift};
\item $\mathfrak L_r$ is a $\Band$-banded stack whose connected components agree with the sheafification of $F_{p,s}$, i.e.
\[
\pi_0(\mathfrak L_r)\cong a_{p,s}F_{p,s}|_{\mathcal C_{p,s}/a};
\]
\item the stackification morphism $\iota_r:\mathfrak P_r\to\mathfrak L_r$ is essentially surjective on the terminal fibre over $a$ in the sense of \Cref{def:global-conservativity};
\item for each global component section supported by the datum, the induced family
\(\Cov_v^{\mathrm{acyc}}(a)\) is part of a \v{C}ech--derived comparison
hypothesis in the sense of \Cref{def:cech-derived-comparison-package}; in
particular, its covers admit \v{C}ech nerves in the slice, trivialize the
band where the obstruction cocycles are computed, and carry the specified
comparison isomorphism to derived cohomology.
\end{enumerate}
If such a datum exists we say that $r$ admits a gluing-sensitive banded lift. This datum is extra structure on top of the bare presheaf of local sections: later sections treat the gerbe-theoretic layer only relative to a chosen datum of this kind and do not assert that every presheaf of local sections admits one.
\end{definition}

The following result applies the fragment of \Cref{def:lower-comparison-class} to the local-section predicates.

\begin{theorem}[Parameter-free one-reference lower-fragment separation in the constant-free reduct]\label{thm:pointwise-irreducibility}
There exist a model $M$, the singleton information state
\[
p=(\varnothing,\{\emptyset\}),
\]
and pairwise distinct elements
\[
a_1,a_2,a_3,a_4\in D_{\mathsf{Ref}_s}^M
\]
of one parameter sort such that:
The comparison language in this theorem is restricted to the
parameter-free one-variable \(\Sigma_1\)-reduct: it excludes admission,
all local-section predicates, site predicates, cover data, and parameters of
that parameter sort. The undefinability conclusions are therefore conclusions on
the admissible subdomain displayed in \textup{(i)}, not definitions of
admission itself in the lower language.
No separation claim is made for languages with such parameters or with
access to the \(\Sigma_2\) presheaf structure.
\begin{enumerate}[label=(\roman*),nosep]
\item each $a_i$ is admissible at $p$, and for every $\varphi(x)\in\mathcal{C}_{s}^{\mathrm{inv}}$,
\[
M,\emptyset[x\mapsto a_i]\models\varphi(x)
\quad\Longleftrightarrow\quad
M,\emptyset[x\mapsto a_j]\models\varphi(x)
\]
for all $i,j\in\{1,2,3,4\}$;
\item
\[
M,p\Vdash \bigl(\CompSec_s(x)\land \Sec_s(x)\bigr)[a_1/x];
\]
\item
\[
M,p\Vdash \bigl(\CompSec_s(x)\land \neg\Sec_s(x)\bigr)[a_2/x];
\]
\item
\[
M,p\Vdash \bigl(\LocSec_s(x)\land \neg\CompSec_s(x)\bigr)[a_3/x];
\]
\item
\[
M,p\Vdash \bigl(\neg\LocSec_s(x)\bigr)[a_4/x].
\]
\end{enumerate}
Consequently, on the displayed admissible subdomain, no $\varphi(x)\in\mathcal{C}_{s}^{\mathrm{inv}}$ can define any of the predicates
\[
\LocSec_s,
\qquad
\CompSec_s,
\qquad
\Sec_s,
\qquad
\NoLoc_s,
\qquad
\NoMatch_s,
\qquad
\NoAmalg_s
\]
on admissible parameter elements. In particular, $\CompSec_s$, $\Sec_s$, and $\NoAmalg_s$ are not reducible to the automorphism-invariant pointwise fragment of the $\mathbb L_1$-reduct.
\end{theorem}

\begin{proof}
Take one designated sort $s$ and let the lower-layer state be the singleton
\[
p=(\varnothing,\{\emptyset\}).
\]
Choose the constant-free $\Sigma_1$-reduct of $M$ so that the reference carrier of sort $\mathsf{Ref}_s$ is
\[
D_{\mathsf{Ref}_s}^M=\{(i,n)\mid i\in\{1,2,3,4\},\ n\in\mathbb N\},
\]
and write $a_i:=(i,0)$. Interpret the remaining symbols of the constant-free $\Sigma_1$-reduct so that no interpretation depends on the first coordinate on this sort: relations involving $\mathsf{Ref}_s$ are chosen to be unions of whole first-coordinate orbits, and functions with arguments in $\mathsf{Ref}_s$ are chosen to factor through the second coordinate or to be constant on that sort. Then every permutation of $\{1,2,3,4\}$ on the first coordinate, fixing the second coordinate, extends to an automorphism of the reduct. In particular, for any $i,j\in\{1,2,3,4\}$ there exists an automorphism $\sigma$ sending $a_i$ to $a_j$. Since formulas in $\mathcal{C}_{s}^{\mathrm{inv}}$ are parameter free and automorphism invariant, \textup{(i)} follows immediately.

Now attach the $\Sigma_2$-data. Declare each $a_i$ admissible, so
\[
M,p\Vdash \Adm_s(x)[a_i/x]
\qquad
(1\le i\le 4).
\]
Let the reference-refinement site $\mathcal C_{p,s}$ be the disjoint union of four rooted thin components, one over each admissible base object $a_i$. In the first three components, write the root as $a_i$ and choose objects
\[
c_i\sqsubseteq u_i\sqsubseteq a_i,
\qquad
c_i\sqsubseteq v_i\sqsubseteq a_i,
\]
with covering family $\{u_i\to a_i,v_i\to a_i\}$ and pullback $c_i=u_i\times_{a_i}v_i$. In the fourth component, let $a_4$ be the unique object and let the trivial cover $\{a_4\to a_4\}$ be the only covering family.

This expansion is deliberately outside the automorphism-invariant one-variable
fragment of the reduct. The $\Sigma_1$-reduct remembers the parameter carrier,
equality, the lower many-sorted structure, and the singleton information
state; it forgets admissibility, the site, covers, presheaf of local sections,
and the predicates $\LocSec_s,\CompSec_s,\Sec_s$. Hence the automorphisms of
the displayed $\Sigma_1$-reduct are the reduct automorphisms used in
\Cref{def:lower-comparison-class}. Since that fragment contains only
constant-free one-variable $\Sigma_1$-formulas and is closed under these
reduct automorphisms, any later distinction produced by the $\Sigma_2$-data
cannot be read by a formula in $\mathcal C_s^{\mathrm{inv}}$.

Define the presheaf of local sections $F_{p,s}$ componentwise as follows.

For $a_1$, set
\[
F_{p,s}(u_1)=F_{p,s}(v_1)=F_{p,s}(c_1)=F_{p,s}(a_1)=\{*\},
\]
with all restriction maps equal to the identity. Then the unique local family on the chosen cover is compatible and glues to the global section in $F_{p,s}(a_1)$, proving \textup{(ii)}.

For $a_2$, set
\[
F_{p,s}(u_2)=F_{p,s}(v_2)=F_{p,s}(c_2)=\{*\},
\qquad
F_{p,s}(a_2)=\varnothing,
\]
again with identity restrictions on the nonempty fibres. Then the local family $(*,*)$ is compatible on the overlap, but there is no global section at $a_2$, proving \textup{(iii)}.

For $a_3$, set
\[
F_{p,s}(u_3)=\{\sigma_u\},
\quad
F_{p,s}(v_3)=\{\sigma_v\},
\quad
F_{p,s}(c_3)=\{0,1\},
\quad
F_{p,s}(a_3)=\varnothing,
\]
with
\[
\res_{c_3}^{u_3}(\sigma_u)=0,
\qquad
\res_{c_3}^{v_3}(\sigma_v)=1.
\]
The pretopology on this component is the smallest one generated by the
identity covers, the cover
\[
\{u_3\to a_3,\ v_3\to a_3\},
\]
its pullbacks along arrows in the component, and transitivity. Pulling this
cover back to \(u_3\) gives the cover \(\{u_3\to u_3,\ c_3\to u_3\}\);
pulling it back to \(v_3\) gives \(\{c_3\to v_3,\ v_3\to v_3\}\); pulling it
back to \(c_3\) gives only identity copies of \(c_3\).

We claim that every generated cover of \(a_3\) that contains local sections
has, after deleting repeated identity refinements, nonempty refinements over
both \(u_3\) and \(v_3\). This is proved by induction on the generation of
covers. The displayed generator has this property. Pullback stability only
creates covers of \(u_3\), \(v_3\), or \(c_3\), and therefore cannot produce a
new cover of \(a_3\) missing one side. For transitivity, a cover of \(a_3\)
is obtained by refining the \(u_3\)-member and the \(v_3\)-member separately;
if the resulting family has local sections, the refinements over \(u_3\) and
over \(v_3\) must each contain a member on which the unique sections
\(\sigma_u\) and \(\sigma_v\) restrict. These restrictions meet over a
refinement of \(c_3=u_3\times_{a_3}v_3\). On every such refinement the
\(u_3\)-side restriction is \(0\) and the \(v_3\)-side restriction is \(1\).
Thus no generated cover of \(a_3\) with local sections admits a compatible
matching family. Local sections exist, for example on the displayed cover,
but every possible generated cover carrying both sides forces the same
incompatible restrictions on \(c_3\). Hence
\[
M,p\Vdash \LocSec_s(a_3)
\qquad\text{and}\qquad
M,p\nVdash \CompSec_s(a_3),
\]
proving \textup{(iv)}.

For $a_4$, set
\[
F_{p,s}(a_4)=\varnothing.
\]
Since the trivial cover is the only cover of $a_4$, there are no local sections anywhere above $a_4$, proving \textup{(v)}.

Because all of this geometric data lives in $\Sigma_2\setminus\Sigma_1$, it does not alter the automorphism orbit computed in \textup{(i)}. The four admissible parameter elements therefore remain indistinguishable in the automorphism-invariant one-variable fragment while separating the higher predicates: $\LocSec_s$ holds at $a_1,a_2,a_3$ and fails at $a_4$; $\CompSec_s$ holds at $a_1,a_2$ and fails at $a_3,a_4$; and $\Sec_s$ holds only at $a_1$.
To spell out the reduct argument, suppose for instance that $\theta(x)\in\mathcal C_s^{\mathrm{inv}}$ defined $\CompSec_s$ on admissible elements in this expansion. The reduct automorphism sending $a_1$ to $a_3$ would imply
\[
M,\emptyset[x\mapsto a_1]\models\theta(x)
\Longleftrightarrow
M,\emptyset[x\mapsto a_3]\models\theta(x),
\]
while the $\Sigma_2$-interpretation gives $\CompSec_s(a_1)$ and $\neg\CompSec_s(a_3)$, a contradiction. The pairs $(a_1,a_2)$ and $(a_2,a_4)$ give the same contradiction for $\Sec_s$ and $\LocSec_s$, respectively. Thus none of these three predicates is definable by a formula from the comparison class.
The three failure predicates are Boolean combinations of these same existence predicates:
\begin{align*}
\NoLoc_s&=\Adm_s\land \neg\LocSec_s,\\
\NoMatch_s&=\Adm_s\land \LocSec_s\land \neg\CompSec_s,\\
\NoAmalg_s&=\Adm_s\land \CompSec_s\land \neg\Sec_s.
\end{align*}
Since admissibility is fixed on all four elements and the lower formulas cannot distinguish the relevant existence predicates, none of the three failure predicates is definable either.
\end{proof}

\begin{remark}[Why tuples and parameters are excluded]\label{rem:tuple-parameter-limitation}
The restriction to one free variable and no parameters is essential for the argument. If two free reference variables are allowed, equality formulas such as $x_1=x_2$ already distinguish tuples by their equality pattern, so a single orbit no longer witnesses the undefinability of $\CompSec_s$ or $\Sec_s$. Likewise, if parameters of sort $\mathsf{Ref}_s$ are available, a formula of the form $x=b$ isolates a chosen witness unless one first arranges that all witnesses lie outside the parameter support. Thus the comparison class $\mathcal{C}_{s}^{\mathrm{inv}}$ records exactly the fragment in which automorphism-invariant pointwise semantics cannot detect existence of a matching family.
\end{remark}
