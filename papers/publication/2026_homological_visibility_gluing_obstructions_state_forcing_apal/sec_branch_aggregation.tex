\section{Aggregation over component sections}\label{sec:branch-aggregation}

We compare the quotients attached to the different global component sections of a single admissible parameter. Retaining the component label leads to a label-aware quotient, while forgetting it leads to a common-character quotient. The comparison results below are formal once the family of componentwise homological image subgroups have been fixed, whereas the construction theorems later in the section remain internal to explicitly chosen finite nerves and constant bands.

\begin{definition}[Family of componentwise homological image subgroups]\label{def:branchwise-intrinsic-data}
Fix an admissible parameter $r:\mathsf{Ref}_s$ at a state $p$, and assume:
\begin{enumerate}[label=(\roman*),nosep]
\item the hypotheses of \Cref{def:cech-gluing-obstruction} hold;
\item the set $\Vis_{p,s}(r)$ is finite and nonempty;
\item every component gerbe $\mathfrak G_v$ is banded by the same finite abelian group $A$ and admits a finite nerve representative of its component obstruction class $\omega_v$.
\end{enumerate}
For each global component section $v\in \Vis_{p,s}(r)$, write
\[
K_v:=K_{\omega_v},
\qquad
Q_v:=Q_{\omega_v}=A/K_v,
\qquad
\pi_v:A\twoheadrightarrow Q_v.
\]
\end{definition}

\begin{theorem}[Label-aware quotient]\label{thm:branch-sensitive-visible-quotient}
In the setting of \Cref{def:branchwise-intrinsic-data}, consider the diagonal map
\[
\Delta_{\mathrm{lab}}:A\longrightarrow \prod_{v\in \Vis_{p,s}(r)} Q_v,
\qquad
x\longmapsto (\pi_v(x))_v.
\]
Then
\[
\ker(\Delta_{\mathrm{lab}})=\bigcap_{v\in \Vis_{p,s}(r)} K_v.
\]
Consequently, the quotient
\[
Q_{\mathrm{lab}}(r):=A\Big/\bigcap_{v\in \Vis_{p,s}(r)} K_v
\]
is canonically isomorphic to $\mathrm{Im}(\Delta_{\mathrm{lab}})$. It is the maximal quotient of $A$ detected by the family of componentwise quotient maps when the component label is retained.
\end{theorem}

\begin{proof}
For $x\in A$ one has $\Delta_{\mathrm{lab}}(x)=0$ if and only if $\pi_v(x)=0$ for every global component section $v$, which is equivalent to $x\in K_v$ for every $v$. Hence
\[
\ker(\Delta_{\mathrm{lab}})=\bigcap_{v\in \Vis_{p,s}(r)} K_v.
\]
The first isomorphism theorem therefore yields
\[
\mathrm{Im}(\Delta_{\mathrm{lab}})
\cong
A\Big/\bigcap_{v\in \Vis_{p,s}(r)} K_v.
\]
Thus, once the global component section is known, two elements of $A$ have the same image exactly when they have the same image in every componentwise quotient.
\end{proof}

\begin{theorem}[Common-character quotient]\label{thm:branch-uniform-visible-quotient}
In the setting of \Cref{def:branchwise-intrinsic-data}, let
\[
X_v:=\{\chi\in \widehat A\mid \chi_*(\omega_v)=0\}
\]
for each global component section $v$, and define the common annihilating characters by
\[
X^{\mathrm{com}}(r):=\bigcap_{v\in \Vis_{p,s}(r)} X_v.
\]
Then
\[
X^{\mathrm{com}}(r)=\mathrm{Ann}\Bigl(\sum_{v\in \Vis_{p,s}(r)} K_v\Bigr).
\]
Consequently, the quotient
\[
Q_{\mathrm{com}}(r):=A\Big/\sum_{v\in \Vis_{p,s}(r)} K_v
\]
is exactly the quotient detected by the characters that annihilate every componentwise homological image simultaneously.
\end{theorem}

\begin{proof}
By \Cref{thm:intrinsic-visible-quotient}, one has
\[
X_v=\mathrm{Ann}(K_v)
\]
for every $v$. Therefore
\[
X^{\mathrm{com}}(r)
=
\bigcap_{v\in \Vis_{p,s}(r)} \mathrm{Ann}(K_v)
=
\mathrm{Ann}\Bigl(\sum_{v\in \Vis_{p,s}(r)} K_v\Bigr),
\]
where the last identity is the standard annihilator formula for finite abelian groups \cite[Part~I, Chap.~2]{Terras1999}. Finite abelian duality then gives
\[
\bigcap_{\chi\in X^{\mathrm{com}}(r)} \ker\chi
=
\sum_{v\in \Vis_{p,s}(r)} K_v.
\]
Hence the quotient detected by all common annihilating characters is precisely
\[
A\Big/\sum_{v\in \Vis_{p,s}(r)} K_v.
\qedhere
\]
\end{proof}

\begin{definition}[Canonical comparison and component-forgetting kernel]\label{def:branch-comparison-map}
In the setting of \Cref{def:branchwise-intrinsic-data}, let
\[
\pi_{\mathrm{lab}}:A\twoheadrightarrow Q_{\mathrm{lab}}(r)
\qquad\text{and}\qquad
\pi_{\mathrm{com}}:A\twoheadrightarrow Q_{\mathrm{com}}(r)
\]
be the canonical quotient maps. Since
\[
\bigcap_{v\in \Vis_{p,s}(r)} K_v
\subseteq
\sum_{v\in \Vis_{p,s}(r)} K_v,
\]
there is a unique surjective homomorphism
\[
\beta_r:Q_{\mathrm{lab}}(r)\twoheadrightarrow Q_{\mathrm{com}}(r)
\]
such that
\[
\beta_r\circ \pi_{\mathrm{lab}}=\pi_{\mathrm{com}}.
\]
Its kernel is denoted
\[
K_{\mathrm{forg}}(r):=
\ker(\beta_r)
\cong
\Bigl(\sum_{v\in \Vis_{p,s}(r)} K_v\Bigr)\Big/\Bigl(\bigcap_{v\in \Vis_{p,s}(r)} K_v\Bigr)
\]
and called the \emph{component-forgetting kernel}.
\end{definition}

\begin{theorem}[Exact comparison of the aggregate quotients]\label{thm:branch-comparison-exact-sequence}
In the setting of \Cref{def:branch-comparison-map}, there is a short exact sequence
\[
0\longrightarrow K_{\mathrm{forg}}(r)\longrightarrow Q_{\mathrm{lab}}(r)\xrightarrow{\ \beta_r\ } Q_{\mathrm{com}}(r)\longrightarrow 0.
\]
Equivalently,
\[
\ker(\beta_r)
\cong
\Bigl(\sum_{v\in \Vis_{p,s}(r)} K_v\Bigr)\Big/\Bigl(\bigcap_{v\in \Vis_{p,s}(r)} K_v\Bigr).
\]
Thus the component-forgetting kernel is the quotient of the sum of the componentwise images by their intersection.
\end{theorem}

\begin{proof}
Existence and uniqueness of $\beta_r$ follow from the universal property of the quotient because
\[
\bigcap_v K_v\subseteq \sum_v K_v.
\]
Surjectivity is immediate from surjectivity of $\pi_{\mathrm{com}}$. By the third isomorphism theorem,
\[
\ker(\beta_r)
\cong
\Bigl(\sum_v K_v\Bigr)\Big/\Bigl(\bigcap_v K_v\Bigr),
\]
which gives the claimed exact sequence.
\end{proof}

\begin{corollary}[Criterion for equality of the aggregate quotients]\label{cor:no-branch-gain-criterion}
In the setting of \Cref{def:branch-comparison-map}, the following are equivalent:
\begin{enumerate}[label=(\roman*),nosep]
\item $\beta_r$ is an isomorphism;
\item $K_{\mathrm{forg}}(r)=0$;
\item
\[
\bigcap_{v\in \Vis_{p,s}(r)} K_v
=
\sum_{v\in \Vis_{p,s}(r)} K_v;
\]
\item the homological image subgroups are independent of the component section:
\[
K_v=K_w
\qquad
(\forall v,w\in \Vis_{p,s}(r)).
\]
\end{enumerate}
Hence retaining the component label changes no quotient information exactly when all global component sections have the same presentation-relative homological image subgroup.
\end{corollary}

\begin{proof}
The equivalence of \textup{(i)} and \textup{(ii)} is exactness of the sequence in \Cref{thm:branch-comparison-exact-sequence}. Condition \textup{(ii)} is equivalent to \textup{(iii)} by the quotient description of $K_{\mathrm{forg}}(r)$.

If \textup{(iii)} holds, write the common subgroup as $H$. Then
\[
H=\bigcap_v K_v\subseteq K_v\subseteq \sum_v K_v=H
\]
for every global component section $v$, so each $K_v=H$, proving \textup{(iv)}. Conversely, if all $K_v$ are equal to one subgroup $H$, then clearly
\[
\bigcap_v K_v=H=\sum_v K_v.
\qedhere
\]
\end{proof}

\begin{corollary}[Dual character description of the component-forgetting kernel]\label{cor:branch-resolution-dual}
In the setting of \Cref{def:branch-comparison-map},
\[
\widehat{Q_{\mathrm{lab}}(r)}
\cong
\sum_{v\in \Vis_{p,s}(r)} X_v,
\qquad
\widehat{Q_{\mathrm{com}}(r)}
\cong
\bigcap_{v\in \Vis_{p,s}(r)} X_v.
\]
Moreover,
\[
\widehat{K_{\mathrm{forg}}(r)}
\cong
\Bigl(\sum_{v\in \Vis_{p,s}(r)} X_v\Bigr)\Big/\Bigl(\bigcap_{v\in \Vis_{p,s}(r)} X_v\Bigr).
\]
Thus the dual of the component-forgetting kernel consists of the characters generated by the componentwise class-trivializing characters, modulo those that are common to every component.
\end{corollary}

\begin{proof}
By \Cref{thm:intrinsic-visible-quotient}, one has
\[
X_v=\mathrm{Ann}(K_v)
\]
for every global component section $v$. Finite abelian duality then gives
\[
\widehat{Q_{\mathrm{lab}}(r)}
\cong
\mathrm{Ann}\Bigl(\bigcap_v K_v\Bigr)
=
\sum_v \mathrm{Ann}(K_v)
=
\sum_v X_v
\]
and, by \Cref{thm:branch-uniform-visible-quotient},
\[
\widehat{Q_{\mathrm{com}}(r)}
\cong
\bigcap_v X_v.
\]
Dualizing the short exact sequence of \Cref{thm:branch-comparison-exact-sequence} yields an exact sequence
\[
0\longrightarrow \widehat{Q_{\mathrm{com}}(r)}
\longrightarrow \widehat{Q_{\mathrm{lab}}(r)}
\longrightarrow \widehat{K_{\mathrm{forg}}(r)}
\longrightarrow 0,
\]
and substituting the two identifications above gives the claimed quotient description of $\widehat{K_{\mathrm{forg}}(r)}$.
\end{proof}

\begin{definition}[Auxiliary register for the component-forgetting map]\label{def:branch-resolution-code}
In the setting of \Cref{def:branch-comparison-map}, a \emph{auxiliary register for the component-forgetting map} consists of a finite set $R$ and an injective map
\[
\iota_{\mathrm{forg}}:Q_{\mathrm{lab}}(r)\hookrightarrow Q_{\mathrm{com}}(r)\times R
\]
such that
\[
\mathrm{pr}_1\circ \iota_{\mathrm{forg}}=\beta_r.
\]
\end{definition}

\begin{theorem}[Exact logarithmic component-forgetting fibre cardinality]\label{thm:branch-resolution-budget}
Every auxiliary register for the component-forgetting map satisfies
\[
|R|\ge |K_{\mathrm{forg}}(r)|
=
\left|\Bigl(\sum_v K_v\Bigr)\Big/\Bigl(\bigcap_v K_v\Bigr)\right|.
\]
Equivalently,
\[
\log_2|R|
\ge
\log_2|K_{\mathrm{forg}}(r)|.
\]
This bound is sharp.
\end{theorem}

\begin{proof}
Since $\mathrm{pr}_1\circ \iota_{\mathrm{forg}}=\beta_r$, every fiber of $\beta_r$ injects into $R$. By \Cref{thm:branch-comparison-exact-sequence}, each fiber of $\beta_r$ is a coset of $K_{\mathrm{forg}}(r)$ and therefore has cardinality $|K_{\mathrm{forg}}(r)|$. Hence
\[
|R|\ge |K_{\mathrm{forg}}(r)|.
\]

For sharpness, choose a finite set $R$ with
\[
|R|=|K_{\mathrm{forg}}(r)|.
\]
Choose a set-theoretic section
\[
s:Q_{\mathrm{com}}(r)\to Q_{\mathrm{lab}}(r)
\]
of $\beta_r$, and choose an injection
\[
j:K_{\mathrm{forg}}(r)\hookrightarrow R.
\]
Define
\[
\iota_{\mathrm{forg}}(x):=
\bigl(\beta_r(x),j(x-s(\beta_r(x)))\bigr).
\]
Since $x-s(\beta_r(x))\in \ker(\beta_r)=K_{\mathrm{forg}}(r)$, the map is well defined. If
\[
\iota_{\mathrm{forg}}(x)=\iota_{\mathrm{forg}}(y),
\]
then $\beta_r(x)=\beta_r(y)$ and
\[
j(x-s(\beta_r(x)))=j(y-s(\beta_r(y))).
\]
Injectivity of $j$ gives $x=y$, so $\iota_{\mathrm{forg}}$ is injective and the bound is attained.
\end{proof}

\begin{corollary}[Factorization of uniform quotient-fibre information]\label{cor:uniform-hidden-budget-factorization}
In the setting of \Cref{def:branch-comparison-map},
\[
\left|\sum_{v\in \Vis_{p,s}(r)} K_v\right|
=
|K_{\mathrm{forg}}(r)|
\cdot
\left|\bigcap_{v\in \Vis_{p,s}(r)} K_v\right|.
\]
Equivalently,
\[
\log_2\left|\sum_v K_v\right|
=
\log_2|K_{\mathrm{forg}}(r)|
+
\log_2\left|\bigcap_v K_v\right|.
\]
Thus the logarithmic fibre cardinality of the common-character quotient splits into the component-forgetting contribution and the logarithmic cardinality of the intersection of the componentwise images.
\end{corollary}

\begin{proof}
By \Cref{def:branch-comparison-map},
\[
|K_{\mathrm{forg}}(r)|
=
\left|\Bigl(\sum_v K_v\Bigr)\Big/\Bigl(\bigcap_v K_v\Bigr)\right|
=
\frac{|\sum_v K_v|}{|\bigcap_v K_v|}.
\]
Rearranging gives the first displayed identity, and the second follows after taking base-$2$ logarithms.
\end{proof}

\begin{definition}[Common auxiliary register for the componentwise quotient maps]\label{def:branch-preserving-strictification-code}
In the setting of \Cref{def:branchwise-intrinsic-data}, a \emph{common auxiliary register for the componentwise quotient maps} consists of a finite set $R$ and, for each global component section $v\in \Vis_{p,s}(r)$, an injective map
\[
\iota_v:A\hookrightarrow Q_v\times R
\]
such that
\[
\mathrm{pr}_1\circ \iota_v=\pi_v.
\]
\end{definition}

\begin{theorem}[Common-register fibre cardinality]\label{thm:multi-branch-strictification-budget}
Every common auxiliary register for the componentwise quotient maps satisfies
\[
|R|\ge \max_{v\in \Vis_{p,s}(r)} |K_v|.
\]
Equivalently,
\[
\log_2|R|\ge \max_{v\in \Vis_{p,s}(r)} \log_2|K_v|.
\]
This bound is sharp.
\end{theorem}

\begin{proof}
Fix a global component section $v$. Since $\mathrm{pr}_1\circ \iota_v=\pi_v$, every fiber of $\pi_v$ injects into $R$. By \Cref{def:intrinsic-visible-quotient}, every fiber of $\pi_v$ is a coset of $K_v$ and hence has cardinality $|K_v|$. Therefore $|R|\ge |K_v|$ for every $v$, which proves the lower bound.

For sharpness, choose a finite set $R$ with
\[
|R|=\max_{v\in \Vis_{p,s}(r)} |K_v|.
\]
For each $v$, choose an injection
\[
j_v:K_v\hookrightarrow R
\]
and a set-theoretic section
\[
s_v:Q_v\to A
\]
of $\pi_v$. Define
\[
\iota_v(x):=\bigl(\pi_v(x),j_v(x-s_v(\pi_v(x)))\bigr).
\]
Since $x-s_v(\pi_v(x))\in K_v$, the formula is well defined. If $\iota_v(x)=\iota_v(y)$, then $\pi_v(x)=\pi_v(y)$ and
\[
j_v(x-s_v(\pi_v(x)))=j_v(y-s_v(\pi_v(y))).
\]
Injectivity of $j_v$ gives $x=y$. Thus each $\iota_v$ is injective, so the bound is attained.
\end{proof}

\begin{corollary}[Single-component reduction]\label{cor:single-branch-reduction}
If
\[
\Vis_{p,s}(r)=\{v\},
\]
then
\[
Q_{\mathrm{lab}}(r)\cong Q_{\mathrm{com}}(r)\cong Q_v
\]
and the minimum common auxiliary-register cardinality is exactly $|K_v|$.
\end{corollary}

\begin{proof}
If there is only one global component section, then
\[
\bigcap_{u\in \Vis_{p,s}(r)} K_u
=
\sum_{u\in \Vis_{p,s}(r)} K_u
=
K_v.
\]
The quotient statements follow from \Cref{thm:branch-sensitive-visible-quotient,thm:branch-uniform-visible-quotient}, and the register-size statement is \Cref{thm:multi-branch-strictification-budget}.
\end{proof}
