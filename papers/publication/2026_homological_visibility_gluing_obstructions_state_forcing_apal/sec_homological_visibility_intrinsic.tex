\subsection{Homological images, cokernels, and annihilating characters}

\begin{definition}[Homological image and cokernel]\label{def:intrinsic-visible-quotient}
In the setting of \Cref{def:finite-nerve-presentation}, define the \emph{presentation-relative homological image subgroup}
\[
K_{\omega}:=\mathrm{Im}(\mathrm{ev}_{\omega})
\]
and the associated \emph{homological cokernel}
\[
Q_{\omega}:=\operatorname{coker}(\mathrm{ev}_{\omega})=A/K_{\omega}.
\]
These terms are relative to the presentation-comparison groupoid of
\Cref{def:finite-nerve-comparison-relation}; no presentation-free quotient is
asserted outside that groupoid.
\end{definition}

\begin{definition}[Class-trivializing characters]\label{def:class-admissible-character}
In the setting of \Cref{def:finite-nerve-presentation}, let
\[
X_{\omega}:=\{\chi\in\widehat A\mid \chi_*(\omega)=0\},
\qquad
\widehat A:=\mathrm{Hom}(A,\mathbb T).
\]
Elements of $X_{\omega}$ are called \emph{class-trivializing} for $\omega$.
\end{definition}

\begin{theorem}[Character-theoretic detection of the homological cokernel]\label{thm:intrinsic-visible-quotient}
In the setting of \Cref{def:class-admissible-character},
\[
X_{\omega}
=
\{\chi\in\widehat A\mid \chi\circ \mathrm{ev}_{\omega}=0\}
=
\mathrm{Ann}(K_{\omega}).
\]
Consequently,
\[
K_{\omega}
=
\bigcap_{\chi\in X_{\omega}}\ker\chi,
\qquad
Q_{\omega}\cong A\Big/\!\bigcap_{\chi\in X_{\omega}}\ker\chi.
\]
\end{theorem}

\begin{proof}
For every character $\chi:A\to\mathbb T$, naturality of the universal coefficient sequence gives a commutative square
\[
\begin{CD}
H^2(N(\mathcal U),A) @>\eta_A>> \mathrm{Hom}(H_2(N(\mathcal U),\mathbb Z),A) \\
@V\chi_*VV @VV(\chi\circ -)V \\
H^2(N(\mathcal U),\mathbb T) @>\eta_{\mathbb T}>> \mathrm{Hom}(H_2(N(\mathcal U),\mathbb Z),\mathbb T).
\end{CD}
\]
Since $\mathbb T$ is divisible and hence injective in $\mathbf{Ab}$, one has
\[
\operatorname{Ext}^1(H_1(N(\mathcal U),\mathbb Z),\mathbb T)=0
\]
\cite[Sec.~2.3]{Weibel1994}. Thus $\eta_{\mathbb T}$ is an isomorphism by \Cref{def:finite-nerve-presentation}. Therefore
\[
\chi_*(\omega)=0
\Longleftrightarrow
\eta_{\mathbb T}(\chi_*(\omega))=0
\Longleftrightarrow
\chi\circ \eta_A(\omega)=0
\Longleftrightarrow
\chi\circ \mathrm{ev}_{\omega}=0.
\]
This proves the first displayed identity, and the second is simply the definition of the annihilator of a subgroup of $A$.

Let $H:=K_{\omega}=\mathrm{Im}(\mathrm{ev}_{\omega})\le A$. By the first part of the proof, $X_{\omega}=\mathrm{Ann}(H)$. Finite abelian duality implies
\[
\bigcap_{\chi\in \mathrm{Ann}(H)}\ker\chi=H
\]
\cite[Part~I, Chaps.~2, 12]{Terras1999}. Hence
\[
K_{\omega}
=
\bigcap_{\chi\in X_{\omega}}\ker\chi.
\]
The final isomorphism follows immediately:
\[
Q_{\omega}
=
A/K_{\omega}
\cong
A\Big/\!\bigcap_{\chi\in X_{\omega}}\ker\chi.
\qedhere
\]
\end{proof}

\begin{corollary}[Character detection of the homological cokernel]\label{cor:intrinsic-character-detection}
The quotient map
\[
\pi_{\omega}:A\twoheadrightarrow Q_{\omega}
\]
induces a natural isomorphism
\[
\widehat{Q_{\omega}}\cong X_{\omega}.
\]
For elements $x,y\in A$, one has
\[
\pi_{\omega}(x)=\pi_{\omega}(y)
\quad\Longleftrightarrow\quad
\chi(x)=\chi(y)\ \text{for every class-trivializing }\chi.
\]
\end{corollary}

\begin{proof}
By \Cref{thm:intrinsic-visible-quotient},
\[
X_{\omega}=\mathrm{Ann}(K_{\omega})=K_{\omega}^{\perp}.
\]
Characters of $Q_{\omega}=A/K_{\omega}$ are exactly the characters of $A$ that kill $K_{\omega}$, so precomposition with $\pi_{\omega}$ identifies $\widehat{Q_{\omega}}$ with $X_{\omega}$. The separation statement follows from finite abelian duality applied to the finite group $Q_{\omega}$ \cite[Part~I, Chap.~2]{Terras1999}.
\end{proof}

\begin{theorem}[Initiality among zero-evaluation quotients]\label{thm:intrinsic-pure-ext-initiality}
In the finite nerve representative of \Cref{def:finite-nerve-presentation}, let
\[
\pi_\omega:A\twoheadrightarrow Q_\omega=A/K_\omega
\]
be the homological cokernel. For every surjective homomorphism $q:A\twoheadrightarrow B$, the following are equivalent:
\begin{enumerate}[label=(\roman*),nosep]
\item $q$ factors uniquely through $\pi_\omega$;
\item $q$ kills $\operatorname{Im}(\mathrm{ev}_\omega)=K_\omega$;
\item $\eta_B(q_*\omega)=0$;
\item $q_*(\omega)$ lies in the UCT $\operatorname{Ext}^1$ subgroup
\[
\operatorname{Ext}^1(H_1(N(\mathcal U),\mathbb Z),B)
\subseteq H^2(N(\mathcal U),B)
\]
of the universal coefficient sequence.
\end{enumerate}
Thus $Q_\omega$ is initial among coefficient quotients for which the pushed-forward obstruction has zero $H_2$ evaluation. This is the pullback-site, finite-nerve form of the companion initiality theorem \cite[Theorem~6.13]{MaZhang2026ConditionalGluingFailure}.
\end{theorem}

\begin{proof}
The equivalence of \textup{(i)} and \textup{(ii)} is the universal property of the quotient
\[
\pi_\omega:A\twoheadrightarrow A/K_\omega.
\]
Indeed, a homomorphism $q:A\to B$ factors uniquely through $\pi_\omega$ if and only if $K_\omega\subseteq\ker q$, which is exactly the assertion that $q$ kills $\operatorname{Im}(\mathrm{ev}_\omega)$.

For \textup{(ii)} and \textup{(iii)}, use naturality of the universal coefficient map with respect to the coefficient homomorphism $q$. The naturality square gives
\[
\eta_B(q_*\omega)=q\circ \eta_A(\omega)=q\circ \mathrm{ev}_\omega.
\]
Thus $\eta_B(q_*\omega)=0$ if and only if $q$ vanishes on the image of $\mathrm{ev}_\omega$, namely on $K_\omega$.

Finally, the universal coefficient exact sequence for the finite nerve is
\[
0\to \operatorname{Ext}^1(H_1(N(\mathcal U),\mathbb Z),B)
\to H^2(N(\mathcal U),B)
\xrightarrow{\eta_B}
\operatorname{Hom}(H_2(N(\mathcal U),\mathbb Z),B)
\to 0.
\]
Therefore $\eta_B(q_*\omega)=0$ if and only if $q_*(\omega)$ lies in the kernel of $\eta_B$, which is precisely the displayed UCT $\operatorname{Ext}^1$ subgroup. This proves all equivalences and the initiality statement.
\end{proof}

\begin{corollary}[Class-trivializing initiality when $H_1$ vanishes]\label{cor:h1-vanishing-class-trivializing}
Assume
\[
H_1(N(\mathcal U),\mathbb Z)=0.
\]
Then, for every surjective homomorphism $q:A\twoheadrightarrow B$, the following are equivalent:
\begin{enumerate}[label=(\roman*),nosep]
\item $q_*(\omega)=0$ in $H^2(N(\mathcal U),B)$;
\item $q$ factors uniquely through
\[
\pi_{\omega}:A\twoheadrightarrow Q_{\omega}.
\]
\end{enumerate}
Hence, when $H_1(N(\mathcal U),\mathbb Z)=0$, the homological cokernel is initial among coefficient quotients that annihilate the obstruction class itself.
\end{corollary}

\begin{proof}
If $H_1(N(\mathcal U),\mathbb Z)=0$, then
\[
\operatorname{Ext}^1(H_1(N(\mathcal U),\mathbb Z),B)=0
\]
for every abelian group $B$. Thus, in the criterion of \Cref{thm:intrinsic-pure-ext-initiality}, the zero-evaluation condition is equivalent to $q_*(\omega)=0$, and the conclusion follows.
\end{proof}

\begin{remark}[Presentation-relative invariance without a canonical splitting]\label{rem:no-canonical-splitting}
For a fixed finite nerve representative, the quotient
$Q_{\omega}$ and the subgroup
$K_{\omega}=\mathrm{Im}(\mathrm{ev}_{\omega})$ are invariants of the chosen
presentation. Across comparison-equivalent presentations they are transported
functorially only along chosen morphisms in the presentation-comparison
groupoid, as in \Cref{prop:invariance-visible-quotients}; different zigzags
with the same endpoints may carry different coefficient automorphisms. By
contrast, the universal coefficient sequence need not split canonically. Thus
the theory isolates a homological cokernel and homological image subgroup
at each presentation, with comparison-zigzag functoriality, without requiring
any noncanonical decomposition of $\omega$ into $H_2$-evaluated and UCT-Ext
summands.
\end{remark}

\begin{corollary}[Comparison with cocycle-coordinate quotients]\label{cor:strict-to-intrinsic-visible}
Let $\alpha$ be any finite \v{C}ech cocycle representative of the class $\omega$. Then the homological cokernel
\[
Q_{\omega}
\]
admits a canonical surjective homomorphism onto the cocycle-coordinate quotient
\[
A/H_\alpha.
\]
Equivalently, every cocycle-coordinate quotient is a quotient of the homological cokernel.
\end{corollary}

\begin{proof}
Let $q:A\twoheadrightarrow A/H_\alpha$ be the canonical quotient. By \Cref{thm:initial-visible-quotient}, the quotient kills every cocycle value $\alpha_{ijk}$ pointwise, so the pushed-forward cocycle representative is zero. Hence $q_*(\omega)=0$, and in particular it lies in the $\operatorname{Ext}$-subgroup of $H^2(N(\mathcal U),A/H_\alpha)$. Apply \Cref{thm:intrinsic-pure-ext-initiality}.
\end{proof}

\begin{definition}[Chain-level evaluation of a presentation]\label{def:chain-level-evaluation}
Let $\alpha\in Z^2(N(\mathcal U),A)$ be a finite \v{C}ech cocycle representative of the class $\omega$. Write
\[
C_2(N(\mathcal U),\mathbb Z)
\]
for the simplicial group of integral $2$-chains on the finite nerve. The \emph{chain-level evaluation map} of $\alpha$ is the homomorphism
\[
\widetilde\alpha:C_2(N(\mathcal U),\mathbb Z)\to A
\]
defined by linear extension from the oriented $2$-simplices:
\[
\widetilde\alpha\!\left(\sum_{\sigma} n_{\sigma}\sigma\right)
:=
\sum_{\sigma} n_{\sigma}\alpha(\sigma).
\]
\end{definition}

\begin{proposition}[Chain-coordinate versus homology-evaluation quotient]\label{prop:chain-vs-homological-visibility}
In the setting of \Cref{def:chain-level-evaluation}, the following hold.
\begin{enumerate}[label=(\roman*),nosep]
\item $\widetilde\alpha$ vanishes on $2$-boundaries and therefore induces a homomorphism
\[
\overline\alpha:H_2(N(\mathcal U),\mathbb Z)\to A.
\]
\item The induced homomorphism $\overline\alpha$ is exactly the homological evaluation map $\mathrm{ev}_{\omega}$.
\item The strict coordinate-image subgroup and the presentation-relative homological image subgroup satisfy
\[
H_\alpha=\mathrm{Im}(\widetilde\alpha),
\qquad
K_{\omega}=\mathrm{Im}(\overline\alpha)=\widetilde\alpha(Z_2(N(\mathcal U),\mathbb Z)).
\]
\end{enumerate}
Thus the gap between chain-coordinate quotient and homology-evaluation quotient is exactly the gap between the image of the cocycle on all $2$-chains and its image on $2$-cycles.
\end{proposition}

\begin{proof}
If $c\in C_3(N(\mathcal U),\mathbb Z)$, then
\[
\widetilde\alpha(\partial c)=(\delta\alpha)(c)=0
\]
because $\alpha$ is a cocycle. Hence $\widetilde\alpha$ vanishes on $B_2(N(\mathcal U),\mathbb Z)$ and induces a homomorphism
\[
\overline\alpha:H_2(N(\mathcal U),\mathbb Z)=Z_2/B_2\to A.
\]
This proves \textup{(i)}.

For any $2$-cycle $z=\sum_{\sigma} n_{\sigma}\sigma\in Z_2(N(\mathcal U),\mathbb Z)$, set
\[
\alpha^{\sharp}(z):=\sum_{\sigma} n_{\sigma}\alpha(\sigma)=\widetilde\alpha(z).
\]
If $z'$ is another $2$-cycle homologous to $z$, then $z-z'=\partial c$ for some $c\in C_3(N(\mathcal U),\mathbb Z)$. By \textup{(i)},
\[
\alpha^{\sharp}(z)-\alpha^{\sharp}(z')
=
\widetilde\alpha(z-z')
=
\widetilde\alpha(\partial c)
=
0.
\]
Therefore $\alpha^{\sharp}$ depends only on the homology class of $z$ and defines a homomorphism
\[
H_2(N(\mathcal U),\mathbb Z)\to A,\qquad
[z]\longmapsto \alpha^{\sharp}(z).
\]
By the definition of the universal coefficient map $\eta_A$, this homomorphism is exactly the image of the cohomology class $\omega=[\alpha]$ under
\[
\eta_A:H^2(N(\mathcal U),A)\to \operatorname{Hom}(H_2(N(\mathcal U),\mathbb Z),A).
\]
Thus
\[
\overline\alpha([z])=\alpha^{\sharp}(z)=\eta_A(\omega)([z])=\mathrm{ev}_{\omega}([z])
\]
for every $[z]\in H_2(N(\mathcal U),\mathbb Z)$, proving \textup{(ii)}.

For \textup{(iii)}, the image of $\widetilde\alpha$ is exactly the subgroup generated by the simplex values $\alpha_{ijk}$, hence it is $H_\alpha$ by \Cref{def:finite-obstruction-presentation}. The equality
\[
\mathrm{Im}(\overline\alpha)=\widetilde\alpha(Z_2(N(\mathcal U),\mathbb Z))
\]
is immediate from \textup{(i)}, and \Cref{def:intrinsic-visible-quotient} identifies this image with $K_{\omega}$.
\end{proof}

\begin{corollary}[Coincidence of the chain-coordinate and homology-evaluation quotients]\label{cor:strict-intrinsic-coincidence-criterion}
Let $\alpha$ be a finite \v{C}ech cocycle representative of $\omega$. The following are equivalent:
\begin{enumerate}[label=(\roman*),nosep]
\item $A/H_\alpha\cong Q_{\omega}$;
\item $H_\alpha=K_{\omega}$;
\item $\widetilde\alpha(C_2(N(\mathcal U),\mathbb Z))=\widetilde\alpha(Z_2(N(\mathcal U),\mathbb Z))$.
\end{enumerate}
Equivalently, the chain-coordinate quotient agrees with the homology-evaluation quotient exactly when every value of the cocycle on a $2$-chain already occurs on a $2$-cycle.
\end{corollary}

\begin{proof}
The equivalence of \textup{(i)} and \textup{(ii)} is immediate from the quotient descriptions
\[
A/H_\alpha
\qquad\text{and}\qquad
Q_{\omega}=A/K_{\omega}.
\]
The equivalence of \textup{(ii)} and \textup{(iii)} is exactly \Cref{prop:chain-vs-homological-visibility}(iii).
\end{proof}

\begin{theorem}[Nonzero UCT-Ext classes and vanishing $H_2$ evaluation]\label{thm:character-blind-obstructions}
In the setting of \Cref{def:finite-nerve-presentation}, with a chosen admissible
parameter, global component section, banded groupoid lift, trivialized constant finite
abelian band $A$, and the \textup{(H4)} comparison datum needed to regard
$\omega$ as a derived component class, the following are equivalent:
\begin{enumerate}[label=(\roman*),nosep]
\item $K_{\omega}=0$;
\item $Q_{\omega}=A$;
\item $\mathrm{ev}_{\omega}=0$;
\item $\omega$ lies in the subgroup
\[
\operatorname{Ext}^1(H_1(N(\mathcal U),\mathbb Z),A)\subseteq H^2(N(\mathcal U),A).
\]
\end{enumerate}
If these conditions hold and $\omega\neq 0$, then the chosen component gerbe $\mathfrak G_v$ is non-neutral, while every character of $A$ is class-trivializing for that component class. If, in addition, the banded groupoid lift is essentially surjective on the terminal fibre, \(F_{p,s}|_{\mathcal C_a}\) is separated on the slice, and $|\Vis_{p,s}(r)|=1$, then
\[
M,p\Vdash \NoAmalg_s(r).
\]
\end{theorem}

\begin{proof}
By \Cref{def:intrinsic-visible-quotient},
\[
K_{\omega}=\mathrm{Im}(\mathrm{ev}_{\omega}),
\]
so \textup{(i)} and \textup{(iii)} are equivalent. Since $Q_{\omega}=A/K_{\omega}$, conditions \textup{(i)} and \textup{(ii)} are equivalent. By the universal coefficient sequence, \textup{(iii)} holds exactly when $\omega\in\ker(\eta_A)$, and this kernel is the stated $\operatorname{Ext}$-subgroup. Hence all four conditions are equivalent.

If they hold and $\omega\neq 0$, then \Cref{prop:derived-branch-classification} shows that the component gerbe $\mathfrak G_v$ is non-neutral.
On the other hand, $\mathrm{ev}_{\omega}=0$ implies
\[
\chi\circ \mathrm{ev}_{\omega}=0
\]
for every $\chi\in\widehat A$, so \Cref{thm:intrinsic-visible-quotient} shows that every character is class-trivializing.

Assume now that \(F_{p,s}|_{\mathcal C_a}\) is separated on the slice and that $|\Vis_{p,s}(r)|=1$. Since the chosen global component section $v$ exists, \Cref{prop:compsecs-vs-visible-branches} gives
\[
M,p\Vdash \CompSec_s(r).
\]
Suppose towards a contradiction that
\[
M,p\Vdash \Sec_s(r).
\]
By \Cref{def:banded-groupoid-lift}(i), this yields an object of the terminal fibre $(\mathfrak P_r)_a$, and its image under the stackification morphism
\[
\iota_r:\mathfrak P_r\to \mathfrak L_r
\]
defines an object of $\mathfrak L_r(a)$. Because there is only one global component section, every object of $\mathfrak L_r(a)$ lies in the unique component gerbe $\mathfrak G_v(a)$. Hence
\[
\mathfrak G_v(a)\neq\varnothing.
\]
By \Cref{lem:terminal-fibre-neutrality}, this implies that $\mathfrak G_v$ is neutral, contradicting the non-neutrality already proved. Therefore
\[
M,p\nVdash \Sec_s(r).
\]
Combining this with $M,p\Vdash \CompSec_s(r)$ and admissibility of $r$, we obtain
\[
M,p\Vdash \NoAmalg_s(r).
\]
\end{proof}

\begin{corollary}[Vanishing evaluation when $H_2=0$]\label{cor:h2-vanishing-blindness}
Assume
\[
H_2(N(\mathcal U),\mathbb Z)=0.
\]
Then, for every obstruction class $\omega\in H^2(N(\mathcal U),A)$,
\[
\mathrm{ev}_{\omega}=0,
\qquad
K_{\omega}=0,
\qquad
Q_{\omega}=A.
\]
Every nonzero component obstruction class on such a nerve therefore has zero $H_2$ evaluation, and every character is class-trivializing.
\end{corollary}

\begin{proof}
If $H_2(N(\mathcal U),\mathbb Z)=0$, then
\[
\mathrm{Hom}(H_2(N(\mathcal U),\mathbb Z),A)=0,
\]
so $\mathrm{ev}_{\omega}=0$ for every $\omega$. Apply \Cref{thm:character-blind-obstructions}.
\end{proof}

\begin{corollary}[No nonzero classes with vanishing $H_2$ evaluation when $H_1$ is free]\label{cor:h1-free-no-blindness}
Assume that $H_1(N(\mathcal U),\mathbb Z)$ is free abelian. Then
\[
K_{\omega}=0
\quad\Longleftrightarrow\quad
\omega=0.
\]
Equivalently, every nonzero component obstruction class has nonzero $H_2$ evaluation. In particular, if $|\Vis_{p,s}(r)|=1$, every nonzero component obstruction class on such a nerve produces a proper homological cokernel.
\end{corollary}

\begin{proof}
If $H_1(N(\mathcal U),\mathbb Z)$ is free abelian, then
\[
\operatorname{Ext}^1(H_1(N(\mathcal U),\mathbb Z),A)=0.
\]
Hence the universal coefficient map $\eta_A$ is injective. By \Cref{thm:character-blind-obstructions},
\[
K_{\omega}=0
\quad\Longleftrightarrow\quad
\mathrm{ev}_{\omega}=0
\quad\Longleftrightarrow\quad
\eta_A(\omega)=0.
\]
Injectivity of $\eta_A$ therefore gives $K_{\omega}=0$ if and only if $\omega=0$. If $\omega\neq 0$, then $K_{\omega}\neq 0$, so $Q_{\omega}=A/K_{\omega}$ is a proper quotient of the finite group $A$.
\end{proof}

\begin{example}[A nonzero class with vanishing $H_2$ evaluation on an $\mathbb{R}P^2$-type nerve]\label{ex:rp2-character-blind}
This is a finite-nerve mechanism example.  Let $\mathcal U$ be a finite cover
whose \v{C}ech nerve is a finite triangulation of $\mathbb{R}P^2$, and take
\[
A=\mathbb Z/2\mathbb Z.
\]
Then
\[
H_2(\mathbb{R}P^2,\mathbb Z)=0,
\qquad
H_1(\mathbb{R}P^2,\mathbb Z)\cong \mathbb Z/2\mathbb Z
\]
\cite[Sec.~2.2]{Hatcher2002}. The universal coefficient sequence therefore yields
\[
H^2(\mathbb{R}P^2,\mathbb Z/2\mathbb Z)
\cong
\operatorname{Ext}^1(\mathbb Z/2\mathbb Z,\mathbb Z/2\mathbb Z)
\cong
\mathbb Z/2\mathbb Z
\]
\cite[Sec.~3.1]{Hatcher2002}. Let $\omega$ be the nonzero class. Then
\[
\mathrm{ev}_{\omega}=0,
\qquad
K_{\omega}=0,
\qquad
Q_{\omega}=A,
\]
by \Cref{thm:character-blind-obstructions}. Thus the homological cokernel is all of $A$, although the obstruction class is nonzero. Every character is therefore class-trivializing. If this is the unique global component section of the admissible parameter under consideration and \(F_{p,s}|_{\mathcal C_a}\) is separated on the slice, then \Cref{thm:gerbe-null-semantics} yields a matching family without global amalgamation whose component class is trivialized by every character.

The last sentence is conditional on the usual banded-stack-lift data: an
admissible parameter, a banded stack lift with constant band
$\mathbb Z/2\mathbb Z$, essential surjectivity on the terminal fibre, and the
\textup{(H4)} comparison identifying the displayed finite-nerve class with the
derived component class; the matching-without-amalgamation conclusion also requires
separatedness on the slice site associated to the parameter.  The unconditional content of the example is the
homological mechanism
\[
\omega\in \operatorname{Ext}^1(H_1(N(\mathcal U),\mathbb Z),A),
\qquad
\eta_A(\omega)=0,
\]
which makes the homological cokernel equal to $A$ although the
obstruction class is nonzero.  Thus the example records the UCT
$\operatorname{Ext}^1$ kernel of the finite nerve, while the component-gerbe conclusion
enters only after the admissible-parameter and banded-lift hypotheses have been
supplied.
\end{example}

\begin{example}[A single-component surjective-homological-evaluation class on $S^2$]\label{ex:s2-single-branch-maximal-collapse}
Let $\mathcal U=\{U_i\to S^2\}_{i\in I}$ be a finite good cover of $S^2$, let $N:=N(\mathcal U)$ be its nerve, and let $\mathcal C_a$ be the corresponding slice site for one admissible parameter $r:\mathsf{Ref}_s$ with $a=[r]_p$. Take the constant band
\[
A=\mathbb Z/2\mathbb Z
\]
on $\mathcal C_a$. Define the presheaf of local sections
\[
F_{p,s}|_{\mathcal C_a}:\mathcal C_a^{op}\to\mathsf{Set}
\]
by
\[
F_{p,s}(U)=\{v\}
\]
for every nonempty proper finite intersection $U$ of members of $\mathcal U$, by
\[
F_{p,s}(a)=\varnothing,
\]
and by the identity map on every restriction between proper intersections. Then the cover
\[
\mathcal U=\{U_i\to a\}_{i\in I}
\]
supports the compatible family constantly equal to $v$, while there is no global section at $a$. Hence
\[
M,p\Vdash \CompSec_s(r),
\qquad
M,p\nVdash \Sec_s(r).
\]
The restricted presheaf is separated on the slice: over \(a\) there are no two
sections to compare, and over every proper nonempty intersection there is
only the section \(v\).
Because
\[
H^2(N,A)\cong \mathbb Z/2\mathbb Z,
\]
choose a simplicial cocycle
\[
\alpha\in Z^2(N,A)
\]
representing the nonzero class. For each $U_i$ choose a local lift object $x_i$ with automorphism group $A$, and choose isomorphisms
\[
\phi_{ij}:x_i|_{U_{ij}}\xrightarrow{\sim} x_j|_{U_{ij}}
\]
on the pairwise overlaps such that
\[
\phi_{ik}^{-1}\phi_{jk}\phi_{ij}=\alpha_{ijk}\in A
\]
on each triple overlap. This is the single-label instance of the
cocycle-to-prestack construction isolated in
\Cref{lem:twisted-branch-prestack-model}, used under the same generated-subsite
admissibility convention as \Cref{lem:generated-subsite-model-interface};
equivalently, it is the standard local-trivialization construction for gerbes
\cite[Chap.~IV]{Giraud1971}, \cite[Sec.~5]{Breen2006}, \cite{Murray1996}. It
yields a prestack presentation $\mathfrak P_r$ on $\mathcal C_a$ with the following properties:
\begin{enumerate}[label=(\roman*),nosep]
\item over each member $U_i$ of the cover there is the chosen local lift object $x_i$ with automorphism group $A$;
\item the chosen transition maps have triple-overlap defect exactly $\alpha$;
\item the presheaf of isomorphism classes satisfies
\[
 \pi_0^{\mathrm{pre}}(\mathfrak P_r)\cong F_{p,s}|_{\mathcal C_a}.
\]
\end{enumerate}
Thus the example begins with an explicit presheaf of local sections and then chooses local lift objects and transition data whose component presheaf is the given functor; the component gerbe is recovered from those data by stackification rather than being taken as primitive input.

Let
\[
\mathfrak L_r:=a(\mathfrak P_r).
\]
Because $\pi_0^{\mathrm{pre}}(\mathfrak P_r)\cong F_{p,s}|_{\mathcal C_a}$, the sheaf of connected components of $\mathfrak L_r$ has exactly one global section at $a$, represented by the constant compatible family above. Therefore
\[
\Vis_{p,s}(r)=\{v\}.
\]
But $\alpha$ is nonzero, so the descent datum is ineffective and no object exists over $a$ in the resulting stack; equivalently, the unique component gerbe is non-neutral. Hence
\[
\mathfrak G_v=\mathfrak L_r,
\qquad
\omega_v=[\alpha]\neq 0.
\]
Combining this with the already computed presheaf-level facts gives
\[
M,p\Vdash \NoAmalg_s(r),
\qquad
M,p\nVdash \Sec_s(r),
\qquad
\Vis_{p,s}(r)=\{v\},
\qquad
\mathfrak G_v=\mathfrak L_r,
\qquad
\omega_v=[\alpha]\neq 0.
\]
The good cover $\mathcal U$ trivializes the constant band, all finite intersections are contractible, and therefore the $H^1$-vanishing and \v{C}ech-to-derived comparison hypotheses of \Cref{def:cech-gluing-obstruction,def:finite-nerve-presentation} are satisfied.

Since
\[
H_1(N,\mathbb Z) = 0,
\qquad
H_2(N,\mathbb Z) \cong \mathbb Z
\]
\cite[Secs.~2.1, 2.2]{Hatcher2002}, the universal coefficient theorem gives
\[
H^2(N, \mathbb{Z}/2\mathbb{Z})
\cong
\mathrm{Hom}(\mathbb{Z}, \mathbb{Z}/2\mathbb{Z})
\cong
\mathbb{Z}/2\mathbb{Z}.
\]
Let $\omega$ be the nonzero class, and let $[S^2]$ denote the fundamental class in $H_2(N,\mathbb Z)$. Then
\[
\mathrm{ev}_\omega([S^2]) = 1 \in \mathbb{Z}/2\mathbb{Z},
\]
so the homological evaluation map is surjective. By \Cref{cor:maximal-collapse},
\[
K_\omega = A = \mathbb{Z}/2\mathbb{Z},
\qquad
Q_\omega = 0.
\]
Hence
\[
M,p\Vdash \NoAmalg_s(r),
\qquad
\omega=\omega_v\neq 0,
\qquad
\mathrm{ev}_{\omega}:H_2(N,\mathbb Z)\twoheadrightarrow A,
\qquad
K_\omega=A,
\qquad
Q_\omega=0.
\]
Since $K_\omega = A$, \Cref{thm:intrinsic-visible-quotient} gives
\[
X_\omega = \mathrm{Ann}(K_\omega) = \mathrm{Ann}(A) = \{1\},
\]
so only the trivial character is class-trivializing. Equivalently, $Q_\omega = A/K_\omega = 0$.

This example therefore verifies the full downstream hypothesis chain
\[
(\mathcal C_a,\mathfrak P_r,A,\omega,\mathrm{ev}_\omega,K_\omega,Q_\omega)
\]
inside a concrete site-theoretic example, once the banded stack lift has been fixed by the chosen cocycle data. It is the opposite extreme from the zero-evaluation case of \Cref{ex:rp2-character-blind}: for an $\mathbb{R}P^2$-type nerve one has $K_\omega=0$, every character is class-trivializing, and the obstruction is entirely in the UCT $\operatorname{Ext}^1$ kernel with $Q_\omega=A$; for the present $S^2$-type nerve one has $K_\omega=A$, only the trivial character is class-trivializing, and the obstruction is entirely detected on $H_2$, with $Q_\omega=0$.
\end{example}

\begin{definition}[Auxiliary register for the homological quotient]\label{def:intrinsic-strictification-code}
An \emph{auxiliary register for the homological quotient} for the obstruction class $\omega$ consists of a finite set $R$ and an injective map
\[
\iota_{\omega}:A\hookrightarrow Q_{\omega}\times R
\]
such that
\[
\mathrm{pr}_1\circ\iota_{\omega}=\pi_{\omega}.
\]
\end{definition}

\begin{corollary}[Fibre size of the homological quotient]\label{cor:intrinsic-hidden-state-lower-bound}
Every auxiliary register for the homological quotient satisfies
\[
|R|\ge |K_{\omega}|.
\]
Equivalently,
\[
\log_2|R|\ge \log_2|K_{\omega}|.
\]
This lower bound is sharp.
\end{corollary}

\begin{proof}
By \Cref{def:intrinsic-visible-quotient}, the fibres of $\pi_{\omega}$ are exactly the cosets of $K_{\omega}$, so every fibre has cardinality $|K_{\omega}|$. The same fibre-counting argument as in \Cref{thm:minimal-information-cost-of-strictification} therefore shows that any auxiliary register preserving the first projection must satisfy $|R|\ge |K_{\omega}|$.

For sharpness, take $R:=K_{\omega}$ and choose a set-theoretic section $s_{\omega}:Q_{\omega}\to A$ of $\pi_{\omega}$. Then
\[
\iota_{\omega}(x):=\bigl(\pi_{\omega}(x),x-s_{\omega}(\pi_{\omega}(x))\bigr)
\]
is injective.
\end{proof}

\begin{theorem}[Prime decomposition of the homological quotient and fibre size]\label{thm:intrinsic-prime-decomposition}
Write the finite abelian group $A$ as a direct sum of its $p$-primary components:
\[
A=\bigoplus_p A_{(p)}.
\]
Let $\omega_p\in H^2(N(\mathcal U),A_{(p)})$ be the image of $\omega$ under the projection onto the $p$-primary summand. Then
\[
\mathrm{ev}_{\omega}=\bigoplus_p \mathrm{ev}_{\omega_p},
\qquad
K_{\omega}=\bigoplus_p K_{\omega,p},
\qquad
Q_{\omega}\cong \bigoplus_p A_{(p)}/K_{\omega,p},
\]
where
\[
K_{\omega,p}:=\mathrm{Im}(\mathrm{ev}_{\omega_p}).
\]
Consequently,
\[
|K_{\omega}|=\prod_p |K_{\omega,p}|,
\qquad
\log_2|K_{\omega}|=\sum_p \log_2|K_{\omega,p}|.
\]
\end{theorem}

\begin{proof}
Both $H^2(N(\mathcal U),-)$ and $\mathrm{Hom}(H_2(N(\mathcal U),\mathbb Z),-)$ commute with finite direct sums, so the universal coefficient map is compatible with the primary decomposition of $A$. Hence
\[
\eta_A(\omega)=\bigoplus_p \eta_{A_{(p)}}(\omega_p),
\]
which is exactly the asserted decomposition of $\mathrm{ev}_{\omega}$.

Taking images gives
\[
K_{\omega}
=
\mathrm{Im}(\mathrm{ev}_{\omega})
=
\bigoplus_p \mathrm{Im}(\mathrm{ev}_{\omega_p})
=
\bigoplus_p K_{\omega,p},
\]
by \Cref{def:intrinsic-visible-quotient}. The quotient decomposition follows immediately:
\[
Q_{\omega}
=
A/K_{\omega}
\cong
\bigoplus_p A_{(p)}/K_{\omega,p}.
\]
Since finite direct sums correspond to products of cardinalities, the logarithmic fibre-cardinality formula is immediate.
\end{proof}

\begin{corollary}[Homological versus chain-coordinate fibre size]\label{cor:strict-intrinsic-budget-comparison}
For every finite \v{C}ech cocycle representative $\alpha$ of the class $\omega$,
\[
|K_{\omega}|\le |H_\alpha|,
\qquad
\log_2|K_{\omega}|\le \log_2|H_\alpha|.
\]
Hence the minimum auxiliary register size for the homological quotient never exceeds that for the cocycle-coordinate quotient. Equality holds if and only if the canonical surjection
\[
Q_{\omega}\twoheadrightarrow A/H_\alpha
\]
is an isomorphism.
\end{corollary}

\begin{proof}
By \Cref{cor:strict-to-intrinsic-visible}, there is a surjective homomorphism
\[
Q_{\omega}=A/K_{\omega}\twoheadrightarrow A/H_\alpha.
\]
For finite groups this implies
\[
|A|/|K_{\omega}|\ge |A|/|H_\alpha|,
\]
hence $|K_{\omega}|\le |H_\alpha|$. The logarithmic inequality is immediate. Equality of cardinalities holds exactly when the displayed surjection is bijective.
\end{proof}

\begin{corollary}[Surjective homological evaluation]\label{cor:maximal-collapse}
In the setting of \Cref{def:finite-nerve-presentation}, the following are equivalent:
\begin{enumerate}[label=(\roman*),nosep]
\item $\mathrm{ev}_{\omega}:H_2(N(\mathcal U),\mathbb Z)\to A$ is surjective;
\item $K_{\omega}=A$;
\item $Q_{\omega}=0$;
\item the logarithmic fibre cardinality of the homological quotient is $\log_2|A|$:
\[
\log_2|K_{\omega}|=\log_2|A|.
\]
\end{enumerate}
\end{corollary}

\begin{proof}
By \Cref{def:intrinsic-visible-quotient}, $K_{\omega}=\mathrm{Im}(\mathrm{ev}_{\omega})$, so \textup{(i)} and \textup{(ii)} are equivalent. Condition \textup{(ii)} is equivalent to \textup{(iii)} by the definition of $Q_{\omega}=A/K_{\omega}$. For finite groups, \textup{(ii)} is also equivalent to $|K_{\omega}|=|A|$, which is exactly \textup{(iv)} after taking $\log_2$.
\end{proof}
