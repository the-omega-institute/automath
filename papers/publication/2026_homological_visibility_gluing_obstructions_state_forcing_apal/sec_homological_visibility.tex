\section{Homological visibility and intrinsic quotients}
\label{sec:homological-visibility}

We now fix one visible branch together with a finite nerve presentation. The quotient attached to a chosen cocycle still depends on that presentation, so in order to isolate the part determined by the cohomology class itself we pass to the homological evaluation map. Throughout the section we work with one fixed finite cover and the comparison maps coming from the identification of \v{C}ech cochains with simplicial cochains on its finite nerve. The resulting hidden subgroup and visible quotient are intrinsic only relative to the homology-equivalent changes of presentation singled out below.

We also assume that the chosen cover $\mathcal U$ trivializes the band, so that the coefficients may be viewed as a constant finite abelian group $A$ on the nerve. The case of nontrivial monodromy, which would require local coefficients, is not considered here.

\subsection{Finite nerve presentations}

\begin{definition}[Finite nerve presentation of the obstruction class]\label{def:finite-nerve-presentation}
Fix an admitted reference $r:\mathsf{Ref}_s$ at a state $p$, and assume conditions \textup{(i)}--\textup{(iv)} of \Cref{def:finite-obstruction-presentation}, with the ordered \v{C}ech convention of \Cref{def:ordered-cech-convention}. A \emph{finite nerve presentation} of the branch obstruction class consists of:
\begin{enumerate}[label=(\roman*),nosep]
\item a finite covering family
\[
\mathcal U=\{a_i\to a\}_{i\in I},
\qquad
a=[r]_p,
\]
whose \v{C}ech nerve $N(\mathcal U)$ is finite;
\item a trivialization of the band on $\mathcal U$, so that the band may be viewed as a constant coefficient finite abelian group $A$ on $N(\mathcal U)$;
\item the induced obstruction class
\[
\omega\in H^2(N(\mathcal U),A),
\]
obtained by identifying \v{C}ech cochains on $\mathcal U$ with simplicial cochains on the finite nerve, transporting the branch obstruction cocycle through that identification, and writing
\[
\omega:=\omega_v
\]
for the chosen visible branch. When the class is also viewed as a derived $H^2$-class, this identification is part of the chosen presentation.
\end{enumerate}
Write
\[
\eta_A:H^2(N(\mathcal U),A)\longrightarrow \mathrm{Hom}(H_2(N(\mathcal U),\mathbb Z),A)
\]
for the map in the universal coefficient exact sequence
\[
0\to \operatorname{Ext}^1(H_1(N(\mathcal U),\mathbb Z),A)\to H^2(N(\mathcal U),A)\xrightarrow{\eta_A}\mathrm{Hom}(H_2(N(\mathcal U),\mathbb Z),A)\to 0
\]
\cite[Sec.~3.1]{Hatcher2002}, and define the \emph{homological evaluation map}
\[
\mathrm{ev}_{\omega}:=\eta_A(\omega).
\]
\end{definition}

\begin{definition}[Finite nerve comparison groupoid]\label{def:finite-nerve-comparison-relation}
For finite nerve presentations of a fixed branch obstruction class, an
\emph{elementary comparison} is one of the three operations listed in
\Cref{prop:invariance-visible-quotients}: homology-equivalent refinement,
cohomologous cocycle replacement, or re-trivialization of the constant band.
The \emph{finite nerve comparison groupoid} has these presentations as
objects and finite zigzags of elementary comparisons and formal inverses as
morphisms, with composition given by concatenation of zigzags. We say that two
presentations are comparison-equivalent when they lie in the same connected
component of this groupoid. Thus the comparison data carried by a finite
zigzag are part of the pseudofunctorial assignment; no path independence
between distinct zigzags is asserted unless an additional coherence
hypothesis is imposed.
\end{definition}

\begin{proposition}[Pseudofunctorial assignment of $K_\omega$ and $A_{\mathrm{vis}}^\omega$]\label{prop:invariance-visible-quotients}
Let $(\mathcal U,A,\omega)$ and $(\mathcal U',A',\omega')$ be finite nerve presentations of the same branch obstruction class. Suppose first that the two presentations differ by one of the following elementary comparisons:
\begin{enumerate}[label=(\roman*),nosep]
\item refinement of the cover by a simplicial map
\[
f:N(\mathcal U')\to N(\mathcal U)
\]
from the refined nerve to the original nerve, such that
\[
\omega'=f^*\omega
\]
and
\[
f_*:H_2(N(\mathcal U'),\mathbb Z)\xrightarrow{\sim} H_2(N(\mathcal U),\mathbb Z)
\]
is an isomorphism, with the coefficient group fixed;
\item replacement of the \v{C}ech cocycle by a cohomologous cocycle in $H^2(N(\mathcal U),A)$;
\item re-trivialization of the band by an isomorphism of coefficient groups
\[
\theta:A\xrightarrow{\sim} A',
\]
so that
\[
\omega'=\theta_*\omega.
\]
\end{enumerate}
Then there exists a specified coefficient comparison isomorphism $\phi:A\to A'$---the identity in \textup{(i)} and \textup{(ii)}, and $\theta$ in \textup{(iii)}---such that
\[
\phi\bigl(K_{\omega}\bigr)=K_{\omega'}
\qquad\text{and}\qquad
\bar\phi:A_{\mathrm{vis}}^{\omega}=A/K_\omega\xrightarrow{\sim}A'/K_{\omega'}=A_{\mathrm{vis}}^{\omega'}
\]
is the induced quotient isomorphism. In particular, both $K_{\omega}$ and $A_{\mathrm{vis}}^{\omega}$ are transported by the three elementary comparison operations listed above, with the refinement direction fixed by the displayed pullback map.

More generally, if the two presentations are joined by a morphism $\Gamma$ in
the finite nerve comparison groupoid of
\Cref{def:finite-nerve-comparison-relation}, then the chosen finite comparison
zigzag $\Gamma$ determines, by composition of the elementary coefficient
comparisons and their inverses, an isomorphism
\[
\phi_\Gamma:A\xrightarrow{\sim} A'
\]
such that
\[
\phi_\Gamma(K_\omega)=K_{\omega'}
\qquad\text{and}\qquad
A/K_\omega\xrightarrow{\sim}A'/K_{\omega'}
\]
is the corresponding composite quotient isomorphism. Hence
$(\mathcal U,A,\omega)\mapsto (K_\omega\le A)$ and
$(\mathcal U,A,\omega)\mapsto A_{\mathrm{vis}}^\omega$ define a pseudofunctorial assignment over the
finite nerve comparison groupoid. They are canonical on objects and functorial
along chosen comparison zigzags; the proposition does not identify the
isomorphisms attached to two different zigzags with the same endpoints.
\end{proposition}

\begin{remark}[Typical refinements]\label{rem:refinement-iso}
The refinements used later are simplicial homotopy equivalences on the chosen nerves---for example, barycentric subdivisions and the standard refinements attached to good covers---and therefore satisfy the hypothesis of case \textup{(i)} of \Cref{prop:invariance-visible-quotients}.
\end{remark}

\begin{remark}[Necessity of homology equivalences and chosen paths]\label{rem:homology-equality}
The equalities in \Cref{prop:invariance-visible-quotients} rely on the assumption that the refinement map induces an isomorphism on $H_2$. Without that hypothesis one only obtains inclusions $\mathrm{Im}(\mathrm{ev}_{\omega'})\subseteq \mathrm{Im}(\mathrm{ev}_{\omega})$ and hence a comparison map $A_{\mathrm{vis}}^{\omega}\twoheadrightarrow A_{\mathrm{vis}}^{\omega'}$, but not equality. The term ``intrinsic visible quotient'' is therefore reserved for refinements and presentations that are homology-equivalent in this precise sense.

For presentations connected by several finite zigzags, the comparison
isomorphism is part of the finite nerve comparison groupoid. A different
zigzag may compose the re-trivialization automorphisms of the coefficient band
in a different way, and the paper does not quotient by that possible holonomy.
Thus the assignment is pseudofunctorial over the finite nerve comparison groupoid rather than a path-independent quotient
on the set of comparison-equivalence classes.
\end{remark}

\begin{proof}
Case \textup{(ii)} is immediate because cohomologous cocycles define the same element of $H^2(N(\mathcal U),A)$ and hence have the same image under $\eta_A$. Here $\phi=\mathrm{id}_A$, so the quotient map is the identity on $A/K_\omega$.

In case \textup{(i)}, the assumption on $f$ gives an isomorphism on $H_2$. Since $\omega'=f^*(\omega)$ and the universal coefficient map is natural, one has
\[
\mathrm{ev}_{\omega'}=\eta_A\bigl(f^*(\omega)\bigr)=\eta_A(\omega)\circ f_*,
\]
where $f_*:H_2(N(\mathcal U'),\mathbb Z)\to H_2(N(\mathcal U),\mathbb Z)$ is induced on homology and is an isomorphism by hypothesis. Hence $\mathrm{Im}(\mathrm{ev}_{\omega'})=\mathrm{Im}(\mathrm{ev}_\omega)$, so $K_{\omega'}=K_{\omega}$ and $A_{\mathrm{vis}}^{\omega'}\cong A_{\mathrm{vis}}^{\omega}$.
The comparison is canonical because the refinement operation supplies the direction of $f$ and hence the specified isomorphism $f_*$; replacing $f_*$ by its inverse would compare the evaluation maps in the wrong direction. Thus the subgroup equality occurs inside the same coefficient group $A$, and the quotient isomorphism is induced by $\phi=\mathrm{id}_A$.

In case \textup{(iii)}, the coefficient isomorphism $\theta$ identifies the two cohomology groups, and $\omega'=\theta_*(\omega)$. Naturality gives the coefficient-isomorphism square
\[
\begin{array}{ccc}
H^2(N(\mathcal U),A) & \xrightarrow{\eta_A} &
\mathrm{Hom}(H_2(N(\mathcal U),\mathbb Z),A)\\
\theta_*\downarrow\phantom{\theta_*} && \phantom{\theta\circ(-)}\downarrow\theta\circ(-)\\
H^2(N(\mathcal U),A') & \xrightarrow{\eta_{A'}} &
\mathrm{Hom}(H_2(N(\mathcal U),\mathbb Z),A'),
\end{array}
\]
and hence
\[
\mathrm{ev}_{\omega'}=\eta_{A'}(\omega')=\eta_{A'}\bigl(\theta_*(\omega)\bigr)=\theta\circ \eta_A(\omega)=\theta\circ \mathrm{ev}_\omega.
\]
Therefore $K_{\omega'}=\theta(K_{\omega})$, and $\theta$ induces an isomorphism between $A_{\mathrm{vis}}^\omega$ and $A_{\mathrm{vis}}^{\omega'}$.

In all three cases the induced quotient map is well defined because the chosen coefficient comparison $\phi$ sends $K_\omega$ onto $K_{\omega'}$. Its inverse is induced by the inverse coefficient comparison: the identity in cases \textup{(i)} and \textup{(ii)}, and $\theta^{-1}$ in case \textup{(iii)}. This proves functoriality of the specified elementary comparison and invariance of the quotient for that elementary morphism.

For a finite zigzag in the comparison groupoid, apply the elementary
case at each step. If a step is traversed in the opposite direction, use the
inverse coefficient and quotient isomorphisms just constructed. Composition of
the finitely many subgroup identifications gives
\[
\phi_\Gamma(K_\omega)=K_{\omega'},
\]
and the universal property of quotients gives the displayed quotient
isomorphism. Concatenating two zigzags concatenates the corresponding composite
coefficient maps, so these assignments respect identity zigzags and composition
of zigzags. Thus finite composites of the three comparison operations preserve
the intrinsic hidden subgroup and visible quotient in the stated
groupoid-indexed sense.
\end{proof}

\begin{proposition}[Finite nerve comparison isomorphism]\label{prop:finite-nerve-comparison}
Work in one finite nerve presentation
$(\mathcal U,A,\omega)$ of \Cref{def:finite-nerve-presentation}; in
particular, the band has been trivialized on $\mathcal U$ and is being
viewed as the constant finite abelian group $A$ on the selected finite
nerve. Use the total order on the finite index set $I$ already fixed in
\Cref{def:ordered-cech-convention} for
\[
\mathcal U=\{a_i\to a\}_{i\in I}.
\]
Use the ordered \v{C}ech convention in which $\check C^n(\mathcal U,A)$
is indexed by tuples
\[
i_0<\cdots<i_n
\]
with nonempty overlap, and use the corresponding ordered oriented
simplices of $N(\mathcal U)$ for simplicial cochains. For each $n\ge 0$,
each nonempty ordered $(n{+}1)$-fold overlap $a_{i_0\cdots i_n}$
determines the oriented $n$-simplex $(i_0,\dots,i_n)$ of the selected
nerve, and the assignment
\[
\Theta_{\mathcal U}^n:\check C^n(\mathcal U,A)\to C^n(N(\mathcal U),A),\qquad
(\alpha_{i_0\cdots i_n})\longmapsto\bigl((i_0,\dots,i_n)\mapsto \alpha_{i_0\cdots i_n}\bigr)
\]
is an isomorphism of cochain groups. Under this correspondence,
\[
\delta\leftrightarrow d,\qquad
B^n\leftrightarrow B^n,\qquad
Z^n\leftrightarrow Z^n.
\]
Hence the \v{C}ech cohomology of this selected ordered finite nerve
presentation with constant coefficients $A$ is identified with the
simplicial cohomology of $N(\mathcal U)$ with coefficients in $A$.
\end{proposition}
\begin{proof}
The hypotheses are part of the finite nerve presentation: the cover is
finite, the coefficient sheaf has been trivialized to the constant group
$A$ on this cover, and the comparison is being made on the chosen finite
nerve rather than on an arbitrary refinement system. With the fixed order
on $I$, the $n$-cochains on $\mathcal U$ are families of coefficients
indexed by nonempty ordered $(n{+}1)$-fold overlaps
$a_{i_0\cdots i_n}$ with $i_0<\cdots<i_n$, while the simplicial
$n$-cochains on $N(\mathcal U)$ are coefficient functions on the
corresponding ordered oriented $n$-simplices $(i_0,\dots,i_n)$. By
definition of the selected nerve, these two index sets agree, so
$\Theta_{\mathcal U}^n$ is a bijection.

The \v{C}ech coboundary of $(\alpha_{i_0\cdots i_n})$ is the alternating
sum obtained by omitting the $k$th ordered index, and the simplicial
coboundary of $\Theta_{\mathcal U}^n(\alpha)$ uses the same ordered face
map with the same sign $(-1)^k$. Since the coefficients are constant
under the chosen trivialization, the restriction maps on coefficients are
the identity maps of $A$. Because the face maps of $N(\mathcal U)$ are
induced by omitting one ordered index, the two alternating sums coincide
term by term. Hence
\[
\Theta_{\mathcal U}^{n+1}(\delta\alpha)=d(\Theta_{\mathcal U}^n(\alpha)).
\]
Therefore $\Theta_{\mathcal U}^\bullet$ is an isomorphism of cochain
complexes, and it induces the stated identifications on cocycles,
coboundaries, and cohomology. This is only the finite ordered
constant-coefficient comparison used in \Cref{def:finite-nerve-presentation};
promotion of a filtered \v{C}ech system to derived cohomology remains the
separate \textup{(H4)} assumption of \Cref{rem:cech-vs-derived}.
\end{proof}

\begin{proposition}[Strict versus intrinsic hidden subgroups]\label{prop:strict-vs-intrinsic-hidden-subgroups}
In the setting of \Cref{def:finite-nerve-presentation}, let $\alpha\in Z^2(N(\mathcal U),A)$ be a simplicial cocycle representing $\omega$, and let
\[
\widetilde\alpha:C_2(N(\mathcal U),\mathbb Z)\to A
\]
be the chain-level evaluation map obtained by linear extension from the oriented $2$-simplices. Then
\[
H_\alpha=\operatorname{Im}(\widetilde\alpha)
\]
and
\[
K_{\omega}:=\mathrm{Im}(\mathrm{ev}_{\omega}).
\]
Thus $H_\alpha$ records presentation-level chain visibility: it is generated by the values of the chosen cocycle on all $2$-chains. By contrast, $K_{\omega}$ records cycle-level class visibility: it is the image that survives after passage to $2$-homology. In particular,
\[
K_{\omega}\subseteq H_\alpha.
\]
\end{proposition}
\begin{proof}
The image of $\widetilde\alpha$ is the subgroup generated by the values of $\alpha$ on the oriented $2$-simplices, equivalently by the values of the original \v{C}ech cocycle on the nonempty triple overlaps of $\mathcal U$. This is exactly the strict subgroup attached to the chosen presentation. The second equality is the definition of $K_\omega$. Since $H_2(N(\mathcal U),\mathbb Z)$ is represented by $2$-cycles inside $C_2(N(\mathcal U),\mathbb Z)$, every homological value counted by $\mathrm{ev}_\omega$ is, a fortiori, a chain-level value of $\widetilde\alpha$, so $K_\omega\subseteq H_\alpha$.
\end{proof}
