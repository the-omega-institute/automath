
\subsection{Quotients and choice of compatible representatives}

We now fix one global component section and a finite \v{C}ech cocycle representative. The cocycle-coordinate quotient $A/H_\alpha$ kills the subgroup generated by the coordinates of this chosen cocycle; the homological cokernel considered in the next section removes dependence on the cocycle representative within the specified presentation-comparison groupoid.

\begin{definition}[Ordered \v{C}ech convention]\label{def:ordered-cech-convention}
For every finite cover
\[
\mathcal U=\{a_i\to a\}_{i\in I}
\]
used in a finite \v{C}ech cocycle representative, fix a total order on $I$. The
group $\check C^n(\mathcal U,A)$ is indexed by strictly increasing
nondegenerate tuples
\[
i_0<\cdots<i_n
\]
whose overlap $a_{i_0\cdots i_n}$ is nonempty. If a value is written on an
arbitrary ordered tuple, it denotes the alternating extension from the
increasing tuple: permuting pairwise distinct indices multiplies the value
by the sign of the permutation, and a tuple with a repeated index has value
$0$. In degree $2$, the cocycle identity on every ordered nondegenerate
quadruple is
\[
\alpha_{jkl}-\alpha_{ikl}+\alpha_{ijl}-\alpha_{ijk}=0.
\]
Thus reversing the orientation of a nondegenerate triple replaces
$\alpha_{ijk}$ by $-\alpha_{ijk}$.
\end{definition}

\begin{definition}[Finite \v{C}ech cocycle representative]\label{def:finite-obstruction-presentation}
Fix an admissible parameter $r:\mathsf{Ref}_s$ at a state $p$, and assume:
\begin{enumerate}[label=(\roman*),nosep]
\item the hypotheses of \Cref{def:cech-gluing-obstruction} hold;
\item a global component section
\[
v\in \Vis_{p,s}(r)
\]
has been chosen;
\item the associated component gerbe $\mathfrak G_v$ is banded by a finite abelian group $A$,
\item a finite covering family
\[
\mathcal U=\{a_i\to a\}_{i\in I},
\qquad
a=[r]_p,
\]
has been chosen together with a \v{C}ech $2$-cocycle
\[
\alpha=(\alpha_{ijk})\in Z^2(\mathcal U,A)
\]
representing the component obstruction class $\omega_v$, written in the ordered
convention of \Cref{def:ordered-cech-convention}.
\end{enumerate}
Define the \emph{coordinate-image subgroup}
\[
H_\alpha:=\langle \alpha_{ijk}\mid i<j<k\text{ and }a_{ijk}\neq\varnothing\rangle\le A,
\]
and the associated \emph{cocycle-coordinate quotient}
\[
Q_{\alpha}^{\mathrm{coord}}:=A/H_\alpha.
\]
\end{definition}

\begin{remark}[Orientation independence of the generated subgroup]\label{rem:strict-subgroup-orientation}
The definition of $H_\alpha$ is independent of whether one generates from
increasing triples or from all oriented nondegenerate triples. Indeed, any
orientation change replaces a generator $\alpha_{ijk}$ by
$-\alpha_{ijk}$, and the subgroup generated by a subset of an abelian group
is unchanged after replacing any generator by its negative. Degenerate
triples contribute $0$ by \Cref{def:ordered-cech-convention}.
\end{remark}

\begin{theorem}[Cocycle-coordinate quotient]\label{thm:initial-visible-quotient}
In the setting of \Cref{def:finite-obstruction-presentation}, let
\[
\pi:A\twoheadrightarrow Q_{\alpha}^{\mathrm{coord}}
\]
be the canonical quotient map. Then:
\begin{enumerate}[label=(\roman*),nosep]
\item the pushed-forward cocycle $\pi_*(\alpha)$ is pointwise zero on every nonempty triple overlap;
\item if $\varphi:A\to B$ is any homomorphism into an abelian group $B$ such that $\varphi(\alpha_{ijk})=0$ for every ordered nondegenerate triple overlap, then there exists a unique homomorphism
\[
\bar\varphi:Q_{\alpha}^{\mathrm{coord}}\to B
\]
with $\varphi=\bar\varphi\circ\pi$.
\end{enumerate}
Hence \(Q_{\alpha}^{\mathrm{coord}}\) is the elementary quotient of \(A\) by the
subgroup generated by the displayed cocycle values; equivalently, it is the
initial coefficient quotient through which this chosen obstruction
cocycle representative becomes pointwise zero.
\end{theorem}

\begin{proof}
By construction and \Cref{rem:strict-subgroup-orientation}, every oriented
cocycle value $\alpha_{ijk}$ lies in the subgroup $H_\alpha$. Therefore
$\pi(\alpha_{ijk})=0$ for every ordered nondegenerate triple overlap, which
proves (i).

For (ii), the hypothesis $\varphi(\alpha_{ijk})=0$ on every ordered
nondegenerate triple overlap implies that every generator of $H_\alpha$
lies in $\ker\varphi$. Hence $H_\alpha\subseteq\ker\varphi$. By the
universal property of the quotient by $H_\alpha$, there exists a unique
homomorphism $\bar\varphi:A/H_\alpha\to B$ satisfying
$\varphi=\bar\varphi\circ\pi$.
\end{proof}

The cocycle-coordinate quotient $A/H_\alpha$ depends on the chosen cocycle representative $\alpha$, not only on the cohomology class $\omega_v$. The next section removes this dependence: the homological evaluation map $\mathrm{ev}_\omega$ is defined from the class itself, and its cokernel $Q_\omega$ is invariant within the homology-equivalent comparison class isolated there. The comparison between the two quotients is given by \Cref{cor:strict-to-intrinsic-visible} and \Cref{prop:chain-vs-homological-visibility}.

\begin{definition}[Coboundary action and annihilating characters]\label{def:gauge-invariance-character}
In the setting of \Cref{def:finite-obstruction-presentation}, the \v{C}ech $1$-cochain group $C^1(\mathcal U,A)$ acts on cocycles by coboundary translation: a cochain
\[
h=(h_{ij})\in C^1(\mathcal U,A)
\]
sends a cocycle $\alpha$ to the cohomologous cocycle
\[
(h\cdot\alpha)_{ijk}:=\alpha_{ijk}+(\delta h)_{ijk}.
\]
A character $\chi\in\widehat A$ is said to \emph{annihilate the coordinate-image subgroup} if it kills $H_\alpha$, equivalently if it factors through the quotient
\[
\pi:A\twoheadrightarrow Q_{\alpha}^{\mathrm{coord}}=A/H_\alpha.
\]
In that case the pushed-forward class $\chi_*[\alpha]$ depends only on the quotient and not on the chosen local trivializations.
\end{definition}

\begin{theorem}[Characters of the cocycle-coordinate quotient]\label{thm:character-definability-barrier}
In the setting of \Cref{def:finite-obstruction-presentation}, let
\[
\widehat A:=\mathrm{Hom}(A,\mathbb T),
\]
where $\mathbb T\subseteq\mathbb C^\times$ is the unit circle, and define
\[
H_\alpha^\perp:=\{\chi\in\widehat A\mid \chi(h)=1\text{ for every }h\in H_\alpha\}.
\]
Call a character $\chi\in\widehat A$ \emph{cocycle-trivializing} for the chosen representative if
\[
\chi(\alpha_{ijk})=1
\]
for every nonempty triple overlap of $\mathcal U$. Then:
\begin{enumerate}[label=(\roman*),nosep]
\item a character $\chi\in\widehat A$ is cocycle-trivializing if and only if $\chi\in H_\alpha^\perp$;
\item the quotient map $\pi:A\twoheadrightarrow Q_{\alpha}^{\mathrm{coord}}$ induces a natural isomorphism
\[
\widehat{Q_{\alpha}^{\mathrm{coord}}}\cong H_\alpha^\perp;
\]
\item consequently, every cocycle-trivializing character factors uniquely through $Q_{\alpha}^{\mathrm{coord}}$, and a character annihilates the coordinate-image subgroup if and only if it lies in $H_\alpha^\perp$.
\end{enumerate}
\end{theorem}

\begin{proof}
Since $H_\alpha$ is generated by the cocycle values $\alpha_{ijk}$, a character $\chi$ is trivial on $H_\alpha$ if and only if it is trivial on each generator $\alpha_{ijk}$. This proves (i).

For (ii), the quotient map $\pi$ induces by precomposition an injective homomorphism
\[
\pi^*:\widehat{Q_{\alpha}^{\mathrm{coord}}}\hookrightarrow \widehat A.
\]
Its image consists exactly of the characters on $A$ that are trivial on $\ker\pi=H_\alpha$, namely $H_\alpha^\perp$. Thus $\pi^*$ identifies $\widehat{Q_{\alpha}^{\mathrm{coord}}}$ with $H_\alpha^\perp$.

Clause \textup{(iii)} is just \Cref{def:gauge-invariance-character} rewritten using \textup{(i)} and \textup{(ii)}.
\end{proof}

\begin{corollary}[Detection by annihilating characters]\label{cor:visible-quotient-character-separation}
In the setting of \Cref{def:finite-obstruction-presentation}, for elements $x,y\in A$ the following are equivalent:
\begin{enumerate}[label=(\roman*),nosep]
\item $\pi(x)=\pi(y)$ in $Q_{\alpha}^{\mathrm{coord}}$;
\item $\chi(x)=\chi(y)$ for every cocycle-trivializing character $\chi\in\widehat A$.
\end{enumerate}
Equivalently, $Q_{\alpha}^{\mathrm{coord}}$ is the quotient of $A$ obtained by identifying exactly those elements that cannot be separated by cocycle-trivializing characters.
\end{corollary}

\begin{proof}
Assume first that $\pi(x)=\pi(y)$. By \Cref{thm:character-definability-barrier}(ii), every cocycle-trivializing character $\chi$ factors as
\[
\chi=\bar\chi\circ\pi
\]
for a unique character $\bar\chi\in\widehat{Q_{\alpha}^{\mathrm{coord}}}$. Hence
\[
\chi(x)=\bar\chi(\pi(x))=\bar\chi(\pi(y))=\chi(y).
\]

Conversely, suppose $\pi(x)\neq\pi(y)$. Then
\[
g:=\pi(x)-\pi(y)\neq 0
\]
in the finite abelian group $Q_{\alpha}^{\mathrm{coord}}$. By the structure theorem for finite abelian groups \cite{DummitFoote2004}, there is an isomorphism
\[
Q_{\alpha}^{\mathrm{coord}}\cong \mathbb Z/n_1\mathbb Z\oplus\cdots\oplus \mathbb Z/n_t\mathbb Z
\]
under which the image of $g$ has some nonzero coordinate, say the $k$th. The coordinate character
\[
\psi(z_1,\dots,z_t):=\exp(2\pi i z_k/n_k)
\]
therefore satisfies $\psi(g)\neq 1$. By \Cref{thm:character-definability-barrier}(ii), the character
\[
\chi:=\psi\circ\pi
\]
is cocycle-trivializing. Since
\[
\chi(x)\chi(y)^{-1}=\psi(\pi(x)-\pi(y))=\psi(g)\neq 1,
\]
we have $\chi(x)\neq\chi(y)$, contradicting (ii). Thus $\pi(x)=\pi(y)$.
\end{proof}

\begin{definition}[Fibre-separating auxiliary register]\label{def:auxiliary-strictification-code}
In the setting of \Cref{def:finite-obstruction-presentation}, an \emph{fibre-separating auxiliary register} consists of a finite set $R$ and an injective map
\[
\iota:A\hookrightarrow Q_{\alpha}^{\mathrm{coord}}\times R
\]
such that the first projection recovers the canonical quotient:
\[
\mathrm{pr}_1\circ\iota=\pi.
\]
Equivalently, the quotient is kept fixed, while the auxiliary register $R$ separates elements inside each fibre of $\pi$.
\end{definition}

\begin{theorem}[Fibre size of an auxiliary register]\label{thm:minimal-information-cost-of-strictification}
In the setting of \Cref{def:auxiliary-strictification-code}, every fibre-separating auxiliary register satisfies
\[
|R|\ge |H_\alpha|.
\]
Equivalently,
\[
\log_2 |R|\ge \log_2 |H_\alpha|.
\]
Moreover, this lower bound is sharp.
\leanverified{paper_gluing_failure_minimal_information_cost_of_strictification}
\end{theorem}

\begin{proof}
Let $\bar a\in Q_{\alpha}^{\mathrm{coord}}$. The fiber $\pi^{-1}(\bar a)$ is a coset of $\ker\pi=H_\alpha$, hence has cardinality $|H_\alpha|$. Since $\iota$ is injective and $\mathrm{pr}_1\circ\iota=\pi$, the restriction of $\iota$ to $\pi^{-1}(\bar a)$ lands inside the subset $\{\bar a\}\times R$ and remains injective. Therefore
\[
|H_\alpha|=|\pi^{-1}(\bar a)|\le |\{\bar a\}\times R|=|R|.
\]
This proves the lower bound.

To see that it is sharp, take $R:=H_\alpha$. Choose a set-theoretic section $s:Q_{\alpha}^{\mathrm{coord}}\to A$ of $\pi$. For each $x\in A$, the difference
\[
h(x):=x-s(\pi(x))
\]
lies in $H_\alpha$. Define
\[
\iota(x):=(\pi(x),h(x))\in Q_{\alpha}^{\mathrm{coord}}\times H_\alpha.
\]
Then $\mathrm{pr}_1\circ\iota=\pi$. If $\iota(x)=\iota(y)$, then $\pi(x)=\pi(y)$ and $h(x)=h(y)$, hence
\[
x=s(\pi(x))+h(x)=s(\pi(y))+h(y)=y.
\]
Thus $\iota$ is injective and $|R|=|H_\alpha|$.
\end{proof}

\begin{corollary}[Minimum auxiliary register size]\label{cor:conservative-recovery-budget}
For a finite \v{C}ech cocycle representative, an injective lift that leaves the quotient coordinate unchanged requires at least \(\log_2|H_\alpha|\) bits in its auxiliary register. This is the elementary fibre-counting consequence of quotienting by the coordinate-image subgroup \(H_\alpha\).
\end{corollary}

\begin{proof}
Any such lift induces a fibre-separating auxiliary register in the sense of \Cref{def:auxiliary-strictification-code}. Apply \Cref{thm:minimal-information-cost-of-strictification}.
\end{proof}

\subsection{A worked finite example}

\begin{example}[A coordinate-image subgroup of order two]\label{ex:one-bit-hidden-obstruction}
Assume an admissible parameter $r:\mathsf{Ref}_s$ together with a three-member cover
\[
\mathcal U=\{a_0,a_1,a_2\to a\},
\qquad
a=[r]_p,
\]
and let the band be the additive group
\[
A=\mathbb Z/4\mathbb Z.
\]
Suppose the chosen obstruction presentation is represented by a cocycle whose only nonzero value on a nonempty triple overlap is
\[
\alpha_{012}=2\in \mathbb Z/4\mathbb Z.
\]
Then
\[
H_\alpha=\langle 2\rangle=\{0,2\},
\qquad
Q_{\alpha}^{\mathrm{coord}}=A/H_\alpha\cong \mathbb Z/2\mathbb Z.
\]

Thus the quotient identifies $0$ with $2$ and identifies $1$ with $3$. By \Cref{thm:initial-visible-quotient}, this is the initial coefficient quotient through which the chosen cocycle representative becomes pointwise zero.

For $m\in\mathbb Z/4\mathbb Z$, define a character
\[
\chi_m(x):=\exp(2\pi i mx/4).
\]
Then
\[
\chi_m(\alpha_{012})=\chi_m(2)=(-1)^m,
\]
so \Cref{thm:character-definability-barrier} implies that the cocycle-trivializing characters are exactly
\[
\chi_0
\qquad\text{and}\qquad
\chi_2.
\]
Hence the only surviving linear detection is parity: $\chi_2$ distinguishes the two cosets of $H_\alpha$, while the phase-sensitive characters $\chi_1$ and $\chi_3$ do not factor through the quotient. By \Cref{cor:visible-quotient-character-separation}, the quotient is therefore exactly the quotient detected by all cocycle-trivializing characters.

Finally, \Cref{thm:minimal-information-cost-of-strictification} yields
\[
|R|\ge |H_\alpha|=2,
\]
so at least one auxiliary bit is required to distinguish all elements of $\mathbb Z/4\mathbb Z$ while keeping the quotient coordinate fixed. This lower bound is attained by the auxiliary register
\[
0\mapsto (\overline{0},0),\qquad
2\mapsto (\overline{0},1),\qquad
1\mapsto (\overline{1},0),\qquad
3\mapsto (\overline{1},1),
\]
where $\overline{0},\overline{1}\in Q_{\alpha}^{\mathrm{coord}}\cong \mathbb Z/2\mathbb Z$ and the second component is a one-bit register.
\end{example}
