\section{Presheaves, sheafification, and matching without amalgamation}\label{sec:null-decomposition}

We study a presheaf of local sections attached to an admissible parameter, extending the meet-semilattice formulation of the companion paper \cite{MaZhang2026ConditionalGluingFailure} to the pullback-based sites used here. The basic distinction is between local nonemptiness, existence of a matching family, and existence of a global section.

The following standing hypothesis will be used throughout the section and cited explicitly when needed.

\begin{enumerate}[label=\textup{(RC)},nosep,leftmargin=3em]
\item\label{hyp:refinement-conservation} \emph{Refinement conservation.} If $q\le p$, then admissible parameters, covers, and compatible local families at $q$ restrict functorially to those at $p$, and a sheafified global section already uniquely determined at $p$ is not split by further refinement.
\end{enumerate}

\begin{remark}[Relation to the companion paper]
The companion paper \cite{MaZhang2026ConditionalGluingFailure} develops the same presheaf and sheafification framework on meet-semilattice sites. In this section those constructions are recast on the pullback-based sites of \Cref{def:local-reference-frame}, so the sheafification criterion, split banded groupoid lifts, stackification comparison, essential surjectivity on the terminal fibre, and sheafified global sections are all stated without assuming that overlaps arise from semilattice meets.
\end{remark}

\subsection{Admissible parameters and presheaves}

Fix a family of designated sorts
\[
S_{\mathrm{ref}}\subseteq S.
\]
For each $s\in S_{\mathrm{ref}}$, the language contains a parameter sort $\mathsf{Ref}_s$ and an admissibility predicate $\Adm_s(\cdot)$.
Layer placement is fixed as follows. The sort \(\mathsf{Ref}_s\) and its
equality belong to \(\Sigma_1\); the admissibility predicate
\(\Adm_s\), together with \(\LocSec_s,\CompSec_s,\Sec_s\) and the
site-structure symbols, belongs to \(\Sigma_2\setminus\Sigma_1\). Thus a
lower-fragment formula may quantify over references but cannot define or
mention admissibility unless the statement explicitly restricts attention to the
admissible subdomain.

\begin{definition}[Base objects]\label{def:reference-classes}
For each state $p$ and each parameter sort $s$, fix a set
\[
\mathcal R_s(p)
\]
and a surjective assignment
\[
r\longmapsto [r]_p\in \mathcal R_s(p)
\]
defined for the admissible parameters $r:\mathsf{Ref}_s$ at $p$. The element $[r]_p$ is called the \emph{base object} of $r$ at $p$. The later reference-refinement site is built on these base objects; the bracket notation does not denote a quotient or an equivalence class.
\end{definition}

\begin{definition}[Reference-refinement site]\label{def:local-reference-frame}
For a state $p$ and a designated sort $s$, a \emph{reference-refinement site} consists of the following data.
\begin{enumerate}[label=(\roman*),nosep]
\item A reflexive, transitive refinement relation $\sqsubseteq_{p,s}$ on $\mathcal R_s(p)$.
\item The associated small thin category $\mathcal B_{p,s}$ whose objects are the elements of $\mathcal R_s(p)$ and in which there is a unique morphism
\[
b\longrightarrow a
\]
if and only if $b\sqsubseteq_{p,s} a$.
\item A Grothendieck pretopology $\Cov_{p,s}$ on $\mathcal B_{p,s}$ satisfying the identity, pullback-stability, and transitivity axioms.
\item The standing pullback hypothesis that, for every covering family
\[
\{a_i\to a\}_{i\in I}\in \Cov_{p,s}(a)
\]
and every pair $(i,j)$, the fibre product
\[
a_i\times_a a_j
\]
exists in $\mathcal B_{p,s}$.
\end{enumerate}
The fibre product clause is a condition on the site, not on a later
\v{C}ech cochain display: if two members of a covering family are
disjoint in a concrete presentation, their pullback must still be
represented by an object of the site, typically a strict initial object.
One may omit such empty pullback objects from a nonzero \v{C}ech nerve or
from a cochain complex whose coefficient sheaf has only the zero section
there, but not from the reference-refinement site itself. Because
$\mathcal B_{p,s}$ is thin, the fibre product $a_i\times_a a_j$ is the
meet of $a_i$ and $a_j$ over $a$ whenever that meet exists. We write
\[
\mathcal C_{p,s}:=(\mathcal B_{p,s},\Cov_{p,s})
\]
for the resulting site.
\end{definition}

\begin{definition}[Presheaf of local sections]\label{def:local-object-functor}
A \emph{presheaf of local sections} on $(p,s)$ is a contravariant functor
\[
F_{p,s}:\mathcal{B}_{p,s}^{op}\to\mathsf{Set}.
\]
If $b\sqsubseteq_{p,s} a$, the induced restriction map is denoted by
\[
\res_b^a:F_{p,s}(a)\to F_{p,s}(b).
\]
\end{definition}

\begin{definition}[Three existence levels]\label{def:existence-levels}
Let $r:\mathsf{Ref}_s$ and assume $M,p\Vdash \Adm_s(r)$. Write $a=[r]_p$.
In the pullback-based reference-refinement site of \Cref{def:local-reference-frame}, overlaps are checked on the fibre products $a_i\times_a a_j$ supplied there, rather than on semilattice meets.
\begin{enumerate}[label=(\roman*),nosep]
\item $M,p\Vdash \LocSec_s(r)$ if some cover $\mathcal U=\{a_i\to a\}_{i\in I}\in\Cov_{p,s}(a)$ has $F_{p,s}(a_i)\neq\varnothing$ for every $i\in I$.
\item $M,p\Vdash \CompSec_s(r)$ if some cover $\mathcal U=\{a_i\to a\}_{i\in I}\in\Cov_{p,s}(a)$ carries a matching family $(\sigma_i)_{i\in I}$ with each $\sigma_i\in F_{p,s}(a_i)$ and
\[
\res_{a_i\times_a a_j}^{a_i}(\sigma_i)=\res_{a_i\times_a a_j}^{a_j}(\sigma_j).
\]
\item $M,p\Vdash \Sec_s(r)$ if the fibre $F_{p,s}(a)$ is nonempty.
\end{enumerate}
\end{definition}

\begin{proposition}[Global sections yield compatible local sections]\label{prop:global-implies-compatible-local}
In the pullback-based site of \Cref{def:local-reference-frame}, so without assuming a meet-semilattice presentation of overlaps, if $M,p\Vdash \Sec_s(r)$, then
\[
M,p\Vdash \CompSec_s(r)
\qquad\text{and}\qquad
M,p\Vdash \LocSec_s(r).
\]
\end{proposition}

\begin{proof}
Let $a=[r]_p$ and choose $\sigma\in F_{p,s}(a)$. The identity axiom for the pretopology gives the trivial cover $\{a\to a\}$. The singleton family $(\sigma)$ is therefore a compatible local family, so both existence of a matching family and local nonemptiness follow immediately.
\end{proof}

\subsection{Sheafification and matching without amalgamation}

For fixed $(p,s)$, the reference-refinement site of \Cref{def:local-reference-frame} determines a presheaf
\[
F_{p,s}\in\mathbf{PSh}(\mathcal{B}_{p,s}).
\]
Write
\[
a_{p,s}:\mathbf{PSh}(\mathcal{B}_{p,s})\to\mathbf{Sh}(\mathcal{C}_{p,s})
\]
for the associated sheafification functor.

\begin{definition}[Separatedness at an object]\label{def:separated-at-a}
Let $\mathcal C$ be a site, let $F$ be a presheaf on $\mathcal C$, and let $a$ be an object of $\mathcal C$. We say that $F$ is \emph{separated at $a$} if for every covering family
\[
\{a_i\to a\}_{i\in I}
\]
and every pair of sections $\sigma,\tau\in F(a)$, the equalities
\[
\res_{a_i}^{a}(\sigma)=\res_{a_i}^{a}(\tau)
\qquad
(\forall i\in I)
\]
imply $\sigma=\tau$.
\end{definition}

\begin{definition}[Separatedness on the slice site associated to the parameter]\label{def:separated-restriction}
For an admissible parameter $r:\mathsf{Ref}_s$ at $p$, write
\[
a=[r]_p,
\qquad
\mathcal C_a:=\mathcal C_{p,s}/a.
\]
The restricted presheaf $F_{p,s}|_{\mathcal C_a}$ is \emph{separated on the slice} if it is separated at every object of $\mathcal C_a$.
Equivalently, separatedness is required not only at the terminal object
$a$, but also at every cover member, every pullback overlap, and every
object introduced by a refinement inside the slice. This distinction is
used essentially below: equality in a sheafification value over a cover
member gives equality of the original presheaf sections only after a
further cover of that cover member, and the upgrade back to equality over
the cover member itself requires separatedness at that object.
\end{definition}

\begin{lemma}[Explicit plus--plus normal form for sheafification]\label{lem:strict-sheafification-representatives}
Let $\mathcal C$ be a site with pullbacks of covering families, let
$F$ be a presheaf of sets on $\mathcal C$, and let
$u:F\to aF$ be the sheafification unit. For every object $X$ and every
section $\xi\in(aF)(X)$ there is a covering family
\[
\{X_i\to X\}_{i\in I}
\]
and sections $\sigma_i\in F(X_i)$ such that:
\begin{enumerate}[label=(\roman*),nosep]
\item for every pair $(i,j)$, the restrictions of $\sigma_i$ and
$\sigma_j$ to the pullback $X_i\times_X X_j$ agree after a covering
refinement of that pullback object;
\item the family $(u(\sigma_i))_i$ glues in the sheaf $aF$ to $\xi$;
\end{enumerate}
No strict matching family of $F$ is asserted in this first conclusion.
For an arbitrary presheaf, the compatibility obtained from a section of
\((aF)(X)\) is only compatibility after refinements of the pairwise
overlaps. This is exactly the loss of information caused by the two
successive plus-construction quotient relations. In particular, the data
in \textup{(i)} cannot be used as cover-level identity descent data unless
the refined equalities have first been upgraded to equalities on the
original pullback overlaps.

If $F$ is separated at every object of the slice $\mathcal C/X$, then the
equalities after covering refinements in \textup{(i)} upgrade to
equalities in $F(X_i\times_X X_j)$; in that case the displayed data are a
strict matching family in $F$ on the cover of $X$.
\end{lemma}

\begin{proof}
We use the usual plus construction
\cite[III.5]{MacLaneMoerdijk1994}, \cite[Tags 00W1, 00WK]{StacksProject},
but we record the flattening explicitly. For an arbitrary presheaf, a
section of \(aF\) is not a matching family of \(F\)-sections on one cover;
it is a matching family of \(F^+\)-sections, and each of those
\(F^+\)-sections is itself represented by \(F\)-sections only up to a
local equivalence relation.

For an object \(T\), a section of \(F^+(T)\) is an equivalence class of
strict \(F\)-matching data
\[
\{T_\alpha\to T\}_{\alpha\in A},
\qquad
s_\alpha\in F(T_\alpha),
\]
whose restrictions agree strictly on each pullback
\(T_\alpha\times_T T_\beta\). Two such matching families define the same
element of \(F^+(T)\) precisely when their pullbacks to a common covering
refinement agree in \(F\). More explicitly, the common refinement is taken
after pulling the two covers back to each \(T_\alpha\times_TT'_\beta\);
there is no assertion that the original sections agree on
\(T_\alpha\times_TT'_\beta\) itself. In particular, if two elements of
\(F^+(T)\) are
represented by matching families
\[
\{T_\alpha\to T,s_\alpha\}_{\alpha\in A},
\qquad
\{T'_\beta\to T,t_\beta\}_{\beta\in B},
\]
then their equality in $F^+(T)$ means that, for every fibre product
$T_\alpha\times_TT'_\beta$, there is a covering family of that fibre
product on which the two pulled-back $F$-sections become equal. This is
the only equality criterion used below.

Since \(aF=(F^+)^+\), a section \(\xi\in(aF)(X)\) is represented by a
covering family
$\{Y_\ell\to X\}_{\ell\in L}$ and sections
\[
\eta_\ell\in F^+(Y_\ell)
\]
which form a strict \(F^+\)-matching family. The word ``strict'' here
refers only to equality of the \(\eta_\ell\) in \(F^+\) on the overlaps
\(Y_\ell\times_XY_{\ell'}\); after expanding the \(\eta_\ell\) to
\(F\)-data, that equality again means equality only after a further cover.
Choose, for each $\ell$, a
plus representative
\[
\{Y_{\ell m}\to Y_\ell\}_{m\in M_\ell},
\qquad
\sigma_{\ell m}\in F(Y_{\ell m})
\]
of $\eta_\ell$. By pullback stability and transitivity, the composites
$Y_{\ell m}\to X$ form a covering family of $X$.

Fix two composites $Y_{\ell m}\to X$ and $Y_{\ell' m'}\to X$, and write
\[
T:=Y_{\ell m}\times_XY_{\ell' m'}.
\]
The strict matching condition in \((F^+)^+\) says that the two restrictions
of \(\eta_\ell\) and \(\eta_{\ell'}\) to
\(Y_\ell\times_XY_{\ell'}\) are equal as elements of \(F^+\). Pull this
equality back to \(T\). Applying the just stated equality criterion in
\(F^+\) to the two chosen plus representatives gives a cover
$\{T_q\to T\}_q$ on which the restrictions of the two $F$-sections
$\sigma_{\ell m}$ and $\sigma_{\ell' m'}$ are equal in $F(T_q)$. These
are covers of the already formed pullback overlap \(T\), not replacements
for \(T\) in the \v{C}ech nerve. They give exactly the local-on-overlaps
compatibility in \textup{(i)} and nothing stronger without a separatedness
hypothesis. The image of the family
$(u(\sigma_{\ell m}))_{\ell,m}$ in $(F^+)^+(X)$ is the original
representative $(\eta_\ell)_\ell$, hence glues to $\xi$; this proves
\textup{(ii)}.

If $F$ is separated at every object of $\mathcal C/X$, then in particular
it is separated at each pullback overlap \(X_i\times_XX_j\). Two sections
over such an overlap that agree after a covering of that overlap are
therefore already equal by separatedness at the overlap object. This
converts the local-on-overlaps compatibility into strict equality on every
$X_i\times_X X_j$. Notice that no cover has been replaced at this last
step: the output is a genuine matching family on the original cover of
\(X\), not a matching family only after changing the \v{C}ech nerve.
\end{proof}

\begin{lemma}[Compatible representatives of a sheafified section]\label{lem:strictify-component-representatives}
Let $\mathcal C$ be a site with pullbacks of covering families, let
$F$ be a presheaf of sets on $\mathcal C$, and let \(X\) be an object.
Assume that \(F\) is separated at every object of the slice
\(\mathcal C/X\). If \(\xi\in(aF)(X)\), then there is a cover
\[
\{X_i\to X\}_{i\in I}
\]
and sections \(\sigma_i\in F(X_i)\) which form a strict matching family on
the original pullback overlaps \(X_i\times_X X_j\), and whose images under
the sheafification unit glue to \(\xi\).
\end{lemma}

\begin{proof}
Apply \Cref{lem:strict-sheafification-representatives}. This gives a
cover and \(F\)-sections whose restrictions to each original pullback
overlap agree after a further cover of that overlap. Since each overlap is
an object of \(\mathcal C/X\), separatedness on the slice lets
\Cref{lem:separated-restriction-refined-equality} upgrade all refined overlap
equalities to equalities over the original overlaps. The same
plus--plus representative still glues under the unit to \(\xi\).
\end{proof}

\begin{lemma}[Descent of refined equality for a separated restriction]\label{lem:separated-restriction-refined-equality}
Let $\mathcal C$ be a site with pullbacks of covering families, let $F$
be a presheaf of sets on $\mathcal C$, and let $X$ be an object. Suppose
that $F$ is separated at every object of the slice $\mathcal C/X$. If
\[
\{X_i\to X\}_{i\in I}
\]
is a cover and sections $\sigma_i\in F(X_i)$ satisfy the following
refined-overlap condition: for every pair $(i,j)$, writing
$X_{ij}:=X_i\times_X X_j$, the two restrictions of $\sigma_i$ and
$\sigma_j$ to $X_{ij}$ become equal after some covering family of
$X_{ij}$, then the family $(\sigma_i)_i$ is a strict matching family:
\[
\res_{X_{ij}}^{X_i}(\sigma_i)=\res_{X_{ij}}^{X_j}(\sigma_j)
\qquad(\forall i,j).
\]
\end{lemma}

\begin{proof}
For each pair $(i,j)$ the overlap $X_{ij}$ is an object of the slice
$\mathcal C/X$. By hypothesis $F$ is separated at $X_{ij}$. The two
sections
\[
\res_{X_{ij}}^{X_i}(\sigma_i),
\qquad
\res_{X_{ij}}^{X_j}(\sigma_j)
\]
agree after a cover of $X_{ij}$, so separatedness at $X_{ij}$ forces
their equality in $F(X_{ij})$. This is exactly the strict matching
condition.
\end{proof}

\begin{lemma}[Equality in the unit is local equality]\label{lem:sheafification-unit-equality-local}
Let $\mathcal C$ be a site with pullbacks of covering families, let
$F$ be a presheaf of sets on $\mathcal C$, and let
$u:F\to aF$ be the sheafification unit. If
\[
u_X(\sigma)=u_X(\tau)
\qquad
(\sigma,\tau\in F(X)),
\]
then there is a covering family $\{X_i\to X\}_{i\in I}$ such that
\[
\res_{X_i}^{X}(\sigma)=\res_{X_i}^{X}(\tau)
\qquad(\forall i\in I).
\]
Conversely, equality after such a cover implies
$u_X(\sigma)=u_X(\tau)$.
\end{lemma}

\begin{proof}
For the plus construction, the two unit representatives of
$\sigma,\tau\in F(X)$ are represented by the identity cover of $X$. Their
images in $F^+(X)$ are equal exactly when, after a covering refinement of
that identity cover, the restrictions of $\sigma$ and $\tau$ agree in
$F$. Applying the same statement to the unit
$F^+\to(F^+)^+=aF$, equality of $u_X(\sigma)$ and $u_X(\tau)$ gives a cover
$\{Y_j\to X\}$ on which the two corresponding sections of $F^+$ agree.
For each $j$, equality in $F^+(Y_j)$ gives a further cover
$\{X_{ji}\to Y_j\}_i$ on which the restrictions of $\sigma$ and $\tau$
agree in $F$. By transitivity, the composites $X_{ji}\to X$ form the
required cover. The converse follows because $aF$ is a sheaf: if the two
unit sections agree after a cover, separatedness of the sheaf $aF$ forces
them to agree over $X$.
\end{proof}

\begin{proposition}[Existence of a matching family gives a sheafified section]\label{prop:compsec-to-sheafification}
Let $r:\mathsf{Ref}_s$ be admissible at $p$, and write $a=[r]_p$. Then, in the pullback-based thin site $\mathcal C_{p,s}$ of \Cref{def:local-reference-frame},
\[
M,p\Vdash \CompSec_s(r)
\quad\Longrightarrow\quad
\bigl(a_{p,s}F_{p,s}\bigr)(a)\neq\varnothing.
\]
\end{proposition}

\begin{proof}
Let
\[
G:=a_{p,s}F_{p,s}.
\]
By \Cref{def:existence-levels}(ii), a proof of $M,p\Vdash\CompSec_s(r)$
supplies a cover $\{a_i\to a\}_{i\in I}$ and local sections
$\sigma_i\in F_{p,s}(a_i)$ satisfying the displayed matching condition on
the pairwise pullbacks. Applying the sheafification unit on each $a_i$
gives a matching family in the sheaf $G$; the sheaf gluing axiom for $G$
therefore gives a section of $G(a)$.
\end{proof}

\begin{theorem}[Sheafification characterizes existence of a matching family on slice sites on which the restricted presheaf is separated]\label{thm:sheafification-characterization}
Let $r:\mathsf{Ref}_s$ be admissible at $p$, write $a=[r]_p$, and assume that
$F_{p,s}|_{\mathcal C_a}$ is separated on the slice in the sense of
\Cref{def:separated-restriction}. Then, in the pullback-based thin
site $\mathcal C_{p,s}$ of \Cref{def:local-reference-frame},
\[
M,p\Vdash \CompSec_s(r)
\quad\Longleftrightarrow\quad
\bigl(a_{p,s}F_{p,s}\bigr)(a)\neq\varnothing.
\]
\end{theorem}

\begin{proof}
The forward implication is \Cref{prop:compsec-to-sheafification}.
Conversely, suppose
\[
\xi\in \bigl(a_{p,s}F_{p,s}\bigr)(a).
\]
Since sheafification is the plus--plus construction for an arbitrary
presheaf, one cannot read $\xi$ directly as a strict $F$-matching family
on the first cover that appears in a plus representative. The needed
strict representative is exactly the conclusion of
\Cref{lem:strictify-component-representatives}, applied to the restricted
presheaf \(F_{p,s}|_{\mathcal C_a}\) on the slice. This gives a covering family
\[
\{b_\ell\to a\}_{\ell\in L}
\]
and sections $\tau_\ell\in F_{p,s}(b_\ell)$ satisfying the strict
equalities
\[
\res_{b_\ell\times_a b_m}^{b_\ell}(\tau_\ell)
=
\res_{b_\ell\times_a b_m}^{b_m}(\tau_m)
\qquad(\ell,m\in L).
\]
This is precisely the compatibility requirement in
\Cref{def:existence-levels}(ii). Thus the refined strict representative
exhibits
\[
M,p\Vdash \CompSec_s(r).
\]
\end{proof}

\begin{corollary}[Sheaf condition at a reference]\label{cor:sheaf-local-to-global}
In the pullback-based thin site of \Cref{def:local-reference-frame}, assume
$F_{p,s}|_{\mathcal C_a}$ is separated on the slice. If $F_{p,s}$ is already a
sheaf at $a=[r]_p$, then existence of a global section and existence of a matching family coincide:
\[
M,p\Vdash \Sec_s(r)
\quad\Longleftrightarrow\quad
M,p\Vdash \CompSec_s(r)
\]
\end{corollary}

\begin{proof}
If $F_{p,s}$ is already a sheaf at $a$, then the unit $F_{p,s}\to a_{p,s}F_{p,s}$ is an isomorphism at $a$. The claim therefore follows from \Cref{thm:sheafification-characterization}.
\end{proof}

\begin{corollary}[Sheafification removes matching without amalgamation]\label{cor:sheafification-removes-glue}
Let $\Sec_s^{\sharp}(r)$ denote existence of a global section computed with the
sheafified presheaf of local sections $a_{p,s}F_{p,s}$. In the pullback-based
thin site of \Cref{def:local-reference-frame}, assume that
$F_{p,s}|_{\mathcal C_a}$ is separated on the slice. Then
\[
M,p\Vdash \Sec_s^{\sharp}(r)
\quad\Longleftrightarrow\quad
M,p\Vdash \CompSec_s(r)
\]
In particular, no admissible parameter remains in matching without amalgamation after sheafification.
\end{corollary}

\begin{proof}
By definition,
\[
M,p\Vdash \Sec_s^{\sharp}(r)
\quad\Longleftrightarrow\quad
\bigl(a_{p,s}F_{p,s}\bigr)(a)\neq\varnothing .
\]
The direct equivalence of
\Cref{thm:sheafification-characterization} identifies the right-hand side
with \(M,p\Vdash\CompSec_s(r)\). This avoids any need to extract strict
compatible local sections from the already sheafified functor.
\end{proof}
