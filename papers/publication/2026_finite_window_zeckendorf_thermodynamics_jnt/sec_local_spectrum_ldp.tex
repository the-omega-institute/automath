\subsection{Large deviations across the corner}

\begin{theorem}[Full large deviations across the freezing corner]
\label{thm:full-fiber-ldp}
Let \(\mathbf X_m\) be uniformly distributed on \(X_m\), and set
\[
 \begin{gathered}
 y_m(x)=\frac1m\log d_m(x),\qquad Y_m=y_m(\mathbf X_m),\\
 t_c=-\sigma_0,\qquad v_c=P'_+(t_c),
 \qquad \alpha_+=\frac12\log\varphi.
 \end{gathered}
\]
Then \((Y_m)\) satisfies a full large deviation principle with speed \(m\)
and good rate function
\begin{equation}
 I(\alpha)=\sup_{t\in\mathbb R}
 \{t\alpha-P(t)+\log\varphi\}.
 \label{eq:full-fiber-ldp-rate}
\end{equation}
Its effective domain is \([0,\alpha_+]\).  On the coexistence interval,
\begin{equation}
 I(\alpha)=\log\varphi-\sigma_0\alpha
 \qquad(0\le\alpha\le v_c).
 \label{eq:frozen-affine-rate}
\end{equation}
More precisely, for every \(0<\alpha<v_c\),
\begin{equation}
 \begin{aligned}
 &\lim_{\varepsilon\downarrow0}\liminf_{m\to\infty}
 \frac1m\log\Pr\{|Y_m-\alpha|<\varepsilon\}\\
 &\quad=\lim_{\varepsilon\downarrow0}\limsup_{m\to\infty}
 \frac1m\log\Pr\{|Y_m-\alpha|<\varepsilon\}\\
 &\quad=-\bigl(\log\varphi-\sigma_0\alpha\bigr).
 \end{aligned}
 \label{eq:frozen-local-large-deviation}
\end{equation}
\end{theorem}

\begin{proof}
For \(\alpha\in\mathbb R\) and \(\varepsilon>0\), write
\[
 A_m(\alpha,\varepsilon)=\card{\{x\in X_m:
 |m^{-1}\log d_m(x)-\alpha|<\varepsilon\}}.
\]
Fix \(0<\alpha<v_c\), choose \(t>t_c\), and put \(v=P'(t)\).  Strict
convexity gives \(v>v_c>\alpha\), so \(\rho=\alpha/v\in(0,1)\).  Define
the tilted law on \(X_n\) by
\[
 \nu_{n,t}(x)=\frac{d_n(x)^t}{S_t(n)}.
\]
For sufficiently small positive \(s\), the exponential Markov inequality and
\Cref{thm:all-real-pressure} give
\[
 \limsup_{n\to\infty}\frac1n
 \log\nu_{n,t}\{y_n\ge v+\delta\}
 \le P(t+s)-P(t)-s(v+\delta)<0.
\]
The same estimate with \(s<0\) treats \(y_n\le v-\delta\).  Hence
\[
 \sum_{\substack{x\in X_n\\|y_n(x)-v|<\delta}}d_n(x)^t
 =\exp\{n(P(t)+o(1))\},
\]
and bounding each summand above yields
\begin{equation}
 \liminf_{n\to\infty}\frac1n\log A_n(v,\delta)
 \ge P(t)-tv-|t|\delta.
 \label{eq:exposed-source-count}
\end{equation}

The map \(\chi_n\) defined after
\Cref{thm:fiber-partition-interval} is a bijection from \(X_n\) onto the
integer points of
\(\mathcal I_n\sqcup\mathcal I_{n+1}\), with
\(d_n(x)=R(\chi_n(x))\).  Choose \(\delta<v/2\), so the source
multiplicities exceed two for large \(n\).  By
\Cref{lem:uniform-layer-count}, each generating orbit supplies at most four
source states.  The orbit-layer formulas show that every represented orbit
has generator length at most \(n+1\).  Selecting its generator produces a
set \(\mathcal G_n\) with
\begin{equation}
 \card{\mathcal G_n}\ge\frac14A_n(v,\delta),
 \qquad L(g)\le n+1\quad(g\in\mathcal G_n),
 \label{eq:source-generating-orbits}
\end{equation}
and \(n^{-1}\log R(g)\in(v-\delta,v+\delta)\).  Distinct elements generate
disjoint orbits.

Let \(n_m=\lfloor\rho m\rfloor\).  Since
\(\rho m-1<n_m\le\rho m\) and \(L(g)\le n_m+1\), every
\(g\in\mathcal G_{n_m}\) satisfies \(L(g)<m\) once
\((1-\rho)m>2\).  The positive integer \(m-L(g)\) has one of the two
parities, and we define
\[
 z_m(g)=
 \begin{cases}
 [a]g,&m-L(g)=2a,\\
 [a]\tau g,&m-L(g)=2a+1.
 \end{cases}
\]
Here \(a=(m-L(g))/2\) or \(a=(m-L(g)-1)/2\), respectively, so it is a
nonnegative integer in both cases.
Then \(L(z_m(g))=m\) and \(R(z_m(g))=R(g)>1\), so
\(z_m(g)\in\mathcal I_m\).  Freeness and \(\chi_m^{-1}\) make the map into
\(X_m\) injective.  Moreover,
\[
 \left|m^{-1}\log R(z_m(g))-\alpha\right|
 \le \rho\delta+\frac{\delta+v}{m}.
\]
Combining the injection with
\eqref{eq:exposed-source-count}--\eqref{eq:source-generating-orbits} gives,
for \(\varepsilon>\rho\delta\),
\begin{equation}
 \liminf_{m\to\infty}\frac1m\log A_m(\alpha,\varepsilon)
 \ge\rho\bigl(P(t)-tv-|t|\delta\bigr).
 \label{eq:orbit-filling-lower-bound}
\end{equation}
Let \(\delta\downarrow0\) and then \(t\downarrow t_c\).  Since
\(P(t)\to0\), \(v\to v_c\), and
\[
 \rho(P(t)-tv)=\alpha\left(\frac{P(t)}v-t\right)
 \longrightarrow\sigma_0\alpha,
\]
we obtain
\begin{equation}
 \lim_{\varepsilon\downarrow0}\liminf_{m\to\infty}
 \frac1m\log A_m(\alpha,\varepsilon)\ge\sigma_0\alpha.
 \label{eq:frozen-count-lower}
\end{equation}

On the same local set,
\(d_m(x)^{-\sigma_0}\ge
e^{-\sigma_0m(\alpha+\varepsilon)}\).  Therefore
\[
 A_m(\alpha,\varepsilon)e^{-\sigma_0m(\alpha+\varepsilon)}
 \le S_{-\sigma_0}(m)=e^{o(m)},
\]
where the last equality follows from
\Cref{thm:critical-finite-window-gibbs}.  This matches
\eqref{eq:frozen-count-lower}.  Division by
\(\card{X_m}=F_{m+2}\) proves \eqref{eq:frozen-local-large-deviation}.

If \(0\le\alpha\le v_c\), then \(P(t)=0\) for \(t\le t_c\), while
convexity gives \(P(t)\ge v_c(t-t_c)\) for \(t\ge t_c\).  Thus
\(t\alpha-P(t)\le t_c\alpha\), with equality at \(t_c\), proving
\eqref{eq:frozen-affine-rate}.

The largest-fiber corollary gives
\[
 \max_{x\in X_m}m^{-1}\log d_m(x)\longrightarrow\alpha_+.
\]
For \(t>0\),
\(D_m^t\le S_t(m)\le F_{m+2}D_m^t\), and hence
\[
 \alpha_+\le\frac{P(t)}t
 \le\alpha_++\frac{\log\varphi}{t}.
\]
Since \(P'\) is increasing, it has a limit in \((0,+\infty]\).  The identity
\[
 \frac{P(t)}t=\frac{P(0)}t+\frac1t\int_0^tP'(u)\,du
\]
and Ces\`aro convergence show that this limit is \(\alpha_+\).  Strict
convexity and analyticity on \((t_c,\infty)\) therefore make every
\(\alpha\in(v_c,\alpha_+)\) exposed by a unique \(t>t_c\).  The tilted
estimate above with \(n=m\) supplies
\begin{equation}
 \lim_{\varepsilon\downarrow0}\liminf_{m\to\infty}
 \frac1m\log\Pr\{|Y_m-\alpha|<\varepsilon\}
 \ge P(t)-t\alpha-\log\varphi=-I(\alpha).
 \label{eq:exposed-local-large-deviation}
\end{equation}

The logarithmic moment generating functions under the uniform law converge
on \(\mathbb R\) to \(P(t)-\log\varphi\).  The variables have common compact
support up to a vanishing endpoint error, so the G\"artner--Ellis upper bound
\cite[Theorem~2.3.6(a)]{DemboZeitouni2010} applies to every closed set.

If \(\alpha<0\), letting \(t\to-\infty\) in
\eqref{eq:full-fiber-ldp-rate} and using \(P(t)=0\) gives
\(I(\alpha)=+\infty\).  If \(\alpha>\alpha_+\), the limit
\(P(t)/t\to\alpha_+\) gives the same conclusion as \(t\to\infty\).
Conversely, for \(0\le\alpha\le\alpha_+\), the function
\(t\alpha-P(t)\) is bounded above on \(( -\infty,t_c]\), on the compact
interval \([t_c,0]\), and on \([0,\infty)\), where
\(P(t)\ge t\alpha_+\).  Thus the effective domain is precisely
\([0,\alpha_+]\).  A finite convex rate is continuous relative to this
interval, including at its endpoints.

Let \(G\subset\mathbb R\) be open and meet \([0,\alpha_+]\), and fix
\(a\in G\cap[0,\alpha_+]\).  The local lower bounds apply directly when
\(a\in(0,\alpha_+)\setminus\{v_c\}\).  At \(v_c\), choose interior points
converging from either side; at \(\alpha_+\), choose exposed points
increasing to the endpoint.  Relative continuity gives convergence of their
rates, while openness provides neighborhoods contained in \(G\).  Applying
the local lower bounds before taking the endpoint limit proves the lower
bound there.  At zero, either multiplicity-one state supplies it directly.
Taking the infimum over \(G\cap[0,\alpha_+]\) proves the full open-set lower
bound.  The compact effective domain makes \(I\) good and completes the LDP.
\end{proof}
