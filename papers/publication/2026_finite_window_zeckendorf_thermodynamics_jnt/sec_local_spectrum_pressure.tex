For a reduced fraction \(p/q\in(0,1)\), write its unique negative continued
fraction as
\[
 \frac pq=
 \cfrac1{e_1-\cfrac1{e_2-\ddots-\cfrac1{e_k}}},
 \qquad e_i\ge2,
\]
and put
\[
 d(p/q)=\sum_{i=1}^k(e_i-1),
 \qquad c(p/q)=2d(p/q)+1.
\]
For real \(s\), define the denominator-weighted letter sums
\[
 b_\ell(s)=
 \sum_{\substack{0<p<q,\ (p,q)=1\\c(p/q)=\ell}}q^{-s}.
\]
Define the negative continuant by
\[
 \begin{gathered}
  \mathcal D(\varnothing)=1,\qquad \mathcal D(e_1)=e_1,\\
  \mathcal D(e_1,\ldots,e_k)
  =e_k\mathcal D(e_1,\ldots,e_{k-1})
   -\mathcal D(e_1,\ldots,e_{k-2}).
 \end{gathered}
\]
Then
\begin{equation}
 b_{2d+1}(s)=
 \sum_{r=1}^d
 \sum_{\substack{f_1+\cdots+f_r=d\\f_i\ge1}}
 \mathcal D(f_1+1,\ldots,f_r+1)^{-s},
 \qquad b_{2d}(s)=0.
 \label{eq:weighted-letter-coefficients}
\end{equation}
Put
\begin{equation}
 u_0(s):=1,\qquad
 u_j(s):=\sum_{\ell=1}^j b_\ell(s)u_{j-\ell}(s).
 \label{eq:weighted-renewal-coefficients}
\end{equation}

\begin{proposition}[Weighted generator renewal]
\label{prop:weighted-generator-renewal}
For \(j\ge1\), let
\[
 \Lambda_j(s):=\sum_{n\in\mathcal I_j}R(n)^{-s}.
\]
Then, for every real \(s\),
\begin{equation}
 \Lambda_j(s)
 =1+u_j(s)+2\sum_{\ell=1}^{j-1}u_\ell(s)
 =2\sum_{\ell=0}^{j-1}u_\ell(s)+u_j(s)-1.
 \label{eq:weighted-layer-renewal}
\end{equation}
Consequently, for every \(m\ge1\),
\begin{equation}
 S_{-s}(m)
 =4\sum_{j=0}^{m-1}u_j(s)+3u_m(s)+u_{m+1}(s)-2.
 \label{eq:weighted-window-renewal}
\end{equation}
Moreover, with formal power series
\[
 B_s(z):=\sum_{\ell\ge1}b_\ell(s)z^\ell,
 \qquad U_s(z):=\sum_{j\ge0}u_j(s)z^j,
\]
one has
\begin{equation}
 U_s(z)=\frac1{1-B_s(z)}.
 \label{eq:weighted-renewal-generating-function}
\end{equation}
\end{proposition}

\begin{proof}
We use the generating-number parameterization in
\cite[Secs.~3--4]{Weinstein2016}, but not its layer-stabilization theorem.
Under Weinstein's free-monoid bijection, a generating number \(g\) corresponds
uniquely to a nonempty word
\[
 w=\frac{p_1}{q_1}\times\cdots\times\frac{p_r}{q_r}
\]
in reduced fractions from \((0,1)\).  Proposition~3.3 and formula~(16) of
that paper give, in the present notation,
\begin{equation}
 R(g)=\prod_{i=1}^r q_i,
 \qquad L(g)=\sum_{i=1}^r c(p_i/q_i).
 \label{eq:generator-weight-length}
\end{equation}
The continued-fraction bijection also gives
\eqref{eq:weighted-letter-coefficients}.  Unique decomposition of a free word
therefore shows that \(u_j(s)\) is exactly the sum of \(R(g)^{-s}\) over all
generating numbers with \(L(g)=j\), and proves
\eqref{eq:weighted-renewal-generating-function} as a formal identity.

It remains to count each generating orbit in a fixed layer.  A generating
number belongs to Weinstein's odd first-index sheet, so its initial-block
parameter is \(1\).  The formulas already established in
\eqref{eq:L-a-shift}--\eqref{eq:L-sigma-shift} consequently specialize to
\[
 L([a]g)=L(g)+2a,\qquad
 L([a]\tau g)=L(g)+1+2a,\qquad
 L(\sigma n)=L(n)+1
\]
for \(a\ge0\).  Hence the two branches in Weinstein's first sheet contribute
one point to every layer \(j\ge L(g)\); their sigma images contribute one
further point precisely when \(j>L(g)\).  The action is free and distinct
generating orbits are disjoint.  Since \(\mathcal I_j\) contains exactly one
point of partition multiplicity \(1\) for \(j\ge1\), summing the orbit weights
gives \eqref{eq:weighted-layer-renewal}.  The two-layer identity
\eqref{eq:real-collision-two-layers} then gives
\eqref{eq:weighted-window-renewal}.  Notice that the restriction \(j\ge1\) is
essential in \eqref{eq:weighted-layer-renewal}, because \(\mathcal I_0\) is
empty.
\end{proof}

Let \(\mathfrak P\) denote the Stern--Brocot pressure
\[
 \mathfrak P(\theta)
 :=\lim_{d\to\infty}\frac1d
   \log\sum_{T\in\mathcal T_d}|T|^\theta .
\]
We use the generation convention in which the first interior fraction
\(1/2\) has depth \(d=1\).  The denominator form of the same pressure
\cite[Sec.~2 and Thm.~1.1]{KessebohmerStratmann2007} is
\begin{equation}
 \mathfrak P(\theta)=
 \lim_{d\to\infty}\frac1d\log
 \sum_{\substack{0<p<q,\ (p,q)=1\\d(p/q)=d}}q^{-2\theta}.
 \label{eq:stern-brocot-denominator-convention}
\end{equation}

\begin{lemma}[Continued-fraction/Knauf normalization]
\label{lem:continued-fraction-knauf-normalization}
For \(d\ge1\), write a composition of \(d\) as
\(d=e_1+\cdots+e_r\), and put \(a_i=e_i+1\).  Encode the composition by the
word \(w=w_1\cdots w_{d-1}\in\{L,R\}^{d-1}\) whose \(j\)-th letter is \(R\)
exactly when \(j=e_1+\cdots+e_i\) for some \(i<r\), and is \(L\) otherwise.
With
\[
 L=\begin{pmatrix}1&0\\1&1\end{pmatrix},\qquad
 R=\begin{pmatrix}1&1\\0&1\end{pmatrix},\qquad M_w=w_1\cdots w_{d-1},
\]
the map \((a_1,\ldots,a_r)\mapsto w\) is a bijection, and
\begin{equation}
 \begin{aligned}
 \frac pq&=[0;a_1,\ldots,a_r]_-=
 \frac{\text{sum of the first row of }M_w}{\|M_w\|_\Sigma},\\
 q&=\mathcal D(a_1,\ldots,a_r)=\|M_w\|_\Sigma.
 \end{aligned}
 \label{eq:negative-cf-word-bijection}
\end{equation}
Thus \(d(p/q)=d\) is Knauf generation \(d\), the corresponding matrix word
has length \(d-1\), and the generator cost is \(c(p/q)=2d+1\).  In particular,
for \(\beta\in\mathbb R\), the even Knauf sum in the convention of
\cite[Sec.~IV]{FialaKlebanOzluk2003} is exactly
\[
 Z^{K,e}_d(2\beta)
 =\sum_{d(p/q)=d}q^{-2\beta}.
\]
There is no \(d=0\) fraction in \((0,1)\); the first four nonempty layers are
\[
\begin{array}{c|c|c|c}
d&\text{words}&q=\|M_w\|_\Sigma\text{ in word order}&c=2d+1\\ \hline
1&\varnothing&2&3\\
2&L,R&3,\,3&5\\
3&LL,LR,RL,RR&4,\,5,\,5,\,4&7\\
4&\{L,R\}^3&5,\,7,\,8,\,7,\,7,\,8,\,7,\,5&9
\end{array}
\]
\end{lemma}

\begin{proof}
Choosing the separator positions of a composition is the standard bijection
between compositions of \(d\) and binary words of length \(d-1\).  Starting
from \(M_\varnothing=I\), simultaneous induction on the word length, using
right multiplication by \(L\) or \(R\), gives the displayed subtractive
Euclidean recursion for the numerator and the negative-continuant recursion
for the denominator.  Explicitly, appending \(L\) replaces the last
composition part \(e_r\) by \(e_r+1\), whereas appending \(R\) replaces it by
\((e_r,1)\); substituting these two operations in
\(\mathcal D(a_1,\ldots,a_r)
=a_r\mathcal D(a_1,\ldots,a_{r-1})
-\mathcal D(a_1,\ldots,a_{r-2})\) agrees with
\(M_{wL}=M_wL\) and \(M_{wR}=M_wR\), respectively.  This proves
\eqref{eq:negative-cf-word-bijection}.
The depth and cost identities follow from the definitions.  The Knauf
identity is then term-by-term: its inverse temperature is \(2\beta\), so its
denominator weight is \(q^{-2\beta}\).  The table also fixes the empty-word
and first-generation conventions explicitly.
\end{proof}

\begin{lemma}[Denominator-layer pressure estimate]
\label{lem:knauf-layer-estimate}
For every fixed real \(s<2\),
\begin{equation}
 a_d(s):=
 \sum_{\substack{0<p<q,\ (p,q)=1\\d(p/q)=d}}q^{-s}
 \asymp_s \exp\{d\mathfrak P(s/2)\}.
 \label{eq:knauf-layer-pressure}
\end{equation}
For every fixed \(s\ge2\), one has instead
\[
 \lim_{d\to\infty}\frac1d\log a_d(s)=0.
\]
\end{lemma}

\begin{proof}
First suppose \(0<s<2\), and put \(\beta=s/2\).  The even Knauf
partition function \(Z^{K,e}_d(2\beta)\) is exactly \(a_d(s)\): its
states are the Stern--Brocot denominators first appearing at depth \(d\).  In the
notation of Fiala, Kleban, and \"Ozl\"uk, the first-coordinate identity
for their Knauf vector, its Perron--Frobenius iteration, and its spectral
asymptotic are given in
\cite[Sec.~IV, especially Eqs.~(23), (26), and (28)]{FialaKlebanOzluk2003}.
They give
\[
 a_d(s)=c_s\lambda(s/2)^d(1+o(1)),\qquad c_s>0.
\]
Their free-energy identification, together with the denominator formula
for Stern--Brocot pressure
\cite[Thm.~1.1 and the denominator formula in Sec.~2]{KessebohmerStratmann2007},
gives \(\log\lambda(s/2)=\mathfrak P(s/2)\).  This proves
\eqref{eq:knauf-layer-pressure} in the range \(0<s<2\).  At \(s=0\),
there are exactly \(2^{d-1}\) such denominators and
\(\mathfrak P(0)=\log2\), so the assertion is immediate.

It remains to supply the estimate for \(s<0\), a range not covered by
the positive-temperature Perron--Frobenius statement just cited.  Put
\(t=-s>0\), and let
\[
 L=\begin{pmatrix}1&0\\1&1\end{pmatrix},\qquad
 R=\begin{pmatrix}1&1\\0&1\end{pmatrix},\qquad
 \lVert M\rVert_\Sigma=\sum_{i,j=1}^2M_{ij}.
\]
 By \Cref{lem:continued-fraction-knauf-normalization}, the denominators at
 depth \(d\) are in bijection with words \(w\in\{L,R\}^{d-1}\), and
\begin{equation}
 q(w)=\lVert M_w\rVert_\Sigma.
 \label{eq:stern-brocot-matrix-denominator}
\end{equation}
Define
\[
 Z_n(t):=\sum_{w\in\{L,R\}^n}\lVert M_w\rVert_\Sigma^t,
 \qquad n\ge0.
\]
For nonnegative matrices \(A,B\),
\(\lVert AB\rVert_\Sigma\le
\lVert A\rVert_\Sigma\lVert B\rVert_\Sigma\), and hence
\begin{equation}
 Z_{m+n}(t)\le Z_m(t)Z_n(t).
 \label{eq:matrix-partition-submultiplicative}
\end{equation}
For the reverse inequality with one bridging letter, write
\(\alpha=A^T(1,1)^T\) and \(\beta=B(1,1)^T\).  Direct multiplication gives
\[
 \lVert ALB\rVert_\Sigma+\lVert ARB\rVert_\Sigma
 =\lVert A\rVert_\Sigma\lVert B\rVert_\Sigma
   +\alpha_1\beta_1+\alpha_2\beta_2.
\]
Thus at least one \(C\in\{L,R\}\) satisfies
\(\lVert ACB\rVert_\Sigma\ge
\frac12\lVert A\rVert_\Sigma\lVert B\rVert_\Sigma\).  Summing over
all pairs of words yields
\begin{equation}
 Z_{m+n+1}(t)\ge2^{-t}Z_m(t)Z_n(t).
 \label{eq:matrix-partition-bridged}
\end{equation}

Let \(\lambda_t=\lim_n Z_n(t)^{1/n}\), whose existence follows from
\eqref{eq:matrix-partition-submultiplicative}.  That inequality gives
\(Z_n(t)\ge\lambda_t^n\).  Iterating
\eqref{eq:matrix-partition-bridged} \(k-1\) times and then letting
\(k\to\infty\) gives
\[
 Z_{k(n+1)-1}(t)\ge2^{-t(k-1)}Z_n(t)^k,
 \qquad
 Z_n(t)\le2^t\lambda_t^{n+1}.
\]
Consequently \(Z_n(t)\asymp_t\lambda_t^n\).  By
\eqref{eq:stern-brocot-matrix-denominator} and the denominator form of
Stern--Brocot pressure cited above,
\(\log\lambda_t=\mathfrak P(-t/2)\).  Since
\(a_d(-t)=Z_{d-1}(t)\), this proves
\eqref{eq:knauf-layer-pressure} for \(s<0\).

Finally, the logarithmic assertion for \(s\ge2\) is precisely the
denominator formulation of the zero-pressure part of
\cite[Thm.~1.1]{KessebohmerStratmann2007}.
\end{proof}

The following pressure statement is retained because its renewal
normalization is used below.  Its existence, implicit branch, frozen phase,
and nondifferentiable critical point recover the golden-ratio
Bernoulli-convolution results of Feng--Olivier and Feng
\cite{FengOlivier2003,Feng2005}; the statement here is their finite-window
translation.

\begin{proposition}[Recovery of the Bernoulli-convolution pressure]
\label{thm:exact-negative-pressure}
\label{thm:all-real-pressure}
For every \(s\in\mathbb R\), the radius of convergence of \(B_s\) is
\begin{equation}
 R_B(s)=\exp\{-\mathfrak P(s/2)/2\}.
 \label{eq:Bs-radius}
\end{equation}
Thus \(R_B(s)<1\) for \(s<2\), and \(R_B(s)=1\) for \(s\ge2\).
The nonnegative real roots of \(B_s(r)=1\) are classified as follows:
\[
\begin{array}{c|c}
s<\sigma_0 & \text{a unique root }r_s\in(0,R_B(s)),\\
s=\sigma_0 & \text{the unique root }r_s=1=R_B(s),\\
s>\sigma_0 & \text{no root in }[0,R_B(s)].
\end{array}
\]
The pressure exists for every real tilt and satisfies
\begin{equation}
 P(-s):=\lim_{m\to\infty}\frac1m\log S_{-s}(m)
 =\begin{cases}
   -\log r_s,&s<\sigma_0,\\
   0,&s\ge\sigma_0.
  \end{cases}
 \label{eq:all-real-pressure}
\end{equation}
Equivalently, for \(t>-\sigma_0\), \(P(t)>0\) is the unique solution of
\begin{equation}
 \sum_{q=2}^{\infty}
 \sum_{\substack{1\le p<q\\(p,q)=1}}
 q^t e^{-P(t)c(p/q)}=1,
 \label{eq:exact-negative-pressure}
\end{equation}
whereas \(P(t)=0\) for \(t\le-\sigma_0\).  In particular,
\begin{equation}
 \{t\in\mathbb R:P(t)>0\}=(-\sigma_0,\infty).
 \label{eq:exact-zero-pressure-set}
\end{equation}

The restriction of \(P\) to \((-\sigma_0,\infty)\) is real analytic and
strictly convex.  At \(t_c=-\sigma_0\), it has the finite one-sided
derivatives
\begin{equation}
 P'_-(t_c)=0,
 \qquad
 P'_+(t_c)=
 \frac{\displaystyle
  \sum_{q\ge2}\sum_{\substack{1\le p<q\\(p,q)=1}}
       q^{-\sigma_0}\log q}
 {\displaystyle
  \sum_{q\ge2}\sum_{\substack{1\le p<q\\(p,q)=1}}
       c(p/q)q^{-\sigma_0}}>0.
 \label{eq:critical-pressure-slopes}
\end{equation}
Consequently, \(P\) is not differentiable at the freezing point.
\end{proposition}

\begin{proof}
Write \(a_d(s):=b_{2d+1}(s)\).  The negative-continued-fraction
parameterization gives
\[
 a_d(s)=\sum_{\substack{0<p<q,\ (p,q)=1\\d(p/q)=d}}q^{-s}.
\]
There are \(2^{d-1}\) fractions in this layer, and their denominators satisfy
\(d+1\le q\le F_{d+2}\).  In particular, every \(B_s\) has positive finite
radius.  The layer is precisely the first-appearance denominator layer of the
Knauf model.
 The precise estimate in \Cref{lem:knauf-layer-estimate} gives, for fixed
\(s<2\), \eqref{eq:knauf-layer-pressure}, and gives
\(d^{-1}\log a_d(s)\to0\) for \(s\ge2\).  Cauchy--Hadamard now proves
\eqref{eq:Bs-radius}: explicitly,
\(B_s(z)=\sum_{d\ge1}a_d(s)z^{2d+1}\), so the cost \(2d+1\) contributes the
factor \(1/2\) in the radius exponent.  Moreover,
\eqref{eq:knauf-layer-pressure} shows that
the terms of \(B_s\) do not tend to zero on \(|z|=R_B(s)\) when \(s<2\).
At \(s=2\), the positive real boundary also diverges.  For \(s>2\), Tonelli's
theorem gives
\begin{equation}
 B_s(1)=\sum_{q\ge2}\frac{\phi_{\rm E}(q)}{q^s}
 =\frac{\zeta(s-1)}{\zeta(s)}-1<\infty.
 \label{eq:weighted-renewal-at-one}
\end{equation}

The coefficients are nonnegative and the support contains both \(3\) and
\(5\), so \(B_s\) is strictly increasing on its positive interval of
convergence.  If \(s\le2\), its positive boundary value is infinite; if
\(2<s<\sigma_0\), \eqref{eq:weighted-renewal-at-one} is greater than one.
These facts give the unique interior root for \(s<\sigma_0\).  At
\(s=\sigma_0\), the boundary value is one, while it is less than one for
\(s>\sigma_0\).  This proves the asserted real-root classification.  The
triangle inequality also shows that, for \(s<\sigma_0\), the only zero of
\(1-B_s(z)\) in \(|z|\le r_s\) is the simple zero \(r_s\): equality forces
the degree-three and degree-five terms to have the same positive phase, and
\(\gcd(3,5)=1\).

Concatenation of generating words gives
\(u_{m+n}(s)\ge u_m(s)u_n(s)\).  The letters of costs \(3\) and \(5\) imply
\(u_j(s)>0\) for every \(j\ge8\).  The shifted Fekete argument used in the
proof of the strip result therefore applies without any restriction on
\(s\), and yields
\[
 \gamma_s:=\lim_{j\to\infty}\frac1j\log u_j(s).
\]
If \(s<\sigma_0\), the formal identity
\(U_s=(1-B_s)^{-1}\), positivity, and the preceding root classification show
that the radius of \(U_s\) is \(r_s\).  At \(s=\sigma_0\), it is one.  If
\(s>\sigma_0\), then \(U_s(1)=(1-B_s(1))^{-1}<\infty\), while
\(u_j(s)\ge b_j(s)\) prevents the radius of \(U_s\) from exceeding
\(R_B(s)=1\).  Hence
\[
 R_U(s)=\begin{cases}r_s,&s<\sigma_0,\\1,&s\ge\sigma_0,
 \end{cases}
 \qquad \gamma_s=-\log R_U(s)\ge0.
\]
For \(H_m(s)=\sum_{j=0}^m u_j(s)\), the exact renewal identity gives
\[
 H_m(s)\le S_{-s}(m)\le4H_{m+1}(s).
\]
Since \(m^{-1}\log H_m(s)\to\gamma_s\), this proves
\eqref{eq:all-real-pressure} and the root equation.

For \(t>-\sigma_0\), put
\[
 \mu_t(p/q):=q^t e^{-P(t)c(p/q)}.
\]
The root equation makes this a probability measure.  The root lies strictly
inside the relevant convergence circle, and
\(\log q\ll d(p/q)\), so the defining series and all its parameter
derivatives converge locally uniformly.  The analytic implicit-function
theorem applies.  With \(X=\log q\) and \(C=c(p/q)\), differentiation gives
\begin{equation}
 P'(t)=\frac{\mathbb E_{\mu_t}X}{\mathbb E_{\mu_t}C},
 \qquad
 P''(t)=
 \frac{\mathbb E_{\mu_t}[(X-P'(t)C)^2]}
      {\mathbb E_{\mu_t}C}.
 \label{eq:pressure-derivatives}
\end{equation}
The fractions \(1/2\) and \(1/3\) have \((C,q)=(3,2)\) and \((5,3)\),
and \(\log2/3\ne\log3/5\).  Thus the variance in
\eqref{eq:pressure-derivatives} is strictly positive.

It remains to justify passage to the critical endpoint.  For the canonical
regular continued fraction \(p/q=[0;a_1,\ldots,a_r]\), with \(a_r>1\),
the conversion between regular and negative continued fractions gives
\[
 d(p/q)=a_1+\cdots+a_r-1,
 \qquad c(p/q)=2(a_1+\cdots+a_r)-1.
\]
The fixed-denominator mean theorem of Panov and Liehl
\cite{Panov1982,Liehl1983,AistleitnerBordaHauke2023} states that
\[
 \frac1{\phi_{\rm E}(q)}
 \sum_{\substack{1\le p<q\\(p,q)=1}}(a_1+\cdots+a_r)
 \sim\frac6{\pi^2}(\log q)^2.
\]
Consequently, both the numerator and denominator in
\eqref{eq:critical-pressure-slopes} converge absolutely, since they are
bounded by constant multiples of
\(\sum q^{1-\sigma_0}\log q\) and
\(\sum q^{1-\sigma_0}(\log q)^2\), respectively.  Dominated convergence in
the first identity of \eqref{eq:pressure-derivatives} gives the stated finite
positive right derivative.  The pressure is identically zero to the left of
\(t_c\), so its left derivative is zero.
\end{proof}
