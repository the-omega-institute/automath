\begin{theorem}[Finite-window conditional phase separation]
\label{thm:conditional-phase-separation}
Fix \(0<\alpha<v_c\) and
\[
 0<\varepsilon<\min\{\alpha,v_c-\alpha\}.
\]
Let \(\mathbf X_{m,\alpha,\varepsilon}\) be uniform on
\[
 \left\{x\in X_m:
 \left|m^{-1}\log d_m(x)-\alpha\right|<\varepsilon\right\}.
\]
This set is nonempty for all sufficiently large \(m\), and
\begin{equation}
 \left(
  \frac1m\log d_m(\mathbf X_{m,\alpha,\varepsilon}),
  \frac1mL(g_m(\mathbf X_{m,\alpha,\varepsilon}))
 \right)
 \xrightarrow[m\to\infty]{\Pr}
 \left(\alpha+\varepsilon,
       \frac{\alpha+\varepsilon}{v_c}\right).
 \label{eq:conditional-phase-separation}
\end{equation}
Consequently, for every \(\eta>0\),
\[
 \lim_{\varepsilon\downarrow0}\lim_{m\to\infty}
 \Pr\left\{
  \left|m^{-1}L(g_m(\mathbf X_{m,\alpha,\varepsilon}))
       -\frac\alpha{v_c}\right|>\eta
 \right\}=0.
\]
For fixed \(\varepsilon\), the outer limit is essential: if
\(0<\eta<\varepsilon/v_c\), then the probability in the last display tends
to one.
\end{theorem}

\begin{proof}
Put \(a=\alpha-\varepsilon\), \(b=\alpha+\varepsilon\), and
\(r=\mu_C/\mu_H=1/v_c\).  We first record the two renewal facts used below.
For each fixed \(h>0\), Blackwell's nonarithmetic renewal theorem
\cite[Ch.~XI]{Feller1971} gives
\begin{equation}
 \mathcal U\{t<T\le t+h\}\longrightarrow\frac h{\mu_H}.
 \label{eq:blackwell-critical-renewal}
\end{equation}
The law of \(H_1\) is nonarithmetic because it charges both \(\log2\) and
\(\log3\), whose ratio is irrational.  The strong law gives
\(S_n/T_n\to r\) almost surely.  Moreover, since \(H_i\ge\log2\), an interval
of length \(h\) contains at most
\(N_h:=\lfloor h/\log2\rfloor+1\) renewal epochs on every sample path.
Dominated convergence therefore yields, for each \(\delta>0\),
\begin{equation}
 \mathcal U\{t<T\le t+h,\ |S/T-r|>\delta\}\longrightarrow0.
 \label{eq:marked-renewal-reward-law}
\end{equation}

Choose \(\delta_0>0\) so that \((r+\delta_0)b<1\), and fix \(h>0\).
On
\[
 J_m=(bm-2h,bm-h]
\]
the part of \(\mathcal U\) satisfying \(|S/T-r|\le\delta_0\) has mass
\(h/\mu_H+o(1)\) by
\eqref{eq:blackwell-critical-renewal}--\eqref{eq:marked-renewal-reward-law}.
For all sufficiently large \(m\), every such point has \(S\le m-1\), and
hence \(K_m(S)=4\).  If \(Z_m\) denotes the cardinality of the conditioning
set, \eqref{main-eq:marked-microcanonical-count} gives
\begin{equation}
 Z_m\ge4e^{\sigma_0(bm-2h)}
       \left(\frac h{\mu_H}+o(1)\right).
 \label{eq:conditional-denominator-lower}
\end{equation}

For fixed \(\delta>0\), set
\[
 \rho_m:=\sup_{t\ge am-h}
 \mathcal U\{t<T\le t+h,\ |S/T-r|>\delta\}.
\]
Equation \eqref{eq:marked-renewal-reward-law} implies \(\rho_m\to0\).
Covering \((am,bm)\) from its upper endpoint by intervals of length \(h\),
and using \(K_m\le4\), bounds the number of exceptional states by
\[
 \frac{4e^{\sigma_0bm}}{1-e^{-\sigma_0h}}\rho_m.
\]
After division by \eqref{eq:conditional-denominator-lower}, this proves
\[
 \frac{L(g_m)}{\log d_m}=\frac ST\xrightarrow{\Pr}r
\]
under the conditional law.

Finally, \(\mathcal U\{t<T\le t+h\}\le N_h\) for every \(t\).  Thus, for
each \(\gamma>0\), the number of conditioned states with
\(T\le(b-\gamma)m\) is
\(O(me^{\sigma_0(b-\gamma)m})\).  Equation
\eqref{eq:conditional-denominator-lower} shows that their conditional
probability tends to zero.  Since the conditioning itself gives
\(T/m<b\), we have \(T/m\to b\) in probability.  Combining this with
\(S/T\to r\) proves \eqref{eq:conditional-phase-separation}; its stated
consequences follow immediately.
\end{proof}

\begin{corollary}[Admissible shrinking windows]
\label{cor:shrinking-window-phase-separation}
Under the hypotheses of \Cref{thm:conditional-phase-separation}, suppose that
\(\varepsilon_m\downarrow0\) and
\(\liminf_{m\to\infty}m\varepsilon_m>0\).  Then the conditioning sets are
nonempty for all sufficiently large \(m\), and
\[
 \frac1mL(g_m(\mathbf X_{m,\alpha,\varepsilon_m}))
 \xrightarrow[m\to\infty]{\Pr}\frac\alpha{v_c}.
\]
\end{corollary}

\begin{proof}
Choose a fixed \(h>0\) smaller than
\(\liminf m\varepsilon_m\).  The interval
\(((\alpha+\varepsilon_m)m-2h,
   (\alpha+\varepsilon_m)m-h]\) then lies inside the conditioning window for
all sufficiently large \(m\).  The proof of
\Cref{thm:conditional-phase-separation}, with this fixed interval, gives the
same denominator lower bound and the same uniform estimate
\(S/T\to1/v_c\).  The conditioning itself now gives \(T/m\to\alpha\), which
proves the assertion.
\end{proof}

\begin{remark}[Arithmetic obstruction for thinner windows]
\label{rem:shrinking-window-arithmetic-obstruction}
The preceding corollary cannot be extended to every sequence
\(\varepsilon_m\downarrow0\).  Put
\(\mathcal D_m=\{\log d_m(x):x\in X_m\}\).  For every
\(0<\alpha<v_c\), there are infinitely many \(m\) for which
\(\alpha m\notin\mathcal D_m\).  Indeed, otherwise \(e^{\alpha m}\) and
\(e^{\alpha(m+1)}\) would be integers for all large \(m\), forcing
\(e^\alpha\) to be an integer; this contradicts
\(0<\alpha<v_c\le\alpha_+=\tfrac12\log\varphi<\log2\).
Enumerate such a subsequence as \((m_j)\).  Starting from any
\(\varepsilon_1>0\), define the remaining widths recursively so that
\(0<\varepsilon_m<\varepsilon_{m-1}/2\), and, when \(m=m_j\), impose also
\[
 \varepsilon_m<\frac1m
 \operatorname{dist}(\alpha m,\mathcal D_m).
\]
This gives a decreasing sequence for which the conditioning set is empty at
every \(m_j\).  Thus windows on a smaller scale require an additional
arithmetic local theorem, not merely a uniform version of Blackwell renewal.
\end{remark}

The preceding obstruction concerns arbitrary thin windows.  A complete
arithmetic conditional limit can nevertheless be obtained along the dyadic
multiplier ray.  For \(r\ge1\), let
\[
 \mathcal G_r:=\{g:\text{\(g\) is a Weinstein generator and }R(g)=2^r\}.
\]

\begin{theorem}[Exact dyadic multiplier limit]
\label{thm:dyadic-exact-multiplier}
The series
\begin{equation}
 \gamma_2:=\frac13\sum_{a\ge1}3^{-a}
 \sum_{\substack{1\le p<2^a\\p\ \mathrm{odd}}}c(p/2^a)
 \label{eq:dyadic-cost-velocity}
\end{equation}
converges.  Moreover,
\[
 \card{\mathcal G_r}=3^{r-1}\qquad(r\ge1).
\]
If \(G_r\) is uniform on \(\mathcal G_r\), then
\begin{equation}
 \frac{L(G_r)}r\xrightarrow[r\to\infty]{\Pr}\gamma_2.
 \label{eq:dyadic-generator-bridge}
\end{equation}

Let \(r_m\to\infty\) be integers such that
\[
 \frac m{r_m}\longrightarrow\beta>\gamma_2.
\]
Then
\begin{equation}
 C_m(2^{r_m})\sim4\cdot3^{r_m-1}.
 \label{eq:dyadic-finite-window-count}
\end{equation}
In particular, the exact level \(\{x\in X_m:d_m(x)=2^{r_m}\}\) is nonempty
for all sufficiently large \(m\).  If \(\mathbf X_m^{(2)}\) is uniform on
this level, then
\begin{equation}
 \frac{L(g_m(\mathbf X_m^{(2)}))}{r_m}
 \xrightarrow[m\to\infty]{\Pr}\gamma_2,
 \qquad
 \frac{L(g_m(\mathbf X_m^{(2)}))}{m}
 \xrightarrow[m\to\infty]{\Pr}\frac{\gamma_2}{\beta}.
 \label{eq:dyadic-finite-window-bridge}
\end{equation}
Equivalently, with \(v_2=\log2/\gamma_2\), if
\(r_m=\lfloor\alpha m/\log2\rfloor\) and \(0<\alpha<v_2\), the last limit is
\(\alpha/v_2\).
\end{theorem}

\begin{proof}
If a free generator word has multiplier \(2^r\), every denominator in the
word is a power of \(2\).  For each \(a\ge1\), there are
\(\phi_{\rm E}(2^a)=2^{a-1}\) letters with denominator \(2^a\).  Hence the
ordinary generating function for such words, including the empty word, is
\[
 \frac1{1-\sum_{a\ge1}2^{a-1}z^a}
 =\frac{1-2z}{1-3z}.
\]
Coefficient extraction proves \(\card{\mathcal G_r}=3^{r-1}\) for \(r\ge1\).

Define an independent letter sequence \((A_i,C_i)_{i\ge1}\) by
\[
 \Pr\{(A_i,C_i)=(a,c(p/2^a))\}=3^{-a}
 \quad(a\ge1,\ 1\le p<2^a,\ p\ \mathrm{odd}).
\]
This is a probability distribution because
\[
 \sum_{a\ge1}2^{a-1}3^{-a}=1,
 \qquad \mathbb E A_1=3.
\]
For the canonical regular continued fraction of \(p/2^a\), the
Panov--Liehl fixed-denominator mean theorem
\cite{Panov1982,Liehl1983,AistleitnerBordaHauke2023} gives, as \(a\to\infty\),
\[
 2^{1-a}\sum_{\substack{1\le p<2^a\\p\ \mathrm{odd}}}
 c(p/2^a)=O(a^2).
\]
Consequently \(\mathbb E C_1<\infty\), and
\(\mathbb E C_1/\mathbb E A_1=\gamma_2\).

Put \(A^{(n)}=\sum_{i=1}^nA_i\) and
\(C^{(n)}=\sum_{i=1}^nC_i\).  Every word with total exponent \(r\) has
probability \(3^{-r}\).  Therefore
\[
 \Pr\{A^{(n)}=r\text{ for some }n\}
 =3^{-r}\card{\mathcal G_r}=\frac13.
\]
Conditioned on this renewal event, the word is uniform on \(\mathcal G_r\).
The strong law gives
\(C^{(n)}/A^{(n)}\to\mathbb E C_1/\mathbb E A_1=\gamma_2\) almost surely.
On every sample path, the indicator that a renewal visit to \(r\) violates a
fixed neighborhood of this limit therefore tends to zero as \(r\to\infty\).
Dominated convergence, followed by division by the constant hitting
probability \(1/3\), proves \eqref{eq:dyadic-generator-bridge}.

For \(j\ge0\), let
\[
 U_j(2^r):=\card{\{g\in\mathcal G_r:L(g)=j\}}.
\]
The four-branch formula \eqref{main-eq:marked-microcanonical-count} gives, for
\(r\ge2\),
\begin{equation}
 C_m(2^r)=\sum_{j\ge0}K_m(j)U_j(2^r).
 \label{eq:dyadic-four-branch-count}
\end{equation}
Under the hypotheses on \(r_m\),
\eqref{eq:dyadic-generator-bridge} implies
\[
 \Pr\{L(G_{r_m})\le m-1\}\longrightarrow1,
\]
because \(m/r_m\to\beta>\gamma_2\).  Since \(0\le K_m\le4\) and
\(K_m(j)=4\) for \(j\le m-1\), division of
\eqref{eq:dyadic-four-branch-count} by \(4\card{\mathcal G_{r_m}}\) proves
\eqref{eq:dyadic-finite-window-count}.  The cost law under the uniform exact
fiber level is obtained from the uniform law on \(\mathcal G_{r_m}\) by the
bounded weight \(K_m/4\), which converges to \(1\) in probability and in
\(L^1\).  Thus the two laws have total-variation distance tending to zero.
Equation \eqref{eq:dyadic-generator-bridge} now gives
\eqref{eq:dyadic-finite-window-bridge}.  The final formulation follows from
\(r_m/m\to\alpha/\log2\).
\end{proof}

\begin{remark}
The theorem identifies the quenched speed on one arithmetic multiplier ray.
The dyadic exponent--cost lattice has no proper sublattice obstruction:
the two letter vectors \((1,3)\) and \((2,7)\) have determinant \(1\).
The comparison with the critical annealed velocity requires the explicit
certificate given next.
\end{remark}
