\section{A one-layer corollary}
\label{sec:standard-thermodynamics}

For \(m\ge1\), define
\[
 Z_m^R(t)=\sum_{n\in\mathcal I_m}R(n)^t,
 \qquad
 Y_m^R=\frac1m\log R(\mathbf N_m),
\]
where \(\mathbf N_m\) is uniform on \(\mathcal I_m\).

\begin{corollary}[One-layer Fibonacci partition thermodynamics]
\label{thm:standard-fibonacci-thermodynamics}
For every \(t\in\mathbb R\),
\begin{equation}
 \lim_{m\to\infty}\frac1m\log Z_m^R(t)=P(t).
 \label{eq:standard-layer-pressure}
\end{equation}
Thus this pressure is zero on \(( -\infty,-\sigma_0]\), positive, real
analytic, and strictly convex on \(( -\sigma_0,\infty)\), with the one-sided
derivatives in \eqref{eq:critical-pressure-slopes}.  For \(s>\sigma_0\),
\begin{equation}
 Z_m^R(-s)\longrightarrow
 \frac2{2-\zeta(s-1)/\zeta(s)}-1,
 \label{eq:standard-layer-frozen-limit}
\end{equation}
and at the critical tilt,
\begin{equation}
 Z_m^R(-\sigma_0)\sim\frac{2m}{\mu_C}.
 \label{eq:standard-layer-critical-size}
\end{equation}

For \(n\in\mathcal I_m\) with \(R(n)>2\), let \(g(n)\) be its Weinstein
generator and put \(\ell_m^R(n)=L(g(n))\); set \(\ell_m^R(n)=0\) when
\(R(n)\le2\).  If
\[
 \Pr\{\widehat{\mathbf N}_m=n\}
 =\frac{R(n)^{-\sigma_0}}{Z_m^R(-\sigma_0)},
\]
then
\begin{equation}
 \left(\frac1m\log R(\widehat{\mathbf N}_m),
       \frac1m\ell_m^R(\widehat{\mathbf N}_m)\right)
 \xrightarrow{\mathrm d}(v_cU,U),
 \label{eq:standard-layer-critical-gibbs}
\end{equation}
where \(U\) is uniform on \([0,1]\).

Finally, \((Y_m^R)\) satisfies a full large deviation principle with speed
\(m\) and good rate function
\begin{equation}
 I(\alpha)=\sup_{t\in\mathbb R}
 \{t\alpha-P(t)+\log\varphi\}.
 \label{eq:standard-layer-ldp-rate}
\end{equation}
Its effective domain is \([0,\alpha_+]\), and
\[
 I(\alpha)=\log\varphi-\sigma_0\alpha
 \qquad(0\le\alpha\le v_c).
\]
\end{corollary}

\begin{proof}
Put \(H_m(s)=\sum_{j=0}^m u_j(s)\).  The one-layer part of
\Cref{prop:weighted-generator-renewal} is
\begin{equation}
 Z_m^R(-s)=2H_{m-1}(s)+u_m(s)-1.
 \label{eq:standard-layer-renewal}
\end{equation}
The proof of \Cref{thm:all-real-pressure} gives the common exponential rate
\(P(-s)\) of \(u_m(s)\) and \(H_m(s)\), proving
\eqref{eq:standard-layer-pressure}.  If \(s>\sigma_0\), then
\[
 H_m(s)\longrightarrow U_s(1)
 =\frac1{2-\zeta(s-1)/\zeta(s)},
 \qquad u_m(s)\longrightarrow0,
\]
which proves \eqref{eq:standard-layer-frozen-limit}.  At \(s=\sigma_0\),
\(u_m(\sigma_0)\to1/\mu_C\), and Ces\`aro summation proves
\eqref{eq:standard-layer-critical-size}.

The changed boundary weight is
\[
 K_m^R(c)=
 \begin{cases}
 2,&c<m,\\
 1,&c=m,\\
 0,&c>m.
 \end{cases}
\]
Indeed, the two nonsigma orbit branches contribute one point in every layer
at or above the generator cost, while the two sigma branches begin one layer
later.  Thus, for Borel
\(B\subset\mathbb R\times(\log2,\infty)\),
\begin{equation}
 \card{\{n\in\mathcal I_m:
 (\ell_m^R(n),\log R(n))\in B\}}
 =\int_BK_m^R(c)e^{\sigma_0h}\,\mathcal U(dc,dh).
 \label{eq:standard-layer-marked-count}
\end{equation}
The renewal Riemann sum used in
\Cref{thm:critical-finite-window-gibbs}, now with interior weight two, and
the normalization \eqref{eq:standard-layer-critical-size} give
\(m^{-1}\ell_m^R(\widehat{\mathbf N}_m)\Rightarrow U\).  The same strong-law
estimate for \(T_n/S_n\) gives
\[
 \frac1m\log R(\widehat{\mathbf N}_m)
 -v_c\frac1m\ell_m^R(\widehat{\mathbf N}_m)
 \xrightarrow{\Pr}0,
\]
proving \eqref{eq:standard-layer-critical-gibbs}.

Since \(\card{\mathcal I_m}=F_m\), \eqref{eq:standard-layer-pressure} gives
the limiting logarithmic moment generating function
\(P(t)-\log\varphi\).  The extremal corollary gives the same upper endpoint
\(\alpha_+\).  The exposed-slope tilting and orbit-filling argument in
\Cref{thm:full-fiber-ldp} applies with two changes already verified above:
each source orbit contributes at most two points to \(\mathcal I_n\), and
the parity choice
\[
 [a]g\quad\text{or}\quad[a]\tau g
\]
places its generator injectively in \(\mathcal I_m\) while preserving
multiplicity.  The counting loss and the corner limit are unchanged, giving
the lower bound \(\sigma_0\alpha\) on \([0,v_c]\); exposed slopes and the
relative-boundary argument are identical.  This proves the full LDP and its
stated rate function.
\end{proof}
