\section{Local spectra, renewal, and the freezing corner}
\label{sec:local-spectrum}

For \(m\ge1\) and \(k\ge1\), put
\[
 C_m(k)=\card{\{x\in X_m:d_m(x)=k\}},
 \qquad
 \nu_m=\sum_{k\ge1}C_m(k)\delta_k.
\]

\begin{theorem}[Local microcanonical spectrum]
\label{thm:local-fiber-spectrum}
For every \(m\ge1\),
\[
 C_m(1)=2.
\]
The two singleton fibers have residues \(F_{m+1}-1\) and \(F_{m+2}-1\).
For every \(k\ge2\),
\[
 C_m(k)\le4\Psi(k),
 \qquad C_m(k)=4\Psi(k)\quad(m\ge2k).
\]
In particular, \(C_m(2)=4\) for \(m\ge4\).  Moreover, \(\nu_m\) converges
vaguely on the discrete space \(\mathbb N\) to
\[
 \nu_\infty=2\delta_1+4\sum_{k\ge2}\Psi(k)\delta_k.
\]
Equivalently, the mass of every finite subset of \(\mathbb N\) eventually
stabilizes.
\end{theorem}

\begin{proof}
The interval identity gives
\begin{equation}
 C_m(k)=N_m(k)+N_{m+1}(k).
 \label{eq:fiber-level-two-layer-count}
\end{equation}
Apply \Cref{lem:uniform-layer-count} to the two layers.  Their value-one
points are the left endpoints \(F_{m+1}-1\) and \(F_{m+2}-1\), which the
two cases of \Cref{thm:fiber-partition-interval} also identify as the fiber
residues.  The displayed pointwise stabilization implies vague convergence,
because compact subsets of the discrete space are finite.
\end{proof}
