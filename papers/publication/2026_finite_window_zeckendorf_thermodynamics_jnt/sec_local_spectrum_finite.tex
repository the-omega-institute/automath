\section{Local fiber spectra and the critical Dirichlet law}
\label{sec:local-spectrum}

The pressure records only exponential growth.  On the proved frozen
half-line, a finer limit survives without exponential normalization.  It is
controlled by the fixed-level stabilization in the preceding section.

For \(m\ge1\) and \(k\ge1\), put
\[
 C_m(k):=\card{\{x\in X_m:d_m(x)=k\}},
 \qquad
 \nu_m:=\sum_{k\ge1}C_m(k)\delta_k.
\]

\begin{theorem}[Local microcanonical spectrum]
\label{thm:local-fiber-spectrum}
For every \(m\ge1\),
\[
 C_m(1)=2.
\]
The two singleton fibers have residues
\(F_{m+1}-1\) and \(F_{m+2}-1\).  For every \(k\ge2\),
\[
 C_m(k)\le4\Psi(k),
 \qquad
 C_m(k)=4\Psi(k)\quad(m\ge2k).
\]
In particular, \(C_m(2)=4\) for \(m\ge4\).  Moreover, the locally finite
measures \(\nu_m\) converge vaguely on the discrete space \(\mathbb N\) to
\[
 \nu_\infty=2\delta_1+4\sum_{k\ge2}\Psi(k)\delta_k.
\]
Equivalently, the masses of every finite subset of \(\mathbb N\) eventually
stabilize.
\end{theorem}

\begin{proof}
The fiber--partition interval identity gives, as multisets, the values of
\(R\) on the disjoint consecutive layers \(\mathcal I_m\) and
\(\mathcal I_{m+1}\).  Hence
\begin{equation}
 C_m(k)=N_m(k)+N_{m+1}(k).
 \label{eq:fiber-level-two-layer-count}
\end{equation}
The uniform layer-count lemma gives one value equal to \(1\) in each layer,
the bound \(N_j(k)\le2\Psi(k)\) for \(k\ge2\), and equality for
\(j\ge2k\).  Summing the two layers proves all the asserted counts.  Under
the interval identity, the two level-one points are the left endpoints
\(F_{m+1}-1\) and \(F_{m+2}-1\); inspection of the two cases in
\Cref{thm:fiber-partition-interval} shows that these are also their fiber
residues.  Finally, every compact subset of the discrete space
\(\mathbb N\) is finite, and each of its point masses eventually has the
displayed value.  This is precisely vague convergence.
\end{proof}

Define the unnormalized limiting spectral partition function by
\[
 \mathcal Z(s):=\frac{4}{2-\zeta(s-1)/\zeta(s)}-2
 \qquad(s>\sigma_0).
\]

\begin{theorem}[Exact negative-tilt spectral limit]
\label{thm:spectral-partition-function}
For every \(s>\sigma_0\),
\[
 \lim_{m\to\infty}S_{-s}(m)=\mathcal Z(s)<\infty.
\]
For an arbitrary real tilt \(t\), the exact dichotomy is
\[
 \begin{cases}
 S_t(m)\longrightarrow
 \displaystyle\frac{4}{2-\zeta(-t-1)/\zeta(-t)}-2,&t< -\sigma_0,\\[8pt]
 S_t(m)\longrightarrow+\infty,&t\ge-\sigma_0.
 \end{cases}
\]
Thus \(-\sigma_0\) is the convergence boundary for the unnormalized
partition functions, independently of whether it is the boundary of the
zero-pressure set.
\end{theorem}

\begin{proof}
For \(s>\sigma_0\), \Cref{thm:local-fiber-spectrum} and dominated convergence
give
\[
 S_{-s}(m)
 =\sum_{k\ge1}C_m(k)k^{-s}
 \longrightarrow
 2+4\sum_{k\ge2}\frac{\Psi(k)}{k^s}.
\]
The domination follows from \(C_m(k)\le4\Psi(k)\) for \(k\ge2\), and the
majorant is summable by
\eqref{eq:weinstein-level-dirichlet-series}.  Weinstein's Dirichlet identity
then reduces the last expression to \(\mathcal Z(s)\).

Suppose next that \(0<s\le\sigma_0\).  If \(2<s\le\sigma_0\), the geometric
series in \eqref{eq:weinstein-level-dirichlet-series} diverges.  If
\(0<s\le2\), the one-factor generating words give
\(\Psi(k)\ge\phi_{\rm E}(k)\), and the totient Dirichlet series diverges.
Thus \(\sum_{k\ge2}\Psi(k)k^{-s}=+\infty\) throughout this range.  For each
fixed \(K\), \Cref{thm:local-fiber-spectrum} gives, once \(m\ge2K\),
\[
 S_{-s}(m)\ge2+4\sum_{2\le k\le K}\Psi(k)k^{-s}.
\]
Letting \(K\to\infty\) proves divergence for
\(-\sigma_0\le t<0\).  For \(t\ge0\), nonempty fibers satisfy
\(d_m(x)^t\ge1\), and hence
\(S_t(m)\ge\card{X_m}=F_{m+2}\to\infty\).
\end{proof}

The singularity of \(\mathcal Z\) also determines a limiting Gibbs law.  Put
\[
 \kappa:=-\left.\frac{d}{ds}\frac{\zeta(s-1)}{\zeta(s)}
          \right|_{s=\sigma_0}
 =\sum_{n\ge2}\frac{\phi_{\rm E}(n)\log n}{n^{\sigma_0}}
 =2.589184379946924\ldots.
\]

\begin{theorem}[Critical Gibbs spectrum]
\label{thm:critical-gibbs-spectrum}
As \(s\downarrow\sigma_0\),
\[
 \mathcal Z(s)=\frac{4}{\kappa(s-\sigma_0)}+O(1).
\]
For \(s>\sigma_0\), define
\[
 Q_{m,s}(k):=\frac{C_m(k)k^{-s}}{S_{-s}(m)}.
\]
Then \(Q_{m,s}\) converges in total variation to the probability measure
\[
 Q_s(1)=\frac2{\mathcal Z(s)},
 \qquad
 Q_s(k)=\frac{4\Psi(k)k^{-s}}{\mathcal Z(s)}\quad(k\ge2).
\]
In particular,
\[
 Q_s(1)\sim\frac\kappa2(s-\sigma_0).
\]
If \(K\) has law \(Q_s\), then
\[
 \mathbb E_s[\log K]
 =\frac1{s-\sigma_0}+O(1),
 \qquad
 \operatorname{Var}_s(\log K)
 =\frac1{(s-\sigma_0)^2}+O(1).
\]
\end{theorem}

\begin{proof}
The totient series is absolutely differentiable in a neighborhood of
\(\sigma_0>2\), and therefore
\[
 2-\frac{\zeta(s-1)}{\zeta(s)}
 =\kappa(s-\sigma_0)+O((s-\sigma_0)^2).
\]
The pole expansion and the asymptotic for \(Q_s(1)\) follow.  For fixed
\(s>\sigma_0\), the pointwise convergence and summable domination used in
\Cref{thm:spectral-partition-function} give convergence of the unnormalized
weights in \(\ell^1(\mathbb N)\).  Normalizing therefore gives total
variation convergence.

Absolute differentiation of the limiting Dirichlet series yields
\[
 -\frac{d}{ds}\log\mathcal Z(s)=\mathbb E_s[\log K],
 \qquad
 \frac{d^2}{ds^2}\log\mathcal Z(s)
 =\operatorname{Var}_s(\log K).
\]
Since the denominator defining \(\mathcal Z\) is analytic at \(\sigma_0\)
and has a simple zero there, its Taylor expansion gives
\[
 \log\mathcal Z(s)
 =-\log(s-\sigma_0)+\log(4/\kappa)+O(s-\sigma_0),
\]
where the remainder is analytic.  Differentiating this local analytic
expansion proves the two moment estimates.
\end{proof}

The same pole has a Tauberian consequence for the generating-orbit counts.

\begin{theorem}[Generating-orbit Tauberian law]
\label{thm:psi-tauberian}
As \(x\to\infty\),
\[
 \sum_{k\le x}\Psi(k)
 \sim\frac{x^{\sigma_0}}{\sigma_0\kappa},
 \qquad
 \sum_{k\le x}\frac{\Psi(k)}{k^{\sigma_0}}
 \sim\frac1\kappa\log x.
\]
Consequently,
\[
 \left(\frac4\kappa+o(1)\right)\log m
 \le S_{-\sigma_0}(m)
 \le\left(\frac{2\log\varphi}{\kappa}+o(1)\right)m.
\]
\end{theorem}

\begin{proof}
Set
\[
 A(s)=\sum_{n\ge2}\frac{\phi_{\rm E}(n)}{n^s}
 =\frac{\zeta(s-1)}{\zeta(s)}-1.
\]
For \(\Re s>\sigma_0\), Weinstein's identity reads
\[
 1+\sum_{k\ge2}\frac{\Psi(k)}{k^s}=\frac1{1-A(s)}.
\]
On the line \(\Re s=\sigma_0\), if \(\tau\ne0\), then
\[
 |A(\sigma_0+i\tau)|<A(\sigma_0)=1.
\]
Indeed, equality in the triangle inequality would force all
\(n^{-i\tau}\) with \(n\ge2\) to have the same phase; since the sum would
have to equal the positive number \(1\), it would in particular force
\(2^{-i\tau}=3^{-i\tau}=1\).  This contradicts
\(\log2/\log3\notin\mathbb Q\).  Thus the Dirichlet series has no other
boundary singularity and has at \(s=\sigma_0\) a simple pole of residue
\(1/\kappa\).  The Wiener--Ikehara theorem
\cite{Korevaar2004} now gives
\[
 \sum_{k\le x}\Psi(k)
 \sim\frac{x^{\sigma_0}}{\sigma_0\kappa}.
\]
Partial summation gives the second asymptotic.

For the critical finite-window sum, stabilization gives
\[
 S_{-\sigma_0}(m)
 \ge2+4\sum_{2\le k\le m/2}\frac{\Psi(k)}{k^{\sigma_0}},
\]
which proves the lower bound.  The uniform layer estimate and the absence of
levels above \(D_m\) give
\[
 S_{-\sigma_0}(m)
 \le2+4\sum_{2\le k\le D_m}\frac{\Psi(k)}{k^{\sigma_0}}.
\]
Now use the second Tauberian asymptotic and
\(\log D_m\sim\tfrac12m\log\varphi\).
\end{proof}

Finally, the same estimates narrow the possible location of the zero-pressure
boundary.  Write \(p_m(t)=m^{-1}\log S_t(m)\), and set
\[
 \beta_*:=\frac{\log\varphi}{\log(2/\varphi)}
 =2.270559453959664\ldots.
\]

\begin{theorem}[Pressure bounds in the transition strip]
\label{thm:pressure-transition-strip}
For \(0<s<\sigma_0\),
\[
 \max\left\{0,\log\varphi-s\log\frac2\varphi\right\}
 \le\liminf_{m\to\infty}p_m(-s)
\]
and
\[
 \limsup_{m\to\infty}p_m(-s)
 \le\min\left\{\log\varphi,
       \frac{\sigma_0-s}{2}\log\varphi\right\}.
\]
In particular,
\[
 \liminf_{m\to\infty}p_m(t)>0\qquad(t>-\beta_*).
\]
Together with \Cref{prop:negative-temperature-freezing}, this confines any
boundary of the zero-pressure set to
\[
 [-\sigma_0,-\beta_*]
 =[-2.478750785733960\ldots,-2.270559453959664\ldots].
\]
The width of this candidate strip is
\[
 \sigma_0-\beta_*=0.208191331774296\ldots.
\]
\end{theorem}

\begin{proof}
Under the uniform distribution on \(X_m\), the mean fiber multiplicity is
\(2^m/F_{m+2}\).  Since \(u\mapsto u^{-s}\) is convex, Jensen's inequality
gives
\[
 \frac{S_{-s}(m)}{F_{m+2}}
 \ge\left(\frac{2^m}{F_{m+2}}\right)^{-s}.
\]
After taking logarithms and dividing by \(m\), this gives the nonzero term in
the lower bound; the term zero follows from the two singleton fibers.

For the upper bound, fix \(a>\sigma_0\).  Since all occurring levels satisfy
\(k\le D_m\), \Cref{thm:local-fiber-spectrum} gives
\[
 S_{-s}(m)
 \le2+4D_m^{a-s}\sum_{k\ge2}\frac{\Psi(k)}{k^a}.
\]
Using \(m^{-1}\log D_m\to\tfrac12\log\varphi\) and then letting
\(a\downarrow\sigma_0\) proves the second term in the upper bound.  The
first follows directly from \(d_m(x)^{-s}\le1\).  If
\(-\beta_*<t<0\), the lower bound is strictly positive by the definition of
\(\beta_*\); if \(t\ge0\), then \(S_t(m)\ge F_{m+2}\).  This proves the
last assertion without assuming that the pressure limit exists.
\end{proof}
