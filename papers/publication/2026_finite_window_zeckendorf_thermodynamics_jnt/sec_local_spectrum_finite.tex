\section{Local spectra, renewal, and the freezing corner}
\label{sec:local-spectrum}

For \(m\ge1\) and \(k\ge1\), put
\[
 C_m(k)=\card{\{x\in X_m:d_m(x)=k\}},
 \qquad
 \nu_m=\sum_{k\ge1}C_m(k)\delta_k.
\]

\begin{theorem}[Local microcanonical spectrum]
\label{thm:local-fiber-spectrum}
For every \(m\ge1\),
\[
 C_m(1)=2.
\]
The two singleton fibers have residues \(F_{m+1}-1\) and \(F_{m+2}-1\).
For every \(k\ge2\),
\[
 C_m(k)\le4\Psi(k),
 \qquad C_m(k)=4\Psi(k)\quad(m\ge2k).
\]
In particular, \(C_m(2)=4\) for \(m\ge4\).  Moreover, \(\nu_m\) converges
vaguely on the discrete space \(\mathbb N\) to
\[
 \nu_\infty=2\delta_1+4\sum_{k\ge2}\Psi(k)\delta_k.
\]
Equivalently, the mass of every finite subset of \(\mathbb N\) eventually
stabilizes.
\end{theorem}

\begin{proof}
The interval identity gives
\begin{equation}
 C_m(k)=N_m(k)+N_{m+1}(k).
 \label{eq:fiber-level-two-layer-count}
\end{equation}
Apply \Cref{lem:uniform-layer-count} to the two layers.  Their value-one
points are the left endpoints \(F_{m+1}-1\) and \(F_{m+2}-1\), which the
two cases of \Cref{thm:fiber-partition-interval} also identify as the fiber
residues.  The displayed pointwise stabilization implies vague convergence,
because compact subsets of the discrete space are finite.
\end{proof}

The fixed-level theorem is uniform on the full range \(k\le m/2\).
Combining this fact with the pole of Weinstein's orbit series yields a
scaling limit for a growing portion of the spectrum.

Put
\[
 \kappa=-\left.\frac{d}{ds}\frac{\zeta(s-1)}{\zeta(s)}
 \right|_{s=\sigma_0}
 =\sum_{n\ge2}\frac{\phi_{\rm E}(n)\log n}{n^{\sigma_0}}.
\]

\begin{theorem}[Mesoscopic fiber spectrum]
\label{thm:mesoscopic-fiber-spectrum}
For Borel sets \(A\subset(0,1/2]\), define
\[
 \eta_m(A)=m^{-\sigma_0}
 \card{\{x\in X_m:d_m(x)/m\in A\}}.
\]
Then, vaguely on \((0,1/2]\),
\begin{equation}
 \eta_m(dy)\longrightarrow \frac4\kappa y^{\sigma_0-1}\,dy.
 \label{eq:mesoscopic-vague-limit}
\end{equation}
Equivalently, for \(0<a<b\le1/2\),
\[
 m^{-\sigma_0}\card{\{x\in X_m:am<d_m(x)\le bm\}}
 \longrightarrow
 \frac4{\sigma_0\kappa}(b^{\sigma_0}-a^{\sigma_0}).
\]
Moreover, if \(K_m\to\infty\) is any integer sequence with \(2K_m\le m\),
then
\begin{equation}
 \card{\{x\in X_m:d_m(x)\le K_m\}}
 \sim\frac4{\sigma_0\kappa}K_m^{\sigma_0}.
 \label{eq:mesoscopic-moving-cutoff}
\end{equation}
\end{theorem}

\begin{proof}
Set
\[
 A(s)=\sum_{n\ge2}\frac{\phi_{\rm E}(n)}{n^s}
 =\frac{\zeta(s-1)}{\zeta(s)}-1.
\]
Weinstein's free-monoid identity \eqref{eq:weinstein-level-dirichlet-series}
gives, for \(\Re s>\sigma_0\),
\[
 1+\sum_{k\ge2}\frac{\Psi(k)}{k^s}=\frac1{1-A(s)}.
\]
The denominator has a simple zero at \(s=\sigma_0\), with derivative
\(\kappa\).  If \(\tau\ne0\), positivity of the coefficients gives
\[
 |A(\sigma_0+i\tau)|<A(\sigma_0)=1;
\]
indeed, equality would force both \(2^{-i\tau}=1\) and
\(3^{-i\tau}=1\), contrary to \(\log2/\log3\notin\mathbb Q\).
Thus the Dirichlet series has no other singularity on its line of
convergence, and Wiener--Ikehara \cite{Korevaar2004} yields
\begin{equation}
 \sum_{k\le y}\Psi(k)
 \sim\frac{y^{\sigma_0}}{\sigma_0\kappa}.
 \label{eq:generator-orbit-counting-asymptotic}
\end{equation}

For every integer \(K\le m/2\), \Cref{thm:local-fiber-spectrum} gives the
exact identity
\begin{equation}
 \card{\{x\in X_m:d_m(x)\le K\}}
 =2+4\sum_{2\le k\le K}\Psi(k).
 \label{eq:mesoscopic-exact-cutoff}
\end{equation}
Taking \(K=K_m\) and applying
\eqref{eq:generator-orbit-counting-asymptotic} proves
\eqref{eq:mesoscopic-moving-cutoff}.  Taking differences of
\eqref{eq:mesoscopic-exact-cutoff} at
\(K=\lfloor bm\rfloor\) and \(K=\lfloor am\rfloor\) proves the interval
limit.  These intervals form a convergence-determining class for Radon
measures on \((0,1/2]\), which proves \eqref{eq:mesoscopic-vague-limit}.
\end{proof}
