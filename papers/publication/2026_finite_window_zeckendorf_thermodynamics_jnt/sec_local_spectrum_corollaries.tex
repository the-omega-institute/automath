\subsection{Local consequences}

Define
\[
 \mathcal Z(s)=\frac4{2-\zeta(s-1)/\zeta(s)}-2
 \qquad(s>\sigma_0).
\]

\begin{corollary}[Negative-tilt spectral limit]
\label{thm:spectral-partition-function}
For every \(s>\sigma_0\),
\[
 \lim_{m\to\infty}S_{-s}(m)=\mathcal Z(s)<\infty.
\]
For an arbitrary real tilt \(t\),
\[
 \begin{cases}
 S_t(m)\longrightarrow
 \displaystyle\frac4{2-\zeta(-t-1)/\zeta(-t)}-2,&t< -\sigma_0,\\[8pt]
 S_t(m)\longrightarrow+\infty,&t\ge-\sigma_0.
 \end{cases}
\]
Thus \(-\sigma_0\) is the convergence boundary for the unnormalized
partition functions.
\end{corollary}

\begin{proof}
For \(s>\sigma_0\), \Cref{thm:local-fiber-spectrum} and dominated
convergence give
\[
 S_{-s}(m)\longrightarrow
 2+4\sum_{k\ge2}\frac{\Psi(k)}{k^s}=\mathcal Z(s),
\]
using \eqref{eq:weinstein-level-dirichlet-series}.  For
\(0<s\le\sigma_0\), the same series diverges: this follows from its
geometric form when \(s>2\), while the one-letter bound
\(\Psi(k)\ge\phi_{\rm E}(k)\) covers \(s\le2\).  Stabilization of each
finite set of levels then gives \(S_{-s}(m)\to\infty\).  For \(t\ge0\),
\(S_t(m)\ge\card{X_m}=F_{m+2}\).
\end{proof}

Recall
\[
 \kappa=-\left.\frac{d}{ds}\frac{\zeta(s-1)}{\zeta(s)}
          \right|_{s=\sigma_0}
 =\sum_{n\ge2}\frac{\phi_{\rm E}(n)\log n}{n^{\sigma_0}}.
\]

\begin{corollary}[Critical Gibbs spectrum]
\label{thm:critical-gibbs-spectrum}
As \(s\downarrow\sigma_0\),
\[
 \mathcal Z(s)=\frac4{\kappa(s-\sigma_0)}+O(1).
\]
For \(s>\sigma_0\), define
\[
 Q_{m,s}(k)=\frac{C_m(k)k^{-s}}{S_{-s}(m)}.
\]
Then \(Q_{m,s}\) converges in total variation to
\[
 Q_s(1)=\frac2{\mathcal Z(s)},
 \qquad Q_s(k)=\frac{4\Psi(k)k^{-s}}{\mathcal Z(s)}\quad(k\ge2).
\]
In particular, \(Q_s(1)\sim\frac\kappa2(s-\sigma_0)\).  If \(K\) has law
\(Q_s\), then
\[
 \mathbb E_s[\log K]=\frac1{s-\sigma_0}+O(1),
 \qquad
 \operatorname{Var}_s(\log K)=\frac1{(s-\sigma_0)^2}+O(1).
\]
\end{corollary}

\begin{proof}
Absolute differentiability of the totient series near \(\sigma_0>2\) gives
\[
 2-\frac{\zeta(s-1)}{\zeta(s)}
 =\kappa(s-\sigma_0)+O((s-\sigma_0)^2).
\]
The pole and the singleton asymptotic follow.  The summable domination in
the preceding proof gives convergence of the unnormalized weights in
\(\ell^1(\mathbb N)\), hence total variation after normalization.  Finally,
\[
 -\frac{d}{ds}\log\mathcal Z(s)=\mathbb E_s[\log K],
 \qquad
 \frac{d^2}{ds^2}\log\mathcal Z(s)=\operatorname{Var}_s(\log K),
\]
and differentiating the local expansion
\(\log\mathcal Z(s)=-\log(s-\sigma_0)+\log(4/\kappa)+O(s-\sigma_0)\)
proves the estimates.
\end{proof}
