\section{Finite-window fibers}\label{sec:preliminaries}

Throughout,
\[
 F_0=0,\qquad F_1=F_2=1,\qquad F_{n+2}=F_{n+1}+F_n\quad(n\ge0).
\]
For \(m\ge0\), let
\[
 \Omega_m=\{0,1\}^m,
 \qquad
 X_m=\{x\in\Omega_m:x_ix_{i+1}=0\text{ for }1\le i<m\},
\]
and define
\[
 V_m(x)=\sum_{j=1}^m x_jF_{j+1},
 \qquad
 N_m(\omega)=\sum_{j=1}^m\omega_jF_{j+1}.
\]
The empty word belongs to \(\Omega_0=X_0\), with both values equal to zero.

\begin{definition}[Finite-window fold]
\label{def:fold-map}
For \(m\ge0\), define
\[
 \Fold_m(\omega)=
 V_m^{-1}\bigl(N_m(\omega)\bmod F_{m+2}\bigr),
 \qquad \omega\in\Omega_m,
\]
and, for \(x\in X_m\), put
\[
 d_m(x)=\card{\Fold_m^{-1}(x)}.
\]
The proposition below supplies the asserted inverse.  In particular,
\(\Fold_0(\varepsilon)=\varepsilon\) and \(d_0(\varepsilon)=1\).
\end{definition}

\begin{proposition}[Finite-window residue model]
\label{prop:Vm-bijection}
\label{prop:single-overflow}
\label{cor:residue-characterization}
For every \(m\ge0\), the map
\[
 V_m:X_m\longrightarrow\{0,1,\ldots,F_{m+2}-1\}
\]
is a bijection.  Moreover, every \(\omega\in\Omega_m\) has a unique
decomposition
\[
 N_m(\omega)=b_m(\omega)F_{m+2}+r,
 \qquad b_m(\omega)\in\{0,1\},\quad 0\le r<F_{m+2},
\]
and \(\Fold_m(\omega)=V_m^{-1}(r)\).  Consequently, if
\(r_x=V_m(x)\), then
\[
 \Fold_m(\omega)=x
 \quad\Longleftrightarrow\quad
 N_m(\omega)\equiv r_x\pmod{F_{m+2}}.
\]
Computing the full Zeckendorf expansion of \(N_m(\omega)\) and deleting its
possible leading \(F_{m+2}\)-digit gives the same fold.
\end{proposition}

\begin{proof}
By Zeckendorf's theorem \cite{Zeckendorf1972}, every integer
\(0\le n<F_{m+2}\) has a unique expansion using nonconsecutive parts among
\(F_2,\ldots,F_{m+1}\).  This is exactly a word in \(X_m\), proving the
bijection.

The Fibonacci sum identity gives
\[
 N_m(\omega)\le\sum_{j=1}^mF_{j+1}=F_{m+3}-2<2F_{m+2}.
\]
Euclidean division therefore produces the stated bit and residue.  The fold
formula and the fiber congruence follow from the definition and bijectivity
of \(V_m\).  Finally, the same bound permits at most the leading
\(F_{m+2}\)-digit in the full Zeckendorf expansion, which proves the last
assertion.
\end{proof}
