\subsection{Critical finite-size law}

Let
\[
 \mathcal A=\{p/q:0<p<q,\ (p,q)=1\},
 \qquad \pi(p/q)=q^{-\sigma_0}.
\]
By \eqref{eq:weighted-renewal-at-one}, \(\pi\) is a probability measure.
Let \((C_i,H_i)_{i\ge1}\) be independent with
\[
 \Pr\{(C_i,H_i)=(c(p/q),\log q)\}=\pi(p/q),
\]
put \(S_n=\sum_{i=1}^nC_i\), \(T_n=\sum_{i=1}^nH_i\), and define
\[
 \mathcal U=\sum_{n\ge1}\mathcal L(S_n,T_n).
\]
The moment estimates in \Cref{thm:all-real-pressure} give
\[
 \mu_C=\mathbb EC_1<\infty,
 \qquad \mu_H=\mathbb EH_1<\infty,
 \qquad \mu_H/\mu_C=v_c.
\]

For \(x\in X_m\) with \(d_m(x)>2\), let \(g_m(x)\) be the generator of the
Weinstein orbit containing the corresponding partition integer, and put
\[
 K_m(c)=
 \begin{cases}
 4,&c\le m-1,\\
 3,&c=m,\\
 1,&c=m+1,\\
 0,&c\ge m+2.
 \end{cases}
\]
For every Borel set \(B\subset\mathbb R\times(\log2,\infty)\),
\begin{equation}
 \card{\{x\in X_m:(L(g_m(x)),\log d_m(x))\in B\}}
 =\int_B K_m(c)e^{\sigma_0h}\,\mathcal U(dc,dh).
 \label{eq:marked-microcanonical-count}
\end{equation}
Indeed, a generator of cost \(c\) has four orbit branches in layers
\[
 c+2a,\qquad c+1+2a,\qquad c+1+2a,\qquad c+2+2a
 \quad(a\ge0).
\]
The two middle points are distinct, so the orbit contributes zero, one, or
two points to \(\mathcal I_j\) according as \(j<c\), \(j=c\), or \(j>c\).
Summing over \(\mathcal I_m\sqcup\mathcal I_{m+1}\) gives \(K_m(c)\).
Under the free-monoid bijection a generator word has cost \(S_n\), logarithmic
multiplicity \(T_n\), and renewal mass \(e^{-\sigma_0T_n}\), which proves
\eqref{eq:marked-microcanonical-count}.

Define
\[
 \widehat Q_m(x)=\frac{d_m(x)^{-\sigma_0}}{S_{-\sigma_0}(m)},
\]
and put \(\ell_m(x)=L(g_m(x))\) when \(d_m(x)>2\), with
\(\ell_m(x)=0\) otherwise.

\begin{theorem}[Critical finite-size and coexistence law]
\label{thm:critical-finite-window-gibbs}
Let
\[
 \mu_C=\sum_{q\ge2}\sum_{\substack{1\le p<q\\(p,q)=1}}
 c(p/q)q^{-\sigma_0}.
\]
Then
\begin{equation}
 S_{-\sigma_0}(m)\sim\frac{4m}{\mu_C}.
 \label{eq:critical-linear-finite-size}
\end{equation}
If \(\widehat{\mathbf X}_m\) has law \(\widehat Q_m\) and \(U\) is uniform
on \([0,1]\), then
\begin{equation}
 \left(
 \frac1m\log d_m(\widehat{\mathbf X}_m),
 \frac1m\ell_m(\widehat{\mathbf X}_m)
 \right)
 \xrightarrow[m\to\infty]{\mathrm d}(v_cU,U).
 \label{eq:critical-uniform-coexistence}
\end{equation}
\end{theorem}

\begin{proof}
At \(s=\sigma_0\), the coefficients \(b_j(\sigma_0)\) form a probability
distribution on the positive integers.  Its span is one because its support
contains \(3\) and \(5\), and its mean is \(\mu_C\).  The arithmetic renewal
theorem gives
\[
 u_j(\sigma_0)\longrightarrow\frac1{\mu_C}.
\]
Substitution in \eqref{eq:weighted-window-renewal} and Ces\`aro summation prove
\eqref{eq:critical-linear-finite-size}.

For every bounded continuous \(f\) on \([0,1]\), the renewal limit and a
Riemann-sum argument give
\[
 \frac1m\sum_{j=0}^{m-1}u_j(\sigma_0)f(j/m)
 \longrightarrow\frac1{\mu_C}\int_0^1f(y)\,dy.
\]
The terms at \(j=m,m+1\) are bounded, \(K_m(j)=4\) for \(j\le m-1\), and
the total critical weight at multiplicity at most two is bounded uniformly
by \Cref{lem:uniform-layer-count}.  Multiplying
\eqref{eq:marked-microcanonical-count} by \(e^{-\sigma_0h}\) and using
\eqref{eq:critical-linear-finite-size} proves
\begin{equation}
 \frac{\ell_m(\widehat{\mathbf X}_m)}m\xrightarrow{\mathrm d}U.
 \label{eq:critical-uniform-layer}
\end{equation}

The strong law gives \(T_n/S_n\to v_c\) almost surely.  For \(\delta>0\),
let
\[
 A_m=\mathcal U\{0<S\le m+1,\ |T/S-v_c|>\delta\}.
\]
Every renewal step has cost at least three.  The number of renewal epochs
with \(S_n\le m+1\), divided by \(m\), is therefore bounded on every path,
while the exceptional proportion tends to zero.  Dominated convergence
gives \(A_m=o(m)\).  Since \(K_m\le4\), the marked identity implies
\[
 \frac1m\log d_m(\widehat{\mathbf X}_m)
 -v_c\frac1m\ell_m(\widehat{\mathbf X}_m)
 \xrightarrow{\Pr}0.
\]
Together with \eqref{eq:critical-uniform-layer}, this proves
\eqref{eq:critical-uniform-coexistence}.
\end{proof}
