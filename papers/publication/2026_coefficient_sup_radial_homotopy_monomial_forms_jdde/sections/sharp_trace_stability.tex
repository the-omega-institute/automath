\subsection{Optimality of the continuous trace-stability constant}

The factor in Corollary~\ref{cor:box-delta-trace-stability} is not a loss in
the passage from the exact signed deficit identity to total variation.  In
dimension at least two, divergence-free boundary layers can concentrate the
negative part of the trace error on sets whose relative face measure tends to
zero.  The required extension is nonlinear in the deficit scale and has no
uniform derivative bound.

\begin{theorem}[Microstructured divergence-free trace extension]
\label{thm:microstructured-trace-extension}
Let $k\ge2$, let $R=\prod_{j=1}^k[0,L_j]$, and let $V_R$ and $m_R$ be as in
Lemma~\ref{lem:box-self-cheeger}.  For every $\gamma>0$ and every
$\delta_0>0$ there exist $\delta\in(0,\delta_0)$ and a vector field $W$,
smooth on a neighborhood of $R$, such that
\[
 \Div W=0,
 \qquad
 \norm{V_R+W}_{L^\infty(R;\ell^\infty)}=m_R+\delta,
\]
and
\begin{equation}
\label{eq:microstructured-trace-lower-bound}
 \int_{\partial R}\abs{W\cdot\nu_R}\,\dd\Haus^{k-1}
 \ge \bigl(2P_1(R)-\gamma\bigr)\delta .
\end{equation}
\end{theorem}

\begin{collarlemma}[Oscillatory collar lemma]
\label{lem:oscillatory-collar}
Let $k\ge2$, write $m=m_R$, and let $F$ be a coordinate face of $R$
perpendicular to $x_j$, with outward normal $\nu$ and inward coordinate $r$.
Fix $\ell\ne j$, put $s=x_\ell$ and $\tau=\partial_{x_\ell}$, and fix a
rectangular core
\[
 G=I\times Z\Subset F,
 \qquad I\Subset(0,L_\ell),
\]
where $Z$ contains the remaining tangential variables.  If $k=2$, the factor
$Z$ and all integrals over it are omitted, with $\abs Z:=1$.  Define the
\emph{tangential coefficient slack}
\begin{equation}
\label{eq:tangential-coefficient-slack}
 d_F:=\dist(\overline I,\{0,L_\ell\}),
 \qquad
 \sigma_F:=\frac{2m}{L_\ell}d_F>0.
\end{equation}
Fix $\epsilon\in(0,1)$, and let $a\in C_c^\infty(Z)$ and $J\Subset I$
satisfy
\begin{equation}
\label{eq:z-cutoff-cost}
 0\le a\le1,
 \qquad \max_Za=1,
 \qquad \int_Za\,\dd z\ge(1-\epsilon/8)\abs Z,
 \qquad \abs J\ge(1-\epsilon/8)\abs I,
\end{equation}
with $a=1$ in dimension two.  For every
\begin{equation}
\label{eq:collar-delta-threshold}
 0<\delta<\frac{\epsilon m}{8},
\end{equation}
every geometric upper bound $h_0>0$, and every $\kappa>0$, there are a collar
width $h\in(0,h_0)$, a smooth $\chi$ on an open interval containing $[0,h]$,
and $q_F,Q_F\in C_c^\infty(G)$ with the following five properties:
\begin{align}
 &\partial_sQ_F=q_F,
 \qquad \operatorname{supp}q_F\cup\operatorname{supp}Q_F\Subset G;
 \tag{i}\label{eq:collar-signal-support}\\
 &\int_Iq_F(s,z)\,\dd s=0
 \quad\text{for every }z;
 \tag{ii}\label{eq:collar-signal-fibre-mean}\\
 &-(2m+\delta)\le q_F\le\delta,
 \qquad \max_Gq_F=\delta;
 \tag{iii}\label{eq:boundary-signal-properties}\\
 &\int_G\abs{q_F}\,\dd\Haus^{k-1}
 \ge(2-\epsilon)\delta\Haus^{k-1}(G);
 \tag{iv}\label{eq:face-signal-mass}\\
 &\norm{Q_F}_{L^\infty(G)}
 <\min\left\{\kappa,
 \frac{\sigma_F}{2\max\{1,\norm{\chi'}_{L^\infty([0,h])}\}}\right\}.
 \tag{v}\label{eq:collar-primitive-bound}
\end{align}
The function $\chi$ satisfies
\begin{equation}
\label{eq:collar-cutoff-condition}
 0\le\chi\le1,
 \quad \chi(0)=1,
 \quad \chi=0\ \text{near }h,
 \quad
 (2m+\delta)(1-\chi(r))\ge \frac{2m}{L_j}r
 \quad(0\le r\le h).
\end{equation}
Consequently, the field
\begin{equation}
\label{eq:local-solenoidal-collar}
 W_F=q_F(s,z)\chi(r)\nu+Q_F(s,z)\chi'(r)\tau,
\end{equation}
extended by zero at the inner and tangential edges, is smooth on a
neighborhood of $R$, satisfies $\Div W_F=0$ and $W_F\cdot\nu=q_F$ on $F$,
and obeys
\begin{equation}
\label{eq:local-collar-capacity}
 \norm{V_R+W_F}_{L^\infty(\operatorname{supp}W_F\cap R;\ell^\infty)}
 \le m+\delta.
\end{equation}
Moreover, the lower bound $\sigma_F$ in
\eqref{eq:tangential-coefficient-slack} is uniform on the collar above
$\operatorname{supp}q_F\cup\operatorname{supp}Q_F$.
\end{collarlemma}

\begin{proof}
We make the choices in the following irreversible order: $G$ and the
direction $\ell$; $\epsilon$; $a$ and $J$ as in
\eqref{eq:z-cutoff-cost}; $\delta$; $h$; $\chi$; and, last, the oscillation
period.  No choice below is altered by a later one.  The two localization
costs in \eqref{eq:z-cutoff-cost} are fixed before $\delta$ and the period.

After $\delta$ satisfying \eqref{eq:collar-delta-threshold} is fixed, set
\[
 A_\delta:=2m+\delta,
 \qquad
 \theta:=\frac{\delta}{2(m+\delta)}<\frac{\epsilon}{16}.
\]
Choose $u\in C_c^\infty((0,1-\theta))$ with $0\le u\le1$,
$\max u=1$, and
\[
 b:=\int_0^1u(t)\,\dd t
 \ge1-\theta-\frac{\epsilon}{16}>1-\frac{\epsilon}{8}.
\]
Because
\[
 \frac{\delta b}{A_\delta}
 <\frac{\delta(1-\theta)}{A_\delta}=\theta,
\]
there is $v\in C_c^\infty((1-\theta,1))$, $0\le v\le1$, with
$\int_0^1v=\delta b/A_\delta$.  Thus
\[
 \varphi:=\delta u-A_\delta v
\]
has integral zero, lies between $-A_\delta$ and $\delta$, attains $\delta$,
and, since the supports of $u$ and $v$ are disjoint,
\begin{equation}
\label{eq:cell-mass-bound}
 \int_0^1\abs\varphi=2\delta b>(2-\epsilon/4)\delta.
\end{equation}

We next choose $h<h_0$.  With $c_j=2m/L_j$, take
$B>c_j/A_\delta$.  Since $1-e^{-Br}=Br+O(r^2)$, reducing $h$ makes
$A_\delta(1-e^{-Br})\ge c_jr$ on $[0,h]$.  Multiply $e^{-Br}$ by a smooth
cutoff which equals one near $0$ and vanishes near $h$.  The product $\chi$
is no larger than $e^{-Br}$, so it satisfies
\eqref{eq:collar-cutoff-condition}.  Extend it smoothly across $r=0$; this
also gives the formula \eqref{eq:local-solenoidal-collar} on the exterior side
of $F$.

Only the period remains.  Write $J=(s_0,s_0+\abs J)$, choose an integer $N$,
and put $p=\abs J/N$.  On the $n$th period define
\[
 q(s)=\varphi\left(\frac{s-s_0-np}{p}\right)
 \quad (s_0+np<s<s_0+(n+1)p),
 \qquad 0\le n<N,
\]
and set $q=0$ off $J$.  Since $\varphi$ is compactly supported in $(0,1)$,
$q\in C_c^\infty(I)$, every period has zero integral, and its primitive
$Q(s)=\int_{-\infty}^sq(t)\,\dd t$ resets to zero at every period endpoint.
The positive part of one period has mass $p\delta b$, hence
\begin{equation}
\label{eq:primitive-period-bound}
 \norm Q_{L^\infty(I)}\le p\delta b<p\delta.
\end{equation}
Set $q_F(s,z)=a(z)q(s)$ and $Q_F(s,z)=a(z)Q(s)$.  Properties
\eqref{eq:collar-signal-support}--\eqref{eq:boundary-signal-properties}
follow immediately.  The exact factorization requested by the two
localizations is
\[
 \int_G\abs{q_F}
 =\left(\int_Za\right)\left(\int_I\abs q\right).
\]
Using \eqref{eq:z-cutoff-cost}, \eqref{eq:cell-mass-bound}, and the choice of
$J$, its right-hand side is at least
\[
 (1-\epsilon/8)^2(2-\epsilon/4)\delta\abs Z\abs I
 \ge(2-\epsilon)\delta\Haus^{k-1}(G),
\]
where the last inequality follows from
$(1-\epsilon/8)^2\ge1-\epsilon/4$.  This proves
\eqref{eq:face-signal-mass} and displays both localization losses.

Finally choose $N$ so large that
\begin{equation}
\label{eq:period-choice}
 p<\min\left\{\frac{\kappa}{\delta},
 \frac{\sigma_F}{2\delta\max\{1,\norm{\chi'}_{L^\infty([0,h])}\}}\right\}.
\end{equation}
Equations \eqref{eq:primitive-period-bound} and \eqref{eq:period-choice} give
\eqref{eq:collar-primitive-bound}.  Increasing $N$ changes neither cutoff nor
the mass factorization, so the mass and primitive demands do not conflict.

It remains to verify the coefficient box.  On the whole collar above the
supports of $q_F$ and $Q_F$, the $\ell$th affine coefficient has the uniform
slack
\begin{equation}
\label{eq:uniform-slack-bound}
 m-\abs{V_R\cdot\tau}
 =\frac{2m}{L_\ell}\min\{s,L_\ell-s\}
 \ge\frac{2m}{L_\ell}d_F=\sigma_F.
\end{equation}
This proves the asserted uniformity on both supports.  By
\eqref{eq:collar-primitive-bound},
\[
 \abs{(V_R+W_F)\cdot\tau}
 \le m-\sigma_F+\norm{Q_F}_{L^\infty}
       \norm{\chi'}_{L^\infty([0,h])}
 <m-\frac{\sigma_F}{2}<m.
\]
For the normal coefficient, $V_R\cdot\nu=m-c_jr$ and
\eqref{eq:boundary-signal-properties}, \eqref{eq:collar-cutoff-condition}
give
\[
 -(m+\delta)
 \le m-c_jr+q_F\chi\le m+\delta.
\]
All other coefficients are unchanged and have magnitude at most $m$.
Moreover,
\[
 \Div W_F=-q_F\chi'+\partial_s(Q_F\chi')=0.
\]
Compact support in the tangential variables, vanishing of $\chi$ near $h$,
and its exterior extension across $r=0$ prove the claimed smoothness.  This
completes the proof.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:microstructured-trace-extension}]
Write $m=m_R$.  We now invoke Lemma~\ref{lem:oscillatory-collar}, keeping its
order of choices explicit.  First choose rectangular cores $G_F\Subset F$,
one on each coordinate face, such that
\begin{equation}
\label{eq:face-core-exhaustion}
 \sum_F\Haus^{k-1}(G_F)>P_1(R)-\frac{\gamma}{4}.
\end{equation}
Choose one tangential direction on each core.  This fixes all positive
numbers $d_F$ and $\sigma_F$ in
\eqref{eq:tangential-coefficient-slack}.  Next choose
\[
 0<\epsilon<\min\left\{1,\frac{\gamma}{2P_1(R)}\right\}.
\]
The cutoffs $a_F$ and intervals $J_F$ are then chosen as in
\eqref{eq:z-cutoff-cost}, and are not changed later.  Now fix a common
\[
 0<\delta<\min\left\{\delta_0,\frac{\epsilon m}{8}\right\}.
\]
The compact face cores are pairwise separated.  Choose geometric upper bounds
$h_{0,F}>0$ small enough that any collars of widths below them are pairwise
disjoint.  Applying Lemma~\ref{lem:oscillatory-collar} with, for example,
$\kappa=1$, then chooses, in order, each actual width $h_F$, its cutoff
$\chi_F$, and finally its oscillation period.
Thus the period is the only choice used to fit the already fixed slack
$\sigma_F$; it does not change either localization loss.

Set $W=\sum_FW_F$.  Disjointness of the supports, the divergence identity, and
\eqref{eq:local-collar-capacity} give
\[
 \Div W=0,
 \qquad
 \norm{V_R+W}_{L^\infty(R;\ell^\infty)}\le m+\delta.
\]
Each face signal attains $\delta$ at a point where $a_F=1$.  Thus the outward
normal component of $V_R+W$ equals $m+\delta$ there, and the last norm
inequality is an equality.  Finally, summing \eqref{eq:face-signal-mass} and using
\eqref{eq:face-core-exhaustion} gives
\[
 \int_{\partial R}\abs{W\cdot\nu_R}\,\dd\Haus^{k-1}
 >(2-\epsilon)\delta\left(P_1(R)-\frac{\gamma}{4}\right)
 \ge(2P_1(R)-\gamma)\delta,
\]
because $\epsilon P_1(R)<\gamma/2$.  This proves the theorem.
\end{proof}

The theorem determines the optimal constant in the main continuous stability
estimate.  The one-dimensional case is different because there is no
tangential direction in which to close a solenoidal boundary layer.

\begin{theorem}[Exact continuous linear stability constant]
\label{thm:sharp-continuous-stability-constant}
For a rectangular box $R=\prod_{j=1}^k[0,L_j]$, let
\[
 \mathsf C_{\rm tr}(R):=
 \inf\left\{C\ge0:
 \begin{array}{l}
 \displaystyle
 \sum_{j=1}^k\sum_{\lambda\in\{0,L_j\}}
 \norm{\iota_{j,\lambda}^*(\eta-\eta_R)}_{L^1(F_{j,\lambda};\coeff)}
 \le C\bigl(\norm{\eta}_{\coeff,\infty}-m_R\bigr)\\[2mm]
 \text{for every }\eta\in\Omega^{k-1}_{\rm sm}(R)
 \text{ with }\dd\eta=\omega_R
 \end{array}
 \right\}.
\]
Then
\begin{equation}
\label{eq:exact-continuous-linear-constant}
 \mathsf C_{\rm tr}(R)=
 \begin{cases}
 2, & k=1,\\
 2P_1(R), & k\ge2.
 \end{cases}
\end{equation}
In particular, the constant $2P_1(R)$ in
Theorem~\ref{thm:box-boundary-stability} is sharp in every dimension
$k\ge2$.
\end{theorem}

\begin{proof}
For $k\ge2$, Theorem~\ref{thm:box-boundary-stability} gives
$\mathsf C_{\rm tr}(R)\le2P_1(R)$.  Given $\gamma>0$, take $W$ from
Theorem~\ref{thm:microstructured-trace-extension} and let $\eta$ be the form
corresponding to $V_R+W$ under \eqref{eq:form-vector-identification}.  Then
$\dd\eta=\omega_R$, its norm deficit is exactly $\delta$, and its total face
trace error is
\[
 \int_{\partial R}\abs{W\cdot\nu_R}\,\dd\Haus^{k-1}
 \ge(2P_1(R)-\gamma)\delta.
\]
Since $\gamma$ is arbitrary, the reverse inequality follows.

If $k=1$, write $R=[0,L]$.  Every primitive has the form
$\eta(x)=x-L/2+a$, while $\eta_R(x)=x-L/2$ and $m_R=L/2$.  Hence
\[
 \norm{\eta}_{L^\infty}-m_R=\abs a,
 \qquad
 \abs{\eta(0)-\eta_R(0)}+\abs{\eta(L)-\eta_R(L)}=2\abs a.
\]
This proves the first line of \eqref{eq:exact-continuous-linear-constant}.
\end{proof}

The construction does not contradict the endpoint obstruction for the
divergence equation.  Bourgain and Brezis proved bounded nonlinear solvability
of $\Div Y=f$ with $Y\in L^\infty\cap W^{1,k}$ for $f\in L^k$ and showed that
no bounded linear solution selection has the same endpoint estimate
\cite{BourgainBrezis2003,BourgainBrezis2007}; the surrounding limiting
Sobolev theory is organized by canceling and cocanceling operators
\cite{VanSchaftingen2013}.  Theorem~\ref{thm:microstructured-trace-extension}
is not a right inverse on a boundary data space.  Its boundary data, collar
widths, and oscillation periods are chosen together, and the period tends to
zero with the norm deficit.  It therefore supplies the sharp trace sequence
needed here without asserting an endpoint-bounded extension operator.
