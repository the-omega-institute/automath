\subsection{Uniform score, stopping, plug-in, and two-point lemmas}
\label{subsec:singular-uniform-lemmas-main}

\begin{lemma}[Uniform sampled-bin and sampled-tail derivatives]
\label{lem:uniform-sampled-bin-tail-score-main}
For fixed \(n,\varepsilon,M,c_0\), and \(\Delta\tau>0\), there is a common
physical one-sided neighbourhood \(\mathcal Q^+\) of
\(\mathcal C_n(\varepsilon,M,c_0)\), in the coordinates
\(q=(\eta,\delta)\), on which \(c\pm\sqrt\delta>0\) and, for every
multi-index \(a\),
\[
\begin{aligned}
 \sup_{q\in\mathcal Q^+}|\partial_q^a\log g_q(k)|
 &\le C_a(1+k)^{2|a|},&& 1\le|a|\le3,\\
 \sup_{q\in\mathcal Q^+}|\partial_q^a\log S_q(k)|
 &\le C_a(1+k)^{2|a|},&& 1\le|a|\le2.
\end{aligned}
\tag{U1}
\]
All parameter derivatives extend continuously to \(\delta=0\).  In addition,
\(g_q(k)\le C(1+k)^{n-1}r^k\) for a common \(r<1\).
\end{lemma}

\begin{proof}
Compress the colliding holding times into \(Y=X_1+X_2\).  Relative to
Lebesgue measure on \((0,\infty)^{n-1}\), the complete holding-time density is
\[
 f_q(y,x_3,\ldots,x_n)
 =(c^2-\delta)e^{-cy}\frac{\sinh(\sqrt\delta\,y)}{\sqrt\delta}
 \prod_{j=3}^n\theta_j e^{-\theta_jx_j},
\tag{U2}
\]
where the quotient at \(\delta=0\) is \(y\).  Write
\(\sinh(\sqrt\delta\,y)/\sqrt\delta=y h(\delta y^2)\), where
\(h(x)=\sinh(\sqrt x)/\sqrt x\) is positive and analytic on
\([0,\infty)\).  On a sufficiently small common neighbourhood,
\(c^2-\delta\) and every rate are bounded away from zero.  The derivatives
through order three of \(\log h(x)\) are bounded on \([0,\infty)\): they are
continuous at zero by the power series, and their elementary hyperbolic
expressions are bounded as \(x\to\infty\).  Differentiating \(\log f_q\)
therefore gives, uniformly on \(\mathcal Q^+\),
\[
 |\partial_q^a\log f_q(y,x_3,\ldots,x_n)|
 \le C_a\{1+(y+x_3+\cdots+x_n)^{2|a|}\},
 \qquad 1\le |a|\le3.
\tag{U3}
\]

Put \(W=Y+X_3+\cdots+X_n\).  Differentiation under the integral in a
sampling bin expresses every derivative of \(\log g_q(k)\) through order
three as a finite universal polynomial in conditional expectations of the
complete-data derivatives in \textup{(U3)}, conditional on
\(k\Delta\tau<W\le(k+1)\Delta\tau\).  On this event every nonnegative
holding time is at most \((k+1)\Delta\tau\), which proves the first bound in
\textup{(U1)}.  This argument controls the ratio by the bin probability
itself and hence does not infer a relative bound from an absolute derivative
bound.

For the tail ratios, each exponential density is log-concave, convolution
preserves log-concavity, and integration over a half-line shows that the
survival function of \(W\) is log-concave.  Thus \(W\) is IFR and its residual
life satisfies
\[
 \PP_q(W-t>x\mid W>t)\le \PP_q(W>x),\qquad t,x\ge0.
\tag{U4}
\]
All rates on \(\mathcal Q^+\) exceed a common \(\underline\theta>0\), so
\(W\) is stochastically dominated by an
Erlang\((n,\underline\theta)\) variable.  Consequently, for every fixed
integer \(m\),
\[
 \sup_{q\in\mathcal Q^+}\EE_q(W^m\mid W>t)
 \le C_m(1+t)^m.
\tag{U5}
\]
The same conditional-score polynomial argument, now conditional on
\(W>k\Delta\tau\), proves the second bound in \textup{(U1)}.  Finally,
stochastic domination by the Erlang variable gives the common
polynomial-exponential mass envelope.  The envelopes justify every
differentiation under the integral and continuity at \(\delta=0\).
\end{proof}

\begin{lemma}[Uniform stopped renewal-score CLT and LLN]
\label{lem:uniform-stopped-renewal-score-main}
Let \(\mathcal Q\) be compact and let the integer Palm gaps \(G\) have full
support, a common exponential moment, and parameter-continuous masses.  Let a

\(D=G+1\), \(\mu_q=\EE_qD\).  Uniformly in \(q\in\mathcal Q\), and also for

\[
 K_N=N/\mu_{q_N}+O_{P_{q_N}}(\sqrt N),
\qquad
 \max\{A_N,R_N,G_1,\ldots,G_{K_N}\}=O_{P_{q_N}}(\log N).
\tag{U6}
\]
If \(b_q(G)\in\RR^d\) is parameter-continuous,
\(\EE_qb_q(G)=0\), and is bounded by a common polynomial in \(G\), then
\[
 \sup_{q\in\mathcal Q}d_{\rm BL}\!\left(
 \mathcal L_q\!\left[N^{-1/2}\sum_{j=1}^{K_N}b_q(G_j)\right],
 \mathcal N_d\!\left(0,\mu_q^{-1}\EE_q[b_qb_q^\top]\right)
 \right)\longrightarrow0.
\tag{U7}
\]
For every similarly dominated matrix reward \(H_q\),
\[
 N^{-1}\sum_{j=1}^{K_N}H_q(G_j)
 -\mu_q^{-1}\EE_qH_q(G)\longrightarrow0
\tag{U8}
\]
uniformly in probability.  If \(B_q\) has a common polynomial envelope, then
\(N^{-3/2}\sum_{j=1}^{K_N}B_q(G_j)=o_P(1)\) uniformly.  These conclusions
hold jointly and hence apply to the score, stopped Hessian, and third-order
likelihood remainder under bounded root-\(N\) local triangular arrays.
\end{lemma}

\begin{proof}
The common exponential moment gives uniform bounds for every moment of
\(D\), with \(0<\inf_q\mu_q\le\sup_q\mu_q<\infty\).  For the ordinary
renewal count, compare the partial sums of \(D_j-\mu_q\) at
\(\lfloor N/\mu_q\pm x\sqrt N\rfloor\).  Chebyshev's inequality and
Kolmogorov's maximal inequality give a bound \(C/x^2\), uniformly in \(q\),
for deviation of the count by more than \(x\sqrt N\).  The stationary delay
changes this count only through the initial incomplete cycle, which is
\(O_P(\log N)\) by the bound below.  This proves the first part of
\textup{(U6)} without requiring independence between a reward and its cycle
length.

The equilibrium formula
\(\PP_q(A_N\ge t)=\mu_q^{-1}\sum_{j\ge t}S_q(j)\), and its terminal analogue,
have a common exponential-polynomial bound.  There are at most \(N\) complete
gaps because \(D\ge1\).  A union bound for those gaps and the two endpoint
variables therefore makes the probability that the maximum exceeds
\(C_0\log N\) tend to zero uniformly for sufficiently large \(C_0\).  This
proves the second part of \textup{(U6)}.

For \(m_N=\lfloor N/\mu_q\rfloor\), the triangular-array Lindeberg theorem
applied to the deterministic sum of \(b_q(G_j)\) gives the normal law in
\textup{(U7)}.  The common polynomial envelope and exponential moment verify
Lindeberg uniformly.  Replacing \(m_N\) by \(K_N\) changes the sum by at most
the maximum of centred partial sums over \(O_P(\sqrt N)\) indices.
Kolmogorov's inequality makes that difference \(o_P(\sqrt N)\) uniformly;
this is the stopped-score, or Anscombe, step.  The same replacement in a
law of large numbers is \(o_P(N)\), proving \textup{(U8)}.  The last assertion
follows from \(K_N=O_P(N)\) and uniform integrability of the envelope.

To verify the displayed uniformity rather than only pointwise convergence,
take an arbitrary parameter sequence.  Compactness supplies a convergent
subsequence; parameter continuity plus the common exponential envelope gives
convergence of the means, covariance matrices, Lindeberg terms, and all LLN
integrals along it.  The preceding inequalities have constants independent of
the subsequence.  Every subsequence therefore has a further subsequence with
the asserted joint limits, which is the uniform subsequence criterion and
also proves the triangular-array statement.
\end{proof}

\begin{lemma}[Efficient-score plug-in equicontinuity]
\label{lem:efficient-score-plugin-main}
On the canonically ordered collision stratum, write the one-cycle null scores
as \(s_\eta\) and \(s_\delta\), and define
\[
 \psi_\eta=s_\delta-I_{\delta\eta}I_{\eta\eta}^{-1}s_\eta,
 \qquad
 \Delta_{{\rm eff},N}(\eta)=N^{-1/2}
 \sum_{j=1}^{K_N}\psi_\eta(G_j).
\tag{U9}
\]
For every fixed \(L<\infty\),
\[
 \sup_{\eta}\sup_{\|\eta'-\eta\|\le L/\sqrt N}
 |\Delta_{{\rm eff},N}(\eta')-\Delta_{{\rm eff},N}(\eta)|
 =o_{P_{\eta,0}}(1).
\tag{U10}
\]
The conclusion also holds under bounded contiguous local alternatives.  The
empirical calendar information and its Schur complement, evaluated anywhere
in the same root-\(N\) ball, converge uniformly to \(I_n(\eta)\) and
\(J_n(\eta)\).  Hence any measurable root-\(N\) null fit may be inserted in
the standardized efficient central sequence with an \(o_P(1)\) change.
\end{lemma}

\begin{proof}
Uniform nondegeneracy makes the projection coefficient in \textup{(U9)} twice
continuously differentiable with polynomial score envelopes.  By construction,
\[
 \EE_{\eta,0}\psi_\eta=0,
 \qquad \EE_{\eta,0}(\psi_\eta s_\eta^\top)=0.
\tag{U11}
\]
Differentiating the first identity, justified by
\Cref{lem:uniform-sampled-bin-tail-score-main}, gives
\[
 \EE_{\eta,0}(\partial_\eta\psi_\eta)
 =-\EE_{\eta,0}(\psi_\eta s_\eta^\top)=0.
\tag{U12}
\]
Thus the derivative of \(\Delta_{{\rm eff},N}\) is a normalized stopped sum
of centred rewards and is \(O_P(1)\) uniformly by
\Cref{lem:uniform-stopped-renewal-score-main}.  A second-order Taylor expansion
over a ball of radius \(L/\sqrt N\) has first term \(O_P(N^{-1/2})\); its
second term is bounded by
\(N^{-1/2}O_P(N)O(N^{-1})=O_P(N^{-1/2})\).  This proves \textup{(U10)}.
The stopped LLN applied to score outer products, followed by the same Taylor
bound and continuity of matrix inversion on the uniformly positive-definite
set, proves information consistency.  The normalization and the two
stationary endpoint scores obey the same conclusion: their parameter
increments are bounded by \(N^{-1/2}\) times a polynomial in
\(\max(A_N,R_N)\), which is \(O_P(\log N)\) by \textup{(U6)}.  Thus they
contribute \(o_P(1)\) uniformly to the plug-in difference.  Contiguity
transfers every null \(o_P(1)\) assertion to bounded local alternatives.
\end{proof}

\begin{lemma}[Two-point threshold-risk lower bound]
\label{lem:two-point-threshold-rate-main}
Fix a collision base \((\eta_0,0)\), hold \(\eta_0\) fixed, and compare
\(P_{0,N}=P_{\eta_0,0}^{(N)}\) with
\(P_{1,N}=P_{\eta_0,v_0/\sqrt N}^{(N)}\), where \(v_0>0\).  Let
\(\mathcal R_{i,N}\) be their unordered rate multisets.  Then
\[
 d_{\rm m}(\mathcal R_{0,N},\mathcal R_{1,N})
 =\sqrt{v_0}\,N^{-1/4},
\tag{U13}
\]
and, for every \(0<a<\sqrt{v_0}/2\),
\[
 \liminf_{N\to\infty}\inf_{\widehat{\mathcal R}_N}
 \max_{i=0,1}P_{i,N}\!\left\{
 N^{1/4}d_{\rm m}(\widehat{\mathcal R}_N,\mathcal R_{i,N})>a
 \right\}
 \ge \Phi_{\rm N}\!\left(
 -\frac{v_0\sqrt{I_{\delta\delta}(\eta_0)}}2\right)>0.
\tag{U14}
\]
Thus the lower conclusion is pointwise at a fixed base, uses two alternatives
and threshold loss, and asserts optimality in rate rather than an exact
minimax-risk theorem over a continuum of local parameters.
\end{lemma}

\begin{proof}
The two balls of radius \(aN^{-1/4}\) around the candidates are disjoint by
\textup{(U13)}.  Assigning an estimate to its nearer candidate therefore
turns simultaneous success of an estimator into correct testing of
\(P_{0,N}\) against \(P_{1,N}\).  The LAN likelihood ratio under \(P_{0,N}\)
converges to
\[
 \exp\{v_0Z-\tfrac12v_0^2I_{\delta\delta}(\eta_0)\},
 \qquad Z\sim\mathcal N(0,I_{\delta\delta}(\eta_0)).
\tag{U15}
\]
Its limit has mean one; mutual contiguity makes the likelihood ratios uniformly
integrable, so their total-variation distances converge to that of the two
Gaussian shifts separated by
\(v_0\sqrt{I_{\delta\delta}(\eta_0)}\).  For any test, its maximum error is at
least its equal-prior average error, whose limiting infimum is
\(\Phi_{\rm N}\{-v_0\sqrt{I_{\delta\delta}(\eta_0)}/2\}\).  Taking a
measurable estimator within \(o(1)\) of the infimum at each \(N\) proves
\textup{(U14)}.  Strict positivity also follows directly from mutual
contiguity, but \textup{(U14)} records the quantitative Gaussian bound.
\end{proof}

\section{Stationary collision experiments for sampled serial phase-type renewals}
\label{sec:stationary-serial-collision-main}

This section changes the inferential object from a deterministic product of
complete Palm gaps to the stationary binary record itself.  We first close
that gap for the present two-state sampled counter and then treat one isolated
double collision in an arbitrary fixed serial order.  Throughout this section
the sampling interval \(\Delta\tau>0\) is known.

For two finite multisets \(A=\{\!\{a_1,\ldots,a_n\}\!\}\) and
\(B=\{\!\{b_1,\ldots,b_n\}\!\}\), write
\[
 d_{\rm m}(A,B)=\min_{\pi\in\mathfrak S_n}
       \max_{1\le i\le n}|a_i-b_{\pi(i)}|
\tag{SC1}
\]
for optimal-matching distance.  This metric retains multiplicities.

\begin{paperthm}[Stationary two-state exchange experiment and sharp recovery]
\label{thm:stationary-two-state-collision-main}
Fix \(0<\varepsilon<M<\infty\).  In the sampled-counter model of
\Cref{ass:model-inference-contract-main}, use the quotient coordinates
\[
 c=\frac{\Gamma+\kappa_{\mathrm r}}2,
 \qquad d=\frac{\Gamma-\kappa_{\mathrm r}}2,
 \qquad \delta=d^2\ge0.
\tag{SC2}
\]
Let \(P_{c,\delta}^{(N)}\) be the law of the stationary binary record
\((A_0,\ldots,A_{N-1})\), and let \(g_{c,\delta}\), \(S_{c,\delta}\), and
\(\mu_{c,\delta}=\sum_{k\ge0}S_{c,\delta}(k)\) denote its Palm gap mass,
tail, and mean cycle length.  At \(\delta=0\), set
\[
 \dot\ell_c(k)=\partial_c\log g_{c,0}(k),\qquad
 \dot\ell_\delta(k)=\partial_\delta\log g_{c,\delta}(k)|_{\delta=0},
\]
and define the calendar information
\[
 I_2(c)=\frac1{\mu_{c,0}}
 \sum_{k\ge0}g_{c,0}(k)
 \begin{pmatrix}\dot\ell_c(k)\\ \dot\ell_\delta(k)\end{pmatrix}
 \begin{pmatrix}\dot\ell_c(k)&\dot\ell_\delta(k)\end{pmatrix}.
\tag{SC3}
\]
Then the following conclusions hold uniformly for \(c\in[\varepsilon,M]\).

\begin{enumerate}[label=\textup{(\roman*)},leftmargin=2.6em]
\item If \(u\in\RR\) and \(v\in[0,\infty)\) range over bounded sets, then
\[
 \log\frac{dP_{c+u/\sqrt N,v/\sqrt N}^{(N)}}
               {dP_{c,0}^{(N)}}
 =\begin{pmatrix}u\\v\end{pmatrix}^{\!\top}\Delta_{N,c}
 -\frac12\begin{pmatrix}u\\v\end{pmatrix}^{\!\top}
 I_2(c)\begin{pmatrix}u\\v\end{pmatrix}+o_{P_{c,0}}(1),
\tag{SC4}
\]
where the remainder is uniform in \((c,u,v)\) and
\(\Delta_{N,c}\Rightarrow\mathcal N_2\{0,I_2(c)\}\) uniformly in bounded
Lipschitz distance.  Thus the stationary-record experiments converge to the
Gaussian shift on the half-plane \(\RR\times[0,\infty)\).

\item The continuous survival transform has a double pole at the collision,
whereas its collision derivative has an exact fourth-order pole:
\[
 \mathcal L\{\overline F_{c,\delta}\}(s)
 =\frac1s\left\{1-
 \frac{c^2-\delta}{(s+c)^2-\delta}\right\},
\quad
 \partial_\delta\mathcal L\{\overline F_{c,\delta}\}(s)|_0
 =\frac{s+2c}{(s+c)^4}.
\tag{SC5}
\]
The centre derivative is \(-2c/(s+c)^3\).  Consequently the sampled
collision derivative contains a nonzero
\(k^3e^{-c\Delta\tau k}\) term, while the centre derivative has degree at
most two at that exponential base.  Hence
\[
 \inf_{c\in[\varepsilon,M]}\lambda_{\min}\{I_2(c)\}>0.
\tag{SC6}
\]

\item Partition \(I_2\) according to \((c,\delta)\) and put
\[
 J_2(c)=I_{\delta\delta}
 -I_{\delta c}I_{cc}^{-1}I_{c\delta}>0.
\tag{SC7}
\]
For testing \(H_0:\delta=0\) against \(H_1:\delta>0\), the residualized
score test, equivalently the local likelihood-ratio test, is asymptotically
level \(\alpha\) uniformly in \(c\).  At
\((c_N,\delta_N)=(c+u/\sqrt N,v/\sqrt N)\) its limiting power is
\[
 1-\Phi_{\rm N}\{z_{1-\alpha}-v\sqrt{J_2(c)}\}.
\tag{SC8}
\]
This is the Gaussian half-plane power envelope among tests with uniform local
asymptotic level \(\alpha\).

\item There is an estimator \(\widehat{\mathcal R}_N\) of the unordered rate
pair such that, uniformly for bounded \((u,v)\),
\[
 d_{\rm m}\left(
 \widehat{\mathcal R}_N,
 \{\!\{c+u/\sqrt N+\sqrt{v/\sqrt N},
          c+u/\sqrt N-\sqrt{v/\sqrt N}\}\!\}\right)
 =O_P(N^{-1/4}).
\tag{SC9}
\]
The estimated centre is root-\(N\), and the estimator first obtains a
root-\(N\) estimate of \(\delta\) and projects it onto \([0,\infty)\).
Conversely, for every fixed \(c\in(\varepsilon,M)\), \(v_0>0\), and
\(0<a<\sqrt{v_0}/2\),
\[
 \liminf_{N\to\infty}\ \inf_{\widehat{\mathcal R}_N}
 \max_{v\in\{0,v_0\}}
 P_{c,v/\sqrt N}^{(N)}\!\left\{
 N^{1/4}d_{\rm m}(\widehat{\mathcal R}_N,
 \{\!\{c+\sqrt{v/\sqrt N},c-\sqrt{v/\sqrt N}\}\!\})>a
 \right\}>0.
\tag{SC10}
\]
Thus \(N^{-1/4}\) is uniformly attainable and pointwise locally
minimax-optimal in rate, as witnessed by this fixed-base two-point threshold
lower bound.  No exact minimax-risk constant is asserted.  The sign of \(d\)
is not identifiable.
\end{enumerate}
\end{paperthm}

We now impose exactly the isolated-collision restrictions used in the
higher-order result.  Fix \(n\ge2\), \(0<\varepsilon<M<\infty\), and
\(c_0>0\).  The compact double-rate stratum is
\[
\mathcal C_{n}(\varepsilon,M,c_0)=
\left\{(\lambda,\lambda,\lambda_3,\ldots,\lambda_n):
\begin{array}{l}
\lambda,\lambda_j\in[\varepsilon,M],\\
\lambda_3<\cdots<\lambda_n,\\
|\lambda_j-\lambda|\ge c_0,\\
|\lambda_i-\lambda_j|\ge c_0\quad(3\le i<j\le n)
\end{array}\right\}.
\tag{SC11}
\]
Only one double collision is allowed; all other rates are simple and uniformly
separated.  The inequalities in \textup{(SC11)} canonically order the
noncollision rates (and are vacuous for \(n=2\)); these coordinates are not
labels of physical serial states.  No assertion below covers simultaneous
collisions, unknown order, unknown sampling interval, or nonserial phase-type
representations.

\begin{paperthm}[Stationary double-pole experiment for sampled generalized-Erlang renewals]
\label{thm:stationary-serial-double-pole-main}
Fix \(n\), \(\varepsilon,M,c_0\), and known \(\Delta\tau>0\) as above, and
suppose \(\mathcal C_n(\varepsilon,M,c_0)\ne\varnothing\).  For
\(\theta=(\theta_1,\ldots,\theta_n)\), let
\[
 Q(\theta)_{ii}=-\theta_i,\qquad
 Q(\theta)_{i,i-1}=\theta_i\quad(2\le i\le n),
\quad K_\theta=e^{\Delta\tau Q(\theta)},
\quad T_{1,\theta}=(I-K_\theta)\mathbf1e_n^\top.
\tag{SC12}
\]
Let \(P_\theta^{(N)}\) be the stationary length-\(N\) binary record from the
killed-reset kernels \((K_\theta,T_{1,\theta})\).  Near the displayed
stratum use
\[
 c=\frac{\theta_1+\theta_2}{2},\qquad
\delta=\left(\frac{\theta_1-\theta_2}{2}\right)^2\ge0,
\qquad \eta=(c,\theta_3,\ldots,\theta_n)\in\RR^{n-1}.
\tag{SC13}
\]
Here and below the separated entries of \(\eta\) are in the canonical order
of \textup{(SC11)}.  Bounded root-\(N\) perturbations preserve this order for
all sufficiently large \(N\), uniformly on the stratum.
Write \(g_{\eta,\delta}\), \(S_{\eta,\delta}\), and
\(\mu_{\eta,\delta}\) for the Palm gap mass, tail, and mean cycle length,
and at \((\eta_0,0)\) put
\[
 \dot\ell_{\eta,\delta}(k)
 =\nabla_{(\eta,\delta)}\log g_{\eta,\delta}(k)|_{(\eta_0,0)},
 \qquad
 I_n(\eta_0)=\frac1{\mu_{\eta_0,0}}
 \sum_{k\ge0}g_{\eta_0,0}(k)
 \dot\ell_{\eta,\delta}(k)\dot\ell_{\eta,\delta}(k)^\top.
\tag{SC14}
\]
Then:

\begin{enumerate}[label=\textup{(\roman*)},leftmargin=2.6em]
\item For bounded \(u\in\RR^{n-1}\) and \(v\in[0,\infty)\), uniformly over
\((\eta_0,0)\in\mathcal C_n(\varepsilon,M,c_0)\),
\[
 \log\frac{dP_{\eta_0+u/\sqrt N,v/\sqrt N}^{(N)}}
               {dP_{\eta_0,0}^{(N)}}
 =\begin{pmatrix}u\\v\end{pmatrix}^{\!\top}\Delta_{N,\eta_0}
 -\frac12\begin{pmatrix}u\\v\end{pmatrix}^{\!\top}
 I_n(\eta_0)\begin{pmatrix}u\\v\end{pmatrix}
 +o_{P_{\eta_0,0}}(1),
\tag{SC15}
\]
and \(\Delta_{N,\eta_0}\Rightarrow
\mathcal N_n\{0,I_n(\eta_0)\}\) uniformly in bounded Lipschitz distance.
The local stationary-record experiments therefore converge to the Gaussian
shift on \(\RR^{n-1}\times[0,\infty)\).

\item The information is uniformly nondegenerate:
\[
 \inf_{\eta_0\in\mathcal C_n(\varepsilon,M,c_0)}
 \lambda_{\min}\{I_n(\eta_0)\}>0.
\tag{SC16}
\]
This follows from sampled pole order, not merely from population
identifiability: the \(\delta\)-derivative has a nonzero
\(k^3e^{-\lambda\Delta\tau k}\) component; the centre derivative reaches
degree two at that base; and the derivative of each separated rate has a
unique degree-one component at its own exponential base.

\item Partition \(I_n\) according to \((\eta,\delta)\), and set
\[
 J_n=I_{\delta\delta}
 -I_{\delta\eta}I_{\eta\eta}^{-1}I_{\eta\delta}>0.
\tag{SC17}
\]
The residualized score test, or the local likelihood-ratio test, is uniformly
asymptotically level \(\alpha\) for \(H_0:\delta=0\) and attains at
\(\delta_N=v/\sqrt N\) the nuisance-adjusted power envelope
\[
 1-\Phi_{\rm N}\{z_{1-\alpha}-v\sqrt{J_n(\eta_0)}\}.
\tag{SC18}
\]

\item A measurable finite-atlas recurrence estimator based on the first
\(2n\) empirical complete-gap tails gives root-\(N\) estimates of the
canonically ordered \((\eta,\delta)\), after projection of \(\delta\) onto
\([0,\infty)\).  The induced unordered-multiset estimator
\(\widehat{\mathcal R}_{n,N}\) satisfies, uniformly over the compact stratum
and bounded \((u,v)\),
\[
 d_{\rm m}(\widehat{\mathcal R}_{n,N},
 \{\!\{c_N+\sqrt{\delta_N},c_N-\sqrt{\delta_N},
          \theta_{3,N},\ldots,\theta_{n,N}\}\!\})=O_P(N^{-1/4}),
\tag{SC19}
\]
where \((\eta_N,\delta_N)=(\eta_0+u/\sqrt N,v/\sqrt N)\).  The centre and
all canonically ordered separated rates are root-\(N\).

For every fixed point of the stratum, every \(v_0>0\), and every
\(0<a<\sqrt{v_0}/2\), the analogue of \textup{(SC10)}, with the remaining
rates held fixed, has a strictly positive lower limit, bounded below by the
Gaussian two-point testing constant displayed in
\Cref{lem:two-point-threshold-rate-main}.  Hence the unordered-multiset rate
\(N^{-1/4}\) is uniformly attainable over compact bounded local alternatives
and is pointwise locally minimax-optimal in rate at every collision base, as
witnessed by a two-point threshold-risk lower bound.  This is a rate statement,
not an exact asymptotic minimax-risk or sharp-constant theorem.
\end{enumerate}
\end{paperthm}

\section{Discussion}\label{sec:discussion}

Theorems~F and G make the collision experiment, rather than the order-two
quotient, the mathematical centre of the article.  For an isolated repeated
rate in a fixed-order sampled generalized-Erlang renewal, the stationary
binary record has a uniform Gaussian half-space experiment in the squared
separation coordinate.  The sampled fourth-order pole proves uniform
nuisance-adjusted information, the residualized score attains the one-sided
power envelope, and the finite-tail recurrence estimator and contiguity
argument match at \(N^{-1/4}\).  The two-state result is the prerequisite
special case, not the endpoint of the extension.

The deterministic marked reset turns the visible output of the constrained
D-MAP into a renewal process.  This is the source of both the explicit record
law and the low-dimensional quotient inverse.  It is also the source of the
model's fragility: a marked transition into the ready state removes the fixed
Palm restart and invalidates the renewal product used by the inverse and the
retained-cycle calculations.

The visible law determines the unordered survival pair, not the hidden labels.
The two-state fibre result makes that distinction concrete.  Away from the
exchange diagonal, weakly equivalent stochastic realizations in the
killed-reset cone form a closed arc with the two triangular rate orderings as
endpoints.  On the diagonal the arc collapses.  In higher state dimension the
\((n-1)^2\) orbit calculation applies to the minimal interior stratum and is
not a statement about the global dimension of every visible fibre.

The lettered two-state inferential statements A--E remain downstream of the
population algebra.  Theorem~C estimates the three inclusion coordinates;
Theorem~D checks the complete visible law under its stated null and alternative
classes; Theorem~E transports Theorem~C through the stable quotient chart of
Theorem~B.  Sorted roots and log-rates are ordinary delta-method consequences.
The physical-image test is a separate finite-dimensional diagnostic.  The
abstract semiparametric, moving-dimensional coupling, complete-Palm-gap split,
and projection calculations are supplementary and are not used to replace the
stationary likelihood proof in Theorems~F--G.

\begin{lemma}[Sampled isolated-double-pole nondegeneracy]
\label{lem:sampled-double-pole-nondegeneracy-main}
On \(\mathcal C_n(\varepsilon,M,c_0)\), the Palm gap family is analytic in
\((\eta,\delta)\) on a fixed physical one-sided neighbourhood of
\(\delta=0\), obeys the relative bounds in \textup{(SL1)} and \textup{(U1)},
and its one-cycle score coordinates are linearly independent.  Consequently
\textup{(SC16)} holds.
\end{lemma}

\begin{proof}
Let \(W=X_1+\cdots+X_n\), where the independent exponential holding times
have rates \(c+\sqrt\delta,c-\sqrt\delta,\theta_3,\ldots,\theta_n\), and put
\[
 B_\eta(s)=\prod_{j=3}^n\frac{\theta_j}{s+\theta_j}.
\]
The empty product is one.  The Laplace transform of the continuous survival
function \(\overline F_{\eta,\delta}(t)=P(W>t)\) is
\[
 H_{\eta,\delta}(s)=\frac1s\left\{1-
 \frac{c^2-\delta}{(s+c)^2-\delta}B_\eta(s)\right\}.
\tag{P1}
\]
At a base point with \(c=\lambda\), direct differentiation gives
\[
 \partial_\delta H_{\eta,0}(s)
 =B_\eta(s)\frac{s+2\lambda}{(s+\lambda)^4},
 \qquad
 \partial_cH_{\eta,0}(s)
 =-\frac{2\lambda B_\eta(s)}{(s+\lambda)^3},
\tag{P2}
\]
and, for \(j\ge3\),
\[
 \partial_{\theta_j}H_{\eta,0}(s)
 =-\frac{\lambda^2}{(s+\lambda)^2}
 \frac{1}{(s+\theta_j)^2}
 \prod_{\substack{\ell=3\\\ell\ne j}}^n
 \frac{\theta_\ell}{s+\theta_\ell}.
\tag{P3}
\]
The displayed expression follows after differentiating
\(\theta_j/(s+\theta_j)\) and cancelling its factor \(s\) against the outer
\(1/s\); all remaining numerator factors are nonzero at the relevant
separated poles.

Uniform separation makes \(B_\eta(-\lambda)\ne0\).  Thus
\(\partial_\delta H\) has an exact order-four pole at \(-\lambda\), with
nonzero leading coefficient \(\lambda B_\eta(-\lambda)\), while
\(\partial_cH\) has order three there.  At \(-\theta_j\), the derivative in
\(\theta_j\) has order two, and every other coordinate derivative has at most
order one there.  Inverse Laplace transformation therefore gives the following
distinguishing sampled terms:
\[
 \begin{array}{c|c}
 \text{coordinate}&\text{term not attainable by the remaining coordinates}\ \\
 \hline
 \delta& k^3e^{-\lambda\Delta\tau k}\\
 c& k^2e^{-\lambda\Delta\tau k}\quad\text{after the \(\delta\) coefficient is zero}\\
 \theta_j& k e^{-\theta_j\Delta\tau k}\quad(3\le j\le n).
 \end{array}
\tag{P4}
\]
All leading coefficients in \textup{(P4)} are nonzero.

Suppose \(a^\top\dot\ell_{\eta,\delta}=0\) in
\(L^2(g_{\eta,0})\).  Full support gives
\(a^\top\partial g_{\eta,0}(k)=0\) for every \(k\).  Absolute summability
then gives the same identity for the tail derivatives.  Exponential-polynomial
sequences at distinct bases are linearly independent: applying successively
the difference operators \(E-z_j\), or equivalently using their rational
generating functions, separates each base and its highest polynomial degree.
The cubic term at \(e^{-\lambda\Delta\tau}\) first forces
\(a_\delta=0\); the unique linear term at every separated base forces
\(a_{\theta_j}=0\); and the remaining quadratic term at the collision base
forces \(a_c=0\).  The scores are therefore linearly independent.

It remains to record the absolute analytic control used for information
continuity.  Formula \textup{(P1)} is rational and analytic in
\((\eta,\delta)\) on a common one-sided neighbourhood whose pole clusters
remain separated by compactness and \(c_0\).  Partial fractions, including
the confluent pole at \(-c\), give for every fixed derivative order
\[
 \sup_{(\eta,\delta)}
 |\partial_{\eta,\delta}^a S_{\eta,\delta}(k)|
 \le C_a(1+k)^{m_a}e^{-\varepsilon\Delta\tau k/2}.
\tag{P5}
\]
First differences give the same absolute bound for \(g\).  The required
relative bounds do not follow from \textup{(P5)}; they are proved independently
in \Cref{lem:uniform-sampled-bin-tail-score-main} by conditioning the complete
holding-time scores on a bin or tail event.  Dominated summation therefore
makes the information continuous.  Pointwise positive definiteness,
continuity, and compactness yield the strictly positive uniform minimum in
\textup{(SC16)}.
\end{proof}

\begin{lemma}[Measurable finite-atlas quotient estimator from sampled tails]
\label{lem:serial-collision-recurrence-estimator-main}
Under the restrictions of \Cref{thm:stationary-serial-double-pole-main}, there
are measurable estimators \((\widehat\eta_N,\widehat\delta_N)\), with the
noncollision entries of \(\widehat\eta_N\) canonically ordered and
\(\widehat\delta_N\ge0\), such that for every bounded set of local parameters
\[
 \widehat\eta_N-\eta_N=O_P(N^{-1/2}),
 \qquad
 \widehat\delta_N-\delta_N=O_P(N^{-1/2})
\tag{P6}
\]
uniformly over the compact stratum.  The induced rate multiset satisfies
\textup{(SC19)}.
\end{lemma}

\begin{proof}
Delete the initial and terminal incomplete cycles and write the observed
complete gaps as \(G_1,\ldots,G_{K_N}\).  Define
\[
 \widehat S_k=K_N^{-1}\sum_{j=1}^{K_N}\mathbf1_{\{G_j\ge k\}},
 \qquad 0\le k\le2n-1,
\tag{P7}
\]
with arbitrary fixed values on \(K_N=0\).  Apply
\Cref{lem:uniform-stopped-renewal-score-main} to the finite vector of indicator
rewards.  Its stopped CLT and count bound imply
\[
 \max_{0\le k\le2n-1}|\widehat S_k-S_k|=O_P(N^{-1/2})
\tag{P8}
\]
uniformly over the stratum and bounded local alternatives.

Write the monic recurrence polynomial as
\[
 \chi(x)=x^n+a_{n-1}x^{n-1}+\cdots+a_0
 =\prod_{i=1}^n(x-z_i),\qquad z_i=e^{-\theta_i\Delta\tau}.
\tag{P9}
\]
The equations
\[
 S_{k+n}+\sum_{j=0}^{n-1}a_jS_{k+j}=0,\qquad0\le k<n,
\tag{P10}
\]
have coefficient matrix \((S_{i+j})_{0\le i,j<n}\).  It is nonsingular even
at collisions by
\Cref{cor:nstate-serial-all-collision-identifiability-main}.  Its determinant
has a positive minimum in absolute value on the compact stratum; the same is
true on a sufficiently small common neighbourhood.  Solving the empirical
version of \textup{(P10)} and using \textup{(P8)} therefore gives
\[
 \max_j|\widehat a_j-a_j|=O_P(N^{-1/2})
\tag{P11}
\]
uniformly.

We now distinguish the cluster from the noncollision roots.  In sampled-root
coordinates, the separation in rates gives the common distance
\(\rho_0=\Delta\tau e^{-M\Delta\tau}c_0>0\).  At each base polynomial choose
one circle enclosing its double root and \(n-2\) disjoint circles enclosing
its simple roots, all with radii below \(\rho_0/4\).  On a coefficient
neighbourhood \(U\), the argument principle keeps the root counts equal to
two and one, respectively.  For a simple-root circle the contour power sum
\[
 z_j(a)=\frac{1}{2\pi i}\int_{\gamma_j}
 z\,\frac{\chi_a'(z)}{\chi_a(z)}\,dz
\tag{P12}
\]
is analytic.  On the double-root circle the corresponding first and second
power sums \(p_1,p_2\) are analytic, and
\[
 A=p_1,\qquad B=(p_1^2-p_2)/2
\tag{P13}
\]
are the elementary symmetric functions of the two-root cluster.  These
formulas remain analytic when the two roots split; no individual label is
assigned inside the cluster.

The compact collision stratum in coefficient space is covered by finitely
many such neighbourhoods \(U_1,\ldots,U_L\).  Shrink them to open sets
\(V_\ell\) whose closures lie in \(U_\ell\) and still cover the stratum.
Define the chart index to be the least \(\ell\) with
\(\widehat a\in V_\ell\).  Membership in a finite collection of open sets is
Borel measurable.  In every chart, order the real simple roots by their
rates \(-\log z/\Delta\tau\); on overlaps the ordered simple-root vector and
the cluster power sums agree, so the selection cannot change the estimand.
The finite maximum of the chart derivatives is bounded.

By \textup{(P11)}, for every deterministic sequence \(C_N\to\infty\),
\[
 \sup_qP_q\{\|\widehat a-a(q)\|>C_N/\sqrt N\}\longrightarrow0.
\tag{P14}
\]
The finite shrunken cover has a positive Lebesgue-number margin in coefficient
space.  Coefficients on bounded local alternatives are \(O(N^{-1/2})\) from
the base stratum and hence lie within half that margin for all sufficiently
large \(N\).  Thus \textup{(P14)} implies that the probability of no valid
chart tends to zero uniformly.  On that exceptional event use a fixed ordered
value.
Otherwise the analytic maps, \textup{(P11)}, and the finite derivative bound
give root-\(N\) errors for every ordered simple root and for \(A,B\).  The
separated rates are \(-\log z_j/\Delta\tau\), while the double-cluster
quotient coordinates are the analytic functions in \textup{(T27)}.  Project
the continued \(\widetilde\delta_N\) onto \([0,\infty)\); projection is
nonexpansive.  This proves \textup{(P6)} and the claimed measurability.

Finally form the estimated double pair
\(\widehat c_N\pm\sqrt{\widehat\delta_N}\) and the estimated separated
rates.  The square-root inequality used after \textup{(T28)} gives
\(O_P(N^{-1/4})\) for the colliding pair, while all other components are
root-\(N\).  The separation \(c_0\) fixes the optimal matching with
probability tending to one uniformly, proving \textup{(SC19)}.
\end{proof}
